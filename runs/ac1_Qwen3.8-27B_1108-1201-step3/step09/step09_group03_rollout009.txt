import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and it > 10:
            break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Paired {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def subgradient_balance(a0, time_limit, top_k=10, lr_init=0.02, verbose=False, label=""):
    """
    Subgradient method using top-k peaks of the convolution.
    Multiplicative weight update (exponential tilting).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = normalize(a)
    conv = fast_convolve(a).astype(np.float64)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.85
                no_improve = 0
                if lr < 1e-10:
                    break

        # Get top-k peaks
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            weights = np.ones(k_act) / k_act
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]
            temp = 0.01 * max(M, 1e-10)
            weights = np.exp((pv - pv.max()) / temp)
            weights /= weights.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if weights[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # sg[i] += w * 2 * a[k-i] for i in [i_start, i_end]
            sg[i_start:i_end + 1] += weights[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Multiplicative weight update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SubGrad {label} it={it} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def block_perturbation(a0, n_blocks=8, verbose=False):
    """
    Apply a structural perturbation: randomly redistribute mass between blocks.
    This helps escape local minima.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    block_size = n // n_blocks
    rng = np.random.default_rng()

    # Pick 2 random blocks and mix them
    b1 = int(rng.integers(0, n_blocks))
    b2 = int(rng.integers(0, n_blocks))
    if b1 == b2:
        b2 = (b2 + 1) % n_blocks

    s1 = a[b1 * block_size:(b1 + 1) * block_size].sum()
    s2 = a[b2 * block_size:(b2 + 1) * block_size].sum()

    # Mix with random mixing ratio
    mix = rng.uniform(0.2, 0.8)
    a[b1 * block_size:(b1 + 1) * block_size] *= (1 - mix)
    a[b2 * block_size:(b2 + 1) * block_size] *= (1 - mix)

    # Redistribute
    total = s1 + s2
    a[b1 * block_size:(b1 + 1) * block_size] += mix * total * (1 - mix) * (s1 / max(total, 1e-15)) / max(block_size, 1)
    a[b2 * block_size:(b2 + 1) * block_size] += mix * total * mix * (s2 / max(total, 1e-15)) / max(block_size, 1)

    # Simpler: just add random noise
    a = a0.copy().astype(np.float64)
    noise_scale = rng.uniform(0.01, 0.05)
    a += rng.standard_normal(n) * noise_scale
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def cosine_init(n, rng):
    """Raised cosine initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.8, 2.5)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_peak_init(n, rng):
    """Two Gaussian peaks."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.25, 0.35)
    c2 = n * rng.uniform(0.65, 0.75)
    w = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def upsample(a, factor=2):
    """Upsample a sequence by factor using FFT interpolation."""
    n = len(a)
    m = n * factor
    x = np.arange(n)
    X = rfft(a, 2 * n)
    # Zero-pad in frequency domain
    Xp = np.zeros(m, dtype=np.complex128)
    half = n // 2
    Xp[:half] = X[:half]
    Xp[-half:] = X[-half:]
    if n % 2 == 0:
        Xp[half] = X[half]
    a_up = irfft(Xp, m)
    a_up = np.maximum(a_up, 0)
    s = a_up.sum()
    if s < 1e-15:
        a_up = np.ones(m) / m
    else:
        a_up = a_up / s
    return a_up


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate with the actual evaluate_sequence and update if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
