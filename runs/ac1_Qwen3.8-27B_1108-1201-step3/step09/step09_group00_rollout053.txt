            # Re-extract from the problem
            b_val = np.asarray(prob.variables[0].value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            b_val = b_val / s
            M_val = float(prob.variables[1].value)
            y = _extract_dual(ineq_con, n)
            return b_val, M_val, y
    except Exception:
        pass
    return None


def _extract_dual(ineq_con, n):
    """Extract dual variable for inequality constraint (2n-1 vector)."""
    try:
        if ineq_con.dual_value is not None:
            dv = np.asarray(ineq_con.dual_value, dtype=np.float64)
            if dv.shape[0] == 2 * n - 1 or dv.shape == (2 * n - 1,):
                dv = np.maximum(dv, 0)
                return dv
            elif dv.size == 2 * n - 1:
                dv = dv.reshape(-1)
                dv = np.maximum(dv, 0)
                return dv
    except Exception:
        pass
    return None


def compute_kkt_weights(a, peak_set, support):
    """Given a, peak positions, and support, find weights w on peak_set
    that make sum_k w[k]*grad_k(a) as constant as possible over support.
    Uses a small LP: minimize range subject to w>=0, sum(w)=1."""
    n = len(a)
    n_peaks = len(peak_set)
    n_sup = len(support)
    if n_peaks == 0 or n_sup == 0:
        return np.ones(n_peaks) / max(n_peaks, 1)

    # grad_k(a)[i] = 2*a[k-i] (or a[i] if k=2i), for valid indices
    # Build matrix G: G[row=i_in_support, col=k_in_peak] = grad_k(a)[i]
    G = np.zeros((n_sup, n_peaks))
    for ii, i in enumerate(support):
        for kk, k in enumerate(peak_set):
            val = 0.0
            kj = k - i
            if 0 <= kj < n:
                val += a[kj]
            if kj != i and 0 <= kj < n:
                val += a[kj]
            G[ii, kk] = val

    # Solve: minimize (max_i G@w - min_i G@w) s.t. w>=0, sum(w)=1
    # This is: minimize t s.t. G@w - mean <= t/2, mean - G@w <= t/2, sum(w)=1, w>=0
    # Simpler: minimize max_i G@w - min_i G@w
    # Use scipy linprog for this small problem
    from scipy.optimize import linprog

    c = np.zeros(n_peaks + 2)  # w (n_peaks), upper, lower
    # Minimize upper - lower
    c[-2] = 1.0
    c[-1] = -1.0

    A_ub = np.zeros((2 * n_sup, n_peaks + 2))
    b_ub = np.zeros(2 * n_sup)
    for ii in range(n_sup):
        # G[ii,:] @ w - upper <= 0
        A_ub[ii, :n_peaks] = G[ii, :]
        A_ub[ii, -2] = -1.0
        b_ub[ii] = 0
        # lower - G[ii,:] @ w <= 0
        A_ub[n_sup + ii, :n_peaks] = -G[ii, :]
        A_ub[n_sup + ii, -1] = 1.0
        b_ub[n_sup + ii] = 0

    A_eq = np.zeros((1, n_peaks + 2))
    A_eq[0, :n_peaks] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n_peaks + [(None, None)] * 2

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        w = res.x[:n_peaks]
        w = np.maximum(w, 0)
        sw = w.sum()
        if sw > 1e-15:
            w = w / sw
        else:
            w = np.ones(n_peaks) / n_peaks
        return w
    else:
        return np.ones(n_peaks) / n_peaks


def kkt_newton_refine(a0, time_limit, max_iters=500):
    """Refine using KKT conditions: alternate between updating weights w
    (via small LP) and updating a (via gradient step toward constant gradient)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0

    # Identify initial support and peak set
    support = np.where(a > 1e-10)[0]
    b = fast_convolve(a)
    max_b = b.max()
    peak_threshold = max_b * 0.99
    peak_set = np.where(b >= peak_threshold)[0]
    if len(peak_set) == 0:
        peak_set = np.array([np.argmax(b)])

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Update weights using small LP
        w = compute_kkt_weights(a, peak_set, support)

        # Step 2: Compute gradient using these weights
        grad = np.zeros(n)
        for kk, k in enumerate(peak_set):
            if w[kk] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[k - idx]
            # For k=2i case, we get 2*a[i] instead of a[i]+a[i], which is the same
            grad[idx] += w[kk] * 2.0 * contrib

        # Step 3: Gradient descent step (projected to sum=1, a>=0)
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                lr *= 0.95
                lr = max(lr, 1e-6)

        # Update peak set every 50 iterations
        if (it + 1) % 50 == 0:
            b = fast_convolve(a)
            max_b = b.max()
            peak_threshold = max_b * 0.995
            peak_set = np.where(b >= peak_threshold)[0]
            if len(peak_set) == 0:
                peak_set = np.array([np.argmax(b)])
            support = np.where(a > 1e-10)[0]

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f} |S|={len(peak_set)}")

        if no_improve > 200:
            break

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing at progressively lower temperatures."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.03]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_z(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_s = _softmax_from_z(z)
            b = fast_convolve(a_s)
            max_b = b.max()
            if max_b < 1e-300:
                return 0.0
            log_se = max_b + np.log(np.sum(np.exp((b - max_b) / stage_T)))
            return (2.0 * n / stage_T) * log_se

        def jac_z(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a_s = _softmax_from_z(z)
            b = fast_convolve(a_s)
            Tb = b / stage_T
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_s[k - idx]
            S_val = np.dot(grad_a, a_s)
            return a_s * (grad_a - S_val)

        try:
            res = minimize(
                fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'eps': 1e-12,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_result = _softmax_from_z(z)
    return a_result


def _softmax_from_z(z):
    """Convert log-domain vector to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with dual-based multiplicative updates."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(42)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        result = solve_lp_response(y, solver)
        if result is None:
            continue
        b_new, M_b, dual_y = result

        # Standard Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # New: multiplicative update using dual info
        if dual_y is not None and dual_y.max() > 1e-15:
            # Compute "pressure" from dual: p[i] = sum_k dual_y[k] * a[k-i]
            p = np.zeros(n)
            for k in range(2 * n - 1):
                if dual_y[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                p[idx] += dual_y[k] * a[k - idx]
            # Multiplicative update: a[i] proportional to a[i] / (p[i] + eps)
            eps = p.max() * 1e-6 + 1e-15
            a_mult = a / (p + eps)
            a_mult = np.maximum(a_mult, 0)
            sm = a_mult.sum()
            if sm > 1e-15:
                a_mult = a_mult / sm
                # Blend with standard Nesterov a
                a_nest = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
                a_nest = np.maximum(a_nest, 0)
                sn = a_nest.sum()
                if sn > 1e-15:
                    a_nest = a_nest / sn
                    a = 0.5 * a_nest + 0.5 * a_mult
                    a = np.maximum(a, 0)
                    sa = a.sum()
                    if sa > 1e-15:
                        a = a / sa
                else:
                    a = a_mult
            else:
                a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
        else:
            a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 6:
                noise_scale = 0.02 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_rounds = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_hessian_descent(a0, time_limit, n_steps=50000):
    """Peak-directed descent using approximate Hessian for line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributing elements to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            # Try several reduction factors with redistribution
            for factor in [0.998, 0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute proportionally to non-peak-contributing elements
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                # Also reduce weight for elements that contribute to peaks
                other_contribs = np.zeros(n)
                # Check a few nearby peak positions
                for pk in [peak_k, peak_k - 1, peak_k + 1, peak_k // 2]:
                    if 0 <= pk < 2 * n - 1:
                        pjs = max(0, pk - n + 1)
                        phe = min(n - 1, pk)
                        pidx = np.arange(pjs, phe + 1)
                        other_contribs[pidx] += a[pidx] * a[pk - pidx]
                others = a_trial[mask]
                others_weight = others + 0.1 * other_contribs[mask]
                others_weight = np.maximum(others_weight, 1e-15)
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others_weight / others_weight.sum()
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous perturbation of top 4
            a_trial = a.copy()
            for pos in top_positions[:4]:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.997
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 300:
                        break
            else:
                stall += 1
                if stall > 300:
                    break
        else:
            stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDesc {elapsed:.0f}s: score={best_score:.8f} stall={stall}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence with good properties."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Multi-frequency oscillation
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.1 * np.cos(2 * np.pi * 11 * x)
        a = np.maximum(a, 0)
    else:
        # Flat with bumps
        a = np.ones(n)
        a[int(n * 0.25)] += 0.5
        a[int(n * 0.75)] += 0.5
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization using LP-dual multiplicative updates, KKT Newton
    refinement, aggressive LogSumExp L-BFGS, and Hessian-informed peak descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP with dual multiplicative updates ===
    print(f"\n=== Phase 1: NesterovLP+dual (n={n_best}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 2: KKT Newton refinement (new!) ===
    print(f"\n=== Phase 2: KKT Newton (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_kkt, _ = kkt_newton_refine(best_seq, t_alloc, max_iters=300)
        try_update(a_kkt, "kkt_newton")

    # === Phase 3: Aggressive L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        T_stages = [40.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.03, 0.01]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "lbgfs_aggro")

    # === Phase 4: Peak Hessian Descent ===
    print(f"\n=== Phase 4: Peak Hessian Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
