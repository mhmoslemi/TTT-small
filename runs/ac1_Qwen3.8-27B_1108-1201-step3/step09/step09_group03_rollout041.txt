"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with adaptive temperature annealing
2. Subgradient method with momentum for the non-smooth objective  
3. FFT-based gradient computation for O(n log n) per step
4. Coordinate descent with exact line search for polishing
5. Multi-start with structured perturbations
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient_smoothed(a, eps):
    """
    Compute gradient of f_eps(a) = eps * log(sum_k exp(c[k]/eps))
    where c = conv(a,a).
    
    grad[j] = sum_k p[k] * (a[k-j] + a[j]) where p = softmax(c/eps)
    
    Uses FFT for the cross-correlation part.
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Softmax with numerical stability
    c_scaled = c / eps
    c_scaled -= c_scaled.max()
    p = np.exp(c_scaled)
    p /= p.sum()
    
    # grad[j] = (p * a_crosscorr)[j] + a[j]
    # where (p * a_crosscorr)[j] = sum_k p[k] * a[k-j]
    
    # Cross-correlation via FFT
    # Need to zero-pad both to at least len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    
    # Cross-correlation: ifft(FFT(p) * conj(FFT(a)))
    # For real signals: ifft(FFT(p) * conj(FFT(a))) gives (p ⋆ a)[j] = sum_k p[k]*a[k-j]
    cross = irfft(P * np.conj(A), fft_size)[:n]
    
    grad = cross + a  # a[j] * sum_k p[k] = a[j]
    
    return grad, c.max()

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def smooth_adam_optimize(a0, t_budget, eps_start=0.1, eps_end=0.001, 
                          lr=0.01, beta1=0.9, beta2=0.999, 
                          n_anneal=2000, verbose=True):
    """
    Adam optimizer on the simplex with smooth (log-sum-exp) objective.
    Uses adaptive temperature annealing.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    while time.monotonic() - t0 < t_budget:
        t += 1
        
        # Anneal temperature
        if t < n_anneal:
            progress = t / n_anneal
            eps = eps_start * (eps_end / eps_start) ** progress
        else:
            eps = eps_end
        
        # Compute gradient
        grad, c_max = compute_gradient_smoothed(a, eps)
        
        # Adam update (in the simplex, we do projected step)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAdam t={t} ({elapsed:.0f}s): score={best_score:.6f} eps={eps:.5f}")
    
    return best_a, best_score

def subgradient_momentum(a0, t_budget, n_restarts=5, verbose=True):
    """
    Subgradient method with momentum for the non-smooth max objective.
    Uses multiple random restarts.
    """
    best_a = a0.copy()
    best_score = 2.0 * len(a0) * fast_convolve(a0).max()
    n = len(a0)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for restart in range(n_restarts):
        # Initialize: either from best or perturbed
        if restart == 0:
            a = best_a.copy()
        else:
            # Perturbation: mix with random
            a = (1 - 0.3) * best_a + 0.3 * rng.dirichlet(np.ones(n))
            a = a / a.sum()
        
        s = np.zeros(n)  # Momentum
        lr = 1.0
        no_improve = 0
        
        t_restart = time.monotonic()
        while time.monotonic() - t0 < t_budget and time.monotonic() - t_restart < t_budget / n_restarts:
            conv = fast_convolve(a)
            M = conv.max()
            score = 2.0 * n * M
            
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 200:
                break
            
            # Find active peaks (top-k)
            k_active = 20
            top_k = np.argpartition(conv, -k_active)[-k_active:]
            c_top = conv[top_k]
            
            # Softmax weights for active peaks
            temp = 0.01 * M
            w = np.exp((c_top - c_top.max()) / temp)
            w /= w.sum()
            
            # Compute subgradient
            grad = np.zeros(n)
            for idx in range(len(top_k)):
                if w[idx] < 1e-10:
                    continue
                k = int(top_k[idx])
                # grad[j] += w[idx] * (a[k-j] + a[j]) for valid j
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                # a[k-j] for j in [j_start, j_end]
                # When j = j_start: a[k - j_start]
                # When j = j_end: a[k - j_end]
                a_part1 = a[k - j_end:k - j_start + 1][::-1]  # a[k-j] for j in [j_start, j_end]
                a_part2 = a[j_start:j_end + 1]  # a[j] for j in [j_start, j_end]
                grad[j_start:j_end + 1] += w[idx] * (a_part1 + a_part2)
            
            # Step size (standard subgradient)
            lr = M / (np.sum(grad ** 2) + 1e-30)
            
            # Momentum update
            s = 0.9 * s + grad
            a_new = a - lr * 0.5 * s  # Damped step
            a_new = project_simplex(a_new)
            
            # Check if we improved
            conv_new = fast_convolve(a_new)
            if conv_new.max() < M:
                a = a_new
            else:
                # Halve the step
                s = 0.5 * s
                a_new = a - lr * 0.25 * s
                a_new = project_simplex(a_new)
                conv_new = fast_convolve(a_new)
                if conv_new.max() < M:
                    a = a_new
                else:
                    # Revert, reduce momentum
                    s = 0.5 * s
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad restart={restart} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def coord_descent_polish(a0, t_budget, verbose=True):
    """Coordinate descent with exact line search for polishing."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < t_budget:
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            # If we change a[i] by delta, the affected convolution values are:
            # conv[k] changes for k in [i, i+n-1] (capped at 2n-2)
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            if k_start > 0:
                left_max = conv[:k_start].max() if k_start < len(conv) else 0
            else:
                left_max = 0
            if k_end + 1 < len(conv):
                right_max = conv[k_end + 1:].max()
            else:
                right_max = 0
            unaff_max = max(left_max, right_max)
            
            # For the affected range, conv[k] = old_conv[k] + 2*delta*a[k-i] + delta^2 (if k=2i)
            # We want to minimize max(unaff_max, max_{k in [k_start,k_end]} new_conv[k])
            # This gives us the optimal delta.
            
            a_shift = a[k_start - i : k_end - i + 1]  # a[k-i] for k in [k_start, k_end]
            
            # new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2 * (1 if k==2i else 0)
            # We want: new_conv[k] <= new_M for all affected k, and new_M as small as possible.
            # Also need a[i] + delta >= 0.
            
            # The optimal new_M is the smallest value such that:
            # conv[k] + 2*delta*a[k-i] + delta^2*(k==2i) <= new_M for all affected k
            # and a[i] + delta >= 0
            # and new_M >= unaff_max
            
            # Binary search on new_M (or delta)
            # For a given delta, new_M = max(unaff_max, max_k (conv[k] + 2*delta*a[k-i] + delta^2*(k==2i)))
            # We want to minimize new_M over delta in [-a[i], inf).
            
            # This is a 1D optimization problem. Let's use a simple approach:
            # Try a few candidate deltas and pick the best.
            
            best_delta = 0.0
            best_new_M = M
            
            # The function new_M(delta) is convex in delta (max of affine functions + possibly one quadratic)
            # So we can use ternary search or just try several points.
            
            # Simple approach: try geometric spacing around 0
            candidates = []
            base = max(a[i], 1e-10)
            for factor in [0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.15, 1.3, 1.5, 2.0]:
                delta = (factor - 1.0) * base
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                candidates.append(delta)
            
            # Also try forcing some values to zero
            candidates.append(-a[i])  # Set a[i] to 0
            
            for delta in candidates:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                # Compute new affected conv values
                new_affected = conv[k_start:k_end+1].copy() + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] += delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update convolution
                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                
                score = 2.0 * n * M
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and no_improve_sweeps == 0 and (no_improve_sweeps + improved) % 500 == 3:
            pass  # Skip verbose to save time
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Adam optimizer and adaptive temperature annealing
2. L-BFGS quasi-Newton refinement via softmax parameterization
3. Coordinate descent with exact line search for final polishing

5. Multi-start with diverse initializations and structured perturbations
6. Multi-size exploration to find optimal length

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_gradient(a, eps):
    """
    Compute gradient of f_eps(a) = eps * logsumexp(conv(a,a) / eps).
    grad[j] = sum_k p[k] * (a[k-j] + a[j]) where p = softmax(c/eps)
    Returns (gradient, max_c).
    """
    n = len(a)
    c = fast_convolve(a)
    c_scaled = c / eps
    c_scaled -= c_scaled.max()
    exp_vals = np.exp(c_scaled)
    p = exp_vals / exp_vals.sum()
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(P * np.conj(A), fft_size)[:n]
    grad = cross + a
    return grad, float(c.max())


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def adam_smooth_optimize(a0, t_budget, eps_start, eps_end, lr=0.1,
                         beta1=0.9, beta2=0.999, n_anneal=2000, verbose=True, label=""):
    """
    Adam optimizer on the simplex with smooth log-sum-exp objective.
    Adaptive temperature annealing from eps_start to eps_end.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    while time.monotonic() - t0 < t_budget:
        t += 1

        if t < n_anneal:
            progress = t / n_anneal
            eps = eps_start * (eps_end / eps_start) ** progress
        else:
            eps = eps_end

        grad, c_max = compute_smooth_gradient(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a - step
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam {label} t={t} ({elapsed:.0f}s): score={best_score:.6f} eps={eps:.6f}")

    return best_a, best_score


def lbfgs_smooth_optimize(a0, t_budget, eps=0.01, m_lbfgs=20, max_iter=5000, verbose=True, label=""):
    """
    L-BFGS on the smooth objective using softmax parameterization.
    z -> a = softmax(z), optimize f(z) = eps * logsumexp(conv(softmax(z), softmax(z))/eps)
    """
    n = len(a0)
    a_init = np.maximum(a0.copy(), 0.0)
    a_init = a_init / a_init.sum()
    # Initialize z from a_init: z = log(a) (approximately)
    z = np.log(np.maximum(a_init, 1e-300)).copy().astype(np.float64)

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()

    t0 = time.monotonic()
    s_hist = []
    y_hist = []
    rho_hist = []
    prev_grad = None
    prev_z = None

    def get_grad_z(z_vec):
        a_vec = softmax(z_vec)
        c = fast_convolve(a_vec)
        c_scaled = c / eps
        c_scaled -= c_scaled.max()
        exp_vals = np.exp(c_scaled)
        p = exp_vals / exp_vals.sum()

        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        P = rfft(p, fft_size)
        A = rfft(a_vec, fft_size)
        cross = irfft(P * np.conj(A), fft_size)[:n]
        grad_a = cross + a_vec

        dot = np.dot(grad_a, a_vec)
        grad_z = a_vec * (grad_a - dot)
        return grad_z, a_vec, float(c.max())

    # Initial gradient
    gz, a_cur, c_max = get_grad_z(z)
    score = 2.0 * n * c_max
    if score < best_score:
        best_score = score
        best_a = a_cur.copy()

    for it in range(max_iter):
        if time.monotonic() - t0 > t_budget:
            break

        # L-BFGS two-loop recursion to compute search direction
        if prev_grad is not None:
            s_vec = z - prev_z
            y_vec = gz - prev_grad
            sy = np.dot(s_vec, y_vec)
            if sy > 1e-16:
                # Add to history
                s_hist.append(s_vec.copy())
                y_hist.append(y_vec.copy())
                rho_hist.append(1.0 / sy)
                if len(s_hist) > m_lbfgs:
                    s_hist.pop(0)
                    y_hist.pop(0)
                    rho_hist.pop(0)

                # Two-loop recursion
                q = gz.copy()
                alphas = []
                for i in range(len(s_hist) - 1, -1, -1):
                    alpha = rho_hist[i] * np.dot(s_hist[i], q)
                    alphas.append(alpha)
                    q -= alpha * y_hist[i]
                alphas.reverse()

                # Initial Hessian approximation
                if len(s_hist) > 0:
                    gamma = np.dot(s_hist[-1], y_hist[-1]) / np.dot(y_hist[-1], y_hist[-1])
                else:
                    gamma = 1.0
                r = gamma * q

                for i in range(len(s_hist)):
                    beta = rho_hist[i] * np.dot(y_hist[i], r)
                    r += (alphas[i] - beta) * s_hist[i]

                direction = -r
            else:
                direction = -gz / (np.linalg.norm(gz) + 1e-30)
        else:
            direction = -gz / (np.linalg.norm(gz) + 1e-30)

        # Line search (backtracking)
        alpha = 1.0
        f_cur = eps * np.log(np.sum(np.exp((fast_convolve(a_cur) - fast_convolve(a_cur).max()) / eps)))
        dg = np.dot(direction, gz)

        for _ in range(30):
            z_trial = z + alpha * direction
            gz_trial, a_trial, c_max_trial = get_grad_z(z_trial)
            f_trial = eps * np.log(np.sum(np.exp((fast_convolve(a_trial) - fast_convolve(a_trial).max()) / eps)))
            if f_trial <= f_cur + 0.0001 * alpha * dg:
                break
            alpha *= 0.5

        z_new = z + alpha * direction
        gz_new, a_new, c_max_new = get_grad_z(z_new)
        score_new = 2.0 * n * c_max_new

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        prev_z = z.copy()
        prev_grad = gz.copy()
        z = z_new
        gz = gz_new
        a_cur = a_new

        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LBFSGS {label} it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_momentum(a0, t_budget, n_restarts=3, verbose=True, label=""):
    """
    Subgradient method with momentum for the non-smooth max objective.
    Uses top-k peak weighting and decreasing step sizes.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    t_per_restart = t_budget / n_restarts

    for restart in range(n_restarts):
        if time.monotonic() - t0 > t_budget:
            break

        if restart == 0:
            a = best_a.copy()
        elif restart == 1:
            a = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n) * 0.5)
            a = a / a.sum()
        else:
            a = best_a + rng.standard_normal(n) * 0.02
            a = np.maximum(a, 0)
            a = a / a.sum()

        momentum = np.zeros(n)
        step_scale = 1.0
        no_improve = 0
        t_restart = time.monotonic()

        while time.monotonic() - t_restart < t_per_restart and time.monotonic() - t0 < t_budget:
            conv = fast_convolve(a)
            M = float(conv.max())
            score = 2.0 * n * M

            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 300:
                break

            # Top-k active peaks with softmax weighting
            k_top = min(30, 2 * n - 1)
            if k_top == 2 * n - 1:
                peaks = np.arange(2 * n - 1)
                weights = np.ones(k_top) / k_top
            else:
                peaks = np.argpartition(conv, -k_top)[-k_top:]
                c_peaks = conv[peaks]
                temp = 0.005 * M
                weights = np.exp((c_peaks - c_peaks.max()) / temp)
                weights /= weights.sum()

            # Compute subgradient
            grad = np.zeros(n)
            for idx in range(len(peaks)):
                if weights[idx] < 1e-12:
                    continue
                k = int(peaks[idx])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                a_part1 = a[k - j_end:k - j_start + 1][::-1]
                a_part2 = a[j_start:j_end + 1]
                grad[j_start:j_end + 1] += weights[idx] * (a_part1 + a_part2)

            # Step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-30:
                break
            lr = step_scale * M / (grad_norm * grad_norm)
            lr = min(lr, 0.5)

            # Momentum update
            momentum = 0.85 * momentum + grad
            a_new = a - lr * momentum
            a_new = project_simplex(a_new)

            # Accept if improvement
            conv_new = fast_convolve(a_new)
            if float(conv_new.max()) < M - 1e-15:
                a = a_new
            else:
                step_scale *= 0.7
                if step_scale < 1e-10:
                    break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} restart={restart} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    """
    Coordinate descent with candidate-based line search for polishing.
    Uses O(n) update per coordinate change.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve_sweeps = 0
    sweep = 0

    while time.monotonic() - t0 < t_budget:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                left_max = float(conv[:k_start].max())
            else:
                left_max = 0.0
            if k_end + 1 < len(conv):
                right_max = float(conv[k_end + 1:].max())
            else:
                right_max = 0.0
            unaff_max = max(left_max, right_max)

            a_shift = a[k_start - i:k_end - i + 1]
            affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Try multiple step sizes
            base = max(a[i], 1e-12)
            factors = [0.3, 0.5, 0.7, 0.85, 0.93, 0.97, 1.0,
                       1.03, 1.07, 1.12, 1.2, 1.35, 1.6, 2.0]

            for factor in factors:
                delta = (factor - 1.0) * base
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_aff = affected + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] += delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            # Also try zeroing
            delta_zero = -a[i]
            new_aff = affected + 2.0 * delta_zero * a_shift
            if 2 * i >= k_start and 2 * i <= k_end:
                idx = 2 * i - k_start
                new_aff = new_aff.copy()
                new_aff[idx] += delta_zero * delta_zero
            new_M = max(unaff_max, float(new_aff.max()))
            if new_M < best_new_M - 1e-16:
                best_new_M = new_M
                best_delta = delta_zero

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                score = 2.0 * n * M
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweep} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def fine_cd_polish(a0, t_budget, verbose=False, label=""):
    """Fine coordinate descent with very small steps for last-mile improvements."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < t_budget:
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                left_max = float(conv[:k_start].max())
            else:
                left_max = 0.0
            if k_end + 1 < len(conv):
                right_max = float(conv[k_end + 1:].max())
            else:
                right_max = 0.0
            unaff_max = max(left_max, right_max)

            a_shift = a[k_start - i:k_end - i + 1]
            affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine steps
            for factor in [0.995, 0.99, 0.997, 0.993, 1.0, 1.003, 1.007, 1.01, 1.015]:
                delta = (factor - 1.0) * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_aff = affected + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] += delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                score = 2.0 * n * M
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def make_initialization(n, method, rng):
    """Create diverse initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "cosine":
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(0.5, 3.0)
        a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
        a = np.maximum(a, 0)
        a = a / a.sum()
        return a
    elif method == "periodic":
        period = int(rng.integers(2, 10))
        a = np.ones(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.5 + 0.5 * np.cos(2 * np.pi * phase) ** 2
        a = a / a.sum()
        return a
    elif method == "two_hump":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
        a = a / a.sum()
        return a
    elif method == "sinc_like":
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        a = np.abs(np.sin(np.pi * (t - center) / n) / (np.pi * (t - center) / n + 1e-10))
        a = a + 0.1
        a = a / a.sum()
        return a
    else:
        return np.ones(n) / n


def propose_candidate():
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        parent = np.ones(n_init) / n_init
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform init: n={n_init}, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    n = len(best_seq)

    def try_update(a_new, label):
        nonlocal best_seq, best_score
        if a_new is None:
            return
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            return
        a_new = a_new / s
        try:
            sc = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            print(f"  NEW BEST [{label}]: {best_score:.6f} (n={len(best_seq)})")

    # === Phase 1: Smooth Adam with multiple starting points ===
    print(f"\n=== Phase 1: Smooth Adam Optimization ===")
    n_anneal = 3000

    # Main run from parent
    remaining = deadline - time.monotonic() - 800
    if remaining > 100:
        t1 = min(remaining * 0.35, 350.0)
        print(f"  From parent, budget={t1:.0f}s")
        a1, s1 = adam_smooth_optimize(best_seq, t1,
                                       eps_start=1.0/n, eps_end=0.02/n,
                                       lr=0.5, n_anneal=n_anneal, verbose=True, label="parent")
        try_update(a1, "adam_parent")

    # From diverse initializations at same n
    remaining = deadline - time.monotonic() - 700
    for init_method in ["cosine", "periodic", "two_hump"]:
        if remaining < 60:
            break
        t_i = min(remaining * 0.2, 150.0)
        init = make_initialization(n, init_method, rng)
        a_i, s_i = adam_smooth_optimize(init, t_i,
                                         eps_start=1.0/n, eps_end=0.02/n,
                                         lr=0.5, n_anneal=n_anneal, verbose=False, label=init_method)
        try_update(a_i, f"adam_{init_method}")
        remaining = deadline - time.monotonic() - 600

    # === Phase 2: L-BFGS refinement ===
    print(f"\n=== Phase 2: L-BFGS Refinement (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 500
    if remaining > 60:
        t2 = min(remaining * 0.3, 200.0)
        print(f"  Budget={t2:.0f}s")
        a2, s2 = lbfgs_smooth_optimize(best_seq, t2, eps=0.05/n, m_lbfgs=20,
                                        max_iter=3000, verbose=True, label="lbfgs_main")
        try_update(a2, "lbfgs_main")

        # L-BFGS from a perturbed version
        remaining = deadline - time.monotonic() - 400
        if remaining > 60:
            perturbed = best_seq + rng.standard_normal(n) * 0.01
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t2b = min(remaining * 0.3, 150.0)
            a2b, s2b = lbfgs_smooth_optimize(perturbed, t2b, eps=0.05/n, m_lbfgs=20,
                                              max_iter=2000, verbose=False, label="lbfgs_pert")
            try_update(a2b, "lbfgs_pert")

    # === Phase 3: Subgradient with momentum ===
    print(f"\n=== Phase 3: Subgradient Momentum (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 350
    if remaining > 60:
        t3 = min(remaining * 0.4, 250.0)
        print(f"  Budget={t3:.0f}s")
        a3, s3 = subgradient_momentum(best_seq, t3, n_restarts=3, verbose=True, label="subgrad")
        try_update(a3, "subgradient")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 60:
        t4 = min(remaining * 0.5, 250.0)
        print(f"  Budget={t4:.0f}s")
        a4, s4 = coord_descent_polish(best_seq, t4, verbose=True, label="cd_main")
        try_update(a4, "coord_descent")

    # === Phase 5: Try different sequence lengths ===
    print(f"\n=== Phase 5: Multi-size Exploration (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 150
    if remaining > 100:
        for n_try in [n * 2, n // 2, n + 128]:
            if remaining < 50:
                break
            if n_try < 64 or n_try > 8192:
                continue
            if n_try == n:
                continue
            t_try = min(remaining * 0.25, 120.0)
            # Initialize from parent (padded or truncated)
            if n_try > n:
                init_try = np.zeros(n_try)
                init_try[:n] = best_seq
                init_try[n:] = best_seq.sum() / (n_try - n)
            else:
                init_try = best_seq[:n_try]
            init_try = np.maximum(init_try, 0)
            init_try = init_try / init_try.sum()

            a_try, s_try = adam_smooth_optimize(init_try, t_try,
                                                 eps_start=1.0/n_try, eps_end=0.02/n_try,
                                                 lr=0.5, n_anneal=n_anneal, verbose=False, label=f"n{n_try}")
            try_update(a_try, f"size_{n_try}")
            remaining = deadline - time.monotonic() - 120

    # === Phase 6: Second round of refinement ===
    print(f"\n=== Phase 6: Second Round Refinement (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 80
    if remaining > 60:
        n_best = len(best_seq)
        # Short Adam pass with lower temperature
        t6a = min(remaining * 0.3, 100.0)
        a6a, s6a = adam_smooth_optimize(best_seq, t6a,
                                         eps_start=0.2/n_best, eps_end=0.005/n_best,
                                         lr=0.2, n_anneal=1000, verbose=False, label="round2_adam")
        try_update(a6a, "round2_adam")

        # L-BFGS with lower eps
        remaining = deadline - time.monotonic() - 40
        if remaining > 30:
            t6b = min(remaining * 0.4, 80.0)
            a6b, s6b = lbfgs_smooth_optimize(best_seq, t6b, eps=0.02/n_best, m_lbfgs=20,
                                              max_iter=2000, verbose=False, label="round2_lbfgs")
            try_update(a6b, "round2_lbfgs")

    # === Phase 7: Final Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Final CD Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 20
    if remaining > 20:
        t7 = min(remaining * 0.7, 60.0)
        a7, s7 = coord_descent_polish(best_seq, t7, verbose=False, label="final_cd")
        try_update(a7, "final_cd")

    # === Phase 8: Ultra-fine polishing ===
    remaining = deadline - time.monotonic() - 5
    if remaining > 10:
        a8, s8 = fine_cd_polish(best_seq, min(remaining * 0.8, 30.0), verbose=False, label="ultra_fine")
        try_update(a8, "ultra_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
