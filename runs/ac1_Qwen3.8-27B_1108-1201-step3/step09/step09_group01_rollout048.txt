**Algorithm: KKT-based iteration**

1. Start with some a.
2. Compute conv(a,a), find the set S of indices where conv[k] is close to max (say within epsilon).
3. Compute g[i] = 2 * sum_{k in S} a[k-i] (with bounds) for each i.
4. Solve for the "equilibrium": we want a[i] * (g[i] - lambda) = 0 (complementary slackness).
5. Update: a_new[i] = max(0, a[i] * exp(-eta * (g[i] - lambda)))
6. Normalize.
7. Repeat.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def cross_corr_fft(a, h):
    """
    Compute cross-correlation: result[i] = sum_j a[j] * h[j+i] for i=0..len(a)-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    m = len(h)
    # Need size >= n + m - 1 for linear correlation
    size = 1 << (n + m - 1).bit_length()
    A = rfft(a, size)
    H = rfft(h, size)
    # Cross-correlation: ifft(conj(A) * H)[i] = sum_j a[j] * h[j+i] (circular)
    # With sufficient padding, circular = linear for the indices we need
    result = irfft(np.conj(A) * H, size)
    return result[:n]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def kkt_mirror_descent(a0, time_limit, beta_init=20.0, eta_init=0.05, verbose=False, label=""):
    """
    KKT active-set mirror descent.
    Uses soft active set (log-sum-exp weights) and FFT-based gradient.
    Operates on the simplex (sum=1).
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    conv = fast_convolve(a)
    best_M = float(conv.max())
    best_score = 2.0 * n * best_M

    beta = beta_init
    eta = eta_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    momentum = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())

        # Compute soft active set weights
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -80, 0)
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
            wsum = 1.0
        weights = weights / wsum

        # Compute gradient via FFT cross-correlation
        # g[i] = 2 * sum_k weights[k] * a[k-i] (where 0 <= k-i <= n-1)
        # = 2 * cross_corr(a, weights)[i] ... need to be careful with indexing
        # g[i] = 2 * sum_j a[j] * weights[i+j] for j=0..n-1 where i+j < 2n-1
        # This is cross_corr with h=weights: result[i] = sum_j a[j] * h[j+i]
        g = 2.0 * cross_corr_fft(a, weights)

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a -= eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)

        # Ensure no underflow
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameters
        if no_improve > 50:
            if eta > 1e-6:
                eta *= 0.9
            if beta < 500:
                beta *= 1.3
            no_improve = 0
            if eta < 1e-7 and beta > 400:
                break
        elif it % 100 == 0:
            beta *= 1.05

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT-MD {label} it={it} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.5f} beta={beta:.1f}")

    return best_a, best_score

                # Update affected convolution values
                # conv[k] changes by 2*delta*a[k-i] for k in [i, i+n-1]
                # except at k=2i where it's 2*delta*a[i] + delta^2
                j_start = max(0, k_start - i)
                j_end = min(n - 1, k_end - i)
                idx_in_affected = np.arange(k_start, k_end + 1) - i
                idx_valid = (idx_in_affected >= 0) & (idx_in_affected < n)

                new_affected = affected.copy()
                new_affected[idx_valid] += 2.0 * delta * a[idx_in_affected[idx_valid]]

                # Handle the k=2i special case
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * ai + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                # Apply update
                ai_old = a[i]
                a[i] = ai_old + best_delta
                sum_a_delta = best_delta

                j_start = max(0, k_start - i)
                j_end = min(n - 1, k_end - i)
                idx_in_affected = np.arange(k_start, k_end + 1) - i
                idx_valid = (idx_in_affected >= 0) & (idx_in_affected < n)
                conv[k_start:k_end + 1][idx_valid] += 2.0 * best_delta * a[idx_in_affected[idx_valid]]

                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    conv[2 * i] = conv[2 * i] + 2.0 * best_delta * ai_old + best_delta * best_delta

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                sum_a = a.sum()
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and (it := int((time.monotonic() - t0) * 2)) % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} ({elapsed:.0f}s): best={best_score:.6f}")

    # Renormalize
    best_a = normalize(best_a)
    final_score = 2.0 * n * fast_convolve(best_a).max()
    return best_a, final_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    step_factors = [1.001, 1.002, 1.005, 1.01, 1.015, 1.02, 0.999, 0.998, 0.995, 0.99, 0.985, 0.98]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1.0
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            ai = a[i]
            best_delta = 0.0
            best_new_max = M

            for f in step_factors:
                delta = ai * (f - 1.0)
                new_val = ai + delta
                if new_val < -1e-15:
                    continue

                idx_in_affected = np.arange(k_start, k_end + 1) - i
                idx_valid = (idx_in_affected >= 0) & (idx_in_affected < n)

                new_affected = conv[k_start:k_end + 1].copy()
                new_affected[idx_valid] += 2.0 * delta * a[idx_in_affected[idx_valid]]

                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * ai + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                ai_old = a[i]
                a[i] = ai_old + best_delta

                idx_in_affected = np.arange(k_start, k_end + 1) - i
                idx_valid = (idx_in_affected >= 0) & (idx_in_affected < n)
                conv[k_start:k_end + 1][idx_valid] += 2.0 * best_delta * a[idx_in_affected[idx_valid]]

                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    conv[2 * i] += 2.0 * best_delta * ai_old + best_delta * best_delta

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                sum_a = a.sum()
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    best_a = normalize(best_a)
    return best_a, 2.0 * n * fast_convolve(best_a).max()


def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """LP fictitious play with Nesterov averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = normalize(initial.astype(np.float64))

    a_sequent = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            a_new, _ = solve_lp_best_response(b_new, solver)
            if a_new is not None:
                a = a_new
            else:
                a = b_new

        gamma_it = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma_it) * a_sequent + gamma_it * a

        for cand in [a, a_sequent, 0.5 * (a + a_sequent)]:
            cand = normalize(cand)
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_diverse(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Bimodal
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.65, 0.8)
    w = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    inits.append(("bimodal", normalize(a)))

    # Trimodal
    centers = [n * rng.uniform(0.1, 0.3), n * 0.5, n * rng.uniform(0.7, 0.9)]
    w = n * rng.uniform(0.04, 0.12)
    a = sum(np.exp(-0.5 * ((t - c) / w) ** 2) for c in centers)
    inits.append(("trimodal", normalize(a)))

    # Periodic ridges
    period = int(n / rng.integers(4, 12))
    ripple = rng.uniform(0.2, 0.5)
    a = 0.5 + ripple * np.cos(2 * np.pi * t / period + rng.uniform(0, np.pi))
    inits.append(("ridges", normalize(a)))

    # Power law (concentrated in middle)
    center = n / 2.0
    sigma = n * rng.uniform(0.1, 0.3)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    inits.append(("gaussian", normalize(a)))

    # Parent interpolation
    if parent is not None:
        inits.append(("parent_interp", interpolate_seq(parent, n)))

    return inits


def perturb_support(a, rng, frac=0.2):
    """Randomly zero out a fraction and redistribute."""
    n = len(a)
    a_new = a.copy()
    n_zero = max(1, int(n * frac))
    zero_idx = rng.choice(n, size=n_zero, replace=False)
    a_new[zero_idx] = 0
    # Add small random values to some zero positions
    zero_mask = (a_new < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.1))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a_new[add_idx] = rng.random(n_add) * a.max() * 0.3
    return normalize(a_new)


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining KKT mirror descent, coordinate descent,
    and LP fictitious play across multiple sequence lengths.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick KKT-CD ===
    print("\n=== Phase 1: Diverse Init + KKT/CD ===")
    t1_start = time.monotonic()
    n_values = [1024, 2048, 3072]
    if parent is not None and len(parent) not in n_values:
        n_values.insert(0, len(parent))

    candidates = []

    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 250:
            break

        inits = init_diverse(n, rng, parent)

        for name, init_cand in inits:
            t_left = 250.0 - (time.monotonic() - t1_start)
            if t_left < 10:
                break

            # Quick KKT mirror descent
            t_kkt = min(20.0, t_left * 0.5)
            a_kkt, s_kkt = kkt_mirror_descent(init_cand, t_kkt, beta_init=30.0, eta_init=0.03)

            # Quick coordinate descent
            t_cd = min(15.0, t_left * 0.3)
            a_cd, s_cd = fast_coord_descent(a_kkt, t_cd)

            candidates.append((s_cd, a_cd, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, top-3: " +
          ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in candidates[:3]]))

    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Intensive KKT mirror descent ===
    print(f"\n=== Phase 2: Intensive KKT (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(5):
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  KKT round {rnd+1}/5, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")

            beta_try = 20.0 + 20.0 * rnd
            eta_try = 0.05 / (1.0 + 0.3 * rnd)
            a_kkt, s_kkt = kkt_mirror_descent(current, t_alloc, beta_init=beta_try, eta_init=eta_try,
                                              verbose=True, label=f"r{rnd}")

            best_seq, best_score = check_and_update(a_kkt, best_seq, best_score, f"kkt_r{rnd}")

            # Perturb for next round
            if rnd < 3:
                current = perturb_support(best_seq, rng, frac=0.1 + 0.05 * rnd)
            else:
                current = best_seq.copy()

    # === Phase 3: Coordinate descent refinement ===
    print(f"\n=== Phase 3: CD Refinement (best={best_score:.6f}) ===")
    t3_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for rnd in range(4):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 120.0)
            print(f"  CD round {rnd+1}/4, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")
            a_cd, s_cd = fast_coord_descent(best_seq, t_alloc, verbose=True, label=f"cd{rnd}")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_r{rnd}")

    # === Phase 4: LP Fictitious Play ===
    print(f"\n=== Phase 4: LP-FP (best={best_score:.6f}) ===")
    t4_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        lp_lengths = [n_best]
        for alt in [2048, 3072]:
            if alt != n_best:
                lp_lengths.append(alt)
        lp_lengths = lp_lengths[:2]

        for n_lp in lp_lengths:
            rem = deadline - time.monotonic() - 250
            if rem < 60:
                break
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  LP-FP n={n_lp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_lp) if best_seq is not None else None
            a_fp, score_fp, iters = lp_fictitious_play(n_lp, t_alloc, initial=initial, solver=solver, verbose=True)
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"lpfp_n{n_lp}")

    # === Phase 5: Fine polishing ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.6f}) ===")
    t5_start = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_fine = min(rem * 0.5, 150.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine, verbose=False, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_1")

    # === Phase 6: Final intensive optimization ===
    print(f"\n=== Phase 6: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            # One more KKT pass with high beta
            t_final = min(rem * 0.4, 60.0)
            a_fin, s_fin = kkt_mirror_descent(best_seq, t_final, beta_init=100.0, eta_init=0.005)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_kkt")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_fin2, s_fin2 = fine_coord_descent(best_seq, min(rem * 0.6, 30.0), label="fine2")
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
