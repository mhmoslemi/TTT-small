import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    return irfft(rfft(a, fft_size) * rfft(b, fft_size), fft_size)[:size]

def simplex_project(v):
    n = len(v)
    if n == 0: return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_T(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_solve_best_response(a, solver):
    """
    Given a, solve LP: min M s.t. (a*b)[k] <= M, b>=0, sum(b)=1.
    Returns optimal b and M.
    """
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n)
    M_var = cp.Variable()
    T = build_T(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1, b_var >= 0]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_var.value

def lp_alternate_optimize(a_init, time_limit, solver, verbose=True):

    a = a / a.sum()

    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Optimize b given a
        b_new, M_b = lp_solve_best_response(a, solver)
        if b_new is None:
            break
        b = b_new
        
        # Optimize a given b
        a_new, M_a = lp_solve_best_response(b, solver)
        if a_new is None:
            break
        a = a_new
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()
        
        # Also try the average
        avg = 0.5 * (a + b)
        avg_score = 2.0 * n * fast_convolve(avg).max()
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg.copy()
        
        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-A it={iteration} ({elapsed:.0f}s): best={best_score:.6f} M_a={M_a:.8f} M_b={M_b:.8f}")
    
    return best_a, best_score
