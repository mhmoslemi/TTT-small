def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def fw_round(a, b, b_conv, steps=200):

    n = len(a)
    
    # Optimize b given a
    for step in range(steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad_slice))
        t = 2.0 / (step + 2.0)

            grad_full[i_start:i_end + 1] = grad_slice

            i_away = int(np.argmax(away_scores))
            if b[i_away] > 1e-12 and grad_full[i_away] > 1e-12:
                t_away = min(t * 0.3, b[i_away] * 0.5)
                ks_s = i_away
                ks_e = min(ks_s + n - 1, 2 * n - 2)
                b_conv[ks_s:ks_e + 1] -= t_away * a
                b[i_away] -= t_away
                b[i_away] = max(b[i_away], 0.0)
        
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv = (1.0 - t) * b_conv
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t
    
    # Optimize a given b
    a_conv = convolve_ab(b, a)
    for step in range(steps):
        k_star = int(np.argmax(a_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]

        a_conv[ks_s:ks_e + 1] += t * b

def fw_optimize(a_init, time_limit, steps_per_round=150, verbose=False):
    """FW optimization with periodic evaluation."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    b = a.copy()
    b_conv = convolve_ab(a, b)
    best_a = a.copy()
    best_score = quick_score(a)
    t0 = time.monotonic()
    round_num = 0
    
    while time.monotonic() - t0 < time_limit:
        round_num += 1
        a, b = fw_round(a, b, b_conv, steps_per_round)
        
        # Re-normalize
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / sa
        
        b = np.maximum(b, 0)
        sb = b.sum()
        if sb < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / sb
        
        # Re-compute convolution (correct for accumulated errors)
        b_conv = convolve_ab(a, b)
        
        # Score
        sa_score = quick_score(a)
        sb_score = quick_score(b)
        if sa_score < best_score:
            best_score = sa_score
            best_a = a.copy()
        if sb_score < best_score:
            best_score = sb_score
            best_a = b.copy()
        
        if round_num % 20 == 0 and verbose:
            print(f"  FW r={round_num}: {best_score:.6f}")
    
    return best_a, best_score

    """
    Solve LP for best response b given a, using constraint generation.

    conv_ab: current convolution of a and b (for initialization)
    """
    import cvxpy as cp
    from scipy import sparse
    
    n = len(a)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Initial constraint set: top n_active peaks
    # We need conv(a, b_var) <= M_var
    # Use the current conv_ab as a guide
    top_k = np.argpartition(conv_ab, -n_active)[-n_active:]
    
    constraints = [cp.sum(b_var) == 1]
    active_set = set(top_k.tolist())
    
    # Build constraint matrix for active constraints
    # conv(a,b)[k] = sum_j a[k-j]*b[j]
    # For k in active_set, we need sum_j a[k-j]*b[j] <= M
    
    def build_constraints(active_set):

            j_min = max(0, k - n + 1)

            for j in range(j_min, j_max + 1):

            cons.append(coeffs @ b_var <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=cp.PDLP)

                prob.solve(solver=cp.ECOS)

                try:
                    prob.solve(solver=cp.SCS)
                except:
                    break
