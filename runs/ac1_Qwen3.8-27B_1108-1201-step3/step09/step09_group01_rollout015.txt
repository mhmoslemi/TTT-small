```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n
```

6. Project b onto the simplex
7. Repeat

```python
from scipy.optimize import linprog

def lp_best_response(a):
    n = len(a)
    # Build T (2n-1 x n) matrix
    # T[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    # Minimize M
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize M
    
    # Constraints: T@b - M <= 0 for each k
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # b_j >= 0, M >= 0 (actually M can be anything, but we know it's positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        b = result.x[:n]
        b = np.maximum(b, 0)
        b = b / b.sum()
        return b
    return np.ones(n) / n
```
