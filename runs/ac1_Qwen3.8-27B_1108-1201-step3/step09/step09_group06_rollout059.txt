```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP best-response with Nesterov acceleration (using HiGHS)
2. Multi-length exploration with adaptive time allocation
3. Spectral flattening (POCS-inspired) as complementary method
4. Coordinate descent and swap-based local search for final polish
5. Diverse initialization (structured + random + interpolated parent)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

# ... (helper functions)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed):
    """Solve min M s.t. T@b <= M, sum(b)=1, b>=0"""
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    # A_ub: T @ b - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def nesterov_lp(a0, time_limit, max_iters=50000):
    """Nesterov-accelerated LP best response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    x = a.copy()  # actual iterate
    y = a.copy()  # extrapolated point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    stall = 0
    prev_score = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute best response to y
        b_br = solve_lp_best_response(y)
        if b_br is None:
            break
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_br
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        
        # Update x (weighted average)
        x = ((it + 1) * x + alpha * (it + 2) * b_br) / (it + 1 + alpha * (it + 2))
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx > 1e-15:
            x = x / sx
        
        y = y_new
        
        # Evaluate
        score_x = 2.0 * n * fast_convolve(x).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_x < best_score:
            best_score = score_x
            best_a = x.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if it % 100 == 0:
            if best_score < prev_score - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_score = best_score
            
            if stall >= 10:
                # Restart with perturbation
                rng = np.random.default_rng(hash((it, n)) % 2**32)
                noise = rng.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
```

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        b_val = res.x[:n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None
        return b_val / s
    except Exception:
        return None

def compute_subgradient_active(a, n_active=5):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n), max_b
    threshold = max_b - 0.15 * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    active = active[:max(n_active, 1)]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def spectral_flatten(a0, time_limit, max_iters=5000, beta=0.7):

        a_pad[:n] = a

        P_shifted = P - P.mean()
        P_comp = P - beta * P_shifted
        P_comp = np.maximum(P_comp, 1e-15)

        A_new = A * (scale * np.sqrt(P_comp / np.maximum(P, 1e-15)))

        a = a_new
        if it % 500 == 0:

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]
        top_idx = np.argsort(contrib)[::-1][:3]

        improved = False
        for si in top_idx:
            if a[si] < 1e-14:

                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial

def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.5, maxiter=5000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    def make_obj(T_val):
        def obj(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_fn(z_in)
            b = fast_convolve(a_in)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * lse
        return obj

    def make_grad(T_val):
        def grad(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_fn(z_in)
            b = fast_convolve(a_in)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        return grad

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        try:
            res = minimize(
                fun=make_obj(stage_T), x0=z, jac=make_grad(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_fn(z)


def softmax_fn(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s
