def fast_convolve(a):

def smooth_objective_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    scaled = beta * (conv - max_conv)
    scaled = np.clip(scaled, -50, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    weights = weights / Z
    obj = max_conv + np.log(Z) / beta
    grad = grad_from_weights(weights, a)
    return obj, grad, weights, conv

def adam_update(theta, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer update step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    return theta, m, v

def softmax(theta):
    """Numerically stable softmax."""
    theta = theta - theta.max()
    exp_theta = np.exp(theta)
    return exp_theta / exp_theta.sum()

def fine_polish(a0, time_limit, verbose=False):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if a[i] < 1e-15:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_contrib = np.zeros(k_end - k_start + 1)
            for idx, k in enumerate(range(k_start, k_end + 1)):
                j = k - i
                if 0 <= j < n:
                    a_contrib[idx] = a[j]

            best_delta = 0.0
            best_new_max = M
            
            for frac in [0.01, 0.03, 0.05, 0.02, 0.04]:
                delta = -frac * a[i]
                new_affected = conv[k_start:k_end+1] + 2.0 * delta * a_contrib
                if 2*i >= k_start and 2*i <= k_end:
                    diag_idx = 2*i - k_start
                    new_affected[diag_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                conv[k_start:k_end+1] += 2.0 * best_delta * a_contrib
                if 2*i >= k_start and 2*i <= k_end:
                    conv[2*i] += best_delta * best_delta
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score

def init_symmetric(n, rng):
    """Symmetric initialization with cosine profile."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(3, 8))
    profile = np.ones(half) * 0.5
    for j in range(n_comp):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = profile[half//2] * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s

def init_bimodal(n, rng):
    """Bimodal Gaussian initialization."""
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    return a / s

def init_ridges(n, rng):
    """Periodic ridge initialization."""
    period = n / int(rng.integers(3, 10))
    t = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * t / period) + 0.1 * np.cos(2 * np.pi * t / (period * 0.7))
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s

def init_from_parent(parent, n, rng):
    """Interpolate parent to new length with perturbation."""
    old_n = len(parent)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    f = np.interp(new_x, old_x, parent)
    f = np.maximum(f, 0)
    # Add small perturbation
    f += rng.random(n) * 0.01 * f.max()
    s = f.sum()
    return f / s

def check_and_update(a, best_seq, best_score, label=""):

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_a(w, a):
    """
    Compute gradient of smoothed max-conv w.r.t. a.
    grad_a[i] = 2 * sum_k w[k] * a[k-i] (with a extended by zeros).
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv_result = irfft(W * Ar, fft_size)[:3 * n - 2]
    c = conv_result[n - 1:2 * n - 1]
    return 2.0 * c


def smooth_objective_and_grad(a, beta, top_k=None):
    """Compute smooth objective and gradient. Optionally use top-k weighting."""
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    scaled = beta * (conv - max_conv)
    scaled = np.clip(scaled, -50, 0)
    weights = np.exp(scaled)
    
    if top_k is not None and top_k < len(conv):
        # Zero out all but top-k weights
        k_smallest = np.argsort(weights)[:len(conv) - top_k]
        weights[k_smallest] = 0.0
    
    Z = weights.sum()
    if Z < 1e-30:
        weights = np.zeros_like(conv)
        weights[conv.argmax()] = 1.0
        Z = 1.0
    weights = weights / Z
    obj = max_conv + np.log(Z) / beta
    grad = compute_grad_a(weights, a)
    return obj, grad, weights, conv


def softmax(theta):
    """Numerically stable softmax."""
    theta = theta - theta.max()
    exp_theta = np.exp(theta)
    s = exp_theta.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return exp_theta / s


def gradient_descent_softmax(a_init, time_limit, lr_init=0.01, beta_start=20.0,
                             beta_end=200.0, top_k=None, verbose=False, seed=0):
    """
    Full gradient descent in softmax parameterization with Adam optimizer.
    Uses log-sum-exp smoothing with annealing beta.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    
    theta = np.log(a)
    theta -= theta.max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    total_time = time_limit
    perturb_interval = max(50, int(total_time / 5))
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / max(1, int(200.0 / (total_time / perturb_interval))))
        beta = beta_start + (beta_end - beta_start) * progress
        
        obj, grad_a, weights, conv = smooth_objective_and_grad(a, beta, top_k=top_k)
        
        # Chain rule for softmax parameterization
        dot_ga_a = np.dot(grad_a, a)
        grad_theta = a * (grad_a - dot_ga_a)
        
        # Adam update with learning rate decay
        lr_t = lr_init / (1.0 + 0.005 * t)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta * grad_theta)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        a = softmax(theta)

def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent for auto-convolution minimization.
    Tries multiple step sizes including aggressive ones.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                right_max = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, right_max)

            # Pre-compute contribution vector
            a_contrib = np.zeros(k_end - k_start + 1)
            for idx in range(k_end - k_start + 1):
                j = k_start + idx - i
                if 0 <= j < n:
                    a_contrib[idx] = a[j]

            best_delta = 0.0
            best_new_max = M

            # Try reducing (most useful for lowering max)
            for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
                delta = -frac * a[i]
                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_contrib
                if 2 * i >= k_start and 2 * i <= k_end:
                    diag_idx = 2 * i - k_start
                    new_aff[diag_idx] += delta * delta
                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            # Try increasing (sometimes helps by redistributing)
            for frac in [0.1, 0.2, 0.4]:
                delta = frac * a[i]
                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_contrib
                if 2 * i >= k_start and 2 * i <= k_end:
                    diag_idx = 2 * i - k_start
                    new_aff[diag_idx] += delta * delta
                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                conv[k_start:k_end + 1] += 2.0 * best_delta * a_contrib
                if 2 * i >= k_start and 2 * i <= k_end:
                    conv[2 * i] += best_delta * best_delta
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

def init_symmetric(n, rng):
    """Symmetric initialization with superposed cosines."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(3, 8))
    profile = np.ones(half) * 0.5
    for j in range(n_comp):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = profile[half // 2] * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def init_bimodal(n, rng):
    """Bimodal Gaussian initialization."""
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    return a / s


def init_ridges(n, rng):
    """Periodic ridge initialization."""
    period = n / float(int(rng.integers(3, 10)))
    t = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * t / period) + 0.1 * np.cos(4 * np.pi * t / period)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def init_from_parent(parent, n, rng):
    """Interpolate parent to new length with small perturbation."""
    old_n = len(parent)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    f = np.interp(new_x, old_x, parent)
    f = np.maximum(f, 0)
    f += rng.random(n) * 0.005 * max(f.max(), 1e-10)
    s = f.sum()
    return f / s

def perturbed_copy(a, rng, strength=0.03):
