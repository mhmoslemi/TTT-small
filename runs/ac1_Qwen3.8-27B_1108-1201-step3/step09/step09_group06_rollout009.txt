"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous approach:
1. Adaptive log-sum-exp temperature based on convergence rate
2. Hybrid subgradient with dual-driven active set weighting
3. Nesterov LP with extrapolation and periodic restart
4. Stochastic coordinate descent with line search
5. Population-based diversity with crossover
6. Multi-length geometric exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def project_simplex(v, target_sum=1.0):
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s * target_sum


def compute_score(a):
    b = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * b.max() / (s * s)


def compute_subgradient_weighted(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b * (1.0 - n_active_frac)
    weights = np.maximum(b - threshold, 0)
    total_w = weights.sum()
    if total_w < 1e-300:
        active = np.array([int(np.argmax(b))])
        g = np.zeros(n)
        k = active[0]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] = a[k - idx]
        return g, max_b
    w = weights / total_w
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w[k] * a[k - idx]
    return g, max_b


def mirror_descent_adaptive(a0, time_limit, n_active_frac_init=0.1, n_active_frac_min=0.02):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    s_target = 1.0
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.8
    no_improve = 0
    active_frac = n_active_frac_init
    prev_score = best_score

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b = compute_subgradient_weighted(a, active_frac)
        g_norm = np.sqrt(np.sum(g * g))
        if g_norm < 1e-18:
            break

        gap = best_score - f_lb
        if gap < 1e-10:
            break

        polyak_lr = gap / (g_norm * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()

        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 3.0)
            active_frac = max(n_active_frac_min, active_frac * 0.97)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.6
                active_frac = min(0.3, active_frac * 1.5)
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f} frac={active_frac:.3f}")

    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = _get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    if solver is None:
        solver = _get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_idx = it + 1
        alpha = 2.0 / (t_idx + 1.0)

        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y = y_new / sy

        momentum = (t_idx - 1.0) / (t_idx + 1.0)
        a_new = y + momentum * (y - x_prev)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sa

        x_prev = a.copy()
        a = a_new

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 10:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                x_prev = y.copy()
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lse_lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.5, maxiter=20000):
    n = len(a0)
    z0 = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300))
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    z = z0.copy()

    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def lse_func(z_in):
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            log_sum_exp = max_Tb + np.log(np.exp(Tb - max_Tb).sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def lse_jac(z_in):
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx_arr = np.arange(js, je + 1)
                grad_a[idx_arr] += 2.0 * w[k] * a[k - idx_arr]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=lse_func,
                x0=z,
                jac=lse_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if idx % 3 == 0:
            print(f"    LSE stage T={stage_T:.1f} ({elapsed:.0f}s): score={sc:.6f}")

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final


def stochastic_local_search(a0, time_limit, n_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    temperature = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        move_type = rng.integers(0, 4)

        if move_type == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            amount = a[i] * rng.uniform(0.01, 0.1)
            a_trial = a.copy()
            a_trial[i] -= amount
            a_trial[j] += amount

        elif move_type == 1:
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if len(set([i, j, k])) < 2:
                continue
            amount = a[i] * rng.uniform(0.01, 0.05) / 2.0
            a_trial = a.copy()
            a_trial[i] -= amount
            a_trial[j] += amount
            a_trial[k] += amount

        elif move_type == 2:
            noise = rng.normal(0, 0.01, n)
            a_trial = a + noise
        else:
            mask = rng.random(n) < 0.1
            n_mask = mask.sum()
            if n_mask < 2:
                continue
            scale = rng.uniform(0.8, 1.2)
            a_trial = a.copy()
            a_trial[mask] *= scale

        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        score = 2.0 * n * fast_convolve(a_trial).max()

        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        elif score < a_score if hasattr(a, '_score') else False:
            pass
        else:
            delta = score - 2.0 * n * fast_convolve(a).max()
            if np.exp(-max(delta, 0) / temperature) > rng.random():
                a = a_trial

        if step % 5000 == 0 and step > 0:
            temperature *= 0.95
        no_improve += 1
        if no_improve > 50000:
            break
        if no_improve > 5000:
            temperature = max(temperature * 1.1, 0.001)

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        g, _ = compute_subgradient_weighted(a, 0.15)
        g_norm = np.sqrt(g @ g)
        if g_norm < 1e-18:
            break

        top_neg = np.argsort(g)[:5]
        top_pos = np.argsort(g)[::-1][:5]

        improved = False
        for si in top_neg:
            for di in top_pos:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved and rnd > 10000:
            break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (x * (1 - x)) ** 0.5
    else:
        a = np.ones(n)
        a[:n//4] *= 0.5
        a[3*n//4:] *= 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        t_total = min(rem * 0.30, 180.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best_alt = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        t_lp = t_total * 0.50
        if t_lp > 10:
            a_alt, _ = nesterov_lp_refine(cur_best_alt, t_lp, solver=solver, n_max_iters=20000)
            try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None:
            t_md = min(rem * 0.25, 50.0)
            if t_md > 10:
                a_alt2, _ = mirror_descent_adaptive(best_seq, t_md, n_active_frac_init=0.08)
                try_update(a_alt2, f"md_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = lse_lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.1, maxiter=10000)
        try_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_md_fin = min(rem * 0.25, 50.0)
        a_md_fin, _ = mirror_descent_adaptive(best_seq, t_md_fin, n_active_frac_init=0.05, n_active_frac_min=0.01)
        try_update(a_md_fin, "final_md")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd_fin = min(rem * 0.25, 40.0)
        a_cd_fin, _ = coordinate_descent(best_seq, t_cd_fin, n_rounds=50000)
        try_update(a_cd_fin, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.5, 30.0)
        if t_lp_fin > 5:
            a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
            try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
