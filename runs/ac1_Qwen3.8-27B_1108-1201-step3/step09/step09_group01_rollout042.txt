            dx = x - x0
            vals = base + coeffs * dx
            if has_quad:
                vals = vals.copy()
                vals[idx_2i] += dx * dx
            M = max(U, float(vals.max()))
            return M, Sx

        # Current score
        M0, S0 = get_M_and_S(x0)
        f0 = 2.0 * n * M0 / (S0 * S0)

        # Try candidate x values
        best_x = x0
        best_f = f0

        if x0 > 1e-10:
            for factor in [0.05, 0.1, 0.2, 0.4, 0.6, 1.4, 1.8, 2.5, 4.0]:
                x_c = x0 * factor
                M_c, S_c = get_M_and_S(x_c)
                f_c = 2.0 * n * M_c / (S_c * S_c)
                if f_c < best_f:
                    best_f = f_c
                    best_x = x_c
            # Also try zeroing
            M_c, S_c = get_M_and_S(0.0)
            f_c = 2.0 * n * M_c / (S_c * S_c)
            if f_c < best_f:
                best_f = f_c
                best_x = 0.0
        else:
            # x0 is near zero
            for scale in [S * 0.001, S * 0.005, S * 0.01, S * 0.02, S * 0.05]:
                x_c = scale
                M_c, S_c = get_M_and_S(x_c)
                f_c = 2.0 * n * M_c / (S_c * S_c)
                if f_c < best_f:
                    best_f = f_c
                    best_x = x_c

        # Golden section search
        if best_x > 1e-10:
            spread = max(best_x * 0.4, 1e-10)
        else:
            spread = max(x0 * 0.5, S * 0.005)
            if spread < 1e-12:
                spread = S * 0.005

        lo = max(0.0, best_x - spread)
        hi = best_x + spread

        c = hi - g * (hi - lo)
        d = lo + g * (hi - lo)
        Mc, Sc = get_M_and_S(c)
        fc = 2.0 * n * Mc / (Sc * Sc)
        Md, Sd = get_M_and_S(d)
        fd = 2.0 * n * Md / (Sd * Sd)

        for _ in range(max_steps_per_coord):
            if hi - lo < 1e-14 * max(1.0, abs(hi)):
                break
            if fc <= fd:
                hi = d
                d = c
                fd = fc
                c = hi - g * (hi - lo)
                Mc, Sc = get_M_and_S(c)
                fc = 2.0 * n * Mc / (Sc * Sc)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + g * (hi - lo)
                Md, Sd = get_M_and_S(d)
                fd = 2.0 * n * Md / (Sd * Sd)

        x_opt = (lo + hi) / 2.0
        Mo, So = get_M_and_S(x_opt)
        fo = 2.0 * n * Mo / (So * So)
        if fo < best_f:
            best_x = x_opt
            best_f = fo

        # Apply update if improved
        if best_f < f0 - 1e-14:
            dx = best_x - x0
            a[i] = best_x
            conv[ks] += coeffs * dx
            if has_quad:
                conv[two_i] += dx * dx
            S = S - x0 + best_x
            improved_count += 1

    return improved_count, S


def cd_until_convergence(a, max_time, max_sweeps=500, verbose=False, label=""):
    """Run coordinate descent until convergence or time exhausted."""
    n = len(a)
    S = a.sum()
    conv = fast_convolve(a)
    t0 = time.monotonic()
    sweep_count = 0
    no_improve_count = 0
    best_score = 2.0 * n * conv.max() / (S * S)
    best_a = a.copy()

    while time.monotonic() - t0 < max_time and sweep_count < max_sweeps:
        # Random coordinate ordering
        order = np.random.permutation(n)
        improved, S = exact_sweep(a, conv, S, order)
        sweep_count += 1

        cur_score = 2.0 * n * conv.max() / (S * S)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if verbose and sweep_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweep_count} ({elapsed:.0f}s): score={best_score:.8f}")

        if no_improve_count >= 3:
            break

    return best_a, best_score


def perturb_thermal(a, rng, beta=0.05):
    """Thermal perturbation: multiplicative noise."""
    n = len(a)
    noise = rng.standard_normal(n)
    a_new = a * np.exp(beta * noise)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def perturb_active(a, conv, rng, strength=0.1):
    """Active-set targeted perturbation."""
    n = len(a)
    M = conv.max()
    threshold = M * 0.9
    active_ks = np.where(conv >= threshold)[0]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in active_ks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Reduce high-pressure, increase low-pressure
    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + noise * strength * 0.5)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def perturb_restructure(a, rng, frac=0.15):
    """Random restructure: zero out some coords, add noise."""
    n = len(a)
    a_new = a.copy()
    zero_mask = rng.random(n) < frac
    a_new[zero_mask] = 0.0
    # Add noise to remaining
    keep_mask = ~zero_mask
    a_new[keep_mask] *= (1.0 + rng.standard_normal(keep_mask.sum()) * 0.15)
    a_new = np.maximum(a_new, 0.0)
    # Add small mass to some zeroed positions
    if zero_mask.sum() > 5:
        n_add = max(1, int(zero_mask.sum() * 0.1))
        zero_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(zero_idx, size=min(n_add, len(zero_idx)), replace=False)
        a_new[add_idx] = rng.random(n_add) * a.max() * 0.3
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def init_symmetric(n, rng):
    """Symmetric initialization with cosine superposition."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 7))
    profile = np.ones(half) * 0.8
    for _ in range(n_comp):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)

    if n % 2 == 0:
        a = np.concatenate([profile, profile[:-1][::-1]])
    else:
        a = np.concatenate([profile, profile[::-1]])
    if len(a) > n:
        a = a[:n]
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Bimodal Gaussian initialization."""
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_ridges(n, rng):
    """Flat base with periodic ridges."""
    period = int(n / rng.integers(3, 8))
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.6 + rng.uniform(0.1, 0.3) * np.cos(2 * np.pi * t / period + rng.uniform(0, np.pi))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_fourier_flat(n, rng):
    """
    Initialize by trying to create flat autocorrelation.
    Use a random phase in frequency domain, take real part, make non-negative.
    """
    # Random complex spectrum with |A(f)|^2 approximately flat
    fft_size = 1 << (2 * n - 1).bit_length()()
    freqs = np.arange(fft_size)
    # Create a spectrum that's "flat" in power
    amp = np.ones(fft_size)
    phase = rng.uniform(0, 2 * np.pi, fft_size)
    # Enforce Hermitian symmetry
    X = amp * np.exp(1j * phase)
    X[1:fft_size//2] = X[1:fft_size//2]
    X[fft_size//2:] = np.conj(X[1:fft_size//2][::-1])
    X[0] = X[0].real
    if fft_size % 2 == 0:
        X[fft_size//2] = X[fft_size//2].real
    a_complex = irfft(X, fft_size)[:n]
    a = np.maximum(a_complex.real, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_polish(a0, time_limit, solver=None):
    """Fictitious play with Nesterov averaging for polishing."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy()
    b = a.copy()
    a_seq = a.copy()
    b_seq = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma = 2.0 / (iters + 2.0)
        a_seq = (1 - gamma) * a_seq + gamma * a
        b_seq = (1 - gamma) * b_seq + gamma * b

        for cand in [a, b, a_seq, b_seq, 0.5 * (a_seq + b_seq)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            s = 2.0 * n * fast_convolve(cand_n).max()
            if s < best_score:
                best_score = s
                best_a = cand_n.copy()

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def update_best(a, best_seq, best_score, label=""):
    """Check candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_seq = a_norm.copy()
        best_score = sc
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")
    return best_seq, best_score
