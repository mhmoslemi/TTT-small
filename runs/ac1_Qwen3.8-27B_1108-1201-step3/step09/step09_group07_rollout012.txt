import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix_T(a):
    """Build matrix T such that T @ b = convolve(a, b) (full)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_T(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a refined multi-phase approach:
1. LP-based best response with adaptive step sizes, line search, and averaging
2. Gradient-based peak shaving exploiting the subgradient structure of auto-convolution
3. Symmetry-aware initialization and multi-length exploration
4. Restored gradient descent with momentum for deep polishing
Key improvements: line search on LP direction, subgradient exploitation,
better time allocation, and coordination between phases.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b) (full)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_best_solver(test_n=200):
    """Benchmark LP solvers and return the fastest one."""
    rng = np.random.default_rng(0)
    a = rng.random(test_n)
    a = a / a.sum()
    T = build_conv_matrix(a)
    results = {}
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.GLOP, cp.APPS_CHOCO, cp.CBC]:
        try:
            b_var = cp.Variable(test_n, nonneg=True)
            M_var = cp.Variable()
            constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results[solver_name] = elapsed
        except Exception:
            continue
    if not results:
        return cp.ECOS
    return min(results, key=results.get)


def solve_lp_br(a_fixed, solver, verbose=False):
    """Given fixed a (sum=1, nonneg), solve LP: min M s.t. T_a @ b <= M*1, b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        alt_solvers = [cp.SCS, cp.ECOS, cp.PDLP]
        if solver in alt_solvers:
            alt_solvers.remove(solver)
        for alt in alt_solvers:
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_game_iteration(a0, time_limit, solver, n_max_iters=100000):
    """
    LP-based game iteration with adaptive step sizes and iterate averaging.
    Uses the "miracle of averaging" for convergence.
    """
    n = len(a0)
    a = a0.copy()
    avg_a = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_br(a, solver)
        if b_new is None:
            continue
        count += 1

        # Adaptive step size: larger early, smaller later
        alpha = 1.0 / np.sqrt(1.0 + 0.5 * count)

        # Update current point
        a = project_simplex((1 - alpha) * a + alpha * b_new)

        # Update average (Polyak averaging)
        avg_a = (count - 1 + 1) * 0 + ((count) * avg_a + b_new) / (count + 1)
        avg_a = project_simplex(avg_a)

        # Evaluate periodically
        if count % 20 == 0:
            sc_a = score_normalized(a)
            sc_avg = score_normalized(avg_a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()

            if count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP it={count} ({elapsed:.0f}s): best={best_score:.6f} M_br={M_b:.6f}")

    # Final check
    sc_avg = score_normalized(avg_a)
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = avg_a.copy()

    return best_a, best_score, avg_a


def lp_with_line_search(a0, time_limit, solver, n_max_iters=50000):
    """
    LP best response with adaptive line search on the actual objective.
    After computing the LP direction, try several step sizes and keep the best.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    count = 0
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_br(a, solver)
        if b_new is None:
            continue
        count += 1

        # Try several step sizes via line search
        best_local_score = best_score
        best_local_a = a.copy()
        base_alpha = 2.0 / (1.0 + 0.3 * count)
        for alpha_mult in [1.0, 0.5, 0.25, 2.0, 0.1]:
            alpha = base_alpha * alpha_mult
            a_trial = project_simplex((1 - alpha) * a + alpha * b_new)
            sc = score_normalized(a_trial)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = a_trial

        if best_local_score < best_score - 1e-12:
            a = best_local_a
            best_score = best_local_score
            best_a = a.copy()
            stall_count = 0
        else:
            # Nudge with small random perturbation
            stall_count += 1
            if stall_count % 50 == 0:
                rng = np.random.default_rng(it)
                a = project_simplex(a + 0.01 * rng.standard_normal(n))
                stall_count = 0

        if count % 100 == 0:
            elapsed = time.monotonic() - t0
            improved = best_score < prev_best - 1e-8
            prev_best = best_score
            print(f"      LP+LS it={count} ({elapsed:.0f}s): best={best_score:.6f} improved={improved}")

    return best_a, best_score


def grad_peak_shave(a0, time_limit, n_steps=50000):
    """
    Gradient-based peak shaving using the subgradient structure.
    The subgradient of max_k conv(a,a)_k at the active peak k* is 2*a_reflected.
    Uses projected gradient with adaptive step size.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta_mom = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        # Compute subgradient from the peak
        # d/d(a_i) conv(a,a)_{k} = a_{k-i} + a_i (if k-i != i) or 2*a_i (if k-i == i)
        # More precisely: sum_j a_j a_{k-j}, derivative w.r.t. a_i:
        # = a_{k-i} (from j=i term) + a_i * [k-i is valid and k-i != i] (from j=k-i term)
        # If k-i == i: derivative = 2*a_i
        # If k-i out of range: derivative = 0 (from that term)
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            val = 0.0
            if 0 <= j < n:
                val += a[j]  # from j=i in the sum
            if j != i and 0 <= j < n:
                val += a[i]  # from i=k-j in the sum (i.e., j=k-i term)
            # Actually let me be more careful:
            # conv(a,a)_k = sum_j a_j * a_{k-j}
            # d/d(a_i) = sum_j [delta_{j,i} * a_{k-j} + a_j * delta_{k-j,i}]
            #           = a_{k-i} (if k-i in range) + a_{k-i} (if k-i in range and k-i != i)
            #           = a_{k-i} + a_{k-i} [if k-i in range and k-i != i]
            #           = 2*a_{k-i} [if k-i in range and k-i != i]
            #           = 2*a_i [if k-i == i, i.e., k=2i]
            #           = a_{k-i} [if k-i in range and k-i == i ... wait that's the same case]
            # Let me redo: d/d(a_i) sum_j a_j a_{k-j}
            # = sum_j (delta_{ji} a_{k-j} + a_j delta_{k-j,i})
            # = a_{k-i} + a_{k-i}  [if 0<=k-i<n, the first term gives a_{k-i} when j=i]
            #   [the second term gives a_j when k-j=i, i.e., j=k-i, so gives a_{k-i}]
            # But if j=i and k-j=i simultaneously (i.e., k=2i), we get 2*a_i (not a_i + a_i from different terms)
            # Actually: when j=i: term is a_{k-i}. When j=k-i: term is a_j = a_{k-i}.
            # If i = k-i (k=2i): both give a_i, total = 2*a_i.
            # If i != k-i: first gives a_{k-i}, second gives a_{k-i}, total = 2*a_{k-i}.
            # So in all cases where k-i is in [0,n-1]: derivative = 2*a_{k-i}
            # If k-i is out of range: only the j=i term contributes (if valid), = a_{k-i} which is 0.
            # Wait, if k-i is out of range, the j=i term gives a_{k-i} = 0 (out of range).
            # And the j=k-i term: j is out of range, so doesn't contribute.
            # So: if 0 <= k-i < n: grad[i] = 2*a_{k-i}, else grad[i] = 0.

        # Cleaner computation:
        grad = np.zeros(n)
        valid_i = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        for i in valid_i:
            j = peak_k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Adaptive step size
        lr = 0.1 / (1.0 + 0.01 * step)

        # Momentum
        momentum = beta_mom * momentum - lr * grad
        a_new = project_simplex(a + momentum)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            momentum = 0.5 * momentum
            a_half = project_simplex(a + momentum)
            sc_half = score_normalized(a_half)
            if sc_half < best_score:
                best_score = sc_half
                best_a = a_half.copy()
                a = a_half
            else:
                momentum = 0.0
                # Small random kick to escape
                if step % 200 == 199:
                    rng = np.random.default_rng(step)
                    a = project_simplex(a + 0.005 * rng.standard_normal(n))

        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PS step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def smoothed_max_grad(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv(a,a)_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_k w_k * (d/d(a_i) conv(a,a)_k) = 2 * sum_k w_k * a_{k-i} (for valid k-i)
    # = 2 * (a * w_flipped reflected) -- use FFT
    # Actually: sum_k w_k * a_{k-i} for k-i in [0,n-1]
    # = sum_j w_{j+i} * a_j for j in [0,n-1] where j+i < 2n-1
    # This is a cross-correlation-like operation
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # We want sum_j a_j * w_{j+i} for each i
    # This is (a correlate w)[i] where correlation means sum_j a_j w_{j+i}
    # = ifft(conj(FFT(w)) * FFT(a)) ... let me think
    # sum_j a_j w_{j+i} = (conv(a_flipped, w))[n-1+i] ... getting complicated
    # Simpler: just do it directly for small n, or use the fact that it's a Toeplitz matrix-vector product
    # For efficiency with FFT: define c_i = sum_j a_j w_{j+i}
    # This is the "anti-convolution" or "cross-correlation"
    # c = cross_correlate(a, w) where (c)_i = sum_j a_j w_{j+i}
    # In terms of convolution: c_i = sum_j a_j w_{j+i} = (a * w_reversed)[?]
    # Let's just use numpy for clarity (n is at most a few thousand)
    # Actually for n up to ~2000, a direct approach is fine:
    c = np.zeros(n)
    for i in range(n):
        # sum_j a_j * w[j+i] for j in [0, n-1] and j+i in [0, 2n-2]
        js = np.arange(0, n)
        valid = js + i < len(w)
        c[i] = np.dot(a[js[valid]], w[js[valid] + i])
    return 2.0 * c


def nesterov_smooth(a0, time_limit, T_start=2.0, T_end=200.0, n_stages=8, steps_per_stage=2000):
    """Nesterov accelerated projected gradient on smoothed max objective."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_x = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = 0.05 / (1.0 + 0.003 * step_count)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count)) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1))) / 2.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = project_simplex(x + beta * (x - x_prev))

            grad = smoothed_max_grad(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_x = x.copy()
                if step_count % 1000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_x, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init(n, rng, style=0):
    """Create initial sequence of various styles."""
    x = np.linspace(0, 1, n)
    if style == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x + 0.3)
    elif style == 1:
        a = 1.0 + 0.7 * (2 * x - 1)
    elif style == 2:
        a = 1.0 + 0.4 * np.sin(4 * np.pi * x)
    elif style == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2) + 0.1
    elif style == 4:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x + 1.0)
    else:
        a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a

def coordinate_descent_polish(a0, time_limit, n_rounds=2000):
    """Fine-grained coordinate descent polishing."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        # Find top contributing indices to the peak
        js = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        contribs = a[js] * a[peak_k - js]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ti in top_idx:
            i = js[ti]
            for factor in [0.98, 0.95, 0.92, 0.88, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing a zero or small element
            min_idx = np.argmin(a)
            if a[min_idx] < best_score / (2.0 * n) * 0.1:
                a_trial = a.copy()
                a_trial[min_idx] += 0.001 / n
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and rnd > 50:
            break

        if rnd % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD round={rnd} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === PHASE 1: Intensive LP game iteration ===
    print(f"\n=== Phase 1: LP Game (n={n_best}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp, avg_lp = lp_game_iteration(best_seq, t_alloc, solver, n_max_iters=200000)
        try_update(a_lp, "lp_game")
        try_update(avg_lp, "lp_avg")

    # === PHASE 2: LP with line search ===
    print(f"\n=== Phase 2: LP + Line Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ls, _ = lp_with_line_search(best_seq, t_alloc, solver, n_max_iters=50000)
        try_update(a_ls, "lp_ls")

    # === PHASE 3: Gradient peak shaving ===
    print(f"\n=== Phase 3: Gradient Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = grad_peak_shave(best_seq, t_alloc, n_steps=100000)
        try_update(a_ps, "grad_ps")

    # === PHASE 4: Nesterov smoothed gradient ===
    print(f"\n=== Phase 4: Nesterov Smooth (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_smooth(best_seq, t_alloc, T_start=5.0, T_end=500.0,
                                  n_stages=6, steps_per_stage=1000)
        try_update(a_ng, "ng_smooth")

    # === PHASE 5: Multi-start with different inits ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for style in [1, 3, 4]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.3, 100.0)
            if t_sub < 30:
                break
            a_init = make_init(n_best, rng, style=style)
            print(f"  style={style}, budget={t_sub:.0f}s")
            # Quick LP then gradient
            a_sub, _, _ = lp_game_iteration(a_init, t_sub * 0.6, solver, n_max_iters=20000)
            try_update(a_sub, f"ms_s{style}_lp")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = grad_peak_shave(a_sub, min(rem * 0.3, 30.0), n_steps=10000)
                try_update(a_sub2, f"ms_s{style}_ps")

    # === PHASE 6: Try different lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = sorted(set([
        n_best - 40, n_best - 20, n_best + 20, n_best + 40,
        n_best + 60, 1024, 1000, 920
    ]))
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 2048 and x != n_best]

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.25, 100.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _, avg_alt = lp_game_iteration(init_alt, t_total * 0.7, solver, n_max_iters=15000)
        try_update(a_alt, f"alt_n{alt_n}_lp")
        try_update(avg_alt, f"alt_n{alt_n}_avg")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = grad_peak_shave(a_alt, min(rem * 0.3, 20.0), n_steps=5000)
            try_update(a_alt2, f"alt_n{alt_n}_ps")

    # === PHASE 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 120.0)
        a_fin, _, avg_fin = lp_game_iteration(best_seq, t_fin, solver, n_max_iters=200000)
        try_update(a_fin, "final_lp")
        try_update(avg_fin, "final_lp_avg")

    # === PHASE 8: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.30, 60.0)
        a_cd, _ = coordinate_descent_polish(best_seq, t_cd, n_rounds=10000)
        try_update(a_cd, "cd_polish")

    # === Phase 9: One more LP pass if time allows ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_extra = min(rem * 0.5, 80.0)
        a_ex, _, avg_ex = lp_game_iteration(best_seq, t_extra, solver, n_max_iters=100000)
        try_update(a_ex, "extra_lp")
        try_update(avg_ex, "extra_lp_avg")

    # Final coordinate polish
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fp, _ = coordinate_descent_polish(best_seq, min(rem * 0.5, 20.0), n_rounds=5000)
        try_update(a_fp, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
