"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-resolution coarse-to-fine optimization
2. Subgradient descent with heavy-ball momentum and adaptive step
3. Log-sum-exp annealing with adaptive temperature scheduling
4. Spectral-domain gradient descent (flatten power spectrum)
5. Structured local search with peak-targeted mass redistribution

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def normalize_a(a):
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_conv_max(a, n_top=None, frac=None):
    """Compute subgradient of max_k (a*a)(k) w.r.t. a.
    Uses weighted average over top peaks."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b, b

    if n_top is not None:
        idx = np.argsort(b)[::-1][:n_top]
        weights = np.exp(-0.3 * np.arange(len(idx)))
    elif frac is not None:
        threshold = max_b * (1.0 - frac)
        idx = np.where(b >= threshold)[0]
        if len(idx) == 0:
            idx = np.array([int(np.argmax(b))])
        weights = np.ones(len(idx))
    else:
        idx = np.array([int(np.argmax(b))])
        weights = np.ones(1)

    weights /= weights.sum()

    g = np.zeros(n)
    for i, k in enumerate(idx):
        k = int(k)
        w = weights[i]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx_range = np.arange(i_min, i_max + 1)
        g[idx_range] += w * (a[k - idx_range] + a[idx_range])

    return g, max_b, b


def logsumexp_conv(a, T):
    """Smooth approximation of max_b using log-sum-exp."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (max_Tb + np.log(exp_Tb.sum())) / T


def logsumexp_grad(a, T, n_active=200):
    """Gradient of the log-sum-exp smoothed objective."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # Only process significant weights for efficiency
    thresh = 1e-10
    active = np.where(w > thresh)[0]

    g = np.zeros(n)
    for k in active:
        k = int(k)
        wk = w[k]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx_range = np.arange(i_min, i_max + 1)
        g[idx_range] += wk * (a[k - idx_range] + a[idx_range])

    return g, b


def mirror_descent_momentum(a0, time_limit, n_active_frac=0.08, momentum=0.9):
    """Subgradient descent with heavy-ball momentum and adaptive step."""
    n = len(a0)
    a = normalize_a(a0)
    z = np.log(np.maximum(a, 1e-300))
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    patience = 50

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b, b = subgradient_conv_max(a, frac=n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        # Adaptive step size
        gap = best_score - f_lb
        polyak_lr = gap / (g_norm_inf * best_score + 1e-15)
        alpha = min(lr, max(polyak_lr, 0.01))
        alpha = min(alpha, 1.0)

        # Heavy ball momentum
        v = momentum * v + (1 - momentum) * g
        z_new = z - alpha * v

        # Softmax projection (stay in interior of simplex)
        z_shifted = z_new - z_new.max()
        exp_z = np.exp(z_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = exp_z / s

        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= patience:
                lr *= 0.6
                v *= 0.5
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 100000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      MD+Mom {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def lses_anneal(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=10):
    """Log-sum-exp annealing with L-BFGS at each temperature stage."""
    n = len(a0)
    z = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 200

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            a_local = softmax_project(z_in)
            b_local = fast_convolve(a_local)
            Tb = stage_T * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            return (2.0 * n / stage_T) * (max_Tb + np.log(exp_Tb.sum()))

        def jac(z_in):
            a_local = softmax_project(z_in)
            g_lse, _ = logsumexp_grad(a_local, stage_T)
            # Chain rule: d/dz = (I - a*a^T) @ (d/da)
            # dL/da = g_lse (already in a-space)
            # dL/dz_i = g_lse_i * a_i - a_i * sum_j g_lse_j * a_j
            S = np.dot(g_lse, a_local)
            return a_local * (g_lse - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_project(z)
    return a_out


def softmax_project(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

            break

        s = a_curr.sum()
        if s < 1e-15:

        a_curr = a_curr / s

        circ_acorr = np.real(irfft(power, N))

        # We want to minimize max of (a*a), approximated by peak of circ_acorr

        mean_power = power.mean()

            # Recompute gradient in a-space and map back

        # Keep log_amp bounded

        a_curr = np.maximum(a_curr, 0)

        if sc < 1e-15:

        k_star = int(np.argmax(b))
        # The peak is sum_i a[i]*a[k_star-i]
        # Find positions where the product is largest
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1)
        products = a[idx] * a[k_star - idx]
        
        # Find the position with the largest product
        max_prod_idx = idx[np.argmax(products)]
        
        # Move mass from max_prod_idx to a position that doesn't contribute to k_star
        # Find a position j where a[j]*a[k_star-j] is small (or j is outside [i_min, i_max])
        contrib_all = np.zeros(n)
        for p in peaks:
            p = int(p)
            pm = max(0, p - n + 1)
            px = min(n - 1, p)
            ii = np.arange(pm, px + 1)
            contrib_all[ii] += a[ii] * a[p - ii]

        # Find position with lowest contribution but non-zero a
        low_contrib = np.argsort(contrib_all)
        
        improved = False
        for frac in [0.03, 0.05, 0.08, 0.12]:
            src = max_prod_idx
            for dst in low_contrib[:5]:
                if src == dst:
                    continue
                if a[src] * frac > a[src]:  # can't move more than we have
                    continue
                amount = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amount
                a_trial[dst] += amount
                a_trial = np.maximum(a_trial, 0)
                sc = a_trial.sum()
                if sc < 1e-15:
                    continue
                a_trial = a_trial / sc
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2 and a[i1] > 0:
                amount = a[i1] * 0.02
                a_trial = a.copy()
                a_trial[i1] -= amount
                a_trial[i2] += amount
                a_trial = np.maximum(a_trial, 0)
                sc = a_trial.sum()
                if sc > 1e-15:
                    a_trial = a_trial / sc
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakEq {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=500):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    order = np.arange(n)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Shuffle order
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Optimize a[i] by trying a few values
            # Current a[i] = v. Try v * (1+delta) for various delta
            v = a[i]
            if v < 1e-14:
                # Try small positive values
                for delta in [0.001, 0.005, 0.01, 0.05]:
                    a_trial = a.copy()
                    a_trial[i] = delta
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break
                continue

            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a_trial = a.copy()
                a_trial[i] = v * mult
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break

        if rnd % 50 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDesc {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def downsample(a, factor):
    """Downsample by averaging groups of 'factor' consecutive elements."""
    n = len(a)
    new_n = n // factor
    if new_n < 4:
        return a
    a_trimmed = a[:new_n * factor]
    a_new = a_trimmed.reshape(new_n, factor).mean(axis=1)
    return a_new


def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new


def multi_resolve_optimize(a0, time_limit):
    """Coarse-to-fine multi-resolution optimization."""
    n_orig = len(a0)
    t0 = time.monotonic()
    best_a = normalize_a(a0)
    best_score = compute_score(best_a)

    # Create a progression of resolutions
    resolutions = [max(8, n_orig // 64), max(16, n_orig // 32),
                   max(32, n_orig // 16), max(64, n_orig // 8),
                   max(128, n_orig // 4), n_orig]
    # Remove duplicates and sort
    resolutions = sorted(set(resolutions))
    resolutions = [r for r in resolutions if r <= n_orig]
    if resolutions[-1] != n_orig:
        resolutions.append(n_orig)

    current_n = min(resolutions[0], n_orig)
    current_a = downsample(a0, n_orig // current_n) if n_orig > current_n else normalize_a(a0)
    current_a = normalize_a(current_a)

    for target_n in resolutions:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Upsample current solution to target_n
        if len(current_a) < target_n:
            current_a = upsample(current_a, target_n)
        elif len(current_a) > target_n:
            current_a = downsample(current_a, len(current_a) // max(1, target_n // (len(current_a) // target_n)))
            current_a = upsample(current_a, target_n)
        current_a = normalize_a(current_a)

        # Optimize at this resolution
        alloc = remaining * 0.3
        if alloc > 5:
            a_md, _ = mirror_descent_momentum(current_a, alloc, n_active_frac=0.1, momentum=0.85)
            sc = compute_score(a_md)
            if sc < compute_score(current_a):
                current_a = a_md

        # Evaluate and update global best
        sc = compute_score(current_a)
        if target_n == n_orig:
            if sc < best_score:
                best_score = sc
                best_a = current_a.copy()
            print(f"    MultiRes n={target_n}: score={sc:.6f}")

    return best_a, best_score


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 1.5)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1 - x))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2 * np.pi * 3 * x)
    elif choice == 6:
        # Two-lobed
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    else:
        # Raised cosine with modulation
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 1.2
        a *= (1.0 + 0.2 * np.cos(2 * np.pi * 5 * x))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Tries multiple strategies and returns the best sequence found."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Generated init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # ===== Phase 1: Multi-resolution coarse-to-fine =====
    print(f"\n=== Phase 1: Multi-Resolution (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.15, 240.0)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_mr, _ = multi_resolve_optimize(best_seq, t1)
        try_update(a_mr, "multi_resolve")

    # ===== Phase 2: Mirror descent with momentum =====
    print(f"\n=== Phase 2: Mirror Descent + Momentum (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_md, _ = mirror_descent_momentum(best_seq, t2, n_active_frac=0.05, momentum=0.92)
        try_update(a_md, "md_mom_1")

    # ===== Phase 3: Log-sum-exp annealing (L-BFGS) =====
    print(f"\n=== Phase 3: LSE Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_lse = lses_anneal(best_seq, t3, T_start=80.0, T_end=0.05, n_stages=12)
        try_update(a_lse, "lse_anneal")

    # ===== Phase 4: Spectral shaping =====
    print(f"\n=== Phase 4: Spectral Shaping (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 15:
        print(f"  Budget: {t4:.0f}s")
        a_spec, _ = spectral_shape_optimize(best_seq, t4, n_iters=3000)
        try_update(a_spec, "spectral")

    # ===== Phase 5: Peak equalization =====
    print(f"\n=== Phase 5: Peak Equalization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 180.0)
    if t5 > 15:
        print(f"  Budget: {t5:.0f}s")
        a_pe, _ = peak_equalization(best_seq, t5)
        try_update(a_pe, "peak_eq")

    # ===== Phase 6: Second mirror descent (tighter) =====
    print(f"\n=== Phase 6: MD Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 120.0)
    if t6 > 10:
        a_md2, _ = mirror_descent_momentum(best_seq, t6, n_active_frac=0.03, momentum=0.95)
        try_update(a_md2, "md_mom_2")

    # ===== Phase 7: Coordinate descent =====
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.08, 100.0)
    if t7 > 10:
        a_cd, _ = coordinate_descent(best_seq, t7, n_rounds=200)
        try_update(a_cd, "coord_desc")

    # ===== Phase 8: Try alternative lengths =====
    print(f"\n=== Phase 8: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [0.75, 1.25, 1.5, 2.0, 0.5]:
        alt_n = int(n_best * mult)
        if 200 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alt = min(rem * 0.20, 150.0)
        if t_alt < 15:
            break
        print(f"  n={alt_n}, budget={t_alt:.0f}s")

        init_alt = upsample(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Upsample improved: {sc_alt:.6f}")

        # Quick optimization at this length
        t_md_alt = t_alt * 0.6
        a_alt, _ = mirror_descent_momentum(cur, t_md_alt, n_active_frac=0.06, momentum=0.9)
        try_update(a_alt, f"alt_n{alt_n}")

    # ===== Phase 9: Final refinement =====
    print(f"\n=== Phase 9: Final Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = lses_anneal(best_seq, t_fin, T_start=5.0, T_end=0.02, n_stages=5)
        try_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_pe2 = min(rem * 0.25, 50.0)
        a_pe2, _ = peak_equalization(best_seq, t_pe2)
        try_update(a_pe2, "final_peak_eq")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_md3 = min(rem * 0.3, 40.0)
        a_md3, _ = mirror_descent_momentum(best_seq, t_md3, n_active_frac=0.02, momentum=0.97)
        try_update(a_md3, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
