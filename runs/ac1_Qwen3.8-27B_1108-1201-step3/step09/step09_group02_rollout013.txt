    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    u = np.sort(v)[::-1]

    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    soft = exp_shifted / exp_shifted.sum()

    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    fval = 2.0 * n * (max_c + tau * log_sum)
    soft = exp_shifted / exp_shifted.sum()

    # Cross-correlation via FFT: corr[i] = sum_k soft[k] * a[k-i]
    L = 3 * n + 1
    fft_size = 1 << (L - 1).bit_length()
    F_soft = np.fft.rfft(soft, fft_size)
    F_a = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(F_soft) * F_a, fft_size)[:n]

    grad = 4.0 * n * corr
    return fval, grad

def fista_continuation(a0, time_limit, tau_start=0.05, tau_end=0.0005, verbose=False, label=""):
    """FISTA on smoothed objective with geometric continuation on tau."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()

    tau = tau_start
    # Number of continuation stages
    n_stages = max(1, int(np.log(tau_start / tau_end) / np.log(4.0)))
    stage_decay = tau_end / tau_start
    decay_per_stage = stage_decay ** (1.0 / n_stages)

    for stage in range(n_stages):
        if time.monotonic() - t0 >= time_limit:
            break
        stage_time = (time_limit - (time.monotonic() - t0)) / (n_stages - stage) if n_stages > stage else 0
        stage_time = max(stage_time, 5.0)

        z = a.copy()
        k = 0
        lr = 1.0 / (2.0 * n)
        lr_min = 1.0 / (50.0 * n * n)
        f_prev = smooth_objective_grad(z, tau)[0]

        while time.monotonic() - t0 < t0 + stage_time + 60:
            if time.monotonic() - t0 >= time_limit:
                break

            k += 1
            fval, grad = smooth_objective_grad(z, tau)

            # FISTA momentum
            momentum = (k - 1) / (k + 2.0) if k > 1 else 0.0
            step_point = z - lr * grad

            # Backtracking line search (simplified)
            a_new = simplex_project(step_point)
            f_new = float(2.0 * n * fast_convolve(a_new).max())

            if f_new > fval + 1e-10:
                lr = max(lr * 0.5, lr_min)
                if lr <= lr_min:
                    a_new = simplex_project(z - lr * grad)
                else:
                    continue
            else:
                lr = min(lr * 1.1, 1.0 / (0.5 * n))

            conv_new = fast_convolve(a_new)
            score = 2.0 * n * float(conv_new.max())
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            z = a_new + momentum * (a_new - z)
            z = np.maximum(z, 0)
            zs = z.sum()
            if zs > 1e-15:
                z = z / zs

            if verbose and k % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      FISTA {label} stage={stage} tau={tau:.6f} k={k} best={best_score:.8f} ({elapsed:.0f}s)")

        tau *= decay_per_stage
        a = best_a.copy()

    return best_a, best_score

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Precompute for fast evaluation
            c_vals_copy = c_vals.copy()

            def compute_score(x, _c=c_vals_copy, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    factors = [0.999, 0.995, 0.99, 0.98, 0.96, 1.04, 1.02, 1.01, 1.005, 1.001]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_local = float(conv[k_lo:k_hi + 1].max())

            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                delta = x - v
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                if local_max < best_local - 1e-16:
                    best_local = local_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_whitened_nonneg(n, rng):
    """White noise clamped to non-negative, then spectrally whitened."""
    a = rng.random(n)
    a = a - a.mean() + 0.3
    a = np.abs(a)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    # Spectral whitening: divide FFT by its magnitude
    n_fft = 1 << (n - 1).bit_length()
    A = rfft(a, n_fft)
    mag = np.abs(A)
    mag[mag < 1e-12] = 1e-12
    A_white = A / mag
    a_white = irfft(A_white, n_fft)[:n]
    a_white = np.abs(a_white)
    s2 = a_white.sum()
    if s2 < 1e-10:
        return np.ones(n) / n
    return a_white / s2

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] via LP."""
    import cvxpy as cp
    if solver is None:
        solver = _get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build conv matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_time=30)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_time=30)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

_lp_solver_cached = None

def _get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, max_time=5)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with line search."""
    if solver is None:
        solver = _get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        best_t = 0.0
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                best_t = trial_t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient descent with active peak weighting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Top peaks
        k_act = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.003 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        a_new = simplex_project(a - lr * grad)
        conv_new = fast_convolve(a_new)
        sc = 2.0 * n * float(conv_new.max())
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
        lr *= 0.9999

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

            if best_score > 1.5035 and rnd == 2:
                # Perturb to escape if stuck
                perturbed = a_opt + rng.standard_normal(n_best) * 0.04
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: FISTA continuation (finer tau) ===
    print(f"\n=== Phase 3: Fine FISTA (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  Fine FISTA n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = fista_continuation(best_seq, t_alloc, tau_start=0.01, tau_end=0.0002,
                                              verbose=True, label="fine_fista")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "fine_fista")
            # Follow up with ECD
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                t_cd = min(rem * 0.4, 60.0)
                a_cd, s_cd, _ = exact_coord_descent(a_opt, t_cd, label="post_fista_ecd")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "post_fista_ecd")

    # === Phase 4: LP Fictitious Play ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  FP n={n_best}, budget={t_alloc:.0f}s")
            try:
                a_fp, s_fp = fictitious_play(best_seq, t_alloc, verbose=True)
                best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "fp")
                # Polish after FP
                rem = deadline - time.monotonic() - 150
                if rem > 20:
                    t_cd = min(rem * 0.4, 60.0)
                    a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label="post_fp")
                    best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "post_fp")
            except Exception as e:
                print(f"    LP error: {e}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  Subgradient, budget={t_alloc:.0f}s")
            a_sg, s_sg = subgradient_polish(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "subgrad")

    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 40:
            n_best = len(best_seq)
            n_perturbs = min(6, int(rem / 30))
            for p in range(n_perturbs):
                rem2 = deadline - time.monotonic() - 60
                if rem2 < 20:
                    break
                t_alloc = min(rem2 * 0.4, 50.0)

                # Structured perturbation targeting peaks
                conv_p = fast_convolve(best_seq)
                top_k = min(8, 2 * n_best - 1)
                peaks = np.argpartition(conv_p, -top_k)[-top_k:]
                perturbed = best_seq.copy()
                for k in peaks:
                    i_start = max(0, int(k) - n_best + 1)
                    i_end = min(n_best - 1, int(k))
                    idx = np.arange(i_start, i_end + 1)
                    n_pick = max(1, len(idx) // 4)
                    picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                    for ii in picks:
                        perturbed[ii] *= (1.0 + rng.standard_normal() * 0.06)
                # Add small global noise
                perturbed += rng.standard_normal(n_best) * 0.01
                perturbed = np.maximum(perturbed, 0)
                ps = perturbed.sum()
                if ps < 1e-10:
                    continue
                perturbed = perturbed / ps

                a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh_p{p}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"bh_p{p}")

    # === Phase 7: Try different n (upsample/downsample) ===
    print(f"\n=== Phase 7: Multi-n attempts (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            n_best = len(best_seq)
            alt_ns = [x for x in [n_best // 2, n_best * 2, n_best + 256, n_best - 256]
                      if 256 <= x <= 4096 and x != n_best]
            for alt_n in alt_ns[:2]:
                rem2 = deadline - time.monotonic() - 40
                if rem2 < 20:
                    break
                t_alloc = min(rem2 * 0.4, 40.0)
                cand = interpolate_seq(best_seq, alt_n)
                a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"multin_{alt_n}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"multin_{alt_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_fine = min(rem * 0.5, 60.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True, label="final")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_final")

        # Another ECD round
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, min(rem * 0.6, 40.0), label="last_ecd")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "last_ecd")

        # Very fine pass
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 20.0), label="very_last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "very_last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic():.1f}s")
    return [float(x) for x in best_seq]
