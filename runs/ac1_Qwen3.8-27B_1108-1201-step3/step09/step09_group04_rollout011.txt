from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score 2*n*max(c)/(sum(a))^2."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def p_norm_obj_and_grad(z, p):
    """Compute p-norm smoothed objective and gradient in log-domain (softmax).
    Objective: 2n * (sum max(c,0)^p)^(1/p) where a = softmax(z), sum(a)=1."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_powers = c_pos ** p
    S = np.sum(p_powers)
    if S < 1e-300:
        val = 0.0
        grad = np.zeros_like(z)
        return val, grad
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    grad = a * (g_a - np.dot(g_a, a))
    val = 2.0 * n * S ** (1.0 / p)
    return val, grad


def pnorm_lbfgs(a0, time_limit, p_start=30.0, p_end=2000.0, n_stages=10, restart_every=0):
    """L-BFGS with adaptive p-norm schedule and optional restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a) - np.log(a).max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, int(6000 // n_stages))

    for stage_idx, stage_p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            v, _ = p_norm_obj_and_grad(z_val, stage_p)
            return v

        def jac(z_val):
            _, g = p_norm_obj_and_grad(z_val, stage_p)
            return g

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = compute_score(a_cur)
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()

        elapsed = time.monotonic() - t0
        if stage_idx % 3 == 0:
            print(f"      pnorm p={stage_p:.0f} ({elapsed:.0f}s): score={score:.6f}, best={best_score:.6f}")

        # Optional restart with perturbation
        if restart_every > 0 and stage_idx > 0 and stage_idx % restart_every == 0:
            if time.monotonic() - t0 > time_limit:
                break
            rng = np.random.default_rng(stage_idx * 100 + 7)
            z_pert = z + 0.1 * rng.standard_normal(n)
            z_pert = z_pert - z_pert.max()
            z = z_pert

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a_out = exp_z / exp_z.sum()
    final_score = compute_score(a_out)
    if final_score < best_score:
        best_a = a_out.copy()
        best_score = final_score
    return best_a


def top_k_obj_and_grad(a, K):
    """Objective: sum of top K values of convolve(a,a). Gradient w.r.t. a (with sum=1 constraint handled externally)."""
    n = len(a)
    c = fast_convolve(a)
    # Find top K indices
    top_k_idx = np.argsort(c)[::-1][:K]
    top_k_vals = c[top_k_idx]
    obj = np.sum(top_k_vals)
    # Gradient: sum over top-k peaks of dc[k]/da[i] = 2 * correlate(contribution, a)[i]
    w = np.zeros(len(c))
    w[top_k_idx] = 1.0
    corr = np.correlate(w, a, mode='valid')
    grad_a = 2.0 * corr
    return obj, grad_a


def top_k_local_opt(a0, time_limit, K=5, n_iters=50000):
    """Local optimization targeting top-K peaks using projected subgradient with averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    avg_a = a.copy()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        obj, grad = top_k_obj_and_grad(y, K)
        # Projected gradient step: minimize |a - (y - alpha*grad)|^2 subject to a>=0, sum(a)=1
        alpha = 1.0 / (np.sqrt(it + 1) * 10.0)
        step = y - alpha * grad
        step = np.maximum(step, 0)
        ss = step.sum()
        if ss < 1e-15:
            step = y.copy()
            ss = 1.0
        step = step / ss

        # Nesterov update
        beta = 2.0 / (it + 2)
        avg_a = (1 - beta) * avg_a + beta * step
        avg_a = np.maximum(avg_a, 0)
        sa = avg_a.sum()
        if sa < 1e-15:
            avg_a = step.copy()
        else:
            avg_a = avg_a / sa
        y = step

        # Evaluate
        score_y = compute_score(y)
        score_avg = compute_score(avg_a)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      TopK local iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

        # Early stopping if no improvement for a while
        if (it + 1) % 10000 == 0:
            current = min(compute_score(y), compute_score(avg_a))
            if current > best_score * (1 + 0.001):
                # Reinitialize y
                y = best_a.copy()
                avg_a = best_a.copy()

    return best_a


def build_conv_matrix(a):
    """Build convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov LP alternating method."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-7:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 5:
                rng = np.random.default_rng(it + 42)
                y = y + 0.03 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0
                prev_best = best_score

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def peak_shaving_refine(a0, time_limit, n_steps=50000):
    """Aggressive peak-shaving: directly reduce the maximum convolution peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Find the global max peak
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Strategy: reduce the top contributor and redistribute
        top_i = idx[np.argmax(contribs)]
        if a[top_i] < 1e-14:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        improved = False
        # Try reducing top_i by various amounts and redistributing to random/low-contribution positions
        for frac in [0.1, 0.2, 0.3, 0.5]:
            reduce_amt = a[top_i] * frac
            if reduce_amt < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[top_i] -= reduce_amt
            # Redistribute to positions with low contribution to the peak
            # Pick positions that don't contribute much to this peak
            non_peak = np.ones(n, dtype=bool)
            non_peak[idx] = False
            if non_peak.sum() > 0:
                # Distribute proportionally to current values (preserves shape)
                redist_mask = non_peak & (a_trial > 1e-14)
                if redist_mask.sum() > 0:
                    weights = a_trial[redist_mask]
                    weights = weights / weights.sum()
                    a_trial[redist_mask] += reduce_amt * weights
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            score = compute_score(a_trial)
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            # Try a different strategy: reduce two contributors
            if no_improve % 50 == 0 and len(idx) >= 2:
                top2 = np.argsort(contribs)[::-1][:2]
                for t1 in top2:
                    for t2 in top2:
                        if t1 == t2:
                            continue
                        i1, i2 = idx[t1], idx[t2]
                        if a[i1] < 1e-14 or a[i2] < 1e-14:
                            continue
                        for f in [0.15, 0.25]:
                            a_trial = a.copy()
                            a_trial[i1] *= (1 - f)
                            a_trial[i2] *= (1 - f)
                            redist_amt = a[i1] * f + a[i2] * f
                            non_peak2 = np.ones(n, dtype=bool)
                            non_peak2[idx] = False
                            redist_mask = non_peak2 & (a_trial > 1e-14)
                            if redist_mask.sum() > 0:
                                weights = a_trial[redist_mask]
                                weights = weights / weights.sum()
                                a_trial[redist_mask] += redist_amt * weights
                            s_t = a_trial.sum()
                            if s_t >= 0.01:
                                a_trial = a_trial / s_t
                                score = compute_score(a_trial)
                                if score < best_score - 1e-12:
                                    a = a_trial
                                    best_score = score
                                    best_a = a.copy()
                                    no_improve = 0
                                    improved = True
                                    break
                        if improved:
                            break
                    if improved:
                        break

        if no_improve > 500:
            # Try random perturbation to escape local minimum
            rng = np.random.default_rng(step + 12345)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.15, 0.15))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = compute_score(a_trial)
                if score < best_score:
                    a = a_trial
                    no_improve = 0
                else:
                    # Don't accept worse, just reset no_improve counter occasionally
                    if no_improve > 2000:
                        break
            else:
                break

    return best_a


def diverse_init(n, variant, rng):
    """Create diverse initializations."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        xc = np.linspace(-1, 1, n)
        a = np.exp(-xc ** 2 * 3)
    elif variant == 5:
        # Bimodal
        xc = np.linspace(-1, 1, n)
        a = np.exp(-4 * (xc - 0.5) ** 2) + np.exp(-4 * (xc + 0.5) ** 2)
    elif variant == 6:
        # Random
        a = rng.random(n) ** 0.5
    elif variant == 7:
        # Exponential decay
        a = np.exp(-2 * x)
        a = a + np.exp(-2 * (1 - x))
    elif variant == 8:
        # Comb pattern
        a = np.ones(n) * 0.5
        a[::3] = 1.0
        a = a + 0.1 * rng.random(n)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses multi-phase optimization:
    1. Diverse initialization and selection
    2. Aggressive p-norm L-BFGS
    3. Top-K peak targeting
    4. Nesterov LP
    5. Peak-shaving local refinement
    6. Length exploration
    7. Final polish
    """
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    best_score = float('inf')
    best_a = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={ps:.6f}")
            best_a = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = diverse_init(n_init, 4, rng)
        best_a = parent.copy()
        best_score = evaluate_safe(best_a)
        print(f"Default init: n={len(parent)}, score={best_score:.6f}")

    n_best = len(best_a)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # === Phase 1: Quick diverse init evaluation ===
    print(f"\n=== Phase 1: Diverse Init Selection (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.08, 80.0)
    if t_phase1 > 10:
        print(f"  Budget: {t_phase1:.0f}s")
        t_per_init = t_phase1 / 8.0
        for variant in range(8):
            if time.monotonic() > deadline - 60:
                break
            a_init = diverse_init(n_best, variant, rng)
            # Quick p-norm optimization
            t_sub = t_per_init * 0.7
            a_opt = pnorm_lbfgs(a_init, t_sub, p_start=20, p_end=300, n_stages=5, restart_every=0)
            sc = compute_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  Variant {variant}: score={sc:.6f} (NEW BEST)")
            else:
                print(f"  Variant {variant}: score={sc:.6f}")

    # === Phase 2: Aggressive p-norm L-BFGS (main workhorse) ===
    print(f"\n=== Phase 2: Aggressive p-norm L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.25, 400.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s, p: 20 -> 5000")
        a_pn = pnorm_lbfgs(best_a, t_phase2, p_start=20.0, p_end=5000.0, n_stages=12, restart_every=4)
        sc = compute_score(a_pn)
        if sc < best_score:
            best_score = sc
            best_a = a_pn.copy()
            print(f"  p-norm improved: {sc:.6f}")
        else:
            print(f"  p-norm: {sc:.6f} (no improvement)")

    # === Phase 3: Top-K local optimization ===
    print(f"\n=== Phase 3: Top-K Local Opt (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.15, 200.0)
    if t_phase3 > 20:
        print(f"  Budget: {t_phase3:.0f}s, K=5")
        a_topk = top_k_local_opt(best_a, t_phase3, K=5, n_iters=80000)
        sc = compute_score(a_topk)
        if sc < best_score:
            best_score = sc
            best_a = a_topk.copy()
            print(f"  Top-K improved: {sc:.6f}")
        else:
            print(f"  Top-K: {sc:.6f} (no improvement)")

    # === Phase 4: Nesterov LP ===
    print(f"\n=== Phase 4: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.20, 300.0)
    if t_phase4 > 30:
        print(f"  Budget: {t_phase4:.0f}s")
        a_lp = nesterov_lp(best_a, t_phase4, solver=solver, n_max_iters=60000)
        sc = compute_score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
            print(f"  LP improved: {sc:.6f}")
        else:
            print(f"  LP: {sc:.6f} (no improvement)")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.12, 150.0)
    if t_phase5 > 15:
        print(f"  Budget: {t_phase5:.0f}s")
        a_ps = peak_shaving_refine(best_a, t_phase5, n_steps=100000)
        sc = compute_score(a_ps)
        if sc < best_score:
            best_score = sc
            best_a = a_ps.copy()
            print(f"  Peak shaving improved: {sc:.6f}")
        else:
            print(f"  Peak shaving: {sc:.6f} (no improvement)")

    # === Phase 6: Second round of p-norm from improved point ===
    print(f"\n=== Phase 6: p-norm round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase6 = min(rem * 0.12, 150.0)
    if t_phase6 > 20:
        print(f"  Budget: {t_phase6:.0f}s, p: 100 -> 10000")
        a_pn2 = pnorm_lbfgs(best_a, t_phase6, p_start=100.0, p_end=10000.0, n_stages=8, restart_every=3)
        sc = compute_score(a_pn2)
        if sc < best_score:
            best_score = sc
            best_a = a_pn2.copy()
            print(f"  p-norm r2 improved: {sc:.6f}")
        else:
            print(f"  p-norm r2: {sc:.6f} (no improvement)")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_base = len(best_a)
        candidates = []
        for delta in [-128, -96, -64, -48, -32, 32, 48, 64, 96, 128, 192]:
            alt = n_base + delta
            if 384 <= alt <= 3072:
                candidates.append(alt)
        # Add some round numbers
        for rn in [512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536, 1920, 2048]:
            if 384 <= rn <= 3072 and abs(rn - n_base) > 48:
                candidates.append(rn)
        # Deduplicate and sort by proximity to current best
        candidates = sorted(set(candidates), key=lambda x: abs(x - n_base))

        n_explored = 0
        for alt_n in candidates[:6]:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 4:
                break
            n_explored += 1
            t_total = min(rem * 0.15, 120.0)
            if t_total < 30:
                break
            print(f"  Exploring n={alt_n} (budget={t_total:.0f}s)")

            a_alt = interpolate_seq(best_a, alt_n)
            # Quick p-norm
            t_pn = t_total * 0.55
            a_opt = pnorm_lbfgs(a_alt, t_pn, p_start=30, p_end=500, n_stages=6, restart_every=0)
            sc = compute_score(a_opt)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"    n={alt_n}: score={sc:.6f} (NEW BEST!)")
                # If improved, do LP refinement
                rem = deadline - time.monotonic()
                if rem > 30:
                    t_lp = min(rem * 0.1, 60.0)
                    a_lp2 = nesterov_lp(best_a, t_lp, solver=solver, n_max_iters=20000)
                    sc2 = compute_score(a_lp2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_a = a_lp2.copy()
                        print(f"    n={alt_n} LP: score={sc2:.6f} (NEW BEST!)")
            else:
                print(f"    n={alt_n}: score={sc:.6f} (no improvement)")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.3, 80.0)
        a_fin = pnorm_lbfgs(best_a, t_fin, p_start=500, p_end=20000, n_stages=5, restart_every=0)
        sc = compute_score(a_fin)
        if sc < best_score:
            best_score = sc
            best_a = a_fin.copy()
            print(f"  Final pnorm: {sc:.6f}")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps_final = min(rem * 0.3, 50.0)
        a_ps_final = peak_shaving_refine(best_a, t_ps_final, n_steps=50000)
        sc = compute_score(a_ps_final)
        if sc < best_score:
            best_score = sc
            best_a = a_ps_final.copy()
            print(f"  Final peak shave: {sc:.6f}")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_topk_final = min(rem * 0.4, 40.0)
        a_topk_final = top_k_local_opt(best_a, t_topk_final, K=3, n_iters=30000)
        sc = compute_score(a_topk_final)
        if sc < best_score:
            best_score = sc
            best_a = a_topk_final.copy()
            print(f"  Final top-K: {sc:.6f}")

    # Final verification with actual evaluator
    if best_a is not None:
        final_sc = evaluate_safe(best_a)
        if np.isfinite(final_sc) and final_sc < best_score:
            best_score = final_sc
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}")

    return [float(x) for x in best_a]
