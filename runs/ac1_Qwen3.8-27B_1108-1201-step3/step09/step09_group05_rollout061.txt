def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_b/(sum_a)^2 for a normalized sequence (sum=1)."""

def lbgfs_smooth(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10):
    """L-BFGS with log-sum-exp smoothing on the probability simplex."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    def softmax(z_in):
        z_shifted = z_in - z_in.max()
        exp_z = np.exp(z_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z_in)) / len(z_in)
        return exp_z / s
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        max_iter = max(20, int(remaining * 100))
        
        def func(z_in):
            a = softmax(z_in)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            return (2.0 * n / T) * (max_Tb + np.log(exp_Tb.sum()))
        
        def jac(z_in):
            a = softmax(z_in)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_frac = f_interp - np.floor(f_interp)
    f_interp = f_frac
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def mirror_descent_optimize(a0, time_limit, n_peaks=3, eta_init=0.5, eta_min=0.01):
    """Mirror descent (multiplicative weight) with multi-peak subgradients.
    Returns (best_a, best_score)."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
        
        # Find active peaks (top n_peaks or those within 1% of max)
        threshold = max_b * 0.99
        active_indices = np.where(b >= threshold)[0]
        # Limit to top n_peaks
        if len(active_indices) > n_peaks:
            top_vals = b[active_indices]
            top_order = np.argsort(top_vals)[::-1][:n_peaks]
            active_indices = active_indices[top_order]
        
        # Compute subgradient: g[i] = sum over active k of a[k-i] (valid range)
        g = np.zeros(n)
        for k in active_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            indices = np.arange(js, je + 1)
            g[indices] += a[k - indices]
        
        # Adaptive learning rate
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        eta = eta_init * (eta_min / eta_init) ** progress
        
        # Normalized multiplicative update
        g_max = g.max()
        if g_max < 1e-15:
            continue
        log_a = np.log(np.maximum(a, 1e-300)) - eta * (g / g_max)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        # Only accept if it doesn't worsen too much (stochastic acceptance)
        score_new = compute_score(a_new)
        if score_new <= score * 1.01 or eta < 0.05:
            a = a_new
    
    return best_a, best_score


def _softmax_local(z_in):
    """Local softmax for internal use."""
    z_shifted = z_in - z_in.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z_in)) / len(z_in)
    return exp_z / s


def _lbgfs_func(z_in, T, n, t0, time_limit):
    """L-BFGS objective: 2n/T * logsumexp(T*b)."""
    if time.monotonic() - t0 > time_limit:
        return 1e15
    a = _softmax_local(z_in)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * n / T) * (max_Tb + np.log(exp_Tb.sum()))


def _lbgfs_jac(z_in, T, n, t0, time_limit):
    """L-BFGS gradient."""
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z_in)
    a = _softmax_local(z_in)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        max_iter = max(30, int(remaining * 20))
        
        def func_wrapper(z_in):
            return _lbgfs_func(z_in, T, n, t0, time_limit)
        
        def jac_wrapper(z_in):
            return _lbgfs_jac(z_in, T, n, t0, time_limit)
        
        try:
            res = minimize(func_wrapper, z, jac=jac_wrapper, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                                   'maxfun': max_iter * 200, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    return _softmax_local(z)


def peak_shave_optimize(a0, time_limit):
    """Direct peak shaving with smart redistribution to safe positions."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0
    rng = np.random.default_rng(777)
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        score = 2.0 * n * max_b
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            # Try perturbation
            a_trial = a + 0.02 * rng.standard_normal(n)
            a_trial = norm_sequence(a_trial)
            s_new = compute_score(a_trial)
            if s_new < best_score:
                a = a_trial
                best_score = s_new
                best_a = a.copy()
                no_improve = 0
            elif no_improve > 1000:
                break
        
        # Find contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        indices = np.arange(js, je + 1)
        contribs = a[indices] * a[peak_k - indices]
        
        # Compute "danger" for each position (total contribution to near-peak values)
        peak_threshold = max_b * 0.9
        active_peaks = np.where(b >= peak_threshold)[0]
        
        danger = np.zeros(n)
        for k in active_peaks:
            dk_s = max(0, k - n + 1)
            dk_e = min(n - 1, k)
            if dk_s > dk_e:
                continue
            for i in range(dk_s, dk_e + 1):
                danger[i] += a[i] * a[k - i]
        
        # Try reducing top contributors
        top_m = min(5, len(contribs))
        top_occ = np.argsort(contribs)[::-1][:top_m]
        
        improved_this_step = False
        for occ in top_occ:
            if improved_this_step:
                break
            i = indices[occ]
            if a[i] < 1e-12:
                continue
            
            # Find safe positions (low danger)
            safe_thresh = np.percentile(danger, 25)
            safe_mask = (danger <= safe_thresh) & (a > 1e-8)
            safe_mask[i] = False
            safe_indices = np.where(safe_mask)[0]
            
            if len(safe_indices) == 0:
                safe_indices = np.where(danger == danger.min())[0]
                safe_indices = safe_indices[safe_indices != i]
            
            if len(safe_indices) == 0:
                continue
            
            for factor in [0.97, 0.95, 0.93, 0.90, 0.85]:
                reduction = a[i] * (1 - factor)
                if reduction < 1e-14:
                    continue
                
                a_trial = a.copy()
                a_trial[i] *= factor
                
                # Redistribute proportionally to safe positions
                safe_vals = a_trial[safe_indices]
                if safe_vals.sum() < 1e-15:
                    continue
                a_trial[safe_indices] += reduction * safe_vals / safe_vals.sum()
                a_trial = a_trial / a_trial.sum()
                
                s_new = compute_score(a_trial)
                if s_new < best_score - 1e-15:
                    a = a_trial
                    best_score = s_new
                    best_a = a.copy()
                    no_improve = 0
                    improved_this_step = True
                    break
    
    return best_a, best_score


def spectral_flatten(a0, time_limit, n_iters=5000):
    """Gerchberg-Saxton inspired spectral flattening."""
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()  # FFT size for linear convolution
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Design a target magnitude that's flatter
        # Use a mixture: current magnitude raised to a power < 1
        # to compress dynamic range
        alpha = 0.5  # flattening strength
        mag_min = mag[mag > 1e-10].min() if np.any(mag > 1e-10) else 1e-10
        mag_max = mag.max()
        
        # Target: geometric mean of current mag and flat
        target_flat = mag.sum() / N  # average magnitude
        target_mag = mag ** alpha * (target_flat / mag.mean()) ** (1 - alpha)
        
        # Reconstruct
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Accept if improved
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Also try a weaker version
            A_weak = mag ** 0.8 * np.exp(1j * phase)
            a_weak = np.real(ifft(A_weak, N))[:n]
            a_weak = np.maximum(a_weak, 0)
            a_weak = a_weak / a_weak.sum()
            s_weak = compute_score(a_weak)
            if s_weak < best_score:
                best_score = s_weak
                best_a = a_weak.copy()
                a = a_weak
    
    return best_a, best_score


def progressive_peak_redu

ce(a0, time_limit):
    """Progressive peak reduction: iteratively target each peak in order.
    Finds all near-max peaks and tries to reduce them one by one."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all peaks above 95% of max
        threshold = max_b * 0.95
        peak_positions = np.where(b >= threshold)[0]
        
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find top contributor to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            indices = np.arange(js, je + 1)
            contribs = a[indices] * a[pk - indices]
            top_i = indices[np.argmax(contribs)]
            
            if a[top_i] < 1e-12:
                continue
            
            # Compute which positions are "safe" for this specific peak
            # A position j is safe if a[j] * a[pk-j] is small
            safe = np.zeros(n, dtype=bool)
            for j in range(n):
                partner = pk - j
                if 0 <= partner < n:
                    contribution = a[j] * a[partner]
                else:
                    contribution = 0.0
                safe[j] = (contribution < 0.05 * max_b) and (j != top_i)
            
            safe_indices = np.where(safe & (a > 1e-8))[0]
            if len(safe_indices) == 0:
                continue
            
            for factor in [0.98, 0.96, 0.94, 0.92]:
                reduction = a[top_i] * (1 - factor)
                a_trial = a.copy()
                a_trial[top_i] *= factor
                
                safe_vals = a_trial[safe_indices]
                if safe_vals.sum() < 1e-15:
                    continue
                a_trial[safe_indices] += reduction * safe_vals / safe_vals.sum()
                a_trial = a_trial / a_trial.sum()
                
                s_new = compute_score(a_trial)
                if s_new < best_score - 1e-15:
                    a = a_trial
                    best_score = s_new
                    best_a = a.copy()
                    break
    
    return best_a, best_score


def perturb_and_reoptimize(a0, time_limit, rng):
    """Apply random perturbation then re-optimize with mirror descent + L-BFGS."""
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    
    # Try multiple perturbation scales
    for scale in [0.05, 0.1, 0.2, 0.35]:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        
        perturbed = a0 + scale * rng.standard_normal(n)
        perturbed = norm_sequence(perturbed)
        
        # Quick mirror descent
        t_md = min(remaining * 0.4, 60.0)
        a_md, s_md = mirror_descent_optimize(perturbed, t_md, n_peaks=3, eta_init=0.3, eta_min=0.01)
        if s_md < best_score:
            best_score = s_md
            best_a = a_md.copy()
        
        # L-BFGS polish
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 15:
            t_lb = min(remaining * 0.4, 50.0)
            a_lb = lbgfs_refine(best_a if s_md < compute_score(a0) else perturbed, t_lb, T_start=30.0, T_end=0.1, n_stages=8)
            s_lb = compute_score(a_lb)
            if s_lb < best_score:
                best_score = s_lb
                best_a = a_lb.copy()
    
    return best_a, best_score


def multi_coord_descent(a0, time_limit):
    """Multi-coordinate descent: adjust groups of coordinates simultaneously."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0
    rng = np.random.default_rng(555)
    
    group_size = max(3, n // 64)
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        score = 2.0 * n * b.max()
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Pick a random group of coordinates to adjust
        start = rng.integers(0, n - group_size)
        group = np.arange(start, start + group_size)
        
        # Determine which are high-danger vs low-danger
        danger = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for i in range(js, je + 1):
            danger[i] += a[i] * a[peak_k - i]
        
        group_danger = danger[group]
        high_mask = group_danger > np.mean(group_danger)
        low_mask = ~high_mask
        
        if not np.any(high_mask) or not np.any(low_mask):
            continue
        
        hi_idx = group[high_mask]
        lo_idx = group[low_mask]
        
        # Reduce high, increase low
        for frac in [0.02, 0.04, 0.08, 0.15]:
            total_red = 0.0
            a_trial = a.copy()
            for i in hi_idx:
                red = a_trial[i] * frac
                total_red += red
                a_trial[i] -= red
            total_inc = total_red * (len(hi_idx) / max(1, len(lo_idx)))
            
            for j in lo_idx[:max(1, len(lo_idx) // 2)]:
                a_trial[j] += total_inc / max(1, len(lo_idx) // 2)
            
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            s_new = compute_score(a_trial)
            if s_new < best_score - 1e-15:
                a = a_trial
                best_score = s_new
                best_a = a.copy()
                no_improve = 0
                break
    
    return best_a, best_score

    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.25, 200.0)
        if t9 > 15:
            print(f"  Mirror descent budget: {t9:.0f}s")
            a_r2, _ = mirror_descent_optimize(best_seq, t9, n_peaks=5, eta_init=0.3, eta_min=0.005)
            try_update(a_r2, "mirror_2")
    
    # === Phase 10: Final L-BFGS polish ===
    print(f"\n=== Phase 10: Final L-BFGS Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t10 = min(rem * 0.30, 150.0)
        if t10 > 10:
            print(f"  Budget: {t10:.0f}s, T: 10->0.03, stages=10")
            a_fin = lbgfs_refine(best_seq, t10, T_start=10.0, T_end=0.03, n_stages=10)
            try_update(a_fin, "final_lbgfs")
    
    # === Phase 11: Final peak shaving ===
    print(f"\n=== Phase 11: Final Peak Shaving ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t11 = min(rem * 0.30, 100.0)
        if t11 > 10:
            print(f"  Budget: {t11:.0f}s")
            a_fin2, _ = peak_shave_optimize(best_seq, t11)
            try_update(a_fin2, "final_peak")
    
    # === Phase 12: Final spectral ===
    print(f"\n=== Phase 12: Final Spectral ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t12 = min(rem * 0.30, 60.0)
        if t12 > 5:
            a_fin3, _ = spectral_flatten(best_seq, t12)
            try_update(a_fin3, "final_spectral")
    
    # === Phase 13: Ultra-fine perturbation ===
    print(f"\n=== Phase 13: Ultra-Fine Perturbation ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        n_cur = len(best_seq)
        rng_fine = np.random.default_rng(seed + 9999)
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            scale = 0.005 + 0.01 * trial
            perturbed = best_seq + scale * rng_fine.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)
            t_sub = min(rem * 0.5, 30.0)
            if t_sub > 5:
                a_fin4, _ = mirror_descent_optimize(perturbed, t_sub, n_peaks=5, eta_init=0.1, eta_min=0.003)
                try_update(a_fin4, f"fine_pert_{trial}")
    
    # Final verification
    if best_seq is not None:
        fs = safe_evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    elapsed = time.monotonic() - (deadline - float(budget_s) + 20.0)
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
