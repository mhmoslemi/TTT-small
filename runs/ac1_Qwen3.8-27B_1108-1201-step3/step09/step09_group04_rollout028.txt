    log_sum = m + np.log(np.sum(np.exp(p_log_c - m)))

def p_norm_value_and_grad(a, p):
    """Compute the p-norm of the convolution and its gradient.
    Returns (value, gradient) where value = 2n * (sum c_k^p)^(1/p)."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    log_c = np.log(c_pos)
    p_log_c = p * log_c
    m = np.max(p_log_c)
    exp_part = np.exp(p_log_c - m)
    S = np.exp(m) * np.sum(exp_part)  # This is sum c_k^p (might overflow!)
    
    # Better: work in log space
    log_S = m + np.log(np.sum(exp_part))
    p_norm_val = np.exp(log_S / p)
    
    # weights: w_k = c_k^p / S = exp(p*log(c_k) - log_S)
    # = exp(p_log_c - log_S)
    w = exp_part / np.sum(exp_part)  # normalized weights
    
    # gradient: df/d(a_i) = 2*n*p_norm_val * sum_k w_k * a[k-i] / c_k
    # = 2*n*p_norm_val * correlate(w/c, a, 'valid')
    wc = w / c_pos  # w_k / c_k
    corr = np.correlate(wc, a, mode='valid')
    grad = 2.0 * n * p_norm_val * corr
    
    value = 2.0 * n * p_norm_val  # This is 2n * p-norm
    return value, grad

`F_p(a) = 2n * exp(log_S / p)` where `log_S = logsumexp(p * log(c_k))`

`dF_p/d(a_i) = 2n * exp(log_S/p) * (1/p) * d(log_S)/d(a_i)`

`d(log_S)/d(a_i) = d/d(a_i) [logsumexp(p*log(c_k))]`
`= sum_k [p * (c_k^p / S) * (1/c_k) * d(c_k)/d(a_i)]`
`= sum_k [p * w_k * d(c_k)/d(a_i) / c_k]`  where `w_k = c_k^p/S`

So `dF_p/d(a_i) = 2n * p_norm * (1/p) * p * sum_k w_k * d(c_k)/d(a_i) / c_k`
`= 2n * p_norm * sum_k w_k * d(c_k)/d(a_i) / c_k`
`= 2n * p_norm * 2 * sum_k w_k * a[k-i] / c_k` (for valid k-i)
`= 4n * p_norm * correlate(w/c, a, 'valid')[i]`

OK so the gradient is `4n * p_norm * correlate(w/c, a, 'valid')`.

Now for the softmax parameterization: if `a = softmax(z)`, then:
`dF/dz_i = sum_j (dF/d(a_j)) * (da_j/dz_i)`

= `a_i * (g_i - sum_j g_j * a_j)`
= `a_i * (g_i - dot(g, a))`

where `g = dF/da`.

This is the standard softmax gradient transformation.

Let me also think about initialization. A good initialization should have a relatively flat convolution. The "flat convolution" problem is related to the "Golay sequence" or "low autocorrelation sequence" problem in signal processing.

For a continuous function, if we want `f*f` to be as flat as possible on the support, we can think of it in the Fourier domain: `F(t)*F(t) = |F(t)|^2` (for real f, the autocorrelation's Fourier transform is the power spectrum). For a function supported on [0, n], the Fourier transform is `F(t) = integral_0^n f(x) e^{-2πitx} dx`. We want `|F(t)|` to be constant. This is related to the "flat-top" window in signal processing!

The optimal continuous solution would be a function whose Fourier transform has constant magnitude. This is related to the "prolate spheroidal wave functions" or "DPSS" (Discrete Prolate Spheroidal Sequences).

Actually, for the discrete case, we want a sequence `a` of length n such that the DFT of `a` has constant magnitude (or as close to constant as possible). This is exactly the problem of finding a sequence with a "flat" power spectrum!

This gives us a great initialization strategy:

2. Use it as the initial sequence

But DPSS sequences are complex in general. For real sequences, we need the DFT to be "as flat as possible" while being real.

Actually, let me think about this differently. The autocorrelation of a real sequence `a` is:
`r[k] = sum_i a[i] * a[i+k]` (for k >= 0)

The Fourier transform of `r` is `|A(t)|^2` where `A(t)` is the DFT of `a`. So minimizing `max(r[k])` is related to making `|A(t)|^2` as flat as possible.

The optimal sequence would have `|A(t)|` constant, which means `a` is a "flat-spectrum" sequence. The simplest such sequence is a rectangular window (constant sequence), but its autocorrelation is triangular (not flat).

To get a flatter autocorrelation, we need a more complex sequence. This is the "minimum autocorrelation" problem, which is related to Barker sequences, Zadoff-Chu sequences, etc.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score(a):
    """Compute 2n * max(convolve(a,a)) / sum(a)^2"""
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * len(a) * np.max(c) / (s * s)

def pnorm_objective_and_grad(z, p):
    """
    Objective and gradient in softmax parameterization.
    a = softmax(z), objective = 2n * (sum c_k^p)^(1/p)
    """
    n = len(z)
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    log_c = np.log(c_pos)
    p_log_c = p * log_c
    m = np.max(p_log_c)
    shifted = p_log_c - m
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    
    # p-norm value
    log_pnorm = (m + np.log(sum_exp)) / p
    pnorm_val = np.exp(log_pnorm)
    
    # weights w_k = c_k^p / S = exp_shifted[k] / sum_exp
    w = exp_shifted / sum_exp
    
    # w/c
    wc = w / c_pos
    
    # correlate(wc, a, 'valid')
    corr = np.correlate(wc, a, mode='valid')
    
    # gradient of 2n * pnorm with respect to a
    g_a = 4.0 * n * pnorm_val * corr
    
    # gradient with respect to z: a * (g_a - dot(g_a, a))
    g_z = a * (g_a - np.dot(g_a, a))
    
    value = 2.0 * n * pnorm_val
    return value, g_z

def pnorm_lbfgs(a0, time_limit, p_stages=None):
    """Run L-BFGS with p-norm smoothing, going through multiple p stages."""
    n = len(a0)
    if p_stages is None:
        p_stages = [20, 50, 100, 200, 500, 1000, 3000, 10000]
    
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.max()
    t0 = time.monotonic()
    
    for p in p_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        def func(z_val):
            val, _ = pnorm_objective_and_grad(z_val, p)
            return val
        
        def jac(z_val):
            _, g = pnorm_objective_and_grad(z_val, p)
            return g
        
        # More iterations for higher p
        if p <= 100:
            maxiter = 500
        elif p <= 1000:
            maxiter = 800
        else:
            maxiter = 1500
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter * 50, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at p={p}: {e}")
        
        # Evaluate
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"    p={p} ({elapsed:.0f}s): score={sc:.8f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

def structured_init(n, variant=0, alpha=None):
    """Create structured initial sequence of length n, normalized to sum=1."""
    if variant == 0:  # Gaussian
        if alpha is None:
            alpha = 3.0
        x = np.linspace(-1, 1, n)
        a = np.exp(-alpha * x**2)
    elif variant == 1:  # Cosine
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2)**2
    elif variant == 2:  # Sine
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)**2
    elif variant == 3:  # Flat-top
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        # Smooth edges
        edge = max(1, n // 10)
        for i in range(edge):
            a[i] *= np.cos(np.pi * i / (2 * edge))**2
            a[n-1-i] *= np.cos(np.pi * i / (2 * edge))**2
    elif variant == 4:  # Bimodal
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x)**2
    elif variant == 5:  # Wide Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
- Numerically stable p-norm L-BFGS with log-sum-exp (p up to 20000)
- Systematic multi-length exploration with multi-start initializations
- Convexity exploitation (problem is convex for fixed n)
- Aggressive local refinement with peak shaving and mass redistribution
- Symmetric sequence constraint option for improved convergence
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(convolve(a,a)) / sum(a)^2."""
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return float(2.0 * len(a) * np.max(c) / (s * s))


def pnorm_value_grad_z(z, p):
    """
    Compute objective value and gradient w.r.t. z in softmax parameterization.
    a = softmax(z), objective = 2n * (sum c_k^p)^(1/p) where c = convolve(a,a).
    Uses log-sum-exp for numerical stability at high p.
    """
    n = len(z)
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    log_c = np.log(c_pos)
    p_log_c = p * log_c
    m = np.max(p_log_c)
    shifted = p_log_c - m
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)

    log_pnorm = (m + np.log(sum_exp)) / p
    pnorm_val = np.exp(log_pnorm)

    # Normalized weights: w_k = c_k^p / sum(c_j^p)
    w = exp_shifted / sum_exp

    # Gradient of pnorm w.r.t. a: 2 * pnorm * sum_k w_k * a[k-i] / c_k
    # = 2 * pnorm * correlate(w/c, a, 'valid')
    wc = w / c_pos
    corr = np.correlate(wc, a, mode='valid')
    g_a = 4.0 * n * pnorm_val * corr

    # Gradient w.r.t. z via chain rule for softmax
    g_z = a * (g_a - np.dot(g_a, a))

    value = 2.0 * n * pnorm_val
    return value, g_z


def pnorm_lbfgs(a0, time_limit, p_stages=None):
    """
    L-BFGS optimization with p-norm smoothing schedule.
    Transitions through increasing p values for progressively better max approximation.
    """
    n = len(a0)
    if p_stages is None:
        p_stages = [20, 50, 100, 200, 500, 1000, 3000, 10000]

    z = np.log(np.maximum(a0, 1e-300))
    z -= z.max()
    t0 = time.monotonic()

    for stage_idx, p in enumerate(p_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def func(z_val):
            val, _ = pnorm_value_grad_z(z_val, p)
            return val

        def jac(z_val):
            _, g = pnorm_value_grad_z(z_val, p)
            return g

        # Scale iterations with p: higher p needs more iterations
        if p <= 50:
            maxiter = 400
        elif p <= 200:
            maxiter = 600
        elif p <= 1000:
            maxiter = 1000
        else:
            maxiter = 2000

        # Limit iterations based on remaining time (rough estimate: ~0.5ms per iter for n~1000)
        est_time_per_iter = len(z) * 5e-6
        maxiter = min(maxiter, int(remaining / (est_time_per_iter * 3)))
        maxiter = max(maxiter, 100)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={
                    'maxiter': maxiter,
                    'ftol': 1e-22,
                    'gtol': 1e-18,
                    'maxfun': maxiter * 100,
                    'disp': False
                }
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at p={p}: {e}")

        # Evaluate current score
        z_m = z.max()
        exp_z = np.exp(z - z_m)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"    p={p:>6d} ({elapsed:5.0f}s): score={sc:.10f}")

    z_m = z.max()
    exp_z = np.exp(z - z_m)
    a = exp_z / exp_z.sum()
    return a


def structured_init(n, variant=0, alpha=None):
    """Create structured initial sequence of length n, normalized to sum=1."""
    if variant == 0:  # Gaussian center
        if alpha is None:
            alpha = 3.0
        x = np.linspace(-1, 1, n)
        a = np.exp(-alpha * x ** 2)
    elif variant == 1:  # Cosine taper
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:  # Sine (zero at edges)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif variant == 3:  # Flat with cosine edge taper
        a = np.ones(n)
        edge = max(1, n // 12)
        for i in range(edge):
            a[i] *= np.cos(np.pi * i / (2 * edge)) ** 2
            a[n - 1 - i] *= np.cos(np.pi * i / (2 * edge)) ** 2
    elif variant == 4:  # Wide Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2)
    elif variant == 5:  # Bimodal
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 6:  # Raised cosine (Hann window)
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 7:  # Flat-top window (approx)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = max(1, n // 8)
        for i in range(edge):
            t = i / edge
            a[i] *= (1 - 0.5 * t ** 2) ** 2
            a[n - 1 - i] *= (1 - 0.5 * t ** 2) ** 2
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def local_refinement(a0, time_limit, n_steps=100000):
    """
    Local refinement via coordinate descent, peak shaving, and mass redistribution.
    Exploits the convex structure: any improvement locally is global.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Focus on interior peaks (ignore endpoints which are a[0]^2, a[n-1]^2)
        if len(c) > 2:
            c_inner = c[1:-1].copy()
        else:
            c_inner = c.copy()

        peak_rel = np.argmax(c_inner)
        peak_idx = peak_rel + (1 if len(c) > 2 else 0)
        peak_val = c[peak_idx]

        # Find near-peak positions (within 5% of max)
        threshold = peak_val * 0.95
        near_peaks = np.where(c_inner >= threshold)[0]
        near_peaks = near_peaks[:10] + (1 if len(c) > 2 else 0)

        improved = False

        # Compute contribution scores to near-peaks
        contrib_scores = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contribs = a[idx] * a[k - idx]
                contrib_scores[idx] += contribs

        # Strategy 1: Single coordinate reduction on top contributors
        top_coords = np.argsort(contrib_scores)[::-1][:20]
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80, 0.75, 0.70]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0.0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        # Strategy 2: Pair reduction at worst peak
        if not improved:
            worst_k = near_peaks[0]
            js = max(0, worst_k - n + 1)
            je = min(n - 1, worst_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[worst_k - idx]
            order = np.argsort(contribs)[::-1][:8]
            for ci in range(len(order) - 1):
                i = idx[order[ci]]
                j = idx[order[ci + 1]]
                if i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85, 0.80]:
                    for f2 in [0.95, 0.90, 0.85, 0.80]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial /= st
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Redirect mass from peak-contributors to valley-contributors
        if not improved and step % 20 == 0:
            min_rel = np.argmin(c_inner)
            min_k = min_rel + (1 if len(c) > 2 else 0)
            # Find element that contributes heavily to peak but little to min
            peak_contrib = {}
            js_p = max(0, peak_idx - n + 1)
            je_p = min(n - 1, peak_idx)
            for i in range(js_p, je_p + 1):
                pc = a[i] * a[peak_idx - i]
                mc = a[i] * a[min_k - i] if 0 <= min_k - i < n else 0.0
                if pc > 0 and mc < pc * 0.3:
                    peak_contrib[i] = (pc, mc)
            if peak_contrib:
                best_i = max(peak_contrib.keys(), key=lambda x: peak_contrib[x][0] - peak_contrib[x][1])
                for factor in [0.98, 0.95, 0.92, 0.90]:
                    a_trial = a.copy()
                    diff = a_trial[best_i] * (1 - factor)
                    a_trial[best_i] -= diff
                    if a_trial[best_i] < 0:
                        a_trial[best_i] = 0
                        diff = a[best_i]
                    # Add to element that contributes to min

                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

            if stall >= 2000:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalRefine step={step} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def symmetric_pnorm_lbfgs(a0, time_limit, p_stages=None):

    L-BFGS with p-norm smoothing, enforcing symmetry a[i] = a[n-1-i].
    Only optimizes the first half of the sequence.
    """

    half = (n + 1) // 2
    if p_stages is None:
        p_stages = [50, 100, 300, 1000, 5000]

    # Initialize with symmetric version
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = (a0[i] + a0[n - 1 - i]) / 2
        a_sym[n - 1 - i] = a_sym[i]
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0:
        a_sym /= s

    # Work in log-domain for the half sequence
    z_half = np.log(np.maximum(a_sym[:half], 1e-300))
    z_half -= z_half.max()
    t0 = time.monotonic()

    for p in p_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func_sym(zh):
            # Reconstruct full symmetric sequence
            zh_shifted = zh - np.max(zh)
            exp_z = np.exp(zh_shifted)
            a_half = exp_z / exp_z.sum()
            a_full = np.zeros(n)
            for i in range(half):
                a_full[i] = a_half[i]
                a_full[n - 1 - i] = a_half[i]
            s = a_full.sum()
            a_full /= s

            c_pos = np.maximum(c, 1e-300)

            shifted = p_log_c - m
            exp_s = np.exp(shifted)
            sum_e = np.sum(exp_s)

            pnorm_val = np.exp(log_pnorm)

        def jac_sym(zh):

            a_full = np.zeros(n)

            a_full /= s
            c = fast_convolve(a_full)
            c_pos = np.maximum(c, 1e-300)
            log_c = np.log(c_pos)
            p_log_c = p * log_c
            m = np.max(p_log_c)
            shifted = p_log_c - m
            exp_s = np.exp(shifted)
            sum_e = np.sum(exp_s)
            log_pnorm = (m + np.log(sum_e)) / p
            pnorm_val = np.exp(log_pnorm)
            w = exp_s / sum_e
            wc = w / c_pos
            corr = np.correlate(wc, a_full, mode='valid')
            g_full = 4.0 * n * pnorm_val * corr

            # Map gradient back to half-sequence parameters
            # da_full[i]/da_half[j] involves the sum normalization
            # Simplified: just use the first half of g_full
            g_half = g_full[:half].copy()
            if n % 2 == 1:
                g_half[-1] *= 2.0  # middle element appears twice

            # Adjust for the normalization da/dz = a*(delta - a)
            dot_ga = np.dot(g_full, a_full)
            g_z = a_half * (g_half - dot_ga * a_half * 2.0)  # approximate
            return g_z

        maxiter = 800 if p <= 100 else 1200
        try:
            res = minimize(
                fun=func_sym,
                x0=z_half,
                jac=jac_sym,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter * 50, 'disp': False}
            )
            z_half = res.x
        except Exception:
            pass

        # Evaluate
        zh_m = z_half.max()
        exp_z = np.exp(z_half - zh_m)
        a_h = exp_z / exp_z.sum()
        a_full = np.zeros(n)
        for i in range(half):
            a_full[i] = a_h[i]
            a_full[n - 1 - i] = a_h[i]
        a_full /= a_full.sum()
        sc = 2.0 * n * fast_convolve(a_full).max()
        elapsed = time.monotonic() - t0
        print(f"    Sym p={p:>6d} ({elapsed:5.0f}s): score={sc:.10f}")

    zh_m = z_half.max()
    exp_z = np.exp(z_half - zh_m)
    a_h = exp_z / exp_z.sum()
    a_full = np.zeros(n)
    for i in range(half):
        a_full[i] = a_h[i]
        a_full[n - 1 - i] = a_h[i]
    a_full /= a_full.sum()
    return a_full


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Length sweep with moderate p-norm L-BFGS (multiple inits per length)
    Phase 2: Deep p-norm optimization at best lengths (very high p)
    Phase 3: Local refinement with peak shaving
    Phase 4: Final polish with highest p values
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None

    # Load parent if available
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        if p_arr.ndim == 1 and p_arr.size > 0 and np.isfinite(p_arr).all() and p_arr.sum() > 1e-10:
            parent = p_arr / p_arr.sum()
            parent_score = compute_score(parent)
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
            best_score = parent_score
            best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Length sweep ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Length Sweep (budget: {(deadline - start_time) * 0.35:.0f}s)")
    print(f"{'='*60}")

    if parent is not None:
        base_n = len(parent)
        lengths = sorted(set([
            base_n - 128, base_n - 64, base_n - 32,
            base_n, base_n + 32, base_n + 64, base_n + 128,
            base_n + 192, base_n + 256,
            768, 896, 1024, 1152, 1280
        ]))
        lengths = [x for x in lengths if 384 <= x <= 3072]
        # Sort by distance to parent
        lengths.sort(key=lambda x: abs(x - base_n))
    else:
        lengths = [768, 896, 960, 1024, 1152, 1280, 832, 1088, 1408, 640]

    # Take top 8 lengths
    lengths = lengths[:8]
    n_phase1 = len(lengths)
    t_phase1_total = min((deadline - start_time) * 0.35, 500.0)
    t_per_length = t_phase1_total / n_phase1

    length_results = {}

    for li, alt_n in enumerate(lengths):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(t_per_length, rem * 0.5, 100.0)
        if t_alloc < 25:
            continue

        print(f"\n  [Phase1] n={alt_n}, budget={t_alloc:.0f}s")

        best_for_n = None
        best_sc_n = float('inf')

        # Multiple initializations
        n_inits = 4
        t_per_init = t_alloc / n_inits

        for vi in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(alt_n, variant=vi)
            a_opt = pnorm_lbfgs(a_init, t_per_init, p_stages=[20, 50, 100, 300])
            sc = 2.0 * alt_n * fast_convolve(a_opt).max()
            if sc < best_sc_n:
                best_sc_n = sc
                best_for_n = a_opt.copy()

        length_results[alt_n] = (best_for_n, best_sc_n)
        print(f"  [Phase1] n={alt_n}: best={best_sc_n:.10f}")

        if best_sc_n < best_score:
            best_score = best_sc_n
            best_seq = best_for_n.copy()
            print(f"  >>> NEW BEST: {best_score:.10f} (n={alt_n})")

    # === Phase 2: Deep optimization at top lengths ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep Optimization (best={best_score:.10f})")
    print(f"{'='*60}")

    sorted_lengths = sorted(length_results.items(), key=lambda x: x[1][1])
    top_k = min(3, len(sorted_lengths))
    rem = deadline - time.monotonic()
    t_per_deep = min(rem * 0.30 / max(1, top_k), 350.0)

    for alt_n, (a_best_n, sc_n) in sorted_lengths[:top_k]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(t_per_deep, rem * 0.4)
        if t_alloc < 40:
            break

        print(f"\n  [Phase2] Deep n={alt_n}, budget={t_alloc:.0f}s, start={sc_n:.10f}")

        # Full p-norm schedule with very high p
        a_deep = pnorm_lbfgs(a_best_n, t_alloc, p_stages=[50, 100, 300, 1000, 3000, 10000])
        sc = 2.0 * alt_n * fast_convolve(a_deep).max()
        print(f"  [Phase2] n={alt_n} deep: score={sc:.10f}")

        if sc < best_score:
            best_score = sc
            best_seq = a_deep.copy()
            print(f"  >>> NEW BEST: {best_score:.10f} (n={alt_n})")

        # Also try symmetric constraint
        rem = deadline - time.monotonic()
        if rem > 40 and alt_n % 2 == 0:
            t_sym = min(60.0, rem * 0.2)
            if t_sym > 20:
                a_sym = symmetric_pnorm_lbfgs(a_best_n, t_sym, p_stages=[100, 500, 2000])
                sc_sym = 2.0 * alt_n * fast_convolve(a_sym).max()
                print(f"  [Phase2] n={alt_n} symmetric: score={sc_sym:.10f}")
                if sc_sym < best_score:
                    best_score = sc_sym
                    best_seq = a_sym.copy()
                    print(f"  >>> NEW BEST: {best_score:.10f} (n={alt_n}, symmetric)")

    # === Phase 3: Local refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Local Refinement (best={best_score:.10f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    if rem > 60:
        t_lr = min(rem * 0.25, 350.0)
        print(f"  Budget: {t_lr:.0f}s")
        a_lr, sc_lr = local_refinement(best_seq, t_lr, n_steps=200000)
        if sc_lr < best_score:
            best_score = sc_lr
            best_seq = a_lr.copy()
            print(f"  >>> NEW BEST after local: {best_score:.10f}")
        else:
            print(f"  Local refinement: score={sc_lr:.10f} (no improvement)")

    # === Phase 4: Final polish ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish (best={best_score:.10f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 250.0)
        print(f"  High-p L-BFGS budget: {t_fin:.0f}s")
        a_fin = pnorm_lbfgs(best_seq, t_fin, p_stages=[500, 2000, 5000, 20000])
        sc = 2.0 * len(a_fin) * fast_convolve(a_fin).max()
        if sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  >>> NEW BEST after polish: {best_score:.10f}")
        else:
            print(f"  Polish: score={sc:.10f} (no improvement)")

    # Final local refinement pass
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lr2 = min(rem * 0.5, 200.0)
        a_lr2, sc_lr2 = local_refinement(best_seq, t_lr2, n_steps=100000)
        if sc_lr2 < best_score:
            best_score = sc_lr2
            best_seq = a_lr2.copy()
            print(f"  >>> NEW BEST after final local: {best_score:.10f}")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            verified_score = float(evaluate_sequence(final_list))
            if np.isfinite(verified_score) and verified_score < best_score:
                best_score = verified_score
        except Exception:
            pass

    total_time = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.10f}, n={len(best_seq)}, time={total_time:.0f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
