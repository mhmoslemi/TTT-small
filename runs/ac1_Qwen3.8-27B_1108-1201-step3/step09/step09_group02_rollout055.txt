def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}. O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) > 0:
        rho = idx[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_Qa(a, q):
    """
    Compute Q*a where Q_{i,j} = q[i+j].
    Uses FFT for efficiency: (Q*a)[i] = sum_j a[j]*q[i+j] = (a_rev * q)[n-1+i]
    where a_rev = a[::-1] and * is linear convolution.

    full = np.convolve(a_rev, q)

            print(f"    SGD n={n} it={it} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score, it

def golden_section_search_1d(f, lo, hi, max_iter=40):
    """Golden section search for minimum of unimodal f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent: for each coordinate, use golden section search
    to find the value that minimizes the full score function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Identify active coordinates (those above threshold)
    threshold = 1e-10

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < threshold:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            # Define score function for golden section
            def _score_fn(x):
                ns = current_sum + (x - v)
                if ns < 0.01:
                    return 1e10
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                nm = max(unaff_max, float(nv.max()))
                return 2.0 * n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n, 0.1 / n)
            best_x = golden_section_search_1d(_score_fn, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.55)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = a + 0.05 * np.ones(n)  # small floor to avoid zero regions
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_fourier_mix(n, rng):
    """Mix of Fourier modes with random coefficients."""
    n_harmonics = int(rng.integers(3, 12))
    amps = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    max_freq = max(1, n // 3)
    freqs = rng.choice(np.arange(1, max_freq + 1), size=min(n_harmonics, max_freq), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.6)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_windowed(n, rng):
    """Raised-cosine window with random phase modulation."""
    t = np.linspace(0, 1, n)
    window = 0.5 * (1 + np.cos(2 * np.pi * t))
    # Add small oscillatory component
    n_mod = int(rng.integers(2, 6))
    mod_freq = rng.integers(1, n // 4)
    window += 0.05 * np.sin(2 * np.pi * mod_freq * t + rng.uniform(0, 2 * np.pi))
    window = np.maximum(window, 0.0)
    s = window.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return window / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=10, verbose=False):
    """Basin hopping: perturb the sequence and re-optimize."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = float(score_normalized(a))
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.55, 90.0)

        # Multiple perturbation strategies
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce values at positions contributing to top peaks
            conv_p = fast_convolve(a)
            top_k = min(8, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 8)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (0.7 + 0.3 * rng.random())
        elif strategy == 2:
            # Subsequence shuffle: swap adjacent blocks
            perturbed = a.copy()
            block_size = max(2, n // 20)
            for _ in range(3):
                i = int(rng.integers(0, n - block_size))
                j = int(rng.integers(0, n - block_size))
                if abs(i - j) > block_size:
                    perturbed[i:i+block_size], perturbed[j:j+block_size] = \
                        perturbed[j:j+block_size].copy(), perturbed[i:i+block_size].copy()
        else:
            # Cross with a random flat-spectrum sequence
            other = init_flat_spectrum(n, rng)
            mix = rng.uniform(0.7, 0.95)
            perturbed = mix * a + (1 - mix) * other

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} (strat={strategy}): improved to {best_score:.8f}")

    return best_a, best_score

def game_cd_hybrid(a0, time_limit, verbose=False, label=""):
    """
    Hybrid: alternate between fictitious play steps and coordinate descent sweeps.
    This combines the global view of the game approach with the local precision of CD.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = float(score_normalized(a))
    t0 = time.monotonic()
    block_time = 30.0  # seconds per block

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        alloc = min(block_time, rem)

        # Alternate: FP then CD
        half = alloc / 2.0
        a_fp, s_fp, _ = fictitious_play(a, half, eta=0.2, verbose=False)
        if s_fp < best_score:
            best_score = s_fp
            best_a = a_fp.copy()
            a = a_fp.copy()

        a_cd, s_cd, _ = exact_coord_descent(a, half, verbose=False, label=label)
        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()
        a = a_cd.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    GCD {label} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization combining game-theoretic fictitious play,
    exact coordinate descent, subgradient methods, and basin hopping.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Fictitious Play on parent and diverse inits ===
    print("\n=== Phase 1: Fictitious Play ===")
    n_parent = len(parent) if parent is not None else 1024

    n_values = [n_parent]
    for alt in [512, 768, 1024, 1536, 2048]:
        if alt != n_parent and alt not in n_values:
            n_values.append(alt)
    n_values = [n for n in n_values if 256 <= n <= 2048][:5]

    candidates = []

    for n in n_values[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break

        # Generate diverse initializations
        inits = []
        if parent is not None and n == n_parent:
            inits.append(('parent', parent.copy()))
        elif parent is not None:
            inits.append(('parent_interp', interpolate_seq(parent, n)))
        inits.append(('flat_spec', init_flat_spectrum(n, rng)))
        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('window', init_windowed(n, rng)))

        for itype, cand in inits[:5]:
            t_left = rem - 30
            if t_left < 10:
                break
            t_alloc = max(10.0, min(60.0, t_left / (len(inits) + 1)))
            print(f"  FP n={n} init={itype} budget={t_alloc:.0f}s")
            a_opt, s_opt, iters = fictitious_play(cand, t_alloc, eta=0.3, verbose=True)
            candidates.append((s_opt, a_opt, n, itype))

            # Quick CD polish
            if s_opt < best_score:
                t_cd = min(20.0, t_left * 0.2)
                a_cd, s_cd, _ = exact_coord_descent(a_opt, t_cd, label=f"fp_{n}_{itype}")
                if s_cd < s_opt:
                    candidates[-1] = (s_cd, a_cd, n, itype)

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best internal: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Intensive Game-CD Hybrid ===
    print(f"\n=== Phase 2: Game-CD Hybrid (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.6, 300.0)
            print(f"  GCD-Hybrid n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_opt, s_opt = game_cd_hybrid(best_seq, t_alloc, verbose=True, label="main")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "gcd_main")

    # === Phase 3: Subgradient descent ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  SGD n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_sg, s_sg, it = subgradient_descent(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "sgd")
            # Follow up with CD
            if s_sg < best_score + 0.01:
                rem2 = deadline - time.monotonic() - 200
                if rem2 > 10:
                    a_cd, s_cd, _ = exact_coord_descent(a_sg, min(rem2 * 0.5, 60.0), label="post_sgd")
                    best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_post_sgd")

    # === Phase 4: Basin Hopping ===
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_bh = min(rem * 0.7, 200.0)
            print(f"  BH n={len(best_seq)}, budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping(best_seq, t_bh, exact_coord_descent, rng,
                                       n_perturbs=12, verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 5: Try different length ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n for n in [n_best // 2, n_best * 2] if 256 <= n <= 2048 and n != n_best]
        for n_alt in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem * 0.5, 100.0)
            init_a = interpolate_seq(best_seq, n_alt)
            print(f"  Multi-length n={n_alt}, budget={t_alloc:.0f}s")
            a_opt, s_opt, _ = fictitious_play(init_a, t_alloc * 0.5, eta=0.3)
            a_cd, s_cd, _ = exact_coord_descent(a_opt, t_alloc * 0.5, label=f"ml_{n_alt}")
            if s_cd < s_opt:
                a_opt, s_opt = a_cd, s_cd
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ml_{n_alt}")

    # === Phase 6: Final intensive CD polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Multiple rounds of exact CD with increasing patience
        for round_i in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  Final CD round {round_i+1}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(best_seq, t_alloc, verbose=True, label=f"fin_{round_i}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"fin_cd_{round_i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
