def fast_convolve(a):

    n = len(a)

    A = rfft(a.astype(np.float64), fft_size)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution a * b via FFT."""

    A = rfft(a.astype(np.float64), fft_size)

    return irfft(A * B, fft_size)[:size]

def exact_subgradient_refine(a0, time_limit, n_steps=50000):
    """Exact subgradient method targeting the argmax of the convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.998
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 20:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)
    
    return best_a, best_score


def peak_directed_descent(a0, time_limit, n_steps=30000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.99, 0.985, 0.98, 0.97, 0.96]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break

                a_trial = a_trial / a_trial.sum()

                    improved = True
                    break

                break
        
        if not improved:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Multi-temperature LogSumExp L-BFGS with verified gradient

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally (fast)."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name

        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def spectral_shaping_init(n, n_iters=300, seed=42):
    """Initialize using iterative spectral shaping to approximate CAZAC property."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    for it in range(n_iters):
        A = np.fft.fft(a)
        power = np.abs(A) ** 2
        target = power[1:].mean()
        alpha = min(1.0, 0.3 + 0.7 * (it + 1) / max(1, n_iters * 0.5))
        phase = np.angle(A)
        new_mag = np.abs(A).copy()
        new_mag[1:] = (1 - alpha) * new_mag[1:] + alpha * np.sqrt(target)
        new_A = new_mag * np.exp(1j * phase)
        new_A[0] = 1.0
        a_new = np.fft.ifft(new_A).real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def structured_init(n, rng):
    """Create various structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.exp(-5 * (x - 0.15) ** 2) + np.exp(-5 * (x - 0.85) ** 2)
    elif choice == 7:
        a = 1.0 + 0.15 * np.cos(2 * np.pi * 7 * x) + 0.1 * np.cos(2 * np.pi * 13 * x)
        a = np.maximum(a, 0)
    elif choice == 8:
        a = np.exp(-8 * (x - 0.5) ** 2)
    else:
        a = x ** 0.3 * (1 - x) ** 0.3 + 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            if stall_rounds >= 8:
                noise_scale = 0.03 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                ss = y.sum()
                if ss > 1e-15:
                    y = y / ss
                stall_rounds = 0
            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_lse_obj(z, n, T_temp):
    """Compute LogSumExp-smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_b.sum())
    return float(2.0 * n * lse)


def compute_lse_grad(z, n, T_temp):
    """
    Gradient of 2n * LSE_T(conv(a,a)) w.r.t. z, where a = softmax(z).
    
    f(a) = 2n * LSE_T(b(a)), b = conv(a,a)
    df/da_j = 2n * sum_k w_k * a_{k-j} where w_k = exp(b_k/T)/sum(exp(b/T))
    df/dz_l = a_l * (df/da_l - sum_j a_j * df/da_j)
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_temp)
    w = exp_b / exp_b.sum()

    # h_l = 2n * sum_k w_k * a_{k-l}
    h = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        h[idx] += wk * a[k - idx]
    h *= 2.0 * n

    S = np.dot(a, h)
    return a * (h - S)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=4000):
    """L-BFGS with LogSumExp smoothing at decreasing temperatures."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if T_stages is None:
        T_stages = [40.0, 20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e20
            return compute_lse_obj(z, n, stage_T)

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return compute_lse_grad(z, n, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        a_cur = softmax_from_z(z)
        sc = evaluate_local(a_cur)
        print(f"    LSG T={stage_T:.2f}: score={sc:.8f} ({time.monotonic()-t0:.0f}s)")
    return softmax_from_z(z)


def exact_subgradient_refine(a0, time_limit, n_steps=50000):
    """Exact subgradient method targeting the argmax positions."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.998
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 20:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)
    return best_a, best_score


def peak_directed_descent(a0, time_limit, n_steps=30000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(12, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.99, 0.985, 0.98, 0.97, 0.96, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try simultaneous reduction
            a_trial = a.copy()
            for pos in top_positions[:4]:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 400:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining CAZAC-inspired spectral shaping initialization,
    Nesterov LP alternating, multi-temperature L-BFGS, and targeted coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    n_target = 1200
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            n_target = len(parent)
    except Exception:
        pass

    if parent is None:
        best_seq = structured_init(n_target, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={n_target}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a.astype(np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Phase 0: Spectral shaping inits
    print(f"\n=== Phase 0: Spectral Shaping (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for sd in [1, 7, 13, 42, 123, 777]:
            if time.monotonic() > deadline - 30:
                break
            a_spec = spectral_shaping_init(n_target, n_iters=250, seed=sd)
            sc = evaluate_local(a_spec)
            print(f"  Spec seed={sd}: {sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_spec.copy()
                print(f"  → New best!")

    # Phase 1: Nesterov LP (main)
    print(f"\n=== Phase 1: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.42, 650.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nlp")

    # Phase 2: L-BFGS LogSumExp
    print(f"\n=== Phase 2: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        T_stages = [40.0, 20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.5, 0.3, 0.15, 0.1, 0.05]
        print(f"  Budget: {t_alloc:.0f}s, T_stages={T_stages}")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=3500)
        try_update(a_lb, "lsg1")

    # Phase 3: Subgradient
    print(f"\n=== Phase 3: Subgrad (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_sg, _ = exact_subgradient_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_sg, "sg")

    # Phase 4: Peak descent
    print(f"\n=== Phase 4: PeakDesc (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_pd, _ = peak_directed_descent(best_seq, t_alloc, n_steps=50000)
        try_update(a_pd, "p")

    # Phase 5: Final L-BFGS at lower T
    print(f"\n=== Phase 5: Final LSG (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        T_stages2 = [3.0, 1.5, 1.0, 0.5, 0.3, 0.15, 0.1, 0.05, 0.03, 0.01]
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages2, maxiter_per_stage=4000)
        try_update(a_lb2, "lsg_f")

    # Phase 6: Perturbation escape
    print(f"\n=== Phase 6: Perturb (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_cur = len(best_seq)
        t_pert = min(rem * 0.3, 200.0)
        x = np.linspace(0, 2 * np.pi, n_cur)
        perts = [
            0.02 * np.cos(x),
            0.02 * np.cos(2 * x + 0.7),
            0.015 * np.sin(3 * x + 0.3),
            0.01 * rng.standard_normal(n_cur),
        ]
        per_pert = t_pert / len(perts)
        for i, pert in enumerate(perts):
            if time.monotonic() > deadline - 30:
                break
            perturbed = best_seq + pert
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            a_p, _ = nesterov_lp_refine(perturbed, min(per_pert, 50.0), solver=solver, n_max_iters=15000)
            try_update(a_p, f"pert{i}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t1 = min(rem * 0.35, 50.0)
        if t1 > 5:
            a_fin = lbgfs_refine(best_seq, t1,
                                 T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                 maxiter_per_stage=3000)
            try_update(a_fin, "fin_lsg")
        rem = deadline - time.monotonic()
        if rem > 15:
            t2 = min(rem * 0.35, 30.0)
            if t2 > 5:
                a_pd2, _ = peak_directed_descent(best_seq, t2, n_steps=25000)
                try_update(a_pd2, "fin_p")
        rem = deadline - time.monotonic()
        if rem > 10:
            t3 = min(rem * 0.4, 20.0)
            if t3 > 3:
                a_sg2, _ = exact_subgradient_refine(best_seq, t3, n_steps=30000)
                try_update(a_sg2, "fin_sg")

    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
