"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a combination of:
1. Spectral design (frequency-domain iterative method)
2. P-norm smoothed L-BFGS optimization
3. Frequency-domain peak shaving
4. Multi-length exploration
Key innovation: spectral initialization and frequency-domain gradient steps
that directly target the flattening of the linear convolution.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]


def score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def spectral_flatness_init(n, n_iters=200):
    """
    Initialize a sequence using spectral method targeting flat convolution.
    The target in frequency domain is the DFT of a flat sequence (Dirichlet kernel).
    We iteratively adjust the phase of A to make IFFT(A^2) as flat as possible.
    """
    N = 1 << (2 * n - 1).bit_length()
    c_target = np.ones(2 * n - 1) / (2 * n - 1)
    C_target = rfft(c_target, N)
    
    # Initialize A: try to make A^2 = C_target
    # Where C_target > 0: A = sqrt(C_target) (real)
    # Where C_target < 0: A = i*sqrt(|C_target|) (imaginary)
    A = np.zeros(N // 2 + 1, dtype=np.complex128)
    for k in range(len(A)):
        if C_target[k] >= 0:
            A[k] = np.sqrt(C_target[k])
        else:
            A[k] = 1j * np.sqrt(-C_target[k])
    
    # Iterate: adjust phases to make a non-negative
    a = np.ones(n) / n
    for it in range(n_iters):
        # Current A from a
        A_cur = rfft(a, N)
        
        # Desired A: same magnitude as sqrt(|C_target|), but adjust phase
        # to make IFFT(A^2) more flat
        # Simple approach: blend A_cur with A_target in magnitude
        A_mag_target = np.sqrt(np.abs(C_target))
        A_phase_cur = np.angle(A_cur)
        
        # New A: use target magnitude, current phase
        A_new = A_mag_target * np.exp(1j * A_phase_cur)
        
        # Inverse FFT
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
    
    return a


def spectral_gradient_step(a, step_size=1.0, N=None):
    """
    One step of spectral gradient descent on the max convolution value.
    
    The idea: in frequency domain, c = a*a has spectrum C = A^2.
    We want to reduce the peaks of c. The gradient of max(c) w.r.t. c 
    is concentrated at the peak. In frequency domain, this becomes
    a modification of A.
    """
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    
    A = rfft(a, N)
    C = A * A
    c = irfft(C, N)[:size]
    
    # Error: we want c to be flat (value 1/size)
    c_flat = 1.0 / size
    e = c - c_flat
    E = rfft(e, N)
    
    # Newton step: A_new = A - E/(2A) (Linearize C = A^2, C_new = C - E = A^2 - E)
    # So A_new^2 = A^2 - E => A_new ≈ A - E/(2A) (first order)
    A_new = A.copy()
    mask = np.abs(A) > 1e-12
    A_new[mask] = A[mask] - step_size * E[mask] / (2 * A[mask])
    
    # Ensure conjugate symmetry (for real output)
    # rfft already handles this, but let's be safe
    a_new = irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    
    return a_new

def p_norm_objective(a, p):
    """P-norm of positive part of convolution, times 2n."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    return 2.0 * n * (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_gradient(a, p):
    """Gradient of p-norm objective w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = np.sum(c_p)
    if S < 1e-300:
        return np.zeros_like(a)
    factor = 2.0 * n * S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    # Gradient: d/d(a_i) of sum_k c_k^p = sum_k p*c_k^{p-1} * d(c_k)/d(a_i)
    # d(c_k)/d(a_i) = 2*a[k-i] (for valid k-i)
    # So gradient[i] = factor * sum_k w_k * 2*a[k-i] = 2*factor * correlate(w, a)[i]
    corr = np.correlate(w, a, mode='valid')
    return 2.0 * factor * corr


def p_norm_lbfgs(a0, time_limit, p_start=20, p_end=500, n_stages=8):
    """P-norm smoothed L-BFGS with logarithmic parameterization."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a = a / a.sum()
    
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    
    for p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(x):
            a_val = np.exp(x) / np.exp(x).sum()
            return p_norm_objective(a_val, p)
        
        def jac(x):
            a_val = np.exp(x) / np.exp(x).sum()
            g_a = p_norm_gradient(a_val, p)
            # Gradient w.r.t. log(a): a * (g_a - dot(g_a, a))
            return a_val * (g_a - np.dot(g_a, a_val))
        
        x0 = np.log(a)
        try:
            res = minimize(func, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12})
            a = np.exp(res.x)
            a = a / a.sum()
        except Exception:
            pass
        
        sc = score(a)
        elapsed = time.monotonic() - t0
        print(f"    p-norm LBFGS p={p:.0f} ({elapsed:.0f}s): {sc:.6f}")
    
    return a

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new

Actually, let me double-check: `np.convolve(a, a)` vs `np.correlate(a, a)`.

`np.convolve(a, a)[k] = sum_j a[j] * a[k-j]` (valid j)
`np.correlate(a, a)[k] = sum_j a[j+k] * conj(a[j])` = `sum_j a[j+k] * a[j]` (for real a)

They're different! The convolution gives `sum a[j]*a[k-j]` while the correlation gives `sum a[j+k]*a[j]`.

For the convolution: the "center" value (k=n-1) is `sum a[j]*a[n-1-j]`.
For the correlation: the "center" value (k=0) is `sum a[j]^2 = ||a||_2^2`.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LSE-smoothed L-BFGS,
spectral gradient descent, frequency-domain peak shaving, and multi-length
exploration. Key innovations: log-sum-exp smoothing (more robust than p-norm
at extreme values), spectral Newton steps in frequency domain, and adaptive
temperature annealing for the LSE objective.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]


def compute_score(a):
    """Compute the objective: 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def lse_value(c, T):
    """Log-sum-exp of c with temperature T."""
    c_max = c.max()
    return T * (c_max + np.log(np.sum(np.exp((c - c_max) / T))))


def lse_objective_a(a, T):
    """2*n*LSE(conv(a,a), T) for normalized a (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * lse_value(c, T)


def lse_gradient_a(a, T):
    """Gradient of 2*n*LSE(conv(a,a), T) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    p = np.exp((c - c_max) / T)
    p = p / p.sum()
    corr = np.correlate(p, a, mode='valid')
    return 4.0 * n * T * corr


def lse_objective_z(z, T):
    """LSE objective in log-domain (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return lse_objective_a(a, T)


def lse_gradient_z(z, T):
    """Gradient of LSE objective w.r.t. log-domain z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    g_a = lse_gradient_a(a, T)
    return a * (g_a - np.dot(g_a, a))


def lse_lbfgs(a0, time_limit, T_start=0.05, T_end=0.001, n_stages=10):
    """LSE-smoothed L-BFGS with temperature annealing in log-domain."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(x):
            return lse_objective_z(x, T)

        def jac(x):
            return lse_gradient_z(x, T)

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': 100000})
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        sc = compute_score(a)
        elapsed = time.monotonic() - t0
        print(f"    LSE LBFGS T={T:.5f} ({elapsed:.0f}s): {sc:.6f}")

    return a


def p_norm_objective_a(a, p):
    """P-norm of convolution, times 2n."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    return 2.0 * n * (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_gradient_a(a, p):
    """Gradient of p-norm objective w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = np.sum(c_p)
    if S < 1e-300:
        return np.zeros_like(a)
    factor = 2.0 * n * S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    return 2.0 * factor * corr

def spectral_gradient_step(a, step_size, N=None):
    """One spectral Newton step to flatten the convolution."""
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()

    A = rfft(a, N)
    C = A * A
    c = irfft(C, N)[:size]

    c_flat = 1.0 / size
    e = c - c_flat
    E = rfft(e, N)

    A_new = A.copy()
    mask = np.abs(A) > 1e-12
    A_new[mask] = A[mask] - step_size * E[mask] / (2.0 * A[mask])

    a_new = irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()

    return a_new

def spectral_flatness_init(n, n_iters=300):
    """Initialize using spectral method targeting flat convolution."""
    N = 1 << (2 * n - 1).bit_length()
    c_target = np.ones(2 * n - 1) / (2 * n - 1)
    C_target = rfft(c_target, N)

    # Build target A from C_target
    A = np.zeros(N // 2 + 1, dtype=np.complex128)
    for k in range(len(A)):
        ct = C_target[k]
        if ct >= 0:
            A[k] = np.sqrt(ct)
        else:
            A[k] = 1j * np.sqrt(-ct)

    a = np.ones(n) / n
    a_prev_score = compute_score(a)

    for it in range(n_iters):
        A_cur = rfft(a, N)
        # Blend: use target magnitude, current phase
        A_mag_target = np.sqrt(np.abs(C_target))
        A_phase_cur = np.angle(A_cur)
        A_new = A_mag_target * np.exp(1j * A_phase_cur)

        # Ensure proper conjugate symmetry for rfft
        # rfft output: A[0] real, A[N/2] real (if N even), others complex
        A_new[0] = np.abs(A_new[0])
        if N % 2 == 0:
            A_new[-1] = np.abs(A_new[-1])

        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            sc = compute_score(a_new)
            if sc < a_prev_score:
                a = a_new
                a_prev_score = sc

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new
