- Just use b*_t as a proxy for the "direction" and do: y <- (1-alpha)*y + alpha*b*_t.

This is what the previous code does, and it works. The Nesterov acceleration just smooths the iterates.

Let me now focus on writing efficient code.

Key efficiency considerations:
- Use FFT for convolutions (O(n log n) per evaluation).
- The LP is the bottleneck: it has 2n-1 constraints and n variables. For n=960, that's ~1919 constraints and 960 variables. With cvxpy, this should be fast enough (maybe 10-100ms per solve).
- For 1800 seconds, we can do maybe 18000-180000 LP solves (depending on speed).

Let me estimate: if each LP solve takes 50ms, we can do 1800/0.05 = 36000 solves. That should be enough for convergence.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response (FTRL/mirror descent on simplex)
2. Top-k peak subgradient refinement
3. Mirror descent with multiplicative weights
4. Local coordinate descent peak shaving
5. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_cols(a):
    """Build matrix T (2n-1) x n such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, solver, T=None):
    """Solve: min_b max_k (y*b)_k s.t. b>=0, sum(b)=1. Returns (b, M, dual_lambda)."""
    n = len(y)
    size = 2 * n - 1
    if T is None:
        T = build_conv_matrix_cols(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    # Try to get dual variables
    lambda_val = np.zeros(size)
    try:
        for i, c in enumerate(constraints[:-1]):  # All except sum(b)=1
            dv = c.dual_value
            if dv is not None:
                lambda_val[i] = max(0, dv)
    except Exception:
        pass
    ls = lambda_val.sum()
    if ls > 1e-15:
        lambda_val = lambda_val / ls
    return b_val, M_val, lambda_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000, use_lambda=False):
    """Nesterov-accelerated LP best-response refinement."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    T_mat = build_conv_matrix_cols(a)  # Will update when a changes
    
    for it in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response to y
        # Note: T depends on y, not a! T @ b = convolve(y, b)
        T_y = build_conv_matrix_cols(y)
        b_new, M_val, lam = solve_lp_best_response(y, solver, T_y)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y_new)
        
        # FTRL-style average
        a = ((it) * a + (alpha * (it + 1)) * b_new) / (it + alpha * (it + 1))
        a = project_simplex(a)
        
        # Check improvement
        if it % 10 == 0:
            sc_a = score_normalized(a)
            if sc_a < best_score - 1e-12:
                best_score = sc_a
                best_a = a.copy()
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            sc = score_normalized(a)
            print(f"    LP iter={it} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f} M={M_val:.8f}")
    
    # Final check
    sc = score_normalized(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def mirror_descent(a0, time_limit, n_max_iters=50000, eta0=0.5, decay=0.9995):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta0
    
    for it in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Subgradient: grad[i] = 2 * a[peak_k - i] (reflected, bounded)
        # conv(a, e_k)[i] = a[k-i] if 0 <= k-i < n, else 0
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Multiplicative update
        log_a = np.log(a + 1e-300)
        log_a -= eta * grad
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a = a_new
        
        eta *= decay
        
        if it % 50 == 0:
            sc = score_normalized(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def topk_mirror_descent(a0, time_limit, k_top=5, n_max_iters=50000, eta0=0.3, decay=0.999):
    """Mirror descent targeting top-k peaks simultaneously."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta0
    
    for it in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Get top-k peaks
        top_k = np.argsort(b)[::-1][:k_top]
        max_val = b[top_k[0]]
        if max_val < 1e-15:
            break
        # Weights proportional to peak values
        weights = b[top_k] / b[top_k].sum()
        
        # Combined subgradient
        grad = np.zeros(n)
        for w, pk in zip(weights, top_k):
            for i in range(n):
                j = int(pk) - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * a[j]
        
        # Multiplicative update
        log_a = np.log(a + 1e-300)
        log_a -= eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a = a_new
        
        eta *= decay
        
        if it % 100 == 0:
            sc = score_normalized(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-MD iter={it} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_peak_shave(a0, time_limit, n_steps=100000):
    """Coordinate descent: reduce elements that contribute to the peak."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_idx = np.argsort(contribs)[::-1][:min(20, len(idx))]
        improved = False
        
        for ti in top_idx:
            i = idx[ti]
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.95]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                b_trial = fast_convolve(a_trial)
                sc = 2.0 * n * b_trial.max()
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break
        
        if not improved:
            # Try random coordinate
            i = rng.integers(n)
            if a[i] > 1e-10:
                a_trial = a.copy()
                a_trial[i] *= 0.99
                a_trial = project_simplex(a_trial)
                b_trial = fast_convolve(a_trial)
                sc = 2.0 * n * b_trial.max()
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
        
        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Perturb and try again
                pert = 0.01 * rng.standard_normal(n)
                a_trial = a + pert
                a_trial = project_simplex(a_trial)
                b_trial = fast_convolve(a_trial)
                sc = 2.0 * n * b_trial.max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                no_improve = 0
        
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PS step={step} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective (softmax parameterization) - main improvement
2. Nesterov-accelerated LP best-response refinement

4. Coordinate descent peak shaving for final polish
5. Multi-length exploration
Key innovation: L-BFGS in softmax space gives superlinear convergence near optimum.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def smoothed_value(a, T):
    """Log-sum-exp smoothed max of conv(a,a)."""
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    return m / T + np.log(np.exp(Tb - m).sum()) / T


def grad_smoothed(a, T):
    """Gradient of smoothed objective w.r.t. a."""
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    n = len(a)
    # grad[i] = 2 * sum_j a[j] * w[j+i] ... actually:
    # d/d a[i] of conv(a,a)[k] = a[k-i] + a[k-i] (if both valid) = 2*a[k-i]
    # So grad[i] = 2 * sum_k w[k] * a[k-i] (over valid k-i)
    # This is 2 * (a correlated with w)[i]
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # Correlation: corr[i] = sum_j a[j] * w[j-i] = ifft(conj(A) * W)[i]
    corr = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * corr


def grad_smoothed_z(z, T):
    """Gradient of smoothed objective w.r.t. z (softmax parameterization)."""
    a = softmax(z)
    g_a = grad_smoothed(a, T)
    # Chain rule: d/dz_i = a_i * (g_i - sum_j a_j * g_j)
    return a * (g_a - np.dot(a, g_a))


def lbfgs_smoothed(a0, time_limit, T_start=30.0, T_end=800.0, n_T_stages=8):
    """L-BFGS on smoothed objective using scipy, with temperature continuation."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    T_values = np.geomspace(T_start, T_end, n_T_stages)

    for stage, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit:
            break

        def obj(z_local):
            a_local = softmax(z_local)
            return smoothed_value(a_local, T)

        def grad(z_local):
            a_local = softmax(z_local)
            g_a = grad_smoothed(a_local, T)
            return a_local * (g_a - np.dot(a_local, g_a))

        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        max_iter = max(20, int(remaining * 80))

        try:
            res = minimize(obj, z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'gtol': 1e-10,
                                    'maxfun': max_iter * 3, 'ftol': 1e-15})
            z_new = res.x
        except Exception:
            z_new = z.copy()

        z_new = z_new - z_new.mean()
        a_curr = softmax(z_new)
        sc = score_normalized(a_curr)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_curr.copy()

        z = z_new
        elapsed = time.monotonic() - t0
        print(f"    LBFGS stage={stage} T={T:.0f} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(y):
    """Build (2n-1) x n matrix T such that T @ b = convolve(y, b)."""
    n = len(y)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def get_lp_solver():
    for name in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, solver):
    """Solve: min_b M s.t. conv(y,b)[k] <= M, b>=0, sum(b)=1."""
    n = len(y)
    T_mat = build_conv_matrix(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def mirror_descent_topk(a0, time_limit, k_top=5, n_max_iters=100000, eta0=0.5, decay=0.9995):
    """Mirror descent (multiplicative weights) targeting top-k peaks."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta0

    for it in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        top_k = np.argsort(b)[::-1][:k_top]
        max_val = b[top_k[0]]
        if max_val < 1e-15:
            break
        weights = b[top_k]
        weights = weights / weights.sum()

        # Combined subgradient
        grad = np.zeros(n)
        for w, pk in zip(weights, top_k):
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for i in range(js, je + 1):
                grad[i] += 2.0 * w * a[pk - i]

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a = a_new
        eta *= decay

        if it % 100 == 0:
            sc = score_normalized(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top_idx = np.argsort(contribs)[::-1][:min(15, len(idx))]
        improved = False

        for ti in top_idx:
            i = idx[ti]
            if a[i] < 1e-15:
                continue
            for factor in [0.9995, 0.999, 0.997, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                b_trial = fast_convolve(a_trial)
                sc = 2.0 * n * b_trial.max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ti in top_idx[:6]:
                i = idx[ti]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue
                for f1 in [0.999, 0.997, 0.995]:
                    for f2 in [0.999, 0.997, 0.995]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        b_trial = fast_convolve(a_trial)
                        sc = 2.0 * n * b_trial.max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve % 200 == 0:
                # Random perturbation
                pert_size = min(0.005, 5.0 / np.sqrt(step + 1))
                pert = pert_size * rng.standard_normal(n)
                a_trial = a + pert
                a_trial = project_simplex(a_trial)
                b_trial = fast_convolve(a_trial)
                sc = 2.0 * n * b_trial.max()
                if sc <= best_score:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                no_improve = 0

            if no_improve > 2000:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PS step={step} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
