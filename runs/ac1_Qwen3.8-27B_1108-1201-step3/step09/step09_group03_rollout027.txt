```python
def lp_best_response(a, M):
    """
    Given fixed a and bound M, find b that maximizes sum(b)
    subject to conv(a,b) <= M and b >= 0.
    """
    import cvxpy as cp
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Build convolution matrix A: A @ b = conv(a, b)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    # Solve
    prob.solve(solver=cp.ECOS)  # or SCS, or other
    return b.value
```

"""
LP-based block coordinate descent with subgradient refinement
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Strategy:
1. LP-based alternating minimization (main phase): alternate between
   solving LP to find optimal response in each direction.
2. Multi-peak subgradient mirror descent for smooth refinement.
3. Coordinate descent with analytical 1D minimization for final polishing.
4. Multiple diverse initializations to avoid local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix A such that A @ b = conv(a, b). Shape: (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_best_response(a, M, solver=None):
    """
    Solve: maximize sum(b) subject to conv(a, b) <= M, b >= 0.
    Returns the optimal b.
    """
    import cvxpy as cp
    n = len(a)
    A = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [A @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    
    if solver is None:
        # Try solvers in order of preference
        for s in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            try:
                prob.solve(solver=s, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solver = s
                    break
            except Exception:
                continue
    else:
        prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0)
    s = val.sum()
    if s < 1e-15:
        return None
    return val / s

def mirror_descent(a0, time_limit, lr_init=0.05, top_k=20, verbose=False):
    """Mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act < 2 * n - 1:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)
        
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Mirror descent update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 1000 == 0:
            print(f"      MD it={it}: best={best_score:.6f}")
    
    return best_a, best_score

def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent with 1D line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            # Try different values for a[i]
            base = a[i]
            affected = conv[k_start:k_end + 1].copy()
            a_shift = a[k_start - i:k_end - i + 1]  # a[k-i] for k in [k_start, k_end]
            
            best_new_M = M
            best_factor = 1.0
            
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                new_val = base * factor
                if new_val < 0:
                    continue
                delta = new_val - base
                new_aff = affected + 2.0 * delta * a_shift
                # Fix quadratic term at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2*i] + 2.0 * delta * base + delta * delta
                
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_factor = factor
            
            if best_factor != 1.0 and best_new_M < M - 1e-14:
                delta = base * (best_factor - 1.0)
                a[i] = base * best_factor
                changes = 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * base + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                score = 2.0 * n * M
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_a, best_score
