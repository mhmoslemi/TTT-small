```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation for O(n log n) iterations
2. Nesterov-accelerated gradient descent with adaptive learning rate
3. KKT-based fixed-point iteration (A^{-1} @ 1 approach)
4. LP fictitious play for global exploration
5. Structured perturbations for escaping local optima
6. Multi-length exploration with interpolation
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(a, p, beta):
    """
    Compute gradient of smooth max of conv(a,a) with respect to a.
    p[k] = softmax(beta * (conv - max))[k], normalized to sum to 1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # grad[m] = 2 * sum_{j=0}^{n-1} p[m+j] * a[j]
    # = 2 * cross-correlation of p and a
    # Compute via: conv(p, reversed(a))[n-1 : 2n-1]
    
    fft_size = 1 << (3 * n - 1 - 1).bit_length()
    P = rfft(p, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv_pa = irfft(P * A_rev, fft_size)
    # Index n-1 to 2n-2 (inclusive) gives us the cross-correlation
    cross = conv_pa[n-1 : 2*n-1]
    return 2.0 * cross

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) > 0:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def nesterov_gradient_descent(a0, time_limit, beta_start=50.0, beta_end=500.0, lr_init=0.01):
    """
    Nesterov-accelerated gradient descent with log-sum-exp smooth max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    y = a.copy()  # Nesterov lookahead
    y_prev = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Adaptive beta
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Nesterov step
        gamma = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * ((it-1) + 1.0) ** 2))
        # Simpler: use standard Nesterov
        a_new = a.copy()
        y = simplex_project(a + (1.0 + 1.0/np.sqrt(it)) * (a - y_prev))
        y_prev = a.copy()
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        max_cv = conv_y.max()
        scaled = beta * (conv_y - max_cv)
        scaled = np.clip(scaled, -100, 0)
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(conv_y))
            p[conv_y.argmax()] = 1.0
        else:
            p = p / p_sum
        
        grad = compute_gradient(y, p, beta)
        
        # Gradient step with adaptive learning rate
        # Backtracking line search (simplified)
        step = lr
        for _ in range(10):
            a_trial = simplex_project(y - step * grad)
            conv_trial = fast_convolve(a_trial)
            score_trial = 2.0 * n * conv_trial.max()
            if score_trial < best_score + 0.01:  # Allow some worsening
                a = a_trial
                break
            step *= 0.5
        else:
            a = a_trial  # Use the last trial
        
        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adapt learning rate
        if it % 100 == 0:
            lr *= 0.9
            
        if it % 1000 == 0:
            print(f"    NGD it={it} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def kkt_fixed_point(a0, time_limit, beta=100.0, damping=0.5, max_iters=1000):
    """
    Fixed-point iteration based on KKT conditions.
    At optimum: A @ a = c * ones (approximately), where A[m,j] = p[m+j].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        
        # Compute p
        conv = fast_convolve(a)
        max_cv = conv.max()
        scaled = beta * (conv - max_cv)
        scaled = np.clip(scaled, -100, 0)
        p = np.exp(scaled)
        p_sum = p.sum()
        p = p / p_sum
        
        # Solve A @ a_new = ones, where A[m,j] = p[m+j]
        # A is n x n
        # A[m,j] = p[m+j] if m+j < 2n-1, else 0
        
        # Build A efficiently (it's a Hankel matrix)
        # A[m,j] = p[m+j]
        # Row 0: p[0], p[1], ..., p[n-1]
        # Row 1: p[1], p[2], ..., p[n]
        # ...
        # Row n-1: p[n-1], p[n], ..., p[2n-2]
        
        A = np.zeros((n, n))
        for m in range(n):
            A[m, :] = p[m:m+n]
        
        # Solve A @ a_new = ones
        try:
            ones = np.ones(n)
            a_new = np.linalg.solve(A, ones)
        except np.linalg.LinAlgError:
            a_new = np.linalg.lstsq(A, np.ones(n), rcond=None)[0]
        
        # Project onto simplex
        a_new = simplex_project(a_new)
        
        # Damped update
        a = (1 - damping) * a + damping * a_new
        
        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if it % 100 == 0:
            print(f"    KKT it={it} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based O(n log n) gradient computation for smooth max objective
2. Nesterov-accelerated gradient descent with adaptive beta and line search
3. KKT-condition-based fixed-point iteration using FFT Hankel solve
4. LP fictitious play (fictitious play with cvxpy) for global exploration
5. Structured gradient-informed perturbations for escaping local optima
6. Multi-length exploration with interpolation
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_p(conv, beta):
    """Compute softmax weights p from convolution values."""
    max_cv = conv.max()
    scaled = beta * (conv - max_cv)
    scaled = np.clip(scaled, -200, 0)
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[conv.argmax()] = 1.0
    else:
        p = p / p_sum
    return p


def compute_gradient_fft(a, p):
    """
    Compute gradient of smooth max of conv(a,a) w.r.t. a using FFT.
    grad[m] = 2 * sum_j p[m+j] * a[j] for valid indices.
    O(n log n) via cross-correlation.
    """
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    P = np.fft.fft(p, N)
    A_rev = np.fft.fft(np.concatenate([a[::-1], np.zeros(N - n)]), N)
    # cross_corr[m] = sum_j p[m+j] * a[j] = ifft(conj(P) * A_rev_full)[m]
    # Actually: cross_corr(p, a)[m] = sum_j p[m+j]*a[j]
    # = ifft(FFT(p) * conj(FFT(a)))[m]
    A = np.fft.fft(np.concatenate([a, np.zeros(N - n)]), N)
    X = P * np.conj(A)
    cross = np.fft.ifft(X).real
    return 2.0 * cross[:n]


def solve_hankel_fft(p, n):
    """
    Solve the Hankel system: sum_j p[m+j] * x[j] = 1 for m = 0,...,n-1.
    p has length 2n-1. Uses FFT. Returns x of length n.
    """
    N = 1 << (3 * n - 1).bit_length()
    p_ext = np.zeros(N, dtype=np.complex128)
    p_ext[:2 * n - 1] = p
    c_ext = np.zeros(N, dtype=np.complex128)
    c_ext[:n] = 1.0

    P = np.fft.fft(p_ext)
    C = np.fft.fft(c_ext)

    # x_ext = ifft(conj(C) / conj(P))
    mask = np.abs(P) > 1e-12
    X = np.zeros(N, dtype=np.complex128)
    X[mask] = np.conj(C[mask]) / np.conj(P[mask])

    x_ext = np.fft.ifft(X)
    return x_ext.real[:n]


def simplex_project(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) > 0:
        rho = idx[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def score_seq(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def nesterov_gradient_descent(a0, time_limit, beta_start=30.0, beta_end=300.0,
                               lr_init=0.05, label="NGD"):
    """
    Nesterov-accelerated gradient descent with log-sum-exp smooth max.
    Uses FFT-based gradient for O(n log n) per iteration.
    Multiplicative updates (mirror descent) for non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)  # momentum

    best_a = a.copy()
    best_score = score_seq(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive beta
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient at current y
        conv_y = fast_convolve(y)
        p = compute_p(conv_y, beta)
        grad = compute_gradient_fft(y, p)

        # Nesterov momentum update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        y_new = y + (v / t_k) * (-grad)
        # Project to simplex
        y_new = simplex_project(y_new)
        v = y_new - y
        y = y_new

        # Mirror descent (multiplicative) update for a
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_trial = np.exp(log_a - log_a.max())
        a_trial = simplex_project(a_trial)

        # Evaluate
        score_trial = score_seq(a_trial)

        # Accept or reject (with backtracking)
        if score_trial < best_score + 0.1:
            a = a_trial
            if score_trial < best_score - 1e-12:
                best_score = score_trial
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            a = y.copy()

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
        elif it % 200 == 0:
            lr = max(lr * 0.995, 1e-6)

        if lr < 1e-8:
            break

        if it % 500 == 0:
            print(f"    {label} it={it} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f} beta={beta:.1f}")

    return best_a, best_score


def kkt_fixed_point(a0, time_limit, beta_start=50.0, beta_end=200.0,
                    damping_start=0.8, damping_end=0.3, label="KKT"):
    """
    Fixed-point iteration targeting KKT conditions:
    At optimum, cross_correlation(p, a) should be (approximately) constant.
    We solve the Hankel system cross_correlation(p, a_new) = 1 using FFT.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_seq(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        beta = beta_start * (beta_end / beta_start) ** progress
        damping = damping_start + (damping_end - damping_start) * progress

        # Compute p
        conv = fast_convolve(a)
        p = compute_p(conv, beta)

        # Solve Hankel system via FFT
        try:
            x = solve_hankel_fft(p, n)
        except Exception:
            x = a.copy()

        # Project to simplex
        x = simplex_project(x)

        # Damped update
        a = (1.0 - damping) * a + damping * x

        # Evaluate
        score = score_seq(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if it % 200 == 0:
            print(f"    {label} it={it} ({elapsed:.0f}s): score={best_score:.8f} beta={beta:.1f} damp={damping:.3f}")

    return best_a, best_score


def combined_kkt_gradient(a0, time_limit, label="CKG"):
    """
    Combined method: alternate between KKT fixed-point and gradient steps.
    KKT gives large moves toward the structure; gradient polishes.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = score_seq(a)
    t0 = time.monotonic()
    it = 0
    beta = 80.0
    lr = 0.02

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        if it % 2 == 0:
            # KKT step
            conv = fast_convolve(a)
            p = compute_p(conv, beta)
            try:
                x = solve_hankel_fft(p, n)
                x = simplex_project(x)
                a = 0.7 * a + 0.3 * x
            except Exception:
                pass
        else:
            # Gradient step
            conv = fast_convolve(a)
            p = compute_p(conv, beta)
            grad = compute_gradient_fft(a, p)
            a_new = simplex_project(a - lr * grad)
            if score_seq(a_new) < score_seq(a) + 0.05:
                a = a_new
            beta *= 1.01
            lr *= 0.999

        score = score_seq(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if it % 300 == 0:
            print(f"    {label} it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lp_best_response(a_fixed, solver=None):
    """Solve LP: min t s.t. conv(a_fixed, b)[k] <= t, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()

    # Build convolution constraints: conv(a,b)[k] = sum_i a[i]*b[k-i]
    # Using numpy convolve structure
    # For each k in 0..2n-2: sum_{i} a[i] * b[k-i] <= t
    # where both indices must be in [0, n-1]

    # Efficient: build the constraint matrix
    # T @ b where T[k, j] = a[j] if k-j is in [0,n-1], else 0
    # T[k, j] = a[j] if 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
    # So T[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)

    constraints = [cp.sum(b_var) == 1]

    # Build matrix T (sparse would be better but let's use dense for simplicity)
    # T has shape (2n-1, n)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[k, j] = a_fixed[j]

    constraints.append(T @ b_var <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def fictitious_play(a0, time_limit, solver=None, label="FP"):
    """
    Fictitious play: alternate best responses, track Nesterov average.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_seq(a)
    a_avg = a.copy()
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Best response to a
        b = lp_best_response(a, solver)
        if b is None:
            b = a.copy()

        # Best response to b
        a_new = lp_best_response(b, solver)
        if a_new is None:
            a_new = b.copy()

        # Nesterov-style average
        gamma = 2.0 / (it + 2.0)
        a_avg = (1.0 - gamma) * a_avg + gamma * a_new
        a = a_new

        # Evaluate candidates
        for cand in [a, b, a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            sc_val = score_seq(cand)
            if sc_val < best_score - 1e-12:
                best_score = sc_val
                best_a = cand.copy()

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_perturbation(a0, rng, strength=0.05):
    """
    Gradient-informed perturbation: push mass away from peak-contributing positions.
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    # Compute pressure: how much each position contributes to peaks
    p = compute_p(conv, 50.0)
    pressure = compute_gradient_fft(a, p)  # This is the gradient direction
    pressure = np.maximum(pressure, 0)  # Only push away
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Perturb: reduce high-pressure, add noise
    noise = rng.standard_normal(n) * 0.1
    a_new = a * (1.0 - strength * pressure) + strength * 0.1 * np.abs(noise) * a.mean()
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_perturbation(a0, rng, frac=0.2):
    """Random restructure: zero out some coords, add noise."""
    n = len(a0)
    a = a0.copy()
    mask = rng.random(n) < frac
    a[mask] = 0
    a *= (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    # Add mass to some zeroed positions
    zeros = np.where(a < 1e-15)[0]
    if len(zeros) > 0:
        n_add = max(1, len(zeros) // 10)
        idx = rng.choice(zeros, n_add, replace=False)
        a[idx] = a.max() * rng.random(n_add) * 0.3
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver_cached
    if '_lp_solver_cached' in globals() and _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining FFT-based gradient descent,
    KKT fixed-point iteration, and LP fictitious play.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(123)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Determine target lengths
    target_lengths = [1024, 2048]
    if parent is not None:
        np_ = len(parent)
        if np_ not in target_lengths:
            target_lengths.insert(0, np_)

    # === Phase 1: Quick KKT + Gradient from multiple starts ===
    print("\n=== Phase 1: Quick KKT + Gradient (multi-start) ===")
    t1_start = time.monotonic()
    candidates = []

    for n in target_lengths:
        t_left = 90.0 - (time.monotonic() - t1_start)
        if t_left < 10:
            break

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_seq(parent, n)))
        # Symmetric cosine init
        t_grid = np.linspace(0, 1, n)
        sym_init = 1.0 + 0.3 * np.cos(np.pi * t_grid) + 0.1 * np.cos(2 * np.pi * t_grid)
        sym_init = np.maximum(sym_init, 0)
        inits.append(("sym", sym_init / sym_init.sum()))
        # Bell-shaped
        bell = np.exp(-2.0 * (t_grid - 0.5) ** 2)
        inits.append(("bell", bell / bell.sum()))
        # Bimodal
        bim = np.exp(-50 * (t_grid - 0.3) ** 2) + np.exp(-50 * (t_grid - 0.7) ** 2)
        inits.append(("bimodal", bim / bim.sum()))
        # Flat
        inits.append(("flat", np.ones(n) / n))

        for name, init in inits:
            t_left = 90.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(10.0, t_left / 2.0))

            # Quick KKT
            a_kkt, s_kkt = kkt_fixed_point(init, t_alloc * 0.5, beta_start=50, beta_end=150,
                                           damping_start=0.7, damping_end=0.4,
                                           label=f"kkt_{name}_n{n}")
            candidates.append((s_kkt, a_kkt, n, f"kkt_{name}"))

            # Quick gradient from KKT result
            t_left = 90.0 - (time.monotonic() - t1_start)
            if t_left > 3:
                a_gd, s_gd = nesterov_gradient_descent(a_kkt, t_alloc * 0.5,
                                                       beta_start=50, beta_end=200,
                                                       lr_init=0.03,
                                                       label=f"gd_{name}_n{n}")
                candidates.append((s_gd, a_gd, n, f"gd_{name}"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best={candidates[0][0]:.6f}")

    for s_c, a_c, n_c, name in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, name)

    # === Phase 2: Intensive Nesterov Gradient Descent ===
    print(f"\n=== Phase 2: Intensive Nesterov GD (best={best_score:.6f}) ===")
    if best_seq is not None:
        t2_start = time.monotonic()
        t_alloc = min(250.0, deadline - time.monotonic() - 600)
        n_best = len(best_seq)
        print(f"  NGD n={n_best}, budget={t_alloc:.0f}s")

        a_opt, s_opt = nesterov_gradient_descent(
            best_seq, t_alloc, beta_start=30, beta_end=400, lr_init=0.05, label="NGD_main"
        )
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "NGD_main")

        # Try with different hyperparameters
        t_left = deadline - time.monotonic() - 400
        if t_left > 30:

    # === Phase 3: KKT Fixed Point (intensive) ===
    print(f"\n=== Phase 3: KKT Fixed Point (best={best_score:.6f}) ===")
    if best_seq is not None:

    # === Phase 4: LP Fictitious Play ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        t_left = deadline - time.monotonic() - 200
        if t_left > 60:

    # === Phase 5: Perturbation + Local Refinement ===
    print(f"\n=== Phase 5: Perturb + Refine (best={best_score:.6f}) ===")

        for rnd in range(8):
            t_left = deadline - time.monotonic() - 100
            if t_left < 30:

            if rnd % 3 == 0:

            elif rnd % 3 == 1:
                perturbed = random_perturbation(best_seq, rng, frac=0.1 + 0.05 * rnd)
            else:

                noisy = best_seq * (1.0 + 0.02 * rng.standard_normal(len(best_seq)))
                noisy = np.maximum(noisy, 0)
                noisy = noisy / noisy.sum()
                perturbed = noisy

            t_alloc = min(t_left * 0.4, 40)
            a_ref, s_ref = nesterov_gradient_descent(
                perturbed, t_alloc, beta_start=50, beta_end=300,
                lr_init=0.03, label=f"refine_r{rnd}"

        # One more gradient pass
        t_left = deadline - time.monotonic() - 10
        if t_left > 10:
            a_fg, s_fg = nesterov_gradient_descent(
                best_seq, min(t_left * 0.8, 30), beta_start=500, beta_end=2000,
                lr_init=0.002, label="FinalGD"
            )
            best_seq, best_score = check_and_update(a_fg, best_seq, best_score, "FinalGD")

    # === Phase 7: Try alternative length ===
    print(f"\n=== Phase 7: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        for alt_n in [n_cur * 2, n_cur + 512, n_cur // 2]:

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
