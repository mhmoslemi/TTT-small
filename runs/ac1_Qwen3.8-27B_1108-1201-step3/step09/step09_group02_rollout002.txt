import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_normalized(a):
    """Compute objective for a sequence (assumes sum=1). Returns 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(np.max(conv))


def precise_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent using closed-form bottleneck analysis.
    For each coordinate i, the optimal x is found by checking:
    - Closed-form critical points for each potential bottleneck
    - Boundary values (x=0, x=current)
    Then picks the best valid candidate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-16 and best_score > 2.0:
                continue

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            U = 0.0
            if k_lo > 0:
                U = max(U, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                U = max(U, float(conv[k_hi + 1:].max()))

            # Partners and constants
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            # Build candidate x values
            candidates = set()
            candidates.add(0.0)
            candidates.add(v)

            # Closed-form critical points
            upper_bound = max(4.0 * v, 6.0 / n, 0.05) if v > 1e-14 else (6.0 / n)

            for idx in range(n_aff):
                k = k_lo + idx
                b_k = a_partners[idx]
                if b_k < 1e-16:
                    continue
                if k == 2 * i:
                    # Quadratic bottleneck: x* = (conv[2i] - v^2) / (1 - v)
                    c_k = conv[k] - v * v
                    denom = 1.0 - v
                    if abs(denom) > 1e-10:
                        x_star = c_k / denom
                        if -1e-10 <= x_star <= upper_bound:
                            candidates.add(max(0.0, x_star))
                else:
                    # Linear bottleneck: x* = 1 - conv[k] / (2 * b_k)
                    x_star = 1.0 - conv[k] / (2.0 * b_k)
                    if -1e-10 <= x_star <= upper_bound:
                        candidates.add(max(0.0, x_star))

            # Add heuristic candidates
            if v > 1e-14:
                candidates.add(0.5 * v)
                candidates.add(1.5 * v)
                candidates.add(0.25 * v)

            # Evaluate all candidates
            best_x = v
            best_f = 2.0 * n * float(conv.max())  # current score (sum=1)

            for x in candidates:

                new_sum = 1.0 + (x - v)
                if new_sum < 0.01:
                    continue

                # Compute new affected convolution entries
                nv = np.empty(n_aff)
                for idx in range(n_aff):
                    k = k_lo + idx
                    if k == 2 * i:
                        nv[idx] = (conv[k] - v * v) + x * x
                    else:
                        nv[idx] = (conv[k] - 2.0 * v * a_partners[idx]) + 2.0 * x * a_partners[idx]

                new_max = max(U, float(nv.max()))
                f = 2.0 * n * new_max / (new_sum * new_sum)

                if f < best_f - 1e-13:
                    best_f = f
                    best_x = x

            # Apply best update

                delta = best_x - v
                a[i] = best_x

                # Update convolution
                for idx in range(n_aff):
                    k = k_lo + idx
                    if k == 2 * i:

                    else:
                        conv[k] += 2.0 * delta * a_partners[idx]

                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:

            no_improve = 0

def spectral_phase_optimize(n, n_iter, rng, time_limit):
    """
    Optimize the DFT phases of a flat-spectrum signal to minimize
    the peak of the linear autocorrelation of the resulting non-negative sequence.
    """
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1

    # Initial random phases
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    t0 = time.monotonic()
    best_score = float('inf')
    best_a = None
    lr = 0.3
    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Build signal from current phases
        spectrum = np.exp(1j * phases)
        a = irfft(spectrum, n_fft)[:n]
        a = np.abs(a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        score = score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.7
                no_improve = 0

        # Compute gradient w.r.t. phases
        conv = fast_convolve(a)
        max_val = float(conv.max())

        # Subgradient: which convolution entries are at the max?
        active = np.where(conv > max_val - 1e-10 * max_val)[0]

        # Gradient of score w.r.t. a[i] (for sum=1 normalized)
        grad_a = np.zeros(n)
        for k in active:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            for ii in range(i_start, i_end + 1):
                j = int(k) - ii
                if 0 <= j < n:
                    grad_a[ii] += a[j]

        # Normalize gradient
        gnorm = np.linalg.norm(grad_a)
        if gnorm > 1e-15:
            grad_a = grad_a / gnorm

        # Convert to phase gradient via chain rule (approximate)
        # d|a|/dphi_j ≈ -Im(a_complex * exp(2*pi*i*j*idx/n_fft))
        # We use a simpler approach: random phase perturbation guided by score
        # Perturb phases
        phase_perturb = rng.standard_normal(n_freq) * lr
        phases_new = phases + phase_perturb

        # Check new phases
        spectrum_new = np.exp(1j * phases_new)
        a_new = irfft(spectrum_new, n_fft)[:n]
        a_new = np.abs(a_new)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-10:
            a_new = a_new / sn
            score_new = score_normalized(a_new)
            if score_new < score:
                phases = phases_new
            else:
                # Try opposite direction
                phases_opp = phases - phase_perturb
                spectrum_opp = np.exp(1j * phases_opp)
                a_opp = irfft(spectrum_opp, n_fft)[:n]
                a_opp = np.abs(a_opp)
                a_opp = np.maximum(a_opp, 0.0)
                so = a_opp.sum()
                if so > 1e-10:
                    a_opp = a_opp / so
                    score_opp = score_normalized(a_opp)
                    if score_opp < score:
                        phases = phases_opp

        # Adaptive learning rate
        lr *= 0.999

    return best_a, best_score


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Gaussian bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_multi_lobe(n, rng):
    """Multi-lobe initialization."""
    n_lobes = rng.integers(2, 6)
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n)

        amp = rng.uniform(0.3, 1.0)

    a = np.maximum(a, 0)
    s = a.sum()

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = a.sum()
    if s < 0.01:

    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score
