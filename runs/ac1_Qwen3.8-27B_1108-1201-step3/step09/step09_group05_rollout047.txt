Let `k = 2n-2-j`. Then this is `sum_m a[m] * w_r[k-m]` which is `(a * w_r)[k]` (the linear convolution).

- `grad[1] = 2 * conv[2n-3]`
- ...
- `grad[n-1] = 2 * conv[n-1]`

So `grad = 2 * conv[n-1 : 2n-1][::-1]` which is `2 * conv[n-1 : 2n-1]` in reverse order.
In numpy: `grad = 2 * np.fliplr(conv[n-1:])` or `grad = 2 * conv[n-1:2n-1][::-1]`.

Since `conv` has length `2n-1`, `conv[n-1:2n-1]` is `conv[n-1:]` which has length `n`. Reversing gives `grad/2`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated L-BFGS with log-sum-exp smoothing (multi-stage annealing to very low T)
2. Dual-memory Frank-Wolfe with away steps
3. Pair-transfer coordinate descent targeting top convolution peaks
4. Multi-scale perturbation and re-optimization
Key: FFT-based gradient makes L-BFGS 10x faster, allowing many more iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    na = len(a)
    size = 2 * na - 1  # assuming len(a) == len(b)
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def eval_score(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def lse_obj_and_grad(z, T, n):
    """Compute objective and gradient for log-sum-exp smoothed problem."""
    a = softmax_from_z(z)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    
    # Gradient
    w = exp_Tb / exp_Tb.sum()
    w_rev = w[::-1]
    conv_aw = cross_conv(a, w_rev)
    grad_a = 2.0 * conv_aw[n-1:][::-1]
    S = np.dot(grad_a, a)
    grad_z = (2.0 * n / T) * a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_schedule, maxiter_per_stage):
    """L-BFGS with log-sum-exp, multi-stage temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        max_iter = max(20, int(maxiter_per_stage * min(1.0, remaining / 60.0)))
        
        def func(z_in):
            obj, _ = lse_obj_and_grad(z_in, T, n)
            return obj
        
        def jac(z_in):
            _, g = lse_obj_and_grad(z_in, T, n)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': max_iter * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def fw_dual_memory(a0, time_limit, n_memory=20):
    """Frank-Wolfe with dual memory (accumulated subgradients)."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    # Dual memory: list of (subgradient, weight) pairs
    memory = []
    
    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        
        # Current subgradient
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]
        
        # Add to memory
        memory.append(g.copy())
        if len(memory) > n_memory:
            memory.pop(0)
        
        # Aggregate gradient
        g_agg = np.zeros(n)
        for mg in memory:
            g_agg += mg
        g_agg /= len(memory)
        
        # FW step: move toward e_j where j = argmin(g_agg)
        j_toward = int(np.argmin(g_agg))
        
        # Try multiple step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = a.copy()
            a_trial[j_toward] += t * (1.0 - a[j_toward])
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial /= s
            sc = eval_score(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g_agg.copy()
            g_masked[~mask] = -np.inf
            j_away = int(np.argmax(g_masked))
            for gamma in [0.3, 0.15, 0.05]:
                a_trial = a.copy()
                a_trial[j_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                sc = eval_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and step > 100:
            # Try adding a small perturbation
            if step % 500 == 0:
                rng_local = np.random.default_rng(step)
                a = a + 0.01 * rng_local.standard_normal(n)
                a = norm_sequence(a)
    
    return best_a, best_score


def pair_transfer_descent(a0, time_limit, n_top=10):
    """Coordinate descent via pair transfers targeting top peaks."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break
        
        b = fast_autocorr(a)
        # Get top peaks
        top_k = np.argsort(b)[::-1][:n_top]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find the pairs (i, k-i) that contribute most to b[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # For each top contributor pair, try transferring mass
            top_contribs = np.argsort(contribs)[::-1][:min(8, len(contribs))]
            
            for ci in top_contribs:
                i = int(idx[ci])
                j = int(k - i)
                if i == j:
                    continue
                if a[i] < 1e-14 or (j < n and a[j] < 1e-14):
                    continue
                
                # Try reducing the larger of a[i], a[j] and increasing a low-value position
                if a[i] >= a[j] if j < n else True:
                    src = i
                else:
                    src = j
                
                # Find a low-value position to transfer to
                b_contrib_to_src = b.copy()  # positions that involve src
                # Low positions: those with small a values
                low_candidates = np.argsort(a)[:n//4]
                
                for dst in low_candidates[:20]:
                    if dst == src:
                        continue
                    if a[dst] > a[src] * 0.5:
                        continue
                    
                    for frac in [0.1, 0.05, 0.02]:
                        dt = frac * a[src]
                        a_trial = a.copy()
                        a_trial[src] -= dt
                        a_trial[dst] += dt
                        if a_trial[src] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        sc = eval_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_score
