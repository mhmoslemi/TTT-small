**Frank-Wolfe (Conditional Gradient) Method:**
The problem is: minimize `f(a) = 2n * max_k (a*a)[k]` over the simplex `{a >= 0, sum(a) = 1}`.

Frank-Wolfe iterates:
1. `s_t = argmin_{s in simplex} <∇f(a_t), s>` (this is the "linear minimization oracle")
2. `a_{t+1} = (1-gamma_t) * a_t + gamma_t * s_t`

The linear minimization oracle for the simplex is simply: pick the coordinate with the minimum gradient component.

The subgradient of `max_k (a*a)[k]` at `a` is: `2 * a[k* - i]` where `k* = argmax_k (a*a)[k]`.

So the gradient of `f(a) = 2n * max_k (a*a)[k]` at `a` (for a specific argmax `k*`) is:
`∇f(a)[i] = 2n * 2 * a[k*-i] = 4n * a[k*-i]` (for valid `k*-i`).

The Frank-Wolfe step would be: `s = e_{argmin_i ∇f(a)[i]}` = unit vector at the coordinate where `a[k*-i]` is smallest.

This means: put all mass at the coordinate that minimizes `a[k*-i]`, which is the coordinate that contributes least to the peak.

This is a very simple algorithm! Let me formalize:

**FW iteration:**
1. Compute `c = conv(a, a)`
2. Find `k* = argmax_k c[k]`
3. Compute gradient: `g[i] = 2*a[k*-i]` for valid `k*-i`, else 0
4. Find `i* = argmin_i g[i]` (the coordinate to "add mass to")
5. Update: `a = (1-gamma)*a + gamma*e_{i*}`

With step size `gamma_t = 2/(t+2)` (standard FW rate).

This is extremely simple but might be slow. However, it's guaranteed to converge and each iteration is O(n) (for the correlation) + O(n) for the argmin.

Actually, this is essentially the same as the "best response" in the LP approach, but without solving the LP. The LP allows a mixture of coordinates (fractional), while FW just picks one coordinate.

A variant: instead of picking a single coordinate, solve the LP to find the best direction (this is the "away-step FW" or "proper FW").

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using log-sum-exp (LSE) smoothing
with adaptive L-BFGS, mirror descent on the simplex, LP-based Frank-Wolfe
alternating optimization, structured asymmetric initialization, and multi-length
exploration. Improves upon p-norm approach by using tighter LSE approximation
of the max, and adds mirror descent for diverse search directions.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective(a, beta):
    """(1/beta)*log(sum exp(beta*c)) where c = conv(a,a). This approximates max(c)."""
    c = fast_convolve(a)
    c_max = c.max()
    if c_max <= 0:
        return 0.0
    shifted = np.exp(beta * (c - c_max))
    return float(c_max + np.log(shifted.sum()) / beta)


def lse_grad(a, beta):
    """Gradient of 2n * (1/beta)*log(sum exp(beta*conv(a,a))) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    if c_max <= 0:
        return np.zeros(n)
    p = np.exp(beta * (c - c_max))
    p = p / p.sum()
    # grad[i] = 4n * sum_m p[i+m] * a[m] for m=0..n-1
    grad = 4.0 * n * np.correlate(p, a, mode='valid')
    return grad


def score(a):
    """Compute the actual score: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def lse_lbfgs(a0, time_limit, beta_start=10.0, beta_end=500.0, n_stages=8):
    """L-BFGS with adaptive LSE smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score(a0)
    
    def func(z_val):
        z_max = z_val.max()
        exp_z = np.exp(z_val - z_max)
        a = exp_z / exp_z.sum()
        return 2.0 * n * lse_objective(a, current_beta)
    
    def jac(z_val):
        z_max = z_val.max()
        exp_z = np.exp(z_val - z_max)
        a = exp_z / exp_z.sum()
        g_a = lse_grad(a, current_beta)
        # Chain rule: d/da * da/dz where a = softmax(z)
        # da_i/dz_j = a_i*(delta_ij - a_j)
        # dF/dz_j = sum_i dF/da_i * da_i/dz_j = a_j*(sum_i g_a_i*a_i ... )
        # Actually: dF/dz = a * (g_a - dot(g_a, a))
        return a * (g_a - np.dot(g_a, a))
    
    betas = np.geomspace(beta_start, beta_end, n_stages)
    iters_per_stage = max(50, 6000 // n_stages)
    
    for beta in betas:
        current_beta = beta
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 100}
            )
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        elapsed = time.monotonic() - t0
        print(f"        LSE-LBFGS beta={beta:.1f} ({elapsed:.0f}s): score={sc:.6f}")
    
    return best_a


def mirror_descent(a0, time_limit, beta=50.0, n_iters=100000, eta=None):
    """Mirror descent with KL regularization on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    
    if eta is None:
        eta = 0.5  # initial step size
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = lse_grad(a, beta)
        
        # Adaptive step size
        if it < 100:
            eta_t = 0.5
        else:
            eta_t = 1.0 / np.sqrt(it + 1)
        
        # Mirror descent update: a[i] *= exp(-eta*g[i]), then normalize
        log_a = np.log(np.maximum(a, 1e-300)) - eta_t * g / (4.0 * n)
        log_a = log_a - log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a = a_new
        
        if it % 1000 == 0:
            sc = score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        MirrorDescend iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    sc = score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a


def build_conv_matrix(a):
    """Build T where T @ b = convolve(a, b). Shape: (2n-1, n)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fw_lp_alternate(a0, time_limit, solver=None, n_max_iters=50000):
    """Frank-Wolfe with LP best response (alternating optimization)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP best response: find b minimizing max_k (a*b)[k]
        b_new, M_val = solve_lp_best_response(a, solver)
        if b_new is None:
            # Fallback: use simple FW step
            c = fast_convolve(a)
            k_star = c.argmax()
            g = np.zeros(n)
            for i in range(n):
                ki = k_star - i
                if 0 <= ki < n:
                    g[i] = 2.0 * a[ki]
                else:
                    g[i] = 0.0
            i_min = g.argmin()
            b_new = np.zeros(n)
            b_new[i_min] = 1.0
            M_val = c.max()
        
        # FW update with diminishing step size
        gamma = 2.0 / (it + 2)
        a = (1.0 - gamma) * a + gamma * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s
        
        if it % 1000 == 0:
            sc = score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"        FW-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, M={M_val:.6f}")
    
    sc = score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP alternating."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    prev_score = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        if it % 500 == 0:
            sc_a = score(a)
            sc_y = score(y)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                prev_score = best_score
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                prev_score = best_score
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    sc = score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a


def local_peak_shave(a0, time_limit, n_steps=20000):
    """Local refinement: reduce peak contributions via coordinate and pair moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score(a)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf
        
        peak_idx = np.argsort(c_work)[::-1][:3]
        peak_val = c_work[peak_idx[0]]
        threshold = peak_val * 0.85
        near_peaks = [k for k in range(len(c_work)) if c_work[k] >= threshold][:6]
        
        improved = False
        
        # Strategy 1: Reduce top contributor to main peak
        k_peak = peak_idx[0]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        top_i = idx[np.argmax(contribs)]
        
        for factor in [0.95, 0.90, 0.85, 0.80]:
            a_trial = a.copy()
            reduced = a_trial[top_i] * (1.0 - factor)
            a_trial[top_i] -= reduced
            if a_trial[top_i] < 0:
                a_trial[top_i] = 0
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a_trial[mask]
            os = others.sum()
            if os < 1e-15:
                break
            a_trial[mask] = others + reduced * others / os
            sc = score(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        # Strategy 2: Pair reduction
        if not improved:
            top_pairs = np.argsort(contribs)[::-1][:4]
            for ci in top_pairs:
                i = idx[ci]
                j = k_peak - i
                if i == j or not (0 <= j < n):
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.93, 0.88, 0.82]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Strategy 3: Random perturbation
        if not improved and step % 50 == 0:
            rng = np.random.default_rng(step + 123)
            a_trial = a * (1.0 + 0.03 * rng.standard_normal(n))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try a few more random attempts before giving up
            if step % 200 == 0:
                rng = np.random.default_rng(step + 456)
                for _ in range(10):
                    a_trial = a * (1.0 + 0.05 * rng.standard_normal(n))
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 0.01:
                        a_trial = a_trial / st
                        sc = score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
            if not improved:
                break
    
    return best_a, best_score


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    
    if variant == 0:
        # Symmetric bell
        a = np.sin(np.pi * x) ** 2
    elif variant == 1:
        # Asymmetric exponential decay
        a = np.exp(-2.0 * x)
    elif variant == 2:
        # Bimodal (peaks at 1/3 and 2/3)
        a = np.exp(-6.0 * (x - 1.0/3)**2) + np.exp(-6.0 * (x - 2.0/3)**2)
    elif variant == 3:
        # Skewed: high at left, decaying
        a = np.exp(-3.0 * x) + 0.3 * np.exp(-1.0 * x)
    elif variant == 4:
        # Power law decay
        a = (1.0 - x + 1e-6) ** 3
    elif variant == 5:
        # Twin peaks at ends
        a = np.exp(-8.0 * x**2) + np.exp(-8.0 * (x-1)**2)
    elif variant == 6:
        # Gentle asymmetric
        a = np.sin(np.pi * x * 0.8) ** 2
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    # Add small noise for diversity
    a += 0.01 * np.abs(rng.standard_normal(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False
