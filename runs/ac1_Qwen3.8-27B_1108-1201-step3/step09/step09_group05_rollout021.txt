import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(np.clip(z_shifted, -500, 500))
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""

    if _lp_solver_cached is not None:
        return _lp_solver_cached

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

                _lp_solver_cached = solver_name
                return solver_name
        except Exception:

    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:

    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:

            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, b, k_star, top_k=1, T_smooth=1.0):
    """
    Compute smoothed subgradient using top-k convolution peaks.
    Returns gradient vector g of same shape as a.
    """
    n = len(a)
    if top_k <= 1:
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]
        return g
    else:
        # Top-k weighted
        sorted_idx = np.argsort(b)[::-1][:top_k]
        weights = b[sorted_idx].copy()
        weights = weights - weights.min()
        weights = np.exp(weights / T_smooth)
        weights = weights / weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                g[i] += 2.0 * w * a[k - i]
        return g

    Works in log-space (z = log(a)), updating z -= eta * g.
    """
    n = len(a0)
    a = norm_sequence(a0)
    z = z_from_a(a).copy()
    v = np.zeros(n)  # momentum velocity
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    eta = eta_init
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        score = b.max()

            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000 and it > 1000:
            # Re-estimate step size
            eta = max(0.01, eta * 0.5)
            stall = 0

        # Compute smoothed subgradient
        k_star = int(np.argmax(b))
        g = compute_subgradient(a, b, k_star, top_k=top_k, T_smooth=T_smooth)

        # Momentum update
        v = momentum * v + (1 - momentum) * g

        # Mirror descent step in z-space
        z = z - eta * v

        # Convert back
        a_new = softmax_from_z(z)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if adapt:
            if score_new > score * 1.01:
                eta = max(0.001, eta * 0.7)
                z = z + eta * v  # undo
                a = a_new  # keep old
            else:
                a = a_new
                if score_new < score * 0.99:
                    eta = min(5.0, eta * 1.1)
        else:
            a = a_new

        # Adaptive top-k: increase smoothing as we converge
        if it % 1000 == 0 and it > 0:
            T_smooth = max(0.01, T_smooth * 0.95)

        # Periodic restart if stuck
        if stall > 20000:
            rng_local = np.random.default_rng(42 + it)
            z = z + rng_local.standard_normal(n) * 0.1
            a = softmax_from_z(z)
            v = np.zeros(n)
            eta = eta_init
            stall = 0

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Adaptive perturbation to escape plateaus
        if it % 2000 == 1999:
            b_check = fast_convolve(a)
            if b_check.max() * 2.0 * n > best_score * 1.001:
                rng_local = np.random.default_rng(777 + it)
                noise = rng_local.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, maxiter=20000, n_stages=12):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def subgradient_projected(a0, time_limit, n_max_iters=100000):
    """
    Projected subgradient method with Nesterov momentum.
    Projects onto simplex explicitly.
    """
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.05

    def proj_simplex(v):
        """Project v onto the standard simplex."""
        n_v = len(v)
        if n_v == 1:
            return np.array([max(0.0, v[0])])
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n_v + 1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * y[k_star - i]

        # Nesterov momentum
        t_curr = it + 1
        beta = (t_curr - 1) / (t_curr + 1.5)
        a_new = proj_simplex(y - eta * g)
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        a = a_new

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Adaptive step size
        if it % 500 == 499:
            if it > 100:
                eta = max(0.001, eta * 0.98)

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x)) * (1 + 0.1 * np.cos(4 * np.pi * x))
    elif mode == 6:
        # Sinc-squared (flat spectrum approximation)
        x = np.linspace(-(n // 2), n // 2 - 1, n)
        a = np.sinc(x / (n * 0.4)) ** 2
    elif mode == 7:
        # Raised cosine with different beta
        x = np.linspace(-1, 1, n)
        a = np.cos(np.pi * x / 2) ** 4
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Mirror descent (fast, novel smoothed subgradient)
    Phase 2: Nesterov LP (main workhorse, extended time)
    Phase 3: L-BFGS multi-stage annealing
    Phase 4: Projected subgradient (different geometry)
    Phase 5: Multi-length exploration
    Phase 6: Final polish iterations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Try multiple initializations and pick best
        n_init = 960
        best_init = None
        best_init_score = float('inf')
        for mode in range(8):
            a_try = structured_init(n_init, rng, mode=mode)
            sc = evaluate_safe(a_try)
            print(f"  Init mode {mode}: score={sc:.6f}")
            if sc < best_init_score:
                best_init_score = sc
                best_init = a_try
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Mirror descent (novel method, fast iterations) ===
    print(f"\n=== Phase 1: Mirror Descent n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s, top_k=3, T_smooth=0.1")
        a_md, _ = mirror_descent_refine(best_seq, t_alloc, n_max_iters=500000,
                                        top_k=3, T_smooth=0.1, eta_init=0.3,
                                        momentum=0.85, adapt=True)
        try_update(a_md, "mirror_descent_1")

        # Second round with different params
        rem = deadline - time.monotonic()
        if rem > 10:
            t_alloc2 = min(rem * 0.10, 120.0)
            if t_alloc2 > 10:
                a_md2, _ = mirror_descent_refine(best_seq, t_alloc2, n_max_iters=300000,
                                                 top_k=5, T_smooth=0.05, eta_init=0.1,
                                                 momentum=0.9, adapt=True)
                try_update(a_md2, "mirror_descent_2")

    # === Phase 2: Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 3: L-BFGS multi-stage ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 80->0.05, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=80.0, T_end=0.05,
                            maxiter=15000, n_stages=14)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Projected subgradient (different geometry) ===
    print(f"\n=== Phase 4: Projected Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ps, _ = subgradient_projected(best_seq, t_alloc, n_max_iters=200000)
        try_update(a_ps, "proj_subgrad")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration (best={best_score:.8f}) ===")
    # Try lengths around the current best
    alt_lengths = []
    base_n = n_best
    for delta in [-100, -80, -60, -40, -20, 20, 40, 60, 80, 100, 120, 150, 200]:
        alt_n = base_n + delta
        if 384 <= alt_n <= 3000 and alt_n != base_n:
            alt_lengths.append(alt_n)
    # Also try some "round" numbers
    for fixed_n in [512, 640, 768, 896, 1024, 1152, 1280, 1536, 2048]:
        if 384 <= fixed_n <= 3000 and fixed_n != base_n:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_total = min(rem * 0.15, 120.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick mirror descent
        t_md = t_total * 0.4
        a_alt, _ = mirror_descent_refine(init_alt, t_md, n_max_iters=100000,
                                         top_k=3, T_smooth=0.1, eta_init=0.3,
                                         momentum=0.85, adapt=True)
        try_update(a_alt, f"md_n{alt_n}")

        # Quick LP
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None:
            t_lp = min(rem * 0.5, t_total * 0.5)
            if t_lp > 10:
                start = interpolate_sequence(best_seq, alt_n) if len(best_seq) != alt_n else best_seq
                a_alt2, _ = nesterov_lp_refine(start, t_lp, solver=solver, n_max_iters=50000)
                try_update(a_alt2, f"lp_n{alt_n}")

        # Quick L-BFGS
        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.4, 40.0)
            if t_lb > 10:
                a_alt3 = lbgfs_refine(best_seq, t_lb, T_start=30.0, T_end=0.1,
                                      maxiter=5000, n_stages=8)
                try_update(a_alt3, f"lbgfs_n{alt_n}")

    # === Phase 6: Second Nesterov LP on best found ===
    print(f"\n=== Phase 6: Nesterov LP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, n={len(best_seq)}")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n=== Phase 7: Final L-BFGS Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 150.0)
    if t_alloc > 15:
        print(f"  L-BFGS T: 10->0.03, budget={t_alloc:.0f}s")
        a_fin = lbgfs_refine(best_seq, t_alloc, T_start=10.0, T_end=0.03,
                             maxiter=10000, n_stages=10)
        try_update(a_fin, "final_lbgfs")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_cur = len(best_seq)
        for scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)

            t_sub = min(rem * 0.4, 50.0)
            if t_sub > 10:
                a_pert, _ = nesterov_lp_refine(perturbed, t_sub, solver=solver,
                                               n_max_iters=50000)
                try_update(a_pert, f"pert_{scale}")

    # === Phase 9: Ultra-final mirror descent polish ===
    print(f"\n=== Phase 9: Final Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 60.0)
    if t_alloc > 8:
        a_fin_md, _ = mirror_descent_refine(best_seq, t_alloc, n_max_iters=200000,
                                            top_k=1, T_smooth=0.01, eta_init=0.05,
                                            momentum=0.95, adapt=True)
        try_update(a_fin_md, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
