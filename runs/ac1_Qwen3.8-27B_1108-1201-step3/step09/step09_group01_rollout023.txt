import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        r = np.nonzero(cond)[0][-1] + 1
        theta = (cssv[r-1] - 1.0) / r
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def mirror_descent_optimize(a0, time_limit, verbose=False):
    """Mirror descent with smooth max approximation."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    beta = 30.0
    lr = 0.5
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Softmax weights (concentrated on peaks)
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        Z = w.sum()
        if Z < 1e-30:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if cond.any():
        r = np.nonzero(cond)[0][-1]
        theta = (cssv[r] - 1.0) / (r + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n
