def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_lse(a, T):
    """Compute gradient of (2n/T)*logsumexp(T*conv(a,a)) w.r.t. a (on simplex)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # Grad of sum_k w[k] * conv(a,a)[k] w.r.t. a[j]:
    # = sum_k w[k] * (a[k-j] + a[j-k]) with zero padding
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]
    # The second term a[j-k] is the same by symmetry of w
    grad_a *= 2.0
    # Project to tangent space of simplex
    S = np.dot(grad_a, a)
    return a * (grad_a - S), 2.0 * n * (max_Tb / T + np.log(exp_Tb.sum()) / T)


def lbgfs_refine_v2(a0, time_limit, T_start=500.0, T_end=0.01, n_stages=15, maxiter_per_stage=3000):

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def adam_simplex_refine(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, n_stages=8):
    """Adam optimizer on the simplex with log-sum-exp smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    # Temperature schedule: start high, end low
    temps = np.geomspace(100.0, 0.05, n_stages)
    steps_per_stage = 5000
    total_steps = n_stages * steps_per_stage
    step = 0

    for si, T in enumerate(temps):
        stage_steps = steps_per_stage
        for _ in range(stage_steps):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z)
            # Compute gradient
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            step += 1
            lr_t = lr / (1.0 + step * 1e-5)
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
            z -= z.mean()  # keep on hyperplane for softmax invariance

        # Check improvement
        a_new = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

    return best_a, best_score


def peak_reduction_direct(a0, time_limit, max_iter=50000):
    """Direct peak reduction: iteratively reduce the max of conv(a,a)."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Weighted direction: reduce high contributors, add to low contributors
        # Find indices that DON'T contribute to this peak
        no_contrib = np.ones(n, dtype=bool)
        no_contrib[idx] = False
        # Also exclude indices that contribute to nearby peaks
        for dk in range(-3, 4):
            k2 = peak_k + dk
            if 0 <= k2 < len(b) and b[k2] > 0.8 * peak_val:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                no_contrib[js2:je2 + 1] = False

        # Redistribute from top contributors to non-contributors
        top_count = min(10, len(idx))
        top_local = np.argsort(contribs)[::-1][:top_count]
        mass_to_move = 0.0
        moved = np.zeros(n)
        for ci in top_local:
            i = idx[ci]
            frac = 0.15  # reduce by 15%
            mass_to_move += a[i] * frac
            moved[i] -= a[i] * frac

        # Where to add
        add_mask = no_contrib & (a > 1e-10)
        if add_mask.sum() == 0:
            add_mask = no_contrib
        if add_mask.sum() == 0:
            # All indices contribute somewhere; use lowest contribution
            contrib_to_peak = np.zeros(n)
            contrib_to_peak[idx] = contribs
            add_mask = np.argsort(contrib_to_peak)[:(n // 3) + 1] < n
            add_mask = np.zeros(n, dtype=bool)
            add_mask[np.argsort(contrib_to_peak)[:(n // 3) + 1]] = True

        add_amounts = a[add_mask] + 1e-10
        moved[add_mask] += mass_to_move * add_amounts / add_amounts.sum()

        a_trial = a + moved
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-12:
            a = a_trial
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturb and reset
                rng_local = np.random.default_rng(42 + it)
                a = a + 0.02 * rng_local.standard_normal(n)
                a = norm_sequence(a)
                no_improve = 0

    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):

    b_val = np.asarray(b_var.value, dtype=np.float64)

def nesterov_lp_v2(a0, time_limit, solver=None, max_iters=80000):

def spectral_init(n, rng):
    """Initialize by making the power spectrum as flat as possible, then project to non-negative."""
    # Start with a random complex spectrum with conjugate symmetry
    N = 1 << (n - 1).bit_length()
    spec = rng.standard_normal(N // 2 + 1) + 1j * rng.standard_normal(N // 2 + 1)
    # Force conjugate symmetry
    spec_full = np.zeros(N, dtype=complex)
    spec_full[:N // 2 + 1] = spec
    spec_full[N // 2 + 1:] = np.conj(spec[:N // 2][::-1])
    # Flatten the magnitude
    mag = np.abs(spec_full)
    mag = mag ** 0.3  # compress dynamic range
    spec_flat = mag * np.exp(1j * np.angle(spec_full))
    # Inverse DFT
    a_complex = irfft(spec_flat, N)[:n]
    a = np.abs(a_complex)  # take magnitude to get non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    f_norm = f / f.sum()
    old_n = len(f_norm)
    if old_n == new_n:
        return f_norm.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                a_trial[j] += delta
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if found:
                break

        if not found:
            no_improve += 1
            if no_improve > 1000:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining L-BFGS annealing, Adam, LP,
    peak reduction, coordinate descent, and multi-length exploration.
    """
    global _start_time
    _start_time = time.monotonic()
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Try multiple initializations and pick the best
        n_init = 768
        best_init = None
        best_init_score = float('inf')
        for mode in range(8):
            a_init = structured_init(n_init, rng, mode=mode)
            sc = evaluate_safe(a_init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a_init
        a_spec = spectral_init(n_init, rng)
        sc_spec = evaluate_safe(a_spec)
        if sc_spec < best_init_score:
            best_init_score = sc_spec
            best_init = a_spec
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: L-BFGS coarse (T: 500 -> 5) ===
    print(f"\n=== Phase 1: L-BFGS Coarse n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 360.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 500->5")
        a_out = lbgfs_refine_v2(best_seq, t_alloc, T_start=500.0, T_end=5.0, n_stages=10, maxiter_per_stage=4000)
        try_update(a_out, "lbgfs_coarse")

    # === Phase 2: L-BFGS fine (T: 5 -> 0.01) ===
    print(f"\n=== Phase 2: L-BFGS Fine n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 5->0.01")
        a_out = lbgfs_refine_v2(best_seq, t_alloc, T_start=5.0, T_end=0.01, n_stages=12, maxiter_per_stage=5000)
        try_update(a_out, "lbgfs_fine")

    # === Phase 3: Nesterov LP ===
    print(f"\n=== Phase 3: Nesterov LP n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = nesterov_lp_v2(best_seq, t_alloc, solver=solver, max_iters=60000)
        try_update(a_out, "nesterov_lp")

    # === Phase 4: Peak reduction ===
    print(f"\n=== Phase 4: Peak Reduction n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = peak_reduction_direct(best_seq, t_alloc, max_iter=80000)
        try_update(a_out, "peak_reduction")

    # === Phase 5: Adam refine ===
    print(f"\n=== Phase 5: Adam n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = adam_simplex_refine(best_seq, t_alloc, lr=0.005, n_stages=10)
        try_update(a_out, "adam")

    # === Phase 6: Frank-Wolfe ===
    print(f"\n=== Phase 6: Frank-Wolfe n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 90.0)
    if t_alloc > 10:
        a_out, _ = frank_wolfe_away(best_seq, t_alloc, max_steps=50000)
        try_update(a_out, "fw_away")

    # === Phase 7: L-BFGS ultra-low-T polish ===
    print(f"\n=== Phase 7: L-BFGS Polish n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s, T: 2->0.005")
        a_out = lbgfs_refine_v2(best_seq, t_alloc, T_start=2.0, T_end=0.005, n_stages=10, maxiter_per_stage=6000)
        try_update(a_out, "lbgfs_polish")

    # === Phase 8: Coordinate descent ===
    print(f"\n=== Phase 8: Coord Descent n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 90.0)
    if t_alloc > 5:
        a_out, _ = coordinate_descent_refine(best_seq, t_alloc, max_rounds=30000)
        try_update(a_out, "coord_descent")

    # === Phase 9: Multi-length exploration ===
    print(f"\n=== Phase 9: Multi-Length (best={best_score:.6f}) ===")
    n_cur = len(best_seq)
    alt_lengths = []
    for alt_n in [512, 640, 768, 896, 960, 1024, 1152, 1280, 1408, 1536, 1792, 2048]:
        if abs(alt_n - n_cur) > 32:
            alt_lengths.append(alt_n)
    alt_lengths.sort(key=lambda x: abs(x - n_cur))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.20, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick L-BFGS
        t_lb = t_total * 0.5
        a_alt = lbgfs_refine_v2(init_alt, t_lb, T_start=50.0, T_end=0.1, n_stages=8, maxiter_per_stage=3000)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        # Peak reduction
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_pr = min(rem * 0.3, 50.0)
            if t_pr > 5:
                a_alt2, _ = peak_reduction_direct(best_seq, t_pr, max_iter=20000)
                try_update(a_alt2, f"peak_n{alt_n}")

    # === Phase 10: Stochastic restarts ===
    print(f"\n=== Phase 10: Stochastic Restarts (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            scale = [0.03, 0.08, 0.15][trial]
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)

            t_sub = min(rem * 0.3, 100.0)
            if t_sub < 20:
                break
            print(f"  Perturb scale={scale}, budget={t_sub:.0f}s")

            # Quick L-BFGS on perturbed
            a_pert = lbgfs_refine_v2(perturbed, t_sub, T_start=20.0, T_end=0.1, n_stages=8, maxiter_per_stage=3000)
            try_update(a_pert, f"perturb_{scale}")

    # === Phase 11: Final polish ===
    print(f"\n=== Phase 11: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.5, 60.0)
        print(f"  L-BFGS T:1->0.005, {t_fin:.0f}s")
        a_fin = lbgfs_refine_v2(best_seq, t_fin, T_start=1.0, T_end=0.005, n_stages=8, maxiter_per_stage=5000)
        try_update(a_fin, "final_polish")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_pr = min(rem * 0.4, 30.0)
        a_pr, _ = peak_reduction_direct(best_seq, t_pr, max_iter=20000)
        try_update(a_pr, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
