Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

3. LogSumExp L-BFGS with multi-stage temperature annealing
4. Greedy coordinate descent on peak-contributing elements
5. Final polynomial-time local refinement

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Fast local evaluation (normalized a, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_reduced_conv_matrix(a, b, K):
    """Build reduced (2n-1) x n matrix, keeping only top K rows by current b values."""
    n = len(a)
    ab = fast_convolve(a)  # current convolution with a fixed
    # Actually we need T such that T @ b = convolve(a, b)
    # T[i,j] = a[j] if 0 <= i-j <= n-1, else 0
    # So T[i,:] = a[max(0,i-n+1):min(n, i+1)] at position min(n, i+1)-max(0,i-n+1)
    
    # For reduced version, pick top K rows of |ab|
    top_idx = np.argsort(ab)[::-1][:K]
    top_idx = np.sort(top_idx)
    
    size = 2 * n - 1
    T_red = np.zeros((K, n))
    for idx_i, i in enumerate(top_idx):
        js = max(0, i - n + 1)
        je = min(n, i + 1)
        T_red[idx_i, js:je] = a[js:je]
    return T_red, top_idx

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solved = False
    try:

        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True

        pass

                    break
            except Exception:

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    return b_val / s


def solve_full_lp(a, solver):

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses reduced-LP mirror descent, full-LP Nesterov acceleration, LogSumExp L-BFGS,
and targeted coordinate descent.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


# ============================================================
# Helper functions (all top-level, no closures)
# ============================================================

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def _get_lp_solver():
    for name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(20, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            continue
    return cp.ECOS


def build_full_conv_matrix(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows, cols, vals = [], [], []
    for j in range(n):
        if a[j] > 1e-15:
            for i in range(j, j + n):
                rows.append(i)
                cols.append(j)
                vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def build_reduced_conv_matrix(a, conv_a, K):
    """Build reduced matrix using top K rows by current convolution values."""
    n = len(a)
    size = 2 * n - 1
    top_idx = np.argsort(conv_a)[::-1][:K]
    top_idx = np.sort(top_idx)
    T = np.zeros((K, n))
    for idx_i, i in enumerate(top_idx):
        js = max(0, i - n + 1)
        je = min(n, i + 1)
        T[idx_i, js:je] = a[js:je]
    return T


def solve_lp_reduced(a, conv_a, K, solver):
    """Solve reduced LP: min M s.t. T_red @ b <= M, sum(b)=1, b>=0."""
    n = len(a)
    T_red = build_reduced_conv_matrix(a, conv_a, K)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_red @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)

            s = b_val.sum()

                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:

    return None
