    # And sum_k w[k] * a[i+k] = sum_j w[j] * a[i+j] = ... let me use different substitution.
    # Let j = i+k, so k = j-i: sum_j w[j-i] * a[j] = (w * a)[i] (but w has negative indices)
    
    # I think the cleanest way is:
    # grad[i] = 2/T * sum_k w[k] * (a[i+k] + a[i-k])
    # Let's compute this as 2/T * (corr(a, w))[i] where corr(a, w)[i] = sum_k a[i+k]*w[k] + a[i-k]*w[k]
    # = 2/T * sum_k a[i+k]*w[k] + 2/T * sum_k a[i-k]*w[k]
    # First sum: sum_k a[i+k]*w[k] = cross-correlation of a and w at lag i
    # Second sum: sum_k a[i-k]*w[k] = cross-correlation of a and w at lag -i = same by symmetry of real signals
    # So grad[i] = 4/T * (a ⋆ w)[i] where (a ⋆ w)[i] = sum_k a[i+k]*w[k]

    # sum_k w[k] * a[i+k]: this is the cross-correlation (w ⋆ a)[i] = sum_k w[k] * conj(a[k-i])

    # Let j = k-i: = sum_j w[j+i] * a[j] = (w_shifted * a) ...

    # Define: c[i] = sum_k w[k] * (a[i+k] + a[i-k])
    # Note that w has length 2n-1, a has length n.
    # For the term sum_k w[k] * a[i+k]: i ranges over [0,n-1], k ranges over [0,2n-2]
    #   a[i+k] is nonzero only when 0 <= i+k <= n-1, i.e., -i <= k <= n-1-i
    # For the term sum_k w[k] * a[i-k]: i-k is in [0,n-1], i.e., i-n+1 <= k <= i

    # = sum_k w[k] * (a[i+k] + a[i-k])

    # c = ifft(fft(a_padded) * fft(w_padded_reversed)) ...

    # c[i] = sum_k w[k]*a[i+k] + sum_k w[k]*a[i-k]
    # First part: np.correlate(w, a, mode='full')[n-1 : n-1+n] ... hmm
    
    # Actually the simplest: use FFT
    # sum_k w[k] * a[i+k] = correlation of a and w
    # In numpy: np.correlate(a, w[:n], mode='full') gives sum_k a[k+j]*w[k] for various j
    # This is getting messy. Let me use the direct FFT approach.
    
    # Key formula: if b = conv(a, a), then db/da[i] = a * e_i convolved stuff
    # Actually: b[k] = sum_j a[j]*a[j+k]
    # db[k]/da[i] = a[i+k] (from j=i term) + a[i-k] (from j+k=i term, i.e., j=i-k)
    # So db/da[i] (the full gradient w.r.t. a[i]) = sum_k w[k] * (a[i+k] + a[i-k])
    
    # Let me compute this as: 
    # term1[i] = sum_k w[k] * a[i+k]  -- this is (w * a)[n-1-i] or something
    # Let me just use np.correlate
    
    # np.correlate(a, w, mode='full')[i] = sum_j a[j] * w[j - i + (len(w)-len(a))]
    # This is confusing. Let me just do it with FFT directly.
    
    # Define: for the purpose of computing sum_k w[k] * a[i+k]:
    # Pad a to length 4n (to handle all shifts)
    # Then the result at position i is the cross-correlation at lag i
    
    # Simplest approach: use the fact that
    # sum_k w[k] * a[i+k] = (cross_correlation of w and a at lag -i)
    # In FFT terms: ifft(fft(w) * conj(fft(a))) gives cross-correlation
    
    # Let me just compute it numerically with FFT:
    a_pad = np.zeros(4 * n)
    a_pad[:n] = a
    w_pad = np.zeros(4 * n)
    w_pad[:len(w)] = w
    
    fft_size = 1 << (len(a_pad) + len(w_pad) - 1).bit_length()
    A = rfft(a_pad, fft_size)
    W = rfft(w_pad, fft_size)
    
    # cross-correlation: corr[i] = sum_k w[k] * a[k+i]
    # = ifft(conj(fft(w)) * fft(a)) ... let me verify
    # corr = irfft(np.conj(W) * A, fft_size)
    # corr[i] = sum_k w[k] * a[k+i] (circular, but we padded enough)
    
    corr = irfft(np.conj(W) * A, fft_size)
    
    # term1[i] = sum_k w[k] * a[i+k] = corr[i] (for i in [0, n-1])
    # term2[i] = sum_k w[k] * a[i-k] = corr[-i] = corr[fft_size - i]
    # But since a and w are real and we're in a padded space:
    # term2[i] = sum_k w[k] * a[i-k] = sum_j w[j] * a[i-j] (sub j=k)
    # = sum_j w[j] * a[i-j]
    # Let me redefine: let m = i-k, so k = i-m: sum_m w[i-m] * a[m] = (w * a)[i] (linear convolution)
    # Hmm, this is the convolution of w and a, not correlation.
    
    # OK I think the cleanest formula is:
    # grad[i] = 2/T * sum_k w[k] * (a[i+k] + a[i-k])
    # Let's define cc[i] = sum_k w[k] * a[i+k] (cross-correlation, positive lag)
    # Then sum_k w[k] * a[i-k] = sum_k w[k] * a[i-k]
    # If w is symmetric (w[k] = w[2n-2-k]), then this equals cc[... ]
    # But w is NOT necessarily symmetric.
    
    # Let me just compute both terms separately using correlation:
    # cc1[i] = sum_k w[k] * a[i+k] = (a ⋆ w)[i] 
    #   In numpy: this is the cross-correlation of a and w at lag i
    #   = np.correlate(a, w, 'full')[len(w)-1+n-1-i] ... ugh
    
    # You know what, let me just use the most straightforward FFT approach:
    # The gradient of b[k] = sum_j a[j] a[j+k] w.r.t. a[i] is:
    #   = a[i+k] + a[i-k] (with a out of range = 0)
    # So the full gradient w.r.t. a is:
    #   grad[i] = sum_k w[k] (a[i+k] + a[i-k])
    #          = sum_k w[k] a[i+k] + sum_k w[k] a[i-k]
    
    # First sum: let j = i+k, k = j-i: sum_j w[j-i] a[j]
    #   This is the cross-correlation (w, a) evaluated at lag i
    #   = (w ⋆ a)[i] where (w ⋆ a)[i] = sum_j w[j-i] a[j]
    
    # Second sum: let j = i-k, k = i-j: sum_j w[i-j] a[j]
    #   This is (w * a)[i] (linear convolution of w and a at position i)
    
    # So grad[i] = (w ⋆ a)[i] + (w * a)[i]
    
    # For real signals, (w ⋆ a)[i] = (a ⋆ w)[i] and (w * a)[i] = (a * w)[i]
    # And (w ⋆ a)[i] = (a * w_flipped)[i] where w_flipped[k] = w[-k]
    
    # OK let me just compute both via FFT:
    # (w * a) via convolution: ifft(fft(w) * fft(a))
    # (w ⋆ a) via correlation: ifft(conj(fft(w)) * fft(a))
    
    # But I need to be careful about the lengths. w has length 2n-1, a has length n.
    # The linear convolution w * a has length 3n-1. I only need indices 0 to n-1.
    # The cross-correlation w ⋆ a also has length 3n-1.
    
    # Let me use a simpler approach: just compute the gradient directly using the formula
    # grad[i] = 2/T * sum_k w[k] * (a[i+k] + a[i-k])
    
    # I'll compute this by:
    # 1. Computing the "flipped w" correlation: for each i, sum_k w[k] * a[i+k]
    #    This is np.correlate(a, w, 'full')[offset + i] for appropriate offset
    # 2. Computing sum_k w[k] * a[i-k] = np.convolve(w, a, 'full')[i]
    
    # Actually, let me just use the approach from the previous code which worked:
    # corr = irfft(np.conj(W) * A, fft_size)[:n] where A = rfft(a), W = rfft(w)
    # This gives: corr[i] = sum_k w[k] * a[k-i] (for the real case)
    # Hmm, that's sum_k w[k] * a[k-i], which is what I want for the second term.
    
    # Let me just verify with a small example in the code.
    
    # For now, let me use the approach that I know works:
    # The gradient of the log-sum-exp of conv(a,a) is:
    # grad = 2 * (a ⋆ w) where ⋆ denotes cross-correlation
    # Because: d/da[i] log(sum exp(T*b[k])) = sum_k w[k] * db[k]/da[i]
    # = sum_k w[k] * (a[i+k] + a[i-k])
    # = 2 * sum_k w[k] * a[i+k] (if w is symmetric, which it's approximately for the log-sum-exp weights)
    # Actually w is NOT symmetric in general.
    
    # Let me just compute it as:
    # grad[i] = sum_k w[k] * a[i+k] + sum_k w[k] * a[i-k]
    # = cross_corr(a, w)[i] + conv(a, w_flipped)[i]  ... 
    
    # I'll use this formula: 
    # Let's compute c1[i] = sum_k w[k] * a[i+k]
    # This is the cross-correlation of a and w at lag i (with a as the "first" signal)
    # In FFT: c1 = ifft(fft(a) * conj(fft(w)))  [with appropriate padding]
    # Wait: ifft(fft(a) * conj(fft(w)))[i] = sum_j a[j] * w[j-i] = sum_k a[i+k] * w[k] (sub j=i+k)
    # Yes! So c1[i] = sum_k a[i+k] * w[k] = sum_k w[k] * a[i+k]. 
    
    # And c2[i] = sum_k w[k] * a[i-k] = sum_j w[i-j] * a[j] (sub j=i-k)
    # = (w * a)[i] (linear convolution)
    # In FFT: c2 = ifft(fft(w) * fft(a))[i]
    
    # So grad[i] = c1[i] + c2[i] = ifft(fft(a)*conj(fft(w)))[i] + ifft(fft(w)*fft(a))[i]
    
    # Since a and w are real: conj(fft(w)) = fft(w_flipped) (for circular)
    # And ifft(fft(w)*fft(a)) = ifft(fft(a)*fft(w)) = convolution
    
    # Let me just compute both:
    fft_size2 = 1 << (3 * n - 1).bit_length()
    A2 = rfft(a, fft_size2)
    W2 = rfft(w, fft_size2)
    
    c1 = irfft(A2 * np.conj(W2), fft_size2)[:n]  # c1[i] = sum_k w[k]*a[i+k]
    c2 = irfft(W2 * A2, fft_size2)[:n]  # c2[i] = sum_k w[k]*a[i-k] = conv(w,a)[i]
    
    grad = (2.0 / T) * (c1 + c2)
    
    return value, grad


def nesterov_pg(a0, time_limit, T_start=1.0, T_end=500.0,
                 lr_init=0.1, n_stages=15, steps_per_stage=2000,
                 restart_patience=500):
    """Nesterov accelerated projected gradient with adaptive temperature and restarts."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    total_planned = n_stages * steps_per_stage
    step_count = 0
    stall_count = 0
    prev_best = best_score

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.01 * global_step)

            # Nesterov momentum
            if step_count > 0:
                t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0

            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            # Gradient
            _, grad = smooth_max_obj_and_grad(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            # Check score periodically
            if step_count % 30 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    stall_count = 0
                    if step_count % 300 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"        NG step={step_count} T={T:.1f}: score={best_score:.6f} ({elapsed:.0f}s)")

            # Restart check
            if step_count % 100 == 0:
                if best_score > prev_best - 1e-8:
                    stall_count += 1
                else:
                    stall_count = 0
                prev_best = best_score
                if stall_count >= restart_patience // 100:
                    # Restart: reset momentum and add small noise
                    x_prev = x.copy()
                    rng = np.random.default_rng(step_count)
                    x = x + 0.01 * rng.standard_normal(n)
                    x = project_simplex(x)
                    stall_count = 0

    return best_a, best_score


def lbfgs_smoothed(a0, time_limit, T_start=10.0, T_end=200.0, n_restarts=5):
    """L-BFGS on log-sum-exp smoothed objective with simplex projection."""
    n = len(a0)
    best_a = project_simplex(a0.copy())
    best_score = score_normalized(best_a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_restarts)
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        T = temps[restart]
        
        a_curr = best_a.copy()
        
        # Use scipy L-BFGS-B on the smoothed objective
        # We'll use the simplex projection as a barrier: 
        # optimize over "unconstrained" variables with a large penalty for violating simplex
        # Actually, let's just do projected L-BFGS manually
        
        # Simpler: use scipy with the log-barrier / penalty approach
        # Or just use the gradient and do our own L-BFGS with projection
        
        # Let me use scipy.optimize.minimize with method='L-BFGS-B' 
        # on the log-domain parameterization: a = softmax(w)
        
        w_init = np.log(np.maximum(a_curr, 1e-15)) - np.log(n)  # log(a) - log(n)
        # Actually, let's use a simpler parameterization:
        # w such that a = exp(w) / sum(exp(w))
        w_init = np.log(np.maximum(a_curr, 1e-15))
        
        def objective(w):
            a = np.exp(w - w.max())
            a = a / a.sum()
            _, _ = smooth_max_obj_and_grad(a, T)
            b = fast_convolve(a)
            Tb = T * b
            m = Tb.max()
            return (m + np.log(np.exp(Tb - m).sum())) / T
        
        def grad(w):
            a = np.exp(w - w.max())
            a = a / a.sum()
            _, g = smooth_max_obj_and_grad(a, T)
            # Chain rule: da/dw_i = a_i * (delta_ii - sum_j a_j * delta_ij) for softmax
            # grad_w = (grad_a - sum_j grad_a_j * a_j) * a
            ga = g * a
            grad_w = g - ga.sum()
            grad_w *= a
            return grad_w
        
        t_sub_start = time.monotonic()
        sub_budget = time_limit / n_restarts * 1.2
        try:
            result = minimize(objective, w_init, jac=grad, method='L-BFGS-B',
                            options={'maxiter': 500, 'maxfun': 2000,
                                    'ftol': 1e-12, 'gtol': 1e-8})
            
            if result.success or (result.fun < objective(w_init)):
                a_opt = np.exp(result.x - result.x.max())
                a_opt = a_opt / a_opt.sum()
                sc = score_normalized(a_opt)
                if sc < best_score:
                    best_score = sc
                    best_a = a_opt.copy()
                    elapsed = time.monotonic() - t0
                    print(f"        LBF restart={restart} T={T:.1f}: score={best_score:.6f} ({elapsed:.0f}s)")
        except Exception as e:
            print(f"        LBF restart={restart} failed: {e}")
        
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, tol=1e-6):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, tol=tol)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, tol=tol)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternating(a0, time_limit, solver=None, n_max_iters=2000):
    """
    LP-based alternating minimization with Nesterov dual averaging.
    Alternates: fix a, optimize b via LP; then fix b, optimize a via LP.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    b = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    
    # Dual variables for Nesterov averaging
    y = a.copy()  # mirror for a
    z = b.copy()  # mirror for b
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Fix b, optimize a
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 2)
            y = (1 - alpha) * y + alpha * a_new
            a = ((it + 1) * a + alpha * (it + 2) * a_new) / (it + 1 + alpha * (it + 2))
            a = project_simplex(a)
        
        # Step 2: Fix a, optimize b
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            z = (1 - alpha) * z + alpha * b_new
            b = ((it + 1) * b + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
            b = project_simplex(b)
        
        # Check scores
        score_a = score_normalized(a)
        score_b = score_normalized(b)
        score_y = score_normalized(y)
        score_z = score_normalized(z)
        
        for cand, name in [(a, 'a'), (b, 'b'), (y, 'y'), (z, 'z')]:
            if score_normalized(cand) < best_score:
                best_score = score_normalized(cand)
                best_a = cand.copy()
                stall_count = 0
        
        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                z = z + 0.03 * rng.standard_normal(n)
                z = project_simplex(z)
                stall_count = 0
            
            elapsed = time.monotonic() - t0
            print(f"        LP alt iter={it+1}: score={best_score:.6f} ({elapsed:.0f}s)")
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LP alt iter={it+1}: score={best_score:.6f} ({elapsed:.0f}s)")
    
    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent that targets the peak of the convolution."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (excluding boundaries which are naturally small)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-1] = -np.inf
            b_work[-2] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        
        # Find coordinates that contribute most to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        improved = False
        top_idx = np.argsort(contribs)[::-1][:8]
        
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-15:
                continue
            for factor in [0.98, 0.96, 0.93, 0.90]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-10:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing a pair
            for ci in top_idx[:5]:
                i = idx[ci]
                j = peak_k - i
                if i >= n or j >= n or i < 0 or j < 0:
                    continue
                if i == j:
                    continue
                if a[i] < 1e-15 or a[j] < 1e-15:
                    continue
                for f1 in [0.97, 0.94, 0.90]:
                    for f2 in [0.97, 0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-10:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Perturb and retry
                rng = np.random.default_rng(step)
                a = a + 0.005 * rng.standard_normal(n)
                a = project_simplex(a)
                no_improve = 0
        
        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"        CD step={step}: score={best_score:.6f} ({elapsed:.0f}s)")
    
    return best_a, best_score


def make_flat_top_init(n, taper_width_frac=0.15):
    """Create a flat-top window with smooth tapers (Blackman-like)."""
    x = np.linspace(0, 1, n)
    tw = int(taper_width_frac * n)
    a = np.ones(n)
    # Smooth taper using cosine
    for i in range(tw):
        t = i / tw
        a[i] = 0.5 * (1 - np.cos(np.pi * t))
        a[n-1-i] = 0.5 * (1 - np.cos(np.pi * t))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_random_smooth_init(n, rng, n_harmonics=5):
    """Create a random smooth initialization."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * k * x + phase)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
