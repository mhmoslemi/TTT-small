            # Max of unaffected portion
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def obj(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                    _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                    _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = max(3.0 * v, 5.0 / n)
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = obj(c_pt)
            fd = obj(d_pt)
            for _ in range(30):
                if hi - lo < 1e-13 * max(1.0, lo):
                    break
                if fc <= fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = obj(c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = obj(d_pt)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

        if verbose and no_improve == 0:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 30 < 1:
                print(f"    CD {label} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def subgradient_descent(a0, time_limit, n_peaks=20, verbose=False):
    """Subgradient descent with adaptive step size and peak weighting."""
    n = len(a0)
    a = normalize(a0).copy()
    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Get top peaks
        k_act = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        a_new = simplex_project(a - lr * grad)
        if np.allclose(a_new, a, atol=1e-15):
            no_improve += 1
        else:
            conv = fast_convolve(a_new)
            sc = 2.0 * n * float(conv.max())
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr *= 0.999

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG ({elapsed:.0f}s) it={it}: best={best_score:.10f}")

    return best_a, best_score


def lp_best_response(a_fixed, time_budget=5.0):
    """Solve LP to find b minimizing max_k conv(a,b)[k] subject to b>=0, sum(b)=1."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix T where T @ b = conv(a_fixed, b)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solver_list = [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK, cp.CBC]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, max_time=time_budget)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                return b_val / s
        except Exception:
            continue
    return None


def fictitious_play(a0, time_limit, verbose=False):
    """LP-based fictitious play with line search."""
    n = len(a0)
    a = normalize(a0).copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        iters += 1
        b = lp_best_response(a, time_budget=min(rem * 0.3, 5.0))
        if b is None:
            continue

        # Line search: a_new = (1-t)*a + t*b
        best_t = 0.0
        best_ls = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            cand = normalize((1.0 - t) * a + t * b)
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls:
                best_ls = sc
                best_t = t

        if best_t > 0:
            a_new = normalize((1.0 - best_t) * a + best_t * b)
            a = a_new
            sc = 2.0 * n * float(fast_convolve(a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # Fallback: averaging with decaying weight
            alpha = 2.0 / (iters + 2.0)
            a = normalize((1.0 - alpha) * a + alpha * b)

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def basin_hopping(a0, time_limit, n_perturbs=6, verbose=False):
    """Basin hopping with coordinate descent refinement."""
    n = len(a0)
    a = normalize(a0).copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for p in range(n_perturbs):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.4, 40.0)

        # Different perturbation strategies
        strategy = p % 3
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce values at positions contributing to peaks
            conv_p = fast_convolve(a)
            top_k = min(5, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                perturbed[picks] *= (1.0 + rng.standard_normal() * 0.08)
            perturbed = a + (perturbed - a) * 0.5
        else:
            # Structured: multiply by a smooth random function
            smooth = np.exp(-0.5 * ((np.arange(n) - n * rng.random()) / (n * 0.3)) ** 2)
            perturbed = a * (0.8 + 0.4 * smooth)

        perturbed = np.maximum(perturbed, 0.0)
        ps = perturbed.sum()
        if ps < 1e-15:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = coord_descent(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Key insight: theoretical minimum is 2 (uniform sequence).
    Strategy: verify, then try to beat parent if somehow possible.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    def check_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_eval = a / s  # normalize for evaluation
        # Scale to avoid any clipping issues
        a_eval = np.clip(a_eval, 0, 1000.0)
        score = float(evaluate_sequence(a_eval.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_eval.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_eval)})")

    # === Step 1: Evaluate parent ===
    print("=== Step 1: Evaluate parent ===")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            check_update(parent, "parent")
            print(f"  Parent n={len(parent)}, score={best_score:.10f}")
    except Exception as e:
        print(f"  Could not load parent: {e}")

    # === Step 2: Try uniform sequences (theoretical optimum) ===
    print("\n=== Step 2: Uniform sequences ===")
    for n_uni in [64, 128, 256, 512, 1024, 2048]:
        uni = np.ones(n_uni) / n_uni
        check_update(uni, f"uniform_n{n_uni}")

    # If uniform gives score 2.0 and that's our best, we're at the theoretical minimum
    print(f"  After uniform: best={best_score:.10f}")

    # === Step 3: If parent is better than uniform (shouldn't happen, but check) ===
    if best_seq is not None and best_score < 1.999:
        print("  Parent beats uniform! Optimizing further...")
        n_best = len(best_seq)

        # Intensive coordinate descent
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.4, 300.0)
            print(f"\n=== Step 3a: Intensive CD, budget={t_alloc:.0f}s ===")
            a_opt, s_opt = coord_descent(best_seq, t_alloc, verbose=True, label="intensive")
            check_update(a_opt, "intensive_cd")

        # LP fictitious play
        rem = deadline - time.monotonic() - 100
        if rem > 30 and best_score < 1.999:
            t_fp = min(rem * 0.4, 200.0)
            print(f"\n=== Step 3b: FP, budget={t_fp:.0f}s ===")
            a_fp, s_fp = fictitious_play(best_seq, t_fp, verbose=True)
            check_update(a_fp, "fp")

            # CD after FP
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_cd, s_cd = coord_descent(a_fp, min(rem * 0.5, 60.0))
                check_update(a_cd, "cd_after_fp")

    else:
        # Score is at or near theoretical minimum of 2
        # Try to see if any non-uniform structure helps (it shouldn't mathematically)
        print("\n=== Step 3: Exploration (near theoretical min) ===")

        # Try various non-trivial initializations and short CD
        n_try = len(best_seq) if best_seq is not None else 1024
        inits = [
            ("flat_spec", _init_flat_spectrum(n_try)),
            ("gaussian", _init_gaussian(n_try)),
            ("random_smooth", _init_random_smooth(n_try)),
        ]

        if parent is not None:
            inits.append(("parent_interp", _interpolate(parent, n_try)))

        for itype, init in inits:
            rem = deadline - time.monotonic() - 100
            if rem < 10:
                break
            t_short = min(rem * 0.15, 30.0)
            a_opt, s_opt = coord_descent(init, t_short)
            check_update(a_opt, f"explore_{itype}")

        # Try different lengths
        for n_alt in [512, 768, 1024, 1536, 2048]:
            if n_alt == n_try:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            uni_alt = np.ones(n_alt) / n_alt
            check_update(uni_alt, f"uniform_alt_n{n_alt}")

    # === Step 4: Final basin hopping attempt ===
    rem = deadline - time.monotonic() - 30
    if rem > 20 and best_seq is not None:
        print(f"\n=== Step 4: Basin hopping, budget={rem * 0.5:.0f}s ===")
        t_bh = min(rem * 0.5, 120.0)
        a_bh, s_bh = basin_hopping(best_seq, t_bh, n_perturbs=4, verbose=True)
        check_update(a_bh, "basin_hopping")

    # === Step 5: Final polish ===
    rem = deadline - time.monotonic() - 10
    if rem > 5 and best_seq is not None:
        print(f"\n=== Step 5: Final polish ===")
        a_final, s_final = coord_descent(best_seq, min(rem * 0.8, 30.0))
        check_update(a_final, "final_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def _init_flat_spectrum(n):
    """Flat power spectrum initialization."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    rng = np.random.default_rng(7)
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    a = irfft(np.exp(1j * phases), n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _init_gaussian(n):
    """Gaussian bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * 0.3
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    return a / s


def _init_random_smooth(n):
    """Random smooth positive function."""
    rng = np.random.default_rng(99)
    n_harm = 5
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n) * 0.5
    for j in range(n_harm):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _interpolate(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
