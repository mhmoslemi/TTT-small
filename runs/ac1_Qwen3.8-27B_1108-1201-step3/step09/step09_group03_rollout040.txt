Key innovation: exact 1D subproblem solution via water-filling formula.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def exact_cd_sweep(a, conv, priority_order=None):
    """
    One sweep of exact 1D coordinate descent using water-filling.
    For each coordinate i, computes the exact optimal delta that minimizes
    max(unaff_max, max_affected(C + 2*delta*v)).
    Returns updated a, conv, M, and number of improvements.
    """
    n = len(a)
    M = conv.max()
    n_improvements = 0

    if priority_order is None:
        order = list(range(n))
    else:
        order = priority_order

    for i in order:
        if a[i] < 1e-15:
            continue

        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        length = k_end - k_start + 1

        # Unaffected max
        unaff_max = 0.0
        if k_start > 0:
            unaff_max = max(unaff_max, float(conv[:k_start].max()))
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

        # Affected range: C[k] + 2*delta*a[k-i] for k in [k_start, k_end]
        v = a[:length]  # a[k-i] for k in [k_start, k_end] = a[0:length]
        affected = conv[k_start:k_end + 1]

        # Check if any affected value exceeds unaff_max
        M_aff = float(affected.max())
        if M_aff <= unaff_max + 1e-15:
            continue  # No improvement possible by decreasing a[i]

        # Water-filling: find delta* = min_k (unaff_max - C[k]) / (2*v[k]) for v[k] > 0
        mask = v > 1e-15
        if not mask.any():
            continue

        ratios = (unaff_max - affected[mask]) / (2.0 * v[mask])
        delta_star = float(ratios.min())

        # Clamp to non-negativity
        if delta_star < -a[i]:
            delta_star = -a[i]

        if delta_star >= -1e-15:
            continue  # No improvement

        old_ai = a[i]
        a[i] = old_ai + delta_star

        # Update convolution incrementally
        changes = 2.0 * delta_star * v
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            changes = changes.copy()
            changes[idx] += delta_star * delta_star
        conv[k_start:k_end + 1] += changes

        new_M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
        if new_M < M:
            M = new_M
            n_improvements += 1

    return a, conv, M, n_improvements

    """
    Main exact coordinate descent with adaptive behavior.
    Uses priority ordering based on peak locations.

    conv = fast_convolve(a)

    total_sweeps = 0

        peak_k = int(np.argmax(conv))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)

def fw_best_response(a_fixed, num_steps=200):
    """
    FW method to approximate the LP best response:
    min_b max_k (a_fixed * b)_k s.t. b >= 0, sum(b) = 1
    Returns the approximate optimal b.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    b_conv = convolve_ab(a_fixed, b)

    for step in range(num_steps):
        t = 2.0 / (step + 2.0)

        # Find peak of current convolution
        k_star = int(np.argmax(b_conv))

        # Gradient: grad[j] = a_fixed[k_star - j] for valid j
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = np.zeros(n)
        # a_fixed[k_star - j] for j in [i_start, i_end]
        # k_star - j ranges from k_star - i_start to k_star - i_end
        grad[i_start:i_end + 1] = a_fixed[k_star - i_end:k_star - i_start + 1][::-1]

        # FW step: move towards argmin of gradient
        j_star = int(np.argmin(grad))

        # Away step
        if step >= 50:
            away_scores = b * grad
            i_away = int(np.argmax(away_scores))
            if b[i_away] > 1e-10 and grad[i_away] > 1e-10:
                t_away = min(t * 0.5, b[i_away] * 0.9)
                # Remove away component
                ks_s = i_away
                ks_e = min(ks_s + n - 1, 2 * n - 2)
                b_conv[ks_s:ks_e + 1] -= t_away * a_fixed
                b[i_away] -= t_away
                b[i_away] = max(b[i_away], 0.0)

        # Add new component
        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a_fixed
        b = (1.0 - t) * b
        b[j_star] += t

    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b / s


def fw_alternating(a0, time_limit, fw_steps=150, alpha=None, verbose=True):
    """
    Alternating FW: optimize b given a (via FW), blend, then optimize a given b.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    a_conv = fast_convolve(a)
    best_M = float(a_conv.max())
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Optimize b given a
        b = fw_best_response(a, fw_steps)

        # Blending
        if alpha is None:
            alpha = 3.0 / (it + 2.0)
        a_new = (1 - alpha) * a + alpha * b
        a_new = normalize(a_new)

        # Evaluate
        a_conv = fast_convolve(a_new)
        M = float(a_conv.max())
        if M < best_M:
            best_M = M
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-Alt it={it} ({elapsed:.0f}s): M={M:.10f}, score={2*n*M:.6f}")

    return best_a, 2.0 * n * best_M


def mirror_descent_multi_peak(a0, time_limit, lr=0.05, top_k=50, temperature=0.01, verbose=True):
    """
    Mirror descent (multiplicative weights) with multi-peak subgradient.
    Uses softmax-weighted combination of top-K peaks.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    best_a = a.copy()
    best_M = float(conv.max())
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]
            temp = temperature * max(M, 1e-10)
            w = np.exp((pv - pv.max()) / temp)
            weights = w / w.sum()

        # Compute subgradient: g[i] = sum_k weights[k] * 2*a[k-i]
        g = np.zeros(n)
        for idx in range(len(peaks)):
            if weights[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # g[i] += weights[idx] * 2 * a[k-i] for i in [i_start, i_end]
            g[i_start:i_end + 1] += weights[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent (multiplicative weight) step
        a_new = a * np.exp(-lr * g)
        a_new = normalize(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2 * n * float(conv.max())
            print(f"    MD it={it} ({elapsed:.0f}s): best={2*n*best_M:.6f}, cur={cur_score:.6f}, lr={lr:.6f}")

    return best_a, 2.0 * n * best_M


def perturbation_restart(a0, best_M, n_perturbations=5, verbose=True):
    """
    Try several perturbations of the current best and re-optimize each briefly.
    """
    n = len(a0)
    best_a = a0.copy()
    best_M_val = best_M
    rng = np.random.default_rng(12345)

    for p in range(n_perturbations):
        # Different perturbation strategies
        if p == 0:
            # Gaussian noise
            noise = rng.standard_normal(n)
            scale = 0.05
        elif p == 1:
            # Shift (circular)
            shift = rng.integers(1, n // 4)
            noise = np.roll(a0, shift) - a0
            scale = 1.0
        elif p == 2:
            # Scale certain regions
            noise = np.zeros(n)
            mid = n // 2
            noise[:mid] = rng.uniform(-0.1, 0.1, mid)
            noise[mid:] = rng.uniform(-0.05, 0.1, n - mid)
            scale = 1.0
        elif p == 3:
            # Random sparsification
            mask = rng.random(n) < 0.3
            noise = np.zeros(n)
            noise[mask] = rng.uniform(-0.2, 0.2, mask.sum())
            scale = 1.0
        else:
            # Frequency-domain perturbation
            A = rfft(a0, n)
            phase = rng.uniform(-0.1, 0.1, len(A))
            A_pert = A * np.exp(1j * phase)
            noise = irfft(A_pert, n) - a0
            scale = 1.0

        a_pert = normalize(a0 + scale * noise)
        conv_pert = fast_convolve(a_pert)
        M_pert = float(conv_pert.max())

        if verbose:
            print(f"    Perturb {p}: M={M_pert:.10f} (best={best_M_val:.10f})")

        if M_pert < best_M_val * 1.001:  # Accept if close
            # Quick CD polish
            a_quick, M_quick = quick_cd(a_pert, max_sweeps=50)
            if M_quick < best_M_val:
                best_M_val = M_quick
                best_a = a_quick.copy()
                if verbose:
                    print(f"      -> Improved to M={M_quick:.10f}")

    return best_a, best_M_val


def quick_cd(a0, max_sweeps=50):
    """Quick coordinate descent for polishing."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M

    for sweep in range(max_sweeps):
        n_improv = 0
        # Randomize order for diversity
        order = np.random.permutation(n)
        for i in order:
            if a[i] < 1e-15:
                continue
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length = k_end - k_start + 1

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            v = a[:length]
            affected = conv[k_start:k_end + 1]
            M_aff = float(affected.max())

            if M_aff <= unaff_max + 1e-15:
                continue

            mask = v > 1e-15
            if not mask.any():
                continue

            ratios = (unaff_max - affected[mask]) / (2.0 * v[mask])
            delta = float(ratios.min())
            if delta < -a[i]:
                delta = -a[i]
            if delta >= -1e-15:
                continue

            a[i] += delta
            changes = 2.0 * delta * v
            if 2 * i >= k_start and 2 * i <= k_end:
                idx = 2 * i - k_start
                changes = changes.copy()
                changes[idx] += delta * delta
            conv[k_start:k_end + 1] += changes

            new_M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
            if new_M < M:
                M = new_M
                n_improv += 1

        if M < best_M:
            best_M = M
            best_a = a.copy()

        if n_improv == 0:
            break

    return best_a, best_M


def structured_init(n, rng, method="cosine_asym"):
    """
    Generate structured initial sequences with anti-palindromic bias.
    These are designed to have low center-convolution-value.
    """
    if method == "cosine_asym":
        # Asymmetric cosine: large on one side, small on other
        t = np.arange(n, dtype=np.float64) / n
        power = rng.uniform(1.0, 3.0)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) ** power
        # Make asymmetric
        asym = rng.uniform(0.3, 0.7)
        t2 = (t - asym) / (1 - 2 * asym + 1e-10)
        t2 = np.clip(t2, -1, 1)
        a2 = (0.5 + 0.5 * np.cos(np.pi * t2)) ** power
        a = a * a2
        a = np.maximum(a, 0)

    elif method == "exponential_ramp":
        # Exponential ramp (large at one end)
        t = np.arange(n, dtype=np.float64) / n
        rate = rng.uniform(1.0, 5.0)
        direction = rng.choice([1, -1])
        a = np.exp(direction * rate * (t - 0.5))
        a = np.maximum(a, 0)

    elif method == "two_hump_asym":
        # Two asymmetric humps
        c1 = rng.uniform(0.1, 0.4) * n
        c2 = rng.uniform(0.6, 0.9) * n
        w1 = rng.uniform(0.05, 0.15) * n
        w2 = rng.uniform(0.05, 0.15) * n
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + 0.5 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
        a = np.maximum(a, 0)

    elif method == "random_smooth":
        # Random smooth sequence (low-pass filtered noise)
        a = rng.standard_normal(n)
        # Simple low-pass
        kernel_size = int(rng.integers(5, 21))
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)

    elif method == "fourier_sparse":
        # Sparse in frequency domain
        n_freq = n // 2 + 1
        spectrum = np.zeros(n_freq)
        n_nonzero = int(rng.integers(3, 15))
        indices = rng.choice(n_freq, n_nonzero, replace=False)
        magnitudes = rng.uniform(0.5, 1.0, n_nonzero)
        phases = rng.uniform(0, 2 * np.pi, n_nonzero)
        spectrum[indices] = magnitudes * np.cos(phases)
        a = irfft(spectrum * 2, n)  # Scale to get real
        a = np.maximum(a, 0)

    else:
        a = np.ones(n)

    return normalize(a)


def propose_candidate():
    """
    Multi-phase optimization:
    1. Load parent, apply exact CD
    2. Mirror descent refinement
    3. FW alternating
    4. More CD polishing
    5. Perturbation restarts
    6. Try different initializations at same n
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = normalize(p)
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        parent = np.ones(n_init) / n_init
        print("No parent found, using uniform init")

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    n = len(best_seq)
    print(f"Starting: n={n}, score={best_score:.6f}")

    def update_best(a, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # === Phase 1: Exact Coordinate Descent on parent ===
    print(f"\n=== Phase 1: Exact CD (budget: 400s) ===")
    t_start = time.monotonic()
    t_alloc = min(400.0, (deadline - t_start) * 0.25)
    a_cd, s_cd = exact_cd_optimize(parent, t_alloc, verbose=True)
    update_best(a_cd, s_cd, "cd_parent")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (budget: 200s) ===")
    t_start = time.monotonic()
    remaining = deadline - t_start - 600
    t_alloc = min(200.0, remaining * 0.25)
    if t_alloc > 30:
        a_md, s_md = mirror_descent_multi_peak(best_seq, t_alloc, lr=0.04, top_k=40, temperature=0.008, verbose=True)
        update_best(a_md, s_md, "mirror")

        # Quick CD polish on MD result
        if deadline - time.monotonic() > 100:
            a_md_cd, s_md_cd = quick_cd(a_md, max_sweeps=200)
            update_best(a_md_cd, s_md_cd, "md_cd")

    # === Phase 3: FW Alternating ===
    print(f"\n=== Phase 3: FW Alternating (budget: 200s) ===")
    t_start = time.monotonic()
    remaining = deadline - t_start - 400
    t_alloc = min(200.0, remaining * 0.3)
    if t_alloc > 30:
        a_fw, s_fw = fw_alternating(best_seq, t_alloc, fw_steps=100, verbose=True)
        update_best(a_fw, s_fw, "fw_alt")

        # CD polish on FW result
        if deadline - time.monotonic() > 80:
            a_fw_cd, s_fw_cd = quick_cd(a_fw, max_sweeps=200)
            update_best(a_fw_cd, s_fw_cd, "fw_cd")

    # === Phase 4: More CD Polishing ===
    print(f"\n=== Phase 4: CD Polishing (budget: 300s) ===")
    t_start = time.monotonic()
    remaining = deadline - t_start - 300
    t_alloc = min(300.0, remaining * 0.5)
    if t_alloc > 30:
        a_cd2, s_cd2 = exact_cd_optimize(best_seq, t_alloc, verbose=True)
        update_best(a_cd2, s_cd2, "cd_polish")

    # === Phase 5: Perturbation Restarts ===
    print(f"\n=== Phase 5: Perturbation Restarts (budget: 300s) ===")
    t_start = time.monotonic()
    remaining = deadline - t_start - 100
    t_alloc = min(300.0, remaining * 0.6)
    if t_alloc > 30:
        a_pert, s_pert = perturbation_restart(best_seq, best_score / (2 * n),
                                              n_perturbations=8, verbose=True)
        update_best(a_pert, s_pert, "perturb")

        # CD on perturbation result
        if deadline - time.monotonic() > 60:
            a_pert_cd, s_pert_cd = quick_cd(a_pert, max_sweeps=100)
            update_best(a_pert_cd, s_pert_cd, "perturb_cd")

    # === Phase 6: Try different structured initializations ===
    print(f"\n=== Phase 6: Structured Inits (budget: remaining) ===")
    t_start = time.monotonic()
    remaining = deadline - t_start - 50
    if remaining > 60:
        methods = ["cosine_asym", "exponential_ramp", "two_hump_asym", "random_smooth", "fourier_sparse"]
        t_per = remaining / len(methods)
        for method in methods:
            if time.monotonic() > deadline - 50:
                break
            init = structured_init(n, rng, method)
            init_score = 2 * n * fast_convolve(init).max()
            print(f"  Init '{method}': initial score={init_score:.6f}")

            if init_score < best_score * 1.1:  # Only proceed if promising
                t_sub = min(t_per, deadline - time.monotonic() - 50)
                if t_sub > 20:
                    a_s, s_s = exact_cd_optimize(init, t_sub, verbose=False)
                    update_best(a_s, s_s, f"init_{method}")

                    if deadline - time.monotonic() > 30:
                        a_s2, s_s2 = mirror_descent_multi_peak(a_s, min(50, deadline - time.monotonic() - 40),
                                                               lr=0.03, top_k=30, verbose=False)
                        update_best(a_s2, s_s2, f"md_{method}")

    # === Phase 7: Final CD polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        a_final, s_final = exact_cd_optimize(best_seq, remaining, verbose=False)
        update_best(a_final, s_final, "final")

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
