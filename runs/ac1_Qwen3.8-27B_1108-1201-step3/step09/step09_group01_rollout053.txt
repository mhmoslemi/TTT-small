"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with Nesterov (Polyak) averaging
2. Efficient peak-tracking coordinate descent
3. Multi-length exploration (1024, 2048)
4. Smooth log-sum-exp gradient descent with adaptive temperature
5. Intensive post-optimization with structured perturbations
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix_from_a(a):
    """Build (2n-1) x n matrix T where (T @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[k, j] = a[k - j]
    return T


def get_lp_solver():
    """Find a working LP solver."""
    cached = getattr(get_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.GLOP, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    get_lp_solver._cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None, T_cached=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    if T_cached is None:
        T = build_conv_matrix_from_a(a_fixed)
    else:
        T = T_cached
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with peak tracking and adaptive step sizes.
    Works on normalized sequences (sum=1).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected_idx = np.arange(k_start, k_end + 1)
            a_idx = affected_idx - i
            a_shifted = a[a_idx]

            best_delta = 0.0
            best_new_max = M

            # Try a set of factor-based deltas
            factors = [0.5, 0.7, 0.8, 0.9, 0.95, 0.99, 1.01, 1.05, 1.1, 1.2, 1.3, 1.5, 2.0]
            for f in factors:
                new_val = a[i] * f
                if new_val > 1.0:
                    continue
                delta = new_val - a[i]

                new_affected = conv[k_start:k_end + 1].copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            # Also try going to near-zero
            delta_zero = -a[i] * 0.999
            new_affected = conv[k_start:k_end + 1].copy() + 2.0 * delta_zero * a_shifted
            if 2 * i >= k_start and 2 * i <= k_end:
                idx_2i = 2 * i - k_start
                new_affected[idx_2i] += delta_zero * delta_zero
            new_max = max(unaff_max, float(new_affected.max()))
            if new_max < best_new_max - 1e-16:
                best_new_max = new_max
                best_delta = delta_zero

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        else:
            no_improve_count = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.10f}")

    return best_a, best_score, sweeps


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    Fictitious play with Nesterov (Polyak) averaging for the LP game.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = float('inf')
    iters = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best responses
        T_a = build_conv_matrix_from_a(a)
        b_new, M_b = solve_lp_best_response(a, solver, T_a)
        if b_new is not None:
            b = b_new

        T_b = build_conv_matrix_from_a(b)
        a_new, M_a = solve_lp_best_response(b, solver, T_b)
        if a_new is not None:
            a = a_new

        # Polyak averaging (Nesterov sequent)
        gamma = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma) * a_sequent + gamma * a
        b_sequent = (1.0 - gamma) * b_sequent + gamma * b

        # Evaluate candidates
        for cand in [a, b, a_sequent, b_sequent, 0.5 * (a_sequent + b_sequent)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()
                stagnant = 0
            else:
                stagnant += 1

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f} M_a={M_a:.8f}")

        # Early stopping if very stagnant
        if stagnant > 500 and iters > 100:
            break

    return best_a, best_score, iters


def smooth_gradient_descent(a0, time_limit, lr_init=0.005, beta_init=50.0):
    """
    Smooth gradient descent using log-sum-exp approximation of max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Project to simplex
        conv_a = fast_convolve(a)
        max_conv = conv_a.max()
        scaled = beta * (conv_a - max_conv)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_a)
            weights[conv_a.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient of smooth max w.r.t. a
        grad = np.zeros(n)
        peak_indices = np.where(weights > 1e-8)[0]
        for k in peak_indices:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * a[k - j]

        # Gradient step with projection
        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-9:
                break
        elif it % 30 == 0:
            beta *= 1.05

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_bimodal(n, rng):
    """Initialize with bimodal distribution."""
    center1 = n * rng.uniform(0.2, 0.35)
    center2 = n * rng.uniform(0.65, 0.8)
    width1 = n * rng.uniform(0.05, 0.12)
    width2 = n * rng.uniform(0.05, 0.12)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_symmetric_cosine(n, rng):
    """Symmetric cosine-based initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 6))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = profile[-1] * rng.uniform(0.8, 1.2)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_perturb(a, rng, strength=0.05):
    """Add multiplicative noise and reproject to simplex."""
    n = len(a)
    noise = 1.0 + rng.standard_normal(n) * strength
    a_new = a * np.maximum(noise, 0.1)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def peak_pressure_perturb(a, rng, strength=0.03):
    """Perturb by reducing peak-contributing coordinates."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.9
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions[:50]:  # Limit to top peaks for speed
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    a_new = a * (1.0 - strength * pressure)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

            if t_left < 3:
                break
            t_alloc = max(2.0, min(8.0, t_left / 2.0))
            a_opt, s_opt, _ = coord_descent(cand, t_alloc, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done ({time.monotonic()-t1_start:.0f}s), top-3: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:3]]))

    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: LP Fictitious Play (main workhorse) ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    # Pick the best length for intensive optimization
    if best_seq is not None:
        n_best = len(best_seq)
    else:
        n_best = 1024

    # Run NFP on the best length
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem * 0.6, 500.0)
        print(f"  NFP n={n_best}, budget={t_alloc:.0f}s")
        initial = interpolate_seq(best_seq, n_best) if best_seq is not None else None
        a_fp, score_fp, iters = nesterov_fictitious_play(
            n_best, t_alloc, initial=initial, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"nfp_n{n_best}")

    # Try NFP on second length if time remains
    alt_n = 2048 if n_best != 2048 else 1024
    rem = deadline - time.monotonic() - 500
    if rem > 120:
        t_alloc = min(rem * 0.4, 200.0)
        print(f"  NFP alt n={alt_n}, budget={t_alloc:.0f}s")
        initial_alt = interpolate_seq(best_seq, alt_n) if best_seq is not None else None
        a_fp2, score_fp2, iters2 = nesterov_fictitious_play(
            alt_n, t_alloc, initial=initial_alt, solver=solver, verbose=False
        )
        best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, f"nfp_alt_n{alt_n}")

    print(f"  Phase 2 done ({time.monotonic()-t2_start:.0f}s), best={best_score:.6f}")

    # === Phase 3: Intensive Coordinate Descent ===
    print(f"\n=== Phase 3: Intensive CD (best={best_score:.6f}) ===")
    t3_start = time.monotonic()

    if best_seq is not None:
        current = best_seq.copy()
        n_cd = len(current)

        for rnd in range(8):
            rem = deadline - time.monotonic() - 300
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 100.0)
            print(f"  CD round {rnd+1}/8, n={n_cd}, budget={t_alloc:.0f}s, score={best_score:.6f}")
            a_opt, s_opt, sweeps = coord_descent(current, t_alloc, verbose=True, label=f"cd3_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"cd3_r{rnd}")

            # Perturb to escape local optima
            if rnd < 5:
                if rnd % 2 == 0:
                    current = peak_pressure_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
                else:
                    current = random_perturb(best_seq, rng, strength=0.03 + 0.02 * rnd)
            else:
                current = best_seq.copy()

    print(f"  Phase 3 done ({time.monotonic()-t3_start:.0f}s), best={best_score:.6f}")

    # === Phase 4: Smooth Gradient Descent ===
    print(f"\n=== Phase 4: Smooth GD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_md = min(rem * 0.5, 120.0)
            print(f"  SGD n={len(best_seq)}, budget={t_md:.0f}s")
            a_sg, s_sg = smooth_gradient_descent(best_seq, t_md, lr_init=0.003, beta_init=40.0)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "smooth_gd_1")

            # Try with different params
            rem = deadline - time.monotonic() - 100
            if rem > 20:
                a_sg2, s_sg2 = smooth_gradient_descent(best_seq, min(rem * 0.5, 60.0),
                                                       lr_init=0.01, beta_init=80.0)
                best_seq, best_score = check_and_update(a_sg2, best_seq, best_score, "smooth_gd_2")

    # === Phase 5: More NFP + CD cycles ===
    print(f"\n=== Phase 5: NFP+CD Cycles (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        for cycle in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 60:
                break

            # Short NFP burst
            t_nfp = min(rem * 0.3, 100.0)
            a_nfp, _, _ = nesterov_fictitious_play(
                n_cur, t_nfp, initial=best_seq, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_nfp, best_seq, best_score, f"nfp5_c{cycle}")

            # CD polish
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            t_cd = min(rem * 0.4, 80.0)
            a_cd, _, _ = coord_descent(best_seq, t_cd, label=f"cd5_c{cycle}")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd5_c{cycle}")

    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best_final = len(best_seq)
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            for alt_n in [n_best_final + 256, n_best_final + 512, n_best_final - 256]:
                if alt_n <= 0 or alt_n > 4096 or rem < 15:
                    continue
                t_alt = min(rem * 0.3, 40.0)
                rem = deadline - time.monotonic() - 20
                if rem < 10:
                    break
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt, _ = coord_descent(init_alt, t_alt, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_final = min(rem * 0.8, 60.0)
            a_fin, s_fin, _ = coord_descent(best_seq, t_final, label="final")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    elapsed_total = time.monotonic() - (deadline - 1785.0)
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}) [total time: {elapsed_total:.0f}s]")
    return [float(x) for x in best_seq]
