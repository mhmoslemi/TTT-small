from scipy.fft import rfft, irfft

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15: return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s*s)

def score_norm(a):
    """Assume sum(a)=1"""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient(a):
    """Exact subgradient of max_k conv(a,a)[k] at point a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    k_star = np.argmax(b)
    grad = np.zeros(n)
    # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k_star - i]
    return grad

def soft_gradient(a, T):
    """Smooth gradient using softmax."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j a[j] * w[i+j]

    W = rfft(w, N)

def nesterov_subgrad(a0, time_budget, eta_init=1.0):

    best_a = a.copy()

def lp_refine(a0, time_budget, max_iters=50000):
    """LP-based best response refinement."""
    solver = get_lp_solver()
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_sc
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Solve LP: min M s.t. conv(a, b)[k] <= M for all k, b >= 0, sum(b) = 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        # Build constraint matrix
        # conv(a,b)[k] = sum_j a[j] * b[k-j] for valid j, k-j
        # For each k: sum_j a[j]*b[k-j] <= M
        # This is: (T @ b)[k] <= M where T[k,j] = a[j] if 0<=k-j<n else 0
        
        # Build T as sparse or dense
        # T[k, j] = a[j] if 0 <= k-j < n else 0, for k=0..2n-2, j=0..n-1
        # T is (2n-1) x n
        T_mat = np.zeros((2*n-1, n))
        for j in range(n):
            T_mat[j:j+n, j] = a[j]  # T[k,j] = a[j] where k = j + i for i=0..n-1
        
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            continue
        b_val = b_val / s
        
        # Nesterov-like averaging
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = simplex_project(y)
        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = simplex_project(a)
        
        if (it + 1) % 100 == 0:
            sc_a = score_norm(a)
            sc_y = score_norm(y)
            if sc_a < best_sc:
                best_sc = sc_a
                best_a = a.copy()
                stall = 0
            if sc_y < best_sc:
                best_sc = sc_y
                best_a = y.copy()
                stall = 0
            if best_sc >= prev_best - 1e-9:
                stall += 1
            else:
                stall = 0
            prev_best = best_sc
            
            if stall >= 5:
                rng = np.random.default_rng(it)
                y = y + 0.03 * rng.standard_normal(n)
                y = simplex_project(y)
                stall = 0
    
    return best_a, best_sc

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def score_norm(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient(a):
    n = len(a)
    b = fast_convolve(a)
    k_star = np.argmax(b)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k_star - i]
    return grad


def mirror_descent_fast(a0, time_budget, T_final=300.0, n_stages=20, eta_scale=8.0):

def nesterov_subgrad(a0, time_budget, eta_init=1.0):

    a = simplex_project(a0.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    t = 1

    while time.monotonic() - t0 < time_budget:
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) * (t - 1))) / 2.0 if t > 1 else 1.0
        beta = (t_prev - 1.0) / t_curr

        y = a + beta * (a - a_prev)
        y = simplex_project(y)

        grad = subgradient(y)

        g_norm = np.linalg.norm(grad, 2)
        eta = eta_init / np.sqrt(t + 1)
        eta = min(eta, 0.5 / (g_norm + 1e-15))

        a_prev = a.copy()
        a = simplex_project(y - eta * grad)
        t += 1

        if t % 100 == 0:

                best_sc = sc

def lp_refine(a0, time_budget, max_iters=50000, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_sc = score_norm(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_sc

    # Pre-build structure (T depends on a, so we rebuild each iteration)
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        T_mat = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_mat[j:j + n, j] = a[j]

        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            continue
        b_val = b_val / s

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = simplex_project(y)
        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = simplex_project(a)

        if (it + 1) % 50 == 0:
            sc_a = score_norm(a)
            sc_y = score_norm(y)
            if sc_a < best_sc:
                best_sc = sc_a
                best_a = a.copy()
                stall = 0
            if sc_y < best_sc:
                best_sc = sc_y
                best_a = y.copy()
                stall = 0
            if best_sc >= prev_best - 1e-9:
                stall += 1
            else:
                stall = 0
            prev_best = best_sc

            if stall >= 6:
                rng = np.random.default_rng(it * 13)
                y = y + 0.04 * rng.standard_normal(n)
                y = simplex_project(y)
                stall = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"        LP iter={it+1} ({elapsed:.0f}s): score={best_sc:.6f}")

    return best_a, best_sc


def make_init(n, style, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    if style == 'uniform':
        a = np.ones(n)
    elif style == 'cosine':
        a = np.sin(np.pi * x) ** 2
    elif style == 'u_shape':
        a = 1.0 / (0.3 + 4.0 * (x - 0.5) ** 2)
    elif style == 'two_peak':
        a = np.exp(-6.0 * (x - 0.25) ** 2) + np.exp(-6.0 * (x - 0.75) ** 2) + 0.2
    elif style == 'bilinear':
        a = 1.0 + 0.6 * (2.0 * x - 1.0) ** 2
    elif style == 'random':
        a = rng.dirichlet(np.ones(n) * 3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if best_seq is None:
        n_init = 768
        best_seq = make_init(n_init, 'cosine', rng)
        best_score = score_of(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_par = len(best_seq)

    # === Phase 1: Quick screening across n values ===
    print(f"\n=== Phase 1: Multi-n Screening (best={best_score:.6f}) ===")
    n_candidates = sorted(set([
        n_par,
        max(128, n_par - 128),
        max(128, n_par - 64),
        n_par + 64,
        n_par + 128,
        n_par + 256,
    ]))
    # Also add some standard sizes
    for ns in [384, 512, 640, 768, 896, 1024, 1152, 1280, 1408, 1600, 1920]:
        if ns not in n_candidates:
            n_candidates.append(ns)
    n_candidates = sorted(set([n for n in n_candidates if 128 <= n <= 2560]))

    results = {}
    for n in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 45:
            break
        t_md = min(rem * 0.12, 45.0)
        if t_md < 15:
            break

        a_init = interpolate_seq(best_seq, n)
        a_md, _ = mirror_descent_fast(a_init, t_md, T_final=500.0, n_stages=15, eta_scale=10.0)
        sc = score_of(a_md)
        results[n] = (sc, a_md)
        marker = " *" if sc < best_score else ""
        print(f"  n={n}: score={sc:.6f}{marker}")

        if sc < best_score:
            best_score = sc
            best_seq = a_md.copy()

    # Sort by score, pick top 3 for detailed optimization
    sorted_results = sorted(results.items(), key=lambda x: x[1][0])
    top_n = [rn for rn, _ in sorted_results[:4]]
    print(f"\nTop n values for detailed optimization: {top_n}")

    # === Phase 2: Detailed optimization on top candidates ===
    print(f"\n=== Phase 2: Detailed Optimization (best={best_score:.6f}) ===")
    for n in top_n:
        rem = deadline - time.monotonic()
        if rem < 120:
            break

        sc_init, a_init = results[n]
        # Use the best known sequence interpolated, or the screened result
        a_start = interpolate_seq(best_seq, n) if len(best_seq) != n else best_seq.copy()
        if sc_init < score_of(a_start):
            a_start = a_init

        # Step 2a: Extended mirror descent
        t_md = min(rem * 0.25, 120.0)
        if t_md > 20:
            a_md, _ = mirror_descent_fast(a_start, t_md, T_final=800.0, n_stages=20, eta_scale=12.0)
            sc = score_of(a_md)
            if sc < best_score:
                best_score = sc
                best_seq = a_md.copy()
                print(f"  n={n} MD: {sc:.6f}")
            a_current = a_md
        else:
            a_current = a_start

        # Step 2b: Nesterov subgradient
        rem = deadline - time.monotonic()
        t_ns = min(rem * 0.15, 60.0)
        if t_ns > 15:
            a_ns, _ = nesterov_subgrad(a_current, t_ns, eta_init=1.5)
            sc = score_of(a_ns)
            if sc < best_score:
                best_score = sc
                best_seq = a_ns.copy()
                print(f"  n={n} NSG: {sc:.6f}")
            a_current = a_ns

        # Step 2c: LP refinement (only for manageable n)
        rem = deadline - time.monotonic()
        if n <= 1400 and rem > 60:
            t_lp = min(rem * 0.25, 180.0)
            if t_lp > 30:
                a_lp, _ = lp_refine(a_current, t_lp, solver=solver)
                sc = score_of(a_lp)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_lp.copy()
                    print(f"  n={n} LP: {sc:.6f}")
                a_current = a_lp

    # === Phase 3: Final polish ===
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        n_final = len(best_seq)

        # LP polish if manageable
        if n_final <= 1400:
            t_pol = min(rem * 0.5, 300.0)
            if t_pol > 30:
                print(f"  LP polish, budget={t_pol:.0f}s")
                a_p, _ = lp_refine(best_seq, t_pol, solver=solver)
                sc = score_of(a_p)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_p.copy()
                    print(f"  After LP polish: {sc:.6f}")

        # Final mirror descent polish
        rem = deadline - time.monotonic()
        if rem > 20:
            t_mp = min(rem * 0.3, 60.0)
            if t_mp > 10:
                a_mp, _ = mirror_descent_fast(best_seq, t_mp, T_final=2000.0, n_stages=10, eta_scale=15.0)
                sc = score_of(a_mp)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_mp.copy()
                    print(f"  After MD polish: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        final_sc = evaluate_safe(best_seq)
        if np.isfinite(final_sc):
            best_score = final_sc
        # Ensure all values are valid
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        s = best_seq.sum()
        if s < 0.01:
            best_seq = np.ones(960) / 960
            best_score = evaluate_safe(best_seq)

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
