    m = conv.max()

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_gradient_fft(a, weights, n, fft_size):
    """
    Compute grad[i] = 2 * sum_k weights[k] * a[k-i] using FFT cross-correlation.
    weights has length 2n-1, a has length n.
    """
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * corr


def soft_max_weights(conv, tau):
    """Compute softmax weights over convolution values (for smoothed gradient)."""
    m = conv.max()
    shifted = tau * (conv - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = w / ws
    return w


def mirrored_smooth_descent(a0, time_limit, tau_start=3.0, tau_end=2000.0,
                            lr_start=0.02, n_restarts=3, verbose=True):
    """
    Mirror descent with soft-max smoothing and Nesterov momentum.
    Progressively increases tau (sharpness) to transition from smooth to sharp.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (3 * n - 1).bit_length()

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        # Progressive tau schedule (log-linear)
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)

        # Adaptive learning rate
        lr = lr_start * (1.0 / (1.0 + 0.01 * it))

        # Compute smoothed gradient
        y_conv = fast_convolve(a)
        weights = soft_max_weights(y_conv, tau)
        grad = compute_gradient_fft(a, weights, n, fft_size)

        # Mirror descent step (multiplicative update on simplex)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        # Nesterov-like momentum (exponential moving average)
        beta = min(0.95, 0.5 + 0.45 * frac)
        a = (1.0 - beta) * a + beta * a_new
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s_a

        it += 1

        # Evaluate
        if it % 10 == 0:
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 10

            # Restart with perturbation if stuck
            if no_improve > patience:
                # Perturb: mix with random
                noise = np.random.dirichlet(np.ones(n) * 0.1)
                a = (1 - 0.3) * best_a + 0.3 * noise
                a = a / a.sum()
                no_improve = 0
                if verbose:
                    print(f"    [restart] it={it}, score={best_score:.8f}")

        if verbose and it % 2000 == 0:
            conv_check = fast_convolve(a)
            print(f"    MDS it={it} ({time.monotonic()-t0:.0f}s): "
                  f"best={best_score:.8f} tau={tau:.1f}")

    return best_a, best_score


def coord_descent_exact(a0, time_limit, n_sweeps_target=50, verbose=True, label=""):
    """
    Exact coordinate descent: for each coordinate, find optimal delta via
    binary search on the 1D convex subproblem.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    def compute_new_M(delta, i, conv, a, n):
        """Compute new max convolution after changing a[i] by delta."""
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)

        # Unaffected max
        unaff_max = 0.0
        if k_start > 0:
            unaff_max = max(unaff_max, float(conv[:k_start].max()))
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

        # Affected: conv[k] + 2*delta*a[k-i] (+ delta^2 if k==2i)
        affected_max = 0.0
        for k in range(k_start, k_end + 1):
            val = conv[k] + 2.0 * delta * a[k - i]
            if k == 2 * i:
                val += delta * delta
            if val > affected_max:
                affected_max = val

        return max(unaff_max, affected_max)

    def optimized_compute_new_M(delta, i, conv, a, n):
        """Vectorized version of compute_new_M."""
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)

        unaff_max = 0.0
        if k_start > 0:
            unaff_max = max(unaff_max, float(conv[:k_start].max()))
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

        k_range = np.arange(k_start, k_end + 1)
        a_shifted = a[k_range - i]
        new_vals = conv[k_start:k_end + 1].copy() + 2.0 * delta * a_shifted
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            new_vals[idx] += delta * delta

        affected_max = float(new_vals.max())
        return max(unaff_max, affected_max)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        any_improved = False

        # Randomize sweep order for diversity
        order = np.random.permutation(n)

        for i in order:
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Binary search for optimal delta in [-a[i], a[i]]
            # (wider range to allow both increase and decrease)
            lo = -a[i]
            hi = a[i] * 0.5  # Don't double any coordinate in one step

            # Quick check: is current already optimal?
            current_M = M

            # Try a few quick steps first
            best_delta = 0.0
            best_new_M = current_M

            # Check delta = -a[i] (zero out)
            new_M_at_neg = optimized_compute_new_M(-a[i] * 0.99, i, conv, a, n)
            if new_M_at_neg < best_new_M:
                best_new_M = new_M_at_neg
                best_delta = -a[i] * 0.99

            # Check small positive delta
            new_M_at_pos = optimized_compute_new_M(a[i] * 0.1, i, conv, a, n)
            if new_M_at_pos < best_new_M:
                best_new_M = new_M_at_pos
                best_delta = a[i] * 0.1

            # Binary search refinement if we found improvement
            if best_new_M < current_M - 1e-14:
                # Narrow the search
                if best_delta < 0:
                    lo = -a[i]
                    hi = 0.0
                else:
                    lo = 0.0
                    hi = a[i] * 0.5

                for _ in range(20):
                    mid = (lo + hi) / 2
                    mid_M = optimized_compute_new_M(mid, i, conv, a, n)
                    # Check left
                    left = (lo + mid) / 2
                    left_M = optimized_compute_new_M(left, i, conv, a, n)
                    # Check right
                    right = (mid + hi) / 2
                    right_M = optimized_compute_new_M(right, i, conv, a, n)

                    if left_M <= mid_M and left_M <= right_M:
                        hi = mid
                    elif right_M <= mid_M and right_M <= left_M:
                        lo = mid
                    else:
                        hi = mid if mid_M <= left_M else lo
                        lo = mid if mid_M <= right_M else hi

                    if mid_M < best_new_M:
                        best_new_M = mid_M
                        best_delta = mid

                # Final check
                for test_d in [best_delta, best_delta * 0.8, best_delta * 1.1,
                               lo * 0.5, hi * 0.5, (lo + hi) / 2]:
                    test_M = optimized_compute_new_M(test_d, i, conv, a, n)
                    if test_M < best_new_M:
                        best_new_M = test_M
                        best_delta = test_d

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                a_shifted = a[k_start - i:k_end - i + 1]
                # Note: a[i] has already been updated, so a_shifted[i] = a[i]_new
                # The change in conv is: 2*delta*old_a[k-i] + delta^2*(k==2i)
                # But we need old_a[k-i], which for k-i == i means old a[i]
                old_ai = a[i] - best_delta
                a_shifted_old = a_shifted.copy()
                a_shifted_old[i - i] = old_ai  # fix the a[i] term (index 0 in a_shifted)
                # Actually a_shifted = a[k-i] for k in [k_start, k_end]
                # a[i] corresponds to k-i = i, i.e., k = 2i
                # So the index in a_shifted for a[i] is (2i - k_start) = 2i - i = i
                a_shifted_old[i] = old_ai

                changes = 2.0 * best_delta * a_shifted_old
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes[idx] += best_delta * best_delta

                conv[k_start:k_end + 1] += changes
                M = max(M, best_new_M)  # M can only decrease or stay
                # Actually recompute M properly
                M = float(conv.max())

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                any_improved = True

        if not any_improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 5:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): "
                  f"best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def multi_peak_subgradient_cd(a0, time_limit, top_k=50, verbose=True, label=""):
    """
    Hybrid: use subgradient info to identify which coordinates to modify,
    then apply exact 1D optimization to those coordinates.

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (3 * n - 1).bit_length()
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())

        # Get active peaks (within 5% of max)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Compute subgradient from active peaks
        # grad[i] = sum_{k in active} (1/|active|) * 2 * a[k-i]
        weights = np.zeros(2 * n - 1)
        weights[active] = 1.0 / len(active)
        grad = compute_gradient_fft(a, weights, n, fft_size)

        # Identify coordinates to modify: those with highest gradient
        # (highest gradient = most "responsible" for the peak)
        # We want to DECREASE high-gradient coordinates
        # And potentially INCREASE low-gradient coordinates

        # Select top coordinates by gradient (to decrease)
        n_modify = min(n // 4, 256)
        top_indices = np.argsort(grad)[-n_modify:]
        # Also select some low-gradient coordinates (to potentially increase)
        bottom_indices = np.argsort(grad)[:n_modify // 2]
        modify_set = np.unique(np.concatenate([top_indices, bottom_indices]))

        # For each coordinate in modify_set, do exact 1D optimization
        improved = False

            lo = -a[i]
            hi = a[i] * 0.3
            best_delta = 0.0
            best_new_M = M

            candidates = [lo, lo * 0.5, -a[i] * 0.1, 0.0,

            for d in candidates:

                ks = i

                if 2 * i >= ks and 2 * i <= ke:
                    idx = 2 * i - ks
                    new_vals[idx] += d * d
                affected_max = float(new_vals.max())
                new_M = max(unaff_max, affected_max)
                if new_M < best_new_M:

                    new_M = max(unaff_max, float(new_vals.max()))

                        best_new_M = new_M

                    else:
                        hi_s = mid

def diverse_init(n, rng, method="random"):
    """Generate diverse initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "gaussian_center":
        t = np.arange(n, dtype=np.float64)
        center = n / 2
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    elif method == "two_peaks":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif method == "periodic":
        period = int(rng.integers(3, 15))
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.5 + 0.5 * np.cos(2 * np.pi * phase)
        a = np.maximum(a, 0) + 1e-10
    elif method == "random_smooth":
        # Generate random then smooth
        raw = rng.standard_normal(n)
        # Simple moving average
        k = max(3, int(n * 0.02))
        kernel = np.ones(k) / k
        from scipy.signal import convolve
        a = convolve(raw, kernel, mode='same')
        a = np.maximum(a, 0) + 1e-10
    elif method == "chirp":
        # Linear chirp - good for low auto-correlation
        t = np.arange(n, dtype=np.float64)
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t / n * (t / n))
        a = np.maximum(a, 0) + 1e-10
    elif method == "moire":
        # Interference pattern
        t = np.arange(n, dtype=np.float64)
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(1.5, 4.0)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * t / n) * np.cos(2 * np.pi * f2 * t / n)
        a = np.maximum(a, 0) + 1e-10
    elif method == "power_law":
        t = np.arange(1, n + 1, dtype=np.float64)
        alpha = rng.uniform(0.5, 2.0)
        a = t ** (-alpha)
        a += rng.uniform(0, 0.5) * n ** (-alpha)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def iterated_refinement(a0, time_limit, verbose=True):
    """
    Alternates between smoothed mirror descent and coordinate descent,
    with progressively tighter parameters.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    # Phase A: Smoothed mirror descent (long)
    frac_a = 0.4
    t_a = time_limit * frac_a
    print(f"  [iter_refine] Phase A (MDS): {t_a:.0f}s")
    a_out, s_out = mirrored_smooth_descent(best_a, t_a, tau_start=2.0, tau_end=500.0,
                                            lr_start=0.03, verbose=True)
    if s_out < best_score:
        best_score = s_out
        best_a = a_out.copy()

    # Phase B: Subgradient-guided CD
    remaining = time_limit - (time.monotonic() - t0) - 10
    frac_b = 0.35
    t_b = min(remaining, time_limit * frac_b)
    print(f"  [iter_refine] Phase B (SubGradCD): {t_b:.0f}s")
    a_out, s_out = multi_peak_subgradient_cd(best_a, t_b, top_k=30, verbose=True, label="B")
    if s_out < best_score:
        best_score = s_out
        best_a = a_out.copy()

    # Phase C: Exact coordinate descent
    remaining = time_limit - (time.monotonic() - t0) - 5
    t_c = min(remaining * 0.6, time_limit * 0.2)
    print(f"  [iter_refine] Phase C (ExactCD): {t_c:.0f}s")
    a_out, s_out = coord_descent_exact(best_a, t_c, verbose=True, label="C")
    if s_out < best_score:
        best_score = s_out
        best_a = a_out.copy()

    # Phase D: Another round of MDS with sharper tau
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        t_d = min(remaining * 0.7, 100.0)
        print(f"  [iter_refine] Phase D (MDS sharp): {t_d:.0f}s")
        a_out, s_out = mirrored_smooth_descent(best_a, t_d, tau_start=50.0, tau_end=5000.0,
                                                lr_start=0.01, verbose=False)
        if s_out < best_score:
            best_score = s_out
            best_a = a_out.copy()

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate with the actual evaluate_sequence function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}) ***")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Use parent sequence as primary starting point
    2. Generate diverse initializations for multi-start
    3. For each promising start: smoothed mirror descent → subgradient CD → exact CD
    4. Final polishing with fine coordinate descent
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, score={best_score:.8f}")

    # === Multi-start exploration ===
    # Define diverse initializations to try
    init_methods = ["chirp", "moire", "two_peaks", "gaussian_center", "random_smooth", "periodic"]
    
    # Evaluate initializations quickly to pick the best few
    inits = [best_seq.copy()]
    scores = [best_score]
    
    for method in init_methods:
        a_init = diverse_init(n_best, rng, method)
        s_init = 2.0 * n_best * fast_convolve(a_init).max()
        inits.append(a_init)
        scores.append(s_init)
        print(f"  Init '{method}': internal_score={s_init:.8f}")

    # Sort by score, keep top 3 + parent
    order = np.argsort(scores)
    top_inits = [inits[0]]  # Always include parent
    for idx in order[1:]:
        if len(top_inits) >= 4:
            break
        top_inits.append(inits[idx])

    # === Main optimization loop ===
    # Allocate time: 60% to parent, 40% split among other inits
    total_time = deadline - time.monotonic()
    print(f"\nTotal available time: {total_time:.0f}s")

    # Phase 1: Optimize with parent (most time)
    t_parent = total_time * 0.5
    print(f"\n=== Optimizing parent (budget: {t_parent:.0f}s) ===")
    a_opt, s_opt = iterated_refinement(best_seq, t_parent, verbose=True)
    best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "parent_refined")

    # Phase 2: Optimize with other inits
    remaining = deadline - time.monotonic() - 60
    n_other = len(top_inits) - 1
    if n_other > 0 and remaining > 30:
        t_per_other = remaining / (n_other + 1)  # +1 for final polish
        for idx in range(1, len(top_inits)):
            remaining = deadline - time.monotonic() - 60
            if remaining < 30:
                break
            t_alloc = min(t_per_other, remaining)
            print(f"\n=== Optimizing init {idx} (budget: {t_alloc:.0f}s) ===")
            a_opt, s_opt = iterated_refinement(top_inits[idx], t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"init_{idx}")

    # Phase 3: Try different n values (shorter sequences might have better ratio)
    remaining = deadline - time.monotonic() - 30
    if remaining > 60:
        print(f"\n=== Trying different lengths (budget: {remaining:.0f}s) ===")
        for n_try in [512, 640, 2048]:
            remaining = deadline - time.monotonic() - 30
            if remaining < 30:
                break
            t_alloc = min(remaining * 0.3, 90.0)
            # Use the best sequence resampled or a new init
            if n_try == n_best:
                continue
            a_init = diverse_init(n_try, rng, "chirp")
            print(f"  Trying n={n_try} (budget: {t_alloc:.0f}s)")
            a_opt, s_opt = iterated_refinement(a_init, t_alloc, verbose=False)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"n_{n_try}")

    # Phase 4: Final polishing on the best
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        print(f"\n=== Final polishing (budget: {remaining:.0f}s) ===")
        # Fine coordinate descent
        t_final = min(remaining * 0.6, 60.0)
        a_opt, s_opt = coord_descent_exact(best_seq, t_final, verbose=False, label="final")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "final_cd")

        # One more MDS pass with very sharp tau
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_md = min(remaining * 0.5, 30.0)
            a_opt, s_opt = mirrored_smooth_descent(best_seq, t_md, tau_start=100.0,
                                                    tau_end=5000.0, lr_start=0.005,
                                                    verbose=False)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "final_mds")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    else:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: score={best_score:.8f}")

    return [float(x) for x in best_seq]
