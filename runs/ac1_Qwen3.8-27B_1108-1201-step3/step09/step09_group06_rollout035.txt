def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def spectral_flatten(a0, n_iters=2000, time_limit=None, t0=None, percentile=95.0):
    """Gerchberg-Saxton style spectral flattening."""
    if t0 is None:
        t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        # Compute DFT of zero-padded sequence
        A = rfft(a, fft_size)
        power = A.real**2 + A.imag**2  # |A|^2
        
        # Flatten: clip peaks
        threshold = np.percentile(power, percentile)
        power_flattened = np.minimum(power, threshold * 1.5)  # Allow some excess
        
        # Actually, let's use a softer approach: blend with mean
        mean_power = power.mean()
        # Reduce peaks more than troughs
        excess = np.maximum(power - mean_power, 0)
        deficit = np.maximum(mean_power - power, 0)
        alpha = 0.5  # Blending factor
        power_new = power - alpha * excess + alpha * deficit * 0.5
        
        # Inverse transform with original phase
        mag = np.sqrt(np.maximum(power_new, 0))
        phase = np.angle(A)
        A_new = mag * (np.cos(phase) + 1j * np.sin(phase))
        
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try with different blending
            a = a_new  # Keep going even if score worsens temporarily
    
    return best_a, best_score

def frequency_dft_init(n, freq_pattern='flat', rng=None):

    if rng is None:

    half = fft_size // 2 + 1
    
    if freq_pattern == 'flat':
        mag = np.ones(half)
    elif freq_pattern == 'raised_cosine':
        f = np.linspace(0, 1, half)
        mag = 0.5 + 0.5 * np.cos(np.pi * f)
    elif freq_pattern == 'gaussian':
        f = np.linspace(0, 1, half)
        mag = np.exp(-2 * (f - 0.5)**2)
    elif freq_pattern == 'random_phase':
        mag = np.ones(half)
    else:
        mag = np.ones(half)
    
    # Add random phase
    phase = 2 * np.pi * rng.random(half)
    mag[0] *= 1.0  # DC component
    
    A = mag * np.exp(1j * phase)
    a = irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def lbgfs_refine(a0, time_limit, T_start=150.0, T_end=0.3, maxiter=10000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-17, 'gtol': 1e-13,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def population_search(initial_solutions, budget, population_size=5, n_generations=None):
    """Population-based search with crossover and mutation."""
    t0 = time.monotonic()
    n = len(initial_solutions[0])
    
    # Initialize population
    population = []
    for sol in initial_solutions[:population_size]:
        population.append(np.array(sol, dtype=np.float64).copy())
    
    # Pad with variations if needed
    rng = np.random.default_rng(99)
    while len(population) < population_size:
        base = population[rng.integers(len(population))]
        noisy = base + rng.random(n) * 0.02
        noisy = np.maximum(noisy, 0)
        noisy /= noisy.sum()
        population.append(noisy)
    
    # Evaluate initial population
    scores = [2.0 * n * fast_convolve(pop).max() for pop in population]
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_a = population[best_idx].copy()
    
    gen = 0
    max_gens = n_generations if n_generations else 10000
    
    while time.monotonic() - t0 < budget and gen < max_gens:
        gen += 1
        # Tournament selection
        def tournament(k=3):
            candidates = rng.integers(population_size, size=k)
            return candidates[np.argmin([scores[i] for i in candidates])]
        
        # Crossover
        p_parent1_idx = tournament()
        p_parent2_idx = tournament()
        p1 = population[p_parent1_idx]
        p2 = population[p_parent2_idx]
        
        # Blend crossover
        alpha = rng.uniform(0.2, 0.8)
        child = alpha * p1 + (1 - alpha) * p2
        
        # Mutation
        if rng.random() < 0.5:
            # Gaussian mutation
            child += rng.normal(0, 0.01, n)

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

def lbgfs_multistage(a0, time_budget, T_start=300.0, T_end=0.1, maxiter=20000):
    """L-BFGS with multi-stage LogSumExp temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_budget:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_budget:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Spectral Peak Shaving (NEW) ===
    print(f"\n=== Phase 1: Spectral Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sf, _ = spectral_peak_shave(best_seq, t_alloc, peak_fraction=0.35, alpha_init=0.5, alpha_end=0.03)
        try_update(a_sf, "peak_shave_1")

        # Also try water-filling
        rem = deadline - time.monotonic()
        t_wf = min(rem * 0.08, 100.0)
        if t_wf > 10:
            a_wf, _ = spectral_water_fill(best_seq, t_wf, strength=0.25)
            try_update(a_wf, "water_fill_1")

    # === Phase 2: Mirror Descent with Nesterov momentum ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_nesterov(best_seq, t_alloc, n_active_frac=0.05)
        try_update(a_md, "mirror_desc_1")

    # === Phase 3: L-BFGS multi-stage ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s, T: 300->0.1")
        a_lb = lbgfs_multistage(best_seq, t_alloc, T_start=300.0, T_end=0.1, maxiter=20000)
        try_update(a_lb, "lbgfs_1")

    # === Phase 4: Coordinate Descent (NEW) ===
    print(f"\n=== Phase 4: Coordinated Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinated_descent(best_seq, t_alloc, n_sweeps=30)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Multi-start diversification ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, 4 starts")
        a_ms, _ = multi_start_refine(best_seq, t_alloc, n_starts=4)
        try_update(a_ms, "multi_start")

    # === Phase 6: Frequency-domain initialized exploration ===
    print(f"\n=== Phase 6: Freq Init (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_total = min(rem * 0.12, 100.0)
        if t_total > 20:
            print(f"  Budget: {t_total:.0f}s")
            for pat in ['flat', 'raised_cosine']:
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                t_sub = min(t_total * 0.5, rem - 10)
                if t_sub < 10:
                    break
                freq_init = frequency_domain_init(n_best, rng, pattern=pat)
                # Quick spectral refine then mirror descent
                a_f1, _ = spectral_peak_shave(freq_init, min(t_sub * 0.4, 30), peak_fraction=0.3, alpha_init=0.4)
                try_update(a_f1, f"freq_{pat}_sf")
                rem = deadline - time.monotonic()
                if rem > 15:
                    a_f2, _ = mirror_descent_nesterov(a_f1, min(t_sub * 0.5, 40), n_active_frac=0.04)
                    try_update(a_f2, f"freq_{pat}_md")

    # === Phase 7: Alternate lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.25, 120.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolation improved: {sc_alt:.6f}")

        # Spectral + gradient at new length
        t_sf = t_total * 0.4
        a_alt, _ = spectral_peak_shave(cur_best, t_sf, peak_fraction=0.3, alpha_init=0.4)
        try_update(a_alt, f"sf_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_md = min(rem * 0.4, 50.0)
            if t_md > 10:
                a_alt2, _ = mirror_descent_nesterov(best_seq, t_md, n_active_frac=0.04)
                try_update(a_alt2, f"md_n{alt_n}")

    # === Phase 8: Final polish pipeline ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Quick spectral shave
        t_sf = min(rem * 0.15, 30.0)
        if t_sf > 5:
            a_p1, _ = spectral_peak_shave(best_seq, t_sf, peak_fraction=0.25, alpha_init=0.3, alpha_end=0.02)
            try_update(a_p1, "polish_sf")

        # L-BFGS at low temperature
        rem = deadline - time.monotonic()
        if rem > 15:
            t_lb = min(rem * 0.25, 40.0)
            if t_lb > 5:
                a_p2 = lbgfs_multistage(best_seq, t_lb, T_start=20.0, T_end=0.05, maxiter=5000)
                try_update(a_p2, "polish_lbgfs")

        # Mirror descent final
        rem = deadline - time.monotonic()
        if rem > 10:
            t_md = min(rem * 0.3, 30.0)
            if t_md > 5:
                a_p3, _ = mirror_descent_nesterov(best_seq, t_md, n_active_frac=0.02)
                try_update(a_p3, "polish_md")

        # One more coordinate descent
        rem = deadline - time.monotonic()
        if rem > 10:
            t_cd = min(rem * 0.3, 25.0)
            if t_cd > 5:
                a_p4, _ = coordinated_descent(best_seq, t_cd, n_sweeps=20)
                try_update(a_p4, "polish_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
