"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP fictitious play for global structure
2. FFT-accelerated smoothed subgradient descent (novel: O(n log n) gradient)
3. Fast coordinate descent with momentum for local polishing
4. Adaptive perturbation cycles for global exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_smoothed_grad(a, w):
    """Compute grad = 2 * (a cross-correlate w) using FFT."""
    n = len(a)
    # w is defined on [0, 2n-2]
    # grad_j = 2 * sum_m a_m * w_{j+m} for valid m
    # This is cross-correlation: (a ⋆ w)_j
    # = conv(a, w_flipped) where w_flipped[i] = w[2n-2-i]
    w_flipped = w[::-1]
    # conv(a, w_flipped) has length n + (2n-1) - 1 = 3n-2
    # We want elements [0, n-1] of this convolution
    size = n + len(w_flipped) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_flipped, fft_size)
    result = irfft(A * W, fft_size)[:n]
    return 2.0 * result

def smoothed_subgradient_descent(a0, time_limit, beta_init=30.0, lr_init=0.01):
    """
    FFT-accelerated smoothed subgradient descent on the simplex.
    Uses multiplicative (exponentiated gradient) updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0
    gamma = 1.0  # momentum
    
    a_prev = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax weights over convolution values
        M = conv.max()
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -60, 0)
        w = np.exp(scaled)
        Z = w.sum()
        if Z < 1e-30:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            Z = 1.0
        w = w / Z
        
        # Compute gradient via FFT
        grad = compute_smoothed_grad(a, w)
        
        # Nesterov-like momentum
        if it == 1:
            a_prev = a.copy()
        
        # Multiplicative update (mirror descent on simplex)
        # a_new_i = a_i * exp(-lr * grad_i)
        log_a = np.log(a + 1e-300)
        log_a -= lr * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)
        
        # Update
        a_prev = a.copy()
        a = a_new
        
        # Recompute convolution
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 20 == 0:
            beta *= 1.01
        
        # Update momentum (Polyak)
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP fictitious play for global structure discovery
2. FFT-accelerated smoothed subgradient descent (O(n log n) gradient computation)
3. Fast coordinate descent with exact 1D optimization for local polishing
4. Peak-targeted perturbation cycles for escaping local optima
5. Adaptive time allocation across phases
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_fft_gradient(a, w):
    """
    Compute grad_j = 2 * sum_m a[m] * w[m+j] efficiently via FFT.
    w is defined on [0, 2n-2] (the convolution lags).
    Returns gradient of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(a_rev, w) has length n + (2n-1) - 1 = 3n-2
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    result = irfft(A * W, fft_size)
    # c_q = result[n-1+q] for q in [0, n-1]
    c = result[n - 1: 2 * n - 1]
    return 2.0 * c


def smoothed_subgradient_descent(a0, time_limit, beta_init=20.0, lr_init=0.02):
    """
    FFT-accelerated smoothed subgradient descent on the simplex.
    Uses multiplicative (exponentiated gradient) updates with adaptive parameters.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        M = conv.max()
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -60, 0)
        w = np.exp(scaled)
        Z = w.sum()
        if Z < 1e-30:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            Z = 1.0
        w = w / Z

        grad = compute_fft_gradient(a, w)

        # Multiplicative update
        log_a = np.log(a + 1e-300)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)

        a = a_new
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 80:
            lr *= 0.85
            beta *= 1.4
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 30 == 0:
            beta *= 1.005

    return best_a, best_score


def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with systematic step size exploration.
    Tracks convolution incrementally for speed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]

            # Try various step sizes
            best_delta = 0.0
            best_new_max = M

            step_sizes = [-0.5, -0.3, -0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.3, 0.5, -0.99]
            for f in step_sizes:
                delta = f * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15 or new_val > 1.0:
                    continue

                new_aff = affected + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            print(f"    FCD {label} sweep={sweeps}: best={best_score:.6f} M={M:.10f}")

    return best_a, best_score

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            best_delta = 0.0
            best_new_max = M

            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 1.002, 1.005, 1.01, 1.02]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_aff = affected.copy() + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta
                changes = 2.0 * best_delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score


def build_conv_matrix(n):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b) for fixed a."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = 1.0
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLOP, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, T, solver):
    """Given fixed a (unit sum), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    T_a = a_fixed[np.arange(n)][:, None]  # We need T * diag(a) effectively
    # T @ (a * b) won't work directly. We need (T_a) @ b where T_a[i,j] = a_i if j in proper range
    # Actually conv(a,b)_k = sum_j a_j * b_{k-j}, so the matrix is different
    # Let's build it properly: M[k,j] = a_{k-j} if 0<=k-j<n else 0
    # This is the same as: for each j, M[j:j+n, j] = a
    M = T * a_fixed[np.arange(n)]  # Scale columns of T by a
    # Wait, T[j:j+n, j] = a (the sequence), not 1. Let me rebuild.
    
    # Actually, my T above has T[j:j+n, j] = 1. I need T2[j:j+n, j] = a.
    # So M = T * a (column scaling)
    M = T * a_fixed[np.arange(n)]
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [M @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLOP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_fictitious_play(n, time_limit, initial, solver):
    """
    LP-based fictitious play with Nesterov averaging.
    """
    t0 = time.monotonic()
    a = np.maximum(np.asarray(initial, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()

    # Build the convolution matrix (Toeplitz structure)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = 1.0

    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP for b (best response to a)
        b_new = solve_lp_best_response(a, T, solver)
        if b_new is not None:
            b = b_new

        # Solve LP for a (best response to b)
        a_new = solve_lp_best_response(b, T, solver)
        if a_new is not None:
            a = a_new

        # Nesterov averaging
        gamma_it = 2.0 / (iters + 2.0)
        a_avg = (1.0 - gamma_it) * a_avg + gamma_it * a
        b_avg = (1.0 - gamma_it) * b_avg + gamma_it * b

        # Evaluate candidates
        for cand in [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            score = 2.0 * n * fast_convolve(cand_n).max()
            if score < best_score:
                best_score = score
                best_a = cand_n.copy()

        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Initialize with bimodal distribution."""
    center1 = n * rng.uniform(0.2, 0.35)
    center2 = n * rng.uniform(0.65, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_quadratic_residue(n, rng):
    """Initialize based on quadratic residue structure."""
    p = n
    a = np.zeros(p)
    for x in range(p):
        a[x] = 1.0
        if x > 0:
            # Legendre symbol-like
            a[x] = 1.0 if (x * x) % p < p // 2 else 0.3
    # Add some randomness
    a += rng.random(p) * 0.2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(p) / p
    return a / s


def peak_targeted_perturb(a0, rng, strength=0.05):
    """Structured perturbation targeting peak regions."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()

    peak_threshold = M * 0.9
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a0[i] * a0[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.2):
    """Randomly restructure the sequence to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.2))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    a_norm = a / s

def naive_subgradient_descent(a0, time_limit, n_peaks=5, lr=0.005):
    """
    Simple subgradient descent using top-k peaks.
    Faster than smoothed version for large n.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        
        # Get top peaks
        top_idx = np.argsort(conv)[::-1][:n_peaks]
        top_vals = conv[top_idx]
        
        # Compute gradient from top peaks
        grad = np.zeros(n)
        for k in top_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * top_vals[k - j]
        
        # Normalize gradient
        g_max = grad.max()
        if g_max < 1e-30:
            break
        grad = grad / g_max
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)
        score = 2.0 * n * conv.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Adaptive lr
        if it % 100 == 0:
            lr *= 0.95
            if lr < 1e-8:
                break

    return best_a, best_score
