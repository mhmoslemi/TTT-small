Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using diverse LP-alternating

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


_lp_solver_cached = None

    if _lp_solver_cached is not None:

    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_logsumexp_obj(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def compute_logsumexp_grad(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def build_conv_matrix(a):

    size = 2 * n - 1
    T = np.zeros((size, n))

    return T

            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = sn
                return sn

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for sn in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sn)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = sn
                return sn
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp(a_fixed, solver=None):
    """Solve LP: min M s.t. T_a @ b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver] + [x for x in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if x != solver]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                bv = np.asarray(b_var.value, dtype=np.float64)
                bv = np.maximum(bv, 0)
                sv = bv.sum()
                if sv < 1e-15:
                    continue
                return bv / sv, float(M_var.value)
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, n, temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(b - mb)
    return (2.0 * n / temp) * (mb + np.log(e.sum()))


def lse_gradient(z, n, temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(b - mb)
    w = e / e.sum()
    ga = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        ga[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(ga, a)
    return a * (ga - S)


def lbgfs_optimize(a0, time_budget, temps=None, maxiter=5000):
    """Multi-stage LogSumExp L-BFGS optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if temps is None:
        temps = [50, 25, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02]
    
    per_stage = max(20, maxiter // len(temps))
    for T in temps:
        if time.monotonic() - t0 > time_budget:
            break
        if time.monotonic() - t0 > time_budget - 5:
            break
            
        def f(z):
            return lse_objective(z, n, T)
        def df(z):
            return lse_gradient(z, n, T)
        
        try:
            res = minimize(f, z, jac=df, method='L-BFGS-B',
                           options={'maxiter': per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': per_stage*200})
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def peak_descent(a0, time_budget, n_steps=50000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        pk = int(np.argmax(b))
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[pk - idx]
        top = idx[np.argsort(contribs)[::-1][:6]]
        
        improved = False
        for pos in top:
            if a[pos] < 1e-10:
                continue
            for f in [0.998, 0.995, 0.99, 0.985, 0.98]:
                at = a.copy()
                at[pos] *= f
                diff = a[pos] - at[pos]
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = at[mask]
                if others.sum() < 1e-15:
                    break
                at[mask] = others + diff * others / others.sum()
                at = at / at.sum()
                sc = 2.0 * n * fast_convolve(at).max()
                if sc < best_score - 1e-14:
                    a = at
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0
    
    return best_a, best_score


def subgradient_method(a0, time_budget, n_steps=50000):
    """Subgradient method with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_convolve(a)
        mb = b.max()
        thresh = mb * 0.9995
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        at = a - lr * g
        at = np.maximum(at, 0)
        st = at.sum()
        if st < 1e-15:
            at = a.copy()
        else:
            at = at / st
        
        sc = 2.0 * n * fast_convolve(at).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = at.copy()
            a = at
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)
    
    return best_a, best_score


def structured_inits(n, rng):

        if a.sum() > 0.01:

    for p in [0.5, 1.5, 2.0]:
        a = x ** p

def evaluate_local(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi
