import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_conv(conv, n):
    """Compute score assuming sum=1 (normalized)."""
    return 2.0 * n * conv.max()


def eval_fast(a):

    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def simplex_project(v):
    """Project vector v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def init_cosine(n, power=2.0):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def init_fourier(n, rng):
    """Initialize using random Fourier features (positive-definite kernel)."""
    n_freqs = int(rng.integers(3, 10))
    freqs = rng.uniform(0, 1, n_freqs) * 2 * np.pi / n
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amps = rng.uniform(0.5, 1.0, n_freqs)
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, p, am in zip(freqs, phases, amps):
        a += am * np.cos(f * t + p)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_sparse_sidon(n, m, rng):

def coord_descent_optimal(a0, time_limit, verbose=True, label="CD"):
    """
    Coordinate descent with optimal delta candidates.
    Uses prefix/suffix max for O(1) unaffected max queries.
    Tries the analytical optimal delta plus heuristic candidates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    # Precompute prefix and suffix max for conv
    # Prefix max: pref[i] = max(conv[0:i])
    # Suffix max: suff[i] = max(conv[i:])
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved_count = 0

        # Recompute prefix/suffix max (O(n) each)
        pref_max = np.maximum.accumulate(conv)
        suff_max = np.maximum.accumulate(conv[::-1])[::-1]

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = pref_max[k_start - 1]
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = suff_max[k_end + 1]
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            if unaff_max >= M - 1e-14:
                # Even if we zero out all affected, the max won't decrease
                # (unless the affected max IS the global max and unaff is close)
                if unaff_max >= M - 1e-12:
                    # Only try if unaff_max is strictly less than M
                    if unaff_max >= M - 1e-12:
                        pass  # still try, might get small improvement

            # Affected conv values
            affected_len = k_end - k_start + 1
            affected = conv[k_start:k_end + 1]
            a_chunk = a[0:affected_len]  # a[k-i] for k in [k_start, k_end]

            # Try candidate deltas
            best_delta = 0.0
            best_new_M = M

            # Candidate 1: optimal delta to make max_affected = unaff_max
            # We need: max_j (affected[j] + 2*delta*a_chunk[j]) = unaff_max
            # For all j: affected[j] + 2*delta*a_chunk[j] <= unaff_max
            # => delta <= (unaff_max - affected[j]) / (2*a_chunk[j]) for a_chunk[j] > 0
            # Also need: a[i] + delta >= 0 => delta >= -a[i]
            # And want max_affected <= unaff_max, i.e., the binding constraint.
            
            positive_mask = a_chunk > 1e-15
            if positive_mask.any():
                upper_bounds = (unaff_max - affected[positive_mask]) / (2.0 * a_chunk[positive_mask])
                delta_opt = upper_bounds.min()
                # Clamp to valid range
                delta_opt = max(delta_opt, -a[i])
                if delta_opt < M * 10:  # sanity check
                    # Compute new max
                    new_aff = affected + 2.0 * delta_opt * a_chunk
                    # Fix quadratic at index i (position 2i in full conv, index i in affected)
                    quad_idx = 2 * i - k_start
                    if 0 <= quad_idx < affected_len:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] = conv[2 * i] + 2.0 * delta_opt * a[i] + delta_opt * delta_opt
                    new_M_candidate = max(unaff_max, float(new_aff.max()))
                    if new_M_candidate < best_new_M - 1e-14:
                        best_new_M = new_M_candidate
                        best_delta = delta_opt

            # Candidate 2: delta = -a[i] (zero out)
            delta = -a[i]
            new_aff = affected - 2.0 * a[i] * a_chunk
            quad_idx = 2 * i - k_start
            if 0 <= quad_idx < affected_len:
                new_aff = new_aff.copy()
                new_aff[quad_idx] = conv[2 * i] - 2.0 * a[i] * a[i]  # delta=-a[i], so 2*delta*a[i]+delta^2 = -2a[i]^2 + a[i]^2 = -a[i]^2
                # Actually: conv[2i] + 2*delta*a[i] + delta^2 = conv[2i] - 2*a[i]^2 + a[i]^2 = conv[2i] - a[i]^2
            new_M_candidate = max(unaff_max, float(new_aff.max()))
            if new_M_candidate < best_new_M - 1e-14:
                best_new_M = new_M_candidate
                best_delta = delta

            # Candidate 3: Fractional steps
            for frac in [0.25, 0.5, 0.75]:
                delta = -a[i] * frac
                new_aff = affected + 2.0 * delta * a_chunk
                if 0 <= quad_idx < affected_len:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M_candidate = max(unaff_max, float(new_aff.max()))
                if new_M_candidate < best_new_M - 1e-14:
                    best_new_M = new_M_candidate
                    best_delta = delta

            # Candidate 4: Small positive steps (if the peak is NOT at this coordinate's position)
            if M > 0:
                for frac in [0.1, 0.3, 0.5]:
                    delta = a[i] * frac
                    new_aff = affected + 2.0 * delta * a_chunk
                    if 0 <= quad_idx < affected_len:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                    new_M_candidate = max(unaff_max, float(new_aff.max()))
                    if new_M_candidate < best_new_M - 1e-14:
                        best_new_M = new_M_candidate
                        best_delta = delta

            # Apply best delta if improvement found
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                changes = 2.0 * best_delta * a_chunk
                if 0 <= quad_idx < affected_len:
                    # The linear approximation already includes 2*delta*a[i] at quad_idx
                    # We need to add delta^2
                    changes = changes.copy()
                    changes[quad_idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved_count += 1

        if improved_count == 0:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f} impr={improved_count}")

    return best_a, best_score

            if k_end < 2 * n - 2:

            affected_len = k_end - k_start + 1
            affected = conv[k_start:k_end + 1]
            a_chunk = a[0:affected_len]
            quad_idx = 2 * i - k_start

            best_delta = 0.0
            best_new_M = M

                    new_aff = affected + 2.0 * delta_opt * a_chunk
                    if 0 <= quad_idx < affected_len:
                        new_aff = new_aff.copy()
                        new_aff[quad_idx] = conv[2 * i] + 2.0 * delta_opt * a[i] + delta_opt * delta_opt
                    new_M_candidate = max(unaff_max, float(new_aff.max()))
                    if new_M_candidate < best_new_M - 1e-15:
                        best_new_M = new_M_candidate
                        best_delta = delta_opt

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_chunk
                if 0 <= quad_idx < affected_len:

                    changes[quad_idx] += best_delta * best_delta

def fw_alternating(a_init, time_limit, n_steps_per_side=200, verbose=True, label="FW"):
    """
    Frank-Wolfe alternating minimization.
    Optimizes a given b, then b given a, iteratively.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    b_conv = convolve_ab(a, b)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Optimize b given a
        for step in range(n_steps_per_side):
            k_star = int(np.argmax(b_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            # Away step
            if step >= 30:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = b * grad_full
                i_away = int(np.argmax(away_scores))
                if b[i_away] > 1e-10 and grad_full[i_away] > 1e-12:
                    t_away = min(t * 0.4, b[i_away] * 0.7)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)
                    b_conv[ks_s:ks_e + 1] -= t_away * a
                    b[i_away] -= t_away
                    b[i_away] = max(b[i_away], 0.0)

            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)
            b_conv = (1.0 - t) * b_conv
            b_conv[ks_s:ks_e + 1] += t * a
            b = (1.0 - t) * b
            b[j_star] += t

        # Evaluate b
        b_score = 2.0 * n * fast_convolve(b).max()
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()

        # Optimize a given b
        a_conv = convolve_ab(b, a)
        for step in range(n_steps_per_side):
            k_star = int(np.argmax(a_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            if step >= 30:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = a * grad_full
                i_away = int(np.argmax(away_scores))
                if a[i_away] > 1e-10 and grad_full[i_away] > 1e-12:
                    t_away = min(t * 0.4, a[i_away] * 0.7)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)
                    a_conv[ks_s:ks_e + 1] -= t_away * b
                    a[i_away] -= t_away
                    a[i_away] = max(a[i_away], 0.0)

            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)
            a_conv = (1.0 - t) * a_conv
            a_conv[ks_s:ks_e + 1] += t * b
            a = (1.0 - t) * a
            a[j_star] += t

        # Evaluate a
        a_score = 2.0 * n * fast_convolve(a).max()
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

        # Evaluate average
        avg = 0.5 * (a + b)
        avg = np.maximum(avg, 0)
        s_avg = avg.sum()
        if s_avg > 1e-15:
            avg = avg / s_avg
            avg_score = 2.0 * n * fast_convolve(avg).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg.copy()

        # Sync occasionally
        a = best_a.copy()
        b = best_a.copy()
        b_conv = convolve_ab(a, b)

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    {label} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the official evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    - Phase 1: Fast exploration with FW from diverse initializations
    - Phase 2: Smoothed mirror descent for refinement
    - Phase 3: Aggressive coordinate descent
    - Phase 4: Periodic perturbation restarts
    - Phase 5: Fine polishing
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = init_uniform(n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n={n}, score={best_score:.6f}")

    print(f"Starting optimization with n={n}, best_score={best_score:.6f}")

    # === Phase 1: FW Global Optimization ===
    print(f"\n=== Phase 1: FW (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    n_init = min(int(remaining * 0.25), 300)
    if n_init > 30:
        # Try parent first
        a_fw, s_fw = fw_alternating(best_seq, min(n_init * 0.4, 120), n_steps_per_side=100,
                                    verbose=True, label="FW_parent")
        best_seq, best_score = check_and_update(a_fw, best_seq, best_score, "fw_parent")

        # Try diverse inits
        remaining = deadline - time.monotonic()
        if remaining > 60:
            inits = [
                ("cos2", init_cosine(n, 2.0)),
                ("cos3", init_cosine(n, 3.0)),
                ("fourier1", init_fourier(n, rng)),
                ("fourier2", init_fourier(n, rng)),
                ("twopenk", init_two_peak(n, rng)),
            ]
            for name, init in inits:
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                t_alloc = min(remaining * 0.2, 60)
                a_fw, s_fw = fw_alternating(init, t_alloc, n_steps_per_side=80,
                                            verbose=False, label=f"FW_{name}")
                best_seq, best_score = check_and_update(a_fw, best_seq, best_score, f"fw_{name}")

    # === Phase 2: Smoothed Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_md = min(remaining * 0.25, 200)
        a_md, s_md = mirror_descent_smoothed(best_seq, t_md, beta_init=30.0, beta_final=300.0,
                                             lr_init=0.08, verbose=True, label="MD_main")
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror_descent")

        # Run MD with different parameters
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_md2 = min(remaining * 0.2, 100)
            a_md2, s_md2 = mirror_descent_smoothed(best_seq, t_md2, beta_init=10.0, beta_final=100.0,
                                                   lr_init=0.05, verbose=True, label="MD_smooth")
            best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "mirror_descent2")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_cd = min(remaining * 0.3, 300)
        a_cd, s_cd = coord_descent_optimal(best_seq, t_cd, verbose=True, label="CD_main")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_descent")

    # === Phase 4: Perturbation Restarts ===
    print(f"\n=== Phase 4: Perturbation Restarts (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    restart_count = 0
    while remaining > 120 and restart_count < 5:
        restart_count += 1
        # Create perturbed version
        noise_level = 0.02 / (1 + restart_count * 0.5)
        perturbed = best_seq + rng.standard_normal(len(best_seq)) * noise_level
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()

        # Short optimization
        t_alloc = min(remaining * 0.15, 60)
        a_p, s_p = coord_descent_optimal(perturbed, t_alloc, verbose=False, label=f"restart_{restart_count}")
        best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"restart_{restart_count}")

        # Also try MD from perturbed
        remaining = deadline - time.monotonic()
        if remaining > 80:
            t_md_r = min(remaining * 0.1, 40)
            a_mr, s_mr = mirror_descent_smoothed(perturbed, t_md_r, beta_init=50.0, beta_final=200.0,
                                                 lr_init=0.05, verbose=False, label=f"MD_r{restart_count}")
            best_seq, best_score = check_and_update(a_mr, best_seq, best_score, f"md_restart_{restart_count}")

        remaining = deadline - time.monotonic()

    # === Phase 5: Fine Polishing ===
    print(f"\n=== Phase 5: Fine Polishing (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_fine = min(remaining * 0.5, 200)
        a_fine, s_fine = fine_cd_polish(best_seq, t_fine, verbose=True, label="FineCD")
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

    # === Phase 6: Final CD pass ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_final = min(remaining * 0.7, 100)
        a_fin, s_fin = coord_descent_optimal(best_seq, t_final, verbose=True, label="FinalCD")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

    # === Phase 7: Ultra-fine polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        t_ultra = min(remaining * 0.8, 60)
        a_ultra, s_ultra = fine_cd_polish(best_seq, t_ultra, verbose=True, label="UltraFine")
        best_seq, best_score = check_and_update(a_ultra, best_seq, best_score, "ultra_fine")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
    fs = float(evaluate_sequence(np.maximum(best_seq, 0).tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in np.maximum(best_seq, 0)]
