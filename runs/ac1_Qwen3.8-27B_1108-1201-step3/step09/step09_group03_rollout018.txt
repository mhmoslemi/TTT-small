def fast_convolve_v2(a):
    """Linear auto-convolution via FFT (optimized)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab_v2(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

        c = np.zeros(n)
        for idx in range(len(peak_idx)):
            if w[idx] < 1e-14:
                continue
            k = int(peak_idx[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            c[i_start:i_end+1] += w[idx] * a[k - i_end : k - i_start + 1][::-1]
        
        # Update: a_i *= exp(-eta * c_i) (push down where constraint is tight)
        a = a * np.exp(-eta * c)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        conv = fast_convolve_v2(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-10:
                break
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT-MD it={it} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.6f}")
    
    return best_a, best_score

def coord_descent_v2(a0, time_limit=300, verbose=True):
    """
    Coordinate descent with adaptive step selection.
    Key improvement: use binary search for the optimal step in each coordinate.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve_v2(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            # The range of conv that's affected by changing a[i]
            k_start = i
            k_end = min(i + n - 1, 2*n - 2)
            affected_len = k_end - k_start + 1
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2*n - 2:
                unaff_max = max(unaff_max, conv[k_end+1:].max())
            
            # a[k-i] for k in [k_start, k_end]: this is a[0 : affected_len] shifted
            # k_start = i, so k - i ranges from 0 to affected_len-1
            a_shifted = a[:affected_len]  # a[k-i] = a[k_i] where k_i = k - i, k in [i, i+affected_len-1]
            
            # conv[k] = sum_j a[j]*a[k-j]
            # When we change a[i] by delta:
            # conv[k] changes by: delta * (a[k-i] + a[i] - delta) for k = 2i (the "diagonal")
            #                            delta * a[k-i] + delta * a[k-i] = 2*delta*a[k-i] for k != 2i
            # More precisely:
            # conv[k] = sum_j a[j]*a[k-j]
            # = a[i]*a[k-i] + a[k-i]*a[i] + sum_{j != i, k-j != i} a[j]*a[k-j]
            # = 2*a[i]*a[k-i] + rest (when i != k-i, i.e., k != 2i)
            # = a[i]^2 + rest (when k = 2i)
            
            # After changing a[i] -> a[i] + delta:
            # conv[k] -> 2*(a[i]+delta)*a[k-i] + rest = conv[k] + 2*delta*a[k-i] (for k != 2i)
            # conv[2i] -> (a[i]+delta)^2 + rest = conv[2i] + 2*delta*a[i] + delta^2
            
            # So the affected conv values become:
            # new_conv[k] = conv[k] + 2*delta*a[k-i] (for k != 2i)
            # new_conv[2i] = conv[2i] + 2*delta*a[i] + delta^2
            
            # We want to find delta that minimizes max(unaff_max, max of new_conv)
            
            # The current affected values:
            cur_affected = conv[k_start:k_end+1]
            
            # Find the best delta by trying several candidates
            # The optimal delta is where the peak in the affected region equals unaff_max
            # or where two peaks in the affected region are equal.
            
            # Simple approach: try several deltas
            best_delta = 0.0
            best_new_M = M
            
            # The "natural" step: make the peak of new_affected = unaff_max
            # new_affected[k] = cur_affected[k] + 2*delta*a_shifted[k-k_start]
            # (approximately, ignoring the quadratic term at k=2i)
            # Peak is at: argmax_k (cur_affected[k] + 2*delta*a_shifted[k-k_start])
            # If delta < 0: peak moves towards where a_shifted is large
            # If delta > 0: peak moves towards where a_shifted is small
            
            # Try: delta that makes max(new_affected) = unaff_max
            # This requires: max_k (cur_affected[k] + 2*delta*a_shifted[k]) = unaff_max
            # For the k where cur_affected is max: cur_affected_max + 2*delta*a_shifted[argmax] = unaff_max
            # delta = (unaff_max - cur_affected_max) / (2 * a_shifted[argmax])
            
            cur_max_idx = int(np.argmax(cur_affected))
            cur_max_val = cur_affected[cur_max_idx]
            
            if cur_max_val > unaff_max and a_shifted[cur_max_idx] > 1e-15:
                # We need delta < 0 (reduce a[i])
                delta_candidate = (unaff_max - cur_max_val) / (2.0 * a_shifted[cur_max_idx])
                delta_candidate = max(delta_candidate, -a[i])  # can't go below 0
                # Check a few candidates around this
                candidates = [delta_candidate, delta_candidate * 0.5, delta_candidate * 0.8,
                             delta_candidate * 1.2, -a[i] * 0.5, -a[i] * 0.1, 0.0]
            else:
                candidates = [0.0, a[i] * 0.1, a[i] * 0.3, a[i] * 0.5]
            
            for delta in candidates:
                new_val = a[i] + delta
                if new_val < -1e-12:
                    continue
                
                # Compute new affected values
                new_affected = cur_affected + 2.0 * delta * a_shifted
                # Fix the quadratic term at k = 2i
                diag_idx = 2*i - k_start
                if 0 <= diag_idx < affected_len:
                    new_affected = new_affected.copy()
                    new_affected[diag_idx] = conv[2*i] + 2.0*delta*a[i] + delta*delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)
                
                # Update conv incrementally
                changes = 2.0 * best_delta * a_shifted
                diag_idx = 2*i - k_start
                if 0 <= diag_idx < affected_len:
                    changes = changes.copy()
                    changes[diag_idx] = 2.0*best_delta*(a[i] - best_delta) + best_delta*best_delta
                
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")
    
    return best_a, best_score

def multi_peak_md(a0, time_limit=300, lr_init=0.05, top_k=30, verbose=True):
    """Multi-peak subgradient mirror descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve_v2(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-9:
                break
        
        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end+1] += w[idx] * 2.0 * a[k - i_end : k - i_start + 1][::-1]
        
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 0.0)
        sn = a.sum()
        if sn < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / sn
        
        avg_a = (it * avg_a + a) / (it + 1)
        conv = fast_convolve_v2(a)
        
        if it % 200 == 0:
            avg_score = 2.0 * n * fast_convolve_v2(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

            elapsed = time.monotonic() - t0
