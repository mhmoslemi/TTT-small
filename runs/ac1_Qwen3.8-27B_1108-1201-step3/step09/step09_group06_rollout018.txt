import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():

    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CBC, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, time_limit=5.0):

        solver = get_lp_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    t_start = time.monotonic()
    try:
        prob.solve(solver=solver)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC]:

            if time.monotonic() - t_start > time_limit:
                break
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient_exp(a, beta=50.0):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    shifted = b - max_b
    weights = np.exp(shifted * beta)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(b)
        weights[np.argmax(b)] = 1.0
    else:
        weights = weights / w_sum
    g = np.zeros(n)
    for k in range(len(b)):
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += weights[k] * a[k - idx]
    g *= 2.0
    return g, max_b


def mirror_descent_adaptive(a0, time_limit, beta_start=20.0, beta_end=200.0, n_steps=300000):

    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.8
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, step / (0.8 * n_steps))
        beta = beta_start * (beta_end / beta_start) ** progress

        g, max_b = compute_subgradient_exp(a, beta=beta)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-10:

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.05)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)

            z = z_new
            no_improve = 0
            lr = min(lr * 1.08, 3.0)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.7
                no_improve = 0

            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f} beta={beta:.1f}")

            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def newton_kkt_refine(a0, time_limit, max_iters=200, active_frac=1.0):
    """Newton refinement solving KKT system on active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n, 2.0 * n
    a = a / s
    b = fast_convolve(a)
    M = b.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Identify active set
        n_active = max(int(n * active_frac), 10)
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:n_active]

        if len(active) > n:
            active = active[:n]

        # Build Jacobian for active set
        m = len(active)
        J = np.zeros((m, n))
        for i, k in enumerate(active):
            for j in range(n):
                diff = k - j
                if 0 <= diff < n:
                    J[i, j] = a[diff]
                    if k != 2 * j:
                        J[i, j] += a[diff]

        # Build KKT system: [J, -1; 1^T, 0] [da; dM] = [M - b_active; 1 - sum(a)]
        if m == n:
            G = np.zeros((n + 1, n + 1))
            G[:n, :n] = J
            G[:n, n] = -1.0
            G[n, :n] = 1.0
            rhs = np.zeros(n + 1)
            rhs[:n] = M - b[active]
            rhs[n] = 1.0 - a.sum()

            try:
                sol = np.linalg.solve(G, rhs)
            except np.linalg.LinAlgError:
                try:
                    sol = np.linalg.lstsq(G, rhs, rcond=None)[0]
                except Exception:
                    break

            da = sol[:n]
            dM = sol[n]
        elif m < n:
            # Underdetermined: use minimum norm solution
            # G^T G is (n+1) x (n+1), G^T rhs is (n+1)
            G = np.zeros((m + 1, n + 1))
            G[:m, :n] = J
            G[:m, n] = -1.0
            G[m, :n] = 1.0
            rhs = np.zeros(m + 1)
            rhs[:m] = M - b[active]
            rhs[m] = 1.0 - a.sum()

            GtG = G.T @ G + 0.1 * np.eye(n + 1)
            Gtrhs = G.T @ rhs
            try:
                sol = np.linalg.solve(GtG, Gtrhs)
            except np.linalg.LinAlgError:
                break
            da = sol[:n]
            dM = sol[n]
        else:
            break

        # Line search
        found = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_trial = a + t * da
            a_trial = np.maximum(a_trial, 0)
            sa = a_trial.sum()
            if sa < 1e-15:
                continue
            a_trial = a_trial / sa
            b_trial = fast_convolve(a_trial)
            score_trial = 2.0 * n * b_trial.max()
            if score_trial < score - 1e-12:
                a = a_trial
                found = True
                break

        if not found:
            break

    return best_a, best_score


def local_swap_search(a0, time_limit, n_steps=50000, seed=12345):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        for si in top_src:
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    amount = a[si] * frac
                    if amount < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve_count = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve_count += 1
            # Random perturbation to escape
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * rng.random() * 0.05
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve_count = 0
                    break

        if no_improve_count > 5000 and step > 10000:
            break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        # Hann-like window
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = np.maximum(a, 0.01)
    else:
        # Two-tone: 1 + alpha*cos(2*pi*j/n)
        alpha = rng.uniform(0.2, 0.6)
        x = np.linspace(0, 1, n)
        a = 1.0 + alpha * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def symmetric_project(a):
    """Project onto symmetric sequences: p_j = (a_j + a_{n-1-j})/2."""
    n = len(a)
    s = (a + a[::-1]) / 2.0
    s = np.maximum(s, 0)
    ss = s.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return s / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update_callback(a, label):
        return a, label

    # Use a list to track best (workaround for no closures)
    state = {'best_seq': best_seq, 'best_score': best_score}

    def try_update(a, label):
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < state['best_score']:
            state['best_score'] = sc
            state['best_seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: Mirror descent (adaptive with exponential weighting)
    print(f"\n=== Phase 1: Mirror Descent (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_adaptive(state['best_seq'], t_alloc, beta_start=15.0, beta_end=300.0)
        try_update(a_md, "mirror_desc_1")

    # Phase 2: Nesterov LP
    print(f"\n=== Phase 2: Nesterov LP (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(state['best_seq'], t_alloc, solver=solver, n_max_iters=200000, lp_time=3.0)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 3: L-BFGS LogSumExp
    print(f"\n=== Phase 3: L-BFGS (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 500->0.05")
        a_lb = lbgfs_refine(state['best_seq'], t_alloc, T_start=500.0, T_end=0.05, maxiter=15000)
        try_update(a_lb, "lbgfs_1")

    # Phase 4: Newton KKT refinement (NEW)
    print(f"\n=== Phase 4: Newton KKT (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_nw, _ = newton_kkt_refine(state['best_seq'], t_alloc, max_iters=300, active_frac=1.0)
        try_update(a_nw, "newton_kkt_1")
        # Second Newton pass with slightly relaxed active set
        a_nw2, _ = newton_kkt_refine(state['best_seq'], min(rem * 0.05, 60.0), max_iters=200, active_frac=0.8)
        try_update(a_nw2, "newton_kkt_2")

    # Phase 5: Second LP round
    print(f"\n=== Phase 5: LP Round 2 (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(state['best_seq'], t_alloc, solver=solver, n_max_iters=100000, lp_time=2.5)
        try_update(a_lp2, "nesterov_lp_2")

    # Phase 6: Local swap
    print(f"\n=== Phase 6: Local Swap (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ls, _ = local_swap_search(state['best_seq'], t_alloc, n_steps=50000)
        try_update(a_ls, "local_swap")

    # Phase 7: Newton polish
    print(f"\n=== Phase 7: Newton Polish (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_nw3, _ = newton_kkt_refine(state['best_seq'], t_alloc, max_iters=500, active_frac=1.0)
        try_update(a_nw3, "newton_polish")

    # Phase 8: Alternative lengths
    print(f"\n=== Phase 8: Alt Lengths (best={state['best_score']:.6f}) ===")
    alt_lengths = []
    for mult in [1.2, 1.5, 0.85, 2.0, 1.1]:
        alt_n = int(n_best * mult)
        if 480 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.30, 180.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(state['best_seq'], alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < state['best_score']:
            state['best_score'] = sc_alt
            state['best_seq'] = init_alt.copy()

        t_lp = t_total * 0.50
        a_alt, _ = nesterov_lp_refine(cur_best, t_lp, solver=solver, n_max_iters=50000, lp_time=2.0)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and state['best_seq'] is not None and len(state['best_seq']) == alt_n:
            t_nw = min(rem * 0.2, 40.0)
            if t_nw > 10:
                a_alt_nw, _ = newton_kkt_refine(state['best_seq'], t_nw, max_iters=200, active_frac=1.0)
                try_update(a_alt_nw, f"newton_n{alt_n}")

    # Phase 9: Final combined polish
    print(f"\n=== Phase 9: Final Polish (best={state['best_score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = lbgfs_refine(state['best_seq'], t_fin, T_start=20.0, T_end=0.05, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_nw_fin = min(rem * 0.20, 50.0)
        a_nw_fin, _ = newton_kkt_refine(state['best_seq'], t_nw_fin, max_iters=500, active_frac=1.0)
        try_update(a_nw_fin, "final_newton")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ls_fin = min(rem * 0.15, 30.0)
        a_ls_fin, _ = local_swap_search(state['best_seq'], t_ls_fin, n_steps=10000)
        try_update(a_ls_fin, "final_swap")

    # Final Newton pass
    rem = deadline - time.monotonic()
    if rem > 10:
        a_nw_final, _ = newton_kkt_refine(state['best_seq'], min(rem * 0.3, 30.0), max_iters=300, active_frac=1.0)
        try_update(a_nw_final, "ultimate_newton")

    # Final verification
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
