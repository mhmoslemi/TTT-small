def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def compute_subgradient(a, conv, n_peaks=10, temperature=0.01):
    """Compute subgradient of max(conv) from active peaks."""
    n = len(a)
    M = float(conv.max())
    
    # Find active peaks with soft weighting
    threshold = M - temperature * M
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Softmax weights
    vals = conv[active]
    exp_vals = np.exp((vals - vals.max()) / (temperature * M + 1e-20))
    w = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(active)):
        if w[idx] < 1e-14:
            continue
        k = int(active[idx])
        # grad[i] += 2 * w[idx] * a[k-i] for valid k-i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        partners = k - np.arange(lo, hi + 1)  # k - i for i in [lo, hi]
        grad[lo:hi + 1] += 2.0 * w[idx] * a[partners]
    
    return grad, M

        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
            # Slightly increase lr if we're making progress
            lr = min(lr * 1.01, 0.02)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
            # Try a smaller step
            if lr > 0.0001:
                lr *= 0.95
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASF {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score, iters


def multiplicative_weights_flow(a0, time_limit, verbose=False, label=""):
    """Mirror descent with multiplicative updates (natural for non-negative variables)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    # Ensure strictly positive for log
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.01
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        grad, M = compute_subgradient(a, conv, n_peaks=10, temperature=0.005)
        
        # Multiplicative update: a[i] *= exp(-lr * grad[i])
        update = np.exp(-lr * grad)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Blend with original for stability
        alpha = 0.5
        a_blended = (1 - alpha) * a + alpha * a_new
        a_blended = np.maximum(a_blended, 1e-15)
        a_blended = a_blended / a_blended.sum()
        
        conv_new = fast_convolve(a_blended)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_blended.copy()
            a = a_blended
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.9
            elif no_improve > 200:
                break
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")

    # Final: allow zeros
    best_a = np.maximum(best_a, 0.0)
    best_a = best_a / best_a.sum()
    return best_a, best_score, iters


def fictitious_play_combined(a0, time_limit, solver=None, verbose=False, label=""):
    """Combined fictitious play: alternates between min-peak and max-sum LP formulations."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    decay = 5.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        # Alternate between formulations
        if iters % 3 == 0:
            # SOTA variant: maximize sum subject to bound
            M_bound = float(fast_convolve(a).max()) * 1.01
            b = solve_lp_max_sum(a, M_bound, solver)
        else:
            # Standard: minimize peak
            b, M_val = solve_lp_min_peak(a, solver)
        
        if b is None:
            continue

        # Line search
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = decay / (iters + decay)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        # Check candidates
        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_asymmetric(n, rng):
    """Asymmetric initialization (mass shifted)."""
    t = np.arange(n, dtype=np.float64)
    shift = rng.uniform(0.2, 0.8)
    center = n * shift
    width = n * rng.uniform(0.15, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(3, 15)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_harmonics, n // 2), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_one_sided(n, rng):
    """One-sided (ramp-like) initialization - exploits non-symmetry insight."""
    t = np.arange(n, dtype=np.float64)
    # Various one-sided shapes
    shape = rng.integers(0, 4)
    if shape == 0:
        a = np.sin(np.pi * t / (2 * n)) ** 2  # ramp up
    elif shape == 1:
        a = np.cos(np.pi * t / (2 * n)) ** 2  # ramp down
    elif shape == 2:
        a = (t / n) ** rng.uniform(1.0, 3.0)  # power ramp
    else:
        center = n * rng.uniform(0.1, 0.4)
        width = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=8, verbose=False):
    """Basin hopping with diverse perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.6, 80.0)

        # Different perturbation strategies
        strat = p % 4
        if strat == 0:
            # Gaussian noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strat == 1:
            # Peak-breaking: reduce values at positions contributing to top peaks
            conv_p = fast_convolve(a)
            top_k = min(5, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 + rng.standard_normal() * 0.08)
        elif strat == 2:
            # Local permutation: swap adjacent blocks
            perturbed = a.copy()
            n_blocks = rng.integers(3, 8)
            block_size = max(1, n // (n_blocks * 4))
            for _ in range(n_blocks):
                i = rng.integers(0, n - block_size - 1)
                j = rng.integers(0, n - block_size - 1)
                perturbed[i:i+block_size], perturbed[j:j+block_size] = \
                    perturbed[j:j+block_size].copy(), perturbed[i:i+block_size].copy()
        else:
            # Multiplicative perturbation
            perturbed = a * np.exp(rng.standard_normal(n) * 0.05)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} (strat={strat}): improved to {best_score:.8f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
