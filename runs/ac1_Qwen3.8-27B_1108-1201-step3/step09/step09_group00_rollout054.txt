The Nesterov acceleration works well when the "landscape" is smooth, but for this problem, the landscape has many plateaus (regions where the same set of peaks is active). The key is to detect when we're on a plateau and take a different action.

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    momentum_window = 10
    improvements = []

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, active_set = solve_lp_fixed_one(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)

        # Nesterov virtual variable update
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Nesterov a update
        w_old = it + 1
        w_new = (it + 2) * alpha
        a = (w_old * a + w_new * b_new) / (w_old + w_new)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = compute_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        improvements.append(prev_best - best_score)
        if len(improvements) > momentum_window:
            improvements.pop(0)

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-11:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Structured perturbation using cosines
                freq = rng.integers(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                x = np.linspace(0, freq * 2 * np.pi, n)
                noise_scale = 0.03 / np.sqrt(stall_count)
                noise = noise_scale * np.cos(x + phase)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall_count = 0
                improvements = []

            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, max_iters=200000, eta0=0.1):
    """Multiplicative (mirror descent) updates with weighted subgradient on peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation
        c = fast_convolve(a)
        max_c = c.max()

        # Find active peaks (within 0.5% of max)
        threshold = max_c * 0.995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Weight by peak height (higher peaks get more weight)
        peak_vals = c[active]
        weights = peak_vals / peak_vals.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for idx in range(len(active)):
            k = active[idx]
            w = weights[idx]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += w * 2.0 * a[k - j]

        # Multiplicative update (mirror descent with exponential family)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        # Normalize
        log_a_new = log_a_new - log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Increase eta slightly when making progress
            eta = min(eta * 1.001, 0.5)
        else:
            no_improve += 1
            if no_improve >= 50:
                eta *= 0.9
                no_improve = 0
                # Occasionally add small noise
                if rng.random() < 0.3:
                    a = a + rng.standard_normal(n) * 0.005
                    a = np.maximum(a, 0)
                    a = a / a.sum()
            elif no_improve >= 200:
                break

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score


def peak_matching_refine(a0, time_limit, max_iters=50000):
    """Iteratively equalize the top peaks by solving a small linear system."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        min_c_active = c.max() * 0.98
        active = np.where(c >= min_c_active)[0]

        if len(active) <= 1:
            break

        # Target: equalize active peaks
        target = c[active].mean()
        # Compute gradient to reduce peaks above target
        g = np.zeros(n)
        for k in active:
            if c[k] > target:
                excess = (c[k] - target) / max_c
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    g[j] += excess * a[k - j]

        if np.max(np.abs(g)) < 1e-12:
            break

        # Projected gradient step on simplex
        lr = 0.1 / (np.linalg.norm(g) + 1e-15)
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        ts = a_trial.sum()
        if ts < 1e-15:
            continue
        a_trial = a_trial / ts

        score = compute_score(a_trial)
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            lr2 = lr * 0.5
            a_trial2 = a - lr2 * g
            a_trial2 = np.maximum(a_trial2, 0)
            ts2 = a_trial2.sum()
            if ts2 > 1e-15:
                a_trial2 = a_trial2 / ts2
                score2 = compute_score(a_trial2)
                if score2 < best_score:
                    best_score = score2
                    best_a = a_trial2.copy()
                    a = a_trial2
                else:
                    break

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakMatch iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, max_iters=50000):
    """Coordinate descent on elements contributing most to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        peak_k = int(np.argmax(c))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(12, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            # Try reducing this element and redistributing
            for factor in [0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                old_val = a_trial[pos]
                a_trial[pos] = old_val * factor
                diff = old_val - a_trial[pos]
                if diff < 1e-15:
                    continue
                # Redistribute to elements NOT contributing to peak
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                # Also mask out other peak contributors
                for p2 in top_positions:
                    if p2 != pos:
                        mask[p2] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()

                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try simultaneous reduction of top few
            a_trial = a.copy()
            for pos in top_positions[:4]:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            ts = a_trial.sum()
            if ts > 1e-15:
                a_trial = a_trial / ts
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 100:
                        break
        else:
            stall = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def spectral_perturb_redesign(a0, time_limit, solver='ECOS'):
    """Create perturbations in frequency domain and re-optimize."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    # Generate diverse spectral perturbations
    n_perts = 8
    per_budget = time_limit / n_perts

    for p in range(n_perts):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Random spectral perturbation
        A = fft(a)
        # Modify magnitudes slightly
        mag = np.abs(A)
        phase = np.angle(A)
        noise_scale = 0.05 * (1 + 0.5 * p)
        mag_new = mag * (1 + noise_scale * rng.standard_normal(len(mag)))
        mag_new = np.maximum(mag_new, 0)
        A_new = mag_new * np.exp(1j * phase)
        a_pert = np.real(ifft(A_new))
        a_pert = np.maximum(a_pert, 0)
        ps = a_pert.sum()
        if ps < 1e-15:
            continue
        a_pert = a_pert / ps

        # Quick LP optimization from this perturbation
        a_opt, sc = nesterov_lp_optimize(a_pert, per_budget * 0.7, solver=solver, max_iters=100000)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    Spectral pert {p}: NEW BEST {sc:.8f}")

        # Also try multiplicative update
        rem = time_limit - (time.monotonic() - t0)
        if rem > 10:
            a_md, sc2 = mirror_descent_optimize(a_pert, min(rem * 0.3, 30), max_iters=20000)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_a = a_md.copy()
                print(f"    Spectral+Mirror pert {p}: NEW BEST {sc2:.8f}")

    return best_a, best_score


def lbgfs_logsumexp_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing of the max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]

    iters_per_stage = max(20, maxiter_per_stage // len(T_stages))

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z):
            a_local = np.exp(z - z.max())
            a_local = a_local / a_local.sum()
            b = fast_convolve(a_local)
            max_b = b.max()
            exp_b = np.exp(b / stage_T - max_b / stage_T)
            lse = max_b + stage_T * np.log(exp_b.sum())
            return (2.0 * n / stage_T) * stage_T * np.log(exp_b.sum()) + 2.0 * n * max_b

        def jac(z):
            a_local = np.exp(z - z.max())
            a_local = a_local / a_local.sum()
            b = fast_convolve(a_local)
            max_b = b.max()
            exp_b = np.exp(b / stage_T - max_b / stage_T)
            w = exp_b / exp_b.sum()

            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_local[k - idx]

            S = np.dot(grad_a, a_local)
            return a_local * (2.0 * n * (grad_a - S))

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
        except Exception:
            pass

    a = np.exp(z - z.max())
    a = a / a.sum()
    return a


def structured_init(n, rng):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1 + 10 * (x - 0.5) ** 4)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score

        pass

        n_init = 1200
        best_seq = structured_init(n_init, rng)

        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best_n = len(best_seq)
    print(f"Starting optimization: n={n_best_n}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:

            best_seq = a.copy()

        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_lp_optimize(init_alt, t_alt * 0.6, solver='ECOS', max_iters=50000)
        try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = mirror_descent_optimize(best_seq, min(rem * 0.4, 30), max_iters=20000)
            try_update(a_alt2, "alt_mirror")

    # Final check
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    # Ensure all values are in valid range
    result = best_seq.copy() if best_seq is not None else np.ones(1200) / 1200
    result = np.clip(result, 0.0, 1000.0)
    return [float(x) for x in result]
