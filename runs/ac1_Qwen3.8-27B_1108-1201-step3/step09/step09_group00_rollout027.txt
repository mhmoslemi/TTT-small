import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k - j] if 0 <= k-j < n else 0
    # So T @ b gives conv(a, b)
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        T[j:j+n, j] = a
    return T


def get_lp_solver():
    # Try solvers in order of speed for medium-sized LPs
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for try_solver in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
        if try_solver is None:
            continue
        try:
            prob.solve(solver=try_solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_logsumexp_obj_and_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    # LogSumExp of b / T_temp
    max_b = b.max()
    scaled = b / T_temp
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum_exp = max_scaled + np.log(exp_sum_scaled.sum())  # wait, this should be sum of exp
    # Actually: LogSumExp(b/T) = max(b)/T + log(sum(exp(b/T - max(b)/T)))
    # = (max(b) + T*log(sum(exp(b - max(b))))) / T
    
    # Let me redo: LTS(b, T) = T * log(sum(exp(b_k / T)))
    # = T * (max(b_k)/T + log(sum(exp((b_k - max(b_k))/T))))
    # = max(b_k) + T * log(sum(exp((b_k - max(b_k))/T)))
    
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = (2.0 * n) * lse  # since sum(a)=1, (sum a)^2 = 1
    
    # Gradient of lse w.r.t. a
    # d/d a_m of lse = T * log(sum exp(b_k/T)) ... let me compute via chain rule
    # lse = T * log(sum_k exp(b_k/T))
    # d lse / d a_m = T * (1/sum_k exp(b_k/T)) * sum_k exp(b_k/T) * (1/T) * db_k/d a_m
    # = (sum_k exp(b_k/T) * db_k/d a_m) / sum_k exp(b_k/T)
    # = sum_k w_k * db_k/d a_m  where w_k = exp(b_k/T) / sum_j exp(b_j/T)
    
    w = exp_terms / exp_terms.sum()  # w_k = exp((b_k - max_b)/T) / sum exp((b_j - max_b)/T)
    
    # db_k / da_m: from our earlier analysis, for the bilinear form conv(a,a),
    # db_k / da_m = 2 * a[k-m] (with zero-padding), but since we're in the
    # simplex parameterization, we need the gradient w.r.t. z.
    
    # Actually, let's compute grad w.r.t. a first, then convert to grad w.r.t. z.
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        # db_k / da_m = 2*a[k-m] for valid m
        # So grad_a[m] += w[k] * 2 * a[k-m] for all m where 0 <= k-m < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Convert to gradient w.r.t. z: grad_z = a * (grad_a - (grad_a . a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    # Multiply by 2n for the full objective
    return (2.0 * n) * lse, (2.0 * n) * grad_z


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.03, 0.01]
    
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = compute_logsumexp_obj_and_grad(z, n, stage_T)
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, grad = compute_logsumexp_obj_and_grad(z, n, stage_T)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) on the active peak subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Use geometric step size decay
    eta0 = 0.1
    decay = 0.9999
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set (peaks within 0.1% of max)
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])
        
        # Compute subgradient: average of 2*a[k-m] over active k
        g = np.zeros(n)
        for k in active:
            # g[m] += 2 * a[k-m] for valid m
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Multiplicative update: a_new = a * exp(-eta * g)
        eta = eta0 * (decay ** step)
        log_scale = -eta * g
        log_scale -= log_scale.max()  # for numerical stability
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            no_improve = 0
            eta0 *= 0.8
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubGrad {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(999)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        # Keep track of average LP time
        avg_lp_time = np.mean(lp_times[-50:]) if len(lp_times) >= 10 else 1.0

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Structured perturbation: add a low-frequency component
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

                elapsed = time.monotonic() - t0

        if len(lp_times) > 100 and avg_lp_time > 3.0:
            print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping after {it+1} iters")
            break

    return best_a, best_score


def make_lbgfs_func(n, T_stage, deadline):
    """Create objective function for L-BFGS at given temperature."""
    def func(z):
        if time.monotonic() > deadline:
            return 1e15
        obj, _ = logsumexp_obj_grad(z, n, T_stage)
        return obj
    return func


def make_lbgfs_jac(n, T_stage, deadline):
    """Create gradient function for L-BFGS at given temperature."""
    def jac(z):
        if time.monotonic() > deadline:
            return np.zeros_like(z)
        _, g = logsumexp_obj_grad(z, n, T_stage)
        return g
    return jac


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        func = make_lbgfs_func(n, stage_T, deadline)
        jac = make_lbgfs_jac(n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within 0.05% of max
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        # Compute subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.01:
                break

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubGrad {elapsed:.0f}s: score={best_score:.8f} eta_scale={scale:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)

    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:

    elif choice == 3:

    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):

        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Try n=1500 ===
    print(f"\n=== Phase 5: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_n1500 = min(rem * 0.18, 280.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_lp_refine(init_1500, t_n1500 * 0.65, solver, n_max_iters=200000)
        local_update(a_1500, "lp_n1500")

        # If n=1500 is the best, apply L-BFGS
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 40:
            T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_1500b = lbgfs_refine(best_seq, min(rem, 120.0), T_stages2, maxiter_per_stage=5000)
            local_update(a_1500b, "lbgfs_n1500")

    # === Phase 6: Try n=2000 ===
    print(f"\n=== Phase 6: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n2000 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_n2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, _ = nesterov_lp_refine(init_2000, t_n2000 * 0.7, solver, n_max_iters=100000)
        local_update(a_2000, "lp_n2000")

    # === Phase 7: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 7: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_pert = min(rem * 0.20, 180.0)
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.02 + 0.01 * trial)
            t_sub = min(rem_sub * 0.3, 60.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
            local_update(a_opt, f"pert_trial{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.35, 50.0)
        if t_fin > 8:
            T_stages3 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages3, maxiter_per_stage=5000)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_final_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.5, 20.0), n_steps=50000)
        local_update(a_final_ms, "final_mult")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_final_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.6, 15.0), solver, n_max_iters=50000)
        local_update(a_final_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
