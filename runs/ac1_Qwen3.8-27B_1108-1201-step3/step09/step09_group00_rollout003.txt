"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Power-smoothed gradient descent with Nesterov momentum (fast bulk optimization)
2. LogSumExp L-BFGS (precise refinement)
3. LP-based Nesterov dual averaging (highest precision)
4. Multi-start with structured initializations and alternative lengths

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)

            s = b_val.sum()

                return None
            return b_val / s
    except Exception:
        pass
    # Fall back to other solvers
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            b_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                return b_val / s
        except Exception:
            continue
    return None

def eval_local(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def power_grad_descent(a0, time_limit, p_start=4.0, p_end=40.0, 
                       lr_init=0.5, momentum=0.9, seed=42):
    """
    Nesterov momentum gradient descent on the power-smoothed objective,
    working in log-space (softmax parametrization).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)  # log-space
    z = z - z.max()  # center for numerical stability
    
    v = np.zeros(n)  # momentum velocity
    beta = momentum
    lr = lr_init
    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    total_steps = 0
    no_improve = 0
    
    # Schedule: p goes from p_start to p_end over the duration
    # Use a smooth schedule
    
    while time.monotonic() - t0 < time_limit:
        # Current p based on progress
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Use a smooth schedule: p = p_start * (p_end/p_start)^progress
        p = p_start * (p_end / p_start) ** progress
        
        # Nesterov lookahead
        z_nest = z + beta * v
        a_nest = np.exp(z_nest - z_nest.max())
        a_nest = a_nest / a_nest.sum()
        
        # Compute gradient of power-smoothed objective at a_nest
        b = fast_convolve(a_nest)
        b_pos = np.maximum(b, 1e-300)
        wpw = b_pos ** (p - 1)
        wsum = wpw.sum()
        if wsum < 1e-300:
            w = np.ones(len(b)) / len(b)
        else:
            w = wpw / wsum
        
        # Gradient in a-space
        grad_a = np.zeros(n)
        threshold = w.max() * 1e-8
        active = np.where(w > threshold)[0]
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            indices = np.arange(i_start, i_end + 1)
            grad_a[indices] += 2.0 * w[k] * a_nest[k - indices]
        
        # Convert to z-space gradient (softmax gradient)
        dot_ga_a = np.dot(grad_a, a_nest)
        grad_z = a_nest * (grad_a - dot_ga_a)
        
        # Gradient step
        v = beta * v + grad_z
        z = z - lr * v
        z = z - z.max()  # recenter
        
        # Convert back to a
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        # Evaluate
        score = eval_local(a_new)
        total_steps += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve >= 50:
            lr *= 0.8
            no_improve = 0
        lr = max(lr, 1e-8)
        
        # Print progress
        if total_steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD {elapsed:.0f}s: p={p:.1f} score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score

def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on LogSumExp-smoothed objective in log-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.max()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            b = fast_convolve(a_s)
            b_max = b.max()
            return b_max + stage_T * np.log(np.sum(np.exp((b - b_max) / stage_T)))
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            b = fast_convolve(a_s)
            b_max = b.max()
            exp_b = np.exp((b - b_max) / stage_T)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            threshold = w.max() * 1e-10
            active = np.where(w > threshold)[0]
            for k in active:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                if i_start > i_end:
                    continue
                indices = np.arange(i_start, i_end + 1)
                grad_a[indices] += 2.0 * w[k] / stage_T * a_s[k - indices]
            dot_ga_a = np.dot(grad_a, a_s)
            return a_s * (grad_a - dot_ga_a)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': maxiter_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        peak_indices = np.argsort(b)[::-1][:3]
        
        improved = False
        for peak_k in peak_indices:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(5, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    if others.sum() < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0
    
    return best_a, best_score
