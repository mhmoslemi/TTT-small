import numpy as np
import cvxpy as cp
import time

def solve_qp(n):
    a = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    for k in range(2*n - 1):
        # conv[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
        terms = []
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                terms.append(a[i] * a[j])
        conv_k = cp.sum(terms)
        constraints.append(conv_k <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=cp.SCS)
    
    a_val = np.array(a.value).flatten()
    return a_val, M.value

def fast_convolve(a):

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using an improved multi-phase approach:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the score 2*n*max(conv)/(sum)^2."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = a.sum()
    return a / s


def p_norm_gradient_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_value_z(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return p_norm_value_z(z_val, stage_p)

        def jac(z_val):
            return p_norm_gradient_z(z_val, stage_p)

        iters_per_stage = max(50, min(2000, int(remaining * 8)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def build_conv_matrix(a_fixed):
    """Build the n x (2n-1) matrix where row i gives contribution of a[i] to each conv lag.
    Actually we need: T @ b = convolve(a_fixed, b), shape (2n-1) x n.
    T[k, i] = a_fixed[k-i] if 0<=k-i<n, else 0.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a_fixed
    return T

                    je2 = min(n - 1, pk)
                    mask[js2:je2+1] = False
                mask[i] = False

                if not mask.any():
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False

                others = a_trial[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial[mask] += reduction * others / osum

                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Pair swap (reduce one, increase another far away)
        if not improved and step % 3 == 0:
            peak_k = peak_idx
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_pos = np.argsort(contribs)[::-1][:5]

            for ci in top_pos:
                i = idx[ci]
                if a[i] < 1e-14:
                    continue
                # Find a far-away index to move mass to
                # Indices that don't appear in any top-5 peak
                top5 = np.argsort(inner)[::-1][:5]
                forbidden = set()
                for pk in top5:
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    for x in range(js2, je2+1):
                        forbidden.add(x)
                forbidden.add(i)
                
                candidates = [x for x in range(n) if x not in forbidden and a[x] < 0.5 * a.max()]
                if not candidates:
                    continue
                # Pick the one that most reduces the objective
                best_j = -1
                best_sc = best_score
                for j in candidates[:5]:
                    for f in [0.5, 0.7, 0.9]:
                        amt = a[i] * (1 - f)
                        a_trial = a.copy()
                        a_trial[i] = a[i] - amt
                        a_trial[j] = a[j] + amt
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_sc:
                            best_sc = sc
                            best_j = j
                            best_a_trial = a_trial.copy()
                if best_j >= 0:
                    a = best_a_trial
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True
                    break

        # Strategy 3: Global small perturbation
        if not improved and step % 200 == 0:
            rng = np.random.default_rng(step + 99)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.08, 0.08))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step % 500 == 499:
            # More aggressive restart
            rng = np.random.default_rng(step + 777)
            a_trial = a * (1.0 + 0.1 * rng.standard_normal(n))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()

        if not improved:
            # Try a slightly different approach: directly minimize the peak
            # by adjusting the two largest contributors
            if step % 1000 == 0:
                b = fast_convolve(a)
                peak_k = b.argmax()
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                if len(idx) >= 2:
                    top2 = np.argsort(contribs)[::-1][:2]
                    i1, i2 = idx[top2[0]], idx[top2[1]]
                    # Reduce both
                    for f1 in [0.95, 0.90, 0.85]:
                        for f2 in [0.95, 0.90, 0.85]:
                            a_trial = a.copy()
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st > 0.01:
                                a_trial = a_trial / st
                                sc = 2.0 * n * fast_convolve(a_trial).max()
                                if sc < best_score - 1e-14:
                                    a = a_trial
                                    best_score = sc
                                    best_a = a.copy()
                                    improved = True
                                    break
                        if improved:
                            break

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalRefine step={step} ({elapsed:.0f}s): score={best_score / (a.sum()**2) if a.sum() > 0 else best_score:.6f}")

    return best_a, best_score


def parametric_init(n, variant, param=None):
    """Generate structured initialization."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        # Power law decay
        p = param if param is not None else 2.0
        a = (1 - x) ** p
    elif variant == 1:
        # Exponential decay
        alpha = param if param is not None else 3.0
        a = np.exp(-alpha * x)
    elif variant == 2:
        # Skewed Gaussian
        mu = param if param is not None else 0.3
        sigma = 0.25
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif variant == 3:
        # Bimodal (mass at both ends)
        w = param if param is not None else 0.3
        a = np.exp(-w * x) + np.exp(-w * (1 - x))
    elif variant == 4:
        # Cosine envelope
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 5:
        # Linear decay with floor
        floor = param if param is not None else 0.1
        a = (1 - x) * (1 - floor) + floor
    elif variant == 6:
        # Quadratic
        a = (1 - x) ** 2
    elif variant == 7:
        # Sinusoidal
        a = np.sin(0.5 * np.pi * (1 - x)) ** 1.5
    elif variant == 8:
        # Rational
        beta = param if param is not None else 3.0
        a = 1.0 / (1 + beta * x)
    elif variant == 9:
        # Uniform (baseline)
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.MOSEK, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"LP solver: {solver_name}")
                return solver_name
        except Exception:
            continue
    print("LP solver: SCS (fallback)")
    return cp.SCS


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining parametric exploration, Nesterov LP,
    p-norm L-BFGS, and local refinement across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}

    def try_update(a, label):
        if a is None:
            return False
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    solver = get_lp_solver()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    n_best = len(best_state['seq']) if parent is not None else 960

    # === Phase 1: Parametric initialization exploration ===
    print(f"\n=== Phase 1: Parametric Init Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)

    # Try different lengths and parametric families
    lengths_to_try = [512, 640, 768, 896, 960, 1024, 1120, 1280, 1536]
    # Prioritize lengths close to current best
    lengths_to_try.sort(key=lambda x: abs(x - n_best))

    candidates = []
    t1_start = time.monotonic()
    for n_try in lengths_to_try[:7]:
        if time.monotonic() - t1_start > t_alloc * 0.6:
            break
        for variant in [0, 1, 2, 3, 5, 8]:
            if time.monotonic() - t1_start > t_alloc:
                break
            for param in [None, 1.5, 2.5, 4.0]:
                if time.monotonic() - t1_start > t_alloc:
                    break
                a_init = parametric_init(n_try, variant, param)
                # Add small noise for diversity
                if variant not in (0, 9):
                    a_init = a_init * (1.0 + 0.05 * rng.standard_normal(n_try))
                    a_init = np.maximum(a_init, 0)
                    a_init = normalize(a_init)
                sc = score_of(a_init)
                candidates.append((sc, a_init, f"p{variant}_{param}_n{n_try}"))

    # Sort by score and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best parametric score: {candidates[0][0]:.6f}")

    # Top candidates for refinement
    top_candidates = []
    seen_lengths = set()
    for sc, a_init, label in candidates[:20]:
        n_c = len(a_init)
        if n_c not in seen_lengths:
            top_candidates.append((sc, a_init, n_c, label))
            seen_lengths.add(n_c)
        if len(top_candidates) >= 5:
            break

    # Also include the parent
    if parent is not None:
        top_candidates.append((best_state['score'], parent, len(parent), "parent"))
    top_candidates.sort(key=lambda x: x[0])

    # === Phase 2: Nesterov LP on top candidates ===
    print(f"\n=== Phase 2: Nesterov LP on top candidates ===")
    rem = deadline - time.monotonic()
    for ci, (sc_init, a_init, n_c, label) in enumerate(top_candidates[:4]):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.25, 300.0)
        if t_sub < 30:
            t_sub = min(rem * 0.4, 200.0)
        if t_sub < 20:
            break
        print(f"  Nesterov LP on {label}, n={n_c}, budget={t_sub:.0f}s")
        try:
            a_res, sc_res = nesterov_lp_fast(a_init, t_sub, solver)
            try_update(a_res, f"nest_{label}")
        except Exception as e:
            print(f"  Error in Nesterov LP: {e}")

    # === Phase 3: p-norm L-BFGS refinement ===
    print(f"\n=== Phase 3: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pn = min(rem * 0.25, 350.0)
        print(f"  Budget: {t_pn:.0f}s")
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_pn, p_start=15.0, p_end=500.0, n_stages=10)
        try_update(a_pn, "pnorm_1")

    # === Phase 4: Nesterov LP round 2 (accelerated) ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_lp2 = min(rem * 0.25, 300.0)
        print(f"  Budget: {t_lp2:.0f}s")
        try:
            a_lp2, _ = nesterov_lp_fast(best_state['seq'], t_lp2, solver)
            try_update(a_lp2, "nest_2")
        except Exception as e:
            print(f"  Error: {e}")

    # === Phase 5: p-norm L-BFGS round 2 (higher p) ===
    print(f"\n=== Phase 5: p-norm L-BFGS r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_pn2 = min(rem * 0.20, 250.0)
        print(f"  Budget: {t_pn2:.0f}s, p: 100->2000")
        a_pn2 = p_norm_lbfgs_refine(best_state['seq'], t_pn2, p_start=100.0, p_end=2000.0, n_stages=6)
        try_update(a_pn2, "pnorm_2")

    # === Phase 6: Local refinement ===
    print(f"\n=== Phase 6: Local Refinement (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lr = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_lr:.0f}s")
        a_lr, _ = local_refine_fast(best_state['seq'], t_lr, n_steps=80000)
        try_update(a_lr, "local_1")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_state['seq'])
    explored_lengths = {n_cur}

    # Try lengths near current best
    candidate_lengths = []
    for delta in [-128, -96, -64, -32, 32, 64, 96, 128, 192, 256]:
        alt_n = n_cur + delta
        if 256 <= alt_n <= 3000 and alt_n not in explored_lengths:
            candidate_lengths.append(alt_n)
    # Add some round numbers
    for rn in [512, 768, 1024, 1280, 1536, 2048]:
        if 256 <= rn <= 3000 and rn not in explored_lengths and abs(rn - n_cur) > 50:
            candidate_lengths.append(rn)

    candidate_lengths.sort(key=lambda x: abs(x - n_cur))
    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 150.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # Quick Nesterov
        t_nest = t_total * 0.5
        try:
            a_alt, _ = nesterov_lp_fast(init_alt, t_nest, solver)
            improved = try_update(a_alt, f"nest_n{alt_n}")
        except Exception:
            a_alt = init_alt
            improved = False

        # If improved, do p-norm polish
        if improved and rem > 20:
            t_polish = min(rem * 0.1, 60.0)
            a_pol = p_norm_lbfgs_refine(best_state['seq'], t_polish, p_start=50.0, p_end=300.0, n_stages=5)
            try_update(a_pol, f"pn_n{alt_n}")
        explored_lengths.add(alt_n)

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur2 = len(best_state['seq'])
        for noise_scale in [0.03, 0.06, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur2)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.2, 80.0)
            a_pert = p_norm_lbfgs_refine(perturbed, t_sub, p_start=30.0, p_end=400.0, n_stages=5)
            try_update(a_pert, f"pert_{noise_scale}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = p_norm_lbfgs_refine(best_state['seq'], t_fin, p_start=200.0, p_end=3000.0, n_stages=5)
        try_update(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr2 = min(rem * 0.30, 60.0)
        a_lr2, _ = local_refine_fast(best_state['seq'], t_lr2, n_steps=30000)
        try_update(a_lr2, "final_local")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.40, 40.0)
        try:
            a_lp_fin, _ = nesterov_lp_fast(best_state['seq'], t_lp_fin, solver)
            try_update(a_lp_fin, "final_lp")
        except Exception:
            pass

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(max(0.0, min(1000.0, x))) for x in best_state['seq']]
