def fast_convolve_auto(a):
    """Linear auto-convolution conv(a, a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution conv(a, b) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

    n = len(a)

    return T

def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp-smoothed objective and gradient in log-domain."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve_auto(a)
    Tb = T * b  # element-wise: T[k,j] * b[k] for each j
    # Actually, we need the gradient with respect to a
    # f(a) = (2n/T) * log(sum_k exp(T*b[k]))  where b = conv(a,a)
    # This is getting complicated. Let me use a simpler approach.
    
    # Simpler: just compute the smooth max of b
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    log_sum_exp = max_b / T + T * np.log(exp_b.sum())
    obj = 2.0 * n * log_sum_exp
    
    # Gradient of log_sum_exp with respect to a
    w = exp_b / exp_b.sum()  # weights for the smooth max
    # grad log_sum_exp / grad a[j] = (1/T) * sum_k w[k] * d(b[k])/d(a[j])
    # d(b[k])/d(a[j]) = 2 * a[k-j] (if 0<=k-j<n)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    grad_obj = 2.0 * n * grad_a / T
    
    # Convert to log-domain gradient
    S = np.dot(grad_obj, a)
    grad_z = a * (grad_obj - S)
    
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing for precise optimization."""
    n = len(a0)

    z = z - z.max()  # normalize
    t0 = time.monotonic()

    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break

        if remaining < 3:
            break

            if time.monotonic() - t0 > time_limit:

    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

def fast_convolve_auto(a):
    """Linear auto-convolution conv(a, a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution conv(a, b) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:

            prob.solve(solver=solver_name)

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for solv in [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]:
        try:
            prob.solve(solver=solv)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:

                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    a = np.maximum(a, 0)
    return a


def peak_descent(a0, time_limit, n_steps=50000):
    """Peak-directed coordinate descent on top contributors to the max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve_auto(a)
        max_b = b.max()
        # Find all near-peak positions
        threshold = max_b * 0.999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for peak_k in peaks[:3]:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.985, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    if others.sum() < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve_auto(a_trial).max()
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            a_trial = a.copy()
            for pk in peaks[:2]:
                pk = int(pk)
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top2 = idx[np.argsort(contribs)[::-1][:2]]
                for pos in top2:
                    if a[pos] > 1e-12:
                        a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = 2.0 * n * fast_convolve_auto(a_trial).max()
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 400:
                    break
        else:
            stall = 0

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Exact subgradient method targeting the argmax of the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    lr = 0.015
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve_auto(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve_auto(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimization:
    Phase 1: Nesterov with fast PG best response + periodic LP (50% time)
    Phase 2: Low-temp LogSumExp L-BFGS (20% time)
    Phase 3: Peak-directed coordinate descent (15% time)
    Phase 4: Subgradient refinement (10% time)
    Phase 5: Final polish + length exploration (5% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.exp(-4 * x * (1 - x)) + 0.2 * np.exp(-10 * (x - 0.5) ** 2)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov Hybrid (fast PG + periodic LP) ===
    print(f"\n=== Phase 1: Nesterov Hybrid n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.50, 850.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, LP every 40 iters, PG iters=300")
        a_ph1, _ = nesterov_hybrid(best_seq, t_alloc, solver=solver, lp_every=40, n_pg=300)
        try_update(a_ph1, "nesterov_hybrid_1")

    # === Phase 2: LogSumExp L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        T_stages = [30.0, 15.0, 7.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s")
        a_ph2 = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=8000)
        try_update(a_ph2, "lbgfs_1")

    # === Phase 3: Peak-directed coordinate descent ===
    print(f"\n=== Phase 3: Peak Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 15:
        a_ph3, _ = peak_descent(best_seq, t_alloc, n_steps=60000)
        try_update(a_ph3, "peak_desc_1")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_ph4, _ = subgradient_refine(best_seq, t_alloc, n_steps=60000)
        try_update(a_ph4, "subgrad_1")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    # 5a: Another L-BFGS pass at very low temp
    if rem > 30:
        t_fin = min(rem * 0.30, 80.0)
        if t_fin > 10:
            T_fin = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages=T_fin, maxiter_per_stage=5000)
            try_update(a_fin, "lbgfs_final")

    # 5b: Another peak descent
    rem = deadline - time.monotonic()
    if rem > 20:
        t_pd = min(rem * 0.30, 50.0)
        if t_pd > 10:
            a_pd2, _ = peak_descent(best_seq, t_pd, n_steps=20000)
            try_update(a_pd2, "peak_desc_2")

    # 5c: Second Nesterov pass (with different parameters)
    rem = deadline - time.monotonic()
    if rem > 40:
        t_ns = min(rem * 0.40, 100.0)
        if t_ns > 20:
            # Perturb slightly to potentially escape
            x = np.linspace(0, 2 * np.pi, len(best_seq))
            perturbed = best_seq + 0.01 * np.cos(2 * x)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_ns2, _ = nesterov_hybrid(perturbed, t_ns, solver=solver, lp_every=30, n_pg=200)
            try_update(a_ns2, "nesterov_2")

    # 5d: Try slightly different lengths
    rem = deadline - time.monotonic()
    if rem > 60:
        for alt_n in [int(n_best * 0.9), int(n_best * 1.1)]:
            if rem < 30:
                break
            t_alt = min(rem * 0.4, 60.0)
            if t_alt < 15:
                break
            print(f"  Trying alt length n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_hybrid(init_alt, t_alt * 0.6, solver=solver, lp_every=30, n_pg=200)
            try_update(a_alt, f"alt_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                a_alt2 = lbgfs_refine(best_seq, min(rem * 0.4, 30.0),
                                      T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                      maxiter_per_stage=3000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
