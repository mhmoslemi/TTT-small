```python
        return np.ones(len(z)) / len(z)
    return exp_z / s

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_active_qp(a_current, n_active=20, tol_frac=0.05, solver=None):
    """
    Solve the active-set QP.
    Given current a (sum=1, non-neg), find the top n_active convolution values,
    then solve: min a^T H a s.t. sum(a)=1, a>=0, where H is the Hankel matrix
    of the active weights.
    
    Returns the new a.
    """
    if solver is None:
        solver = get_lp_solver()
    
    a = norm_sequence(a_current)
    n = len(a)
    b = fast_convolve(a)
    
    # Find active set
    b_max = b.max()
    threshold = b_max * (1 - tol_frac)
    active_mask = b >= threshold
    
    # Ensure we have at least n_active in the active set
    if active_mask.sum() < n_active:
        # Add the next largest values
        sorted_idx = np.argsort(b)[::-1]
        for idx in sorted_idx[:n_active]:
            active_mask[idx] = True
    
    # Create weights: uniform on active set
    w = np.zeros(2 * n - 1)
    w[active_mask] = 1.0
    
    # Build Hankel Hessian: H[i,j] = 2 * w[i+j]
    # For the QP: minimize a^T H a = 2 * sum_k w_k * b_k
    # We can use cvxpy with the quadratic form
    
    # Actually, let's compute the quadratic form directly using the structure
    # a^T H a = 2 * sum_k w_k * (sum_i a_i * a_{k-i})
    # = 2 * sum_k w_k * b_k(a)
    
    # For cvxpy, we need the matrix. Let's build it efficiently.
    # H[i,j] = 2 * w[i+j]
    
    # Build H as a Hankel matrix
    # H[i,j] = 2*w[i+j] for i,j in 0..n-1, so i+j ranges from 0 to 2n-2
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = 2.0 * w[i:i+n]
    
    # Solve QP
    a_var = cp.Variable(n, nonneg=True)
    objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(H)))
    constraints = [cp.sum(a_var) == 1]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternative solvers
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    a_new = np.asarray(a_var.value, dtype=np.float64)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    a_new = a_new / s
    return a_new


def active_qp_refine(a0, time_limit, n_active=20, tol_frac=0.05, solver=None, max_iters=500):
    """
    Iteratively solve the active-set QP, updating the active set each iteration.
    """
    if solver is None:
        solver = get_lp_solver()
    
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * len(a) * fast_convolve(a).max()
    t0 = time.monotonic()
    
    prev_score = best_score
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_start = time.monotonic()
        a_new = solve_active_qp(a, n_active=n_active, tol_frac=tol_frac, solver=solver)
        t_qp = time.monotonic() - t_start
        
        if a_new is None:
            stall_count += 1
            if stall_count > 5:
                # Try with larger active set
                a_new = solve_active_qp(a, n_active=n_active * 2, tol_frac=tol_frac * 2, solver=solver)
                if a_new is None:
                    break
        else:
            score = 2.0 * len(a) * fast_convolve(a_new).max()
            if score < best_score - 1e-15:
                best_score = score
                best_a = a_new.copy()
                a = a_new.copy()
                stall_count = 0
            else:
                # Accept anyway if it's not much worse (to escape plateaus)
                if score < best_score + 0.001:
                    a = 0.5 * a + 0.5 * a_new
                    a = norm_sequence(a)
                stall_count += 1
                if stall_count > 10:
                    # Try expanding active set
                    tol_frac = min(0.5, tol_frac * 1.5)
                    n_active = min(100, n_active + 10)
                    stall_count = 0
                    continue
                if stall_count > 20:
                    break
        
        # Reset expanded parameters if we made progress
        if stall_count == 0:
            n_active = max(10, n_active // 2)
            tol_frac = max(0.01, tol_frac * 0.5)
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000, n_stages=None):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(9999 + it)
                noise = rng_local.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score

def structured_init(n, rng, mode=0):
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP alternating optimization for global structure
2. L-BFGS with log-sum-exp smoothing and multi-stage annealing for local refinement
3. Active-set QP (with psd_wrap) for precise optimization on binding constraints
4. Multi-length exploration with warm-start interpolation
5. Subgradient polishing with the exact max function
Key new idea: Active-set QP formulation and improved time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_deadline = 0.0


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_active_qp(a_current, n_active, tol_frac, solver=None):
    """
    Solve the active-set QP: minimize weighted sum of top convolution coefficients.
    Uses psd_wrap for the Hankel Hessian.
    """
    if solver is None:
        solver = get_lp_solver()

    a = norm_sequence(a_current)
    n = len(a)
    b = fast_convolve(a)

    # Find active set
    b_max = b.max()
    if b_max < 1e-30:
        return None
    threshold = b_max * (1.0 - tol_frac)
    active_mask = b >= threshold

    # Ensure minimum active set size
    if active_mask.sum() < n_active:
        sorted_idx = np.argsort(b)[::-1]
        for idx in sorted_idx[:n_active]:
            active_mask[idx] = True

    # Cap active set size
    active_count = active_mask.sum()
    if active_count > 2 * n_active:
        sorted_idx = np.argsort(b)[::-1]
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        for idx in sorted_idx[:2 * n_active]:
            active_mask[idx] = True

    # Create weights
    w = np.zeros(2 * n - 1)
    w[active_mask] = 1.0

    # Build Hankel Hessian: H[i,j] = 2 * w[i+j]
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = 2.0 * w[i:i + n]

    # Solve QP using cvxpy
    a_var = cp.Variable(n, nonneg=True)
    objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(H)))
    constraints = [cp.sum(a_var) == 1]
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    a_new = np.asarray(a_var.value, dtype=np.float64)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    a_new = a_new / s
    return a_new


def active_qp_refine(a0, time_limit, n_active, tol_frac, solver=None, max_iters=200):
    """
    Iteratively solve the active-set QP, updating the active set each iteration.
    """
    if solver is None:
        solver = get_lp_solver()

    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * len(a) * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0

    cur_n_active = n_active
    cur_tol = tol_frac

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_new = solve_active_qp(a, cur_n_active, cur_tol, solver)

        if a_new is None:
            stall_count += 1
            if stall_count > 5:
                break
            continue

        score = 2.0 * len(a) * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            stall_count = 0
            # Tighten parameters on success
            cur_tol = max(0.005, cur_tol * 0.7)
            cur_n_active = max(5, int(cur_n_active * 0.8))
        else:
            # Try accepting with slight mixing
            mixed = 0.5 * a + 0.5 * a_new
            mixed = norm_sequence(mixed)
            mixed_score = 2.0 * len(a) * fast_convolve(mixed).max()
            if mixed_score < best_score:
                best_score = mixed_score
                best_a = mixed.copy()
                a = mixed.copy()
            stall_count += 1
            if stall_count > 8:
                # Expand active set to escape
                cur_tol = min(0.5, cur_tol * 2.0)
                cur_n_active = min(100, cur_n_active + 10)
                stall_count = 0
            if stall_count > 25:
                break

    return best_a, best_score


def lbgfs_func(z_in, T_val):
    """L-BFGS objective: smoothed max via log-sum-exp."""
    if time.monotonic() > _global_deadline:
        return 1e15
    n = len(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lbgfs_jac(z_in, T_val):
    """L-BFGS Jacobian for the smoothed objective."""
    if time.monotonic() > _global_deadline:
        return np.zeros_like(z_in)
    n = len(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000, n_stages=None):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=lbgfs_func,
                x0=z,
                jac=lbgfs_jac,
                args=(stage_T,),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(7777 + it)
                noise = rng_local.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=5000):
    """Subgradient method on the exact max function with decreasing steps."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.1
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient: g_i = 2 * a[k_star - i]
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]

        # Projected subgradient step
        eta = 0.1 / (1.0 + 0.01 * step)
        a_trial = a - eta * g
        a_trial = np.maximum(a_trial, 0)
        sa = a_trial.sum()
        if sa < 1e-15:
            break
        a_trial = a_trial / sa

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            eta *= 0.9

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with active-set QP for precise final optimization.
    Combines Nesterov LP, L-BFGS log-sum-exp, active-set QP, and subgradient polish.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    _global_deadline = deadline

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # Use a list to hold best values (works around closure restriction)
    best_state = {'seq': best_seq.copy(), 'score': best_score}

    def try_update(a, label):
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov LP (global structure) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 2: L-BFGS main (multi-temperature) ===
    print(f"\n=== Phase 2: L-BFGS main (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 80->0.3, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_state['seq'], t_alloc, T_start=80.0, T_end=0.3, maxiter=15000, n_stages=14)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: Active-set QP ===
    print(f"\n=== Phase 3: Active-set QP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_qp, _ = active_qp_refine(best_state['seq'], t_alloc, n_active=20, tol_frac=0.04, solver=solver, max_iters=300)
        try_update(a_qp, "active_qp")

    # === Phase 4: L-BFGS low temperature ===
    print(f"\n=== Phase 4: L-BFGS low-T (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 220.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 5->0.03, budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_refine(best_state['seq'], t_alloc, T_start=5.0, T_end=0.03, maxiter=10000, n_stages=12)
        try_update(a_lb2, "lbgfs_lowT")

    # === Phase 5: Active-set QP round 2 (tighter) ===
    print(f"\n=== Phase 5: Active-set QP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 160.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_qp2, _ = active_qp_refine(best_state['seq'], t_alloc, n_active=12, tol_frac=0.02, solver=solver, max_iters=400)
        try_update(a_qp2, "active_qp2")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_polish(best_state['seq'], t_alloc, n_steps=10000)
        try_update(a_sg, "subgrad")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_state['seq'])

    candidate_lengths = []
    for mult in [1.05, 1.1, 0.95, 0.9, 1.15, 1.2, 1.3]:
        alt_n = int(n_cur * mult)
        if 320 <= alt_n <= 4096 and alt_n != n_cur:
            candidate_lengths.append(alt_n)
    for fixed_n in [832, 896, 1024, 1152, 1280, 1408]:
        if 320 <= fixed_n <= 4096 and fixed_n != n_cur:
            candidate_lengths.append(fixed_n)
    candidate_lengths = sorted(set(candidate_lengths))

    for alt_n in candidate_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.25, 250.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # LP on new length
        t_lp = t_total * 0.35
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=15000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS
        rem = deadline - time.monotonic()
        if rem > 30 and best_state['seq'] is not None and len(best_state['seq']) == alt_n:
            t_lb = min(rem * 0.3, 80.0)
            if t_lb > 15:
                a_alt2 = lbgfs_refine(best_state['seq'], t_lb, T_start=30.0, T_end=0.1, maxiter=5000, n_stages=8)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

        # QP
        rem = deadline - time.monotonic()
        if rem > 20 and best_state['seq'] is not None and len(best_state['seq']) == alt_n:
            t_qp_alt = min(rem * 0.25, 60.0)
            if t_qp_alt > 10:
                a_alt3, _ = active_qp_refine(best_state['seq'], t_qp_alt, n_active=15, tol_frac=0.03, solver=solver, max_iters=80)
                try_update(a_alt3, f"qp_n{alt_n}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_qp_fin = min(rem * 0.25, 100.0)
        if t_qp_fin > 10:
            a_fin_qp, _ = active_qp_refine(best_state['seq'], t_qp_fin, n_active=10, tol_frac=0.015, solver=solver, max_iters=150)
            try_update(a_fin_qp, "final_qp")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lb_fin = min(rem * 0.3, 100.0)
        if t_lb_fin > 10:
            a_fin_lb = lbgfs_refine(best_state['seq'], t_lb_fin, T_start=3.0, T_end=0.01, maxiter=8000, n_stages=10)
            try_update(a_fin_lb, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_sg_fin = min(rem * 0.2, 50.0)
        if t_sg_fin > 5:
            a_fin_sg, _ = subgradient_polish(best_state['seq'], t_sg_fin, n_steps=5000)
            try_update(a_fin_sg, "final_sg")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_qp_final = min(rem * 0.3, 60.0)
        if t_qp_final > 5:
            a_final_qp, _ = active_qp_refine(best_state['seq'], t_qp_final, n_active=8, tol_frac=0.01, solver=solver, max_iters=50)
            try_update(a_final_qp, "final_qp2")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
```
