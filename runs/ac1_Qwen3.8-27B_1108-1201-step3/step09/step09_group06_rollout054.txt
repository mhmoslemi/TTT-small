import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_and_grad(a, p):
    """
    Return (||b||_p, gradient_wrt_a) where b = conv(a,a).
    Gradient uses FFT-based weighted cross-correlation.
    """
    n = len(a)
    b = fast_convolve(a)
    # Clip to avoid issues with tiny negative values from FFT
    b = np.maximum(b, 0.0)
    sum_b_p = np.sum(b ** p)
    norm_p = sum_b_p ** (1.0 / p)
    if norm_p < 1e-300:
        return 1e-300, np.zeros(n)
    # b_k^(p-1) as weights
    b_w = b ** (p - 1)
    # cross-correlation: g[i] = sum_m a[m] * b_w[m+i] for i=0..n-1
    # = conv(rev(a), b_w)[n-1 : 2n-1]
    fft_size = 1 << (n + (2 * n - 1) - 1).bit_length()
    a_rev = a[::-1].copy()
    A = rfft(a_rev, fft_size)
    B = rfft(b_w, fft_size)
    conv_res = irfft(A * B, fft_size)
    g = conv_res[n - 1:2 * n - 1]
    grad = 2.0 * (norm_p ** (1.0 - p)) * g
    # Ensure non-negative gradient direction makes sense
    return norm_p, grad


def softmax_from_z(z):

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

            grad_z = a * (grad_a - S)
            gnorm = np.linalg.norm(grad_z)
            if gnorm > 1e-15:
                step = min(0.1, 0.5 / gnorm)
                z_new = z_cur - step * grad_z / gnorm
                a_pn = softmax_from_z(z_new)
                score_pn = 2.0 * n * fast_convolve(a_pn).max()
                if score_pn < 2.0 * n * fast_convolve(a).max():
                    a = a_pn
                    z = z_new
                    y = a.copy()

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 200 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 10:
                rng = np.random.default_rng(it * 13 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def targeted_peak_reduction(a0, time_limit, n_steps=200000):
    """
    Aggressively target the peak of conv(a,a) by redistributing mass
    from high-contributing positions to low-contributing ones.
    Uses weighted cross-correlation to identify optimal moves.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    plateau = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find top few peaks
        top_frac = 0.03
        threshold = peak_val * (1 - top_frac)
        near_peaks = np.where(b >= threshold)[0]

        # Compute contribution to peaks for each position
        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Find best move: reduce high-contrib, add to low-contrib
        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        for si in top_src:
            if a[si] < 1e-14:
                continue
            for di in low_dst:
                if si == di:
                    continue
                for frac in [0.01, 0.02, 0.04, 0.07, 0.12]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        plateau = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape
            if step % 100 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-14:
                    amount = a[i] * 0.02
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            plateau = 0

            plateau += 1
            if plateau > 5000:
                break

    return best_a, best_score


def mirror_descent_adaptive(a0, time_limit, n_active_frac=0.05):
    """Adaptive mirror descent with Polyak step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-10:
            break

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.6
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(4 * x) + 0.1 * np.cos(6 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Quick multi-start (60s) ===
    print(f"\n=== Phase 1: Multi-start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.04, 60.0)
    if t_alloc > 10:
        n_start = len(best_seq)
        candidates = [best_seq.copy()]
        # Add a few diverse inits at same length
        for i in range(3):
            candidates.append(structured_init(n_start, rng))
        # Quick LP warmup on each
        t_per = t_alloc / len(candidates)
        for i, cand in enumerate(candidates):
            if time.monotonic() > deadline - 10:
                break
            a_c, _ = nesterov_lp_refine(cand, t_per, solver=solver, p_inject_every=20)
            try_update(a_c, f"multi_start_{i}")

    # === Phase 2: P-norm L-BFGS annealing (400s) ===
    print(f"\n=== Phase 2: P-norm L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, p: 2->64")
        a_pn, _ = lbgfs_p_norm(best_seq, t_alloc, p_start=2.0, p_end=64.0,
                               n_stages=12, maxiter_per_stage=400)
        try_update(a_pn, "p_norm_1")

    # === Phase 3: Nesterov LP with p-norm injections (500s) ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                     p_inject_every=8, p_val=16.0)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 4: Second P-norm pass with higher p (200s) ===
    print(f"\n=== Phase 4: P-norm High-p (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_pn2, _ = lbgfs_p_norm(best_seq, t_alloc, p_start=8.0, p_end=128.0,
                                n_stages=8, maxiter_per_stage=300)
        try_update(a_pn2, "p_norm_2")

    # === Phase 5: Targeted peak reduction (200s) ===
    print(f"\n=== Phase 5: Peak Reduction (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        a_tr, _ = targeted_peak_reduction(best_seq, t_alloc, n_steps=100000)
        try_update(a_tr, "peak_reduce_1")

    # === Phase 6: Mirror descent for final squeeze (150s) ===
    print(f"\n=== Phase 6: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_md, _ = mirror_descent_adaptive(best_seq, t_alloc, n_active_frac=0.03)
        try_update(a_md, "mirror_desc_final")

    # === Phase 7: Length exploration (250s) ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_total_len = min(rem * 0.16, 250.0)
    if t_total_len > 40:
        alt_lengths = []
        for mult in [1.1, 1.25, 0.9, 1.5]:
            alt_n = int(n_best * mult)
            if 400 <= alt_n <= 2500 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))[:2]
        t_per_len = t_total_len / max(1, len(alt_lengths))

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_use = min(t_per_len, rem - 5)
            if t_use < 20:
                break
            print(f"  n={alt_n}, budget={t_use:.0f}s")

            init_alt = interpolate_sequence(best_seq, alt_n)
            sc_alt = evaluate_safe(init_alt)
            if sc_alt < best_score:
                best_score = sc_alt
                best_seq = init_alt.copy()

            # Quick LP + p-norm on this length
            t_lp = t_use * 0.55
            a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, p_inject_every=15)
            try_update(a_alt, f"lp_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
                t_pn = min(rem * 0.3, 50.0)
                if t_pn > 10:
                    a_alt2, _ = lbgfs_p_norm(best_seq, t_pn, p_start=4.0, p_end=32.0,
                                             n_stages=6, maxiter_per_stage=200)
                    try_update(a_alt2, f"pn_n{alt_n}")

    # === Phase 8: Final combined polish (remaining) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Short LP
        t_lp = min(rem * 0.25, 60.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, p_inject_every=10, p_val=32.0)
            try_update(a_fin, "final_lp")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_tr = min(rem * 0.3, 40.0)
            if t_tr > 10:
                a_tr2, _ = targeted_peak_reduction(best_seq, t_tr, n_steps=50000)
                try_update(a_tr2, "final_tr")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_md = min(rem * 0.3, 30.0)
            if t_md > 5:
                a_md2, _ = mirror_descent_adaptive(best_seq, t_md, n_active_frac=0.02)
                try_update(a_md2, "final_md")

        rem = deadline - time.monotonic()
        if rem > 10:
            t_pn = min(rem * 0.4, 25.0)
            if t_pn > 5:
                a_pn_f, _ = lbgfs_p_norm(best_seq, t_pn, p_start=16.0, p_end=128.0,
                                         n_stages=4, maxiter_per_stage=100)
                try_update(a_pn_f, "final_pn")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
