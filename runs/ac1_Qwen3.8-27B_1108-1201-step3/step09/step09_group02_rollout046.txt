import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_conv(a, conv):
    """Compute score from a and its convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    spectrum[0] = amps[0]  # DC component is real
    if n_fft % 2 == 0:
        spectrum[-1] = amps[-1]  # Nyquist is real
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    return a / s

    t = np.arange(n, dtype=np.float64)

    a = np.exp(-0.5 * ((t - center) / width) ** 2)

    return a / s

def init_spectral_factor(n, rng):
    """
    Spectral factorization-based initialization.
    Choose a smooth spectrum, factor it, and take the real part.
    """
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    
    # Create a smooth spectrum that's close to flat but with some structure
    # This gives a sequence with low peak autocorrelation
    freqs = np.arange(n_freq)
    # Add some low-frequency content to make the sequence more "spread out"
    base = np.ones(n_freq)
    n_modes = rng.integers(2, 8)
    for _ in range(n_modes):
        f0 = rng.integers(1, n_freq // 3)
        width_f = rng.uniform(1, 5)
        base += 0.1 * np.exp(-0.5 * ((freqs - f0) / width_f) ** 2)
    
    # Ensure positive
    base = np.maximum(base, 0.1)
    
    # Random phases
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    A = np.sqrt(base) * np.exp(1j * phases)
    A[0] = np.sqrt(base[0])  # DC real
    if n_fft % 2 == 0:

    a = irfft(A, n_fft)
    a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    return a / s

def init_waterfill(n, rng):
    """
    Water-filling initialization: start flat and iteratively reduce peaks.
    """
    a = np.ones(n) / n
    for _ in range(20):
        conv = fast_convolve(a)
        # Find the peak
        k_peak = np.argmax(conv)
        # Reduce contribution to this peak
        # The peak at lag k involves pairs (i, k-i) for i in [max(0,k-n+1), min(n-1,k)]
        i_start = max(0, k_peak - n + 1)

        # Reduce the affected elements slightly

            if i != j:
                a[i] *= 0.98
                a[j] *= 0.98
            else:
                a[i] *= 0.97
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            quad_idx = (2 * i - k_lo) if has_quad else -1

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 4.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    factors = [0.999, 0.998, 0.995, 0.99, 0.985, 1.005, 1.01, 1.015, 1.02, 1.03]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            quad_idx = (2 * i - k_lo) if has_quad else -1

            cur_conv_local = float(conv[k_lo:k_hi+1].max())
            best_delta = 0.0
            best_local = cur_conv_local

            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = max(unaff_max, float(nv.max()))
                ns = current_sum + (x - v)
                sc = 2.0 * n * local_max / (ns * ns)
                if sc < best_score - 1e-15 and local_max < best_local - 1e-16:
                    best_local = local_max
                    best_delta = x - v

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = _get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

_lp_solver_cached = None

def _get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = 'ECOS'
    return 'ECOS'

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with time-averaging and line search."""
    if solver is None:
        solver = _get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    decay = 3.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        # Solve LP best response
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Line search
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = decay / (iters + decay)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        # Check candidates
        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def peak_targeted_perturb(a, conv, rng, strength=0.05):
    """Perturb the sequence to specifically reduce the top peaks."""
    n = len(a)
    perturbed = a.copy()
    num_peaks = min(5, 2 * n - 1)
    peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
    
    for k in peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        # Reduce elements that contribute most to this peak
        n_reduce = max(1, len(idx) // 10)
        picks = rng.choice(idx, size=min(n_reduce, len(idx)), replace=False)
        for ii in picks:
            perturbed[ii] *= (1.0 - strength * rng.random())
    
    # Also add some noise
    perturbed += rng.standard_normal(n) * 0.01
    perturbed = np.maximum(perturbed, 0)
    ps = perturbed.sum()
    if ps < 1e-10:
        return a
    return perturbed / ps

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=12, verbose=False):
    """Basin hopping with peak-targeted perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    conv = fast_convolve(a)

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.5, 45.0)

        # Peak-targeted perturbation
        perturbed = peak_targeted_perturb(a, conv, rng, strength=0.03 + 0.04 * rng.random())
        
        # Also try random perturbation
        if rng.random() < 0.3:
            perturbed = a + rng.standard_normal(n) * (0.02 + 0.03 * rng.random())
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 1e-10:
                perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            conv = fast_convolve(a_opt)
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score

def subgradient_step(a, conv, lr, n_peaks=20):
    """One subgradient step using active peak weighting."""
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    temp = 0.003 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    a_new = simplex_project(a - lr * grad)
    return a_new

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
