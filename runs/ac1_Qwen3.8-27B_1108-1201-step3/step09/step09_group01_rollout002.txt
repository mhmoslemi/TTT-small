def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (rho[idx])
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_topk_gradient(conv, a, k=10):
    """
    Compute the gradient of the top-k average of conv w.r.t. a.
    top-k avg = (1/k) * sum of k largest values in conv.
    Gradient w.r.t. a_j = (2/k) * sum_{i in top_k} a[conv_idx - j]
    """
    n = len(a)
    # Find top-k indices in conv
    if k >= len(conv):
        topk_idx = np.argsort(conv)[::-1]
        k_actual = len(conv)
    else:
        topk_idx = np.argpartition(conv, -k)[-k:]
        topk_idx = topk_idx[np.argsort(conv[topk_idx])[::-1]]
        k_actual = k

    # Gradient: g_j = (2/k) * sum_{m in topk} a[m-j] (with bounds checking)
    g = np.zeros(n)
    for m in topk_idx:

        valid = (j_val >= 0) & (j_val < n)
        if valid.any():
            g[valid] += 2.0 * a[j_val[valid]]
    g /= k_actual
    return g

        y = simplex_project(y)

        # Compute gradient at y
        conv_y = fast_convolve(y)
        g = get_topk_gradient(conv_y, y, k=topk)

        # Update
        a_new = simplex_project(y - step / gamma * g)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_prev = a.copy()
        a = a_new

        # Adaptive step size
        step *= step_decay
        if no_improve > 200:
            step = max(step * 0.7, step_min)
            no_improve = 0
        elif it % 100 == 0:
            # Occasionally increase step to escape shallow minima
            step = min(step * 1.3, 1.0 / (n ** 0.5))

        if step < step_min:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NPS {label} it={it} ({elapsed:.1f}s): score={best_score:.8f} step={step:.2e}")

    return best_a, best_score


def block_coord_descent(a0, time_limit, block_size=32, verbose=False, label=""):
    """
    Block coordinate descent: optimize blocks of coordinates simultaneously.
    For each block, we minimize max(conv) over the block variables (others fixed).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_M = conv.max()
    it = 0
    no_improve = 0

    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        for b in range(n_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            idx = np.arange(start, end)

            # Compute current conv
            conv_cur = fast_convolve(a)
            M_cur = conv_cur.max()
            sum_a = a.sum()

            # For this block, compute the "fixed" contribution to conv
            # conv_fixed[k] = conv contribution from pairs where neither index is in the block
            # conv_cross[k] = conv contribution from pairs where exactly one index is in the block
            a_block = a[idx].copy()
            a_rest = np.delete(a, idx)

            # Contribution of block to conv:
            # conv = conv_rest + 2 * conv_cross + conv_block
            # where conv_block = convolve(a_block, a_block) (padded)
            # and conv_cross = convolve(a_block, a_rest_padded) + convolve(a_rest_padded, a_block)

            # Simpler: just compute the full conv and track how it changes
            # When we change a_block to a_block_new (with same sum to maintain sum(a)=1):
            # delta_conv[k] = 2 * (a_block_new - a_block) convolved with (a - a_block) 
            #                   + (a_block_new - a_block) convolved with (a_block - a_block)
            # This gets complicated. Let's use a simpler approach:
            # Try various updates to the block and pick the best.

            best_block_update = a_block
            best_block_score = M_cur

            # Strategy: try reducing the block's contribution to peaks
            peak_thresh = M_cur * 0.9
            peak_mask = conv_cur >= peak_thresh

            # Compute "pressure" on this block from peaks
            pressure = np.zeros(end - start)
            for m in np.where(peak_mask)[0]:
                for li, gi in enumerate(idx):
                    j = m - gi
                    if 0 <= j < n and j not in idx:
                        pressure[li] += a[j]
            p_max = pressure.max()
            if p_max > 1e-20:
                pressure = pressure / p_max

            # Try: reduce high-pressure coords, increase low-pressure
            for scale in [0.7, 0.8, 0.9]:
                a_try = a_block.copy()
                a_try -= (1 - scale) * a_block * pressure
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try * a_block.sum()  # maintain block sum
                # Actually let's just try the full vector
                a_full_try = a.copy()
                a_full_try[idx] = a_try
                s_full = a_full_try.sum()
                if s_full < 1e-15:
                    continue
                a_full_try = a_full_try / s_full
                conv_try = fast_convolve(a_full_try)
                M_try = conv_try.max()
                if M_try < best_block_score - 1e-14:
                    best_block_score = M_try
                    best_block_update = a_full_try[idx].copy()

            # Also try: redistribute within block based on pressure
            if not improved or best_block_score < best_M - 1e-14:
                a[idx] = best_block_update
                s = a.sum()
                a = a / s
                conv = fast_convolve(a)
                M_new = conv.max()
                score = 2.0 * n * M_new
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    best_M = M_new
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD {label} it={it} ({elapsed:.1f}s): score={best_score:.8f}")

    return best_a, best_score


def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fast coordinate descent with vectorized peak tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            for factor in [0.85, 0.9, 0.95, 0.97, 0.99, 1.01, 1.03, 1.05, 1.1, 1.15]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                if new_val > 1.0:
                    continue

                new_affected = affected.copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        else:
            no_improve_count = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.1f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            break

    return best_a, best_score


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver_cached
    if hasattr(__builtins__, '__dict__') or True:
        pass
    # Check if already cached
    try:
        if globals().get('_lp_solver_cached') is not None:
            return globals()['_lp_solver_cached']
    except Exception:
        pass
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.GLOP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                globals()['_lp_solver_cached'] = solver_name
                return solver_name
        except Exception:
            continue
    globals()['_lp_solver_cached'] = cp.ECOS
    return cp.ECOS


def lp_fictitious_play(a0, time_limit, solver=None, n_iters_max=50, verbose=False, label=""):
    """
    Fictitious play: alternate between solving LP best responses.
    Uses averaging for the output.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    a_avg = a.copy()
    b_avg = a.copy()

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        a_new, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:
            break

        # Update averages
        alpha = 1.0 / (it + 1)
        a_avg = (1 - alpha) * a_avg + alpha * a_new
        b_avg = (1 - alpha) * b_avg + alpha * b_new

        # Evaluate candidates
        for cand in [a_new, b_new, a_avg, b_avg, 0.5 * (a_avg + b_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand.copy()

        a = a_new
        if verbose and it % 5 == 0:
            print(f"    LPFP {label} it={it}: best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 6))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = profile[len(profile) // 2]
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Initialize with bimodal distribution."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_copy(a, rng, strength=0.05):
    """Add multiplicative noise to a sequence."""
    a_new = a * (1.0 + rng.standard_normal(len(a)) * strength)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:
    - Nesterov projected subgradient with top-k smoothing (primary optimizer)
    - Fast coordinate descent (polishing)

    - Block coordinate descent (medium-scale optimization)

            if np.isfinite(p_score):

                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Generate diverse candidates ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t1_start = time.monotonic()

    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    for n_try in n_values:
        t_left = 120.0 - (time.monotonic() - t1_start)

        init_cands.append(("symmetric", init_symmetric(n_try, rng)))
        init_cands.append(("bimodal", init_bimodal(n_try, rng)))
        init_cands.append(("uniform", np.ones(n_try) / n_try))

        for name, cand in init_cands:
            t_left = 120.0 - (time.monotonic() - t1_start)

            t_alloc = max(3.0, min(8.0, t_left / 4.0))

    print(f"Phase 1 done. Top-3: " + ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in candidates[:3]]))

        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{name_c}_n{n_c}")

    # === Phase 2: Nesterov Projected Subgradient (primary optimizer) ===
    print(f"\n=== Phase 2: Nesterov Projected Subgradient (best={best_score:.8f}) ===")
    t2_start = time.monotonic()

    print(f"\n=== Phase 3: Intensive FCD (best={best_score:.8f}) ===")

                best_seq, best_score = check_and_update(a_opt2, best_seq, best_score, f"fcd_p{rnd}")

    # === Phase 4: Block Coordinate Descent ===
    print(f"\n=== Phase 4: Block CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  BCD n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = block_coord_descent(best_seq, t_alloc, block_size=16,
                                               verbose=True, label="bcd")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "bcd")

    # === Phase 5: LP Fictitious Play ===
    print(f"\n=== Phase 5: LP Fictitious Play (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try at the current size and one alternative
        fp_sizes = [n_best]
        if n_best != 1024:
            fp_sizes.append(1024)
        fp_sizes = fp_sizes[:2]

        for n_fp in fp_sizes:
            rem = deadline - time.monotonic() - 200
            if rem < 60:
                break
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  LPFP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp)
            a_fp, s_fp = lp_fictitious_play(initial, t_alloc, solver=solver,
                                            n_iters_max=30, verbose=True, label=f"fp_n{n_fp}")
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"lpfp_n{n_fp}")

    # === Phase 6: Nesterov MD with different parameters ===
    print(f"\n=== Phase 6: NPS variants (best={best_score:.8f}) ===")
    if best_seq is not None:
        for step_init in [None, 2.0 / (len(best_seq) ** 0.5), 0.5 / (len(best_seq) ** 0.5)]:
            for topk_v in [8, 15]:
                rem = deadline - time.monotonic() - 150
                if rem < 20:
                    break
                t_alloc = min(rem * 0.25, 80.0)
                a_opt, s_opt = nesterov_projected_subgrad(best_seq, t_alloc, topk=topk_v,
                                                          init_step=step_init, label=f"nps2_k{topk_v}")
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score,
                                                        f"nps2_k{topk_v}_s{step_init}")

    # === Phase 7: Fine polishing ===
    print(f"\n=== Phase 7: Fine Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine CD
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_fine = min(rem * 0.4, 120.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd_1")

        # Fast CD
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_cd = min(rem * 0.5, 80.0)
            a_cd, s_cd = fast_coord_descent(best_seq, t_cd, label="fcd_fin")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "fcd_final")

        # Final fine pass
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_f2, s_f2 = fine_coord_descent(best_seq, min(rem * 0.6, 40.0), label="fine2")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_cd_2")

    # === Phase 8: Try alternative sizes ===
    print(f"\n=== Phase 8: Alt Sizes (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [n_best + 128, n_best - 128, n_best + 256]:
            if alt_n < 256 or alt_n > 3000:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alt = min(rem * 0.4, 60.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt = nesterov_projected_subgrad(init_alt, min(t_alt, 40.0), topk=10,
                                                      label=f"alt{alt_n}")
            a_alt2, s_alt2 = fast_coord_descent(a_alt, min(t_alt * 0.5, 30.0), label=f"alt_fcd{alt_n}")
            best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
