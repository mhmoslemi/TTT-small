import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_projected_gradient(a0, time_limit, T_start=1.0, T_end=200.0,

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                                                lr_init=0.08, n_stages=5, steps_per_stage=1500)
        try_update(a_alt, f"alt_ng_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = nesterov_lp_refine(a_alt, min(rem * 0.3, 35.0), solver=solver, n_max_iters=3000)
            try_update(a_alt2, f"alt_lp_n{alt_n}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=20000)
        try_update(a_fin, "final_lp")

    # === Phase 9: Final coord descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.20, 40.0)
        a_cd2, _ = coordinate_descent_polish(best_seq, t_cd, n_steps=50000)
        try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
