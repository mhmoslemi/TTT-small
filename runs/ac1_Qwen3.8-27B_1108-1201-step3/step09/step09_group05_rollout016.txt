    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        b = fast_convolve(a)
        b_max = b.max()
        if b_max < 1e-300:
            break

        # Find top peaks (within 5% of max)
        threshold = b_max * 0.95
        peak_indices = np.where(b >= threshold)[0]

        # Compute weighted subgradient
        # grad_j = sum_{k in peaks} alpha_k * 2 * a[k-j] (valid)
        # Use normalized weights on peaks
        peak_weights = b[peak_indices] / b[peak_indices].sum()

        # c[j] = sum_k' peak_weights[k'] * 2 * a[peak_indices[k'] - j]
        # Use FFT: c = 2 * cross_correlation(weighted_v, a)
        v = np.zeros(2 * n - 1)
        v[peak_indices] = peak_weights
        v_flip = v[::-1]
        conv_len = 3 * n - 2
        fft_size = 1 << (conv_len - 1).bit_length()
        V = rfft(v_flip, fft_size)
        A = rfft(a, fft_size)
        conv_va = irfft(V * A, fft_size)[:conv_len]
        c = conv_va[n - 1:2 * n - 1][::-1]

        grad_a = 2.0 * c

        # Adaptive step size
        eta_curr = eta / (1.0 + 0.01 * step)

        # Multiplicative update: a_j <- a_j * exp(-eta * grad_j)
        # But we want to MINIMIZE, so move in -gradient direction
        # a_j <- a_j * exp(eta * (-grad_j)) = a_j * exp(-eta * grad_j)
        # Actually the objective is to minimize max_b, and grad_a points in 
        # direction of increasing the peaks. So we should decrease a in 
        # directions where grad_a is large.
        # a_j <- a_j * exp(-eta * grad_j / max(grad))
        
        grad_max = grad_a.max()
        if grad_max < 1e-300:
            break
        a_new = a * np.exp(-eta_curr * grad_a / grad_max)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Try smaller step
            a_half = a * np.exp(-0.3 * eta_curr * grad_a / grad_max)
            a_half = np.maximum(a_half, 0)
            a_half = a_half / a_half.sum()
            sc_half = compute_score(a_half)
            if sc_half < best_score:
                best_score = sc_half
                best_a = a_half.copy()
                a = a_half
                no_improve = 0
            else:
                no_improve += 1
                # Decay eta
                if no_improve % 500 == 0:
                    eta *= 0.95

    return best_a, best_score


def lbgfs_smooth_max(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=10):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return float(log_sum_exp)
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # grad_a_j = 2 * sum_k w_k * a[k-j]
            w_flip = w[::-1]
            conv_len = 3 * n - 2
            fft_size = 1 << (conv_len - 1).bit_length()
            W = rfft(w_flip, fft_size)
            A = rfft(a, fft_size)
            conv_wa = irfft(W * A, fft_size)[:conv_len]
            c = conv_wa[n - 1:2 * n - 1][::-1]

            grad_a = 2.0 * c
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 1000

    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(T),
                x0=z,
                jac=make_jac(T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
            a_curr = softmax_from_z(z)
            s = compute_score(a_curr)
            if s < best_score:
                best_score = s
                best_a = a_curr.copy()
        except Exception:
            pass
        z = z_from_a(best_a)

    return best_a, best_score


def direct_peak_shave(a0, time_limit, n_steps=20000):
    """Directly shave the maximum peak by reducing contributing coordinates."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            break

        b = fast_convolve(a)
        # Find the peak (exclude very edge positions which are naturally small)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -1
            b_work[-1] = -1
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total = contribs.sum()
        if total < 1e-300:
            break

        # Reduce top contributors
        top_n = min(10, len(idx))
        top_local = np.argsort(contribs)[::-1][:top_n]

        improved = False
        for ti in top_local:
            i = idx[ti]
            for factor in [0.95, 0.90, 0.85, 0.75]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                # Redistribute to low-contributing positions
                low_local = np.argsort(contribs)[:n // 3]
                low_positions = idx[low_local]
                low_positions = low_positions[low_positions != i]
                if len(low_positions) > 0:
                    low_vals = a_trial[low_positions]
                    if low_vals.sum() > 1e-15:
                        a_trial[low_positions] = low_vals + diff * low_vals / low_vals.sum()
                    else:
                        a_trial[low_positions] = low_vals + diff / len(low_positions)
                a_trial = norm_sequence(a_trial)
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
        else:
            stall = 0

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 1:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif mode == 2:
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif mode == 4:
        # Multi-scale structure
        a = np.exp(-3 * x * (1 - x)) * (1 + 0.3 * np.cos(6 * np.pi * x))
    elif mode == 5:
        # Symmetric bump with ripples
        base = np.exp(-4 * (x - 0.5) ** 2)
        ripple = np.cos(8 * np.pi * x) * 0.2 * base
        a = base + ripple
    elif mode == 6:
        # Wide flat top
        a = np.exp(-8 * (x - 0.5) ** 4)
    elif mode == 7:
        # Uniform-ish with edge decay
        a = 1.0 - 0.3 * x * (1 - x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def random_init(n, rng):
    """Random initialization with some structure."""
    a = rng.random(n)
    # Add some smooth structure
    from scipy.ndimage import gaussian_filter1d
    a = gaussian_filter1d(a, sigma=3)
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Multi-start initialization with scoring
    2. L-BFGS with power-sum smoothing (adaptive p)
    3. Log-sum-exp L-BFGS refinement
    4. Mirror descent peak equalization
    5. Direct peak shaving
    6. Final low-temperature polish
    7. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 0: Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, best_score={best_score:.6f}")

    # === Phase 1: Multi-start at current length ===
    print(f"\n=== Phase 1: Multi-Start (budget remaining {deadline - time.monotonic():.0f}s) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_per_init = min(rem * 0.05, 15.0)
        for mode in range(8):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_init = structured_init(n_best, rng, mode=mode)
            sc_init = compute_score(a_init)
            if sc_init < best_score * 1.05:  # Within 5% of best
                print(f"  Mode {mode}: init score = {sc_init:.6f}")
                # Quick refinement
                t_quick = min(t_per_init, rem * 0.3)
                if t_quick > 5:
                    a_ref, _ = lbgfs_power_sum(a_init, t_quick, p_start=4, p_end=16, n_stages=4, maxiter_per_stage=500)
                    try_update(a_ref, f"init_mode{mode}")

    # === Phase 2: Main L-BFGS with power-sum ===
    print(f"\n=== Phase 2: L-BFGS Power-Sum (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, p: 8->64")
        a_res, _ = lbgfs_power_sum(best_seq, t_alloc, p_start=8, p_end=64, n_stages=10, maxiter_per_stage=2000)
        try_update(a_res, "lbgfs_powersum_main")

    # === Phase 3: Log-sum-exp L-BFGS ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 30->0.5")
        a_res2, _ = lbgfs_smooth_max(best_seq, t_alloc, T_start=30.0, T_end=0.5, n_stages=10)
        try_update(a_res2, "lbgfs_logsumexp")

    # === Phase 4: Mirror Descent Peak Equalization ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res3, _ = mirror_descent_peak_eq(best_seq, t_alloc, n_steps=200000)
        try_update(a_res3, "mirror_descent")

    # === Phase 5: Direct Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res4, _ = direct_peak_shave(best_seq, t_alloc, n_steps=50000)
        try_update(a_res4, "peak_shave")

    # === Phase 6: Low-temperature polish ===
    print(f"\n=== Phase 6: Low-T Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s, T: 5->0.1")
        a_res5, _ = lbgfs_smooth_max(best_seq, t_alloc, T_start=5.0, T_end=0.1, n_stages=8)
        try_update(a_res5, "lowT_polish")

    # === Phase 7: Second round of power-sum with higher p ===
    print(f"\n=== Phase 7: High-p L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s, p: 32->256")
        a_res6, _ = lbgfs_power_sum(best_seq, t_alloc, p_start=32, p_end=256, n_stages=8, maxiter_per_stage=1000)
        try_update(a_res6, "high_p_lbgfs")

    # === Phase 8: Multi-length exploration ===
    print(f"\n=== Phase 8: Multi-Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        alt_lengths = []
        for mult in [1.25, 1.5, 0.75, 1.33, 2.0]:
            alt_n = int(n_best * mult)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                alt_lengths.append(alt_n)
        for fixed_n in [512, 768, 1024, 1280, 1536, 2048]:
            if 256 <= fixed_n <= 4096 and fixed_n != n_best:
                alt_lengths.append(fixed_n)
        alt_lengths = sorted(set(alt_lengths))

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_total = min(rem * 0.20, 120.0)
            if t_total < 20:
                break
            print(f"  n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_seq, alt_n)
            t_sub = t_total * 0.6
            a_alt, _ = lbgfs_power_sum(init_alt, t_sub, p_start=8, p_end=32, n_stages=6, maxiter_per_stage=1000)
            try_update(a_alt, f"alt_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
                t_sub2 = min(rem * 0.5, t_total * 0.3)
                if t_sub2 > 10:
                    a_alt2, _ = lbgfs_smooth_max(best_seq, t_sub2, T_start=10.0, T_end=0.5, n_stages=5)
                    try_update(a_alt2, f"alt2_n{alt_n}")

    # === Phase 9: Perturbation + re-optimization ===
    print(f"\n=== Phase 9: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        for scale in [0.05, 0.10, 0.20]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)
            t_pert = min(rem * 0.3, 60.0)
            if t_pert > 10:
                a_pert, _ = lbgfs_power_sum(perturbed, t_pert, p_start=8, p_end=32, n_stages=5, maxiter_per_stage=800)
                try_update(a_pert, f"pert_{scale}")

    # === Phase 10: Final combined polish ===
    print(f"\n=== Phase 10: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t1 = min(rem * 0.3, 40.0)
        if t1 > 5:
            a_f1, _ = lbgfs_smooth_max(best_seq, t1, T_start=3.0, T_end=0.1, n_stages=5)
            try_update(a_f1, "final_lse")
        rem = deadline - time.monotonic()
        if rem > 20:
            t2 = min(rem * 0.3, 30.0)
            if t2 > 5:
                a_f2, _ = direct_peak_shave(best_seq, t2, n_steps=10000)
                try_update(a_f2, "final_ps")
        rem = deadline - time.monotonic()
        if rem > 15:
            t3 = min(rem * 0.4, 25.0)
            if t3 > 5:
                a_f3, _ = mirror_descent_peak_eq(best_seq, t3, n_steps=50000)
                try_update(a_f3, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
