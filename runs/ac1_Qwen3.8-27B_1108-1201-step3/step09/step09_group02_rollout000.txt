```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast exact coordinate descent with adaptive search range
2. Pair coordinate descent for quadratic structure
3. Spectral initialization with non-negacity enforcement
4. Iterative peak-lag targeting
5. Nesterov-accelerated subgradient with momentum
6. Multi-length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_score(a):
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.maximum(a[:n], 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_bell(n, rng, width_scale=None):
    t = np.arange(n, dtype=np.float64)
    center = n / 2.0
    if width_scale is None:
        width_scale = rng.uniform(0.2, 0.5)
    width = n * width_scale
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    return a / s

def init_dual_peak(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.65, 0.8)
    w = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    s = a.sum()
    return a / s

def init_fourier_random(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    # Random phases, slightly peaked at low frequencies
    freqs = np.arange(n_freq)
    envelope = 1.0 / (1.0 + 0.1 * freqs)
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    spectrum = envelope * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.maximum(a[:n], 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    if old_n == 1:
        return np.full(new_n, f_norm[0] / new_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_subgradient(a0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient method with simplex projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    
    # Nesterov momentum
    y = a.copy()  # lookahead point
    beta = 0.0  # momentum parameter
    
    t0 = time.monotonic()
    lr = 2.0 / np.sqrt(n)  # initial learning rate
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        max_val = float(conv_y.max())
        
        # Find peaks (top few for smoothing)
        top_k = min(8, 2 * n - 1)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        
        # Soft weighting
        pv = conv_y[peaks]
        temp = 0.01 * max_val
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()
        
        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-6:
                continue
            k = int(peaks[idx])
            # d/d(a[i]) conv[k] = a[k-i] + a[i-k] (with bounds)
            # conv[k] = sum_j a[j]*a[k-j]
            # d/da[i] = a[k-i] (from j=i term) + a[i-k] (from j=k-i term) if i != k-i, else a[i]^2 derivative = 2*a[i]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            indices = np.arange(i_start, i_end + 1)
            partners = k - indices
            valid = (partners >= 0) & (partners < n)
            indices_valid = indices[valid]
            partners_valid = partners[valid]
            
            if np.any(indices_valid == partners_valid):
                # Some indices have i = k-i, i.e., i = k/2
                is_self = indices_valid == partners_valid
                grad[indices_valid[is_self]] += 2.0 * w[idx] * y[indices_valid[is_self]]
                not_self = ~is_self
                if np.any(not_self):
                    grad[indices_valid[not_self]] += w[idx] * (y[partners_valid[not_self]] + y[indices_valid[not_self]])
            else:
                grad[indices_valid] += w[idx] * (y[partners_valid] + y[indices_valid])
        
        # Update beta (Nesterov)
        t_k = it + 1
        beta = (1.0 - 1.0 / np.sqrt(t_k + 1.0)) / 2.0  # standard Nesterov beta
        
        # Decay learning rate
        lr = 2.0 / (np.sqrt(n) * np.sqrt(t_k))
        
        # Projected gradient step
        y_new = simplex_project(y - lr * grad)
        
        # Nesterov update
        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
            s_new = 1.0
        a_new = a_new / s_new
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        sc = 2.0 * n * float(conv_new.max())
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = a_new.copy()
        a = a_new.copy()
        
        if no_improve > 500:
            break
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_targeting_descent(a0, time_limit, verbose=False):
    """
    Targeted descent: identify the peak lag and specifically reduce it
    by adjusting the pairs of coordinates that contribute to it.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    s = 1.0
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    
    outer_it = 0
    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = float(conv[peak_k])
        
        # Find all pairs (i, j) such that i + j = peak_k (for convolution)
        # conv[peak_k] = sum_i a[i] * a[peak_k - i] where 0 <= i < n, 0 <= peak_k - i < n
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        
        if i_start > i_end:
            break
        
        indices = np.arange(i_start, i_end + 1)
        partners = peak_k - indices
        
        # We want to reduce sum_i a[i]*a[partner_i]
        # Strategy: for each pair, reduce the product by moving mass from the larger to the smaller
        # (this reduces the product while maintaining sum)
        
        improved = False
        for rep in range(3):  # multiple passes
            if time.monotonic() - t0 > time_limit:
                break
            
            for idx in range(len(indices)):
                i = indices[idx]
                j = partners[idx]
                if i == j:
                    # Single term: a[i]^2, can't reduce without changing a[i]
                    continue
                ai = a[i]
                aj = a[j]
                if ai < 1e-10 or aj < 1e-10:
                    continue
                
                # Move small amount of mass from larger to smaller
                # This reduces ai*aj
                diff = ai - aj
                if abs(diff) < 1e-10:
                    continue
                transfer = 0.1 * abs(diff)  # transfer 10% of the difference
                if ai > aj:
                    a[i] -= transfer
                    a[j] += transfer
                else:
                    a[i] += transfer
                    a[j] -= transfer
            
            # Re-normalize
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss < 1e-15:
                break
            a = a / ss
            
            # Check improvement
            conv_check = fast_convolve(a)
            sc = 2.0 * n * float(conv_check.max())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Try a different strategy: add small noise and re-optimize
            break
    
    return best_a, best_score
```
