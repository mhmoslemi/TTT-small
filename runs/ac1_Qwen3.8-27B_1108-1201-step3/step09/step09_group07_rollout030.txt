import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_grad(a, T):
    """Gradient of (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * cross_correlation(w, a)[j]
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr

def mstep_update(a, grad, eta):
    """Multiplicative weight (mirror descent) update."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def mstep_optimize(a0, time_limit, T_start=5.0, T_end=200.0, 
                    eta_start=0.1, n_stages=10, steps_per_stage=2000):
    """Mirror descent (MWU) with temperature annealing."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage in range(n_stages):
        T = temps[stage]
        eta = eta_start * (T_start / T)
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            grad = smooth_max_grad(a, T)
            a = mstep_update(a, grad, eta)
            if step % 100 == 0:
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if step % 500 == 0:
                        print(f"      MWU stage={stage} step={step}: score={best_score:.6f} T={T:.1f}")
    
    return best_a, best_score

        j_star = np.argmin(grad)
        gamma = 2.0 / (it + 2)
        b = (1 - gamma) * b
        b[j_star] += gamma
    
    return b

    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve_auto(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, T):
    """Gradient of (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a."""
    n = len(a)
    b = fast_convolve_auto(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def mstep_update(a, grad, eta):
    """Multiplicative weight (mirror descent) update on simplex."""
    log_a = np.log(np.maximum(a, 1e-30))
    new_log_a = log_a - eta * grad
    new_log_a = new_log_a - new_log_a.max()
    a_new = np.exp(new_log_a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def mstep_optimize(a0, time_limit, T_start=5.0, T_end=300.0,
                    eta_start=0.05, n_stages=10, steps_per_stage=2000):
    """Mirror descent (MWU) with temperature annealing."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage in range(n_stages):
        T = temps[stage]
        eta = eta_start * (T_start / T)
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            grad = smooth_max_grad(a, T)
            a = mstep_update(a, grad, eta)
            if step % 200 == 0:
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    return best_a, best_score


def nesterov_pg_optimize(a0, time_limit, T_start=2.0, T_end=200.0,
                         lr_init=0.08, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev_calc = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0
                beta = (t_prev_calc - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = smooth_max_grad(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    return best_a, best_score


def frank_wolfe_best_response(a_fixed, n_iters=5000, warm_start=None):
    """Fast best response using Frank-Wolfe (conditional gradient).
    Minimizes max_k conv(a_fixed, b)_k over b in simplex.
    Returns (b, max_val).
    """
    n = len(a_fixed)
    if warm_start is not None:
        b = project_simplex(warm_start.copy())
    else:
        b = np.ones(n) / n

    t = fast_convolve_ab(a_fixed, b)
    best_M = t.max()

    for it in range(n_iters):
        k_star = int(np.argmax(t))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad = a_fixed[k_star - idx]
        if grad.size == 0:
            break
        j_star = js + int(np.argmin(grad))
        gamma = 2.0 / (it + 2)
        b = (1.0 - gamma) * b
        b[j_star] += gamma
        t = fast_convolve_ab(a_fixed, b)
        M = t.max()
        if M < best_M:
            best_M = M
    return b, float(best_M)


def nesterov_fr_wolfe(a0, time_limit, n_iters_max=50000, fr_iters=2000):
    """Nesterov-accelerated Frank-Wolfe best-response fixed-point iteration."""
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    iter_count = 0

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break
        # Nesterov lookahead
        t_curr = 1.0 + np.sqrt(1.0 + 4.0 * iter_count * iter_count) / 2.0
        if iter_count > 0:
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (iter_count - 1) * (iter_count - 1)) / 2.0
            beta = (t_prev - 1.0) / t_curr
        else:
            beta = 0.0
        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        # Frank-Wolfe best response to y
        b_new, M_new = frank_wolfe_best_response(y, n_iters=fr_iters, warm_start=a)

        # Update
        alpha = 2.0 / (iter_count + 2)
        a_prev = a.copy()
        a = (1.0 - alpha) * a + alpha * b_new
        a = project_simplex(a)
        iter_count += 1

        if iter_count % 50 == 0:
            sc_a = score_normalized(a)
            sc_y = score_normalized(y)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
            if iter_count % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      FW-NS iter={iter_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, warm_start=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b) via LP."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if warm_start is not None:
        b_var.value = warm_start
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=5000)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=5000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refinement(a0, time_limit, solver=None, n_max_iters=20000, use_fw=True):
    """LP-based (or FW-based) Nesterov dual averaging refinement."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Nesterov
        t_curr = 1.0 + np.sqrt(1.0 + 4.0 * it * it) / 2.0
        if it > 0:
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (it - 1) * (it - 1)) / 2.0
            beta = (t_prev - 1.0) / t_curr
        else:
            beta = 0.0
        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        # Best response
        if use_fw:
            b_new, _ = frank_wolfe_best_response(y, n_iters=1500, warm_start=a)
        else:
            b_new, _ = solve_lp_best_response(y, solver, warm_start=a)
            if b_new is None:
                b_new, _ = frank_wolfe_best_response(y, n_iters=1000, warm_start=a)

        alpha = 2.0 / (it + 2)
        a_prev = a.copy()
        a = (1.0 - alpha) * a + alpha * b_new
        a = project_simplex(a)

        if it % 20 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP-NS iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=30000):
    """Multi-peak shaving with coordinate adjustments."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve_auto(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        # Find top contributors across all near peaks
        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:6]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:12]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.88, 0.82]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction at the main peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:5]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.90]:
                    for f2 in [0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Try simultaneous reduction
            a_trial = a.copy()
            for i in top_coords[:4]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.96
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                # Try small random perturbation
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_asymmetric_init(n, alpha, rng):
    """Create asymmetric initial sequence."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + alpha * np.sin(2 * np.pi * x + 0.3)
    elif choice == 1:
        a = 1.0 + alpha * (2 * x - 1)
    elif choice == 2:
        a = 1.0 + alpha * np.sin(4 * np.pi * x)
    elif choice == 3:
        a = np.exp(-2.5 * (x - 0.3) ** 2) + 0.4 * np.exp(-2.5 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 1.0 + alpha * np.cos(2 * np.pi * x * 1.5)
    elif choice == 5:
        a = 1.0 + alpha * x * (1 - x) * 4
    else:
        a = 1.0 + alpha * np.sin(3 * np.pi * x + 1.0)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining MWU, Frank-Wolfe best-response,
    LP refinement, peak shaving, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_asymmetric_init(n_init, 0.6, rng)
        best_score = score_normalized(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov Projected Gradient ===
    print(f"\n=== Phase 1: NG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg_optimize(best_seq, t_alloc, T_start=2.0, T_end=200.0,
                                       lr_init=0.06, n_stages=8, steps_per_stage=2000)
        try_update(a_ng, "ng_main")

    # === Phase 2: MWU (Mirror Descent) ===
    print(f"\n=== Phase 2: MWU (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, _ = mstep_optimize(best_seq, t_alloc, T_start=5.0, T_end=200.0,
                                 eta_start=0.04, n_stages=8, steps_per_stage=2000)
        try_update(a_mw, "mwu_1")

    # === Phase 3: Frank-Wolfe Best-Response (main workhorse) ===
    print(f"\n=== Phase 3: FW-NS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fw, _ = nesterov_fr_wolfe(best_seq, t_alloc, n_iters_max=20000, fr_iters=1500)
        try_update(a_fw, "fw_ns_1")

    # === Phase 4: LP Refinement ===
    print(f"\n=== Phase 4: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refinement(best_seq, t_alloc, solver=solver,
                                n_max_iters=10000, use_fw=False)
        try_update(a_lp, "lp_1")

    # === Phase 5: FW refinement (faster, more iterations) ===
    print(f"\n=== Phase 5: FW-NS Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fw2, _ = nesterov_fr_wolfe(best_seq, t_alloc, n_iters_max=30000, fr_iters=1000)
        try_update(a_fw2, "fw_ns_2")

    # === Phase 6: Peak Shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")

    # === Phase 7: Multi-Start with different inits ===
    print(f"\n=== Phase 7: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        alphas = [0.3, 0.7, 1.0, 0.5]
        for alpha in alphas:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.20, 100.0)
            if t_sub < 25:
                break
            a_init = make_asymmetric_init(n_best, alpha, rng)
            # Quick FW

            try_update(a_sub, f"fw_a{alpha}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = lp_refinement(a_sub, min(rem * 0.2, 40.0), solver=solver,
                                          n_max_iters=3000, use_fw=True)
                try_update(a_sub2, f"fwref_a{alpha}")

    # === Phase 8: Alternative lengths ===
    print(f"\n=== Phase 8: Alternative Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for delta in [-40, -20, -10, 10, 20, 40, 60, 80]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 4096:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.15, 120.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_fr_wolfe(init_alt, t_total * 0.6, n_iters_max=5000, fr_iters=1000)
        try_update(a_alt, f"fw_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = lp_refinement(a_alt, min(rem * 0.25, 40.0), solver=solver,
                                      n_max_iters=3000, use_fw=False)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 90.0)
        a_fin, _ = lp_refinement(best_seq, t_fin, solver=solver,
                                 n_max_iters=5000, use_fw=False)
        try_update(a_fin, "final_lp")

    # === Phase 10: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.25, 60.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=50000)
        try_update(a_ps2, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
