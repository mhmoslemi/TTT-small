    # Compute subgradient: g = (1/|A|) * sum_{k in A} 2 * Q_k * a
    g = np.zeros(n)
    for k in A:

        for j in range(n):

            if 0 <= l < n:

    g /= len(A)

    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0)

    if s > 0:

    else:

    return a_new

Let me also think about the "peak equalization" step more carefully:

Given a current sequence a and the top K peaks at lags k_1, ..., k_K with values M_1 >= M_2 >= ... >= M_K, we want to find a small perturbation delta such that:
- (a+delta)^T Q_{k_i} (a+delta) ≈ (a+delta)^T Q_{k_j} (a+delta) for all i,j
- sum(delta) = 0
- a + delta >= 0
- The common value is < M_1

To first order, (a+delta)^T Q_k (a+delta) ≈ a^T Q_k a + 2 delta^T Q_k a = c_k + 2 (Q_k a) · delta

So we want: 2 (Q_{k_i} a) · delta - 2 (Q_{k_j} a) · delta ≈ c_{k_j} - c_{k_i} for all i,j

Or equivalently: (Q_{k_i} a - Q_{k_j} a) · delta ≈ (c_{k_j} - c_{k_i})/2

This is a linear system! We can solve it (least squares) to find delta.

Let me code this up:

```python
def peak_equalize(a, K=5):
    """Try to equalize the top K peaks of conv(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    
    # Get top K peaks
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    M = top_vals[0]
    
    # Only work with peaks that are within 5% of max
    active = [i for i in top_idx if c[i] > 0.95 * M]
    
    if len(active) < 2:
        return a
    
    # Build the linear system
    # For each pair (i, j) in active, we want:
    # 2*(Q_{active[i]} a - Q_{active[j]} a) · delta = c[active[j]] - c[active[i]]
    
    # But this is over-determined. Use least squares.
    # Also add constraint sum(delta) = 0.
    
    m = len(active)
    # Number of equations: C(m,2) + 1 (for sum constraint)
    # But let's just use m-1 independent equations (make all equal to the first) + 1 sum constraint
    
    # Actually, let's use: for each active peak i (except the first):
    # 2*(Q_{active[i]} a - Q_{active[0]} a) · delta = c[active[0]] - c[active[i]]
    # Plus: sum(delta) = 0
    
    rows = []
    rhs = []
    
    for i in range(1, m):
        k_i = active[i]
        k_0 = active[0]
        # (Q_k a)_j = a[k-j] if 0 <= k-j < n
        grad_i = np.zeros(n)
        grad_0 = np.zeros(n)
        for j in range(n):
            l_i = k_i - j
            if 0 <= l_i < n:
                grad_i[j] = a[l_i]
            l_0 = k_0 - j
            if 0 <= l_0 < n:
                grad_0[j] = a[l_0]
        rows.append(2.0 * (grad_i - grad_0))
        rhs.append(c[k_0] - c[k_i])
    
    # Add sum constraint
    rows.append(np.ones(n))
    rhs.append(0.0)
    
    A_mat = np.array(rows)
    b_vec = np.array(rhs)
    
    # Solve least squares
    delta, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
    
    # Apply small step
    for alpha in [0.1, 0.05, 0.01, 0.005]:
        a_new = a + alpha * delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            continue
        a_new /= s
        score_new = 2.0 * n * fast_convolve(a_new).max()
        score_old = 2.0 * n * M
        if score_new < score_old:
            return a_new
    return a
```

Now let me also think about the **frequency domain** approach:

The auto-convolution a*a has DFT: DFT(a*a) = DFT(a) * DFT(a) = |DFT(a)|^2 (for real a, the DFT of the convolution is the product of DFTs).

Wait, that's for cyclic convolution. For linear convolution, it's a bit different, but if we zero-pad to length >= 2n-1, then the DFT of the zero-padded convolution is the product of the DFTs.

So: DFT(a*a)[f] = |A[f]|^2 where A = DFT(a) (with zero padding).

We want to minimize the peak of the inverse DFT of |A[f]|^2. By Parseval, the sum of |A[f]|^2 is constant (= (2n-1) * sum(a^2) for the zero-padded version). So we want to make |A[f]|^2 as flat as possible, which means making |A[f]| as flat as possible.

This is the "flat spectrum" design problem! We want a non-negative sequence a whose DFT has as flat a magnitude as possible.

This is related to "maximum entropy" spectral design. The sequence with the flattest spectrum (among non-negative sequences with fixed sum) would be the one that maximizes the entropy of |A[f]|^2.

One approach: minimize the variance of |A[f]|^2 (or equivalently, minimize the sum of |A[f]|^4 / (sum |A[f]|^2)^2).

This is the "fourth-order spectrum" criterion, which is a smooth surrogate for the peak criterion.

Let me add this as another optimization objective:

```python
def spectral_flatness_objective(a):
    """Minimize the peak of |A(f)|^2, equivalently maximize spectral flatness."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    power = np.abs(A) ** 2
    # The peak of the inverse DFT of power is what we want to minimize
    # But minimizing the peak of power directly is easier
    return power.max()
```

Actually, minimizing the peak of |A(f)|^2 is NOT the same as minimizing the peak of a*a. The peak of a*a is the peak of the inverse DFT of |A(f)|^2, which is different from the peak of |A(f)|^2 itself.

However, if |A(f)|^2 is very flat, then its inverse DFT will also be relatively flat (concentrated at lag 0 for the auto-correlation). So there's a correlation.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_conv(c, n):
    return 2.0 * n * c.max()

def p_norm_obj_and_grad_z(z, p):
    """Objective and gradient in log-domain for p-norm of top convolution values."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return 2.0 * n, np.zeros_like(z)
    val = (S) ** (1.0 / p)
    obj = 2.0 * n * val
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    # Gradient w.r.t. a: 2 * factor * correlate(w, a)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * n * factor * corr
    # Gradient w.r.t. z
    g_z = a * (g_a - np.dot(g_a, a))
    return obj, g_z

def lbfgs_highp(a0, time_limit, p_values, max_iter_per_p=5000):
    """L-BFGS with schedule of increasing p values."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    
    for p in p_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        def func(zv):
            val, _ = p_norm_obj_and_grad_z(zv, p)
            return val
        
        def jac(zv):
            _, g = p_norm_obj_and_grad_z(zv, p)
            return g
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_p, 'ftol': 1e-20, 
                                   'gtol': 1e-15, 'maxfun': max_iter_per_p * 50, 'disp': False})
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score_from_conv(fast_convolve(a_cur), n)
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS p={p} ({elapsed:.0f}s): score={sc:.8f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def peak_equalize(a, K=5, n_trials=20):
    """Equalize top K peaks using linear system."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    
    for trial in range(n_trials):
        c = fast_convolve(best_a)
        M = c.max()
        
        # Get active set: peaks within 10% of max
        threshold = M * 0.9
        active = np.where(c >= threshold)[0]
        
        if len(active) < 2:
            break
        
        # Take top K
        active = active[np.argsort(c[active])[::-1]][:K]
        
        # Build linear system: for each active peak i (except first),
        # 2*(Q_{k_i} a - Q_{k_0} a) · delta = c[k_0] - c[k_i]
        k0 = active[0]
        rows = []
        rhs = []
        
        for i in range(1, len(active)):
            ki = active[i]
            grad_i = np.zeros(n)
            grad_0 = np.zeros(n)
            for j in range(n):
                li = ki - j
                if 0 <= li < n:
                    grad_i[j] = best_a[li]
                l0 = k0 - j
                if 0 <= l0 < n:
                    grad_0[j] = best_a[l0]
            rows.append(2.0 * (grad_i - grad_0))
            rhs.append(c[k0] - c[ki])
        
        # Add sum constraint
        rows.append(np.ones(n))
        rhs.append(0.0)
        
        A_mat = np.array(rows)
        b_vec = np.array(rhs)
        
        delta, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        
        # Try different step sizes
        improved = False
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_new = best_a + alpha * delta
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 0.01:
                continue
            a_new /= s
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score - 1e-15:
                best_a = a_new
                best_score = sc
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score

def subgradient_refine(a, time_limit, n_max_iters=500000):
    """Nesterov-accelerated subgradient method on the exact max function."""
    n = len(a)
    a = np.maximum(a.copy(), 0)
    a /= a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    a_prev = a.copy()
    y = a.copy()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c_y = fast_convolve(y)
        M_y = c_y.max()
        
        # Active set: top 3-5 peaks
        top_k = np.argsort(c_y)[::-1][:5]
        active = [k for k in top_k if c_y[k] > 0.95 * M_y]
        
        if not active:
            active = [top_k[0]]
        
        # Subgradient: average of 2*Q_k*y for k in active
        g = np.zeros(n)
        for k in active:
            for j in range(n):
                l = k - j
                if 0 <= l < n:
                    g[j] += 2.0 * y[l]
        g /= len(active)
        
        # Nesterov step
        t_val = it + 1
        a_new = y - (2.0 / (t_val + 3.0)) * g * n  # scale by n for normalization
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            a_new = y.copy()
        else:
            a_new /= s
        
        beta = (t_val - 1) / (t_val + 2) if t_val > 1 else 0
        y = a_new + beta * (a_new - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 0.01:
            y /= sy
        else:
            y = a_new.copy()
        
        a_prev = a_new
        
        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-stage high-p L-BFGS in log-domain (p up to 20000)
2. Active-set peak equalization via least-squares linear systems
3. Nesterov-accelerated exact subgradient refinement
4. Structured multi-perturbation restarts
5. Narrow length exploration with interpolation
6. Iterative KKT-based active set refinement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def p_norm_func_z(z, p):
    """Objective: 2*n*(sum c^p)^(1/p) in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    S = np.sum(c_pos ** p)
    if S < 1e-300:
        return 2.0 * n
    return 2.0 * n * S ** (1.0 / p)


def p_norm_grad_z(z, p):
    """Gradient of p-norm objective w.r.t. log-domain z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * n * factor * corr
    g_z = a * (g_a - np.dot(g_a, a))
    return g_z


def lbfgs_stage(a0, time_limit, p_schedule, max_iter_per_stage=4000):
    """Run L-BFGS through a schedule of p values in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    for p_val in p_schedule:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 10:
            break

        max_it = max(500, min(max_iter_per_stage, int(remaining * 30)))

        def obj(zv):
            return p_norm_func_z(zv, p_val)

        def grad(zv):
            return p_norm_grad_z(zv, p_val)

        try:
            res = minimize(obj, z, jac=grad, method='L-BFGS-B',
                           options={'maxiter': max_it, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxfun': max_it * 100,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS p={p_val} ({elapsed:.0f}s): score={sc:.8f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def qk_times_a(k, a):
    """Compute Q_k * a where (Q_k a)_j = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    result = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        result[idx] = a[k - idx]
    return result


def peak_equalize(a, max_trials=100, K=5):
    """Equalize top K convolution peaks using linearized least-squares."""
    n = len(a)
    best_a = np.maximum(a.copy(), 0)
    s = best_a.sum()
    if s < 0.01:
        return best_a, compute_score(best_a)
    best_a = best_a / s
    best_score = compute_score(best_a)
    t0 = time.monotonic()

    for trial in range(max_trials):
        if time.monotonic() - t0 > 300:
            break

        c = fast_convolve(best_a)
        M = c.max()
        threshold = M * 0.92
        active_all = np.where(c >= threshold)[0]

        if len(active_all) < 2:
            break

        active = active_all[np.argsort(c[active_all])[::-1]][:K]
        m = len(active)
        k0 = active[0]

        rows = []
        rhs = []
        for i in range(1, m):
            ki = active[i]
            gi = qk_times_a(ki, best_a)
            g0 = qk_times_a(k0, best_a)
            rows.append(2.0 * (gi - g0))
            rhs.append(c[k0] - c[ki])

        rows.append(np.ones(n))
        rhs.append(0.0)

        A_mat = np.array(rows)
        b_vec = np.array(rhs)

        try:
            delta, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except Exception:
            break

        improved = False
        for alpha in [0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]:
            a_new = best_a + alpha * delta
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 0.01:
                continue
            a_new = a_new / sn
            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                best_a = a_new
                best_score = sc
                improved = True
                break

        if not improved:
            # Try shrinking the peak coordinate directly
            peak_j = int(np.argmax(best_a))
            for factor in [0.98, 0.95, 0.92]:
                a_new = best_a.copy()
                diff = a_new[peak_j] * (1 - factor)
                a_new[peak_j] -= diff
                mask = np.ones(n, dtype=bool)
                mask[peak_j] = False
                others = a_new[mask].sum()
                if others < 1e-10:
                    continue
                a_new[mask] += diff * a_new[mask] / others
                sc = compute_score(a_new)
                if sc < best_score - 1e-14:
                    best_a = a_new
                    best_score = sc
                    improved = True
                    break

            if not improved:
                break

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, max_iters=2000000):
    """Nesterov-accelerated subgradient on exact max of conv(a,a)."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 0.01:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    a_prev = a.copy()
    y = a.copy()
    scale = n * best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_convolve(y)
        M_y = c_y.max()
        top_k = 5
        top_idx = np.argsort(c_y)[::-1][:top_k]
        active = [int(k) for k in top_idx if c_y[k] > 0.93 * M_y]
        if not active:
            active = [int(top_idx[0])]

        g = np.zeros(n)
        for k in active:
            g += qk_times_a(k, y)
        g *= 2.0 / len(active)

        t_val = float(it + 1)
        step = 2.0 / (t_val + 3.0) * scale
        a_new = y - step * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 0.01:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        if it > 0:
            beta = (t_val - 1.0) / (t_val + 2.0)
            y = a_new + beta * (a_new - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 0.01:
                y = y / sy
            else:
                y = a_new.copy()

        a_prev = a_new

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequences."""
    if variant == 0:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3.0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6.0 * x * (1.0 - x))
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def kkt_refine(a0, n_active=5, n_iters=200):
    """
    KKT-based refinement: iteratively solve for the dual variables
    and update the primal to satisfy complementary slackness.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 0.01:
        return a, compute_score(a)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        c = fast_convolve(a)
        M = c.max()

        # Active set
        thresh = M * 0.95
        active = np.where(c >= thresh)[0]
        if len(active) > n_active:
            active = active[np.argsort(c[active])[::-1]][:n_active]

        if len(active) < 1:
            break

        # Solve for dual weights d and lambda
        # System: sum_k d_k * (Q_k a) = lambda * 1 (for support coords)
        #         sum_k d_k = 1
        #         d_k >= 0
        m = len(active)
        support = np.where(a > 1e-12)[0]
        n_s = len(support)

        if n_s < 2 or m < 1:
            break

        # Build matrix: rows = support coords, cols = [d_0,...,d_{m-1}, lambda]
        A_rows = np.zeros((n_s, m + 1))
        for j_idx, j in enumerate(support):
            for i, k in enumerate(active):
                A_rows[j_idx, i] = 2.0 * a[k - j] if 0 <= (k - j) < n else 0.0
            A_rows[j_idx, m] = -1.0

        # Last row: sum d = 1
        A_full = np.zeros((n_s + 1, m + 1))
        A_full[:n_s, :] = A_rows
        A_full[n_s, :m] = 1.0
        b_full = np.zeros(n_s + 1)
        b_full[n_s] = 1.0

        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_full, b_full, rcond=None)
        except Exception:
            break

        d = sol[:m]
        d = np.maximum(d, 0)
        ds = d.sum()
        if ds < 1e-10:
            d = np.ones(m) / m
        else:
            d = d / ds

        # Weighted gradient
        g_weighted = np.zeros(n)
        for i, k in enumerate(active):
            g_weighted += d[i] * qk_times_a(k, a)
        g_weighted *= 2.0

        # Projected step: reduce along g_weighted, maintain sum=1, a>=0
        # Use line search
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005]:
            a_new = a - alpha * (g_weighted - g_weighted.mean())
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 0.01:
                continue
            a_new = a_new / sn
            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                a = a_new
                best_score = sc
                best_a = a.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_score
