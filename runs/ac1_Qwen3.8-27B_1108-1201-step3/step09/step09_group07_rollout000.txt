"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-based fictitious play
with dual-guided updates, spectral equalization, and multi-length exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def build_convolution_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        row_start = j
        T[row_start:row_start + n, j] = a
    return T

        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_best_response(a_fixed, solver=None, get_dual=False):
    """
    Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b).
    Returns (b_star, M_star, dual_z) where dual_z are the dual variables.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_convolution_matrix(a_fixed)
    
    if get_dual:
        constraints = []
        dual_vars = []
        for k in range(size):
            cv = cp.Variable()
            constraints.append(T[k, :] @ b_var - cv <= 0)
            dual_vars.append(cv)
        # Actually this isn't the right way to get duals in cvxpy
        # Let me use a different approach
        
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), None


def lp_fictitious_play(a0, time_limit, solver=None, n_max_iters=100000):
    """
    LP-based fictitious play with Nesterov acceleration.
    Maintains y (virtual coordinate) and a (running average).
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b, _ = solve_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y / y_sum
        
        # Running average (Hedge/Exponentiated Gradient step)
        a = ((it + 1) * a + b_new) / (it + 2)
        a = project_simplex(a)
        
        # Evaluate
        if it % 10 == 0:
            score_a = score_normalized(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            
            score_y = score_normalized(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        
        # Periodic reporting
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FictPlay iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}, current_a={score_normalized(a):.6f}")
    
    return best_a, best_score


def spectral_equalization(a0, time_limit, n_iters=500):
    """
    Try to flatten the power spectrum |FFT(a)|^2, which is related to
    flattening the autocorrelation.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current spectrum
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum (but peak at DC is unavoidable due to sum constraint)
        # We want to reduce the peaks in the spectrum
        target = np.ones(fft_size // 2 + 1)
        target[0] = power[0]  # Keep DC component
        
        # Error in spectrum domain
        error = power - target * power.mean()
        
        # Convert error to time domain
        # d(a)/d(power) is complicated, so use gradient-free approach
        # Instead, use the fact that modifying a in time domain affects
        # the spectrum via the FFT
        
        # Simple approach: compute the IDFT of the error and use it as a direction
        # This is related to the "error reduction" algorithm
        error_full = np.zeros(fft_size)
        error_full[:fft_size // 2 + 1] = error
        # Make it conjugate symmetric for real output
        error_full[1:fft_size // 2 + 1] = error_full[fft_size // 2::-1]
        if fft_size % 2 == 0:
            error_full[fft_size // 2 + 1:] = error_full[fft_size // 2 - 1:0:-1]
        else:
            error_full[fft_size // 2 + 1:] = error_full[fft_size // 2 - 1:0:-1]
        
        direction = irfft(rfft(error_full, fft_size), fft_size)[:n]
        # This isn't quite right either. Let me use a simpler approach.
        
        # Alternative: directly modify a to reduce peaks in convolution
        b = fast_convolve(a)
        peak_k = np.argmax(b)
        peak_val = b[peak_k]
        
        # Find coordinates contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Reduce top contributors
        top_n = min(5, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_n]
        
        a_new = a.copy()
        for ci in top_idx:
            i = idx[ci]
            a_new[i] *= 0.95
            j = peak_k - i
            if j != i and 0 <= j < n:
                a_new[j] *= 0.98
        
        a_new = project_simplex(a_new)
        score = score_normalized(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller reduction
            a_new = a.copy()
            for ci in top_idx[:2]:
                i = idx[ci]
                a_new[i] *= 0.98
            a_new = project_simplex(a_new)
            score = score_normalized(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                break
    
    return best_a, best_score


def nesterov_smoothed_gradient(a0, time_limit, T_start=1.0, T_end=100.0,
                                 lr=0.05, n_steps=None):
    """
    Nesterov-accelerated gradient descent on log-sum-exp smoothed objective.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    if n_steps is None:
        n_steps = 10000
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = step / n_steps
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov step
        if step == 0:
            y = a
        else:
            beta = (np.sqrt(1 + 4 * (step - 1) ** 2) - 1) / (2 * step)
            y = a + beta * (a - a_prev)
            y = project_simplex(y)
        
        # Compute smoothed gradient
        b = fast_convolve(y)
        Tb = T * b
        m = Tb.max()
        e = np.exp(Tb - m)
        w = e / e.sum()
        
        # Gradient: d/da [log(sum exp(T * conv(a,a)))] 
        # = 2 * (a correlated with w)  [using the chain rule through convolution]
        # conv(a,a)[k] = sum_i a[i] * a[k-i]
        # d/da[j] conv(a,a)[k] = a[k-j] + (a[j-k] if j!=k else 0) ... 
        # Actually: d/da[j] (a^T T_k a) = 2 * (T_k a)[j] (since T_k is symmetric)
        # So grad[j] = 2 * sum_k w[k] * (T_k a)[j] = 2 * sum_k w[k] * sum_i T_k[j,i] * a[i]
        # = 2 * sum_k w[k] * sum_i (a[k-j] if valid) * a[i]... 
        # 
        # Simpler: grad of log-sum-exp of conv(a,a) w.r.t. a
        # = sum_k w[k] * grad_a conv(a,a)[k]
        # grad_a conv(a,a)[k] = 2 * (the vector whose j-th component is a[k-j] (if valid))
        # = 2 * (a reversed and shifted, padded)
        #
        # So grad = 2 * sum_k w[k] * (a reversed, shifted by k, padded to length n)
        # = 2 * (w * a_reversed) cross-correlation... 
        #
        # More precisely: grad[j] = 2 * sum_k w[k] * a[k-j] (where a[k-j] = 0 if out of range)
        # = 2 * sum_k w[k] * a[max(0, min(n-1, k-j))]... no, just 0 if out of range
        # = 2 * (cross-correlation of w and a, evaluated at appropriate positions)
        #
        # Let me just compute it directly using FFT:
        # grad[j] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-j] (with a[i]=0 for i<0 or i>=n)
        # = 2 * sum_{k=j}^{j+2n-2} w[k] * a[k-j]  (changing variable l=k-j)
        # = 2 * sum_{l=0}^{n-1} w[l+j] * a[l]  (where w[m]=0 for m<0 or m>=2n-1)
        # = 2 * (w padded) correlated with a, at lag j
        
        # Compute using FFT: correlation of w (length 2n-1) with a (length n)
        w_padded = np.zeros(2 * n - 1)
        w_padded[:] = w
        corr_size = n + (2 * n - 1) - 1
        corr_fft_size = 1 << (corr_size - 1).bit_length()
        W = rfft(w_padded, corr_fft_size)
        A = rfft(a, corr_fft_size)
        # Cross-correlation: corr[j] = sum_l w_padded[l+j] * a[l]
        # This is the convolution of reversed w_padded with a
        w_rev = w_padded[::-1]
        W_rev = rfft(w_rev, corr_fft_size)
        corr = irfft(W_rev * A, corr_fft_size)[:corr_size]
        # corr[j] = sum_l w_rev[corr_size-1-l] * a[l] ... hmm this is getting confusing
        # Let me just do it more carefully
        
        # Actually, the simplest way:
        # grad[j] = 2 * sum_k w[k] * a[k-j] for k-j in [0,n-1]
        # Let's flip a: a_flip[i] = a[n-1-i]
        # Then a[k-j] = a_flip[n-1-(k-j)] = a_flip[(n-1-j)-k+1-j]... 
        # 
        # Let me just compute it as:
        # For each j, grad[j] = 2 * sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} w[k] * a[k-j]
        # = 2 * <w[j:j+n], a>  (dot product of subvector of w with a)
        # = 2 * (sliding dot product of w with a)
        #
        # This can be computed as a cross-correlation!
        # cross_corr[j] = sum_i w[j+i] * a[i] for i=0..n-1
        # = correlation of w and a at lag j
        
        # Using FFT for cross-correlation:
        # corr = ifft(fft(w) * conj(fft(a)))  (with appropriate padding)
        pad_size = 1 << (2 * n - 1).bit_length()
        W_pad = rfft(w_padded, pad_size)
        A_pad = rfft(a, pad_size)
        # Cross-correlation: sum_i w[i+j] * a[i] = ifft(conj(fft(w_pad)) * fft(a_pad))[j]
        # Actually: ifft(conj(W) * A) gives sum_i w[i] * a[i+j]... 
        # Let me use: corr[j] = sum_i a[i] * w[i+j]
        # FFT of a: A, FFT of w: W
        # ifft(A * conj(W))[j] = sum_i a[i] * w[i-j]... 
        # Hmm, the sign conventions are tricky. Let me just use scipy.
        
        from scipy.signal import fftconvolve
        # corr[j] = sum_{i=0}^{n-1} a[i] * w[i+j]
        # This is the cross-correlation of a and w at lag j
        # = (a * w_flipped)[j + (2n-2)]  where * is linear convolution
        w_flipped = w_padded[::-1]
        conv_result = fftconvolve(a, w_flipped, mode='full')
        # conv_result[k] = sum_i a[i] * w_flipped[k-i] = sum_i a[i] * w_padded[2n-2-(k-i)]
        # = sum_i a[i] * w_padded[2n-2-k+i]
        # We want corr[j] = sum_i a[i] * w_padded[i+j]
        # So 2n-2-k = -(j) => k = 2n-2+j
        corr = conv_result[n-1:n-1+n]  # Should give us what we want... let me verify
        # Actually this is getting too complicated. Let me just use a direct method for the gradient.
        
        # Direct computation (O(n^2) but fine for n up to ~2000)
        # grad[j] = 2 * sum_k w[k] * a[k-j] (with a[m]=0 for m not in [0,n-1])
        # = 2 * sum_{k=j}^{min(j+2n-2, 2n-2)} w[k] * a[k-j]
        # For efficiency, use the fact that this is a cross-correlation
        
        # Let me just do it with numpy's correlate
        # np.correlate(w_padded, a, mode='valid') gives sum_i w_padded[i+j] * a[i] for j=0..(len(w)-len(a))
        # = sum_i w_padded[i+j] * a[i] for j=0..(2n-2)-(n-1) = 0..n-1
        grad_half = np.correlate(w_padded, a, mode='valid')
        # Wait, np.correlate(a, v) = sum_n a[n+k] * conj(v[n])
        # So np.correlate(w_padded, a, 'valid') = sum_n w_padded[n+k] * a[n] for n=0..len(a)-1
        # = sum_n w_padded[n+k] * a[n] = what we want for grad[k]/2!
        # The 'valid' mode gives output of length max(2n-1, n) - min(2n-1, n) + 1 = n
        # So grad_half[k] = sum_{n=0}^{n-1} w_padded[n+k] * a[n] for k=0..n-1
        # And grad = 2 * grad_half
        
        grad = 2.0 * grad_half
        
        # Gradient descent step with projection
        a_prev = a
        lr_step = lr / (1.0 + 0.01 * step)
        a_new = project_simplex(y - lr_step * grad)
        a = a_new
        
        if step % 200 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                
    return best_a, best_score
