import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def coord_descent_optimized(a0, time_limit, verbose=False, label=""):

    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

        -0.5, -0.3, -0.2, -0.1, -0.05, -0.02, -0.01,
        0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length = k_end - k_start + 1

            # Unaffected max
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            # a[k-i] for k in [k_start, k_end] means a[0:length]
            a_seg = a[0:length]

            best_delta = 0.0
            best_new_max = M

            # Evaluate multiple step sizes
            for factor in step_factors:
                delta = factor * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15 or new_val > 1.0:
                    continue

                # New affected = affected + 2*delta*a_seg + delta^2 at position 2i
                new_affected = affected + 2.0 * delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:

                    new_affected[idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = a[i] + best_delta
                sum_a += best_delta

                changes = 2.0 * best_delta * a_seg

                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)

                    best_score = cur_score
                    best_a = a.copy()
                improved = True

            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.10f}")

    return best_a, best_score


def fine_polish(a0, time_limit):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    fine_factors = np.array([
        -0.5, -0.3, -0.1, -0.05, -0.02, -0.01, -0.005,
        0.005, 0.01, 0.02, 0.05, 0.1, 0.3, 0.5
    ])

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length = k_end - k_start + 1

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_seg = a[0:length]

            best_delta = 0.0
            best_new_max = M

            for factor in fine_factors:
                delta = factor * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = a[i] + best_delta
                sum_a += best_delta

                changes = 2.0 * best_delta * a_seg
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score


def nesterov_md(a0, time_limit, lr_init=0.01, beta_init=40.0):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it == 1:
            y = a.copy()
        else:
            y = (1.0 + gamma) * a - gamma * y
            y = simplex_project(y)

        # Compute smooth gradient via log-sum-exp
        conv_y = fast_convolve(y)
        max_conv = float(conv_y.max())
        scaled = beta * (conv_y - max_conv)
        scaled = np.clip(scaled, -60, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient computation (sparse: only non-negligible weights)
        grad = np.zeros(n)
        active = np.where(weights > 1e-8)[0]
        for k in active:
            w = weights[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y[k - j]

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-9:
                break
        elif it % 20 == 0:
            beta *= 1.01

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        a_prev = a
        a = a_new

    return best_a, best_score


def lp_fictitious_play(n, time_limit, initial=None, solver=None):
    """LP fictitious play with Nesterov-style averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma_it = 2.0 / (iters + 2.0)
        a_avg = (1.0 - gamma_it) * a_avg + gamma_it * a
        b_avg = (1.0 - gamma_it) * b_avg + gamma_it * b

        for cand in [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        if iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_u_shaped(n, rng, curve=2.0):
    """U-shaped initialization: high at edges, low in middle."""
    t = np.linspace(0, 1, n)
    a = (1.0 - t) ** curve + t ** curve + 0.1
    s = a.sum()
    return a / s


def init_exp_decay(n, rng, alpha_range=(0.5, 3.0)):
    """Exponential decay from left to right."""
    alpha = rng.uniform(*alpha_range)
    t = np.linspace(0, 1, n)
    a = np.exp(-alpha * t)
    s = a.sum()
    return a / s


def init_power_asym(n, rng, p_range=(0.5, 4.0)):
    """Power-law asymmetric profile."""
    p = rng.uniform(*p_range)
    t = np.linspace(0, 1, n)
    a = (1.0 - t) ** p + 0.01
    s = a.sum()
    return a / s


def init_compensated(n, rng):
    """Compensated uniform: slightly U-shaped to flatten autocorrelation."""
    t = np.linspace(0, 1, n)
    # Start from uniform and add small U-shaped perturbation
    amp = rng.uniform(0.1, 0.4)
    a = 1.0 + amp * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def init_multi_gaussian(n, rng):
    """Sum of a few Gaussian bumps at random positions."""
    a = np.zeros(n)
    n_bumps = int(rng.integers(2, 6))
    for _ in range(n_bumps):
        center = rng.uniform(0.05, 0.95) * n
        width = rng.uniform(0.02, 0.15) * n
        amp = rng.uniform(0.5, 1.5)
        t = np.arange(n)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    a += 0.01  # floor
    s = a.sum()
    return a / s


def perturb_asymmetric(a0, rng, strength=0.1):
    """Asymmetric perturbation: shift mass from one side to the other."""
    n = len(a0)
    a = a0.copy()
    # Create a tilt
    t = np.linspace(0, 1, n)
    tilt = rng.standard_normal()
    a = a * (1.0 + strength * tilt * (t - 0.5))
    # Add noise
    a = a * (1.0 + rng.standard_normal(n) * strength * 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses asymmetric initialization, fast coordinate descent, LP fictitious play,
    and Nesterov mirror descent with adaptive time allocation.
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-length diverse initialization + quick CD ===
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()
    p1_budget = 200.0

    n_values = [1024, 1280, 1536, 2048, 2560, 3072]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > p1_budget - 30:
            break

        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_seq(parent, n)))
        inits.append(("u_shape", init_u_shaped(n, rng, curve=rng.uniform(1.5, 3.0))))
        inits.append(("exp", init_exp_decay(n, rng)))
        inits.append(("power", init_power_asym(n, rng)))
        inits.append(("comp", init_compensated(n, rng)))
        inits.append(("gauss", init_multi_gaussian(n, rng)))
        inits.append(("uniform", np.ones(n) / n))

        for name, cand in inits:
            t_left = p1_budget - (time.monotonic() - t1_start) - 20
            if t_left < 5:
                break
            t_alloc = max(3.0, min(15.0, t_left / max(1, len(inits))))
            a_opt, s_opt = coord_descent_optimized(cand, t_alloc, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done. Generated {len(candidates)} candidates.")
    for c in candidates[:5]:
        print(f"    score={c[0]:.6f} n={c[2]} init={c[3]}")

    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    print(f"  After Phase 1: best_score={best_score:.6f}")

    # === Phase 2: Intensive coordinate descent on best ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()
        n_rounds = 5
        for rnd in range(n_rounds):
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem * 0.2, 120.0)
            print(f"  CD round {rnd+1}/{n_rounds}, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")
            a_opt, s_opt = coord_descent_optimized(current, t_alloc, verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_cd_r{rnd}")

            # Perturb for next round (except last)
            if rnd < n_rounds - 2:
                current = perturb_asymmetric(best_seq, rng, strength=0.05 + 0.03 * rnd)
            else:
                current = best_seq.copy()

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Use a slightly smaller n for LP speed if n is very large
        lp_n = min(n_best, 1536)

        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem * 0.35, 180.0)
            print(f"  LPFP n={lp_n}, budget={t_alloc:.0f}s")
            initial = interpolate_seq(best_seq, lp_n)
            a_fp, score_fp = lp_fictitious_play(lp_n, t_alloc, initial=initial, solver=solver)
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"lpfp_n{lp_n}")

            # CD polish on LPFP result
            rem = deadline - time.monotonic() - 200
            if rem > 20 and score_fp < best_score + 0.01:
                t_cd = min(rem * 0.3, 60.0)
                a_pol, s_pol = coord_descent_optimized(a_fp, t_cd, label="lpfp_polish")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "lpfp_cd_polish")

    # === Phase 4: Nesterov Mirror Descent variants ===
    print(f"\n=== Phase 4: Nesterov MD (best={best_score:.6f}) ===")
    if best_seq is not None:
        for lr_try, beta_try in [(0.01, 40.0), (0.005, 60.0), (0.02, 30.0)]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_md = min(rem * 0.25, 80.0)
            print(f"  NMD lr={lr_try} beta={beta_try}, budget={t_md:.0f}s")
            a_md, s_md = nesterov_md(best_seq, t_md, lr_init=lr_try, beta_init=beta_try)
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, f"md_lr{lr_try}")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best + 256, n_best + 512, n_best + 1024]
        for alt_n in alt_lengths:
            if alt_n > 4096 or alt_n < 512:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alt = min(rem * 0.3, 60.0)
            print(f"  Alt n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt = coord_descent_optimized(init_alt, t_alt, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

            # Quick polish
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_f, s_f = fine_polish(a_alt, min(rem * 0.4, 30.0))
                best_seq, best_score = check_and_update(a_f, best_seq, best_score, f"alt{alt_n}_polish")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Fine polish
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_fine = min(rem * 0.4, 100.0)
            print(f"  Fine polish, budget={t_fine:.0f}s")
            a_f1, s_f1 = fine_polish(best_seq, t_fine)
            best_seq, best_score = check_and_update(a_f1, best_seq, best_score, "fine_1")

        # Another CD round
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_cd_final = min(rem * 0.5, 60.0)
            a_cdf, s_cdf = coord_descent_optimized(best_seq, t_cd_final, label="final_cd")
            best_seq, best_score = check_and_update(a_cdf, best_seq, best_score, "final_cd")

        # Final fine polish
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_f2, s_f2 = fine_polish(best_seq, min(rem * 0.6, 30.0))
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
