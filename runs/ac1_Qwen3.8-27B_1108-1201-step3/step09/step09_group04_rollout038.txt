```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective: 2*n*max(convolve(a,a))/(sum(a))^2"""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def p_norm_gradient_a(a, p):
    """Gradient of (sum c_k^p)^(1/p) w.r.t. a, where c = convolve(a,a)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(a)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    n = len(a)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * factor * corr
    return g_a


def p_norm_objective_a(a, p):
    """Compute 2n * (sum c_k^p)^(1/p) / (sum(a))^2 for given p."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    s = a.sum()
    return 2.0 * len(a) * val / (s * s)


def p_norm_log_z_objective(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_log_z_gradient(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=10):
    """L-BFGS with adaptive p-norm smoothing schedule in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            return p_norm_log_z_objective(z_val, stage_p)

        def jac(z_val):
            return p_norm_log_z_gradient(z_val, stage_p)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': 10000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

                    g_b[i] = a[j]
            # Projected subgradient step on simplex
            lr = 1.0 / np.sqrt(n * (_inner + 1))
            b = b - lr * g_b
            b = np.maximum(b, 0)
            sb = b.sum()
            if sb > 0:
                b = b / sb
        
        # Update a using Nesterov
        alpha = 2.0 / (it + 2)
        a = (1 - alpha) * a + alpha * b
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 0:
            a = a / sa
        
        score = score_of(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def local_peak_shaving(a0, time_limit, n_steps=20000):
    """Local refinement by shaving peaks in the autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find the peak
        peak_k = c.argmax()
        peak_val = c[peak_k]
        
        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Reduce the top contributor
        if len(idx) == 0:
            break
        
        top_i_local = np.argmax(contribs)
        i = idx[top_i_local]
        j = peak_k - i
        
        if a[i] < 1e-14:
            break
        
        # Try reducing a[i] and redistributing
        improved = False
        for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
            diff = a[i] * (1 - factor)
            if diff < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[i] -= diff
            if a_trial[i] < 0:
                a_trial[i] = 0
                diff = a[i]
            # Redistribute to random positions (not near the peak)
            # Positions that don't contribute to the peak
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            # Avoid positions that would create new peaks
            # Simply redistribute proportionally
            others_sum = a_trial[mask].sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] += diff * a_trial[mask] / others_sum
            score = score_of(a_trial)
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try pair reduction
            for ci in range(min(5, len(idx))):
                ci_sorted = np.argsort(contribs)[::-1][ci]
                i2 = idx[ci_sorted]
                j2 = peak_k - i2
                if i2 == j2 or j2 >= n or j2 < 0 or a[i2] < 1e-14 or a[j2] < 1e-14:
                    continue
                for f1 in [0.90, 0.80]:
                    for f2 in [0.90, 0.80]:
                        a_trial = a.copy()
                        a_trial[i2] *= f1
                        a_trial[j2] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        score = score_of(a_trial)
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random perturbation every 500 steps
            if step % 500 == 0:
                rng = np.random.default_rng(step + 123)
                a_trial = a + 0.02 * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    score = score_of(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
            
            if not improved and step > 100:
                break
    
    return best_a, best_score


def asymmetric_init(n, variant=0, seed=42):
    """Create asymmetric initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    if variant == 0:
        # Skewed Gaussian
        center = 0.35 + 0.3 * rng.random()
        width = 0.15 + 0.1 * rng.random()
        a = np.exp(-((x - center) / width) ** 2)
    elif variant == 1:
        # Exponential decay from one end
        rate = 3 + 4 * rng.random()
        a = np.exp(-rate * x)
        a = a + 0.3 * np.exp(-rate * (1 - x))  # small bump on other side
    elif variant == 2:
        # Sine wave (asymmetric due to phase)
        phase = rng.uniform(0, 2 * np.pi)
        freq = 1.0 + rng.random() * 0.5
        a = (np.sin(2 * np.pi * freq * x + phase) + 1.5) ** 2
    elif variant == 3:
        # Power law
        alpha = 1 + 2 * rng.random()
        beta = 1 + 2 * rng.random()
        a = x ** alpha * (1 - x) ** beta
    elif variant == 4:
        # Step-like
        a = np.ones(n)
        mid = int(n * (0.3 + 0.4 * rng.random()))
        a[:mid] = 1.0
        a[mid:] = 0.3 + 0.7 * rng.random()
    elif variant == 5:
        # Two bumps
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w1 = 0.1 + 0.05 * rng.random()
        w2 = 0.1 + 0.05 * rng.random()
        a = 0.6 * np.exp(-((x - c1) / w1) ** 2) + 0.4 * np.exp(-((x - c2) / w2) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
- p-norm L-BFGS with aggressive scheduling (main workhorse)
- Active-set subgradient with momentum (escape mechanism)
- Multi-length exploration with diverse asymmetric initializations
- Local peak-shaving refinement
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def pnorm_obj_z(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def pnorm_jac_z(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def run_pnorm_lbfgs(a0, time_limit, p_start=10.0, p_end=800.0, n_stages=12):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                pnorm_obj_z,
                x0=z,
                jac=pnorm_jac_z,
                args=(stage_p,),
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': 15000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def run_subgradient(a0, time_limit, n_steps=10000, momentum=0.7):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    step_scale = 2.0 / np.sqrt(n)
    
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        gamma = (momentum + np.sqrt(1.0 + momentum * momentum)) / 2.0
        y = a + gamma * (y - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        # Compute subgradient at y
        c = fast_convolve(y)
        max_c = c.max()
        threshold = max_c * (1.0 - 1e-5)
        active = c >= threshold
        num_active = active.sum()
        if num_active == 0:
            active[np.argmax(c)] = True
            num_active = 1
        active_weight = active.astype(np.float64) / num_active
        g = 2.0 * np.correlate(active_weight, y, mode='valid')
        
        # Step
        step_size = step_scale / np.sqrt(it + 1)
        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        score = score_of(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score


def run_local_shave(a0, time_limit, n_steps=30000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        peak_k = c.argmax()
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        
        # Strategy 1: reduce top contributor
        top_local = np.argmax(contribs)
        i = idx[top_local]
        if a[i] > 1e-14:
            for factor in [0.95, 0.90, 0.85, 0.75]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                sc = score_of(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        # Strategy 2: pair reduction
        if not improved:
            sorted_idx = np.argsort(contribs)[::-1][:8]
            for ci in sorted_idx:
                i2 = idx[ci]
                j2 = peak_k - i2
                if i2 == j2 or j2 >= n or j2 < 0:
                    continue
                if a[i2] < 1e-14 or a[j2] < 1e-14:
                    continue
                for f1 in [0.90, 0.80]:
                    for f2 in [0.90, 0.80]:
                        a_trial = a.copy()
                        a_trial[i2] *= f1
                        a_trial[j2] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        # Strategy 3: random perturbation
        if not improved and step % 200 == 0:
            n_pert = rng.integers(1, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for ii in indices:
                a_trial[ii] *= (1.0 + rng.uniform(-0.04, 0.04))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved and step > 200:
            break
    
    return best_a, best_score


def make_asymmetric_init(n, variant, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if variant == 0:
        center = 0.3 + 0.4 * rng.random()
        width = 0.1 + 0.1 * rng.random()
        a = np.exp(-((x - center) / width) ** 2)
    elif variant == 1:
        rate = 2 + 5 * rng.random()
        a = np.exp(-rate * x) + 0.3 * np.exp(-rate * (1 - x))
    elif variant == 2:
        alpha = 0.5 + 3.0 * rng.random()
        beta = 0.5 + 3.0 * rng.random()
        a = x ** alpha * (1 - x) ** beta
    elif variant == 3:
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w1 = 0.08 + 0.08 * rng.random()
        w2 = 0.08 + 0.08 * rng.random()
        a = 0.6 * np.exp(-((x - c1) / w1) ** 2) + 0.4 * np.exp(-((x - c2) / w2) ** 2)
    elif variant == 4:
        freq = 1.0 + rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = (np.sin(2 * np.pi * freq * x + phase) + 1.5) ** 2
    else:
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
- p-norm L-BFGS with aggressive multi-stage smoothing (main workhorse)
- Active-set subgradient with momentum (escape from local minima)
- Multi-length exploration with diverse asymmetric initializations
- Local peak-shaving refinement
- Cycling between smooth and rough optimization phases
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def pnorm_obj_z(z, p):
    """Objective: 2n * (sum c^p)^(1/p) with a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def pnorm_jac_z(z, p):
    """Gradient of pnorm_obj_z w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def run_pnorm_lbfgs(a0, time_limit, p_start=10.0, p_end=800.0, n_stages=12):
    """L-BFGS with adaptive p-norm schedule in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                pnorm_obj_z,
                x0=z,
                jac=pnorm_jac_z,
                args=(stage_p,),
                method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': 20000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def run_subgradient(a0, time_limit, n_steps=10000, momentum=0.7, tol=1e-8):
    """Subgradient method with Nesterov momentum for min max(convolve(a,a))."""

    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    step_scale = 3.0 / np.sqrt(n)
    no_improve = 0
    
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        gamma = (momentum + np.sqrt(1.0 + momentum * momentum)) / 2.0
        y = a + gamma * (y - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        c = fast_convolve(y)
        max_c = c.max()
        threshold = max_c * (1.0 - 1e-5)
        active = c >= threshold
        num_active = active.sum()
        if num_active == 0:
            active[np.argmax(c)] = True
            num_active = 1
        active_weight = active.astype(np.float64) / num_active
        g = 2.0 * np.correlate(active_weight, y, mode='valid')
        
        step_size = step_scale / np.sqrt(it + 1)
        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        score = score_of(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 500:
            break
    
    return best_a, best_score


def run_local_shave(a0, time_limit, n_steps=40000):
    """Local refinement: peak shaving + pair reduction + perturbation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    fail_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        peak_k = c.argmax()
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        
        # Strategy 1: reduce top contributor, redistribute proportionally
        top_local = np.argmax(contribs)
        i = idx[top_local]
        if a[i] > 1e-14:
            for factor in [0.96, 0.92, 0.87, 0.80]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                sc = score_of(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        # Strategy 2: pair reduction at peak
        if not improved:
            sorted_idx = np.argsort(contribs)[::-1][:10]
            for ci in sorted_idx:
                i2 = idx[ci]
                j2 = peak_k - i2
                if i2 == j2 or j2 >= n or j2 < 0:
                    continue
                if a[i2] < 1e-14 or a[j2] < 1e-14:
                    continue
                for f1 in [0.92, 0.85]:
                    for f2 in [0.92, 0.85]:
                        a_trial = a.copy()
                        a_trial[i2] *= f1
                        a_trial[j2] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        # Strategy 3: multi-peak aware reduction
        if not improved:
            # Find second peak
            c2 = c.copy()
            c2[peak_k] = -1
            peak_k2 = c2.argmax()
            if c2[peak_k2] > 0.85 * c[peak_k]:
                js2 = max(0, peak_k2 - n + 1)
                je2 = min(n - 1, peak_k2)
                idx2 = np.arange(js2, je2 + 1)
                contribs2 = a[idx2] * a[peak_k2 - idx2]
                top2 = np.argmax(contribs2)
                i3 = idx2[top2]
                if a[i3] > 1e-14:
                    a_trial = a.copy()
                    a_trial[i] *= 0.90
                    a_trial[i3] *= 0.90
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 0.01:
                        a_trial = a_trial / st
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
        
        # Strategy 4: random perturbation
        if not improved and step % 100 == 0:
            n_pert = rng.integers(2, 8)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for ii in indices:
                a_trial[ii] *= (1.0 + rng.uniform(-0.05, 0.05))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if improved:
            fail_count = 0
        else:
            fail_count += 1
            if fail_count > 300:
                break
    
    return best_a, best_score


def make_asym_init(n, variant, seed):
    """Create asymmetric initial sequences of various types."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if variant == 0:
        center = 0.25 + 0.5 * rng.random()
        width = 0.08 + 0.12 * rng.random()
        a = np.exp(-((x - center) / width) ** 2)
    elif variant == 1:
        rate = 2.0 + 5.0 * rng.random()
        a = np.exp(-rate * x) + 0.3 * np.exp(-rate * (1.0 - x))
    elif variant == 2:
        alpha = 0.5 + 3.0 * rng.random()
        beta = 0.5 + 3.0 * rng.random()
        a = x ** alpha * (1.0 - x) ** beta
    elif variant == 3:
        c1 = rng.uniform(0.05, 0.35)
        c2 = rng.uniform(0.65, 0.95)
        w1 = 0.06 + 0.08 * rng.random()
        w2 = 0.06 + 0.08 * rng.random()
        a = 0.6 * np.exp(-((x - c1) / w1) ** 2) + 0.4 * np.exp(-((x - c2) / w2) ** 2)
    elif variant == 4:
        freq = 1.0 + rng.random() * 0.5
        phase = rng.uniform(0, 2.0 * np.pi)
        a = (np.sin(2.0 * np.pi * freq * x + phase) + 1.5) ** 2
    elif variant == 5:
        # Triangular with asymmetry
        peak = 0.2 + 0.6 * rng.random()
        a = np.maximum(0, 1.0 - np.abs(x - peak) / (peak + 0.1))
        a = a ** (1.0 + rng.random() * 2.0)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def eval_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def update_best(state, a, label):
    """Try to update best state. state = [score, sequence]."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = eval_safe(a)
    if np.isfinite(sc) and sc < state[0]:
        state[0] = sc
        state[1] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: p-norm L-BFGS + subgradient momentum + local shaving
    with aggressive multi-length exploration and asymmetric initialization.
    Uses cycling between smooth (p-norm) and rough (subgradient) phases to escape
    local minima.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)
    
    state = [float('inf'), None]
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            state[0] = ps
            state[1] = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_asym_init(960, 0, seed)
        state[0] = eval_safe(parent)
        state[1] = parent.copy()
        print(f"Default init: n=960, score={state[0]:.6f}")
    
    n_best = len(state[1])
    
    # Phase 1: p-norm L-BFGS (main workhorse, generous time)
    print(f"\n=== P1: p-norm L-BFGS n={n_best} (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, p: 8 -> 1200, 14 stages")
        a1 = run_pnorm_lbfgs(state[1], t1, p_start=8.0, p_end=1200.0, n_stages=14)
        if not update_best(state, a1, "pnorm1"):
            # Use local score to decide
            ls = score_of(a1)
            if ls < state[0] * 0.999:
                update_best(state, a1, "pnorm1_local")
    
    # Phase 2: Subgradient to explore different region
    print(f"\n=== P2: Subgradient (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 170.0)
    if t2 > 15:
        a2, _ = run_subgradient(state[1], t2, n_steps=8000, momentum=0.6)
        update_best(state, a2, "subgrad1")
    
    # Phase 3: p-norm again from potentially new position
    print(f"\n=== P3: p-norm again (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 300.0)
    if t3 > 20:
        a3 = run_pnorm_lbfgs(state[1], t3, p_start=20.0, p_end=2000.0, n_stages=10)
        update_best(state, a3, "pnorm2")
    
    # Phase 4: Local shaving
    print(f"\n=== P4: Local Shave (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.07, 100.0)
    if t4 > 10:
        a4, _ = run_local_shave(state[1], t4, n_steps=50000)
        update_best(state, a4, "shave1")
    
    # Phase 5: Multi-length exploration
    print(f"\n=== P5: Multi-Length (best={state[0]:.6f}) ===")
    lengths = []
    base = n_best
    for d in [-192, -160, -128, -96, -64, -32, 32, 64, 96, 128, 160, 192, 256, 320]:
        alt = base + d
        if 400 <= alt <= 2560:
            lengths.append(alt)
    for rn in [512, 640, 704, 768, 832, 896, 960, 1024, 1088, 1152, 1216, 1280, 1408, 1536, 1664]:
        if 400 <= rn <= 2560:
            lengths.append(rn)
    lengths = list(set(lengths))
    lengths.sort(key=lambda ln: abs(ln - n_best))
    
    explored = 0
    for alt_n in lengths[:10]:
        rem = deadline - time.monotonic()
        if rem < 70 or explored >= 7:
            break
        explored += 1
        t_total = min(rem * 0.13, 160.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s (explored={explored})")
        
        # Try multiple inits and pick best
        init_interp = interpolate_seq(state[1], alt_n)
        init_asym = make_asym_init(alt_n, explored % 6, seed + alt_n + explored)
        
        si = score_of(init_interp)
        sa = score_of(init_asym)
        a_start = init_interp if si < sa else init_asym
        
        # p-norm optimization
        t_pn = t_total * 0.65
        a_alt = run_pnorm_lbfgs(a_start, t_pn, p_start=15.0, p_end=500.0, n_stages=8)
        update_best(state, a_alt, f"ml_n{alt_n}")
        
        # If this length is promising, do subgradient too
        rem = deadline - time.monotonic()
        if rem > 20 and explored <= 4:
            t_sg = min(rem * 0.08, 50.0)
            a_alt2, _ = run_subgradient(state[1], t_sg, n_steps=3000, momentum=0.5)
            update_best(state, a_alt2, f"msg_n{alt_n}")
    
    # Phase 6: Cycling - alternate p-norm and subgradient
    print(f"\n=== P6: Cycling (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(state[1])
    cycle_budget = min(rem * 0.15, 200.0)
    if cycle_budget > 30:
        t_cycle = cycle_budget / 4.0
        for cyc in range(4):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            # Subgradient step (rough)
            t_sg = min(t_cycle * 0.4, rem * 0.3)
            if t_sg > 5:
                a_cyc, _ = run_subgradient(state[1], t_sg, n_steps=2000, momentum=0.65)
                update_best(state, a_cyc, f"cycle_sg{cyc}")
            # p-norm step (smooth)
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_pn = min(t_cycle * 0.6, rem * 0.4)
            if t_pn > 5:
                a_cyc2 = run_pnorm_lbfgs(state[1], t_pn, p_start=50 + cyc * 100,
                                          p_end=500 + cyc * 500, n_stages=5)
                update_best(state, a_cyc2, f"cycle_pn{cyc}")
    
    # Phase 7: Perturbation exploration
    print(f"\n=== P7: Perturbation (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        for ns in [0.03, 0.08, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            pert = state[1] + ns * rng.standard_normal(n_cur)
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            t_sub = min(rem * 0.25, 50.0)
            if t_sub < 8:
                break
            a_pert = run_pnorm_lbfgs(pert, t_sub, p_start=30.0, p_end=600.0, n_stages=6)
            update_best(state, a_pert, f"pert_{ns}")
    
    # Phase 8: Final high-p polish + local shave
    print(f"\n=== P8: Final (best={state[0]:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.35, 70.0)
        a8 = run_pnorm_lbfgs(state[1], t8, p_start=500.0, p_end=10000.0, n_stages=5)
        update_best(state, a8, "final_pn")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        a8b, _ = run_local_shave(state[1], min(rem * 0.40, 35.0), n_steps=20000)
        update_best(state, a8b, "final_shave")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        a8c, _ = run_subgradient(state[1], min(rem * 0.50, 20.0), n_steps=3000, momentum=0.8)
        update_best(state, a8c, "final_sg")
    
    # Final verification with actual evaluate_sequence
    if state[1] is not None:
        fs = eval_safe(state[1])
        if np.isfinite(fs):
            state[0] = fs
        else:
            # Fallback to parent
            try:
                p = np.asarray(height_sequence_1, dtype=np.float64)
                p = np.clip(p, 0.0, 1000.0)
                if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
                    state[1] = p / p.sum()
                    state[0] = eval_safe(state[1])
            except Exception:
                state[1] = np.ones(960) / 960
                state[0] = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: {state[0]:.6f} (n={len(state[1])}, time={elapsed:.0f}s)")
    return [float(x) for x in state[1]]
```
