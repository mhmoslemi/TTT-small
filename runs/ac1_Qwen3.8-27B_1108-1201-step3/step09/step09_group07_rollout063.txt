import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_A_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    n = len(a)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        # A[k,j] = a[k-j] for k=0,...,2n-2 where 0 <= k-j < n
        # So k ranges from j to j+n-1
        A[j:j+n, j] = a
    return A

def solve_best_response_lp(a):
    """Solve LP: min M s.t. A@a' <= M, sum(a')=1, a'>=0. Returns (b, M)."""
    n = len(a)
    A = build_A_matrix(a)
    size = 2 * n - 1
    
    # CVX form: min M
    # s.t. A @ b - M <= 0  (size constraints)
    #      sum(b) = 1
    #      b >= 0
    #      M free (but will be >= 0 automatically)
    
    # Using scipy linprog: minimize c @ x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    # x = [b_0, ..., b_{n-1}, M]
    # c = [0, ..., 0, 1]
    # A_ub = [A, -ones(size)]  (each row: A[k,:] @ b - M <= 0)
    # b_ub = zeros(size)
    # A_eq = [ones(n), 0]  (sum(b) = 1)
    # b_eq = [1]
    # bounds: b_i in [0, inf) for i=0..n-1, M in (-inf, inf)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    b = result.x[:n]
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return None, None
    b = b / s
    M = result.x[n]
    return b, float(M)

def solve_best_response_cvxpy(a, solver=None):
    """Solve best response LP using cvxpy."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    A = build_A_matrix(a)
    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob.solve()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def make_init_fourier(n, rng):
    """Initialize with a sequence whose DFT has controlled magnitude."""
    # We want a real non-negative sequence with "flat" autocorrelation
    # Start with a random phase in frequency domain, then take real part
    freq = np.rfft(np.ones(n))
    # Apply a window to control the spectral shape
    x = np.linspace(0, 1, n)
    # Raised cosine window in frequency → smoother time domain
    w = 0.5 * (1 + np.cos(2 * np.pi * np.linspace(0, 1, len(freq))))
    target_mag = w ** 0.5  # Half-power window
    phase = 2 * np.pi * rng.random(len(freq))
    spectrum = target_mag * np.exp(1j * phase)
    a = np.irfft(spectrum, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def make_init_smooth(n, rng):
    """Initialize with a smooth non-negative sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 1:
        # Sum of Gaussians
        centers = rng.uniform(0.2, 0.8, size=3)
        widths = rng.uniform(0.05, 0.2, size=3)
        a = sum(np.exp(-0.5 * ((x - c) / w) ** 2) for c, w in zip(centers, widths))
    elif choice == 2:
        # Quadratic
        a = 1.0 - (2 * x - 1) ** 2
    elif choice == 3:
        # Sine-modulated
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x + rng.random() * 2 * np.pi)
    elif choice == 4:
        # Two bumps
        a = np.exp(-4 * (x - 0.3) ** 2) + np.exp(-4 * (x - 0.7) ** 2)
    else:
        # Plateau
        a = np.where(np.abs(x - 0.5) < 0.3, 1.0, 0.3)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def make_init_random(n, rng, smoothness=1.0):
    """Random initialization with controlled smoothness."""
    a = rng.random(n)
    # Smooth with a Gaussian kernel
    if smoothness > 0:
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, smoothness)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def lp_fictitious_play(a0, time_limit, solver_func, n_max_iters=100000, verbose=True):
    """
    LP-based fictitious play / best-response iteration.
    Maintains both the current iterate and a running average.
    Uses adaptive mixing and periodic restarts.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    running_avg = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    stall_count = 0
    prev_best = best_score
    it = 0
    
    while it < n_max_iters:
        # Time check
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve best response
        b_new, M_val = solver_func(a)
        if b_new is None:
            it += 1
            continue
        
        # Adaptive mixing: start aggressive, become more conservative
        # Phase 1 (first 20%): alpha = 1.0 (full update)
        # Phase 2 (next 30%): alpha = 0.5
        # Phase 3 (next 30%): alpha = 0.2
        # Phase 4 (last 20%): alpha = 0.05
        frac = it / n_max_iters
        if frac < 0.2:
            alpha = 1.0
        elif frac < 0.5:
            alpha = 0.5
        elif frac < 0.8:
            alpha = 0.2
        else:
            alpha = 0.05
        
        # Update
        a = (1 - alpha) * a + alpha * b_new
        a = project_simplex(a)
        
        # Running average (fictitious play)
        running_avg = ((it) * running_avg + b_new) / (it + 1)
        running_avg = project_simplex(running_avg)
        
        it += 1
        
        # Evaluate periodically
        if it % 20 == 0:
            sc = score_normalized(a)
            sc_avg = score_normalized(running_avg)
            
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = running_avg.copy()
                stall_count = 0
            
            # Stall detection and perturbation
            if it % 200 == 0:
                if best_score < prev_best - 1e-8:
                    prev_best = best_score
                    stall_count = 0
                else:
                    stall_count += 1
                
                if stall_count >= 3:
                    # Restart from running average with perturbation
                    rng_local = np.random.default_rng(it)
                    perturbation = 0.05 * rng_local.standard_normal(n)
                    a = running_avg + perturbation
                    a = project_simplex(a)
                    running_avg = a.copy()
                    stall_count = 0
                    if verbose:
                        print(f"      FP iter={it}: perturbed, best={best_score:.6f}")
        
    return best_a, best_score

def coordinate_polish(a0, time_limit, n_steps=10000):
    """Coordinate-wise polish: reduce the largest contributors to the peak."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        improved = False
        top = np.argsort(contribs)[::-1][:5]
        for ci in top:
            i = idx[ci]
            for factor in [0.98, 0.95, 0.92, 0.88]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Also try the peak-1 and peak+1 positions
        for pk_offset in [-1, 1]:
            pk = peak_k + pk_offset
            if pk < 0 or pk >= len(b):
                continue
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            idx2 = np.arange(js2, je2 + 1)
            contribs2 = a[idx2] * a[pk - idx2]
            top2 = np.argsort(contribs2)[::-1][:3]
            for ci in top2:
                i = idx2[ci]
                if a[i] < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[i] *= 0.95
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try small random perturbation
            rng_local = np.random.default_rng(step)
            a_trial = a + 0.01 * rng_local.standard_normal(n)
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv)."""
    b = fast_convolve(a)

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_A_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_best_response_scipy(a):
    """Solve LP: min M s.t. A@b <= M*1, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a)
    A = build_A_matrix(a)
    size = 2 * n - 1

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.hstack([A, -np.ones((size, 1))])
    b_ub = np.zeros(size)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    if result.status != 0:
        return None, None, None

    b = result.x[:n]
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return None, None, None
    b = b / s
    M = result.x[n]

    # Extract dual variables (y_k for the inequality constraints)
    # These give the "active" lags
    duals = result.ineqlin.marginals if hasattr(result.ineqlin, 'marginals') else None
    return b, float(M), duals

def make_init_smooth(n, rng):

def make_init_fourier(n, rng):
    """Initialize using controlled frequency content."""
    nfreq = n // 2 + 1
    # Target: flat-ish magnitude with some structure
    mag = np.ones(nfreq)
    # Apply a smooth window
    k = np.arange(nfreq) / nfreq
    mag *= (0.5 + 0.5 * np.cos(np.pi * k)) ** 0.3
    phase = 2 * np.pi * rng.random(nfreq)
    spectrum = mag * np.exp(1j * phase)
    a = np.irfft(spectrum * n, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

    it = 0
    fail_count = 0

        result = solver_func(a)

            if fail_count > 20:
                break
            it += 1
            continue

        fail_count = 0

        if frac < 0.15:
            alpha = 1.0
        elif frac < 0.40:
            alpha = 0.7
        elif frac < 0.70:
            alpha = 0.3
        elif frac < 0.90:
            alpha = 0.1
        else:
            alpha = 0.03

        # Update
        a = (1 - alpha) * a + alpha * b_new
        a = project_simplex(a)

        # Running average
        running_avg = (it * running_avg + b_new) / (it + 1)
        running_avg = project_simplex(running_avg)

        it += 1

        # Evaluate periodically

            sc = score_normalized(a)
            sc_avg = score_normalized(running_avg)

            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = running_avg.copy()
                stall_count = 0

        # Stall detection and perturbation

            if best_score < prev_best - 1e-9:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1

                # Dual-guided perturbation
                rng_local = np.random.default_rng(it)

                    # Perturb in direction related to active constraints

                    if n_active > 0:
                        # Subgradient: grad_i = 2 * sum_{k: active} duals_k * a[k-i]

                            je = min(n - 1, pk)
                            idx = np.arange(js, je + 1)
                            a[idx] *= (1 - 0.02 * rng_local.random(len(idx)))
                        a = project_simplex(a)
                    else:
                        a = running_avg + 0.03 * rng_local.standard_normal(n)
                        a = project_simplex(a)
                else:
                    a = running_avg + 0.03 * rng_local.standard_normal(n)
                    a = project_simplex(a)

                running_avg = a.copy()
                stall_count = 0
                if verbose and it % 500 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      FP iter={it} ({elapsed:.0f}s): perturbed, best={best_score:.6f}")

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0

def subgradient_polish(a0, time_limit, n_steps=20000):

    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    x_avg = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Subgradient: grad_i = 2 * a[peak_k - i] (for valid indices)
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
            if j == i and j != peak_k - i:
                pass  # already counted

        # Actually, the subgradient of max_k conv_k at the point where k=peak_k is active:
        # d(conv_peak_k)/da_i = a[peak_k - i] + a[i] (if peak_k - i != i)
        # or 2*a[i] if peak_k - i == i
        # More carefully: conv_k = sum_j a_j a_{k-j}
        # d/d a_i = a_{k-i} + a_i (if i != k-i) or 2*a_i (if i = k-i)
        # But if k-i is out of bounds [0,n-1], then only one term survives.
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                grad[i] += a[j]
            # The other term: when k-j = i, i.e., j = k-i (same as above)
            # Wait, I'm double counting. Let me redo.
            # conv_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
            # d/d a_i: 
            #   From term j=i: a_{k-i} (if 0<=k-i<n)
            #   From term j=k-i (if k-i != i): a_{k-i} (if 0<=j<n)
            #   If k-i == i: the term is a_i^2, derivative is 2*a_i
            pass

        # Cleaner computation:
        grad = np.zeros(n)
        for i in range(n):
            # Contribution from j=i in conv_peak: a_{peak-i} if 0 <= peak-i < n
            j1 = peak_k - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            # Contribution from j such that k-j = i, i.e., j = k-i = peak-i (same as j1)
            # So if j1 == i (peak = 2i), we get 2*a_i; otherwise just a[j1]
            # Wait: the sum is over j. The term with a_i appears when:
            #   j = i: gives a_i * a_{peak-i}, derivative w.r.t. a_i is a_{peak-i}
            #   peak - j = i, i.e., j = peak - i: gives a_{peak-i} * a_i, derivative is a_{peak-i}
            #   If i = peak - i (peak = 2i): both are the same term a_i * a_i, derivative 2*a_i
            # So: grad[i] = a_{peak-i} (if peak-i in range) + a_i (if peak-i in range and peak-i != i)
            #   = a_{peak-i} + a_i if peak-i in range and peak-i != i
            #   = 2*a_i if peak = 2*i
            #   = a_{peak-i} if peak-i in range and peak-i == i... wait that's the same as 2*a_i
            pass

        # Let me just compute it directly:
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                if j == i:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = a[j] + a[i]  # a[j] from j=i term + a[i] from j=k-i term
                    # Wait no. From j=i: derivative of a_i * a_j w.r.t. a_i is a_j = a[j]
                    # From j=k-i: derivative of a_j * a_i w.r.t. a_i is a_j = a[j]... 
                    # No! From j=k-i: the term is a_{k-i} * a_i = a[j] * a_i
                    # derivative w.r.t. a_i is a[j]
                    # So total: a[j] + a[j] = 2*a[j] if j != i
                    # And if j == i: term is a_i * a_i = a_i^2, derivative 2*a_i
                    grad[i] = 2.0 * a[j] if j != i else 2.0 * a[i]

        # Hmm wait, I think I'm overcomplicating this. Let me just use the FFT approach.
        # The subgradient at the peak is: 2 * (delta_peak cross-correlate a)
        # where delta_peak is 1 at position peak_k and 0 elsewhere.
        # (delta_peak cross-correlate a)[i] = a[peak_k - i] (if valid)
        # So grad[i] = 2 * a[peak_k - i] (if 0 <= peak_k - i < n)
        # But this ignores the "symmetry" of the convolution. Let me verify.

        # conv_k = sum_j a_j a_{k-j}
        # d(conv_k)/da_i: 
        #   The terms involving a_i are: j=i (gives a_i * a_{k-i}) and j=k-i (gives a_{k-i} * a_i)
        #   If i != k-i: d/d a_i = a_{k-i} + a_{k-i} = 2*a_{k-i}... 
        #   Wait no! d/d a_i of (a_i * a_{k-i}) = a_{k-i} (treating a_{k-i} as constant)
        #   d/d a_i of (a_{k-i} * a_i) = a_{k-i} (treating a_{k-i} as constant)
        #   Total: 2 * a_{k-i} (if i != k-i and both j=i and j=k-i are valid)
        #   If i == k-i: only one term a_i^2, derivative 2*a_i
        #   If i != k-i but k-i is out of range: only j=i term, derivative a_{k-i}... but k-i out of range means this term doesn't exist!
        
        # OK so the correct formula is:
        grad = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                if j == i:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = 2.0 * a[j]
            # else: no contribution (both j=i and j=k-i terms require k-i to be in range)
            # Wait, if j = peak-i is out of range, then:
            #   j=i term: requires k-i = j to be in range for a_j to exist. If j out of range, term doesn't exist.
            #   j=k-i term: requires k-i to be in range (which is j). Same condition.
            # So if j is out of range, grad[i] = 0. Correct.

        # Learning rate
        lr = 0.1 / (1.0 + 0.0005 * step)

        a_new = project_simplex(a - lr * grad)

        # Averaging
        x_avg = ((step + 1) * x_avg + a_new) / (step + 2)

        a = a_new

        # Track best
        if step % 50 == 0:
            sc = score_normalized(a)
            sc_avg = score_normalized(x_avg)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = x_avg.copy()

    return best_a, best_score


def gradient_polish_smoothed(a0, time_limit, n_steps=30000):
    """Gradient polish using log-sum-exp smoothed max with Nesterov."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0).astype(np.float64))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()

    T_start = 3.0
    T_end = 400.0
    lr_init = 0.06

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        frac = step / n_steps
        T = T_start * (T_end / T_start) ** frac
        lr = lr_init / (1.0 + 0.002 * step)

        # Nesterov
        beta = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / (2.0 * step + 2.0) if step > 0 else 0.5
        y = x + beta * (x - x_prev)
        y = project_simplex(np.maximum(y, 0))

        # Smoothed max gradient
        b = fast_convolve(y)
        Tb = T * b
        m = Tb.max()
        e = np.exp(Tb - m)
        w = e / e.sum()

        # Gradient: 2 * cross_correlation(w, y)
        # cross_corr[i] = sum_k w[k] * y[k-i]
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        Y = rfft(y, fft_size)
        corr = irfft(np.conj(W) * Y, fft_size)
        grad = 2.0 * corr[:n]

        x_prev = x.copy()
        x = project_simplex(np.maximum(y - lr * grad, 0))

        if step % 200 == 0:
            sc = score_normalized(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()
                if step % 2000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      GradSm step={step} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, n_steps=20000):
    """Coordinate-wise and pair-wise polish focusing on peak reduction."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        improved = False
        # Try reducing top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top = np.argsort(contribs)[::-1][:8]
        for ci in top:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.94, 0.90]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pair reduction
        if not improved:
            for ci in top[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.93]:
                    for f2 in [0.97, 0.93]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Also check peaks near the main peak
        if not improved:
            for offset in [-1, 1, -2, 2]:
                pk = peak_k + offset
                if pk < 0 or pk >= len(b):
                    continue
                if b[pk] < 0.95 * b[peak_k]:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                idx2 = np.arange(js2, je2 + 1)
                contribs2 = a[idx2] * a[pk - idx2]
                top2 = np.argsort(contribs2)[::-1][:3]
                for ci in top2:
                    i = int(idx2[ci])
                    if a[i] < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[i] *= 0.96
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Small random perturbation to escape
            rng_local = np.random.default_rng(step)
            a_trial = a + 0.005 * rng_local.standard_normal(n)
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            elif step % 500 == 0 and step > 0:
                break

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP fictitious play, gradient methods,
    and coordinate descent. Uses diverse initializations and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_init_smooth(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Initialize LP solver
    _init_lp_solver()
    print(f"LP solver: {_lp_solver_name}")

    # Choose solver function
    if _lp_solver_name is not None and _lp_solver_name != 'none':
        def solve_br(a):
            return solve_best_response_cvxpy(a)
    else:
        def solve_br(a):
            return solve_best_response_scipy(a)

    # Test LP solve
    t_test = time.monotonic()
    test_result = solve_br(best_seq)
    t_lp = time.monotonic() - t_test
    if test_result[0] is not None:
        print(f"LP solve time: {t_lp * 1000:.1f}ms, M={test_result[1]:.6f}")
    else:
        print(f"LP solve FAILED, falling back to gradient only")
        # Use scipy
        def solve_br(a):
            return solve_best_response_scipy(a)
        t_test = time.monotonic()
        test_result = solve_br(best_seq)
        t_lp = time.monotonic() - t_test
        if test_result[0] is not None:
            print(f"SciPy LP solve time: {t_lp * 1000:.1f}ms")

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fp, _ = lp_fictitious_play(best_seq, t_alloc, solve_br, n_max_iters=80000)
        try_update(a_fp, "fp_main")

    # === Phase 2: Gradient polish (smoothed) ===
    print(f"\n=== Phase 2: Gradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_gp, _ = gradient_polish_smoothed(best_seq, t_alloc, n_steps=20000)
        try_update(a_gp, "grad_sm")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 40:
        a_fp2, _ = lp_fictitious_play(best_seq, t_alloc, solve_br, n_max_iters=60000)
        try_update(a_fp2, "fp_2")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_sg, _ = subgradient_polish(best_seq, t_alloc, n_steps=15000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Multi-start exploration ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        inits = [
            ("fourier", make_init_fourier(n_best, rng)),
            ("smooth_a", make_init_smooth(n_best, rng)),
            ("smooth_b", make_init_smooth(n_best, rng)),
            ("rand_sm", make_init_random_smooth(n_best, rng, sigma=10.0)),
        ]
        for name, init_a in inits:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_sub = min(rem * 0.18, 250.0)
            if t_sub < 40:
                break
            print(f"  Start: {name}, budget={t_sub:.0f}s")
            a_sub, _ = lp_fictitious_play(init_a, t_sub * 0.75, solve_br, n_max_iters=40000)
            try_update(a_sub, f"ms_{name}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_sub2, _ = gradient_polish_smoothed(a_sub, min(rem * 0.25, 30.0), n_steps=5000)
                try_update(a_sub2, f"ms_{name}_grad")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for delta in [-128, -64, -32, 32, 64, 128]:
        alt_n = n_best + delta
        if 512 <= alt_n <= 2048:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t_total = min(rem * 0.15, 200.0)
        if t_total < 50:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = lp_fictitious_play(init_alt, t_total * 0.7, solve_br, n_max_iters=40000)
        try_update(a_alt, f"alt_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = gradient_polish_smoothed(a_alt, min(rem * 0.3, 25.0), n_steps=5000)
            try_update(a_alt2, f"alt_n{alt_n}_grad")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.30, 120.0)
        a_fin, _ = lp_fictitious_play(best_seq, t_fin, solve_br, n_max_iters=50000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Coordinate polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.20, 60.0)
        a_ps, _ = coordinate_polish(best_seq, t_ps, n_steps=15000)
        try_update(a_ps, "coord")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
