import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_soft_max_gradient(a, conv, T, top_k=50):
    """
    Compute approximate gradient of the soft-max of conv values.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Get top k peaks
    if top_k >= len(conv):
        top_k = len(conv)
    
    # Use log-sum-exp for numerical stability
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / T)
    Z = exp_c.sum()
    w = exp_c / Z  # Normalized weights
    
    # Only keep top-k weights for efficiency
    if top_k < len(w):
        # Zero out small weights
        threshold = np.partition(w, -top_k)[top_k - 1]
        w[w < threshold] = 0
    
    # Compute gradient: g[i] = sum_k w[k] * 2*a[k-i] (for valid k-i)
    g = np.zeros(n)
    active_k = np.nonzero(w > 1e-12)[0]
    
    for k in active_k:
        wk = w[k] * 2.0
        if wk < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # g[i] += wk * a[k-i] for i in [i_start, i_end]
        g[i_start:i_end+1] += wk * a[k-i_end:k-i_start+1][::-1]
    
    return g

def mirror_descent_step(a, conv, T, lr, top_k=50):
    """One mirror descent step with entropic regularization."""
    n = len(a)
    g = compute_soft_max_gradient(a, conv, T, top_k)
    
    # Multiplicative update
    a_new = a * np.exp(-lr * g)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def tempered_mirror_descent(a0, time_limit, T_start=0.01, T_end=0.001, 
                            lr_start=0.05, lr_end=0.001, top_k=30, verbose=False):
    """
    Tempered mirror descent: anneal temperature from T_start to T_end.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    iters = 0
    no_improve = 0
    
    total_iters = 2000  # Will adjust based on time
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        progress = min(1.0, iters / 1500.0)
        
        # Anneal temperature
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        conv = fast_convolve(a)
        cur_score = 2.0 * n * float(conv.max())
        
        # Mirror descent step
        a_new = mirror_descent_step(a, conv, T, lr, top_k)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TMD it={iters} T={T:.5f} lr={lr:.5f} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if no_improve > 200 and progress > 0.5:
            break
    
    return best_a, best_score, iters

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:

        # Projected gradient step (simple: just subtract and clip)

        a_new = a * np.exp(-lr * g * g_norm)

            lr *= 0.95

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    # Force DC component
    amps[0] = 1.0
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_random_phase_nongtrivial(n, rng):
    """Random phase with zero mean constraint for better peak reduction."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    # Sinusoidal amplitude: more energy at high freq
    amps = np.sqrt(np.linspace(0.1, 1.0, n_freq))
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = a[:n]
    a = a - a.min()  # Shift to be non-negative
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss

    for k in active_k:
        wk = 2.0 * w[k]
        if wk < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        g[i_start:i_end + 1] += wk * a[k - i_end:k - i_start + 1][::-1]
    return g


def compute_l2_gradient_fft(a, conv):
    """Gradient of sum(conv^2) wrt a, computed via FFT."""
    n = len(a)
    a_flipped = a[::-1]
    result_length = (2 * n - 1) + n - 1
    fft_size = 1 << (result_length - 1).bit_length()
    C = rfft(conv, fft_size)
    Af = rfft(a_flipped, fft_size)
    conv_result = irfft(C * Af, fft_size)
    return 4.0 * conv_result[n - 1:2 * n - 1]


def l2_warmup(a0, time_limit, verbose=False, label=""):
    """Minimize sum(conv^2) using gradient descent with backtracking line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    l2_obj = float(np.sum(conv ** 2))
    best_a = a.copy()
    best_l2 = l2_obj
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    lr = 0.005
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        conv = fast_convolve(a)
        g = compute_l2_gradient_fft(a, conv)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / ss

        conv_new = fast_convolve(a_new)
        l2_new = float(np.sum(conv_new ** 2))
        score_new = 2.0 * n * float(conv_new.max())

        if l2_new < best_l2 - 1e-15:
            best_l2 = l2_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 0.02)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 100:
                break

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    L2WU {label} it={iters} ({elapsed:.0f}s): l2={best_l2:.8f} score={best_score:.8f}")

    return best_a, best_score, iters


def tempered_mirror_descent(a0, time_limit, T_start=0.01, T_end=0.001,
                            lr_start=0.05, lr_end=0.001, top_k=30, verbose=False, label=""):
    """Tempered mirror descent: anneal temperature from T_start to T_end."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    total_planned = 2000

    while time.monotonic() - t0 < time_limit:
        iters += 1
        progress = min(1.0, iters / total_planned)

        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        conv = fast_convolve(a)
        g = compute_soft_max_gradient(a, conv, T, top_k)

        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            lr *= 0.5
            continue
        a_new = a_new / ss

        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TMD {label} it={iters} T={T:.5f} ({elapsed:.0f}s): best={best_score:.8f}")

        if no_improve > 300 and progress > 0.5:
            break

    return best_a, best_score, iters


def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = np.zeros(n_aff)
                changes[:] = 2.0 * delta * a_partners
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine_bell(n, rng):
    """Cosine bell with random width."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.5)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def gradient_softmax_descent(a0, time_limit, verbose=False, label=""):
    """Gradient descent using soft-max gradient with backtracking line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()

    T = 0.005
    top_k = 25
    lr = 0.01
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        conv = fast_convolve(a)
        g = compute_soft_max_gradient(a, conv, T, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Multiplicative update for non-negativity
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / ss

        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly decrease temperature
            T = max(0.0005, T * 0.9995)
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 80:
                T = min(0.03, T * 1.3)
                lr = 0.01
                no_improve = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GSD {label} it={iters} T={T:.5f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters


def propose_candidate():
    """
    Multi-phase optimization combining:
    - L2 warm-up (smooth objective)
    - Soft-max mirror descent (annealed temperature)
    - Gradient soft-max descent with adaptive step
    - Exact coordinate descent (local refinement)
    - Basin hopping and multi-length exploration
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(777)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = init_flat_spectrum(1024, rng)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Generated initial: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: L2 warm-up ===
    print(f"\n=== Phase 1: L2 Warm-up (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 30:
        t_alloc = min(rem * 0.2, 120.0)
        print(f"  L2WU budget={t_alloc:.0f}s")
        a_l2, s_l2, iters = l2_warmup(best_seq, t_alloc, verbose=True, label="warm")
        best_seq, best_score = check_and_update(a_l2, best_seq, best_score, "l2_warmup")
        conv_check = fast_convolve(best_seq)
        print(f"  After L2WU: max_conv={float(conv_check.max()):.8f}, score={best_score:.8f}")

    # === Phase 2: Tempered Mirror Descent (multiple configs) ===
    print(f"\n=== Phase 2: Tempered Mirror Descent (best={best_score:.8f}) ===")

    tmd_configs = [
        (0.015, 0.003, 0.04, 0.005, 40, "TMD_warm"),
        (0.008, 0.002, 0.03, 0.003, 25, "TMD_mid"),
        (0.004, 0.0008, 0.02, 0.002, 15, "TMD_cold"),
    ]

    for T_s, T_e, lr_s, lr_e, tk, label in tmd_configs:
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        t_alloc = min(rem * 0.2, 100.0)
        print(f"  {label}: T=[{T_s},{T_e}] lr=[{lr_s},{lr_e}] top_k={tk} budget={t_alloc:.0f}s")
        a_tmd, s_tmd, iters = tempered_mirror_descent(
            best_seq, t_alloc, T_start=T_s, T_end=T_e,
            lr_start=lr_s, lr_end=lr_e, top_k=tk, verbose=True, label=label
        )
        best_seq, best_score = check_and_update(a_tmd, best_seq, best_score, label)

    # === Phase 3: Gradient soft-max descent ===
    print(f"\n=== Phase 3: Gradient SoftMax Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem * 0.25, 120.0)
        print(f"  GSD budget={t_alloc:.0f}s")
        a_gsd, s_gsd, iters = gradient_softmax_descent(
            best_seq, t_alloc, verbose=True, label="gsd_main"
        )
        best_seq, best_score = check_and_update(a_gsd, best_seq, best_score, "gsd_main")

    # === Phase 4: Exact coordinate descent (multiple rounds) ===
    print(f"\n=== Phase 4: Exact CD (best={best_score:.8f}) ===")
    for rnd in range(4):
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(rem * 0.2, 120.0)
        print(f"  ECD round {rnd + 1}, budget={t_alloc:.0f}s")
        a_ecd, s_ecd, sweeps = exact_coord_descent(
            best_seq, t_alloc, verbose=True, label=f"ecd_r{rnd}"
        )
        best_seq, best_score = check_and_update(a_ecd, best_seq, best_score, f"ecd_r{rnd}")

        # Alternate with TMD to escape local minima
        if rnd < 3:
            rem = deadline - time.monotonic() - 350
            if rem > 20:
                t_tmd = min(rem * 0.15, 50.0)
                a_t, s_t, _ = tempered_mirror_descent(
                    best_seq, t_tmd, T_start=0.006, T_end=0.0015,
                    lr_start=0.025, lr_end=0.004, top_k=20,
                    verbose=False, label=f"tmd_alt{rnd}"
                )
                best_seq, best_score = check_and_update(a_t, best_seq, best_score, f"tmd_alt{rnd}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    alt_lengths = [n_best]
    for alt in [768, 1024, 1280, 1536, 2048]:
        if alt != n_best and 256 <= alt <= 2560:
            alt_lengths.append(alt)
    alt_lengths = list(set(alt_lengths))[:4]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 20:
            break
        t_total = min(rem * 0.2, 80.0)
        print(f"  n={n_alt}, budget={t_total:.0f}s")

        a_interp = interpolate_seq(best_seq, n_alt)

        # TMD on new length
        t_t = t_total * 0.4
        a_t, s_t, _ = tempered_mirror_descent(
            a_interp, t_t, T_start=0.01, T_end=0.003,
            lr_start=0.03, lr_end=0.004, top_k=25, verbose=False, label=f"ml_tmd_{n_alt}"
        )
        # L2 warmup on new length
        t_l = t_total * 0.2
        a_l, s_l, _ = l2_warmup(a_t, t_l, verbose=False, label=f"ml_l2_{n_alt}")
        # CD polish
        t_c = t_total * 0.4
        a_c, s_c, _ = exact_coord_descent(a_l, t_c, verbose=False, label=f"ml_cd_{n_alt}")
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"ml_n{n_alt}")

    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin Hopping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 40:
        n_best = len(best_seq)
        t_bh_total = min(rem * 0.6, 150.0)
        t0_bh = time.monotonic()

        for p in range(8):
            if time.monotonic() - t0_bh > t_bh_total:
                break
            rem_p = t_bh_total - (time.monotonic() - t0_bh)
            if rem_p < 15:
                break
            t_sub = min(rem_p * 0.3, 40.0)

            # Perturb
            sigma = 0.015 + 0.02 * rng.random()
            perturbed = best_seq + rng.standard_normal(n_best) * sigma
            perturbed = np.maximum(perturbed, 0.0)
            ps = perturbed.sum()
            if ps < 1e-10:
                continue
            perturbed = perturbed / ps

            # Optimize with TMD
            t_t = t_sub * 0.5
            a_t, s_t, _ = tempered_mirror_descent(
                perturbed, t_t, T_start=0.005, T_end=0.001,
                lr_start=0.02, lr_end=0.003, top_k=20, verbose=False, label=f"bh_tmd{p}"
            )
            # Then CD
            t_c = t_sub * 0.5
            a_c, s_c, _ = exact_coord_descent(a_t, t_c, verbose=False, label=f"bh_cd{p}")
            best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"bh_{p}")

            # Also try L2 on the perturbed
            rem_p2 = t_bh_total - (time.monotonic() - t0_bh)
            if rem_p2 > 10:
                a_l, s_l, _ = l2_warmup(perturbed, min(rem_p2 * 0.3, 20.0),
                                         verbose=False, label=f"bh_l2{p}")
                best_seq, best_score = check_and_update(a_l, best_seq, best_score, f"bh_l2_{p}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")

    # TMD with very cold temperature
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_f = min(rem * 0.3, 50.0)
        a_f, s_f, _ = tempered_mirror_descent(
            best_seq, t_f, T_start=0.002, T_end=0.0003,
            lr_start=0.01, lr_end=0.0005, top_k=10, verbose=False, label="tmd_cold_final"
        )
        best_seq, best_score = check_and_update(a_f, best_seq, best_score, "tmd_cold_final")

    # ECD
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_f = min(rem * 0.4, 40.0)
        a_f, s_f, _ = exact_coord_descent(best_seq, t_f, verbose=True, label="ecd_final")
        best_seq, best_score = check_and_update(a_f, best_seq, best_score, "ecd_final")

    # Another TMD + CD cycle
    rem = deadline - time.monotonic() - 10
    if rem > 8:
        t_t = min(rem * 0.3, 15.0)
        a_t, s_t, _ = tempered_mirror_descent(
            best_seq, t_t, T_start=0.001, T_end=0.0002,
            lr_start=0.005, lr_end=0.0003, top_k=8, verbose=False, label="tmd_last"
        )
        best_seq, best_score = check_and_update(a_t, best_seq, best_score, "tmd_last")
        rem2 = deadline - time.monotonic() - 3
        if rem2 > 3:
            a_c, s_c, _ = exact_coord_descent(best_seq, min(rem2 * 0.9, 10.0),
                                              verbose=False, label="ecd_last")
            best_seq, best_score = check_and_update(a_c, best_seq, best_score, "ecd_last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
