def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    """
    Subgradient method: at each step, find the peak of conv(a,a),
    compute the subgradient, and take a projected step.
    """

    best_a = a.copy()

            best_a = a_new.copy()
        
        a = a_new

def mirror_descent_refine(a0, time_limit, n_iters=50000):
    """
    Multiplicative weights (mirror descent on simplex) for peak-directed optimization.
    At each step, find the peak, compute subgradient, update via exp(-eta * g).
    """
    n = len(a0)
    a = norm_seq(a0)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Additional peaks: also consider top few peaks for more robust updates
    n_top = 3
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        
        # Find top peaks
        # Use top-3 peaks weighted by their values
        peak_vals = np.sort(b)[::-1][:n_top]
        peak_fracs = peak_vals / peak_vals[0]
        
        # Weighted subgradient
        g = np.zeros(n)
        top_k = np.argsort(b)[::-1][:n_top]
        for fi, k in enumerate(top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += peak_fracs[fi] * 2.0 * a[k - j]
        
        # Step size: eta = c / (1 + it * decay)
        eta = 0.5 / (1.0 + it * 0.005)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        a_new = np.exp(log_a_new - log_a_new.max())
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
