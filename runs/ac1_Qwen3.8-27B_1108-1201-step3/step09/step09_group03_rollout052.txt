"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Soft-max mirror descent with momentum and adaptive temperature annealing
2. Subgradient-informed coordinate descent for local polishing
3. Spectral and structured diverse initializations
4. Multi-scale exploration (different n values)
5. EMA of iterates for improved convergence

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Fast score computation."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_mirror_descent(a0, time_limit, verbose=True, beta_start=100.0, beta_end=5000.0,
                           lr_start=0.05, momentum=0.3, label=""):
    """
    Soft-max mirror descent (multiplicative weight update) with momentum
    and adaptive temperature annealing.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_start

    # Momentum buffer
    v = np.zeros(n)

    # EMA of iterates
    ema_a = a.copy()
    ema_weight = 0.99

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)

        # Adaptive temperature: starts warm (flat), cools (peaked)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        # Compute soft-max weights over convolution
        M = conv.max()
        shifted = conv - M
        # Clip to avoid underflow
        shifted = np.maximum(shifted, -50.0 / beta)
        exp_vals = np.exp(beta * shifted)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / w_sum

        # Compute gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
        # Using np.correlate(w, a, mode='valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Update momentum
        v = momentum * v + (1.0 - momentum) * grad

        # Adaptive learning rate
        # Scale lr by the norm of the gradient for stability
        grad_norm = np.linalg.norm(v)
        if grad_norm > 1e-15:
            effective_lr = lr * (1.0 / (1.0 + grad_norm))
        else:
            effective_lr = lr

        # Mirror descent step: a_new = a * exp(-lr * v)
        log_update = -effective_lr * v
        # Clip to avoid overflow/underflow
        log_update = np.clip(log_update, -50.0, 50.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Update convolution
        conv = fast_convolve(a_new)
        cur_score = 2.0 * n * conv.max()

        # Track best

            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update EMA
        ema_a = ema_weight * ema_a + (1.0 - ema_weight) * a_new
        ema_a = ema_a / ema_a.sum()

        a = a_new

        # Learning rate decay
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
            v *= 0.3
            if lr < 1e-10:
                break

        # Check EMA periodically
        if it % 30 == 0:
            ema_score = 2.0 * n * fast_convolve(ema_a).max()
            if ema_score < best_score:
                best_score = ema_score
                best_a = ema_a.copy()
                no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    SMD{label} it={it} beta={beta:.0f} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

def subgradient_coord_descent(a0, time_limit, verbose=True, label=""):

    Coordinate descent that uses the current subgradient to prioritize
    which coordinates to update. Updates the convolution incrementally.
    """

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute prefix and suffix maxima for fast unaffected-max queries
    prefix_max = np.zeros(2 * n - 1)
    prefix_max[0] = conv[0]
    for i in range(1, 2 * n - 1):
        prefix_max[i] = max(prefix_max[i - 1], conv[i])
    suffix_max = np.zeros(2 * n - 1)
    suffix_max[2 * n - 2] = conv[2 * n - 2]
    for i in range(2 * n - 3, -1, -1):
        suffix_max[i] = max(suffix_max[i], conv[i + 1])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Identify active peaks for subgradient computation
        top_k = min(10, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        pw = np.exp((pv - pv.max()) / temp)
        pw = pw / pw.sum()

        # Compute subgradient to prioritize coordinates
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if pw[idx] < 1e-6:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += pw[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Process coordinates in order of subgradient magnitude (high first)
        order = np.argsort(-sg)

        for i in order:
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = prefix_max[k_start - 1]
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = suffix_max[k_end + 1]
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Only try to update if this coordinate affects the max
            if conv[k_start:k_end + 1].max() < M - 1e-10 and unaff_max >= M - 1e-10:
                # This coordinate doesn't affect the max, only try reducing it slightly
                # Actually skip for efficiency
                continue

            a_shifted = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            # Adaptive step factors based on subgradient
            step_scale = a[i] * max(0.1, sg[i] / max(M, 1e-10))
            factors = [-step_scale, -0.5 * step_scale, -0.1 * step_scale,
                       0.1 * step_scale, 0.5 * step_scale, step_scale,
                       -0.5 * a[i], 0.5 * a[i], -0.1 * a[i], 0.1 * a[i]]

            for delta in factors:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx2 = (2 * i) - k_start
                    new_affected[idx2] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-13:
                # Apply update
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                changes = 2.0 * best_delta * a_shifted.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx2 = (2 * i) - k_start

                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                # Update prefix/suffix maxima (approximate: recompute if needed)

                    prefix_max = np.maximum.accumulate(conv)
                if k_end == 2 * n - 2:
                    suffix_max = np.maximum.accumulate(conv[::-1])[::-1]

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:

    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1
        # Optimize b given a
        for step in range(n_steps):
            k_star = int(np.argmax(b_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            b_conv = (1.0 - t) * b_conv
            ks_s = j_star
            ks_e = min(j_star + n - 1, 2 * n - 2)
            b_conv[ks_s:ks_e + 1] += t * a
            b = (1.0 - t) * b
            b[j_star] += t

        b_score = 2.0 * n * fast_convolve(b).max()
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()

        # Optimize a given b
        a_conv = convolve_ab(b, a)
        for step in range(n_steps):
            k_star = int(np.argmax(a_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            a_conv = (1.0 - t) * a_conv
            ks_s = j_star
            ks_e = min(j_star + n - 1, 2 * n - 2)
            a_conv[ks_s:ks_e + 1] += t * b
            a = (1.0 - t) * a
            a[j_star] += t

        a_score = 2.0 * n * fast_convolve(a).max()
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

        # Periodic sync
        if round_num % 5 == 0:
            a = best_a.copy()
            b = best_a.copy()
            b_conv = convolve_ab(a, b)

        if verbose and round_num % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW{label} round={round_num} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_init(n, rng):
    """Generate a non-negative sequence with flat-ish power spectrum."""
    # Random phase in frequency domain
    n_fft = 1 << (2 * n - 2).bit_length()
    freq = rng.standard_normal(n_fft // 2 + 1) + 1j * rng.standard_normal(n_fft // 2 + 1)
    # Make it conjugate symmetric for real output
    if n_fft % 2 == 0:
        freq[n_fft // 2] = freq[n_fft // 2].real
        freq[1:n_fft // 2] = np.conj(freq[n_fft // 2:1:-1])
    else:
        freq[(n_fft // 2 + 1):] = np.conj(freq[1:n_fft // 2 + 1][::-1])

    # Modify to have flatter magnitude
    mag = np.abs(freq)
    mag = mag ** 0.5  # Compress dynamic range
    phase = np.angle(freq)
    freq_modified = mag * np.exp(1j * phase)

    a = irfft(freq_modified, n_fft)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_init(n, rng, power=None):
    """Cosine window initialization."""
    if power is None:
        power = rng.uniform(0.5, 3.0)
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def multi_bump_init(n, rng, n_bumps=3):
    """Multi-bump Gaussian initialization."""
    centers = np.sort(rng.uniform(0.1, 0.9, size=n_bumps)) * n
    widths = rng.uniform(0.05, 0.2, size=n_bumps) * n
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    for c, w in zip(centers, widths):
        a += np.exp(-0.5 * ((t - c) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def periodic_init(n, rng):
    """Periodic pattern initialization."""
    period = int(rng.integers(3, 12))
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * t / period + phase) + 0.2 * np.cos(4 * np.pi * t / period)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def check_and_update(cand, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 0.01:
        return best_seq, best_score
    cand_norm = cand / s
    try:
        score = float(evaluate_sequence(cand_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = cand_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(cand_norm)})")
    return best_seq, best_score
