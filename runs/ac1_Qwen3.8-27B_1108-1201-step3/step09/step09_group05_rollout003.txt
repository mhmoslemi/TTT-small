import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def objective_normalized(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Evaluate sequence using the external evaluate_sequence."""
    try:

        return float(evaluate_sequence(a_list))

_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b_opt, M_opt, dual_weights) where dual_weights are the LP dual variables."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    # Use dual variables to get constraint tightness
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s

    # Try to extract dual variables
    dual_weights = None
    try:
        # The dual for constraint T @ b <= M is a vector of length (2n-1)
        # In cvxpy, we can access via the constraint object
        for con in constraints:
            if con.value is not None:
                pass
        # Alternative: compute residual to estimate tightness
        conv_result = T @ b_val
        M_val = float(M_var.value)
        residual = M_val - conv_result
        # Dual weights are proportional to how tight each constraint is
        if residual.max() > 1e-10:
            dual_weights = np.maximum(0, 1.0 - residual / max(residual.max(), 1e-10))
            dual_weights = dual_weights / dual_weights.sum()
    except Exception:
        dual_weights = None

    return b_val, float(M_var.value), dual_weights


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov LP refinement with adaptive perturbation and dual-guided noise."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = objective_normalized(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    dual_history = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, dual_w = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        if dual_w is not None:
            dual_history.append(dual_w)
            if len(dual_history) > 50:
                dual_history.pop(0)

        alpha = 2.0 / (it + 2)

        # Nesterov momentum update
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Averaged iterate
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = objective_normalized(a)
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Adaptive perturbation based on dual information
        if it % 200 == 199:
            if best_score < prev_score - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score

            if stall_count >= 4:
                # Dual-guided perturbation: increase where dual is high (constraint is tight)
                if dual_history:
                    avg_dual = np.mean(dual_history[-10:], axis=0)
                    # Perturb in direction that reduces tight constraints
                    rng_local = np.random.default_rng(777 + it)
                    noise = rng_local.random(n) * 0.05
                    # Weight noise by dual importance
                    noise_weighted = noise * (1.0 + avg_dual)
                    y = y + 0.1 * (noise_weighted - noise_weighted.mean())
                else:
                    rng_local = np.random.default_rng(777 + it)
                    noise = rng_local.random(n) * 0.05
                    y = y + 0.1 * (noise - noise.mean())
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, maxiter=10000, n_stages=None):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if n_stages is None:
        n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave_direct(a0, time_limit, n_steps=30000):
    """Direct peak shaving: find top contributors to max and reduce them."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = objective_normalized(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-15:
            break

        top_count = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_count]]

        improved = False
        for i in top_positions:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80, 0.75]:
                a_trial = a.copy()
                a_trial[i] *= factor
                diff = a[i] * (1 - factor)
                # Redistribute to positions NOT contributing to peak
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                mask[i] = False
                candidates = np.where(mask & (a > 1e-14))[0]
                if len(candidates) == 0:
                    candidates = np.where(mask)[0]
                if len(candidates) == 0:
                    continue
                # Distribute proportionally to existing values among non-contributors
                weights = a_trial[candidates] + 1e-10
                a_trial[candidates] += diff * weights / weights.sum()
                a_trial = norm_sequence(a_trial)
                score = objective_normalized(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ci in range(min(5, len(top_positions))):
                i = top_positions[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = objective_normalized(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0

    return best_a, best_score


def subgradient_descent(a0, time_limit, n_steps=100000):
    """Projected subgradient descent with Nesterov acceleration for the max objective."""
    n = len(a0)
    a = norm_sequence(a0).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = objective_normalized(a)
    t0 = time.monotonic()

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        k_star = int(np.argmax(b))

        # Subgradient
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = y[k_star - i]

        alpha = 2.0 / (it + 2)

        # Nesterov update
        y = (1 - alpha) * y + alpha * a
        y = np.maximum(y, 0)
        y = y / y.sum()

        # Gradient step with projection
        step = 1.0 / (np.sqrt(it + 1) * np.sqrt(n))
        a_new = y - step * g
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s

        # Check improvement
        score = objective_normalized(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        if it % 500 == 0:
            score_curr = objective_normalized(a)
            if score_curr < best_score:
                best_score = score_curr
                best_a = a.copy()

        a = a_new

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2 * x - 1) ** 2
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.sin(2 * np.pi * 3 * x)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.2 * np.cos(2 * np.pi * x)
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.05 * np.sin(2 * np.pi * 7 * x) - 0.03 * np.sin(2 * np.pi * 13 * x)
    else:
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing the autocorrelation peak ratio.
    Uses Nesterov LP as the primary engine, L-BFGS for polishing,
    and direct peak shaving for final adjustments.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.6f}, budget={budget_s:.0f}s")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov LP - main global optimization (45% of time) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: L-BFGS multi-stage refinement (25% of time) ===
    print(f"\n=== Phase 2: L-BFGS Multi-Stage (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 30->0.05, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=30.0, T_end=0.05, maxiter=20000, n_stages=14)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: Direct peak shaving (10% of time) ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave_direct(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave_1")

    # === Phase 4: Subgradient with Nesterov (8% of time) ===
    print(f"\n=== Phase 4: Subgradient Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_sg, _ = subgradient_descent(best_seq, t_alloc, n_steps=200000)
        try_update(a_sg, "subgrad")

    # === Phase 5: Second LP round on refined solution (7% of time) ===
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 100.0)
    if t_alloc > 15:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 6: Very low temperature L-BFGS polish (5% of time) ===
    print(f"\n=== Phase 6: L-BFGS Polish T=5->0.02 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 15:
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_start=5.0, T_end=0.02, maxiter=15000, n_stages=10)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 7: Multi-length exploration (remaining time) ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for alt_n in [896, 1024, 1152, 1280, 2048]:
        if alt_n != n_best and 256 <= alt_n <= 4096:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 45:
            break
        t_total = min(rem * 0.25, 120.0)
        if t_total < 30:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick LP
        t_lp = t_total * 0.5
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=100000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.4, 50.0)
            if t_lb > 10:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=10.0, T_end=0.05,
                                      maxiter=8000, n_stages=8)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        for scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)

            t_sub = min(60.0, rem * 0.5)
            if t_sub < 15:
                break
            a_pert, _ = nesterov_lp_refine(perturbed, t_sub, solver=solver, n_max_iters=50000)
            try_update(a_pert, f"pert_{scale}")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_pert2 = lbgfs_refine(best_seq, min(rem * 0.4, 40.0),
                                       T_start=5.0, T_end=0.05, maxiter=5000, n_stages=6)
                try_update(a_pert2, f"pert_lbgfs_{scale}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.3, 50.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=3.0, T_end=0.02, maxiter=10000, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.3, 40.0)
        a_ps2, _ = peak_shave_direct(best_seq, t_ps, n_steps=30000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.3, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=100000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
