import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, T):
    """Compute log-sum-exp objective and its gradient w.r.t. z."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    
    # Gradient w.r.t. a
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_refine(a0, time_limit, T_schedule, maxiter_per_stage=5000):
    """L-BFGS with log-sum-exp smoothing and given temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            obj, _ = lse_objective_and_grad(z_in, T)
            return obj
        
        def jac(z_in):
            _, g = lse_objective_and_grad(z_in, T)
            return g
        
        iters = max(20, min(maxiter_per_stage, int(remaining / 0.05)))
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxfun': iters * 200, 'disp': False}
            )
            if res.x is not None:
                z = res.x
                a_trial = softmax_from_z(z)
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
        except Exception:
            pass
    
    return best_a, best_score


def subgradient_peak_reduce(a0, time_limit, n_steps=10000):

        # Subgradient of max at the peak
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)

            g[i] = 2.0 * a[k_star - i]
        
        # Project gradient onto simplex tangent

        # Simple projection: subtract mean

        # Adaptive step size
        step_size = 0.5 / (step + 10)
        
        a_trial = a - step_size * g_proj
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step

                a_trial2 = a - step_size * mult * g_proj
                a_trial2 = np.maximum(a_trial2, 0)

                if s2 < 1e-15:

                if score2 < best_score:

                    best_a = a_trial2.copy()
                    a = a_trial2
                    break

def directional_perturb(a0, rng, scale=0.05):
    """Perturb in a direction that's likely to reduce the peak."""
    n = len(a0)
    a = norm_sequence(a0)
    b = fast_convolve(a)
    k_star = int(np.argmax(b))
    
    # Compute the gradient at the peak
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for i in range(js, je + 1):
        g[i] = a[k_star - i]
    
    # The peak is reduced by moving mass from high-g positions to low-g positions
    # High-g positions are those where a[k*-i] is large
    g_proj = g - g.mean()
    direction = -g_proj  # Move opposite to gradient
    direction = direction / (np.linalg.norm(direction) + 1e-15)
    
    perturbed = a + scale * (direction * a + rng.standard_normal(n) * 0.01)
    return norm_sequence(perturbed)


def peak_surgery(a0, time_limit, n_steps=20000):
    """Direct peak reduction by adjusting individual coordinates."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (exclude boundaries which are always smaller)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(10, len(idx))
        top_ranking = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        for rank in top_ranking:
            i = idx[rank]
            j = peak_k - i
            if i == j:
                # Single coordinate contributes
                for factor in [0.95, 0.90, 0.85, 0.80]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    # Redistribute to positions with lowest contribution
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    other_contribs = contribs.copy()
                    other_contribs[rank] = -np.inf
                    low_ranking = np.argsort(contribs)[:top_n]
                    redistribute_mask = np.zeros(n, dtype=bool)
                    for lr in low_ranking:
                        if lr != rank:
                            redistribute_mask[idx[lr]] = True
                    if redistribute_mask.sum() > 0:
                        diff = old_val - a_trial[i]
                        weights = a_trial[redistribute_mask] + 1e-10

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_subgradient_refine(a0, time_limit, n_max_iters=5000):
    """Nesterov-accelerated subgradient method for the peak."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        
        # Subgradient
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]  # Note: using a, not y, for the gradient
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * (y - g)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a_new = ((it + 1) * a + alpha * (it + 2) * (y - g)) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
    
    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage L-BFGS with adaptive log-sum-exp annealing (many low-T stages)
2. Nesterov-accelerated subgradient for peak reduction
3. Peak surgery with coordinate descent for final polish
4. Multi-length exploration with interpolation
5. Directional perturbation informed by peak structure
Key improvements: more low-temperature stages, hybrid L-BFGS + subgradient,
better time allocation, and more aggressive final polishing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, T):
    """Compute log-sum-exp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse

    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_refine(a0, time_limit, T_schedule, maxiter_per_stage=6000):
    """L-BFGS with log-sum-exp smoothing and given temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            obj, _ = lse_objective_and_grad(z_in, T)
            return obj

        def jac(z_in):
            _, g = lse_objective_and_grad(z_in, T)
            return g

        iters = max(20, min(maxiter_per_stage, int(remaining / 0.03)))

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-16,
                         'maxfun': iters * 200, 'disp': False}
            )
            if res.x is not None:
                z = res.x
                a_trial = softmax_from_z(z)
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_peak_reduce(a0, time_limit, n_steps=20000):
    """Subgradient method with line search focused on reducing the peak."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    momentum = 0.0
    v = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]

        # Project onto simplex tangent
        g_proj = g - g.mean()

        # Heavy ball momentum
        momentum = 0.9 * momentum
        v = momentum * v + (1 - momentum) * g_proj

        step_size = 0.3 / (np.sqrt(step + 1) + 1.0)
        direction = -v

        # Line search
        best_trial = None
        best_trial_score = best_score
        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a_trial = a + step_size * mult * direction
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a_trial.copy()
        else:
            v = 0.5 * v

    return best_a, best_score


def nesterov_subgradient_refine(a0, time_limit, n_max_iters=20000):
    """Nesterov-accelerated subgradient method."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        k_star = int(np.argmax(b))

        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]
        g = g - g.mean()

        alpha = 2.0 / (it + 2)
        y = y - alpha * g
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a_new = ((it + 1) * a - alpha * (it + 2) * g) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

    return best_a, best_score


def peak_surgery(a0, time_limit, n_steps=30000):
    """Direct peak reduction via coordinate adjustments."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[:2] = -np.inf
            b_work[-2:] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(12, len(idx))
        top_ranking = np.argsort(contribs)[::-1][:top_count]
        low_ranking = np.argsort(contribs)[:top_count]

        improved = False
        for rank in top_ranking:
            i = idx[rank]
            j = peak_k - i
            if i == j:
                for factor in [0.96, 0.92, 0.88, 0.84, 0.78]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    diff = old_val - a_trial[i]
                    # Redistribute to low-contribution positions
                    redist_positions = []
                    for lr in low_ranking:
                        if lr != rank:
                            pos = idx[lr]
                            if pos != i and a[pos] > 1e-10:
                                redist_positions.append(pos)
                    if redist_positions:
                        redist_arr = a_trial[redist_positions] + 1e-10
                        a_trial[redist_positions] += diff * redist_arr / redist_arr.sum()
                    a_trial = norm_sequence(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        break
            else:
                if j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            best_score = score
                            best_a = a_trial.copy()
                            a = a_trial
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 150:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def directional_perturb(a0, rng, scale=0.05):
    """Perturb in a direction informed by peak structure."""
    n = len(a0)
    a = norm_sequence(a0)
    b = fast_convolve(a)
    k_star = int(np.argmax(b))

    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for i in range(js, je + 1):
        g[i] = a[k_star - i]

    g_proj = g - g.mean()
    direction = -g_proj
    norm_dir = np.linalg.norm(direction)
    if norm_dir > 1e-15:
        direction = direction / norm_dir

    noise = rng.standard_normal(n) * 0.02
    perturbed = a + scale * (direction * a + noise)
    return norm_sequence(perturbed)


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
