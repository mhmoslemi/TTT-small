For a flat aperiodic power spectrum, we need |A(f)|^2 = constant for all f. This means the DFT of a should have constant magnitude, i.e., a is a "linear phase sequence with flat magnitude response" - which is essentially a "maximum length sequence" or "m-sequence" translated to continuous values.

For a non-negative sequence with flat DFT magnitude, one approach is:
1. Generate a random complex spectrum S(f) with |S(f)| = constant.
2. Compute a = IDFT(S).
3. Project a onto the non-negative cone (a_i = max(0, a_i)) and normalize.
4. Repeat: update S(f) = DFT(a) * (|S_target(f)| / |DFT(a)(f)|) where S_target has constant magnitude.
5. This is like a "Gerchberg-Saxton" or "error reduction" algorithm.

But we need the APERIODIC autocorrelation to be flat, not the periodic one. The difference is subtle but important.

Actually, for the purpose of initialization, let's just use the periodic DFT as an approximation. For large n, the aperiodic and periodic correlations are similar in the central region.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
spectral initialization, aggressive multi-length Nesterov LP screening,
p-norm L-BFGS refinement, mirror descent, and local peak shaving.
Key innovations: flat-spectrum initialization, wider length exploration,
adaptive Nesterov with momentum, and entropic mirror descent.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_smoothing_z(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_gradient_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 6000 // n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            return p_norm_smoothing_z(z_val, stage_p)
        
        def jac(z_val):
            return p_norm_gradient_z(z_val, stage_p)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm LBFGS p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 500 == 0:
            if best_score < prev_stall_score - 1e-7:

                prev_stall_score = best_score

                rng = np.random.default_rng(it + 42)
                noise_scale = 0.03 + 0.02 * stall_count
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                # Also perturb a

                a = a + noise_a
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                stall_count = 0
                prev_stall_score = best_score
        
        if (it + 1) % 2000 == 0:

def local_refinement(a0, time_limit, n_steps=20000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        
        peak_indices = np.argsort(b_work)[::-1][:5]
        peak_val = b_work[peak_indices[0]]
        near_threshold = peak_val - 0.05 * peak_val
        near_peaks = [k for k in range(len(b_work)) if b_work[k] >= near_threshold][:8]
        
        improved = False
        contrib_scores = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contribs = a[idx] * a[k - idx]
                contrib_scores[idx] += contribs
        
        top_coords = np.argsort(contrib_scores)[::-1][:15]
        
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score

        if not improved:
            peak_k = peak_indices[0]

            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_pairs = np.argsort(contribs)[::-1][:6]
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved and step % 100 == 0:
            rng = np.random.default_rng(step + 7)
            n_pert = rng.integers(1, 5)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.05, 0.05))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()

def spectral_init(n, n_iters=50, seed=42):
    """Initialize a sequence with approximately flat DFT magnitude using alternating projections."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    for _ in range(n_iters):
        A = np.fft.fft(a)
        target_mag = np.full(n, np.abs(A[0]) / np.sqrt(n))
        target_mag[0] = np.abs(A[0])
        phase = np.angle(A + 1e-30)
        A_new = target_mag * np.exp(1j * phase)
        A_new[0] = target_mag[0]  # DC: real positive
        a = np.real(np.fft.ifft(A_new))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        # Asymmetric: skewed Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) * 8)
    elif variant == 6:
        # Asymmetric: skewed Gaussian other side
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.7) ** 2) * 8)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using multi-phase optimization:
spectral initialization, aggressive multi-length Nesterov LP screening,
p-norm L-BFGS refinement, entropic mirror descent, and local peak shaving.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_smoothing_z(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_gradient_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax param)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 8000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return p_norm_smoothing_z(z_val, stage_p)

        def jac(z_val):
            return p_norm_gradient_z(z_val, stage_p)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm LBFGS p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

    n = len(a)

def spectral_init(n, n_iters=40, seed=42):
    """Initialize with approximately flat DFT magnitude via alternating projections."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    for it in range(n_iters):
        A = np.fft.fft(a)
        target_mag = np.full(n, np.abs(A[0]) / np.sqrt(n))
        target_mag[0] = np.abs(A[0])
        # Blend: partially enforce flat magnitude
        blend = 0.5 + 0.5 * (it / n_iters)
        mag = np.abs(A)
        new_mag = blend * target_mag + (1 - blend) * mag
        phase = np.angle(A + 1e-30)
        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = np.abs(A_new[0])
        a = np.real(np.fft.ifft(A_new))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a


def structured_init(n, variant=0):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state with a new candidate."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def mirror_descent_refine(a0, time_limit, n_steps=3000):
    """Mirror descent with entropic regularization in log-domain (p-smoothed)."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    p = 8.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()

        c = fast_convolve(a)
        c_pos = np.maximum(c, 1e-300)
        p_val = c_pos ** p
        S = np.sum(p_val)
        if S < 1e-300:
            break
        factor = S ** (1.0 / p - 1.0)
        w = c_pos ** (p - 1.0)
        corr = np.correlate(w, a, mode='valid')
        g_a = 4.0 * n * factor * corr
        g_z = a * (g_a - np.dot(g_a, a))

        z = z - lr * g_z
        z = z - z.max()

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_new = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        if step % 500 == 0 and step > 0:
            lr *= 0.85
            elapsed = time.monotonic() - t0
            print(f"      MirrorDesc step={step} ({elapsed:.0f}s): score={best_score:.6f}, lr={lr:.5f}")

    return best_a


def nesterov_lp_quick(a0, n_iters, solver=None):
    """Quick Nesterov LP for screening (no time budget, fixed iterations)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining spectral init, multi-length Nesterov LP
    screening, p-norm L-BFGS, mirror descent, and local refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: Multi-length screening (180s) ===
    print(f"\n=== Phase 1: Multi-length screening (best={best_state['score']:.6f}) ===")
    t_screen_start = time.monotonic()
    t_screen = min(deadline - t_screen_start, 180.0)

    candidate_lengths = sorted(set([
        512, 640, 768, 832, 896, 960, 1024, 1088, 1152,
        1216, 1280, 1344, 1408, 1472, 1536, 1600, 1664, 1728,
        1792, 1856, 1920, 1984, 2048
    ]))

    # Filter to reasonable range
    candidate_lengths = [n for n in candidate_lengths if 512 <= n <= 2048]

    # Prioritize: include parent length, nearby lengths, and spread out
    scored_lengths = []
    for n_c in candidate_lengths:
        init_variants = [4, 0, 5, 8]  # Gaussian center, sine, asymmetric, ramp
        for v in init_variants:
            scored_lengths.append((n_c, v))

    # Sort: parent length first, then nearby, then spread
    scored_lengths.sort(key=lambda x: (abs(x[0] - n_best), x[1]))

    # Limit to fit in screen time
    n_screen_candidates = min(len(scored_lengths), 16)
    # Make sure we include diverse lengths
    seen_lengths = set()
    selected = []
    for n_c, v in scored_lengths:
        if n_c in seen_lengths:
            continue
        seen_lengths.add(n_c)
        selected.append((n_c, v))
        if len(selected) >= n_screen_candidates:
            break
    # Fill remaining with different variants of already-seen lengths
    for n_c, v in scored_lengths:
        if (n_c, v) not in selected:
            selected.append((n_c, v))
        if len(selected) >= n_screen_candidates:
            break

    screen_results = []
    for n_c, v in selected:
        rem = deadline - time.monotonic()
        t_per = min(rem / max(1, len(selected) - len(screen_results)), 15.0)
        if t_per < 3:
            break

        if n_c == n_best and parent is not None:
            init = parent.copy()
        else:
            init = structured_init(n_c, variant=v)

        t0_scr = time.monotonic()
        a_scr, scr_score = nesterov_lp_quick(init, n_iters=30, solver=solver)
        elapsed_scr = time.monotonic() - t0_scr
        screen_results.append((scr_score, n_c, a_scr.copy()))
        print(f"  Screen n={n_c} v={v}: score={scr_score:.6f} ({elapsed_scr:.1f}s)")

    # Sort by score
    screen_results.sort(key=lambda x: x[0])
    print(f"  Top screening results:")
    for sc, nc, _ in screen_results[:5]:
        print(f"    n={nc}: {sc:.6f}")

    # Use the best screening result as our new starting point
    if screen_results:
        best_scr_score, best_scr_n, best_scr_a = screen_results[0]
        if best_scr_score < best_state['score'] + 0.01:
            best_state['seq'] = best_scr_a.copy()
            best_state['score'] = min(best_state['score'], best_scr_score)
            n_best = best_scr_n
            print(f"  Selected n={n_best} for main optimization (score={best_state['score']:.6f})")

    # === Phase 2: Main Nesterov LP (500s) ===
    print(f"\n=== Phase 2: Main Nesterov LP n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=100000)
        improved = try_update(best_state, a_lp, "nesterov_lp_main")

        # Interleave p-norm if time remains
        rem = deadline - time.monotonic()
        if rem > 60:
            t_pn = min(rem * 0.3, 100.0)
            a_pn = p_norm_lbfgs_refine(best_state['seq'], t_pn, p_start=20.0, p_end=300.0, n_stages=6)
            try_update(best_state, a_pn, "pn_after_lp")
            # Then do more LP
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp2 = min(rem * 0.4, 150.0)
                a_lp2, _ = nesterov_lp_refine(best_state['seq'], t_lp2, solver=solver, n_max_iters=50000)
                try_update(best_state, a_lp2, "lp_after_pn")

    # === Phase 3: p-norm L-BFGS intensive (300s) ===
    print(f"\n=== Phase 3: p-norm L-BFGS intensive (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, p: 50 -> 800")
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_start=50.0, p_end=800.0, n_stages=7)
        try_update(best_state, a_pn, "pn_intensive")

    # === Phase 4: Mirror descent (150s) ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_md = mirror_descent_refine(best_state['seq'], t_alloc, n_steps=5000)
        try_update(best_state, a_md, "mirror_descent")

    # === Phase 5: Alternative length exploration (300s) ===
    print(f"\n=== Phase 5: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        # Try lengths from screening that weren't the main one
        alt_lengths = [nc for sc, nc, _ in screen_results if nc != len(best_state['seq'])][:5]
        # Also add some new ones
        for extra in [800, 1100, 1300, 1500, 1700, 1900]:
            if extra != len(best_state['seq']) and extra not in alt_lengths and 512 <= extra <= 2048:
                alt_lengths.append(extra)

        alt_lengths = sorted(set(alt_lengths))[:5]
        n_explored = 0

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 4:
                break
            n_explored += 1

            t_total = min(rem * 0.20, 120.0)
            if t_total < 40:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_state['seq'], alt_n)

            # Quick LP to warm up
            t_lp_short = t_total * 0.4
            a_alt, _ = nesterov_lp_refine(init_alt, t_lp_short, solver=solver, n_max_iters=20000)
            improved = try_update(best_state, a_alt, f"lp_n{alt_n}")

            # If improved or close, do p-norm
            rem = deadline - time.monotonic()
            if rem > 20:
                t_pn_alt = min(rem * 0.15, 60.0)
                a_alt2 = p_norm_lbfgs_refine(best_state['seq'] if not improved else a_alt,
                                              t_pn_alt, p_start=30.0, p_end=400.0, n_stages=5)
                try_update(best_state, a_alt2, f"pn_n{alt_n}")

    # === Phase 6: Perturbation + re-optimization (150s) ===
    print(f"\n=== Phase 6: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.12, 120.0)
        n_cur = len(best_state['seq'])

        for noise_scale in [0.05, 0.10, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.3, 50.0)
            a_pert = p_norm_lbfgs_refine(perturbed, t_sub, p_start=50.0, p_end=500.0, n_stages=5)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 7: Final polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = p_norm_lbfgs_refine(best_state['seq'], t_fin, p_start=200.0, p_end=1000.0, n_stages=4)
        try_update(best_state, a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_lr = min(rem * 0.30, 60.0)
        a_lr, _ = local_refinement(best_state['seq'], t_lr, n_steps=20000)
        try_update(best_state, a_lr, "final_local")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.40, 40.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=15000)
        try_update(best_state, a_lp_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr2 = min(rem * 0.50, 30.0)
        a_lr2, _ = local_refinement(best_state['seq'], t_lr2, n_steps=10000)
        try_update(best_state, a_lr2, "final_local2")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    total_time = time.monotonic() - (deadline - float(budget_s) + 30.0)
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={total_time:.0f}s)")
    return [float(x) for x in best_state['seq']]
