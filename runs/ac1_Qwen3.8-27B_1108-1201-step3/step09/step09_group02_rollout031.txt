def simplex_project(v):
    """Project vector v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

def _gs_commit(a_raw, n):
    """Project raw time-domain signal to valid sequence."""
    a_raw = np.maximum(a_raw, 0.0)

    - Time domain: project to non-negative, sum=1

    no_improve = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_budget:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(3.0 * v, 5.0 / n)
            lo = 0.0
            hi = upper
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = _cd_eval(c, c_vals, a_partners, unaff_max, has_quad, quad_idx,
                          current_sum, v, n)
            fd = _cd_eval(d, c_vals, a_partners, unaff_max, has_quad, quad_idx,
                          current_sum, v, n)
            for _ in range(30):
                if hi - lo < 1e-16:
                    break
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = _cd_eval(c, c_vals, a_partners, unaff_max, has_quad,
                                  quad_idx, current_sum, v, n)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = _cd_eval(d, c_vals, a_partners, unaff_max, has_quad,
                                  quad_idx, current_sum, v, n)
            best_x = (lo + hi) * 0.5

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            print(f"    ECD {label} sweep={sweeps} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_budget, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0
    factors = [0.995, 0.99, 0.985, 0.97, 0.95, 1.05, 1.03, 1.01, 1.005]

    while time.monotonic() - t0 < time_budget:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(k_hi - k_lo + 1):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            cur_local_max = float(conv[k_lo:k_hi + 1].max())
            best_delta = 0.0
            best_local_max = cur_local_max

            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                lm = float(nv.max())
                if lm < best_local_max - 1e-16:
                    best_local_max = lm
                    best_delta = x - v

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

    return best_a, best_score


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_asymmetric(n, rng):
    """Asymmetric Gaussian initialization."""
    t = np.arange(n, dtype=np.float64)
    shift = rng.uniform(0.15, 0.85)
    center = n * shift
    width = n * rng.uniform(0.1, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_quadratic_residues(n):
    """Number-theoretic initialization based on quadratic residues."""
    # Find a prime p >= n
    p = n
    while True:
        if all(p % d != 0 for d in range(2, int(p ** 0.5) + 1)):
            break
        p += 1
    residues = set()
    for i in range(1, p):
        residues.add((i * i) % p)
    a = np.zeros(n)
    for i in range(n):
        if i in residues:
            a[i] = 1.0
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of low-frequency cosines."""
    n_harmonics = rng.integers(3, 12)
    amps = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, min(n // 2, 50) + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def combined_local_opt(a0, time_budget, verbose=False, label=""):
    """Run GS → subgradient → ECD in sequence as a local optimizer."""
    a = np.asarray(a0, dtype=np.float64).copy()
    best_a = a.copy()
    best_score = float('inf')

    # Phase A: GS iteration (60% of budget)
    t_gs = time_budget * 0.5
    a_gs, s_gs = gs_iteration(a, t_gs, verbose=verbose, label=label + "_gs")
    if s_gs < best_score:
        best_score = s_gs
        best_a = a_gs.copy()

    # Phase B: Subgradient (20% of budget)
    t_sg = time_budget * 0.2
    a_sg, s_sg = subgradient_descent(a_gs, t_sg, verbose=verbose, label=label + "_sg")
    if s_sg < best_score:
        best_score = s_sg
        best_a = a_sg.copy()

    # Phase C: Exact coordinate descent (30% of budget)
    t_cd = time_budget * 0.3
    a_cd, s_cd = exact_coord_descent(best_a, t_cd, verbose=verbose, label=label + "_cd")
    if s_cd < best_score:
        best_score = s_cd
        best_a = a_cd.copy()

    return best_a, best_score


def basin_hopping_gs(a0, time_budget, rng, n_perturbs=6, verbose=False):
    """Basin hopping using GS-based local optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_budget:
            break
        rem = time_budget - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.6, 120.0)

        # Multiple perturbation strategies
        strategy = p % 3
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce elements at high auto-correlation lags
            conv_p = fast_convolve(a)
            top_k = np.argpartition(conv_p, -5)[-5:]
            perturbed = a.copy()
            for k in top_k:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - rng.random() * 0.2)
            perturbed = perturbed + rng.standard_normal(n) * 0.01
        else:
            # Sparse perturbation: change a random subset
            perturbed = a.copy()
            n_change = max(5, n // 20)
            changes = rng.choice(n, size=n_change, replace=False)
            perturbed[changes] *= (1.0 + rng.standard_normal(n_change) * 0.3)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = combined_local_opt(perturbed, t_alloc, verbose=False)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} (strat={strategy}): improved to {best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining GS iteration, subgradient descent,
    coordinate descent, and basin hopping for minimizing the auto-correlation peak.
    """
    deadline = time.monotonic() + 1780.0

    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick GS optimization ===
    print("\n=== Phase 1: Diverse Init + Quick GS ===")
    t1_start = time.monotonic()

    n_values = [768, 1024, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values[:4]:
        elapsed = time.monotonic() - t1_start
        if elapsed > 200:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))
        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('asym', init_asymmetric(n, rng)))
        inits.append(('quadr', init_quadratic_residues(n)))
        inits.append(('fourier', init_fourier_mix(n, rng)))

        for itype, cand in inits[:7]:
            t_left = 200.0 - (time.monotonic() - t1_start)
            if t_left < 10:
                break
            t_alloc = max(5.0, min(25.0, t_left / 10.0))
            a_gs, s_gs = gs_iteration(cand, t_alloc, label=f"p1_n{n}_{itype}")
            candidates.append((s_gs, a_gs, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best local: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Combined local optimization on top candidates ===
    print(f"\n=== Phase 2: Combined Local Opt (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_start = time.monotonic()

        # Run combined optimization on the best
        current = best_seq.copy()
        for rnd in range(3):
            rem = deadline - time.monotonic() - 400
            if rem < 60:
                break
            t_alloc = min(rem * 0.35, 200.0)
            print(f"  Combined round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = combined_local_opt(current, t_alloc, verbose=True, label=f"ph2_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_r{rnd}")

            if best_score > 1.5035:
                perturbed = a_opt + rng.standard_normal(n_best) * 0.03
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 3: Subgradient focus on best ===
    print(f"\n=== Phase 3: Subgradient Focus (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 50:
            t_sg = min(rem * 0.5, 150.0)

    # === Phase 5: Try different length from best ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [n_best + 256, max(256, n_best - 256)]:

            rem = deadline - time.monotonic() - 150

            init_alt = interpolate_seq(best_seq, alt_n)

    if best_seq is not None:
        # GS polish
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_gs = min(rem * 0.3, 60.0)
            print(f"  GS polish, budget={t_gs:.0f}s")
            a_gs, s_gs = gs_iteration(best_seq, t_gs, label="polish_gs")
            best_seq, best_score = check_and_update(a_gs, best_seq, best_score, "polish_gs")

        # ECD polish
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_cd = min(rem * 0.4, 50.0)
            print(f"  ECD polish, budget={t_cd:.0f}s")
            a_cd, s_cd = exact_coord_descent(best_seq, t_cd, verbose=True, label="polish_ecd")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "polish_ecd")

        # Fine CD
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_fine = min(rem * 0.5, 30.0)
            print(f"  Fine CD, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "final_fine")

        # Last ECD pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_last, s_last = exact_coord_descent(best_seq, min(rem, 15.0), label="last")
            best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
