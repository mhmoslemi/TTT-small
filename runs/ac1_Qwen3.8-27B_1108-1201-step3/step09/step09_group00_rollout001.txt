- Solve a QP: minimize 0.5*a^T H a + g^T a subject to a>=0, sum(a)=1, where H and g encode the active constraints
- Actually this is still non-convex. Instead, use a "penalty" approach:
  - minimize sum_{k in active} (c[k] - t)^2 + lambda * sum_{k not in active} max(0, c[k]-t)^2
  - This is smooth and can be optimized with L-BFGS

Phase 3 (20% time): Multi-n exploration:
- Interpolate to n*1.1, n*1.2, n*1.5
- Quick optimize each (50 iterations of LP)
- Keep the best

Phase 4 (20% time): Final ultra-precise optimization:
- Very low temperature log-sum-exp
- Then exact subgradient with tiny steps
- Coordinate descent on the top 20 contributing elements

Let me also add a new idea: **"Spectral preconditioning"**

The idea: the Hessian of the objective (at a peak position k*) is:
H[j,m] = (1/n)*delta(j, k*-m) + (1/n)*delta(m, k*-j) (approximately, for the quadratic part)

The eigenvalues of this Hessian are related to the structure of a. If a is smooth, the Hessian has a wide range of eigenvalues, making gradient descent slow in some directions.

Preconditioning: use the diagonal of the Hessian as a preconditioner:
d[j] = 2*a[k*-j] (approximately)

Then the preconditioned step is: a[j] -= lr * (grad[j] / d[j])

This is similar to AdaGrad.

**Actually, let me try something really different: "Genetic Algorithm" approach**

1. Maintain a population of sequences
2. Crossover: take two sequences, split them at a random point, and combine (first half from one, second half from other)
3. Mutation: add Gaussian noise, or smooth/despel the sequence
4. Selection: keep the top M by score
5. Elitism: always keep the best

This explores the search space more broadly and might escape local minima.

But for n=1200, storing and manipulating a population might be expensive. Let's use a small population (5-10).

**Let me now write the code. I'll combine several approaches:**

1. Start from parent
2. Phase 1: Nesterov LP (improved with backtracking and Nesterov momentum)
3. Phase 2: LogSumExp L-BFGS with adaptive temperature
4. Phase 3: Multi-n exploration (try n±20%)
5. Phase 4: "Active set" subgradient with AdaGrad
6. Phase 5: Final coordinate descent polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Helper functions (all top-level, no closures)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

# LP solver
def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    return None

# Nesterov LP with improvements
def nesterov_lp(a0, time_limit, solver, max_iters=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:

            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y

            if stall >= 10:

                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
    
    return best_a, best_score

def adagrad_subgrad(a0, time_limit, n_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # AdaGrad accumulator
    G = np.zeros(n)
    base_lr = 0.1
    eps = 1e-8
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set (positions within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Project gradient to sum-zero subspace (to maintain sum=1)
        g -= g.mean()
        
        # AdaGrad update
        G += g * g
        lr = base_lr / (np.sqrt(G) + eps)
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Don't update a, but still update G (this is standard AdaGrad)
            pass
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdaGrad {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score

def lbgfs_lse(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T) * lse
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            iters = max(50, min(maxiter_per_stage, int(remaining * 100)))
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                                   'maxfun': iters * 100, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax(z)

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def coord_descent(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Get contributing positions
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try adjusting top contributors
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for new_val_frac in [0.999, 0.998, 0.995, 0.99]:
                a_trial = a.copy()
                old_val = a_trial[pos]
                a_trial[pos] = old_val * new_val_frac
                diff = old_val - a_trial[pos]
                if diff < 1e-15:
                    continue
                # Redistribute to non-peak contributors
                mask = np.ones(n, dtype=bool)
                # Don't redistribute to top contributors
                for p in top_positions:
                    mask[p] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 1000:
                break
        else:
            stall = 0
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
