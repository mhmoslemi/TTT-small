    """Linear convolution of a and b via FFT."""

    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Compute score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def make_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def make_two_peak(n, spread):
    t = np.arange(n, dtype=np.float64)
    center1 = n * 0.3
    center2 = n * 0.7
    a = np.exp(-0.5 * ((t - center1) / spread) ** 2) + np.exp(-0.5 * ((t - center2) / spread) ** 2)
    s = a.sum()
    return a / s


def make_spectral_init(n, rng):
    """Initialize using spectral method: choose random Fourier coefficients, square magnitude."""
    m = int(rng.integers(2, n // 4 + 1))
    # Random low-frequency spectrum
    freqs = np.arange(n)
    A = np.zeros(n, dtype=np.complex128)
    for k in range(m):
        A[k] = complex(rng.standard_normal(), rng.standard_normal())
        A[n-k] = np.conj(A[k])
    A[0] = abs(A[0])  # keep DC real and positive
    a = np.abs(irfft(A, n)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def coordinate_descent_exact(a0, time_limit, verbose=True, label="", n_ternary=50):
    """
    Coordinate descent with exact 1D subproblem solving.
    For each coordinate i, find the optimal delta that minimizes max(conv) 
    using ternary search on the convex function g(delta).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_improvement = 0.0

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            aff_len = k_end - k_start + 1

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            C = max(unaff_left, unaff_right)

            # Affected values: conv[k] + 2*delta*a[k-i] + delta^2*(k==2i)
            # Base values
            base = conv[k_start:k_end + 1].copy()
            # Linear coefficients: 2*a[k-i] for k in [k_start, k_end]
            # k-i ranges from k_start-i=0 to k_end-i
            lin_coef = 2.0 * a[0:aff_len]  # a[k-i] for k from k_start to k_end, k-i from 0 to aff_len-1
            # Quadratic term at position 2i (if in range)
            quad_pos = 2 * i - k_start  # position of 2i in the affected range
            has_quad = (quad_pos >= 0 and quad_pos < aff_len)

            # Define g(delta) = max(C, max_j (base[j] + delta*lin_coef[j]), quad_term)
            # where quad_term = base[quad_pos] + delta*lin_coef[quad_pos] + delta^2 if has_quad
            
            # Ternary search for optimal delta in [-a[i], a[i]*2]
            lo = -a[i]
            hi = a[i] * 2.0

            # Evaluate g at a point
            def eval_g(delta):
                vals = base + delta * lin_coef
                if has_quad:
                    vals = vals.copy()
                    vals[quad_pos] += delta * delta
                return max(C, float(vals.max()))

            # Check if current delta=0 is already optimal (gradient check)
            g0 = eval_g(0.0)
            if g0 <= M + 1e-14:
                continue  # No improvement possible from this base

            # Ternary search
            for _ in range(n_ternary):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                g1 = eval_g(m1)
                g2 = eval_g(m2)
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1

            delta_opt = (lo + hi) / 2.0
            g_opt = eval_g(delta_opt)

            if g_opt < M - 1e-14:
                # Apply update
                a[i] = max(a[i] + delta_opt, 0.0)
                # Update convolution
                conv[k_start:k_end + 1] = base + delta_opt * lin_coef
                if has_quad:
                    conv[k_start + quad_pos] += delta_opt * delta_opt
                M = max(C, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
                sweep_improvement += (M - g_opt)  # approximate

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.8f}")

    return best_a, best_score


def cd_with_restart(a0, time_limit, n_restarts=5, verbose=True):
    """Coordinate descent with periodic random perturbations."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    time_per_phase = time_limit / (n_restarts + 1)

    for restart in range(n_restarts + 1):
        if restart > 0:
            # Perturb
            perturb_scale = 0.1 / restart
            a = a + rng.standard_normal(n) * perturb_scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()

        remaining = time_limit - (time.monotonic() - t0)
        t_alloc = min(time_per_phase, remaining)
        if t_alloc < 5:
            break

        a, s_curr = coordinate_descent_exact(a, t_alloc, verbose=False, label=f"r{restart}")
        if s_curr < best_score:
            best_score = s_curr
            best_a = a.copy()
            if verbose:
                print(f"  Restart {restart}: new best {best_score:.6f}")

    return best_a, best_score


def fw_fast(a0, time_limit, n_steps=2000, verbose=True):
    """
    Fast Frank-Wolfe for the problem:
    minimize max_k (a*a)[k] s.t. a >= 0, sum a = 1
    
    Using the subgradient: at the maximizing k*, the subgradient is 2*a shifted.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    conv = fast_convolve(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        k_star = int(np.argmax(conv))
        
        # Subgradient direction: want to minimize a[k_star - j] for j
        # The "descent direction" in FW: find j that minimizes the subgradient
        # subgrad[j] = 2 * a[k_star - j] if 0 <= k_star - j <= n-1, else 0
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        sg = np.zeros(n)
        sg[i_start:i_end + 1] = 2.0 * a[k_star - i_end:k_star - i_start + 1][::-1]
        
        # FW step: move towards e_j where j minimizes sg
        j_star = i_start + int(np.argmin(sg[i_start:i_end + 1]))
        
        step_size = 2.0 / (it + 2.0)
        
        # Update convolution incrementally
        # New a = (1-t)*a + t*e_{j_star}
        # New conv = (1-t)^2 * conv + 2*t*(1-t) * conv(a, e_{j_star}) + t^2 * conv(e_{j_star}, e_{j_star})
        # conv(a, e_{j_star})[k] = a[k - j_star] (shifted a)
        # This is complex to do incrementally. Let's just recompute.
        
        a = (1.0 - step_size) * a
        a[j_star] += step_size
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        conv = fast_convolve(a)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it} ({elapsed:.0f}s): best={best_score:.6f} cur={cur_score:.6f}")

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent
with exact 1D convex subproblem solving via ternary search.
Multi-phase: parent refinement, diverse inits, length exploration, polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_norm(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())


def init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(n - 1, 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def init_two_peak(n, frac=0.15):
    t = np.arange(n, dtype=np.float64)
    c1, c2 = 0.3 * n, 0.7 * n
    w = n * frac
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def init_spectral(n, rng):
    m = max(2, int(n * 0.05))
    A = np.zeros(n, dtype=np.complex128)
    for k in range(1, m):
        A[k] = complex(rng.standard_normal(), rng.standard_normal())
        A[n - k] = np.conj(A[k])
    A[0] = abs(A[0]) + 1.0
    a = np.abs(irfft(A, n)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def _eval_g(base, lin_coef, quad_pos, has_quad, C, delta):
    """Evaluate g(delta) = max(C, max(base + delta*lin_coef [+ delta^2 at quad_pos]))."""
    vals = base + delta * lin_coef
    if has_quad:
        vals = vals.copy()
        vals[quad_pos] += delta * delta
    m = float(vals.max())
    return m if m > C else C


def cd_exact(a0, time_limit, n_ternary=45, verbose=True, label=""):
    """
    Fast coordinate descent with exact 1D subproblem.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Pre-allocate buffers
    base_buf = np.empty(n, dtype=np.float64)
    lin_buf = np.empty(n, dtype=np.float64)

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            aff_len = k_end - k_start + 1

            # Quick check: if global max is not in affected range, skip
            k_star = int(np.argmax(conv))
            if k_star < k_start or k_star > k_end:
                continue

            # Unaffected max (approximate with global max if k_star not in range, 
            # but we already checked, so k_star IS in range here)
            # Compute exact unaffected max
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            C = max(unaff_left, unaff_right)

            # Affected base and linear coefficients
            np.copyto(base_buf[:aff_len], conv[k_start:k_end + 1])
            np.copyto(lin_buf[:aff_len], 2.0 * a[:aff_len])

            quad_pos = 2 * i - k_start
            has_quad = (0 <= quad_pos < aff_len)

            # Ternary search for optimal delta in [-ai, 2*ai]
            lo = -ai
            hi = 2.0 * ai

            # Quick check at delta=0
            g0 = _eval_g(base_buf[:aff_len], lin_buf[:aff_len], quad_pos, has_quad, C, 0.0)
            if g0 <= M + 1e-13:
                continue

            for _ in range(n_ternary):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                g1 = _eval_g(base_buf[:aff_len], lin_buf[:aff_len], quad_pos, has_quad, C, m1)
                g2 = _eval_g(base_buf[:aff_len], lin_buf[:aff_len], quad_pos, has_quad, C, m2)
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1

            delta = (lo + hi) / 2.0
            g_new = _eval_g(base_buf[:aff_len], lin_buf[:aff_len], quad_pos, has_quad, C, delta)

            if g_new < M - 1e-13:
                a[i] = max(ai + delta, 0.0)
                conv[k_start:k_end + 1] = base_buf[:aff_len] + delta * lin_buf[:aff_len]
                if has_quad:
                    conv[k_start + quad_pos] += delta * delta
                M = g_new
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={2.0*n*M:.6f} best={best_score:.6f}")

    return best_a, best_score


def cd_multi_restart(a0, time_limit, n_restarts=4, verbose=True):
    """CD with periodic random perturbations to escape local minima."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    phase_time = time_limit / (n_restarts + 1)

    for r in range(n_restarts + 1):
        if r > 0:
            scale = 0.05 / r
            a = a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()

        remaining = time_limit - (time.monotonic() - t0)
        t_alloc = min(phase_time, remaining)
        if t_alloc < 10:
            break

        a, s = cd_exact(a, t_alloc, n_ternary=45, verbose=(r == 0 and verbose), label=f"r{r}")
        if s < best_score:
            best_score = s
            best_a = a.copy()
            if verbose:
                print(f"  MultiCD r={r}: new best {best_score:.6f}")

    return best_a, best_score


def spectral_polish(a0, time_limit, verbose=True):
    """
    Use the fact that in frequency domain, conv(a,a) has DFT = A^2.
    Try to make A^2 more "spread out" by adjusting a.
    This is a heuristic spectral method.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Find top few peaks
        k_act = min(10, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.01 * M
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        if it % 200 == 0:
            lr *= 0.85
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SpectPolish it={it} ({elapsed:.0f}s): best={best_score:.6f}")

        if it > 5000:
            break

    return best_a, best_score
