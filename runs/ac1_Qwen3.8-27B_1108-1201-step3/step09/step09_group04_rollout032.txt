import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    c = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * c.max()


def p_norm_value_z(z, p):
    """Compute 2n * (sum c_k^p)^(1/p) where a = softmax(z), c = conv(a,a)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_grad_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(z0, time_limit, p_vals, max_iter_per_stage=200):
    """L-BFGS with schedule of p values."""
    z = z0.copy()
    t0 = time.monotonic()
    for p_val in p_vals:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj(z_v):
            return p_norm_value_z(z_v, p_val)

        def grad(z_v):
            return p_norm_grad_z(z_v, p_val)

        try:
            res = minimize(
                fun=obj, x0=z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': max_iter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    p={p_val:.0f} ({elapsed:.0f}s): score={sc:.6f}")
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def build_conv_matrix(a):
    """Build convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    y = a.copy()

    stall = 0
    prev_best = best_score

            break
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()

        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0

            best_score = sc_y
            best_a = y.copy()
            stall = 0

def ansatz_linear_ramp(n, alpha=1.0, gamma=0.0, shift=0.5):
    """a[i] = (i+shift+1)^alpha, optionally with decay (n+1-i)^gamma."""
    i = np.arange(n, dtype=np.float64) + 1.0 + shift
    if alpha == 0 and gamma == 0:
        a = np.ones(n)
    elif gamma == 0:
        a = i ** alpha
    elif alpha == 0:
        a = (n + 1 - np.arange(n, dtype=np.float64)) ** gamma
    else:
        a = i ** alpha * (n + 1 - np.arange(n, dtype=np.float64)) ** gamma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def ansatz_exp_curvature(n, a1, a2, a3=0.0):
    """a[i] = exp(a1*x + a2*x^2 + a3*x^3) where x = i/(n-1)."""
    x = np.linspace(0, 1, n)
    log_a = a1 * x + a2 * x**2 + a3 * x**3
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_state['seq'] = ans_a
        best_state['score'] = evaluate_safe(ans_a)
        best_state['n'] = 960
        print(f"  Ansatz init: n=960, score={ans_s:.6f}")
    else:
        # Try ansatz at the parent length and nearby
        n_base = best_state['n']
        for alt_n in [n_base - 64, n_base, n_base + 64]:
            if alt_n < 64:
                continue
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            ans_a, ans_s = ansatz_sweep(alt_n)
            real_s = evaluate_safe(ans_a)
            if real_s < best_state['score']:
                best_state['seq'] = ans_a
                best_state['score'] = real_s
                best_state['n'] = alt_n
                print(f"  Ansatz n={alt_n}: NEW BEST {real_s:.6f}")
            else:
                print(f"  Ansatz n={alt_n}: {ans_s:.6f} (not better)")

    n_best = best_state['n']
    print(f"  After Phase 1: best={best_state['score']:.6f}, n={n_best}")

    # Phase 2: p-norm L-BFGS at multiple lengths
    print(f"\n=== Phase 2: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    lengths_to_try = sorted(set([n_best, n_best - 48, n_best + 48,
                                  n_best - 96, n_best + 96,
                                  896, 1024]), key=lambda x: abs(x - n_best))
    lengths_to_try = [x for x in lengths_to_try if 512 <= x <= 1600]

    for trial_n in lengths_to_try[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.20, 300.0)
        if t_alloc < 30:
            break
        print(f"  L-BFGS n={trial_n}, budget={t_alloc:.0f}s")

        # Initialize
        if trial_n == n_best:
            init_a = best_state['seq'].copy()
        else:
            init_a = interpolate_seq(best_state['seq'], trial_n)

        z = np.log(np.maximum(init_a, 1e-300))
        z = z - z.max()

        # p schedule: start moderate, go high
        p_schedule = np.array([20, 50, 100, 200, 400, 800])
        a_out = p_norm_lbfgs(z, t_alloc, p_schedule, max_iter_per_stage=300)
        sc = evaluate_safe(a_out)
        if sc < best_state['score']:
            best_state['seq'] = a_out
            best_state['score'] = sc
            best_state['n'] = trial_n
            print(f"  NEW BEST n={trial_n}: {sc:.6f}")

    # Phase 3: Nesterov LP
    print(f"\n=== Phase 3: Nesterov LP (best={best_state['score']:.6f}, n={best_state['n']}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp(best_state['seq'], t_alloc, solver, max_iters=40000)
        sc_real = evaluate_safe(a_lp)
        if sc_real < best_state['score']:
            best_state['seq'] = a_lp
            best_state['score'] = sc_real
            print(f"  NEW BEST (LP): {sc_real:.6f}")
        else:
            print(f"  LP result: {sc_real:.6f} (not better)")

    # Phase 4: Local peak shaving
    print(f"\n=== Phase 4: Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, sc_ps = local_peak_shave(best_state['seq'], t_alloc, n_steps=80000)
        sc_real = evaluate_safe(a_ps)
        if sc_real < best_state['score']:
            best_state['seq'] = a_ps
            best_state['score'] = sc_real
            print(f"  NEW BEST (peak shave): {sc_real:.6f}")
        else:
            print(f"  Peak shave: {sc_real:.6f} (not better)")

    # Phase 5: Another round of L-BFGS with higher p
    print(f"\n=== Phase 5: High-p L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        z = np.log(np.maximum(best_state['seq'], 1e-300))
        z = z - z.max()
        p_schedule = np.array([200, 500, 1000, 2000])
        a_out = p_norm_lbfgs(z, t_alloc, p_schedule, max_iter_per_stage=400)
        sc = evaluate_safe(a_out)
        if sc < best_state['score']:
            best_state['seq'] = a_out
            best_state['score'] = sc
            print(f"  NEW BEST (high-p): {sc:.6f}")

    # Phase 6: Length fine-tuning
    print(f"\n=== Phase 6: Length Fine-tuning (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = best_state['n']
    fine_lengths = sorted(set([n_cur - 16, n_cur - 32, n_cur + 16, n_cur + 32,
                                n_cur - 8, n_cur + 8]))
    fine_lengths = [x for x in fine_lengths if x >= 512]

    for alt_n in fine_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(rem * 0.08, 60.0)
        if t_sub < 15:
            break
        init_alt = interpolate_seq(best_state['seq'], alt_n)
        z = np.log(np.maximum(init_alt, 1e-300))
        z = z - z.max()
        p_schedule = np.array([100, 300, 600])
        a_alt = p_norm_lbfgs(z, t_sub, p_schedule, max_iter_per_stage=200)
        sc_alt = evaluate_safe(a_alt)
        if sc_alt < best_state['score']:
            best_state['seq'] = a_alt
            best_state['score'] = sc_alt
            best_state['n'] = alt_n
            print(f"  NEW BEST n={alt_n}: {sc_alt:.6f}")

    # Phase 7: Final LP + peak shave
    print(f"\n=== Phase 7: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lp = min(rem * 0.4, 100.0)
        a_lp, _ = nesterov_lp(best_state['seq'], t_lp, solver, max_iters=20000)
        sc = evaluate_safe(a_lp)
        if sc < best_state['score']:
            best_state['seq'] = a_lp
            best_state['score'] = sc
            print(f"  NEW BEST (final LP): {sc:.6f}")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.5, 80.0)
        a_ps, _ = local_peak_shave(best_state['seq'], t_ps, n_steps=50000)
        sc = evaluate_safe(a_ps)
        if sc < best_state['score']:
            best_state['seq'] = a_ps
            best_state['score'] = sc
            print(f"  NEW BEST (final PS): {sc:.6f}")

    # Final L-BFGS squeeze
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.6, 40.0)
        z = np.log(np.maximum(best_state['seq'], 1e-300))
        z = z - z.max()
        a_fin = p_norm_lbfgs(z, t_fin, np.array([500, 1000, 2000]), max_iter_per_stage=300)
        sc = evaluate_safe(a_fin)
        if sc < best_state['score']:
            best_state['seq'] = a_fin
            best_state['score'] = sc
            print(f"  NEW BEST (final LBFGS): {sc:.6f}")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        best_state['score'] = fs
        print(f"\nFINAL: score={fs:.6f}, n={len(best_state['seq'])}")

    return [float(x) for x in best_state['seq']]
