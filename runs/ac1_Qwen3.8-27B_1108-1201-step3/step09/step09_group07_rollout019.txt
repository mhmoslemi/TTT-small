"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Top-k peak targeting gradient descent with adaptive step size
2. LP-based best response with Nesterov averaging
3. Multi-length and multi-start exploration
4. Local coordinate descent polish
Key innovations: FFT-based gradient via reversed-convolution trick,
adaptive active-set sizing, diverse initializations.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_wrt_a(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    Uses the identity: grad[i] = 2 * conv(a_rev, w)[n-1+i]
    where a_rev = a[::-1].
    """
    n = len(a)
    a_rev = a[::-1]
    size = len(a_rev) + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)[:size]
    return 2.0 * conv[n - 1:2 * n - 1]


def topk_gradient(a, k, T_smooth):
    """Compute gradient focusing on top-k peaks with soft weighting."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    if k >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -k)[-k:]
    b_top = b[top_idx]
    weights = np.exp((b_top - b_max) / max(T_smooth, 1e-10))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(len(b))
        weights[int(np.argmax(b))] = 1.0
    else:
        weights = weights / wsum
    w = np.zeros(2 * n - 1)
    w[top_idx] = weights
    grad = grad_wrt_a(a, w)
    return grad, b, b_max


def build_conv_sparse(a):
    """Build sparse matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(size):

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))

    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:

    T = build_conv_sparse(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with restarts."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it * 7 + 3)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def topk_gradient_descent(a0, time_limit, k_start=20, k_end=1, T_start=0.01,
                          T_end=0.001, lr_init=0.02, n_restarts=5):
    """Top-k peak targeting gradient descent with adaptive parameters."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0).copy())
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    total_time = max(time_limit, 1.0)
    step_count = 0
    stale_steps = 0

    for restart in range(n_restarts):
        step_count = 0
        stale_steps = 0
        # Anneal k and T over the restart duration
        restart_time = total_time / n_restarts
        for step in range(1000000):
            if time.monotonic() - t0 > time_limit:
                break
            frac = min(step / max(restart_time * 50, 1), 1.0)
            k_curr = max(1, int(round(k_start * (1 - frac) + k_end * frac)))
            T_curr = T_start * (T_end / T_start) ** frac

            grad, b, b_max = topk_gradient(x, k_curr, T_curr)

            # Adaptive learning rate: use norm of gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            lr = lr_init / (1.0 + 0.01 * step)

            # Take step
            x_new = x - lr * grad
            x_new = project_simplex(x_new)

            # Check score every few steps
            if step % 20 == 0 or step < 10:
                sc = score_normalized(x_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = x_new.copy()
                    stale_steps = 0
                else:
                    stale_steps += 1
                x = x_new
            else:
                x = x_new

            if stale_steps > 200:
                if step < 50:
                    break
                # Small perturbation and continue
                rng = np.random.default_rng(step)
                x = x + 0.01 * rng.standard_normal(n)
                x = project_simplex(x)
                stale_steps = 0

            step_count += 1

        # After each restart, evaluate
        sc = score_normalized(x)
        if sc < best_score:
            best_score = sc
            best_a = x.copy()

        # Check time
        if time.monotonic() - t0 > time_limit:
            break

        # Perturb for next restart
        rng = np.random.default_rng(restart * 1000 + 42)
        x = best_a + 0.02 * rng.standard_normal(n)
        x = project_simplex(x)

    return best_a, best_score


def local_polish(a0, time_limit, n_steps=20000):
    """Local coordinate descent polishing the best sequence."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        # Try reducing top contributors
        for ci in np.argsort(contribs)[::-1][:8]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.94, 0.92, 0.90]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pair reduction
        if not improved:
            for ci in np.argsort(contribs)[::-1][:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.95]:
                    for f2 in [0.97, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init_n(n, mode, rng):
    """Create various initialization patterns."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    elif mode == 1:
        a = 1.0 + 0.7 * np.cos(4 * np.pi * x)
    elif mode == 2:
        a = 1.0 + 0.5 * (2 * x - 1)
    elif mode == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif mode == 4:
        a = 1.0 + 0.3 * np.sin(6 * np.pi * x + 1.0)
    elif mode == 5:
        a = 1.0 + 0.8 * np.cos(2 * np.pi * x) ** 2
    elif mode == 6:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x) * np.cos(4 * np.pi * x)
    else:
        a = 1.0 + rng.standard_normal(n) * 0.3
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate():
    """
    Multi-phase optimization for minimizing the auto-convolution peak score.
    Uses top-k gradient descent, LP refinement, and multi-start exploration.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_init_n(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Top-k gradient descent on parent ===
    print(f"\n=== Phase 1: Top-k GD n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 420.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = topk_gradient_descent(best_seq, t_alloc, k_start=30, k_end=1,
                                         T_start=0.05, T_end=0.001, lr_init=0.03,
                                         n_restarts=4)
        try_update(a_out, "topk_1")

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 340.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_out, "lp_1")

    # === Phase 3: Fine gradient descent ===
    print(f"\n=== Phase 3: Fine GD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 170.0)
    if t_alloc > 20:
        a_out, _ = topk_gradient_descent(best_seq, t_alloc, k_start=5, k_end=1,
                                         T_start=0.01, T_end=0.0005, lr_init=0.01,
                                         n_restarts=3)
        try_update(a_out, "fine_gd")

    # === Phase 4: Local polish ===
    print(f"\n=== Phase 4: Local Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        a_out, _ = local_polish(best_seq, t_alloc, n_steps=50000)
        try_update(a_out, "polish_1")

    # === Phase 5: Multi-start with different inits ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        modes = [1, 3, 4, 6]
        for mode in modes:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.20, 120.0)
            if t_sub < 30:
                break
            a_init = make_init_n(n_best, mode, rng)
            # Blend with current best for warm start
            a_init = 0.5 * a_init + 0.5 * best_seq
            a_init = project_simplex(a_init)
            print(f"  Mode={mode}, budget={t_sub:.0f}s")
            a_out, _ = topk_gradient_descent(a_init, t_sub * 0.5, k_start=20, k_end=1,
                                             T_start=0.03, T_end=0.001, lr_init=0.04,
                                             n_restarts=3)
            try_update(a_out, f"ms_mode{mode}")
            rem = deadline - time.monotonic()
            if rem > 30:
                a_out2, _ = nesterov_lp_refine(a_out, min(rem * 0.25, 40.0),
                                                solver=solver, n_max_iters=20000)
                try_update(a_out2, f"ms_mode{mode}_lp")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for target in [800, 896, 1024, 1152, 1280]:
        if abs(target - n_best) > 50:
            alt_lengths.append(target)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.15, 100.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_out, _ = topk_gradient_descent(init_alt, t_total * 0.5, k_start=20, k_end=1,
                                         T_start=0.03, T_end=0.001, lr_init=0.04,
                                         n_restarts=3)
        try_update(a_out, f"alt_n{alt_n}")
        rem = deadline - time.monotonic()

                                            solver=solver, n_max_iters=10000)
            try_update(a_out2, f"alt_n{alt_n}_lp")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 60.0)
        a_out, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=50000)
        try_update(a_out, "final_lp")

    # === Phase 8: Final local polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.15, 30.0)
        a_out, _ = local_polish(best_seq, t_ps, n_steps=20000)
        try_update(a_out, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
