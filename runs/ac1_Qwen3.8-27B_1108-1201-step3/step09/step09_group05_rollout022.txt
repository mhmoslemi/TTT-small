import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def evaluate_safe(a):
    """Evaluate sequence safely using the external evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective(a, T):
    """Log-sum-exp smoothed objective: (2n/T) * log(sum(exp(T*b)))."""
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * len(a) / T) * lse


def lse_gradient(a, T):
    """Gradient of log-sum-exp smoothed objective w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    return grad_a


def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.05, maxiter=8000, n_stages=10):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    def make_func(T_val):
        def func(z_in):
            a = softmax_from_z(z_in)
            return lse_objective(a, T_val)
        return func

    def make_jac(T_val):
        def jac(z_in):
            a = softmax_from_z(z_in)
            grad_a = lse_gradient(a, T_val)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        except Exception:
            pass

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, max_iters=200000):
    """Nesterov-accelerated projected subgradient with Polyak step size."""
    n = len(a0)
    a = norm_sequence(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Polyak step: eta_t = (f(a_t) - f*) / ||g_t||^2
    # We don't know f*, so use a running estimate
    f_star_est = best_score * 0.999  # optimistic estimate

    beta = 0.0  # momentum parameter, will ramp up
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta = min(0.4, 0.1 * (1 + np.log1p(it)))
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Compute subgradient at y
        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = y[k_star - idx]

        # Polyak step size
        f_y = score_normalized(y)
        gap = f_y - f_star_est
        if gap < 1e-12:
            gap = 1e-12
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        eta = gap / g_norm_sq
        eta = min(eta, 0.5)  # cap step size

        # Gradient step + projection
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = score_normalized(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            f_star_est = best_score * (1 - 0.001 / (it + 1))

        a_prev = a
        a = a_new

    return best_a, best_score


def coordinate_descent_peak(a0, time_limit, max_steps=50000):
    """Peak-specific coordinate descent: reduce top contributors to the peak."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Get top peaks (exclude edges)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[1] = -np.inf
            b_work[-2] = -np.inf
            b_work[-1] = -np.inf
        peak_idx = np.argsort(b_work)[::-1][:5]

        improved = False
        for pk in peak_idx:
            pk = int(pk)
            if b[pk] < best_score / (2.0 * n) * 1.001:
                break
            # Find contributors to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]

            # Try reducing top contributors
            top_c = np.argsort(contribs)[::-1][:10]
            found_improve = False
            for ci in top_c:
                i = int(idx[ci])
                j = pk - i
                if i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-15:
                    continue
                # Try reducing a[i] (and a[j] if j != i)
                for frac in [0.98, 0.95, 0.92, 0.88, 0.83]:
                    a_trial = a.copy()
                    a_trial[i] *= frac
                    if j != i and j < n:
                        a_trial[j] *= frac
                    # Redistribute to low-contribution positions
                    diff = a[i] - a_trial[i] + (a[j] - a_trial[j] if j != i else 0)
                    if diff < 1e-16:
                        continue
                    # Add to positions far from peak
                    far_mask = np.zeros(n, dtype=bool)
                    for ii in range(n):
                        # Check if position ii contributes to any top peak
                        contributes = False
                        for pp in peak_idx[:3]:
                            pp = int(pp)
                            jj = pp - ii
                            if 0 <= jj < n and a[ii] > 1e-10 and a[jj] > 1e-10:
                                if a[ii] * a[jj] > 0.3 * contribs.max():
                                    contributes = True
                                    break
                        if not contributes:
                            far_mask[ii] = True
                    far_mask[i] = False
                    if j != i and j < n:
                        far_mask[j] = False
                    if far_mask.sum() > 0:
                        far_vals = a_trial[far_mask] + 1e-12
                        a_trial[far_mask] = far_vals + diff * far_vals / far_vals.sum()
                    a_trial = norm_sequence(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        found_improve = True
                        improved = True
                        break
                if found_improve:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

    return best_a, best_score


def parametric_init(n, mode=0, rng=None):
    """Create parametric window initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)

    if mode == 0:
        # Raised cosine squared
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif mode == 1:
        # Generalized Gaussian
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif mode == 2:
        # Polynomial window (Blackman-like)
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif mode == 3:
        # Asymmetric: heavy on one side
        a = np.exp(-3 * x) * (1 + 0.5 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif mode == 4:
        # Asymmetric Gaussian
        a = np.exp(-((x - 0.4) ** 2) / (2 * 0.2 ** 2))
    elif mode == 5:
        # Flat-top with cosine taper
        t = 0.2
        a = np.ones(n)
        mask_l = x < t
        mask_r = x > 1 - t
        a[mask_l] = 0.5 * (1 + np.cos(np.pi * x[mask_l] / t))
        a[mask_r] = 0.5 * (1 + np.cos(np.pi * (1 - x[mask_r]) / t))
    elif mode == 6:
        # Asymmetric flat-top
        t = 0.25
        a = np.ones(n)
        mask_l = x < t
        mask_r = x > 1 - t * 0.5
        a[mask_l] = 0.5 * (1 + np.cos(np.pi * x[mask_l] / t))
        a[mask_r] = 0.5 * (1 + np.cos(np.pi * (1 - x[mask_r]) / (t * 0.5)))
    elif mode == 7:
        # Power-law taper
        p = 1.5
        a = np.cos(np.pi * (x - 0.5)) ** (2 * p)
    elif mode == 8:
        # Bump function
        a = np.exp(-1 / (x * (1 - x) + 1e-10))
    else:
        # Random smooth
        k = rng.integers(2, 8)
        coeffs = rng.standard_normal(k)
        a = sum(c * np.cos(2 * np.pi * m * x) for m, c in enumerate(coeffs))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def basin_hopping(a0, n, time_budget, rng, n_restarts=5):
    """Basin hopping: perturb and re-optimize."""
    best_a = norm_sequence(a0)
    best_score = score_normalized(best_a)
    t0 = time.monotonic()

    for r in range(n_restarts):
        rem = time_budget - (time.monotonic() - t0)
        if rem < 30:
            break
        t_sub = min(rem / (n_restarts - r), 250.0)

        # Adaptive perturbation scale
        scale = 0.05 + 0.1 * r / max(1, n_restarts - 1)
        perturbed = best_a + scale * rng.standard_normal(n)
        perturbed = norm_sequence(perturbed)

        a_new, s_new = full_optimize(perturbed, t_sub, n)
        if s_new < best_score:
            best_a, best_score = a_new, s_new
            print(f"    Basin hop {r}: improved to {best_score:.8f}")

    return best_a, best_score

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = parametric_init(n_init, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting optimization. Best so far: {best_score:.6f}, n={n_parent}")

    # === Phase 1: Optimize parent at its length ===
    print(f"\n=== Phase 1: Optimize parent n={n_parent} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 350.0)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a1, s1 = full_optimize(best_seq, t1, n_parent)
        try_update(a1, "parent_opt")

    # === Phase 2: Basin hopping on parent length ===
    print(f"\n=== Phase 2: Basin hopping n={n_parent} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = basin_hopping(best_seq, n_parent, t2, rng, n_restarts=4)
        try_update(a2, "basin_hop")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    # Try lengths around the parent and some other interesting values
    candidate_lengths = set()
    for mult in [0.75, 0.85, 0.9, 1.1, 1.15, 1.25, 1.4, 1.5, 2.0]:
        cl = int(n_parent * mult)
        if 384 <= cl <= 4096:
            candidate_lengths.add(cl)
    # Add some "round" numbers
    for fixed_n in [512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536, 2048]:
        if 384 <= fixed_n <= 4096:
            candidate_lengths.add(fixed_n)
    candidate_lengths.discard(n_parent)
    candidate_lengths = sorted(candidate_lengths)

    # Sort by proximity to parent length (try nearby first)
    candidate_lengths.sort(key=lambda x: abs(x - n_parent))

    for alt_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.18, 200.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        # Multiple initializations for this length
        best_alt_a = None
        best_alt_s = float('inf')

        inits = [
            interpolate_sequence(best_seq, alt_n),
            parametric_init(alt_n, mode=rng.integers(0, 9), rng=rng),
            parametric_init(alt_n, mode=rng.integers(0, 9), rng=rng),
        ]

        n_inits = len(inits)
        per_init_budget = t_total / n_inits

        for ii, init_a in enumerate(inits):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(per_init_budget, rem * 0.8)
            if t_sub < 10:
                break
            a_alt, s_alt = full_optimize(init_a, t_sub, alt_n)
            if s_alt < best_alt_s:
                best_alt_s = s_alt
                best_alt_a = a_alt

        try_update(best_alt_a, f"multilen_n{alt_n}")

    # === Phase 4: Fine-tune the best found ===
    print(f"\n=== Phase 4: Fine-tune best (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        n_cur = len(best_seq)
        t4 = min(rem * 0.3, 200.0)
        print(f"  n={n_cur}, budget={t4:.0f}s")
        a4, s4 = full_optimize(best_seq, t4, n_cur)
        try_update(a4, "finetune")

    # === Phase 5: Final coordinate descent + subgradient polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        n_cur = len(best_seq)
        # Coordinate descent
        t_cd = min(rem * 0.3, 60.0)
        if t_cd > 5:
            a_cd, s_cd = coordinate_descent_peak(best_seq, t_cd, max_steps=20000)
            try_update(a_cd, "coord_desc")

        # Nesterov subgradient
        rem = deadline - time.monotonic()
        t_ns = min(rem * 0.3, 50.0)
        if t_ns > 5:
            a_ns, s_ns = nesterov_subgradient(best_seq, t_ns, max_iters=50000)
            try_update(a_ns, "nesterov_final")

        # Very low temperature L-BFGS
        rem = deadline - time.monotonic()
        t_lb = min(rem * 0.3, 50.0)
        if t_lb > 5:
            a_lb, s_lb = lbgfs_refine(best_seq, t_lb, T_start=2.0, T_end=0.01, maxiter=4000, n_stages=8)
            try_update(a_lb, "lbgfs_ultimatelow")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
