import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_T(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP method with ergodic averaging for global optimization
2. Multi-peak subgradient descent with adaptive steps for refinement
3. Fast coordinate descent with O(n) sensitivity for local polishing
4. LP polishing for final tightening

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    import cvxpy as cp
    for sname in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sname)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return sname
        except:
            continue
    return cp.ECOS


def alternating_lp(a_init, time_limit, verbose=True, label=""):
    import cvxpy as cp
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    solver = find_lp_solver()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    avg_a = np.zeros(n)
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1

        T_a = build_T(a)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        prob = cp.Problem(cp.Minimize(M_var), [T_a @ b_var <= M_var, cp.sum(b_var) == 1])
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        sb = b_val.sum()
        if sb < 1e-15:
            break
        b_val /= sb
        avg_a += b_val / it

        sc = 2.0 * n * fast_convolve(b_val).max()
        if sc < best_score:
            best_score = sc
            best_a = b_val.copy()
        sc_avg = 2.0 * n * fast_convolve(avg_a).max()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        T_b = build_T(b_val)
        a_var = cp.Variable(n, nonneg=True)
        M_var2 = cp.Variable()
        prob2 = cp.Problem(cp.Minimize(M_var2), [T_b @ a_var <= M_var2, cp.sum(a_var) == 1])
        try:
            prob2.solve(solver=solver, verbose=False)
        except:
            break
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            break
        a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0)
        sa = a_val.sum()
        if sa < 1e-15:
            break
        a_val /= sa
        avg_a += a_val / (it + 0.5)

        sc = 2.0 * n * fast_convolve(a_val).max()
        if sc < best_score:
            best_score = sc
            best_a = a_val.copy()
        sc_avg = 2.0 * n * fast_convolve(avg_a).max()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        a = a_val

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALP[{label}] it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    sc_avg = 2.0 * n * fast_convolve(avg_a).max()
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = avg_a.copy()

    return best_a, best_score


def subgradient_multi_peak(a0, time_limit, lr_init=0.02, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-9:
                break

        # Find all peaks above 0.999*M
        threshold = 0.999 * M
        peak_mask = conv >= threshold
        peaks = np.where(peak_mask)[0]

        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(conv))])
        elif len(peaks) > 50:
            peak_vals = conv[peaks]
            top_idx = np.argsort(peak_vals)[::-1][:50]
            peaks = peaks[top_idx]

        # Weight by peak value (softmax)
        pv = conv[peaks]
        temp = 0.001 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Multiplicative weight update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        avg_a = (it * avg_a + a_new) / (it + 1)
        a = a_new
        conv = fast_convolve(a)

        if it % 200 == 0:
            sc_avg = 2.0 * n * fast_convolve(avg_a).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG[{label}] it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    sc_avg = 2.0 * n * fast_convolve(avg_a).max()
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = avg_a.copy()

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    factors = [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            best_delta = 0.0
            best_new_M = M

            for factor in factors:
                delta = a[i] * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def cosinit(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.8, 2.5)
    a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_hump_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.25, 0.4)
    c2 = n * rng.uniform(0.6, 0.75)
    w = n * rng.uniform(0.1, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def periodic_init(n, rng):
    period = int(rng.integers(2, 8))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    if remaining > 60:
        t_cd = min(remaining * 0.5, 250.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_cd, verbose=True, label="main")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_main")

    # === Phase 4: LP polishing on best ===
    print(f"\n=== Phase 4: LP Polishing (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 40:
        t_lp = min(remaining * 0.5, 150.0)
        print(f"  LP budget={t_lp:.0f}s")
        a_lp, s_lp = alternating_lp(best_seq, t_lp, verbose=True, label="polish")
        best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "lp_polish")

    # === Phase 5: Fine coordinate descent ===
    print(f"\n=== Phase 5: Fine CD (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 100
    if remaining > 30:
        t_fine = min(remaining * 0.6, 100.0)
        print(f"  FineCD budget={t_fine:.0f}s")
        a_fine, s_fine = fine_cd(best_seq, t_fine, verbose=True, label="main")
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 50
    if remaining > 60:
        for pidx in range(3):
            if deadline - time.monotonic() < 40:
                break
            noise_scale = 0.01 * (pidx + 1)
            perturbed = best_seq + rng.standard_normal(n_best) * noise_scale
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_budget = min(remaining * 0.25, 80.0)
            a_p, s_p = coord_descent(perturbed, t_budget, verbose=False, label=f"pert{pidx}")
            best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"pert_{pidx}")

    # === Phase 7: Final fine polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        t_fin = min(remaining * 0.8, 60.0)
        a_fin, s_fin = fine_cd(best_seq, t_fin, verbose=False, label="final")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
