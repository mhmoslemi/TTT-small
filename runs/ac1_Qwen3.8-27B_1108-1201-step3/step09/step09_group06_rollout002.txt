    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_exact_gradient_smoothed(a, T_beta):
    n = len(a)
    b = fast_convolve(a)
    Tb = T_beta * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    return grad, b.max()


def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_dual_lower_bound(a_current, solver=None):
    """Solve the dual LP to get a lower bound on min max(conv(a,a))."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_current)
    y_var = cp.Variable(2 * n - 1, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(y_var) == 1]
    # (T^T y)_j >= t for all j
    # (T^T y)_j = sum_k T[k,j] * y[k]
    # T[k,j] = 1 iff 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    # So (T^T y)_j = sum_{k=max(0,j)}^{min(2n-2, j+n-1)} y[k]
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 2, j + n - 1)
        constraints.append(cp.sum(y_var[lo:hi + 1]) >= t_var)
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return float(t_var.value)


def lp_nesterov_refine(a0, time_limit, solver=None, n_max_iters=500000):
    """LP best-response with Nesterov acceleration, dual-gap monitoring, and smart restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    lb = best_score / (2.0 * n)  # initial lower bound estimate (0)
    dual_check_interval = 5000

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Nesterov x update
        if it == 0:
            a = b_new.copy()
        else:
            a = ((it + 1) * a + (alpha * (it + 2.0)) * b_new) / (it + 1 + alpha * (it + 2.0))
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
            else:
                a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Periodic logging and stalling check
        if (it + 1) % 500 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score

            # Dual check for gap estimation (less frequent)
            if (it + 1) % dual_check_interval == 0:
                lb_val = solve_lp_dual_lower_bound(a, solver)
                if lb_val is not None:
                    lb = lb_val
                    gap = best_score / (2.0 * n) - lb
                    print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.7f}, LB={lb:.7f}, gap={gap:.7f}")
                else:
                    print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.7f}")

            # Stall recovery: perturb and restart Nesterov
            if stall_count >= 12:
                rng = np.random.default_rng(it * 31 + 17)
                noise_scale = 0.01 * (1.0 + stall_count * 0.1)
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                a = y.copy()
                stall_count = 0
                print(f"    LP restart at iter={it+1}, noise_scale={noise_scale:.4f}")

            # Early stop if gap is tiny
            if (it + 1) % dual_check_interval == 0:
                lb_val = solve_lp_dual_lower_bound(a, solver)
                if lb_val is not None and (best_score / (2.0 * n) - lb_val) < 1e-8:
                    print(f"    LP converged (gap < 1e-8) at iter={it+1}")
                    break

    return best_a, best_score


def mirror_descent_fast(a0, time_limit, n_active_frac=0.08):
    """Fast mirror descent with Polyak step sizing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    t0 = time.monotonic()
    lr = 1.0
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_b = compute_subgradient(a, n_active_frac)
        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        score_now = 2.0 * n * max_b
        polyak_lr = max(0.01, (score_now - best_score + 1e-6) / (2.0 * g_norm_inf * score_now + 1e-15))
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 5.0)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 100000 == 0 and step > 0:
            print(f"    MirrorDesc {time.monotonic()-t0:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def lbgfs_multistage(a0, time_limit, T_start=500.0, T_end=0.05, n_stages=12):
    """Multi-stage L-BFGS on LogSumExp smoothed objective with aggressive annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    total_iters = 60000
    iters_per_stage = max(20, total_iters // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_T(z_in):
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb = T_val * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def jac_T(z_in):
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb = T_val * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_local)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_local[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)

        T_val = stage_T
        # Define proper closures via top-level helper
        func = lambda z_in: _lbgfs_fun(z_in, n, T_val)
        jac = lambda z_in: _lbgfs_jac(z_in, n, T_val)

        try:
            res = minimize(
                fun=_lbgfs_fun,
                x0=z,
                jac=_lbgfs_jac,
                args=(n, T_val),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx} error: {e}")

        a_curr = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_curr).max()
        if stage_idx % 3 == 0:
            print(f"    L-BFGS stage {stage_idx}, T={T_val:.2f} ({time.monotonic()-t0:.0f}s): score={sc:.7f}")

    return softmax_from_z(z)


def _lbgfs_fun(z_in, n, T_val):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def _lbgfs_jac(z_in, n, T_val):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def local_peak_search(a0, time_limit, n_steps=40000):
    """Local search focused on reducing peak contributions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 8000:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        # Compute contribution of each a[i] to the peaks
        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Try moving mass from high-contrib to low-contrib positions
        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        for si in top_src:
            if a[si] < 1e-12:
                continue
            for di in low_dst:
                if si == di:
                    continue
                for frac in [0.01, 0.02, 0.04, 0.08]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Random perturbation
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                amount = a[i] * rng.uniform(0.01, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x)) * (0.5 + 0.5 * np.cos(4 * np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * x * (1 - x))
    a = np.maximum(a, 0)
    sv = a.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    return a / sv


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def gradient_flow_refine(a0, time_limit, n_steps=50000):
    """Gradient flow using exact gradient with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        g, max_b = compute_subgradient(a, 0.05)
        g = g - np.mean(g)  # project to sum-zero
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            a_trial = a - alpha * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.1, 2.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr *= 0.9

    return best_a, best_score

        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_nesterov_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "lp_nesterov_1")

    # === Phase 3: L-BFGS LogSumExp (multi-stage) ===
    print(f"\n=== Phase 3: L-BFGS Multi-stage (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 500->0.05, 12 stages")
        a_lb = lbgfs_multistage(best_seq, t_alloc, T_start=500.0, T_end=0.05, n_stages=12)
        try_update(a_lb, "lbgfs_1")

    # === Phase 4: Second LP round (longer, exploiting improved starting point) ===
    print(f"\n=== Phase 4: LP Nesterov r2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 20:
        a_lp2, _ = lp_nesterov_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp2, "lp_nesterov_2")

    # === Phase 5: Gradient flow with line search ===
    print(f"\n=== Phase 5: Gradient Flow (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_gf, _ = gradient_flow_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_gf, "grad_flow")

    # === Phase 6: Local peak search ===
    print(f"\n=== Phase 6: Local Peak Search (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_ls, _ = local_peak_search(best_seq, t_alloc, n_steps=60000)
        try_update(a_ls, "local_peak")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.7f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0, 1.1]:
        alt_n = int(n_base * mult)
        if 384 <= alt_n <= 3500 and alt_n != n_base:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.35, 200.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolated n={alt_n}: score={sc_alt:.7f}")

        t_lp = t_total * 0.6
        a_alt, _ = lp_nesterov_refine(init_alt, t_lp, solver=solver, n_max_iters=200000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 10:
                a_alt2 = lbgfs_multistage(best_seq, t_lb, T_start=100.0, T_end=0.1, n_stages=6)
                try_update(a_alt2, f"lbgfs_n")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.3, 60.0)
        a_fin = lbgfs_multistage(best_seq, t_fin, T_start=20.0, T_end=0.05, n_stages=6)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_ls = min(rem * 0.3, 40.0)
        a_ls2 = local_peak_search(best_seq, t_ls, n_steps=30000)
        try_update(a_ls2, "final_local")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.4, 50.0)
        a_lp_fin, _ = lp_nesterov_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=100000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
