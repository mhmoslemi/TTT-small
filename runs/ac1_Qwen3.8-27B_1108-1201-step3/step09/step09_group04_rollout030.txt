import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the score 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2 * len(a) * c.max() / (s * s))

def max_subgradient(a):
    """Compute the (sub)gradient of max(conv(a,a)) w.r.t. a.
    Returns gradient vector of length n."""
    c = fast_convolve(a)
    n = len(a)
    # Find all positions where c achieves its maximum
    max_val = c.max()
    # Get all indices where c is at max (within numerical tolerance)
    threshold = max_val - 1e-12 * max_val

    grad = np.zeros(n)
    for i in range(n):
        for k in S:
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    return grad

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-start with diverse asymmetric initializations and length exploration
2. Direct max-norm subgradient with multiplicative weight updates
3. LP-based alternating optimization (Best Response LP)
4. High-p L-BFGS polishing in log-space
5. Peak-targeted coordinate descent for final refinement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2 * len(a) * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def max_subgradient(a):
    """Subgradient of max(conv(a,a)) w.r.t. a, using top peaks."""
    c = fast_convolve(a)
    n = len(a)
    max_val = c.max()
    # Use top few peaks weighted by proximity to max
    sorted_idx = np.argsort(c)[::-1]
    top_k = min(10, 2*n - 1)
    peaks = sorted_idx[:top_k]
    # Exponential weighting
    diffs = max_val - c[peaks]
    scale = max_val if max_val > 1e-15 else 1.0
    weights = np.exp(-20.0 * diffs / scale)
    if weights.sum() < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights /= weights.sum()
    
    I_S = np.zeros(2*n - 1)
    for j, k in enumerate(peaks):
        I_S[k] = weights[j]
    
    grad = np.correlate(I_S, a, 'valid')
    return grad

        grad_z = a * (grad - mean_grad)

def p_norm_obj_z(z, p):
    """2n * (sum max(c,0)^p)^(1/p) in log-space."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val

def p_norm_grad_z(z, p):
    """Gradient of p-norm objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0/p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, 'valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))

def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=6):
    """Multi-stage p-norm L-BFGS."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    
    for i in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        p_val = p_start * (p_end / p_start) ** (i / max(1, n_stages - 1))
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Time-based iteration limit
        max_iter = max(50, int(remaining * 50))  # ~50 iters per second budget
        
        try:
            res = minimize(
                fun=lambda z_val: p_norm_obj_z(z_val, p_val),
                x0=z,
                jac=lambda z_val: p_norm_grad_z(z_val, p_val),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm L-BFGS p={p_val:.1f} ({elapsed:.0f}s): score={sc:.6f}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def build_conv_matrix(a):
    """Build sparse-like conv matrix T where T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, max_time=30.0):
    """LP: min M s.t. conv(a_fixed, b) <= M, b >= 0, sum(b) = 1."""
    import cvxpy as cp
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    t0 = time.monotonic()
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if time.monotonic() - t0 > max_time:
            break
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                return b_val / s
        except Exception:
            continue
    return None

def lp_alternating(a0, time_limit, n_max_iters=200):

        remaining = time_limit - (time.monotonic() - t0)

            break

        b_new = solve_lp_best_response(a, max_time=min(30, remaining * 0.3))
        if b_new is None:
            continue
        
        # Update a (weighted average)
        alpha = 2.0 / (it + 2)
        a_new = (1 - alpha) * a + alpha * b_new

        if score < best_score - 1e-12:

def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:  # Sine
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:  # Cosine (ramp-like, asymmetric)
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:  # Exponential decay
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x)
    elif variant == 3:  # Ramped sine
        x = np.linspace(0, 1, n)
        a = x * np.sin(np.pi * x)
    elif variant == 4:  # Quadratic ramp
        x = np.linspace(0, 1, n)
        a = x * (1 - x) ** 0.5
    elif variant == 5:  # Inverse ramp
        x = np.linspace(0, 1, n)
        a = (1 - x) * np.sin(np.pi * x)
    elif variant == 6:  # Asymmetric gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 2) * (1 + 0.5 * x)
    elif variant == 7:  # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-start with diverse asymmetric initializations
2. Direct max-norm subgradient with multiplicative weight updates and momentum
3. Adaptive p-norm L-BFGS in log-space
4. LP-based best-response alternating optimization
5. Peak-targeted coordinate descent for final refinement
6. Length exploration with interpolated initializations
Key insight: asymmetry in the sequence allows the self-convolution to be
flatter than what is possible with symmetric (or uniform) sequences.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2 * len(a) * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def max_subgradient(a):
    """Approximate subgradient of max(conv(a,a)) w.r.t. a using weighted top peaks."""
    c = fast_convolve(a)
    n = len(a)
    max_val = c.max()
    sorted_idx = np.argsort(c)[::-1]
    top_k = min(10, 2 * n - 1)
    peaks = sorted_idx[:top_k]
    diffs = max_val - c[peaks]
    scale = max_val if max_val > 1e-15 else 1.0
    weights = np.exp(-15.0 * diffs / scale)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights /= wsum
    I_S = np.zeros(2 * n - 1)
    for j in range(top_k):
        I_S[peaks[j]] = weights[j]
    grad = np.correlate(I_S, a, 'valid')
    return grad


def multiplicative_descent(a0, time_limit, lr0=0.3, decay=0.99995, momentum=0.7):
    """Multiplicative weight update with max-subgradient and momentum in log-space."""
    n = len(a0)
    a = normalize(a0.copy())
    z = np.log(np.maximum(a, 1e-300))
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr0
    stall = 0

        if time.monotonic() - t0 > time_limit:
            break
        grad = max_subgradient(a)
        mean_grad = np.dot(a, grad)
        grad_z = a * (grad - mean_grad)
        v = momentum * v + (1.0 - momentum) * grad_z
        z = z - lr * v
        z = z - z.max()
        z = np.clip(z, -50, 50)
        a_new = np.exp(z)
        a_new = normalize(a_new)

def p_norm_obj(z, p):
    """Objective: 2n * (sum max(c,0)^p)^(1/p) computed in log-space."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_grad(z, p):
    """Gradient of p-norm objective w.r.t. log-space z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, 'valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=6):
    """Multi-stage p-norm L-BFGS in log-space with increasing p."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    for i in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        if i < n_stages - 1:
            p_val = p_start * (p_end / p_start) ** (i / (n_stages - 1))
        else:
            p_val = p_end
        max_iter = max(30, int(remaining * 40))

        try:
            res = minimize(
                fun=lambda z_val: p_norm_obj(z_val, p_val),
                x0=z,
                jac=lambda z_val: p_norm_grad(z_val, p_val),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm p={p_val:.1f} ({elapsed:.0f}s): score={sc:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, time_budget=20.0):
    """LP: min M s.t. conv(a_fixed, b) <= M, b >= 0, sum(b) = 1."""
    import cvxpy as cp
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    t0 = time.monotonic()
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        if time.monotonic() - t0 > time_budget:
            break
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                return b_val / s
        except Exception:
            continue
    return None

        a_new = normalize(a_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if (it + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=50000):
    """Coordinate descent reducing contributions to the peak of the convolution."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        peak_k = c.argmax()
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:12]
        improved = False

        for ci in top:
            i = idx[ci]
            for factor in [0.97, 0.94, 0.90, 0.85]:
                if a[i] < 1e-14:
                    break
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - factor)
                if diff < 1e-15:
                    continue
                a_trial[i] -= diff
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] += diff * others / os
                a_trial = normalize(a_trial)
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            for ci in top[:6]:
                i = idx[ci]
                j = peak_k - i
                if j < 0 or j >= n or j == i:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.96, 0.92, 0.88]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = normalize(a_trial)
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 2000 and step % 1000 == 0:
                rng = np.random.default_rng(step)
                k_pert = int(rng.integers(2, 8))
                indices = rng.choice(n, size=k_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.1, 0.1))
                a_trial = normalize(a_trial)
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    if no_improve > 5000:
                        break
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDescent step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, variant=0):
    """Create diverse structured initializations with asymmetry."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = x * np.sin(np.pi * x)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x * (1.0 - x) ** 0.5
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = (1.0 - x) * np.sin(np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.0) * (1.0 + 0.5 * x)
    elif variant == 7:
        x = np.linspace(0.01, 1, n)
        a = x ** 0.3
    elif variant == 8:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (0.3 + 0.7 * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update_best(best_score_ref, a, label):
    """Try to update best. Returns (updated, new_score, new_seq)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return False, float('inf'), None
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score_ref:
        return True, sc, a_norm.copy()
    return False, sc, None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Combines multiplicative descent, p-norm L-BFGS, LP alternating,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=0)
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Multi-start p-norm with diverse initializations ===
    print(f"\n=== Phase 1: Multi-start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p1 = min(rem * 0.14, 250.0)
    if t_p1 > 30:
        print(f"  Budget: {t_p1:.0f}s")
        inits = [
            (0, 'sine'),
            (1, 'ramp'),
            (3, 'ramped_sine'),
            (8, 'asym_sine'),
        ]
        t_per = t_p1 / (len(inits) + 1)
        # First: optimize parent with quick p-norm
        if t_per > 15:
            a_opt = p_norm_lbfgs(best_seq, t_per, p_start=10.0, p_end=150.0, n_stages=4)
            upd, sc, seq = try_update_best(best_score, a_opt, "parent_quick")
            if upd:
                best_score = sc
                best_seq = seq
        for var, name in inits:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(t_per, rem * 0.9)
            if t_sub < 10:
                break
            init_a = structured_init(n_best, var)
            a_opt = p_norm_lbfgs(init_a, t_sub, p_start=10.0, p_end=200.0, n_stages=4)
            upd, sc, seq = try_update_best(best_score, a_opt, f"init_{name}")
            if upd:
                best_score = sc
                best_seq = seq

    # === Phase 2: Multiplicative Descent ===
    print(f"\n=== Phase 2: Multiplicative Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.14, 250.0)
    if t_p2 > 20:
        print(f"  Budget: {t_p2:.0f}s")
        a_md, _ = multiplicative_descent(best_seq, t_p2, lr0=0.3, decay=0.99995, momentum=0.7)
        upd, sc, seq = try_update_best(best_score, a_md, "mult_desc")
        if upd:
            best_score = sc
            best_seq = seq

    # === Phase 3: Main p-norm L-BFGS ===
    print(f"\n=== Phase 3: p-norm L-BFGS main (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.24, 430.0)
    if t_p3 > 30:
        print(f"  Budget: {t_p3:.0f}s, p: 5 -> 500")
        a_pn = p_norm_lbfgs(best_seq, t_p3, p_start=5.0, p_end=500.0, n_stages=7)
        upd, sc, seq = try_update_best(best_score, a_pn, "pnorm_main")
        if upd:
            best_score = sc
            best_seq = seq

    # === Phase 4: LP Alternating ===
    print(f"\n=== Phase 4: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p4 = min(rem * 0.16, 290.0)
    if t_p4 > 30:
        print(f"  Budget: {t_p4:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_p4, n_max_iters=80)
        upd, sc, seq = try_update_best(best_score, a_lp, "lp_alt")
        if upd:
            best_score = sc
            best_seq = seq

    # === Phase 5: Length Exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p5 = min(rem * 0.14, 250.0)
    if t_p5 > 30:
        print(f"  Budget: {t_p5:.0f}s")
        n_cur = len(best_seq)
        candidate_lengths = []
        for delta in [-128, -64, -32, 32, 64, 96, 128, 192]:
            alt_n = n_cur + delta
            if 512 <= alt_n <= 2048:
                candidate_lengths.append(alt_n)
        for rn in [768, 832, 896, 928, 960, 992, 1024, 1056, 1088, 1152, 1216, 1280]:
            if 512 <= rn <= 2048 and rn not in candidate_lengths:
                candidate_lengths.append(rn)
        candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_cur))
        n_explored = 0
        for alt_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 3:
                break
            n_explored += 1
            t_total = min(rem * 0.3, 150.0)
            if t_total < 30:

            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
            init_alt = interpolate_seq(best_seq, alt_n)
            t_pn = t_total * 0.6
            a_alt = p_norm_lbfgs(init_alt, t_pn, p_start=10.0, p_end=300.0, n_stages=5)
            upd, sc, seq = try_update_best(best_score, a_alt, f"len_{alt_n}")
            if upd:
                best_score = sc
                best_seq = seq
                rem = deadline - time.monotonic()
                if rem > 30:
                    t_lp = min(rem * 0.2, 80.0)
                    a_alt2, _ = lp_alternating(best_seq, t_lp, n_max_iters=25)
                    upd2, sc2, seq2 = try_update_best(best_score, a_alt2, f"lp_{alt_n}")
                    if upd2:
                        best_score = sc2
                        best_seq = seq2

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p6 = min(rem * 0.12, 215.0)
    if t_p6 > 15:
        print(f"  Budget: {t_p6:.0f}s")
        a_cd, _ = peak_coordinate_descent(best_seq, t_p6, n_steps=80000)
        upd, sc, seq = try_update_best(best_score, a_cd, "coord_descent")
        if upd:
            best_score = sc
            best_seq = seq

    # === Phase 7: Second MD round (from improved point) ===
    print(f"\n=== Phase 7: MD Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p7 = min(rem * 0.08, 140.0)
    if t_p7 > 15:
        print(f"  Budget: {t_p7:.0f}s")
        a_md2, _ = multiplicative_descent(best_seq, t_p7, lr0=0.15, decay=0.99998, momentum=0.8)
        upd, sc, seq = try_update_best(best_score, a_md2, "md_round2")
        if upd:
            best_score = sc
            best_seq = seq

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.30, 70.0)
        a_fin = p_norm_lbfgs(best_seq, t_fin, p_start=200.0, p_end=1500.0, n_stages=4)
        upd, sc, seq = try_update_best(best_score, a_fin, "final_pnorm")
        if upd:

    rem = deadline - time.monotonic()
    if rem > 15:

        upd, sc, seq = try_update_best(best_score, a_cd2, "final_cd")
        if upd:
            best_score = sc
            best_seq = seq

    rem = deadline - time.monotonic()
    if rem > 10:
        t_md = min(rem * 0.4, 30.0)
        a_md3, _ = multiplicative_descent(best_seq, t_md, lr0=0.05, decay=0.99999, momentum=0.85)
        upd, sc, seq = try_update_best(best_score, a_md3, "final_md")
        if upd:
            best_score = sc
            best_seq = seq

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
