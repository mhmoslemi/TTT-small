"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based von Neumann algorithm
with Nesterov acceleration, targeted peak reduction, and multi-length exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C such that C @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def get_lp_solver():
    """Find the best available LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Given fixed a (sum=1, nonneg), find b (sum=1, nonneg) 
    minimizing max_k conv(a, b)[k].
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    C = build_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [C @ b_var <= M_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, eps_abs=1e-9, eps_rel=1e-9, max_iter=100000)
    except Exception:
        try:
            prob.solve(solver=solver)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternative solvers
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def vn_nesterov(a0, time_limit, solver, max_iters=20000):
    """
    VN (von Neumann) algorithm with Nesterov acceleration.
    Alternates between computing LP best responses for a and b.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    b = a.copy()  # Initialize b = a
    y_a = a.copy()  # Nesterov extrapolated a
    y_b = b.copy()  # Nesterov extrapolated b
    
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    
    # Running averages (these converge to the optimal solution)
    a_avg = a.copy()
    b_avg = b.copy()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov interpolation
        alpha = 2.0 / (it + 2)
        y_a = (1 - alpha) * y_a + alpha * b_avg
        y_a = project_simplex(y_a)
        y_b = (1 - alpha) * y_b + alpha * a_avg
        y_b = project_simplex(y_b)
        
        # Solve LP for best response to y_a (finding optimal b)
        b_new, M_b = solve_lp_best_response(y_a, solver)
        # Solve LP for best response to y_b (finding optimal a)
        a_new, M_a = solve_lp_best_response(y_b, solver)
        
        if b_new is None:
            b_new = b_avg
        if a_new is None:
            a_new = a_avg
        
        # Update running averages
        a_avg = ((it + 1) * a_avg + (alpha * (it + 2)) * a_new) / (it + 1 + alpha * (it + 2))
        b_avg = ((it + 1) * b_avg + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a_avg = project_simplex(a_avg)
        b_avg = project_simplex(b_avg)
        
        # Track best
        score_a = score_normalized(a_avg)
        score_b = score_normalized(b_avg)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a_avg.copy()
            stall_count = 0
        if score_b < best_score:
            best_score = score_b
            best_a = b_avg.copy()
            stall_count = 0
        
        if it % 50 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            # Perturb if stalled
            if stall_count >= 8:
                rng = np.random.default_rng(it * 7 + 3)
                a_avg = a_avg + 0.05 * rng.standard_normal(n)
                a_avg = project_simplex(a_avg)
                b_avg = b_avg + 0.05 * rng.standard_normal(n)
                b_avg = project_simplex(b_avg)
                stall_count = 0
            
            elapsed = time.monotonic() - t0
            print(f"    VN iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_a={M_a:.6f} M_b={M_b:.6f}")
    
    return best_a, best_score

def interpolate_to_length(a, new_n):

        return a.copy()

        return np.ones(new_n) / new_n

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C such that C @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def solve_lp_best_response(a_fixed, solver, tight=False):

    prob = cp.Problem(cp.Minimize(M_var), constraints)

            return None, None

        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

        for _ in range(5):
            i = rng.integers(n)
            for direction in [-1, 1]:
                delta = direction * step_size * rng.uniform(0.5, 1.5)
                a_trial = a.copy()
                a_trial[i] = max(0, a[i] + delta)
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    shrink_count = 0
                    break
            if improved:
                break
        
        if not improved:
            # Try pair moves
            for _ in range(3):
                i1 = rng.integers(n)
                i2 = rng.integers(n)
                if i1 == i2:
                    continue
                delta = step_size * rng.uniform(0.5, 1.0)
                a_trial = a.copy()
                a_trial[i1] = max(0, a[i1] - delta)
                a_trial[i2] = max(0, a[i2] + delta)
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    shrink_count = 0
                    break
        
        if not improved:
            shrink_count += 1
            if shrink_count >= 10:
                step_size *= 0.7
                if step_size < 0.001:
                    break
                shrink_count = 0
            # Random restart within small neighborhood
            a_trial = a + step_size * 0.5 * rng.standard_normal(n)
            a_trial = project_simplex(np.maximum(a_trial, 0))
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score


def make_smooth_init(n, rng, mode=0):
    """Create a smooth initial sequence."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
    elif mode == 1:
        a = 1.0 + 0.4 * (2 * x - 1)
    elif mode == 2:
        a = np.exp(-5 * (x - 0.3) ** 2) + np.exp(-5 * (x - 0.7) ** 2) + 0.2
    elif mode == 3:
        a = 1.0 + 0.5 * np.sin(4 * np.pi * x + 0.5)
    elif mode == 4:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    else:
        coefs = rng.standard_normal(5)
        a = 1.0
        for k in range(1, 6):
            a += coefs[k - 1] * 0.08 * np.cos(2 * np.pi * k * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def nesterov_gradient_refine(a0, time_limit, T_start=5.0, T_end=200.0,
                              lr_init=0.05, max_steps=10000):
    """
    Quick Nesterov gradient descent with log-sum-exp smoothing
    for fast initial refinement.
    """
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    
    total_stages = 8
    steps_per_stage = max_steps // total_stages
    temps = np.geomspace(T_start, T_end, total_stages)
    
    step_count = 0
    for stage in range(total_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            lr = lr_init / (1.0 + 0.003 * step_count)
            
            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            # Gradient of smoothed max
            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            e = np.exp(Tb - m)
            w = e / e.sum()
            # grad[i] = 2 * sum_j w[j] * y[j-i offset]... use FFT
            fft_size = 1 << (n + len(w) - 1).bit_length()
            Y = rfft(y, fft_size)
            W = rfft(w, fft_size)
            corr = irfft(np.conj(W) * Y, fft_size)[:n]
            grad = 2.0 * corr
            
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based VN, gradient refinement,
    dual-guided peak shaving, pattern search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None:
        n_init = 960
        best_seq = make_smooth_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 0: Quick gradient refinement to warm up ===
    print(f"\n=== Phase 0: Quick Gradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 15:
        a_grad, _ = nesterov_gradient_refine(best_seq, t_alloc)
        try_update(a_grad, "grad_warm")
    
    # === Phase 1: Main LP-based VN ===
    print(f"\n=== Phase 1: Main LP VN n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, max_iters derived from budget")
        max_it = int(t_alloc * 5)  # estimate ~200ms per LP
        a_vn, _ = vn_algorithm(best_seq, t_alloc, solver, max_iters=max_it)
        try_update(a_vn, "vn_main")
    
    # === Phase 2: Dual-guided peak shaving ===
    print(f"\n=== Phase 2: Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 20:
        a_peak, _ = dual_guided_peak_shave(best_seq, t_alloc, solver)
        try_update(a_peak, "peak_shave")
    
    # === Phase 3: Second VN round (from perturbed start) ===
    print(f"\n=== Phase 3: VN Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 30:
        a_pert = best_seq + 0.015 * rng.standard_normal(n_best)
        a_pert = project_simplex(np.maximum(a_pert, 0))
        a_vn2, _ = vn_algorithm(a_pert, t_alloc, solver, max_iters=int(t_alloc * 5))
        try_update(a_vn2, "vn_round2")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = sorted(set([
        n_best - 40, n_best - 20, n_best + 20, n_best + 40, n_best + 60
    ]))
    alt_lengths = [x for x in alt_lengths if 200 <= x <= 5000]
    
    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_sub = min(rem * 0.25, 100.0)
        if t_sub < 20:
            break
        print(f"  n={alt_n}, budget={t_sub:.0f}s")
        init_alt = interpolate_to_length(best_seq, alt_n)
        a_alt, _ = vn_algorithm(init_alt, t_sub * 0.8, solver, max_iters=int(t_sub * 5))
        try_update(a_alt, f"vn_n{alt_n}")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt_ps, _ = dual_guided_peak_shave(a_alt, min(rem * 0.3, 30.0), solver)
            try_update(a_alt_ps, f"ps_n{alt_n}")
    
    # === Phase 5: Pattern search polish ===
    print(f"\n=== Phase 5: Pattern Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 80.0)
    if t_alloc > 10:
        a_ps, _ = pattern_search(best_seq, t_alloc)
        try_update(a_ps, "pattern")
    
    # === Phase 6: Final VN polish ===
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.5, 120.0)
        print(f"\n=== Phase 6: Final VN (best={best_score:.6f}) ===")
        a_fin, _ = vn_algorithm(best_seq, t_fin, solver, max_iters=int(t_fin * 6))
        try_update(a_fin, "final_vn")
    
    # === Phase 7: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.4, 40.0)
        a_final_ps, _ = dual_guided_peak_shave(best_seq, t_ps, solver)
        try_update(a_final_ps, "final_ps")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
