Let me implement the active set LP first.

For the active set LP, we maintain a set of "active" lags `S`. At each iteration:
1. Solve the LP with only the constraints for lags in `S`.
2. Compute the full convolution `(T_a m)_k` for the LP solution `m`.
3. If any lag `k not in S` has `(T_a m)_k > M + ε`, add it to `S`.
4. Repeat until `S` is stable.

This is the "cutting plane" method. The key is that `S` is usually small (maybe 10-50 lags), so the LP is much faster.

def solve_lp_best_response_active(a, solver=None, max_active=100, max_iter=20):
    n = len(a)
    size = 2*n - 1
    active = list(range(size))  # Start with all (or a subset)
    
    for _ in range(max_iter):
        # Build LP with only active constraints
        T_active = build_conv_matrix(a)[active, :]  # |active| x n
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_active @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        
        b_val = np.asarray(b_var.value)
        b_val = np.maximum(b_val, 0)
        b_val = b_val / b_val.sum()
        
        # Check if we need to add more constraints
        full_conv = fast_convolve(a * b_val)  # Wait, this is conv(a, b), not conv(a*a, b)
        # Actually, (T_a b)[k] = conv(a, b)[k]
        # We need to check if any non-active lag violates the M constraint
        M_val = M_var.value
        violations = np.where(full_conv > M_val + 1e-6)[0]
        violations = violations[~np.isin(violations, active)]
        
        if len(violations) == 0:
            return b_val, M_val
        
        # Add violated lags to active set
        active.extend(violations.tolist())
        if len(active) > max_active:

            conv_values = full_conv[active]
            top_idx = np.argsort(conv_values)[::-1][:max_active]
            active = [active[i] for i in top_idx]
    
    return b_val, M_val
