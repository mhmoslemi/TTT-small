def pnorm_obj_and_grad(z, p):

def pnorm_refine(a0, time_limit, p_schedule=None):
    """Refine using p-norm L-BFGS with given p schedule."""
    n = len(a0)
    if p_schedule is None:
        p_schedule = [50, 100, 200, 500, 1000, 2000, 5000]

    z -= z.max()

    best_score = compute_score(a)
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)

        max_iter = min(500, max(50, int(remaining * 5)))

            val, _ = pnorm_obj_and_grad(z_val, p)
            return val

            _, g = pnorm_obj_and_grad(z_val, p)
            return g
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15, 'disp': False})
            z = res.x
        except Exception:
            pass

def peak_equalization(a0, time_limit, n_steps=50000):
    """Local search that tries to equalize the top autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        # Find peaks above 95% of max
        threshold = c_max * 0.95
        peak_idx = np.where(c >= threshold)[0]
        
        if len(peak_idx) == 0:
            break
        
        improved = False
        
        # For each peak, find the coordinates that contribute most
        for k in peak_idx[:10]:
            # c[k] = sum_j a[j]*a[k-j]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            total = contribs.sum()
            if total < 1e-15:
                continue
            
            # Try reducing the top contributor
            i_top = idx[np.argmax(contribs)]
            for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
                a_trial = a.copy()
                reduction = a_trial[i_top] * (1 - factor)
                a_trial[i_top] -= reduction
                if a_trial[i_top] < 0:
                    a_trial[i_top] = 0
                    reduction = a[i_top]
                # Redistribute to other coordinates (proportional)
                mask = np.ones(n, dtype=bool)
                mask[i_top] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] += reduction * others / others.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try random perturbation
            if step % 200 == 0:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(2, min(8, n // 4))
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial /= st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
        
        if not improved and step % 1000 == 0 and step > 0:
            # Check if we're stuck
            if step > 5000:
                break
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq step={step:>5d} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(1.0/new_n, 1.0 - 1.0/new_n, new_n)
    # Actually let me use a simpler approach
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def pnorm_obj_grad(z, p):
    """Objective and gradient in log-space (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    cp = c_pos ** p
    S = cp.sum()
    val = 2.0 * n * S ** (1.0 / p)
    if S < 1e-300:
        return val, np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    g_z = a * (g_a - np.dot(g_a, a))
    return val, g_z


def pnorm_refine(a0, time_limit, p_schedule=None):
    """Refine using p-norm L-BFGS with given p schedule."""
    n = len(a0)
    if p_schedule is None:
        p_schedule = [50, 100, 200, 500, 1000, 2000, 5000]
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    z -= z.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        max_iter = min(400, max(30, int(remaining * 3)))
        try:
            res = minimize(
                lambda zv: pnorm_obj_grad(zv, p)[0],
                z,
                jac=lambda zv: pnorm_obj_grad(zv, p)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-15, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        a_cur = np.maximum(a_cur, 0)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        elapsed = time.monotonic() - t0
        print(f"    p={p:>5d} ({elapsed:.0f}s): score={sc:.6f}, best={best_score:.6f}")
    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        import cvxpy as cp2
        for alt in [cp2.ECOS, cp2.SCS, cp2.PDLP, cp2.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[b_val < 1e-15] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        if it % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 5:
                rng = np.random.default_rng(it + 123)
                noise = rng.standard_normal(n) * (0.02 + 0.01 * stall)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall = 0
                prev_best = best_score
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1:>5d} ({elapsed:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def peak_equalization(a0, time_limit, n_steps=50000):
    """Local search targeting top autocorrelation peaks for reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        c_max = c.max()
        threshold = c_max * 0.95
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) == 0:
            break
        improved = False
        for k in peak_idx[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            if contribs.sum() < 1e-15:
                continue
            top_n = min(10, len(idx))
            top_coords = idx[np.argsort(contribs)[::-1][:top_n]]
            for i in top_coords:
                if a[i] < 1e-14:
                    continue
                for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90, 0.85]:
                    a_trial = a.copy()
                    reduction = a_trial[i] * (1 - factor)
                    a_trial[i] -= reduction
                    if a_trial[i] < 0:
                        a_trial[i] = 0
                        reduction = a[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial[mask] += reduction * others / others.sum()
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            if step % 100 == 0:
                rng = np.random.default_rng(step + 7)
                n_pert = int(rng.integers(2, min(6, n // 8)))
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.04, 0.04))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            if not improved and step > 10000:
                break
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq step={step:>5d} ({elapsed:.0f}s): score={best_score:.6f}")
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
