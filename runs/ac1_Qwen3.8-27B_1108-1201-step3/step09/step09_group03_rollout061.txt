"""

2. Fast vectorized coordinate descent with priority ordering
3. Peak-targeted refinement focusing on high-convolution coordinates

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def spectral_shaping(a, n_iters=200, time_limit=60.0, verbose=True):
    """
    Spectral shaping: iterate between spatial domain (simplex projection)
    and frequency domain (spectrum capping) to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Go to frequency domain
        A = rfft(a, fft_size)
        magnitude = np.abs(A)
        phase = np.angle(A)

        # Compute current autocorrelation from spectrum
        power = magnitude ** 2
        autocorr = irfft(power, fft_size)
        M = autocorr.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Cap the spectrum to reduce peaks
        # The idea: reduce the largest spectral components
        # which correspond to the sharpest features in a
        # Use an adaptive cap
        target_peak = best_score / (2.0 * n) * 0.95
        # Scale down the power spectrum where it's too large
        power_capped = power.copy()
        excess = (power - target_peak * fft_size / n)
        mask = excess > 0
        if mask.any():
            # Reduce excess proportionally
            power_capped[mask] = target_peak * fft_size / n + 0.3 * excess[mask]

        magnitude_new = np.sqrt(np.maximum(power_capped, 0))
        A_new = magnitude_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]

        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        a_new = simplex_project(a_new)

        a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

                    if k_start <= k2 <= k_end:
                        idx = k2 - k_start
                        new_affected = new_affected.copy()
                        new_affected[idx] = conv[k2] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                k2 = 2 * i
                if k_start <= k2 <= k_end:
                    idx = k2 - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (old_val) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def vectorized_cd(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with numpy vectorization over the factor grid.
    Processes all coordinates in a sweep, using vector ops where possible.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    factors = np.array([-0.3, -0.15, -0.07, -0.02, 0.02, 0.07, 0.15, 0.3])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved_count = 0

        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max (vectorized prefix/suffix)
            unaff = 0.0
            if k_start > 0:
                unaff = max(unaff, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff = max(unaff, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]
            base = conv[k_start:k_end + 1]
            k2 = 2 * i
            has_center = (k_start <= k2 <= k_end)
            center_idx = k2 - k_start if has_center else -1

            best_delta = 0.0
            best_new_M = M

            for factor in factors:
                delta = a[i] * factor
                if a[i] + delta < -1e-15:
                    continue

                new_affected = base + 2.0 * delta * a_shifted
                if has_center:
                    new_affected = new_affected.copy()
                    new_affected[center_idx] = conv[k2] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if has_center:
                    changes = changes.copy()
                    changes[center_idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved_count += 1

        if improved_count == 0 and sweeps > 5:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    VCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_a_score if False else best_score


def peak_shaving(a0, time_limit, verbose=True, label=""):
    """
    Direct peak shaving: identify the top convolution peaks and
    directly reduce contributing coordinates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Find top peaks
        n_peaks = min(5, 2 * n - 1)
        peak_pos = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_pos = peak_pos[np.argsort(-conv[peak_pos])]

        for k in peak_pos:
            if time.monotonic() - t0 > time_limit - 1:
                break
            if conv[k] < M * 0.9:
                break  # Only shave significant peaks

            # Find the coordinate i in [max(0,k-n+1), min(n-1,k)]
            # that has the largest a[i]*a[k-i]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            candidates = np.arange(i_start, i_end + 1)
            products = a[candidates] * a[k - candidates]

            # Pick the coordinate with max product
            best_local = int(np.argmax(products))
            i = int(candidates[best_local])
            j = k - i

            if i == j:
                # Quadratic term: reduce a[i]
                # conv[k] has term a[i]^2, reducing a[i] by delta:
                # new term = (a[i]-delta)^2 = a[i]^2 - 2*delta*a[i] + delta^2
                # We want to reduce conv[k] by approximately (conv[k] - M_target)
                target_reduction = conv[k] * 0.1
                delta = min(a[i] * 0.1, target_reduction / (2.0 * a[i] + 1e-15))
            else:
                # Reduce a[i] (the one with larger value for less damage)
                if a[i] >= a[j]:
                    delta = a[i] * 0.05
                else:
                    # Reduce a[j] instead
                    i, j = j, i
                    delta = a[i] * 0.05

            delta = min(delta, a[i] * 0.5)
            if delta < 1e-15:
                continue

            # Apply reduction
            old_val = a[i]
            a[i] = max(a[i] - delta, 0.0)

            # Update convolution
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            a_shifted = a[k_start - i:k_end - i + 1]  # updated a
            changes = -2.0 * delta * a_shifted
            k2 = 2 * i
            if k_start <= k2 <= k_end:
                idx = k2 - k_start
                changes = changes.copy()
                changes[idx] = -2.0 * delta * old_val + delta * delta
            conv[k_start:k_end + 1] += changes

            # Renormalize
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
                conv = fast_convolve(a)
                break
            a = a / s
            conv = conv / (s * s) * (s / s)  # Actually just recompute
            # Better: just recompute for safety after renormalization
            conv = fast_convolve(a)
            M = float(conv.max())

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

        # Early stop if no improvement for a while
        if it > 100:
            recent_scores = [2.0 * n * float(fast_convolve(best_a).max())]
            if abs(recent_scores[0] - best_score) < 1e-10 and it > 200:
                break

    return best_a, best_score


def fw_fast(a_init, time_limit, n_fw_steps=100, verbose=True):
    """
    Fast Frank-Wolfe alternating minimization.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    b_conv = convolve_ab(a, b)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1

        # Optimize b given a
        for step in range(n_fw_steps):
            k_star = int(np.argmax(b_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            # Away step
            if step >= 20:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = b * grad_full
                i_away = int(np.argmax(away_scores))
                if b[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                    t_away = min(t * 0.5, b[i_away] * 0.8)
                    ks = i_away
                    ke = min(ks + n - 1, 2 * n - 2)
                    b_conv[ks:ke + 1] -= t_away * a
                    b[i_away] = max(b[i_away] - t_away, 0.0)

            ks = j_star
            ke = min(ks + n - 1, 2 * n - 2)
            b_conv = (1.0 - t) * b_conv
            b_conv[ks:ke + 1] += t * a
            b = (1.0 - t) * b
            b[j_star] += t

        # Check b
        b_score = 2.0 * n * fast_convolve(b).max()
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()

        # Optimize a given b
        a_conv = convolve_ab(b, a)
        for step in range(n_fw_steps):
            k_star = int(np.argmax(a_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            if step >= 20:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = a * grad_full
                i_away = int(np.argmax(away_scores))
                if a[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                    t_away = min(t * 0.5, a[i_away] * 0.8)
                    ks = i_away
                    ke = min(ks + n - 1, 2 * n - 2)
                    a_conv[ks:ke + 1] -= t_away * b
                    a[i_away] = max(a[i_away] - t_away, 0.0)

            ks = j_star
            ke = min(ks + n - 1, 2 * n - 2)
            a_conv = (1.0 - t) * a_conv
            a_conv[ks:ke + 1] += t * b
            a = (1.0 - t) * a
            a[j_star] += t

        # Check a
        a_score = 2.0 * n * fast_convolve(a).max()
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

        # Check average
        avg = 0.5 * (a + b)
        avg_score = 2.0 * n * fast_convolve(avg).max()
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg.copy()

        if round_num % 5 == 0:
            a = best_a.copy()
            b = best_a.copy()

        if verbose and round_num % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW round={round_num} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, lr_init=0.05, top_k=20, verbose=True):
    """Mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-8:
                break

        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def make_cosine_init(n, rng):
    """Cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    power = rng.uniform(0.5, 3.0)
    a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def make_periodic_init(n, rng):
    """Periodic initialization."""
    period = int(rng.integers(3, 10))
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def make_disjoint_init(n, rng):
    """Two disjoint blocks initialization (non-palindromic)."""
    gap = rng.uniform(0.2, 0.4)
    block_width = (1.0 - gap) / 2.0
    start1 = 0
    end1 = int(block_width * n)
    start2 = int((1.0 - block_width) * n)
    end2 = n
    a = np.zeros(n)
    a[start1:end1] = 1.0
    a[start2:end2] = 1.0
    # Add some structure within blocks
    a[start1:end1] *= (0.5 + 0.5 * np.cos(np.pi * np.arange(end1 - start1) / max(1, end1 - start1)))
    a[start2:end2] *= (0.5 + 0.5 * np.cos(np.pi * np.arange(end2 - start2) / max(1, end2 - start2)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def make_offset_init(n, rng):
    """Offset paired blocks - creates non-palindromic structure."""
    offset = rng.uniform(0.1, 0.3) * n
    width = rng.uniform(0.15, 0.3) * n
    a = np.zeros(n)
    center1 = n * 0.3
    center2 = center1 + offset + width
    if center2 + width < n:
        t = np.arange(n)
        a += np.exp(-0.5 * ((t - center1) / (width / 2)) ** 2)
        a += np.exp(-0.5 * ((t - center2) / (width / 2)) ** 2)
    else:
        a = make_disjoint_init(n, rng)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def run_optimization_pipeline(a_init, n, time_budget, verbose=True):
    """Run the full optimization pipeline on a single initialization."""
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()
    t_start = time.monotonic()

    def remaining():
        return time_budget - (time.monotonic() - t_start)

    # Phase 1: Spectral shaping (global)
    if remaining() > 120:
        t_alloc = min(remaining() * 0.2, 80.0)
        if verbose:
            print(f"  [Spectral] budget={t_alloc:.0f}s")
        a_s, s_s = spectral_shaping(best_a, time_limit=t_alloc, verbose=False)
        if s_s < best_score:
            best_score = s_s
            best_a = a_s.copy()
        a_s2, s_s2 = spectral_shaping_v2(best_a, time_limit=min(t_alloc * 0.5, remaining() * 0.1), verbose=False)
        if s_s2 < best_score:
            best_score = s_s2
            best_a = a_s2.copy()

    # Phase 2: FW (medium speed)
    if remaining() > 200:
        t_alloc = min(remaining() * 0.25, 200.0)
        if verbose:
            print(f"  [FW] budget={t_alloc:.0f}s")
        a_fw, s_fw = fw_fast(best_a, time_limit=t_alloc, n_fw_steps=80, verbose=False)
        if s_fw < best_score:
            best_score = s_fw
            best_a = a_fw.copy()

    # Phase 3: Mirror descent
    if remaining() > 150:
        t_alloc = min(remaining() * 0.2, 150.0)
        if verbose:
            print(f"  [MD] budget={t_alloc:.0f}s")
        a_md, s_md = mirror_descent(best_a, time_limit=t_alloc, lr_init=0.03, top_k=25, verbose=False)
        if s_md < best_score:
            best_score = s_md
            best_a = a_md.copy()

    # Phase 4: Peak shaving
    if remaining() > 100:
        t_alloc = min(remaining() * 0.2, 100.0)
        if verbose:
            print(f"  [PeakShave] budget={t_alloc:.0f}s")
        a_ps, s_ps = peak_shaving(best_a, time_limit=t_alloc, verbose=False)
        if s_ps < best_score:
            best_score = s_ps
            best_a = a_ps.copy()

    # Phase 5: Priority coordinate descent
    if remaining() > 80:
        t_alloc = min(remaining() * 0.3, 300.0)
        if verbose:
            print(f"  [PCD] budget={t_alloc:.0f}s")
        a_cd, s_cd = priority_coord_descent(best_a, time_limit=t_alloc, verbose=False)
        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()

    # Phase 6: Final vectorized CD
    if remaining() > 30:
        t_alloc = min(remaining() * 0.5, 200.0)
        if verbose:
            print(f"  [VCD] budget={t_alloc:.0f}s")
        a_vc, s_vc = vectorized_cd(best_a, time_limit=t_alloc, verbose=False)
        if s_vc < best_score:
            best_score = s_vc
            best_a = a_vc.copy()

    # Final peak shave
    if remaining() > 20:
        t_alloc = min(remaining() * 0.5, 60.0)
        a_pf, s_pf = peak_shaving(best_a, time_limit=t_alloc, verbose=False)
        if s_pf < best_score:
            best_score = s_pf
            best_a = a_pf.copy()

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining spectral shaping, FW, mirror descent,
    peak shaving, and coordinate descent. Explores multiple sequence lengths
    and diverse initializations starting from the parent sequence.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(2024)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using uniform init: n={n_init}, score={best_score:.6f}")

    n_parent = len(best_seq)

    # List of (n, initializations) to try
    configs = []

    # Primary: parent length
    inits_primary = [best_seq.copy()]
    inits_primary.append(make_cosine_init(n_parent, rng))
    inits_primary.append(make_disjoint_init(n_parent, rng))
    inits_primary.append(make_offset_init(n_parent, rng))
    # Noisy parent
    noisy = best_seq + rng.standard_normal(n_parent) * 0.03
    noisy = np.maximum(noisy, 0)
    noisy = noisy / noisy.sum()
    inits_primary.append(noisy)
    configs.append((n_parent, inits_primary))

    # Secondary: 2x length (more degrees of freedom)
    n2 = 2 * n_parent
    if n2 <= 4096:
        # Upsample parent
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n2)
        f = interp1d(x_old, best_seq, kind='cubic', fill_value=0, bounds_error=False)
        upsampled = np.maximum(f(x_new), 0)
        upsampled = upsampled / upsampled.sum()
        inits_2x = [upsampled, make_cosine_init(n2, rng), make_disjoint_init(n2, rng)]
        configs.append((n2, inits_2x))

    # Tertiary: slightly different length
    n3 = n_parent + 64
    if n3 != n_parent and n3 != n2:
        inits_3rd = [make_cosine_init(n3, rng), make_offset_init(n3, rng)]
        configs.append((n3, inits_3rd))

    # Allocate time across configs
    primary_budget = 900.0
    secondary_budget = 500.0
    tertiary_budget = 200.0

    # Allocate more time to the first init in each config
    for config_idx, (n_cfg, inits) in enumerate(configs):
        if config_idx == 0:
            total_budget = primary_budget
        elif config_idx == 1:
            total_budget = secondary_budget
        else:
            total_budget = tertiary_budget

        # Check remaining time
        remaining_global = deadline - time.monotonic()
        if remaining_global < 60:
            break

        total_budget = min(total_budget, remaining_global - 30)
        if total_budget < 60:
            break

        n_inits = len(inits)
        # First init gets 50% of budget, rest split equally
        init_budgets = [total_budget * 0.5]
        rest = total_budget * 0.5 / max(1, n_inits - 1)
        init_budgets.extend([rest] * (n_inits - 1))

        print(f"\n=== Config {config_idx}: n={n_cfg}, {n_inits} inits, budget={total_budget:.0f}s ===")

        for init_idx, (init_a, t_b) in enumerate(zip(inits, init_budgets)):
            if time.monotonic() > deadline - 20:
                break
            if t_b < 30:
                break

            label = f"cfg{config_idx}_init{init_idx}"
            print(f"  Running {label} (budget={t_b:.0f}s), current best={best_score:.6f}")

            a_result, s_result = run_optimization_pipeline(init_a, n_cfg, t_b, verbose=False)

            # Check with evaluate_sequence
            best_seq, best_score = check_and_update(a_result, best_seq, best_score, label)

            # Also try the intermediate results from the pipeline
            # (the pipeline already tracks best internally)

    # Final polishing on the best sequence found
    remaining = deadline - time.monotonic() - 10
    if remaining > 20 and best_seq is not None:
        n_fin = len(best_seq)
        print(f"\n=== Final Polish (n={n_fin}, budget={remaining:.0f}s, best={best_score:.6f}) ===")

        # Coordinate descent
        t_cd = min(remaining * 0.5, 100.0)
        a_cd, s_cd = priority_coord_descent(best_seq, time_limit=t_cd, verbose=True, label="final")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "final_cd")

        # Peak shaving
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_ps = min(remaining * 0.6, 50.0)
            a_ps, s_ps = peak_shaving(best_seq, time_limit=t_ps, verbose=True, label="final_ps")
            best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "final_ps")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
