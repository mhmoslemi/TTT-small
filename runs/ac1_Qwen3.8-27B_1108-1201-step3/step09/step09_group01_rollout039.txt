"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(a, weights):
    """
    Compute gradient of sum_k w[k]*b[k] w.r.t. a.
    grad[j] = 2 * sum_m w[j+m] * a[m]
    Using FFT-based cross-correlation.
    """
    n = len(a)
    # g[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    # = conv(w_rev, a)[2n-2-j] where w_rev = w[::-1]
    w_rev = weights[::-1]
    conv_len = len(w_rev) + n - 1  # (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    Wr = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_result = irfft(Wr * A, fft_size)[:conv_len]
    # g[j] = conv_result[2n-2-j] for j=0,...,n-1
    # g = conv_result[n-1:2n-1][::-1]
    g = conv_result[n - 1:2 * n - 1][::-1]
    return 2.0 * g

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_of(a):
    """Compute the score for a normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

    Gradient descent with adaptive logsumexp smoothing.
    a0 should be normalized (sum=1) and non-negative.

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    no_stall = 0

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution
                changes = 2.0 * best_delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score, sweeps


def fine_coordinate_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1].copy()

            best_delta = 0.0
            best_new_max = M

            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 1.0, 1.005, 1.01, 1.02, 1.01, 1.002]:
                delta = a[i] * (factor - 1.0)
                if abs(delta) < 1e-20:
                    continue
                new_val = a[i] + delta
                if new_val < -1e-18 or new_val > 1.0:
                    continue

                new_affected = affected + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Gradient descent from parent with adaptive smoothing
    Phase 2: Perturb-and-reoptimize cycles
    Phase 3: Multi-length exploration
    Phase 4: Coordinate descent polishing
    Phase 5: Final fine-tuning
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(123)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = score_of(parent)
        print(f"Using uniform init: n=1024, score={best_score:.7f}")

    # === Phase 1: Gradient descent from parent ===
    print(f"\n=== Phase 1: Gradient Descent (best={best_score:.7f}) ===")
    t1_start = time.monotonic()
    n_best = len(best_seq)

    # Run GD with different beta schedules
    for beta_start in [10.0, 30.0, 100.0]:
        rem = deadline - time.monotonic() - 800
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 200.0)
        a_opt, s_opt = gradient_descent_optimize(
            best_seq, t_alloc, beta_init=beta_start, beta_max=500.0,
            lr_init=0.005, lr_min=1e-6, verbose=True, label=f"b{beta_start:.0f}"
        )
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"gd_b{beta_start:.0f}")

    # === Phase 2: Perturb and re-optimize ===
    print(f"\n=== Phase 2: Perturb + Re-optimize (best={best_score:.7f}) ===")

    for rnd in range(5):
        rem = deadline - time.monotonic() - 600
        if rem < 60:
            break

        # Choose perturbation type
        if rnd % 3 == 0:
            perturbed = perturb_nonsymmetric(best_seq, rng, strength=0.2 + 0.1 * rnd)
            pname = "nonsym"
        elif rnd % 3 == 1:
            perturbed = perturb_peak_targeted(best_seq, rng, strength=0.15 + 0.05 * rnd)
            pname = "peak"
        else:
            perturbed = perturb_shape_change(best_seq, rng)
            pname = "shape"

        t_opt = min(rem * 0.4, 150.0)
        a_opt, s_opt = gradient_descent_optimize(
            perturbed, t_opt, beta_init=20.0, beta_max=400.0,
            lr_init=0.008, lr_min=1e-6, verbose=False, label=f"p2r{rnd}"
        )
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_{pname}_r{rnd}")

        # Also try CD from the perturbed point
        rem2 = deadline - time.monotonic() - 500
        if rem2 > 30:
            t_cd = min(rem2 * 0.3, 80.0)
            a_cd, s_cd, _ = coordinate_descent(perturbed, t_cd, n_trials=10, label=f"p2cd_r{rnd}")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"p2cd_{pname}_r{rnd}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.7f}) ===")

    n_current = len(best_seq)
    alt_lengths = [l for l in [n_current + 128, n_current + 256, 1536, 2048]
                   if l != n_current and 512 <= l <= 2560]
    for l in [n_current - 128]:
        if 512 <= l <= 2560 and l not in alt_lengths:
            alt_lengths.append(l)
    alt_lengths = alt_lengths[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        init_alt = interpolate_seq(best_seq, alt_n)
        t_opt = min(rem * 0.35, 120.0)
        a_alt, s_alt = gradient_descent_optimize(
            init_alt, t_opt, beta_init=15.0, beta_max=300.0,
            lr_init=0.006, lr_min=1e-6, verbose=False, label=f"alt{alt_n}"
        )
        best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

        # CD polish on the best alternative
        rem2 = deadline - time.monotonic() - 300
        if rem2 > 30:
            t_cd = min(rem2 * 0.3, 60.0)
            a_cd, s_cd, _ = coordinate_descent(a_alt, t_cd, label=f"altcd{alt_n}")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"altcd_n{alt_n}")

    # === Phase 4: Coordinate descent polishing ===
    print(f"\n=== Phase 4: CD Polish (best={best_score:.7f}) ===")

    # Interpolate best to its natural length if needed
    n_final = len(best_seq)
    current = best_seq.copy()

    for rnd in range(3):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_cd = min(rem * 0.35, 150.0)
        a_cd, s_cd, sweeps = coordinate_descent(
            current, t_cd, n_trials=12, verbose=True, label=f"cd4_r{rnd}"
        )
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd4_r{rnd}")
        current = a_cd.copy()

        # Small perturbation between rounds
        if rnd < 2:
            current = perturb_peak_targeted(a_cd, rng, strength=0.05)

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Fine-tune (best={best_score:.7f}) ===")

    # Fine CD
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t_fine = min(rem * 0.5, 100.0)
        a_fine, s_fine = fine_coordinate_descent(best_seq, t_fine, label="fine")
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

    # One more GD pass with very high beta
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_gd = min(rem * 0.6, 60.0)
        a_gd, s_gd = gradient_descent_optimize(
            best_seq, t_gd, beta_init=200.0, beta_max=800.0,
            lr_init=0.002, lr_min=1e-7, label="final_gd"
        )
        best_seq, best_score = check_and_update(a_gd, best_seq, best_score, "final_gd")

    # Final fine CD
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a_f2, s_f2 = fine_coordinate_descent(best_seq, min(rem * 0.7, 30.0), label="fine2")
        best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
