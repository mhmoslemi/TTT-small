    size = 2 * n - 1
    U = np.ones(size)  # coefficients of 1+x+...+x^{2n-2}

        with np.errstate(divide='ignore', invalid='ignore'):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using polynomial-square-root initialization,
fast subgradient-based alternating optimization with Nesterov acceleration,
LogSumExp-smoothed L-BFGS refinement, and peak-directed coordinate descent.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_best_response(a, b_init, n_steps=50, lr=None):
    """Approximate best response: find b minimizing max_k (a*b)_k via subgradient."""
    n = len(a)
    b = np.maximum(b_init.copy(), 0)
    bs = b.sum()
    if bs < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / bs
    
    if lr is None:
        lr = 1.0 / (n * 0.1)  # initial learning rate
    
    for step in range(n_steps):
        c = convolve_ab(a, b)
        max_c = c.max()
        # Find active peaks (within 1% of max)
        threshold = max_c * 0.99
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient: average over active peaks
        g = np.zeros(n)
        for k in active:
            # (a*b)_k = sum_j a_j b_{k-j}, grad w.r.t. b is a reversed/shifted
            # ∂(a*b)_k / ∂b_m = a_{k-m} if 0 ≤ k-m < n, else 0

        # Update b

        b = proj_simplex(b)
        lr *= 0.95  # decay
        
    return b

    Alternates between computing best response and updating.

    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_rounds = 0
    prev_best = best_score

        if time.monotonic() - t0 > time_limit:

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
