"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage LogSumExp L-BFGS-B
with FFT-accelerated gradients, Nesterov LP best-response, peak-directed
coordinate descent, and multi-length exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_obj_and_grad(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = b / T - max_b / T
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    lse = max_b + T * np.log(exp_scaled.sum())
    obj = 2.0 * n * lse
    g = np.correlate(b.astype(np.float64), a, mode='valid')
    S = np.dot(w, b)
    grad = 2.0 * n * a * (g - S)
    return obj, grad


def lbgfs_multi_stage(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    n = len(a0)
    a = normalize(a0)
    z = z_from_a(a)
    t0 = time.monotonic()
    if T_stages is None:
        T_stages = [40.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    n_stages = len(T_stages)
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        T_val = stage_T
        def func(zx):
            obj, _ = compute_obj_and_grad(zx, n, T_val)
            return obj
        def jac(zx):
            _, grad = compute_obj_and_grad(zx, n, T_val)
            return grad
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 800, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        if (stage_i + 1) % 4 == 0:
            a_check = normalize(softmax_from_z(z))
            sc = evaluate_local(a_check)
            print(f"    L-BFGS stage {stage_i+1}/{n_stages} T={stage_T}: score={sc:.8f}")
    return normalize(softmax_from_z(z))


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = None
    return None


def solve_lp_best_response(a_fixed, solver=None):
    if not HAS_CVXPY:
        return None, None
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=500000):
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(777)
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            if stall_rounds >= 15:
                noise_scale = 0.05 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0
    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.998, 0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-18:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=30000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g_centered = g - g.mean()
        a_trial = a - lr * g_centered
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 50:
            lr *= 0.95
            no_improve = 0
        lr = max(1e-7, lr)
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.3 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 7 * x + 0.5)
        a = np.maximum(a, 0.01)
    else:
        a = np.exp(-((x - rng.uniform(0.2, 0.8)) ** 2) / (2 * 0.1 ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def full_pipeline(init_a, time_budget, solver=None, T_stages=None):

        if sc < best_score:

            best_a = a_lb

    elapsed = time.monotonic() - t0
    rem = time_budget - elapsed
    t_cd_actual = min(t_cd, rem * 0.5)
    if t_cd_actual > 10:
        a_cd, _ = peak_coordinate_descent(best_a, t_cd_actual, n_steps=30000)
        sc = evaluate_local(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = a_cd
            print(f"    After CD: score={best_score:.8f}")

    elapsed = time.monotonic() - t0
    rem = time_budget - elapsed
    t_sg_actual = min(t_sg, rem * 0.5)
    if t_sg_actual > 10:
        a_sg, _ = subgradient_refine(best_a, t_sg_actual, n_steps=20000)
        sc = evaluate_local(a_sg)
        if sc < best_score:
            best_score = sc
            best_a = a_sg
            print(f"    After SG: score={best_score:.8f}")

    elapsed = time.monotonic() - t0
    rem = time_budget - elapsed
    t_final_actual = min(t_final, rem * 0.8)
    if t_final_actual > 10:
        T_final = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        a_fin = lbgfs_multi_stage(best_a, t_final_actual, T_stages=T_final, maxiter_per_stage=5000)
        sc = evaluate_local(a_fin)
        if sc < best_score:
            best_score = sc
            best_a = a_fin
            print(f"    After Final L-BFGS: score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):

            parent = p / p.sum()
            p_score = evaluate_local(parent)

    n_main = len(best_seq)
    T_stages_main = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]

    print(f"\n=== Main pipeline n={n_main} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.55, 900.0)
    if t_main > 60:
        print(f"  Budget: {t_main:.0f}s")
        a_main, sc_main = full_pipeline(best_seq, t_main, solver=solver, T_stages=T_stages_main)
        try_update(a_main, "main_pipeline")

    # === Alternate length 1 ===
    print(f"\n=== Alt length 1 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        alt_n1 = max(800, int(n_main * 0.83))
        if alt_n1 != n_main:
            t_alt1 = min(rem * 0.25, 350.0)
            print(f"  Trying n={alt_n1}, budget={t_alt1:.0f}s")
            init_alt1 = interpolate_sequence(best_seq, alt_n1)
            a_alt1, _ = full_pipeline(init_alt1, t_alt1, solver=solver, T_stages=T_stages_main)
            try_update(a_alt1, f"alt_n{alt_n1}")

    # === Alternate length 2 ===
    print(f"\n=== Alt length 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        alt_n2 = int(n_main * 1.2)
        if alt_n2 != n_main and alt_n2 > 500:
            t_alt2 = min(rem * 0.25, 350.0)
            print(f"  Trying n={alt_n2}, budget={t_alt2:.0f}s")
            init_alt2 = interpolate_sequence(best_seq, alt_n2)
            a_alt2, _ = full_pipeline(init_alt2, t_alt2, solver=solver, T_stages=T_stages_main)
            try_update(a_alt2, f"alt_n{alt_n2}")

    # === Perturbation + re-optimize ===
    print(f"\n=== Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        n_cur = len(best_seq)
        t_pert = min(rem * 0.35, 300.0)
        perturbations = [
            0.01 * np.cos(2 * np.pi * 2 * np.linspace(0, 1, n_cur)),
            0.015 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n_cur)),
            0.008 * rng.standard_normal(n_cur),
        ]
        for pi, pert in enumerate(perturbations):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            perturbed = best_seq + pert
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            t_p = min(rem * 0.4, 200.0)
            a_p, _ = full_pipeline(perturbed, t_p, solver=solver, T_stages=T_stages_main[:10])
            try_update(a_p, f"pert{pi}")

    # === Final ultra-precise polish ===
    print(f"\n=== Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        T_ultra = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        t_fin = min(rem * 0.6, 200.0)
        a_fin = lbgfs_multi_stage(best_seq, t_fin, T_stages=T_ultra, maxiter_per_stage=8000)
        try_update(a_fin, "ultra_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_cd, _ = peak_coordinate_descent(best_seq, min(rem * 0.5, 60.0), n_steps=30000)
        try_update(a_cd, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_sg, _ = subgradient_refine(best_seq, min(rem * 0.5, 40.0), n_steps=20000)
        try_update(a_sg, "final_sg")

    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
