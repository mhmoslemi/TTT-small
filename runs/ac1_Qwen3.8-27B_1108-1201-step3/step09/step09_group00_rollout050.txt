def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient(a):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)

    max_b = b.max()
    # Find active set (positions within 1e-6 of max)
    active = np.where(b >= max_b - 1e-10)[0]

            if 0 <= i - k < n:
                g[i] += a[i - k]
    g /= len(active)

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient_vec(a, b, tol=1e-10):
    """Compute subgradient of max_k conv(a,a)[k].
    Returns (sg, max_b, n_active) where sg is the subgradient vector."""
    n = len(a)
    max_b = b.max()
    active = np.where(b >= max_b - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    sg = np.zeros(n)
    for k in active:
        # g_k[i] = 2*a[k-i] for i in [max(0,k-n+1), min(n-1,k)]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end] = a[k-i_end : k-i_start+1] reversed
        seg = a[k - i_end : k - i_start + 1][::-1]
        sg[i_start:i_end+1] += seg
    sg *= 2.0 / len(active)
    return sg, max_b, len(active)

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) - (cssv - 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    t0 = time.monotonic()

        a_trial = np.maximum(a_trial, 1e-300)

        if score < best_score - 1e-15:

            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0

                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    no_improve = 0
        
        # Periodic print
        # (keep it concise)
    
    return best_a, best_score

For each active k, the contribution to sg is:
- sg[i] += 2*a[k-i] for i in [max(0, k-n+1), min(n-1, k)]

Since n can be up to ~2000 and |A| can be up to a few hundred, this is O(n * |A|) per iteration. With n=1200 and |A|~100, that's 120K operations per iteration, which is fast.

But we can do better: note that the inner sum is (a * delta_shifted) evaluated at each i, which is a convolution-like operation. Specifically:

sg[i] = (2/|A|) * sum_{k in A} a[k-i] * 1[0 <= k-i < n]

Let w[k] = 1 if k in A, else 0. Then:
sg[i] = (2/|A|) * sum_{k} w[k] * a[k-i] * 1[0 <= k-i < n]

This is (w * a)[i] (cross-correlation) where we only sum over valid k-i.

Actually, it's simpler: for each i, sg[i] = (2/|A|) * sum_{k in A, 0 <= k-i < n} a[k-i]
= (2/|A|) * sum_{j: j in (A-i) ∩ [0,n-1]} a[j]

where j = k-i. So sg[i] = (2/|A|) * (sum of a[j] for j in the set (A-i) ∩ [0,n-1]).

This is like a "sliding window sum" of a, where the window is determined by the active set A shifted by -i.

If A is a contiguous interval [k_min, k_max], then (A-i) ∩ [0,n-1] = [max(0, k_min-i), min(n-1, k_max-i)], and we can use a prefix sum of a to compute this in O(1) per i, giving O(n) total.

But A might not be contiguous. However, in practice, the active set (lags where conv is close to max) is often roughly contiguous (or a few isolated peaks). For a non-contiguous set, we can still use the O(n * |A|) approach, which is fine for n=1200 and |A| < 500.

Let me just use the direct O(n * |A|) approach for simplicity and correctness.

Now let me write the complete code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
with adaptive active-set management, multiplicative weights for coarse
optimization, and targeted coordinate descent for final polish.
Key differences from prior approach: mirror-descent initialization phase,
BB-adapted step sizes in refinement, and active-set equalization strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, b, rel_tol=1e-6):
    """
    Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    Returns (sg, max_b, active_indices).
    """
    n = len(a)
    max_b = b.max()
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    sg = np.zeros(n)
    for k in active:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end]
        # = a[k-i_end : k-i_start+1] reversed
        seg = a[k - i_end : k - i_start + 1][::-1].copy()
        sg[i_start:i_end + 1] += seg
    sg *= 2.0 / len(active)
    return sg, max_b, active


def proj_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1.0)
    rho = np.nonzero(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def multiplicative_phase(a0, time_limit, alpha_init=0.1, rel_tol=1e-4):
    """Mirror descent with exponential weights for coarse optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        b = fast_convolve(a)
        sg, max_b, active = compute_subgradient(a, b, rel_tol=rel_tol)
        
        # Multiplicative update
        a_trial = a * np.exp(-alpha * sg)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial / a_trial.sum()
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 50:
                alpha *= 0.9
                if no_improve >= 200:
                    # Add noise to escape
                    a = a * (1.0 + 0.005 * np.random.standard_normal(n))
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    no_improve = 0
        
        if elapsed > time_limit:
            break
    
    return best_a, best_score


def nesterov_subgradient_phase(a0, time_limit, rel_tol_start=1e-4, rel_tol_end=1e-8):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    t = 1
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Step size parameters
    c0 = 1.0
    alpha_decay = 0.999
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        
        # Adaptive tolerance
        progress = min(1.0, elapsed / time_limit)
        rel_tol = rel_tol_start * (rel_tol_end / rel_tol_start) ** progress
        
        # Compute subgradient at lookahead point y
        b_y = fast_convolve(y)
        sg, max_b, active = compute_subgradient(y, b_y, rel_tol=rel_tol)
        
        # Step size (diminishing)
        alpha = c0 / (t ** 0.7)
        alpha = min(alpha, 0.5)
        
        # Gradient step
        a_new = a - alpha * sg
        
        # Project to simplex
        a_new = proj_simplex(a_new)
        
        # Nesterov update for y
        momentum = (t - 1.0) / (t + 2.0)
        y = a_new + momentum * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy
        
        a_prev = a.copy()
        a = a_new
        t += 1
        
        # Track best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score


def bb_subgradient_phase(a0, time_limit, rel_tol=1e-6):
    """Barzilai-Borwein accelerated subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    b = fast_convolve(a)
    sg_prev, max_b_prev, _ = compute_subgradient(a, b, rel_tol=rel_tol)
    
    best_a = a.copy()
    best_score = 2.0 * n * max_b_prev
    t0 = time.monotonic()
    s_prev = None
    y_prev = None
    alpha = 0.01
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        a_new = a - alpha * sg_prev
        a_new = proj_simplex(a_new)
        
        b_new = fast_convolve(a_new)
        sg_new, max_b_new, _ = compute_subgradient(a_new, b_new, rel_tol=rel_tol)
        
        # BB step size update
        s_vec = a_new - a
        y_vec = sg_new - sg_prev
        sy = np.dot(s_vec, y_vec)
        ss = np.dot(s_vec, s_vec)
        if sy > 1e-15 and ss > 1e-30:
            alpha_bb = ss / sy
            alpha_bb = np.clip(alpha_bb, 1e-6, 0.1)
            # Blend with previous alpha for stability
            alpha = 0.5 * alpha + 0.5 * alpha_bb
        
        a = a_new
        sg_prev = sg_new
        iteration += 1
        
        score = 2.0 * n * max_b_new
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def coordinate_descent_phase(a0, time_limit, n_top=5):
    """Targeted coordinate descent on peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_positions = idx[np.argsort(contribs)[::-1][:n_top]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        
        if improved:
            stall = 0
    
    return best_a, best_score


def active_set_equalization(a0, time_limit):
    """Try to equalize the values of the active lags."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = 0.005
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set (within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) < 2:
            break
        
        # Compute "excess" for each active lag
        avg_b = b[active].mean()
        
        # Subgradient: for each active lag k with b[k] > avg,
        # push in direction that reduces b[k]
        # For each active lag with b[k] < avg, push in direction that increases b[k]
        # Net effect: equalize
        sg = np.zeros(n)
        for k in active:
            weight = b[k] - avg_b  # positive if above average, negative if below
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            seg = a[k - i_end : k - i_start + 1][::-1].copy()
            sg[i_start:i_end + 1] += weight * seg
        sg *= 2.0 / len(active)
        
        # Step
        a_trial = a - alpha * sg
        a_trial = proj_simplex(a_trial)
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            alpha *= 0.95
            if alpha < 1e-8:
                break
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
