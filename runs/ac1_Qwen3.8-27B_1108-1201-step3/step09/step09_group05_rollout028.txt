import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

    """Compute 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

    """Smooth approximation of max via log-sum-exp."""
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    return max_Tc + np.log(np.exp(Tc - max_Tc).sum())

    """Gradient of (2n/T) * logsumexp(T*c) with respect to a (in softmax parameterization)."""
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    
    # d(c[k])/d(a[i]) = a[k-i] (for valid indices)
    # d(logsumexp)/da = sum_k w[k] * d(c[k])/da
    grad_a = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    return 2.0 * n * grad_a  # Before softmax Jacobian

def adam_optimize(a0, time_limit, T_schedule=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on the simplex using softmax parameterization."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))  # Log-domain parameterization
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    if T_schedule is None:
        T_schedule = np.geomspace(50.0, 0.5, 20)
    
    T = T_schedule[0]
    T_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Compute gradient in z-space
        c = fast_convolve(a)
        Tc = T * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        
        # Gradient of logsumexp(T*c) w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * a[k - idx]
        
        # Scale by 2n/T for the smoothed objective (2n/T) * logsumexp(T*c)
        # The actual gradient of (2n/T)*logsumexp(T*c) w.r.t. a is 2n * (weighted avg of d(c)/da)
        # which is 2n * sum_k w[k] * d(c[k])/da = 2n * grad_a (as computed above)
        # But we want the gradient w.r.t. z: grad_z = a * (grad_a - sum(a * grad_a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S) * 2.0 * n
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back and evaluate
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        score = score_normalized(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Update temperature
        if t % 50 == 0 and T_idx < len(T_schedule) - 1:
            T_idx += 1
            T = T_schedule[T_idx]
    
    return best_a, best_score

def subgradient_nesterov(a0, time_limit, n_iters=100000):
    """Nesterov-accelerated subgradient method (no LP)."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at current y
        c = fast_convolve(y)
        k_star = int(np.argmax(c))
        
        # Subgradient: g[i] = y[k_star - i] for valid i
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = y[k_star - idx]
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        step_size = 1.0 / np.sqrt(it + 1)
        
        # Gradient step
        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Nesterov update
        y = (1 - alpha) * y + alpha * a_new
        
        # Midpoint (actual iterate)
        a = ((it + 1) * a + alpha * (it + 2) * a_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        
        score = score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def multi_peak_nesterov(a0, time_limit, n_peaks=5):
    """Nesterov method that considers multiple peaks simultaneously."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(y)
        
        # Find top n_peaks positions
        # Use weighted subgradient from top peaks
        peak_vals = np.sort(c)[::-1][:n_peaks]
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Weight by how much above average
        avg_val = c.mean()
        weights = np.maximum(peak_vals - avg_val, 0)
        weights = weights / weights.sum() if weights.sum() > 0 else np.ones(n_peaks) / n_peaks
        
        # Combined subgradient
        g = np.zeros(n)
        for wi, ki in zip(weights, peak_idx):
            if wi < 1e-10:
                continue
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            g[idx] += wi * y[ki - idx]
        
        alpha = 2.0 / (it + 2)
        step_size = 1.0 / np.sqrt(it + 1)
        
        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        y = (1 - alpha) * y + alpha * a_new
        a = ((it + 1) * a + alpha * (it + 2) * a_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        
        if it % 100 == 0:
            score = score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=50000):
    """Targeted peak shaving with coordinate descent."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Find coordinates that contribute most to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Try reducing top contributors
        top_coords = np.argsort(contribs)[::-1][:10]
        improved = False
        
        for ci in top_coords:
            i = idx[ci]
            if a[i] < 1e-10:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = norm_sequence(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=10):
    """L-BFGS with log-sum-exp smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            log_sum_exp = max_Tc + np.log(exp_Tc.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            grad_a = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S) * 2.0 * n
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 150000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a = np.exp(z - z.max())
    return a / a.sum()

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_normalized(a):
    """Compute 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def compute_subgradient(a, k_star):
    """Compute subgradient of max_k conv(a,a) at peak k_star."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idx = np.arange(js, je + 1)
    g[idx] = a[k_star - idx]
    return g


def compute_weighted_subgradient(a, peak_indices, peak_weights):
    """Compute weighted combination of subgradients at multiple peaks."""
    n = len(a)
    g = np.zeros(n)
    for w, k in zip(peak_weights, peak_indices):
        if w < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w * a[k - idx]
    return g


def mirror_descent(a0, time_limit, n_iters=200000, n_peaks=3):
    """Mirror descent with entropy regularizer on the simplex."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        
        # Find top peaks
        if n_peaks == 1:
            k_star = int(np.argmax(c))
            g = compute_subgradient(a, k_star)
        else:
            top_idx = np.argsort(c)[::-1][:n_peaks]
            top_vals = c[top_idx]
            avg_val = c.mean()
            weights = np.maximum(top_vals - avg_val, 0)
            if weights.sum() < 1e-15:
                weights = np.ones(n_peaks) / n_peaks
            else:
                weights = weights / weights.sum()
            g = compute_weighted_subgradient(a, top_idx, weights)
        
        # Mirror descent step: a_new[i] = a[i] * exp(-eta * g[i])
        eta = 1.0 / (np.sqrt(it + 1) + 1.0)
        log_update = -eta * g
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        # Nesterov acceleration
        alpha = 2.0 / (it + 2)
        a = (1 - alpha) * a + alpha * a_new
        
        if it % 50 == 0:
            score = score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=12):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            log_sum_exp = max_Tc + np.log(exp_Tc.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = np.exp(z_in - z_in.max())
            a = a / a.sum()
            c = fast_convolve(a)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            grad_a = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S) * 2.0 * n
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 240000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a = np.exp(z - z.max())
    return a / a.sum()


def peak_shave(a0, time_limit, n_steps=50000):
    """Targeted peak shaving with coordinate and pairwise adjustments."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Find top contributors to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        improved = False
        
        # Try single coordinate reduction
        top_coords = np.argsort(contribs)[::-1][:8]
        for ci in top_coords:
            i = idx[ci]
            if a[i] < 1e-12:
                continue
            for factor in [0.97, 0.93, 0.88, 0.82, 0.75]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = norm_sequence(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Try pairwise adjustment
        if not improved:
            if len(top_coords) >= 2:
                for ci1 in range(min(5, len(top_coords))):
                    for ci2 in range(ci1 + 1, min(5, len(top_coords))):
                        i1, i2 = idx[top_coords[ci1]], idx[top_coords[ci2]]
                        if a[i1] < 1e-12 or a[i2] < 1e-12:
                            continue
                        for f1, f2 in [(0.92, 0.92), (0.88, 0.95), (0.95, 0.88), (0.85, 0.85)]:
                            a_trial = a.copy()
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            a_trial = norm_sequence(a_trial)
                            score = score_normalized(a_trial)
                            if score < best_score - 1e-13:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 150:
                break
        else:
            no_improve = 0
    
    return best_a, best_score


def freq_domain_optimize(a0, time_limit, n_iters=100000):
    """Frequency-domain inspired optimization: flatten the power spectrum."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Subgradient pointing away from peak
        g = compute_subgradient(a, k_star)
        
        # Also consider second peak
        c2 = c.copy()
        c2[k_star] = -np.inf
        k_star2 = int(np.argmax(c2))
        g2 = compute_subgradient(a, k_star2)
        
        # Weighted combination
        w1 = c[k_star] / max(c[k_star] + c[k_star2], 1e-15)
        w2 = 1 - w1
        g_combined = w1 * g + w2 * g2
        
        # Softmax (mirror) update with adaptive step
        eta = 0.5 / (np.sqrt(it + 1) + 1.0)
        update = a * np.exp(-eta * g_combined)
        s = update.sum()
        if s < 1e-15:
            break
        a_new = update / s
        
        # Blend with current for stability
        blend = 0.7
        a = blend * a + (1 - blend) * a_new
        a = norm_sequence(a)
        
        if it % 30 == 0:
            score = score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def simulated_annealing(a0, time_limit, n_steps=5000, T_start=0.05, T_end=0.001):
    """Simulated annealing with structured perturbations on the simplex."""
    n = len(a0)
    a = norm_sequence(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = T_start * (T_end / T_start) ** (step / max(n_steps, 1))
        
        # Structured perturbation: modify a few coordinates
        n_mod = rng.integers(1, max(2, n // 20) + 1)
        coords = rng.choice(n, size=n_mod, replace=False)
        
        a_trial = a.copy()
        for i in coords:
            # Multiplicative perturbation
            factor = rng.lognormal(0, 0.3)
            a_trial[i] *= factor
        
        a_trial = norm_sequence(a_trial)
        score_trial = score_normalized(a_trial)
        
        delta = score_trial - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
    
    return best_a, best_score
