import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlate(a, y):
    """Compute g_j = sum_m a_m * y_{j+m} for j=0..n-1 (sliding dot product)."""
    n = len(a)
    # g_j = sum_{m=0}^{n-1} a_m * y_{j+m}
    # This is the cross-correlation of a and y, evaluated at lags 0..n-1
    # corr[j] = sum_m a_m * y_{j+m}
    # This is equivalent to: if we reverse a, then convolve with y
    a_rev = a[::-1]
    size = n + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    Y = rfft(y, fft_size)
    conv = irfft(A * Y, fft_size)
    # conv[k] = sum_m a_rev[m] * y[k-m] = sum_m a[n-1-m] * y[k-m]
    # Let i = n-1-m: conv[k] = sum_i a[i] * y[k-n+1+i]
    # We want g_j = sum_m a_m * y_{j+m} = sum_i a[i] * y_{j+i}
    # So g_j = conv[j + n - 1 + 1] = conv[j + n]? Let me recheck.
    # conv[k] = sum_{m=0}^{n-1} a[n-1-m] * y[k-m] (for valid k)
    # Let i = n-1-m, so m = n-1-i:
    # conv[k] = sum_{i=0}^{n-1} a[i] * y[k-(n-1-i)] = sum_{i=0}^{n-1} a[i] * y[k-n+1+i]
    # We want: g_j = sum_{i=0}^{n-1} a[i] * y[j+i]
    # So set k-n+1 = j, i.e., k = j + n - 1
    # g_j = conv[j + n - 1] for j = 0, ..., n-1
    return conv[n-1 : n-1+n]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def nesterov_subgradient(a0, time_limit, lr_scale=1.0):
    """
    Nesterov accelerated subgradient method for min max_k (a*a)_k
    on the simplex. Uses exact (unsmoothed) max.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    step_count = 0
    
    while True:
        if time.monotonic() - t0 > time_limit:
            break
        # Nesterov momentum
        t = step_count + 1
        beta = (1.0 - np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t * t) if t > 0 else 0.0
        # Actually standard Nesterov: beta_t = (1 - t_{t-1}) / t_t where t_t = (1+sqrt(1+4t^2))/2
        # Let me use the simpler form: beta = (t_prev - 1) / t_curr
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t-1) * (t-1))) / 2.0 if t > 1 else 1.0
        beta = (t_prev - 1.0) / t_curr
        
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        
        # Compute subgradient at y
        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        # grad_i = 2 * a_{k*-i} if 0 <= k*-i < n, else 0
        grad = np.zeros(n)
        # k*-i in [0, n-1] means i in [k*-n+1, k*]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        grad[i_lo:i_hi+1] = 2.0 * y[k_star - i_lo : k_star - i_hi : -1]
        # Wait: grad_i = 2 * y[k_star - i] for i in [i_lo, i_hi]
        # When i = i_lo: k_star - i = k_star - i_lo
        # When i = i_hi: k_star - i = k_star - i_hi
        # So grad[i_lo..i_hi] = 2 * y[k_star-i_lo, k_star-i_lo-1, ..., k_star-i_hi]

        indices = np.arange(i_lo, i_hi + 1)
        grad[indices] = 2.0 * y[k_star - indices]

def mw_best_response(a, n_mw_iters=500, eta=0.5):
    """
    Solve min_b max_k (a*b)_k using multiplicative weights on the dual.
    Returns b (normalized to sum=1).
    """
    n = len(a)
    y = np.ones(2 * n - 1) / (2 * n - 1)
    
    for _ in range(n_mw_iters):
        # Compute g_j = sum_m a_m * y_{j+m} for j=0..n-1
        g = cross_correlate(a, y)
        j_min = int(np.argmin(g))
        
        # Update y: y_k *= exp(eta * C_{j_min, k})
        # C_{j_min, k} = a_{k-j_min} if 0 <= k-j_min < n, else 0
        loss = np.zeros(2 * n - 1)
        k_lo = j_min
        k_hi = j_min + n - 1
        loss[k_lo:k_hi+1] = a  # a_{k-j_min} for k = j_min...j_min+n-1 gives a_0...a_{n-1}
        
        y *= np.exp(eta * loss)
        y_sum = y.sum()
        if y_sum > 0:
            y /= y_sum
        else:
            y = np.ones(2 * n - 1) / (2 * n - 1)
    
    # From the dual solution y, derive b
    # b should be concentrated where (C^T y)_j is minimal
    g = cross_correlate(a, y)
    # Soft version: b_j proportional to exp(-alpha * g_j)
    alpha = 5.0
    b = np.exp(-alpha * (g - g.min()))
    b /= b.sum()
    
    return b, g.min()


def mw_alternating(a0, time_limit, n_mw_iters=300, eta=0.3, outer_lr=0.3):
    """
    Alternating optimization: alternate between finding best response b for a,
    and updating a towards b.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    outer_iter = 0
    
    while True:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find best response b
        b, dual_val = mw_best_response(a, n_mw_iters, eta)
        
        # Update a towards b
        a = (1.0 - outer_lr) * a + outer_lr * b
        a = project_simplex(a)
        outer_iter += 1
        
        # Check score
        if outer_iter % 5 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            if outer_iter % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"        MW-alternating iter={outer_iter} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, y):
    """g[j] = sum_m a[m] * y[j+m] for j=0..n-1."""
    n = len(a)
    size = n + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    A = rfft(a_rev, fft_size)
    Y = rfft(y, fft_size)
    conv = irfft(A * Y, fft_size)
    return conv[n - 1: n - 1 + n]


def score_norm(a):
    """Objective for unit-sum sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def proj_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

    """Subgradient of (a*a)_k w.r.t. a."""

    idx = np.arange(i_lo, i_hi + 1)

    return grad

def interp_seq(f, new_n):
    """Interpolate to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    f_i = np.maximum(f_i, 0.0)
    f_i /= f_i.sum()
    return f_i

    elif shape == 'sin':
        a = np.sin(np.pi * x)
    elif shape == 'gauss':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif shape == 'parabolic':
        a = 1.0 - (2.0 * x - 1.0) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a
