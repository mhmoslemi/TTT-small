        if remaining < 30:
            break

        # FW optimization
        t_fw = min(remaining * 0.2, 200)
        if t_fw > 20:
            n_steps = min(int(t_fw * 30), 3000)
            a_fw = fw_optimize(init, time_limit=t_fw)
            try_candidate(a_fw, f"fw_{idx}")

        # Mirror descent
        remaining = deadline - time.monotonic() - 60
        if remaining > 60:
            t_md = min(remaining * 0.2, 150)
            a_md = mirror_descent(a_fw if 'a_fw' in dir() else init,
                                  time_limit=t_md, lr_init=0.03, top_k=30)
            try_candidate(a_md, f"md_{idx}")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            best_for_cd = a_md if 'a_md' in dir() and compute_score(a_md) < best_score else \
                          (a_fw if 'a_fw' in dir() else init)
            t_cd = min(remaining * 0.4, 120)
            a_cd = coord_descent(best_for_cd, max_sweeps=200, time_limit=t_cd)
            try_candidate(a_cd, f"cd_{idx}")

    # --- Final polishing on the best ---
    remaining = deadline - time.monotonic() - 10
    if remaining > 20 and best_a is not None:
        # Another round of coordinate descent with finer steps
        t_final = min(remaining * 0.8, 100)
        a_final = coord_descent(best_a, max_sweeps=300, time_limit=t_final)
        try_candidate(a_final, "final_cd")

    # --- Final fallback ---
    if best_a is None:
        best_a = np.ones(n_base) / n_base

    # Verify final answer
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nFINAL: score={final_score:.12f}, n={len(best_a)}")

    return [float(x) for x in best_a]
