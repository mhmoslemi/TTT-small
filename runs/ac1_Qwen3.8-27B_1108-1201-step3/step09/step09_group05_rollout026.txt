"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp L-BFGS with aggressive cooling schedule

3. FFT-based gradient computation

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(a, w):
    """
    Compute grad_a[i] = sum_k w[k] * a[k-i] for i=0..n-1
    where w has length 2n-1 (the LogSumExp weights).
    Uses FFT for efficiency.
    """
    n = len(a)
    size_b = 2 * n - 1
    # Need N >= size_b + n - 1 = 3n - 2 for linear cross-correlation
    N = 1 << (3 * n - 2 - 1).bit_length()
    wa = np.zeros(N)
    wa[:size_b] = w
    A = fft(a, N)
    W = fft(wa, N)
    result = ifft(np.conj(A) * W, N).real[:n]
    return result


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lbgfs_stage(a0, T, maxiter, time_limit, t0):
    """Single L-BFGS stage at temperature T."""
    n = len(a0)
    z = z_from_a(a0).copy()

    def func(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * log_sum_exp

    def jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # grad_a[i] = (4n/T) * sum_k w[k] * a[k-i]
        raw_grad = compute_gradient(a, w)
        grad_a = (4.0 * n / T) * raw_grad
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=func, x0=z, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxfun': maxiter * 500, 'disp': False}
        )
        return softmax_from_z(res.x)
    except Exception:
        return a0


def lbgfs_multi_stage(a0, time_limit, T_list, iters_per_stage=200):
    """Multi-stage L-BFGS with given temperature list."""
    t0 = time.monotonic()
    a = a0.copy()
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        a = lbgfs_stage(a, T, iters_per_stage, time_limit, t0)
    return a


def palindrome_project(a):
    """Project to palindrome: a[i] = a[n-1-i]."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return a.copy()
    return a_pal / s


def subgradient_polish(a0, time_limit, n_steps=5000):
    """
    Direct subgradient method: at each step, find k*=argmax(b), 
    compute subgradient, take a small step in the negative direction,
    project onto simplex.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step_size = 0.1
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        score = 2.0 * n * peak_val

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            step_size *= 0.9
            if step_size < 1e-5:
                break

        # Subgradient of max(b) is the rows of the convolution matrix
        # corresponding to the argmax indices
        peak_indices = np.where(b >= peak_val - 1e-10)[0]
        g = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        
        # Normalize subgradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Step
        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_rounds=1000):
    """
    Coordinate descent: at each round, find the peak of b, then 
    adjust the coordinate that most contributes to the peak.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        # Find the coordinate i that most contributes to b[k_star]
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Try reducing the top contributor and redistributing
        top_i = idx[np.argmax(contribs)]
        improved = False
        
        for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
            a_trial = a.copy()
            reduction = a_trial[top_i] * (1 - factor)
            a_trial[top_i] *= factor
            if reduction < 1e-15:
                continue
            # Redistribute to coordinates far from the peak
            # (coordinates that don't contribute to b[k_star])
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je+1] = False
            non_contrib[top_i] = False
            if non_contrib.sum() > 0:
                a_trial[non_contrib] += reduction * a_trial[non_contrib] / a_trial[non_contrib].sum()
            a_trial = norm_sequence(a_trial)
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try pair adjustments
            top_idx = np.argsort(contribs)[::-1][:4]
            found = False
            for ci in top_idx:
                i = idx[ci]
                j = k_star - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial = norm_sequence(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        found = True
                        improved = True
                        break
                if found:
                    break
            if not improved:
                # Try adding to a zero coordinate
                zero_coords = np.where(a < 1e-10)[0]
                if len(zero_coords) > 0:
                    # Pick the one furthest from the peak
                    dists = np.abs(zero_coords - k_star)
                    best_zero = zero_coords[np.argmax(dists)]
                    for eps in [0.001, 0.005, 0.01]:
                        a_trial = a.copy()
                        a_trial[best_zero] = eps
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                if not improved:
                    break

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        # Bump function
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 12)
    elif mode == 7:
        # Multi-bump
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * x * (1 - x)) + 0.5 * np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif mode == 8:
        # Random smooth
        a = np.ones(n) + 0.3 * np.sin(2 * np.pi * rng.random() * 5 * np.linspace(0, 1, n))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive multi-stage LogSumExp L-BFGS with very low final temperatures
2. Palindrome symmetry exploitation to reduce effective search space
3. FFT-based cross-correlation gradient computation
4. Direct subgradient and coordinate descent polishing
5. Multi-start perturbation with local re-optimization
6. Adaptive length exploration with interpolation warm-starts
7. Ultra-low-temperature final polish for sub-1e-4 improvements

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from functools import partial


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(a, w):
    """
    Compute cross-correlation g[i] = sum_k w[k]*a[k-i] for i=0..n-1.
    w has length 2n-1, a has length n. Uses FFT.
    """
    n = len(a)
    size_b = 2 * n - 1
    N = 1 << (3 * n - 2).bit_length()
    wa = np.zeros(N)
    wa[:size_b] = w
    A = rfft(a, N)
    W = rfft(wa, N)
    result = irfft(np.conj(A) * W, N)[:n]
    return result


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _lbgfs_func(z, T, n, time_limit, t0):
    """Objective function for L-BFGS with LogSumExp smoothing."""
    if time.monotonic() - t0 > time_limit:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def _lbgfs_jac(z, T, n, time_limit, t0):
    """Gradient for L-BFGS with LogSumExp smoothing."""
    if time.monotonic() - t0 > time_limit:
        return np.zeros(len(z))
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    raw_grad = compute_gradient(a, w)
    grad_a = (4.0 * n / T) * raw_grad
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_stage(a0, T, maxiter, time_limit, t0):
    """Single L-BFGS stage at given temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    func = partial(_lbgfs_func, T=T, n=n, time_limit=time_limit, t0=t0)
    jac = partial(_lbgfs_jac, T=T, n=n, time_limit=time_limit, t0=t0)
    try:
        res = minimize(
            fun=func, x0=z, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxfun': maxiter * 500, 'dispe': False}
        )
        return softmax_from_z(res.x)
    except Exception:
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12}
            )
            return softmax_from_z(res.x)
        except Exception:
            return a0.copy()


def lbgfs_multi_stage(a0, time_limit, T_list, iters_per_stage=150):
    """Multi-stage L-BFGS with given temperature schedule."""
    t0 = time.monotonic()
    a = a0.copy()
    for i, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        a = lbgfs_stage(a, T, iters_per_stage, time_limit, t0)
    return a


def palindrome_project(a):
    """Project to palindrome: a[i] = a[n-1-i]."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return a.copy()
    return a_pal / s


def subgradient_polish(a0, time_limit, n_steps=50000):
    """
    Direct subgradient method on the max of the convolution.
    Steps away from the peak-contributing coordinates.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step_size = 0.05
    no_improve = 0
    decay = 0.95

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        score = 2.0 * n * peak_val

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            step_size *= decay
            no_improve = 0
            if step_size < 1e-6:
                break

        peak_indices = np.where(b >= peak_val - 1e-8 * peak_val)[0]
        g = np.zeros(n)
        for k in peak_indices[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        a_new = a - step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

    return best_a, best_score

    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses aggressive multi-stage L-BFGS, palindrome symmetry, subgradient/CD polishing,
    multi-start perturbation, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        best_init_score = float('inf')
        for n_try in [960, 768, 1024, 1280]:
            for mode in range(10):
                a_try = structured_init(n_try, rng, mode)
                sc = evaluate_safe(a_try)
                if sc < best_init_score:
                    best_init_score = sc
                    best_seq = a_try.copy()
                    best_score = sc
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}, target=1.5027")
    print(f"Time budget: {deadline - time.monotonic():.0f}s")

    # === Phase 1: Aggressive multi-stage L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: Multi-stage L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.32, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        # Very aggressive cooling: many stages down to extremely low T
        temps = list(np.geomspace(100.0, 0.1, 20))
        temps.extend([0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001])
        a_lb = lbgfs_multi_stage(best_seq, t_alloc, temps, iters_per_stage=120)
        sc = evaluate_safe(a_lb)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lb.copy()
            print(f"  NEW BEST [lbgfs_main]: {sc:.8f}")

    # === Phase 2: Palindrome projection + re-optimization ===
    print(f"\n=== Phase 2: Palindrome + re-opt (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        a_pal = palindrome_project(best_seq)
        pal_score = evaluate_safe(a_pal)
        print(f"  After palindrome proj: {pal_score:.8f}")
        t_pal = min(rem * 0.14, 200.0)
        if t_pal > 20:
            temps_pal = list(np.geomspace(40.0, 0.05, 15))
            temps_pal.extend([0.02, 0.01, 0.005])
            a_pal_opt = lbgfs_multi_stage(a_pal, t_pal, temps_pal, iters_per_stage=120)
            sc = evaluate_safe(a_pal_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_pal_opt.copy()
                print(f"  NEW BEST [palindrome]: {sc:.8f}")

    # === Phase 3: Subgradient + coordinate descent ===
    print(f"\n=== Phase 3: Subgrad/CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sg = min(rem * 0.08, 80.0)
        if t_sg > 10:
            a_sg, _ = subgradient_polish(best_seq, t_sg, n_steps=50000)
            sc = evaluate_safe(a_sg)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_sg.copy()
                print(f"  NEW BEST [subgrad]: {sc:.8f}")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.06, 60.0)
            if t_cd > 10:
                a_cd, _ = coordinate_descent_polish(best_seq, t_cd, n_rounds=50000)
                sc = evaluate_safe(a_cd)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_cd.copy()
                    print(f"  NEW BEST [cd]: {sc:.8f}")

    # === Phase 4: Multi-start perturbation ===
    print(f"\n=== Phase 4: Multi-start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        perturb_configs = [
            (0.02, 3), (0.04, 3), (0.08, 3), (0.12, 2), (0.20, 2)
        ]
        for scale, n_trials in perturb_configs:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            for trial in range(n_trials):
                rem = deadline - time.monotonic()
                if rem < 15:
                    break
                # Mix of Gaussian and structured perturbations
                if trial % 3 == 0:
                    noise = scale * rng.standard_normal(n_cur)
                elif trial % 3 == 1:
                    # Low-frequency perturbation
                    freq = rng.random(n_cur) * 4 * np.pi
                    noise = scale * np.sin(freq + rng.random() * np.pi)
                else:
                    # Block perturbation
                    noise = np.zeros(n_cur)
                    block_size = max(1, n_cur // 8)
                    bstart = int(rng.random() * (n_cur - block_size))
                    noise[bstart:bstart + block_size] = scale * rng.standard_normal(block_size)

                perturbed = norm_sequence(best_seq + noise)
                t_sub = min(rem * 0.12, 50.0)
                if t_sub < 8:
                    continue
                temps_sub = list(np.geomspace(25.0, 0.1, 10))
                temps_sub.extend([0.05, 0.02, 0.01])
                a_sub = lbgfs_multi_stage(perturbed, t_sub, temps_sub, iters_per_stage=80)
                sc = evaluate_safe(a_sub)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_sub.copy()
                    print(f"  NEW BEST [pert_s{scale}_t{trial}]: {sc:.8f}")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        candidates_n = set()
        for mult in [1.25, 1.5, 2.0, 0.75, 1.33, 0.875, 1.125]:
            alt_n = int(n_cur * mult)
            if 384 <= alt_n <= 3072 and alt_n != n_cur:
                candidates_n.add(alt_n)
        for fixed_n in [512, 768, 1024, 1280, 1536, 1920, 2048]:
            if 384 <= fixed_n <= 3072 and fixed_n != n_cur:
                candidates_n.add(fixed_n)
        sorted_ns = sorted(candidates_n)

        for alt_n in sorted_ns[:4]:
            rem = deadline - time.monotonic()
            if rem < 35:
                break
            t_total = min(rem * 0.18, 100.0)
            if t_total < 15:
                break
            print(f"  Trying n={alt_n}, budget={t_total:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            temps_alt = list(np.geomspace(50.0, 0.1, 12))
            temps_alt.extend([0.05, 0.02, 0.01])
            a_alt = lbgfs_multi_stage(init_alt, t_total, temps_alt, iters_per_stage=100)
            sc = evaluate_safe(a_alt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_alt.copy()
                print(f"  NEW BEST [len_{alt_n}]: {sc:.8f}")

    # === Phase 6: Ultra-low-temperature final polish ===
    print(f"\n=== Phase 6: Ultra-low-T polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.22, 100.0)
        if t_fin > 5:
            temps_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin = lbgfs_multi_stage(best_seq, t_fin, temps_fin, iters_per_stage=200)
            sc = evaluate_safe(a_fin)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_fin.copy()
                print(f"  NEW BEST [ultra_low_T]: {sc:.8f}")

    # === Phase 7: Final coordinate descent + subgradient sweep ===
    print(f"\n=== Phase 7: Final CD/SG (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_cd2 = min(rem * 0.3, 50.0)
        if t_cd2 > 5:
            a_cd2, _ = coordinate_descent_polish(best_seq, t_cd2, n_rounds=50000)
            sc = evaluate_safe(a_cd2)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_cd2.copy()
                print(f"  NEW BEST [final_cd]: {sc:.8f}")

        rem = deadline - time.monotonic()
        if rem > 8:
            t_sg2 = min(rem * 0.25, 30.0)
            if t_sg2 > 3:
                a_sg2, _ = subgradient_polish(best_seq, t_sg2, n_steps=30000)
                sc = evaluate_safe(a_sg2)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_sg2.copy()
                    print(f"  NEW BEST [final_sg]: {sc:.8f}")
