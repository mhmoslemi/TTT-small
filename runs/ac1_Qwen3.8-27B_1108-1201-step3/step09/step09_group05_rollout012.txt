1. Multiplicative weights (mirror descent) for fast global structure
2. Multi-stage L-BFGS with log-sum-exp smoothing for local refinement

4. Coordinated peak shaving for final polish

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_conv(a_norm)
    return 2.0 * n * b.max()


def norm_a(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def mirror_descent(a0, time_limit, eta0=0.1, eta_decay=0.999, n_phases=5):
    """Mirror descent with exponential weights on the simplex."""
    n = len(a0)
    a = norm_a(a0)
    best_a = a.copy()
    best_s = score(a)
    t0 = time.monotonic()
    
    phase_len = n_phases
    eta = eta0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient
        log_a = np.log(np.maximum(a, 1e-300))
        # g[i] = a[k_star - i] for valid i
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        indices = np.arange(js, je + 1)
        grads = a[k_star - indices]
        
        # Multiplicative update
        log_a[indices] -= eta * grads
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new /= a_new.sum()
        
        s_new = 2.0 * n * b.max()  # approximate (using old b)
        # Actually let's compute the real score
        s_new = 2.0 * n * fast_conv(a_new).max()
        
        if s_new < best_s:
            best_s = s_new
            best_a = a_new.copy()
        
        a = a_new
        eta *= eta_decay
    
    return best_a, best_s


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=12):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_a(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_conv(a_in)
            Tb = T * b
            m = Tb.max()
            lse = m + np.log(np.exp(Tb - m).sum())
            return (2.0 * n / T) * lse
        
        def jac(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b = fast_conv(a_in)
            Tb = T * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w /= w.sum()
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_in[k - idx]
            # Chain rule for softmax parameterization
            S = np.dot(grad_a, a_in)
            return a_in * (2.0 * grad_a - S)
        
        maxiter = max(20, min(2000, int(remaining * 50)))
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out, score(a_out)

    n = len(a0)

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        top = np.argsort(contribs)[::-1][:10]
        
        improved = False
        for ti in top:
            i = idx[ti]
            j = k_star - i

            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            # Try reducing both i and j, redistributing

                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                freed = a[i]*(1-f) + a[j]*(1-f)
                
                # Redistribute to random low-contribution positions
                low_mask = np.ones(n, dtype=bool)
                low_mask[idx] = False  # exclude peak contributors
                if low_mask.sum() > 0:
                    candidates = np.where(low_mask)[0]
                    # Pick positions with lowest contribution to peak
                    if len(candidates) > 0:
                        redist = rng.dirichlet(np.ones(len(candidates)) * 0.5)

                        a_trial = a_trial / a_trial.sum()
                        s_trial = 2.0 * n * fast_conv(a_trial).max()
                        if s_trial < best_s - 1e-15:
                            a = a_trial
                            best_s = s_trial
                            best_a = a.copy()
                            improved = True
                            break
            if improved:
                break
        
        if not improved:

                i = idx[ti]
                if a[i] < 1e-12:
                    continue
                for f in [0.97, 0.93, 0.88]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    freed = a[i] * (1 - f)
                    # Redistribute uniformly to zeros or low positions

                        a_trial[zero_mask] += freed / zero_mask.sum()

                        improved = True
                        break

                    break

            break
    
    return best_a, best_s

def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    if len(f) == new_n:
        return f.copy()
    old_n = len(f)
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new
