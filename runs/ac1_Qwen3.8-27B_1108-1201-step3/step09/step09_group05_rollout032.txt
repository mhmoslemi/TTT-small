import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def mirror_descent(a0, time_limit, eta_start=0.1, eta_end=0.001, n_decays=5):
    """Mirror descent with exponential weights on the simplex."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for decay in range(n_decays + 1):
        eta = eta_start * (eta_end / eta_start) ** (decay / n_decays)
        while time.monotonic() - t0 < time_limit:
            b = fast_convolve(a)
            k_star = int(np.argmax(b))
            
            # Subgradient of conv(a,a)[k_star] w.r.t. a
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            for i in range(js, je + 1):
                g[i] = 2.0 * a[k_star - i]  # gradient of a[i]*a[k*-i] w.r.t. a[i]
            
            # Mirror descent update: a_new[i] ∝ a[i] * exp(-eta * g[i])
            log_update = -eta * g
            log_update -= log_update.max()  # for numerical stability
            weights = np.exp(log_update)
            a_new = a * weights
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a, best_score

def pair_coord_descent(a0, time_limit, n_top=20, patience=500):

        low_coords = np.argsort(g)[:n_top]

            for j in low_coords:

                    a_trial[j] += delta
                    if a_trial[i] < 0:

                    a_trial = a_trial / a_trial.sum()

                    if score < best_score - 1e-14:

                    break

                    # Add to bottom half of coordinates by pressure

                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def structured_init(n, rng, mode=0):
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x)) * (1 + 0.2 * np.sin(6 * np.pi * x))
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 0.5 * (1 + 0.1 * np.cos(8 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def aggressive_local_search(a0, time_limit):
    """Multi-strategy local search: alternates between different descent directions."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    strategy = 0
    stall = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = 2.0 * a[k_star - i]

        improved = False

        if strategy == 0:
            # Single coordinate: move mass from highest pressure to lowest
            i_src = int(np.argmax(g * np.where(a > 1e-10, 1, 0)))
            i_dst = int(np.argmin(g))
            if a[i_src] > 1e-10:
                for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    delta = a[i_src] * frac
                    a_trial = a.copy()
                    a_trial[i_src] -= delta
                    a_trial[i_dst] += delta
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        elif strategy == 1:
            # Batch: move mass from top-5 to bottom-5
            top5 = np.argsort(g)[::-1][:5]
            bot5 = np.argsort(g)[:5]
            for scale in [0.02, 0.05, 0.1]:
                a_trial = a.copy()
                total_freed = 0.0
                for i in top5:
                    if a_trial[i] > 1e-10:
                        move = a_trial[i] * scale
                        a_trial[i] -= move
                        total_freed += move
                if total_freed < 1e-15:
                    continue
                for j in bot5:
                    if j not in top5:
                        a_trial[j] += total_freed / max(1, len([x for x in bot5 if x not in top5]))
                    else:
                        a_trial[j] += total_freed / 5
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        elif strategy == 2:
            # Pair reduction: reduce top-2 pair and redistribute
            top2 = np.argsort(g)[::-1][:2]
            for f in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                freed = 0.0
                for i in top2:
                    freed += a_trial[i] * (1 - f)
                    a_trial[i] *= f
                if freed < 1e-15:
                    continue
                # Redistribute to coordinates with lowest g
                bottom = np.argsort(g)[:n // 3]
                bottom = [x for x in bottom if x not in top2]
                if bottom:
                    per = freed / len(bottom)
                    a_trial[bottom] += per
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        elif strategy == 3:
            # Multiplicative: scale all by exp(-small * g)
            for eta in [0.01, 0.03, 0.05]:
                log_w = -eta * g
                log_w -= log_w.max()
                w = np.exp(log_w)
                a_trial = a * w
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
            strategy = (strategy + 1) % 4
        else:
            stall += 1
            strategy = (strategy + 1) % 4
            if stall > 200:
                # Perturb to escape
                rng_local = np.random.default_rng(42 + int(time.monotonic() * 1000))
                a_pert = a + 0.01 * rng_local.standard_normal(n)
                a_pert = normalize(a_pert)
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score:
                    a = a_pert
                    best_score = score_pert
                    best_a = a.copy()
                else:
                    a = a_pert  # explore anyway
                stall = 0

        # Periodic check
        if (best_score < prev_score - 1e-10):
            prev_score = best_score
            stall = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Mirror Descent (200s) ===
    print(f"\n=== Phase 1: Mirror Descent n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _, iters = mirror_descent(best_seq, t_alloc, eta0=0.3, decay_rate=0.9997, min_eta=0.0005, momentum=0.8)
        try_update(a_md, "mirror_descent")

        # Second run with different params
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc2 = min(rem * 0.15, 150.0)
            a_md2, _, _ = mirror_descent(best_seq, t_alloc2, eta0=0.1, decay_rate=0.9999, min_eta=0.001, momentum=0.5)
            try_update(a_md2, "mirror_descent2")

    # === Phase 2: Alternating LP (200s) ===
    print(f"\n=== Phase 2: Alternating LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = alternating_lp(best_seq, t_alloc, solver, max_iters=15000)
        try_update(a_lp, "alt_lp")

    # === Phase 3: Aggressive Local Search (300s) ===
    print(f"\n=== Phase 3: Aggressive Local Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_local, _ = aggressive_local_search(best_seq, t_alloc)
        try_update(a_local, "agg_local")

    # === Phase 4: L-BFGS Polish (150s) ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=30.0, T_end=0.05, maxiter=6000, n_stages=10)
        try_update(a_lb, "lbgfs_polish")

    # === Phase 5: More local search after L-BFGS (150s) ===
    print(f"\n=== Phase 5: Local Search r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        a_local2, _ = aggressive_local_search(best_seq, t_alloc)
        try_update(a_local2, "agg_local2")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 1.33, 0.67, 2.0]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [640, 800, 1024, 1280, 1440, 1600]:
        if 256 <= fixed_n <= 4000 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.2, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        # Try both interpolated and fresh init
        init_alt = interpolate_sequence(best_seq, alt_n)
        t_sub1 = t_total * 0.4
        a_alt, _, _ = mirror_descent(init_alt, t_sub1, eta0=0.2, decay_rate=0.9998, min_eta=0.001, momentum=0.7)
        try_update(a_alt, f"md_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_sub2 = min(rem * 0.3, 80.0)
            a_alt2, _ = aggressive_local_search(best_seq, t_sub2)
            try_update(a_alt2, f"local_n{alt_n}")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # L-BFGS with very low temperature
        t_fin = min(rem * 0.3, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.02, maxiter=5000, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_local_fin = min(rem * 0.4, 50.0)
        a_local_fin, _ = aggressive_local_search(best_seq, t_local_fin)
        try_update(a_local_fin, "final_local")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_md_fin = min(rem * 0.5, 30.0)
        a_md_fin, _, _ = mirror_descent(best_seq, t_md_fin, eta0=0.05, decay_rate=0.9999, min_eta=0.005, momentum=0.9)
        try_update(a_md_fin, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
