import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def interp_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def compute_gradient_smoothed(a, conv, beta, top_k=None):
    """Compute smoothed gradient of max(conv) w.r.t. a using logsumexp."""
    n = len(a)
    M = conv.max()
    scaled = beta * (conv - M)
    scaled = np.clip(scaled, -60, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    if Z < 1e-30:
        weights = np.zeros_like(conv)
        weights[conv.argmax()] = 1.0
        Z = 1.0
    weights = weights / Z

    grad = np.zeros(n)
    if top_k is not None:
        top_idx = np.argsort(weights)[::-1][:top_k]
        active = top_idx[weights[top_idx] > 1e-10 / n]
    else:
        active = np.where(weights > 1e-10 / n)[0]

    for k in active:
        w = weights[k]
        if w < 1e-14:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w * a[k - j]

    return grad


def compute_gradient_stochastic(a, conv, n_peaks=5):
    """Compute stochastic subgradient by randomly selecting from argmax positions."""
    n = len(a)
    M = conv.max()
    # Find positions within 1% of max
    threshold = M * 0.995
    candidate_peaks = np.where(conv >= threshold)[0]
    if len(candidate_peaks) == 0:
        candidate_peaks = np.array([conv.argmax()])

    # Select random subset
    k = min(n_peaks, len(candidate_peaks))
    selected = np.random.choice(candidate_peaks, size=k, replace=False)

    grad = np.zeros(n)
    for k_pos in selected:
        i_start = max(0, k_pos - n + 1)
        i_end = min(n - 1, k_pos)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * a[k_pos - j] / k

    return grad

                factors.append(-f * a[i])
                factors.append(f * a[i])
            factors.append(-0.999 * a[i])

            for d in factors:
                nv = a[i] + d
                if nv < -1e-15 or nv > 1.5:
                    continue

                new_aff = affected + 2.0 * d * a
                # Fix the a[i]*a[i] term
                if 2 * i >= ki and 2 * i <= kmax:
                    idx = 2 * i - ki
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * d * a[i] + d * d

                nm = max(unaff_max, float(new_aff.max()))
                if nm < best_new_max - 1e-16:
                    best_new_max = nm
                    best_delta = d

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                old_ai = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a
                if 2 * i >= ki and 2 * i <= kmax:
                    idx = 2 * i - ki
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[ki:kmax + 1] += changes
                M = max(unaff_max, float(conv[ki:kmax + 1].max()))
                improved = True

                s = a.sum()
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweep % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            ki = i
            kmax = min(i + n - 1, 2 * n - 2)

            if ki > 0:
                unaff_l = float(conv[:ki].max())
            else:
                unaff_l = -1e30
            if kmax < 2 * n - 2:
                unaff_r = float(conv[kmax + 1:].max())
            else:
                unaff_r = -1e30
            unaff_max = max(unaff_l, unaff_r)

            affected = conv[ki:kmax + 1]
            best_delta = 0.0
            best_new_max = M

            for factor in [0.999, 0.998, 0.997, 0.995, 0.993,
                           1.001, 1.002, 1.003, 1.005, 1.007]:
                d = a[i] * (factor - 1.0)
                nv = a[i] + d
                if nv < -1e-15:
                    continue

                new_aff = affected.copy() + 2.0 * d * a
                if 2 * i >= ki and 2 * i <= kmax:
                    idx = 2 * i - ki
                    new_aff[idx] = conv[2 * i] + 2.0 * d * a[i] + d * d

                nm = max(unaff_max, float(new_aff.max()))
                if nm < best_new_max - 1e-17:
                    best_new_max = nm
                    best_delta = d

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                old_ai = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a
                if 2 * i >= ki and 2 * i <= kmax:
                    idx = 2 * i - ki
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[ki:kmax + 1] += changes
                M = max(unaff_max, float(conv[ki:kmax + 1].max()))
                improved = True

                s = a.sum()
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_targeted_perturb(a0, rng, strength=0.03):
    """
    Perturbation that specifically reduces coordinates contributing to convolution peaks.
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    # Find peak positions
    threshold = M * 0.97
    peak_positions = np.where(conv >= threshold)[0]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[j]

    # Normalize
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Apply: reduce high-pressure, add noise
    noise = rng.standard_normal(n) * 0.3
    a_new = a * np.exp(-strength * pressure + strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def multiplicative_perturb(a0, rng, strength=0.05):
    """Pure multiplicative noise perturbation."""
    n = len(a0)
    noise = rng.standard_normal(n)
    a_new = a0 * np.exp(strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restart_perturb(a0, rng, frac_zero=0.1):
    """Zero out a random fraction and add small random mass elsewhere."""
    n = len(a0)
    a = a0.copy()
    mask = rng.random(n) < frac_zero
    a[mask] = 0
    # Add small mass to some zeroed positions
    zero_idx = np.where(a < 1e-15)[0]
    if len(zero_idx) > 0:
        n_add = min(len(zero_idx), max(5, n // 20))
        add_idx = rng.choice(zero_idx, size=n_add, replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.3
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def init_gap(n, rng):
    """Uniform with a dip in the middle (to reduce central convolution peak)."""
    t = np.arange(n)
    center = n / 2.0
    width = n * 0.15
    a = np.ones(n) - 0.5 * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_w_shape(n, rng):
    """W-shaped: four peaks at quartile positions."""
    t = np.arange(n)
    a = np.ones(n) * 0.3
    for frac in [0.15, 0.35, 0.65, 0.85]:
        center = n * frac
        width = n * 0.05
        a += 0.8 * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_random_structured(n, rng):
    """Random superposition of Gaussians."""
    a = np.zeros(n)
    n_peaks = int(rng.integers(3, 10))
    t = np.arange(n)
    for _ in range(n_peaks):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.02, n * 0.15)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence, returning inf on error."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-length initialization + quick MGD
    Phase 2: Intensive MGD with varied hyperparameters
    Phase 3: Iterated perturbation + MGD (simulated annealing style)
    Phase 4: Fine coordinate descent polishing
    Phase 5: Length exploration
    Phase 6: Final fine polish
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Multi-length initialization + quick MGD ===
    print("\n=== Phase 1: Diverse Init + Quick MGD ===")
    t1_start = time.monotonic()
    phase1_budget = 200.0

    n_values = [1024, 1280, 1536, 2048]
    if parent is not None and len(parent) not in n_values:
        n_values.insert(0, len(parent))

    candidates = []

    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > phase1_budget:
            break

        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interp_seq(parent, n)))
        init_cands.append(("gap", init_gap(n, rng)))
        init_cands.append(("w_shape", init_w_shape(n, rng)))
        init_cands.append(("rand_struct", init_random_structured(n, rng)))
        init_cands.append(("uniform", np.ones(n) / n))

        for name, cand in init_cands:
            t_left = phase1_budget - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            t_alloc = max(3.0, min(10.0, t_left / 2.0))
            a_opt, s_opt = multiplicative_gradient_descent(
                cand, t_alloc, lr=0.1, beta=30, stochastic=False,
                verbose=False, label=f"p1_{name}_n{n}"
            )
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, top-3: " +
          ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in candidates[:3]]))

    for s_c, a_c, n_c, name_c in candidates[:6]:
        sc = evaluate_safe(a_c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST [P1 {name_c} n={n_c}]: {sc:.8f}")

    # === Phase 2: Intensive MGD ===
    print(f"\n=== Phase 2: Intensive MGD (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    phase2_budget = 350.0

    if best_seq is not None:
        n_best = len(best_seq)
        # Try different MGD configurations
        configs = [
            # (lr, beta, stochastic, n_peaks, label)
            (0.05, 50, False, None, "smooth_b50"),
            (0.03, 80, False, None, "smooth_b80"),
            (0.1, 30, False, None, "smooth_b30_fast"),
            (0.05, None, True, 3, "stoch_3"),
            (0.08, None, True, 5, "stoch_5"),

        for lr_c, beta_c, stoch_c, npeaks_c, label_c in configs:

                break

                best_seq, t_alloc, lr=lr_c,

                verbose=False, label=f"p2_{label_c}"
            )
            sc = evaluate_safe(a_opt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc

                print(f"  NEW BEST [P2 {label_c}]: {sc:.8f}")

    # === Phase 3: Iterated Perturbation + MGD (SA-style) ===
    print(f"\n=== Phase 3: Iterated Perturbation (best={best_score:.8f}) ===")

    if best_seq is not None:

            elif ptype == 1:

            elif ptype == 2:
                perturbed = random_restart_perturb(current, rng, frac_zero=0.05 + 0.1 * temp)
            else:
                perturbed = current * (1.0 + rng.standard_normal(n_best) * 0.02 * temp)
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()

            # Optimize from perturbed point
            t_alloc = min(30.0, t_left * 0.5)
            lr_try = 0.05 * (1.0 + temp)
            a_opt, s_opt = multiplicative_gradient_descent(
                perturbed, t_alloc, lr=lr_try, beta=40,
                stochastic=(rng.random() > 0.5), n_peaks=4,
                verbose=False, label=f"p3_{it}"

            sc = evaluate_safe(a_opt)

            # SA acceptance
            if np.isfinite(sc) and sc < best_score:
                improvement = best_score - sc
                best_score = sc
                best_seq = a_opt.copy()
                current = a_opt.copy()
                temp = max(0.1, temp * 0.95)
                no_improve = 0
                print(f"  NEW BEST [P3 it={it}]: {sc:.8f} (temp={temp:.3f})")
            elif np.isfinite(sc):
                # Accept with SA probability (only for close scores)
                delta = sc - best_score
                if delta < 0.01 and rng.random() < np.exp(-delta / (0.001 * temp + 1e-6)):
                    current = a_opt.copy()
                no_improve += 1
            else:
                no_improve += 1

            if no_improve > 20:

                no_improve = 0

                best_seq = a_opt.copy()

            a_opt, s_opt = fine_coord_descent(

                best_seq = a_opt.copy()

    if best_seq is not None:
