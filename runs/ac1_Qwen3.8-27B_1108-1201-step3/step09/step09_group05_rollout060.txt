    w = exp_b / exp_b.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP alternating for global optimization
2. FFT-based Adam gradient descent on log-sum-exp smoothed objective
3. Multi-length exploration with interpolation
4. Peak shaving for final refinement
Key: efficient gradient via FFT, smart time allocation, multi-start.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_general(a, b):
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    return irfft(rfft(a, fft_size) * rfft(b, fft_size), fft_size)[:size]


def norm_seq(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_score(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def smooth_grad(a, T):
    """Gradient of T*log(sum(exp(conv(a,a)/T))) w.r.t. a, via FFT."""
    n = len(a)
    b = fast_convolve(a)
    bs = b / T
    bs -= bs.max()
    w = np.exp(bs)
    w /= w.sum()
    # grad[m] = 2 * sum_j a[j] * w[j+m]
    # = 2 * (conv(a, flip(w)))[2n-2-m]
    cw = fast_conv_general(a, w[::-1])
    # cw has length n + (2n-1) - 1 = 3n-2
    # We want indices 2n-2 down to n-1
    return 2.0 * cw[n-1:2*n-1][::-1]


def build_T_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def lp_best_response(a_fixed):
    """Solve LP: min M s.t. conv(a_fixed, b)[k] <= M for all k, b>=0, sum(b)=1."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T_matrix(a_fixed)
    prob = cp.Problem(cp.Minimize(M_var), [T @ b_var <= M_var, cp.sum(b_var) == 1])
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

def adam_smoothed(a0, time_limit, T_start=50.0, T_end=0.5, max_iters=20000):
    """Adam gradient descent on smoothed log-sum-exp objective in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-20))
    z -= z.max()  # for numerical stability
    best = norm_seq(a0)
    best_sc = get_score(best)
    t0 = time.monotonic()
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    it = 0
    
    n_T = 8
    temps = np.geomspace(T_start, T_end, n_T)
    iters_per_T = max(10, max_iters // n_T)
    
    for Ti in temps:
        T = float(Ti)
        lr = 0.3 / (1 + it * 0.01)
        
        for _ in range(iters_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            if it >= max_iters:
                break
            
            a_cur = np.exp(z)
            a_cur /= a_cur.sum()
            
            g_a = smooth_grad(a_cur, T)
            # Chain rule for softmax parameterization
            S = np.dot(g_a, a_cur)
            g_z = a_cur * (g_a - S)
            
            it += 1
            m_adam = beta1 * m_adam + (1 - beta1) * g_z
            v_adam = beta2 * v_adam + (1 - beta2) * g_z ** 2
            m_hat = m_adam / (1 - beta1 ** it)
            v_hat = v_adam / (1 - beta2 ** it)
            z -= lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.max()  # renormalize for stability
            
            if it % 200 == 0:
                a_check = np.exp(z)
                a_check /= a_check.sum()
                sc = get_score(a_check)
                if sc < best_sc:
                    best_sc = sc
                    best = a_check.copy()
        
        if time.monotonic() - t0 > time_limit:
            break
    
    # Final check
    a_final = np.exp(z)
    a_final /= a_final.sum()
    sc = get_score(a_final)
    if sc < best_sc:
        best_sc = sc
        best = a_final.copy()
    
    return best, best_sc


def peak_shave(a0, time_limit, max_steps=10000):
    """Iterative peak shaving: reduce the peak of autocorrelation."""
    n = len(a0)
    a = norm_seq(a0)
    best = a.copy()
    best_sc = get_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find interior peak (avoid boundaries)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        k_peak = int(np.argmax(b_work))
        
        # Find top contributors to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        top = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top:
            i = idx[ci]
            j = k_peak - i
            if i == j:
                # Single element contributes; reduce it
                for factor in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = norm_seq(a_trial)
                    sc = get_score(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best = a_trial.copy()
                        a = a_trial
                        improved = True
                        break
            else:
                # Pair (i, j) contributes; reduce one or both
                for fi in [0.95, 0.90]:
                    for fj in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= fi
                        a_trial[j] *= fj
                        a_trial = norm_seq(a_trial)
                        sc = get_score(a_trial)
                        if sc < best_sc:
                            best_sc = sc
                            best = a_trial.copy()
                            a = a_trial
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best, best_sc


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp = f_interp / s
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP alternating for global optimization
2. FFT-based Adam gradient descent on log-sum-exp smoothed objective
3. Multi-length exploration with warm-start interpolation
4. Peak shaving for final refinement
5. Adaptive time allocation and multi-start perturbation
Summary: Combines LP best-response (global) with smoothed gradient descent (local),
explores multiple lengths and perturbations to escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_general(a, b):
    """Linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    return irfft(rfft(a, fft_size) * rfft(b, fft_size), fft_size)[:size]


def norm_seq(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_score(a):
    """Compute 2*n*max_conv for normalized sequence (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def smooth_grad_fft(a, T):
    """Gradient of T*log(sum(exp(conv(a,a)/T))) w.r.t. a, computed via FFT in O(n log n)."""
    n = len(a)
    b = fast_convolve(a)
    bs = b / T
    bs -= bs.max()
    w = np.exp(bs)
    w /= w.sum()
    # grad[m] = 2 * sum_j a[j] * w[j+m]
    # = 2 * (conv(a, flip(w)))[2n-2-m] for m=0..n-1
    cw = fast_conv_general(a, w[::-1])
    return 2.0 * cw[n - 1:2 * n - 1][::-1]


def build_T_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a
    return T


def lp_best_response(a_fixed):
    """Solve LP: min M s.t. conv(a_fixed, b)[k] <= M for all k, b>=0, sum(b)=1."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T_matrix(a_fixed)
    prob = cp.Problem(cp.Minimize(M_var), [T @ b_var <= M_var, cp.sum(b_var) == 1])
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

def peak_shave(a0, time_limit, max_steps=15000):
    """Iterative peak shaving: reduce the peak of autocorrelation."""
    n = len(a0)
    a = norm_seq(a0)
    best = a.copy()
    best_sc = get_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        k_peak = int(np.argmax(b_work))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        top = np.argsort(contribs)[::-1][:6]

        improved = False
        for ci in top:
            i = idx[ci]
            j = k_peak - i
            if i == j:
                for factor in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = norm_seq(a_trial)
                    sc = get_score(a_trial)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best = a_trial.copy()
                        a = a_trial

            else:

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp = f_interp / s
    return f_interp

        try_update(a2, "adam_main")

    # === Phase 3: Second LP round (shorter) ===
    print(f"\n=== Phase 3: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, s3 = run_phase_lp(best_seq, t_alloc, 200000, 0.04)
        try_update(a3, "lp_r2")

    # === Phase 4: Adam polish (low temperature) ===
    print(f"\n=== Phase 4: Adam Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s, T: 10->0.1")
        a4, s4 = run_phase_adam(best_seq, t_alloc, 10.0, 0.1, 15000)
        try_update(a4, "adam_polish")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 1.33, 2.0, 0.85]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [512, 640, 896, 1024, 1152, 1280, 1536, 2048]:
        if 256 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))[:4]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.2, 120.0)
        if t_total < 25:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_seq(best_seq, alt_n)

        # LP on alt length
        t_lp = t_total * 0.55
        a_alt, _ = run_phase_lp(init_alt, t_lp, 100000, 0.05)
        try_update(a_alt, f"lp_n{alt_n}")

        # Adam on alt length
        rem = deadline - time.monotonic()
        if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
            t_ad = min(rem * 0.35, 50.0)
            if t_ad > 10:
                a_alt2, _ = run_phase_adam(best_seq, t_ad, 20.0, 0.2, 8000)
                try_update(a_alt2, f"adam_n{alt_n}")

    # === Phase 6: Perturbation escape ===
    print(f"\n=== Phase 6: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        for scale in [0.06, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_seq(perturbed)
            t_sub = min(rem * 0.3, 80.0)
            t_lp_p = t_sub * 0.5
            a_p, _ = run_phase_lp(perturbed, t_lp_p, 50000, 0.03)
            try_update(a_p, f"pert_{scale}")

            rem = deadline - time.monotonic()
            if rem > 15 and len(best_seq) == n_cur:
                t_ad_p = min(rem * 0.4, 30.0)
                if t_ad_p > 10:
                    a_p2, _ = run_phase_adam(best_seq, t_ad_p, 5.0, 0.15, 5000)
                    try_update(a_p2, f"pert_adam_{scale}")

    # === Phase 7: Peak shaving ===
    print(f"\n=== Phase 7: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.25, 60.0)
        a_ps, _ = peak_shave(best_seq, t_ps, max_steps=20000)
        try_update(a_ps, "peak_shave")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.35, 40.0)
        a_fin, _ = run_phase_adam(best_seq, t_fin, 3.0, 0.05, 5000)
        try_update(a_fin, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.4, 25.0)
        a_fp, _ = run_phase_lp(best_seq, t_lp_fin, 20000, 0.02)
        try_update(a_fp, "final_lp")

    # Final peak shave if time allows
    rem = deadline - time.monotonic()
    if rem > 10:
        a_sp, _ = peak_shave(best_seq, min(rem * 0.5, 20.0), max_steps=5000)
        try_update(a_sp, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(256) / 256
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
