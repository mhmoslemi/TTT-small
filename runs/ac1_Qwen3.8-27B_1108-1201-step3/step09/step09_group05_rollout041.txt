            x = cp.Variable(3, nonneg=True)

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    global _lp_solver
    if solver is None:
        solver = _lp_solver if _lp_solver else get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_sym(a0, time_limit, solver=None, n_max_iters=80000):
    """Nesterov LP with symmetrization and line search."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    a = symmetrize(a)
    a = norm_sequence(a)
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Symmetrize the LP solution
        b_new = symmetrize(b_new)
        b_new = norm_sequence(b_new)

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Line search: try different mixtures
        best_t_score = best_score
        best_t_a = a.copy()
        candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        if it < 10:
            candidates = [0.1, 0.2, 0.3, 0.5]
        for t_val in candidates:
            a_trial = (1.0 - t_val) * a + t_val * b_new
            a_trial = np.maximum(a_trial, 0)
            a_trial = symmetrize(a_trial)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = score_normalized(a_trial)
            if sc < best_t_score:
                best_t_score = sc
                best_t_a = a_trial.copy()

        a = best_t_a
        if best_t_score < best_score:
            best_score = best_t_score
            best_a = a.copy()
            stall_count = 0

        if it % 200 == 0 and it > 0:
            if best_score < prev_score - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 5:
                rng_local = np.random.default_rng(777 + it)
                noise = rng_local.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_function(a0, time_limit, T_start, T_end, n_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing, with symmetrization."""
    n = len(a0)
    a = symmetrize(norm_sequence(a0))
    z = z_from_a(a).copy()
    t0 = time.monotonic()

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        T_val = T_start * (T_end / T_start) ** (stage / max(n_stages - 1, 1))

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = softmax_from_z(z_in)
            aa = symmetrize(aa)
            aa = norm_sequence(aa)
            b = fast_convolve(aa)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            aa = softmax_from_z(z_in)
            aa = symmetrize(aa)
            aa = norm_sequence(aa)
            b = fast_convolve(aa)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * aa[k - idx]
            # Symmetrize gradient
            grad_a = (grad_a + grad_a[::-1].copy()) / 2.0
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a = softmax_from_z(z)
        a = symmetrize(a)
        a = norm_sequence(a)

    return a


def coord_descent_polish(a0, time_limit, n_steps=200000):
    """O(n) coordinate descent with direct score delta evaluation."""
    n = len(a0)
    a = symmetrize(norm_sequence(a0))
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(42)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 2000:
            break

        # Pick a coordinate to modify (cycle through, with some randomness)
        if step % 3 == 0:
            # Target the peak contributors
            peak_k = int(np.argmax(b))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contribs = a[js:je+1] * a[peak_k - (js:je+1)]
            top_idx = np.argsort(contribs)[::-1][:10] + js
            i = int(top_idx[int(rng.integers(len(top_idx)))])
        else:
            i = int(rng.integers(n))

        if a[i] < 1e-14:
            # Try adding mass
            deltas = [0.001, 0.005, 0.01]
            best_d = None
            best_ds = best_score
            for d in deltas:
                # delta_b[k] = 2*d*a[k-i] for k in [i, i+n-1]
                db = np.zeros_like(b)
                k_start = i
                k_end = min(i + n, len(b))
                db[k_start:k_end] = 2.0 * d * a[:k_end - k_start]
                if 2 * i < len(b):
                    db[2 * i] += d * d
                new_max = (b + db).max()
                sc = 2.0 * n * new_max
                if sc < best_ds:
                    best_ds = sc
                    best_d = d
            if best_d is not None:
                a[i] += best_d
                a = symmetrize(a)
                a = norm_sequence(a)
                b = fast_convolve(a)
                sc = 2.0 * n * b.max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                continue
            else:
                no_improve += 1
                continue

        # Try reducing a[i]
        best_d = None
        best_ds = best_score
        for frac in [0.95, 0.90, 0.85, 0.80, 0.70]:
            d = a[i] * (frac - 1.0)
            # delta_b[k] = 2*d*a[k-i] for k in [i, i+n-1]
            db = np.zeros_like(b)
            k_start = i
            k_end = min(i + n, len(b))
            db[k_start:k_end] = 2.0 * d * a[:k_end - k_start]
            if 2 * i < len(b):
                db[2 * i] += d * d
            new_max = (b + db).max()
            sc = 2.0 * n * new_max
            if sc < best_ds:
                best_ds = sc
                best_d = d

        if best_d is not None:
            a[i] += best_d
            a = np.maximum(a, 0)
            a = symmetrize(a)
            a = norm_sequence(a)
            b = fast_convolve(a)
            sc = 2.0 * n * b.max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Try pair move: reduce a[i], add to a[j]
            j = n - 1 - i  # symmetric partner
            if j == i:
                j = int(rng.integers(n))
            if a[j] >= a[i]:
                j = int(rng.integers(n))
            if a[j] < 1e-14 or j == i:
                no_improve += 1
                continue

            # Move delta from i to j
            best_d = None
            best_ds = best_score
            for frac in [0.1, 0.2, 0.3, 0.5]:
                d = a[i] * frac
                # delta_b from changing a[i] by -d and a[j] by +d
                db = np.zeros_like(b)
                # Effect of a[i] -= d
                k_start_i = i
                k_end_i = min(i + n, len(b))
                db[k_start_i:k_end_i] += 2.0 * (-d) * a[:k_end_i - k_start_i]
                if 2 * i < len(b):
                    # Second order: (a[i]-d)^2 - a[i]^2 = -2*d*a[i] + d^2
                    # Already accounted in linear term, add d^2
                    db[2 * i] += d * d
                # Effect of a[j] += d
                k_start_j = j
                k_end_j = min(j + n, len(b))
                db[k_start_j:k_end_j] += 2.0 * d * a[:k_end_j - k_start_j]
                if 2 * j < len(b):
                    db[2 * j] += d * d
                # Cross term: 2*d*(-d) at k=i+j (if valid)
                if i + j < len(b):
                    db[i + j] += 2.0 * (-d) * d

                new_max = (b + db).max()
                sc = 2.0 * n * new_max
                if sc < best_ds:
                    best_ds = sc
                    best_d = d

            if best_d is not None:
                a[i] -= best_d
                a[j] += best_d
                a = np.maximum(a, 0)
                a = symmetrize(a)
                a = norm_sequence(a)
                b = fast_convolve(a)
                sc = 2.0 * n * b.max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

    return best_a, best_score


def peak_shave_direct(a0, time_limit, n_steps=100000):
    """Direct peak shaving: reduce top contributors to the maximum peak."""
    n = len(a0)
    a = symmetrize(norm_sequence(a0))
    b = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break

        # Find the peak (exclude edges which are structurally smaller)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing each top contributor
        top_idx = np.argsort(contribs)[::-1][:12]
        improved = False
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-12:
                continue
            for frac in [0.97, 0.94, 0.90, 0.85, 0.78]:
                d = a[i] * (frac - 1.0)
                db = np.zeros_like(b)
                k_start = i
                k_end = min(i + n, len(b))
                db[k_start:k_end] = 2.0 * d * a[:k_end - k_start]
                if 2 * i < len(b):
                    db[2 * i] += d * d
                new_max = (b + db).max()
                sc = 2.0 * n * new_max
                # After normalization the score changes slightly, but this is a good approximation
                if sc < best_score * 0.9999:
                    a[i] *= frac
                    a = np.maximum(a, 0)
                    # Redistribute the removed mass to low-contrib positions
                    removed = a[i] * (1 - frac) + a[i] * (1 - frac)  # symmetrized
                    # Actually just renormalize
                    a = symmetrize(a)
                    a = norm_sequence(a)
                    b = fast_convolve(a)
                    sc = 2.0 * n * b.max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break
            if improved:
                break

        if not improved:
            no_improve += 1

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence (palindromic)."""
    if mode == 0:
        x = np.linspace(0, np.pi / 2, n)
        a = np.cos(x) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.2
    elif mode == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x)) * (1 + 0.2 * np.cos(6 * np.pi * x))
    elif mode == 5:
        # Bimodal: peaks at edges
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * x * (1 - x)) + 0.3 * np.exp(-5 * x * (1 - x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    a = symmetrize(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize(f_interp)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lbgfs_aggressive(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=15):
    """Aggressive L-BFGS with more stages and tighter tolerance."""
    n = len(a0)
    a_start = symmetrize(norm_sequence(a0))
    z = z_from_a(a_start).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = softmax_from_z(z_in)
            aa = symmetrize(aa)
            aa = norm_sequence(aa)
            b = fast_convolve(aa)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            aa = softmax_from_z(z_in)
            aa = symmetrize(aa)
            aa = norm_sequence(aa)
            b = fast_convolve(aa)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * aa[k - idx]
            grad_a = (grad_a + grad_a[::-1].copy()) / 2.0
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)

        # More iterations at lower temperatures
        if si < n_stages // 3:
            maxiter = 2000
        elif si < 2 * n_stages // 3:
            maxiter = 4000
        else:
            maxiter = 6000

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxfun': maxiter * 1000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    a_out = symmetrize(a_out)
    a_out = norm_sequence(a_out)
    return a_out


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Nesterov LP with symmetrization
    Phase 2: Aggressive L-BFGS
    Phase 3: O(n) coordinate descent
    Phase 4: Peak shaving
    Phase 5: Multi-length exploration
    Phase 6: Perturbation + re-optimize
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent = symmetrize(parent)
            parent = norm_sequence(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = symmetrize(a)
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov LP (300s) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 320.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_sym(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 2: Aggressive L-BFGS (400s) ===
    print(f"\n=== Phase 2: L-BFGS Aggressive (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 50->0.05")
        a_lb = lbgfs_aggressive(best_seq, t_alloc, T_start=50.0, T_end=0.05, n_stages=15)
        try_update(a_lb, "lbgfs_agg")

    # === Phase 3: Coordinate Descent (200s) ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coord_descent_polish(best_seq, t_alloc, n_steps=300000)
        try_update(a_cd, "coord_descent")

    # === Phase 4: Peak Shaving (100s) ===
    print(f"\n=== Phase 4: Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shave_direct(best_seq, t_alloc, n_steps=100000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Second L-BFGS polish (200s) ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.13, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 10->0.02")
        a_lb2 = lbgfs_aggressive(best_seq, t_alloc, T_start=10.0, T_end=0.02, n_stages=12)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 6: Multi-length exploration (300s) ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 2.0, 1.33, 0.75]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [768, 1024, 1280, 1536, 2048, 2560, 3072]:
        if 384 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.3, 150.0)
        if t_total < 40:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick L-BFGS
        t_lb_alt = t_total * 0.6
        a_alt = lbgfs_aggressive(init_alt, t_lb_alt, T_start=30.0, T_end=0.05, n_stages=10)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        # Quick coord descent
        rem = deadline - time.monotonic()
        if rem > 20 and len(best_seq) == alt_n:
            t_cd_alt = min(rem * 0.3, 40.0)
            if t_cd_alt > 10:
                a_alt2, _ = coord_descent_polish(best_seq, t_cd_alt, n_steps=50000)
                try_update(a_alt2, f"cd_n{alt_n}")

    # === Phase 7: Return to best length, final polish (200s) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin_cd = min(rem * 0.3, 60.0)
        if t_fin_cd > 10:
            a_fin1, _ = coord_descent_polish(best_seq, t_fin_cd, n_steps=100000)
            try_update(a_fin1, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin_lb = min(rem * 0.4, 80.0)
        if t_fin_lb > 15:
            a_fin2 = lbgfs_aggressive(best_seq, t_fin_lb, T_start=5.0, T_end=0.01, n_stages=10)
            try_update(a_fin2, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin_cd2 = min(rem * 0.3, 30.0)
        if t_fin_cd2 > 5:
            a_fin3, _ = coord_descent_polish(best_seq, t_fin_cd2, n_steps=50000)
            try_update(a_fin3, "final_cd2")

    # === Phase 8: Perturbation escape (100s) ===
    print(f"\n=== Phase 8: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        for scale in [0.02, 0.05]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = symmetrize(perturbed)
            perturbed = norm_sequence(perturbed)

            t_sub = min(rem * 0.4, 40.0)
            if t_sub > 10:
                a_pert = lbgfs_aggressive(perturbed, t_sub, T_start=20.0, T_end=0.05, n_stages=8)
                try_update(a_pert, f"pert_{scale}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
