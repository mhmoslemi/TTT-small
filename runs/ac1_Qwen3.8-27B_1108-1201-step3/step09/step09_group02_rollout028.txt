"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed max (log-sum-exp) gradient with adaptive temperature cooling
2. Multiplicative weights (mirror descent) and Euclidean projection updates
3. Exact coordinate descent with golden section for polishing
4. Basin hopping with structured perturbations
5. Multi-length exploration with adaptive time allocation
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_extract(W, a):
    """
    Compute g[i] = sum_{j=0}^{n-1} W[i+j] * a[j] for i in [0, n-1].
    W has length 2n-1, a has length n.
    Uses FFT for efficiency.
    """
    n = len(a)
    W_rev = W[::-1]  # reverse W, length 2n-1
    # Linear convolution of W_rev (len 2n-1) and a (len n) -> length 3n-2
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    Wf = rfft(W_rev, fft_size)
    Af = rfft(a, fft_size)
    conv_result = irfft(Wf * Af, fft_size)[:conv_len]
    # g[i] = conv_result[2n-2-i] for i = 0, ..., n-1
    g = conv_result[2 * n - 2 - np.arange(n)]
    return g

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_smooth_grad(a, conv, T):
    """
    Compute gradient of F(a) = T * log(sum(exp(conv[k]/T)))
    where conv = auto-convolution of a.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax weights
    shifted = (conv - conv.max()) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    g = cross_corr_extract(w, a)
    return 2.0 * g, w

def smooth_obj_value(conv, T):
    """Value of smooth objective T*log(sum(exp(conv/T)))."""
    shifted = (conv - conv.max()) / T
    return T * (np.log(np.sum(np.exp(shifted))) + conv.max() / T)

def multiplicative_update(a, grad, eta):
    """Mirror descent step: a * exp(-eta*grad) normalized."""
    log_update = -eta * grad
    # Clip for numerical stability
    log_update = np.clip(log_update, -50.0, 50.0)
    a_new = a * np.exp(log_update)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def euclidean_update(a, grad, eta):
    """Euclidean gradient step with simplex projection."""
    a_new = a - eta * grad
    return simplex_project(a_new)

def smooth_gradient_descent(a0, time_limit, T_init=None, T_min=None, 
                             use_multiplicative=True, verbose=True, label=""):
    """
    Smoothed max gradient descent with cooling schedule.
    a0 should be normalized (sum=1, non-negative).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    init_max = conv.max()
    
    if T_init is None:
        T_init = init_max * 0.1
    if T_min is None:
        T_min = init_max * 0.001

    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    
    # Cooling schedule
    T = T_init
    cool_factor = 0.65
    min_steps_per_phase = 50
    max_steps_per_phase = 5000
    
    phase = 0
    global_step = 0
    
    while time.monotonic() - t0 < time_limit:
        phase += 1
        phase_start = time.monotonic()
        steps = 0
        no_improve = 0
        best_phase_score = best_score
        best_phase_a = best_a.copy()
        prev_score = 2.0 * n * float(conv.max())
        eta = 1.0 / (2.0 * float(conv.max()) + 1e-10)
        
        while steps < max_steps_per_phase:
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - phase_start > time_limit * 0.15 and steps > min_steps_per_phase:
                break
            
            # Compute gradient
            grad, w = compute_smooth_grad(a, conv, T)
            steps += 1
            global_step += 1
            
            # Adaptive learning rate
            grad_norm = np.abs(grad).max()
            if grad_norm < 1e-15:
                break
            eta = 1.0 / (grad_norm + 1e-10)
            
            # Try update
            if use_multiplicative:
                a_trial = multiplicative_update(a, grad, eta)
            else:
                a_trial = euclidean_update(a, grad, eta)
            
            # Check if it improved
            conv_trial = fast_convolve(a_trial)
            score_trial = 2.0 * n * float(conv_trial.max())
            
            if score_trial < prev_score:
                a = a_trial
                conv = conv_trial
                prev_score = score_trial
                no_improve = 0
                if score_trial < best_phase_score:
                    best_phase_score = score_trial
                    best_phase_a = a.copy()
            else:
                no_improve += 1
                eta *= 0.5
                if no_improve > 100:
                    break
        
        # Check improvement
        if best_phase_score < best_score:
            best_score = best_phase_score
            best_a = best_phase_a.copy()
            if verbose and phase % 2 == 0:
                elapsed = time.monotonic() - t0
                print(f"    SGD {label} phase={phase} T={T:.6f} step={steps} "
                      f"({elapsed:.0f}s): best={best_score:.8f}")
        
        # Cool down
        T *= cool_factor
        if T < T_min:
            break
        
        if verbose and phase % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} phase={phase} T={T:.6f} "
                  f"({elapsed:.0f}s): cur={2*n*float(conv.max()):.8f} best={best_score:.8f}")

    return best_a, best_score

def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def make_score_func(c_vals, a_partners, unaff_max, has_quad, quad_idx, 
                    current_sum, v, n, x):
    """Compute score for a given coordinate value x."""
    ns = current_sum + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            quad_idx = 2 * i - k_lo if has_quad else -1

            upper = max(2.0 * v, 5.0 / n)
            best_x = golden_section_search(
                lambda x, _cv=c_vals, _ap=a_partners, _um=unaff_max,
                _hq=has_quad, _qi=quad_idx, _cs=current_sum, _v=v, _n=n:
                make_score_func(_cv, _ap, _um, _hq, _qi, _cs, _v, _n, x),
                0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v

                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_quadratic_residue(n, rng):
    """Initialize using quadratic residues mod p (known good autocorrelation)."""
    # Find a prime p close to n
    p = int(np.round(n))
    while True:
        if p > 2 and all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)):
            break
        p += 1
    
    # Create sequence based on quadratic residues
    a = np.zeros(p)
    for x in range(1, p):
        a[x * x % p] = 1.0
    a[0] = 1.0
    
    # Add small random perturbation and reshape to desired length
    a = a + rng.standard_normal(p) * 0.05
    a = np.maximum(a, 0.0)
    
    # Resample to length n if needed
    if p != n:
        old_x = np.linspace(0, 1, p)
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, a)
    
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_random_smooth(n, rng):
    """Random smooth initialization using low-pass filtered noise."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    # Low-pass: only keep low frequencies
    cutoff = rng.integers(3, max(4, n // 10))
    spectrum = np.zeros(n_freq, dtype=complex)
    for k in range(min(cutoff, n_freq)):
        r = rng.uniform(0.5, 1.5)
        phi = rng.uniform(0, 2 * np.pi)
        spectrum[k] = r * np.exp(1j * phi)
        if k > 0:
            spectrum[-k] = np.conj(spectrum[k])
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def basin_hopping_smooth(a0, time_limit, rng, n_perturbs=6, verbose=False):
    """Basin hopping using smoothed gradient descent for re-optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.7, 120.0)

        # Perturb
        sigma = 0.02 + 0.04 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Peak-breaking perturbation
        conv_p = fast_convolve(a)
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 4)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 + rng.standard_normal() * 0.06)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Re-optimize with smoothed gradient
        a_opt, s_opt = smooth_gradient_descent(perturbed, t_alloc, 
                                                T_init=conv_p.max() * 0.05,
                                                T_min=conv_p.max() * 0.001,
                                                verbose=False, label=f"BH{p}")
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score

    best_a = a.copy()
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    
    T = conv.max() * 0.08
    eta_mult = 1.0
    eta_add = 1.0
    step = 0
    cool_count = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        if step % 5000 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 5:
                break
        
        # Compute gradient
        grad, w = compute_smooth_grad(a, conv, T)
        grad_max = np.abs(grad).max()
        if grad_max < 1e-15:
            break
        
        # Adaptive step sizes
        eta_mult = 0.8 / grad_max
        eta_add = 0.5 / grad_max
        
        # Multiplicative update
        a_mult = multiplicative_update(a, grad, eta_mult)
        conv_mult = fast_convolve(a_mult)
        score_mult = 2.0 * n * float(conv_mult.max())
        
        # Additive update
        a_add = euclidean_update(a, grad, eta_add)
        conv_add = fast_convolve(a_add)
        score_add = 2.0 * n * float(conv_add.max())
        
        # Pick better
        if score_mult <= score_add:
            a_new, conv_new, score_new = a_mult, conv_mult, score_mult
        else:
            a_new, conv_new, score_new = a_add, conv_add, score_add
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            cool_count = 0
        else:
            a = a_new  # Still accept, just not best
            conv = conv_new
            cool_count += 1
            if cool_count > 200:
                T *= 0.7
                cool_count = 0
        
        # Slow cooling
        if step % 200 == 0:
            T *= 0.95
            T = max(T, conv.max() * 0.001)
        
        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualAvg {label} step={step} T={T:.6f} ({elapsed:.0f}s): "
                  f"best={best_score:.8f}")

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Smoothed gradient descent from parent with cooling
    Phase 2: Dual averaging / mirror descent
    Phase 3: Multi-length exploration
    Phase 4: Basin hopping
    Phase 5: Exact coordinate descent polish
    Phase 6: Fine coordinate descent
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent found, using uniform. score={best_score:.8f}")

    # === Phase 1: Smoothed gradient descent from parent ===
    print(f"\n=== Phase 1: Smoothed GD from parent (best={best_score:.8f}) ===")
    n_parent = len(parent)
    t1_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    print(f"  n={n_parent}, budget={t1_budget:.0f}s")
    
    a_sg, s_sg = smooth_gradient_descent(
        parent, t1_budget, 
        T_init=fast_convolve(parent).max() * 0.08,
        T_min=fast_convolve(parent).max() * 0.001,
        use_multiplicative=True,
        verbose=True, label="p1")
    
    best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "smooth_gd")
    
    # Also try additive variant
    rem = deadline - time.monotonic()
    if rem > 100:
        t1b_budget = min(200.0, rem * 0.15)
        print(f"\n  Phase 1b: Additive variant, budget={t1b_budget:.0f}s")
        a_sg2, s_sg2 = smooth_gradient_descent(
            a_sg, t1b_budget,
            T_init=fast_convolve(a_sg).max() * 0.05,
            T_min=fast_convolve(a_sg).max() * 0.001,
            use_multiplicative=False,
            verbose=True, label="p1b")
        best_seq, best_score = check_and_update(a_sg2, best_seq, best_score, "smooth_gd_add")

    # === Phase 2: Dual averaging / mirror descent ===
    print(f"\n=== Phase 2: Dual Avg (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t2_budget = min(350.0, rem * 0.3)
        print(f"  budget={t2_budget:.0f}s")
        a_da, s_da = dual_avg_descent(best_seq, t2_budget, verbose=True, label="p2")
        best_seq, best_score = check_and_update(a_da, best_seq, best_score, "dual_avg")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        n_parent = len(best_seq)
        n_try_list = [n_parent]
        for alt_n in [768, 1024, 1280, 1536, 1920, 2048]:
            if alt_n not in n_try_list and 256 <= alt_n <= 2560:
                n_try_list.append(alt_n)
        n_try_list = list(set(n_try_list))[:5]
        
        for n_try in n_try_list:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  n={n_try}, budget={t_alloc:.0f}s")
            
            # Different initializations
            inits = []
            inits.append(('interp', interpolate_seq(best_seq, n_try)))
            inits.append(('flat', init_flat_spectrum(n_try, rng)))
            inits.append(('quadres', init_quadratic_residue(n_try, rng)))
            inits.append(('smooth_rand', init_random_smooth(n_try, rng)))
            
            best_for_n = None
            best_score_n = float('inf')
            
            for itype, init in inits[:3]:
                rem_inner = deadline - time.monotonic()
                if rem_inner < 20:
                    break
                t_inner = min(t_alloc / 3.0, rem_inner * 0.4)
                a_init_opt, s_init_opt = smooth_gradient_descent(
                    init, t_inner,
                    T_init=fast_convolve(init).max() * 0.06,
                    T_min=fast_convolve(init).max() * 0.002,
                    use_multiplicative=True,
                    verbose=False, label=f"p3_n{n_try}_{itype}")
                if s_init_opt < best_score_n:
                    best_score_n = s_init_opt
                    best_for_n = a_init_opt
            
            if best_for_n is not None:
                best_seq, best_score = check_and_update(
                    best_for_n, best_seq, best_score, f"multi_n{n_try}")

    # === Phase 4: Basin hopping ===
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t4_budget = min(300.0, rem * 0.3)
        print(f"  budget={t4_budget:.0f}s")
        a_bh, s_bh = basin_hopping_smooth(best_seq, t4_budget, rng, 
                                           n_perturbs=6, verbose=True)
        best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 5: Exact coordinate descent polish ===
    print(f"\n=== Phase 5: ECD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple rounds of ECD
        for rnd in range(5):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem * 0.35, 200.0)
            print(f"  ECD round {rnd+1}, budget={t_alloc:.0f}s")
            a_cd, s_cd, sweeps = exact_coord_descent(
                best_seq, t_alloc, verbose=True, label=f"ecd_r{rnd}")
            best_seq, best_score = check_and_update(
                a_cd, best_seq, best_score, f"ecd_r{rnd}")
            
            # Small perturbation between rounds if stuck
            if rnd < 4:
                perturbed = best_seq + rng.standard_normal(n_best) * 0.01
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                best_seq, best_score = check_and_update(
                    perturbed, best_seq, best_score, f"perturb_r{rnd}")

    # === Phase 6: Final fine polishing ===
    print(f"\n=== Phase 6: Fine Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_fine = min(rem * 0.6, 100.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(
                best_seq, t_fine, verbose=True, label="final")
            best_seq, best_score = check_and_update(
                a_fine, best_seq, best_score, "fine_cd")
        
        # One more ECD pass
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_cd2, s_cd2, _ = exact_coord_descent(
                best_seq, min(rem * 0.7, 60.0), label="last")
            best_seq, best_score = check_and_update(
                a_cd2, best_seq, best_score, "ecd_last")
        
        # Final fine
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_fin, s_fin, _ = fine_coord_descent(
                best_seq, min(rem * 0.8, 30.0), label="ultimate")
            best_seq, best_score = check_and_update(
                a_fin, best_seq, best_score, "ultimate")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
