import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(conv, a, tau):
    """Compute gradient of smoothed max using FFT. O(n log n)."""
    n = len(a)
    conv_shifted = (conv - conv.max()) / tau
    w = np.exp(conv_shifted)
    w /= w.sum()
    
    # g[i] = 2 * sum_j w[i+j] * a[j] = 2 * (w * a_rev)[n-1+i]
    a_rev = a[::-1]
    total_len = (2*n - 1) + n - 1  # = 3n-2
    N = 1 << (total_len - 1).bit_length()
    W_fft = rfft(w, N)
    a_rev_fft = rfft(a_rev, N)
    conv_res = irfft(W_fft * a_rev_fft, N)
    g = 2.0 * conv_res[n-1:2*n-1]
    return g

def adam_update(state, gradient, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. state = (m, v, t)."""
    m, v, t = state
    t += 1
    m = beta1 * m + (1 - beta1) * gradient
    v = beta2 * v + (1 - beta2) * gradient * gradient
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return update, (m, v, t)

        a = a / s

    v = np.zeros(n)
    t = 0
    lr = lr_init
    tau = tau_init

    it = 0

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_smoothed_gradient(conv, a, tau):
    """
    Compute gradient of tau*log(sum(exp(conv/tau))) w.r.t. a using FFT.
    O(n log n) computation.
    
    g[i] = 2 * sum_j w[i+j] * a[j] where w = softmax(conv/tau)
    Computed via: g = 2 * (w * a_rev)[n-1 : 2n-1]
    """
    n = len(a)
    conv_shifted = (conv - conv.max()) / tau
    w = np.exp(conv_shifted)
    w /= w.sum()
    
    a_rev = a[::-1]
    total_len = (2 * n - 1) + n - 1
    N = 1 << (total_len - 1).bit_length()
    W_fft = rfft(w, N)
    a_rev_fft = rfft(a_rev, N)
    conv_res = irfft(W_fft * a_rev_fft, N)
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    return g

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

        a = a / s

def _eval_delta(delta, aff_conv, a_shift, unaff, quad_idx):

        if val_quad > new_max:
            new_max = val_quad
    return max(unaff, new_max)

def fine_polish(a0, time_limit, verbose=True, label=""):

def fw_one_round(a, b, n_steps=50):
    """One round of FW alternating: optimize b given a, then a given b."""
    n = len(a)
    
    # Optimize b given a
    b_conv = convolve_ab(a, b)
    for step in range(n_steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad_slice))
        t = 2.0 / (step + 2.0)
        
        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t
    
    # Optimize a given b
    a_conv = convolve_ab(b, a)
    for step in range(n_steps):
        k_star = int(np.argmax(a_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad_slice))
        t = 2.0 / (step + 2.0)
        
        a_conv = (1.0 - t) * a_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        a_conv[ks_s:ks_e + 1] += t * b
        a = (1.0 - t) * a
        a[j_star] += t

    return a, b


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluator and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def optimize_at_length(n, time_budget, rng, verbose=True):
    """Run the full optimization pipeline at a given sequence length n."""
    label = f"n={n}"
    t0 = time.monotonic()
    
    # Generate diverse initializations
    inits = []
    inits.append(antipal_init(n, alpha=1.0, rng=rng))
    inits.append(antipal_init(n, alpha=2.0, rng=rng))
    inits.append(power_ramp_init(n, p=1.5, rng=rng))
    inits.append(power_ramp_init(n, p=3.0, rng=rng))
    inits.append(two_hump_antipal_init(n, rng=rng))
    
    # Add parent if available and same length
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.shape == (n,):
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 0.01:
                inits.insert(0, p / p.sum())
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    # Score initializations
    for idx, init in enumerate(inits):
        s = init.sum()
        if s < 1e-15:
            continue
        init_norm = init / s
        conv = fast_convolve(init_norm)
        score = 2.0 * n * float(conv.max())
        if best_seq is None or score < best_score:
            best_score = score
            best_seq = init_norm.copy()

    if best_seq is None:
        best_seq = antipal_init(n, 1.0, rng)
        best_score = 2.0 * n * fast_convolve(best_seq).max()

    print(f"  {label} init best: {best_score:.6f}")

    # Phase 1: Smoothed gradient descent (50% of budget)
    t1 = time_budget * 0.45
    t1_start = time.monotonic()
    a_sg, s_sg = smoothed_gradient_optimize(best_seq, t1, tau_init=0.03, tau_min=0.003,
                                            lr_init=1.0, verbose=verbose, label="main")
    best_seq, best_score = check_and_update(a_sg, best_seq, best_score, f"sg_{label}")

    # Also run SG on a different init
    remaining = time_budget - (time.monotonic() - t0) - 30
    if remaining > 30:
        t1b = min(remaining * 0.2, 60.0)
        init2 = inits[1] if len(inits) > 1 else antipal_init(n, 1.5, rng)
        a_sg2, s_sg2 = smoothed_gradient_optimize(init2, t1b, tau_init=0.05, tau_min=0.005,
                                                   lr_init=0.8, verbose=False, label="alt")
        best_seq, best_score = check_and_update(a_sg2, best_seq, best_score, f"sg2_{label}")

    # Phase 2: Coordinate descent (35% of budget)
    remaining = time_budget - (time.monotonic() - t0) - 20
    if remaining > 20:
        t2 = min(remaining * 0.65, time_budget * 0.35)
        t2_start = time.monotonic()
        a_cd, s_cd = coord_descent_ternary(best_seq, t2, verbose=verbose, label="main")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_{label}")

    # Phase 3: Fine polish (15% of budget)
    remaining = time_budget - (time.monotonic() - t0) - 10
    if remaining > 15:
        t3 = min(remaining * 0.7, time_budget * 0.15)
        a_fp, s_fp = fine_polish(best_seq, t3, verbose=verbose, label="main")
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_{label}")

    # Phase 4: Another CD pass on the polished result
    remaining = time_budget - (time.monotonic() - t0) - 5
    if remaining > 15:
        t4 = min(remaining * 0.8, 60.0)
        a_cd2, s_cd2 = coord_descent_ternary(best_seq, t4, verbose=False, label="polish")
        best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, f"cd2_{label}")

    elapsed = time.monotonic() - t0
    print(f"  {label} done: best={best_score:.6f} ({elapsed:.0f}s)")
    
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2:
    1. Diverse anti-palindromic initializations
    2. FFT-based smoothed gradient descent
    3. Coordinate descent with ternary search (exact per-coordinate optimization)
    4. Peak-pressure fine polishing
    5. Multiple sequence lengths
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    parent_n = 1024
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={parent_n}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = antipal_init(1024, 1.5, rng)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=1024, score={best_score:.6f}")

    # Strategy: allocate time across different n values
    # Main effort on parent length, secondary on other lengths
    
    n_main = parent_n if parent is not None else 1024
    
    # Time allocation: 55% main, 25% n=2048, 20% n=1536
    remaining = deadline - time.monotonic() - 30
    
    # Main optimization at n_main
    t_main = remaining * 0.55
    print(f"\n=== Main optimization at n={n_main} (budget={t_main:.0f}s) ===")
    seq_main, score_main = optimize_at_length(n_main, t_main, rng, verbose=True)
    if score_main < best_score:
        best_seq = seq_main
        best_score = score_main

    # Secondary: try n=2048
    remaining = deadline - time.monotonic() - 30
    if remaining > 60:
        t_2048 = min(remaining * 0.5, 400.0)
        print(f"\n=== n=2048 optimization (budget={t_2048:.0f}s) ===")
        seq_2048, score_2048 = optimize_at_length(2048, t_2048, rng, verbose=True)
        if score_2048 < best_score:
            best_seq = seq_2048
            best_score = score_2048

    # Tertiary: try n=1536
    remaining = deadline - time.monotonic() - 30
    if remaining > 50:
        t_1536 = min(remaining * 0.6, 300.0)
        print(f"\n=== n=1536 optimization (budget={t_1536:.0f}s) ===")
        seq_1536, score_1536 = optimize_at_length(1536, t_1536, rng, verbose=True)
        if score_1536 < best_score:
            best_seq = seq_1536
            best_score = score_1536

    # Final: one more CD pass on the best found
    remaining = deadline - time.monotonic() - 10
    if remaining > 30 and best_seq is not None:
        t_final = min(remaining * 0.8, 120.0)
        print(f"\n=== Final CD polish (budget={t_final:.0f}s) ===")
        a_fin, s_fin = coord_descent_ternary(best_seq, t_final, verbose=True, label="FINAL")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

    # Final fine polish
    remaining = deadline - time.monotonic() - 5
    if remaining > 15 and best_seq is not None:
        t_fp = min(remaining * 0.8, 60.0)
        a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=False, label="final_fp")
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "final_fp")

    # Fallback
    if best_seq is None:
        best_seq = antipal_init(1024, 1.5, rng)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification with the actual evaluator
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
