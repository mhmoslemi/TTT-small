def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            print(f"      SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses aggressive L-BFGS with multi-temperature log-sum-exp, multi-length
    exploration, asymmetric perturbations, and peak-directed descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = asymmetric_init(n_init, rng, style=2)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Main L-BFGS optimization on parent n ===
    print(f"\n=== Phase 1: L-BFGS n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [80.0, 40.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
        a_opt = lbgfs_optimize(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=20000)
        try_update(a_opt, "lbgfs_phase1")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_sg, "subgrad_phase2")

    # === Phase 3: Low-temperature L-BFGS polish ===
    print(f"\n=== Phase 3: Low-T L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003]
        a_opt2 = lbgfs_optimize(best_seq, t_alloc, T_stages=T_stages2, maxiter_per_stage=15000)
        try_update(a_opt2, "lbgfs_lowT")

    # === Phase 4: Peak-directed coordinate descent ===
    print(f"\n=== Phase 4: Peak Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pd, _ = peak_directed_descent(best_seq, t_alloc, n_steps=80000)
        try_update(a_pd, "peak_desc")

    # === Phase 5: Try different n values ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    candidate_ns = []
    for factor in [0.75, 0.85, 1.15, 1.25, 1.5]:
        alt_n = int(n_best * factor)
        if 200 <= alt_n <= 5000 and alt_n not in candidate_ns:
            candidate_ns.append(alt_n)

    for alt_n in candidate_ns[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alt = min(rem * 0.15, 180.0)
        if t_alt < 20:
            break
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        T_stages3 = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        a_alt = lbgfs_optimize(init_alt, t_alt, T_stages=T_stages3, maxiter_per_stage=8000)
        try_update(a_alt, f"lbgfs_n{alt_n}")

    # === Phase 6: Asymmetric perturbation + re-optimize ===
    print(f"\n=== Phase 6: Asymmetric Perturb (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_pert = min(rem * 0.20, 200.0)
        if t_pert > 30:
            print(f"  Budget: {t_pert:.0f}s")
            for style in range(5):
                rem = deadline - time.monotonic()
                if rem < 15:
                    break
                t_style = t_pert / 5.0
                if rem < t_style:
                    t_style = rem * 0.5
                if t_style < 10:
                    break
                pert = asymmetric_init(n_cur, rng, style=style)
                blended = 0.7 * best_seq + 0.3 * pert
                blended = np.maximum(blended, 0)
                blended = blended / blended.sum()
                T_stages4 = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
                a_p = lbgfs_optimize(blended, t_style, T_stages=T_stages4, maxiter_per_stage=5000)
                try_update(a_p, f"perturb_style{style}")

    # === Phase 7: Final ultra-precise polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.35, 200.0)
        if t_fin > 15:
            print(f"  Budget: {t_fin:.0f}s")
            T_final = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002]
            a_fin = lbgfs_optimize(best_seq, t_fin, T_stages=T_final, maxiter_per_stage=20000)
            try_update(a_fin, "final_lbgfs")

    # === Phase 8: Final peak descent ===
    print(f"\n=== Phase 8: Final Peak Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.4, 60.0)
        if t_ps > 10:
            a_ps, _ = peak_directed_descent(best_seq, t_ps, n_steps=50000)
            try_update(a_ps, "final_peak")

    # === Phase 9: One more subgradient pass ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_sg = min(rem * 0.5, 40.0)
        if t_sg > 10:
            a_sg2, _ = subgradient_refine(best_seq, t_sg, n_steps=50000)
            try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
