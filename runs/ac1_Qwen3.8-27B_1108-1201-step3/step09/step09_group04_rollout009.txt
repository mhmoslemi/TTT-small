"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using log-sum-exp smoothed L-BFGS
in log-domain, fast Nesterov accelerated subgradient method, adaptive active-set
coordinate descent, and systematic multi-length/multi-init exploration.
Key innovations: LSE smoothing (tighter L-infinity approximation than p-norm),
O(n log n) Nesterov iterations (vs LP-based), spectral-aware initialization,
and structured peak-pair coordinate moves.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_value(a, eps):
    """Compute 2n * LSE(conv(a,a), eps) = 2n * eps * log(sum exp(c/eps))."""
    c = fast_convolve(a)
    c_shift = c - c.max()
    val = c.max() + eps * np.log(np.sum(np.exp(c_shift / eps)))
    return 2.0 * len(a) * val


def lse_grad_a(a, eps):
    """Gradient of 2n * LSE(conv(a,a), eps) w.r.t. a."""
    c = fast_convolve(a)
    c_shift = c - c.max()
    w = np.exp(c_shift / eps)
    w = w / w.sum()
    # grad[i] = 2n * 2 * sum_k w_k * a[k-i] = 4n * correlate(w, a, 'valid')[i]
    n = len(a)
    corr = np.correlate(w, a, mode='valid')
    return 4.0 * n * corr


def lse_obj_z(z, eps):
    """Objective in log-domain: a = softmax(z), return 2n * LSE(conv(a,a), eps)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return lse_value(a, eps)


def lse_grad_z(z, eps):
    """Gradient of 2n * LSE(conv(softmax(z),a), eps) w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_shift = c - c.max()
    w = np.exp(c_shift / eps)
    w = w / w.sum()
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * corr
    return a * (g_a - np.dot(g_a, a))


def lse_lbfgs_refine(a0, time_limit, eps_start=0.1, eps_end=0.002, n_stages=10):
    """L-BFGS with adaptive LSE smoothing schedule (log-domain)."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    eps_schedule = np.geomspace(eps_start, eps_end, n_stages)
    iters_per_stage = max(50, 6000 // n_stages)

    for stage_eps in eps_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            return lse_obj_z(z_val, stage_eps)

        def jac(z_val):
            return lse_grad_z(z_val, stage_eps)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      LSE-LBFGS eps={stage_eps:.5f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def nesterov_subgrad_refine(a0, time_limit, n_max_iters=200000):
    """
    Nesterov accelerated subgradient method in log-domain for LSE-smoothed objective.
    Much faster per iteration than LP-based approach (O(n log n) vs polynomial in n).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.max()

    y = z.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Adaptive epsilon that decreases over time
    eps_init = 0.05
    eps_floor = 0.003
    stall_count = 0
    prev_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive epsilon
        progress = min(1.0, (it + 1) / (n_max_iters * 0.5))
        eps = eps_floor + (eps_init - eps_floor) * (1.0 - progress) ** 1.5

        # Compute gradient at lookahead point y
        y_max = y.max()
        exp_y = np.exp(y - y_max)
        a_y = exp_y / exp_y.sum()

        c = fast_convolve(a_y)
        c_shift = c - c.max()
        w = np.exp(c_shift / eps)
        w = w / w.sum()
        corr = np.correlate(w, a_y, mode='valid')
        g_a = 4.0 * n * corr
        g_z = a_y * (g_a - np.dot(g_a, a_y))

        # Nesterov update
        alpha = 2.0 / (it + 2)
        # Step size: use a simple diminishing schedule
        lr = 0.01 / (1.0 + 0.01 * it)
        z_new = y - lr * g_z
        y_new = (1.0 - alpha) * z_new + alpha * (y - z)  # lookahead
        # Actually standard Nesterov: y_{t+1} = (1-alpha)*z_{t+1} + alpha*(y_t - z_t)
        # But simpler: just use z = z - lr * grad with momentum
        z = z_new
        y = y_new

        # Evaluate current point
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()

        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart on stall
        if stall_count > 2000:
            # Add noise and reset momentum
            rng = np.random.default_rng(it)
            z = z + 0.05 * rng.standard_normal(n)
            z = z - z.max()
            y = z.copy()
            stall_count = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovSG iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def coordinate_refine(a0, time_limit, n_steps=50000):
    """
    Smart coordinate descent with peak-pair moves.
    At each step, identify the top peaks and find the best single or paired
    coordinate adjustment that reduces the maximum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        # Find top peaks
        top_k = 3
        peak_indices = np.argsort(c)[::-1][:top_k]
        peak_val = c[peak_indices[0]]
        threshold = peak_val * 0.85
        near_peaks = [k for k in range(len(c)) if c[k] >= threshold]

        improved = False

        # Strategy 1: Single coordinate reduction on peak contributors
        if len(near_peaks) > 0:
            # For each peak, compute contribution of each coordinate
            contrib = np.zeros(n)
            for k in near_peaks:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                if len(idx) > 0:
                    contrib[idx] += a[idx] * a[k - idx]

            top_coords = np.argsort(contrib)[::-1][:10]
            for i in top_coords:
                if a[i] < 1e-12:
                    continue
                for factor in [0.95, 0.90, 0.85, 0.75, 0.60]:
                    a_trial = a.copy()
                    diff = a[i] * (1.0 - factor)
                    a_trial[i] = a[i] * factor
                    # Redistribute to others proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    osum = others.sum()
                    if osum < 1e-15:
                        continue
                    a_trial[mask] += diff * others / osum
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 2: Paired reduction
        if not improved and len(near_peaks) > 0:
            k_peak = near_peaks[0]
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) > 1:
                contribs = a[idx] * a[k_peak - idx]
                top_pairs = np.argsort(contribs)[::-1][:4]
                for ci in top_pairs:
                    i = idx[ci]
                    j = k_peak - i
                    if i == j or j < 0 or j >= n:
                        continue
                    if a[i] < 1e-12 or a[j] < 1e-12:
                        continue
                    for f1 in [0.92, 0.85, 0.75]:
                        for f2 in [0.92, 0.85, 0.75]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-15:
                                continue
                            a_trial /= st
                            sc = 2.0 * n * fast_convolve(a_trial).max()
                            if sc < best_score - 1e-14:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break

        # Strategy 3: Pair of coordinates at different peaks
        if not improved and len(near_peaks) >= 2:
            k1, k2 = near_peaks[0], near_peaks[1]
            js1 = max(0, k1 - n + 1)
            je1 = min(n - 1, k1)
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            # Find coordinate that contributes to both peaks
            common = set(range(js1, je1 + 1)) & set(range(js2, je2 + 1))
            if common:
                common_list = list(common)
                for i in common_list[:5]:
                    if a[i] < 1e-12:
                        continue
                    for factor in [0.90, 0.80, 0.70]:
                        a_trial = a.copy()
                        diff = a[i] * (1.0 - factor)
                        a_trial[i] *= factor
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        others = a_trial[mask]
                        osum = others.sum()
                        if osum < 1e-15:
                            continue
                        a_trial[mask] += diff * others / osum
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        # Strategy 4: Small random perturbation (staircase escape)
        if not improved and step % 50 == 0:
            rng = np.random.default_rng(step * 31 + 7)
            n_pert = min(n, rng.integers(2, 8))
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.08, 0.08))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial /= st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step > 0 and step % 2000 == 0:

            rng = np.random.default_rng(step * 13 + 3)
            a_trial = a + 0.05 * rng.standard_normal(n)
            a_trial = np.maximum(a_trial, 0)

            if st > 1e-15:

                if sc < best_score - 1e-14:

        if not improved and step > 500:

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordRefine step={step} ({elapsed:.0f}s): score={best_score:.6f}")

def spectral_init(n, variant=0):
    """
    Generate initial sequence using spectral thinking.
    Try to make the auto-convolution as flat as possible by designing in frequency domain.
    """
    if variant == 0:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 1:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.5 * x ** 2)
    elif variant == 2:
        # Bimodal (two bumps)
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2)
    elif variant == 3:
        # Triangle (linear up then down)
        x = np.linspace(0, 1, n)
        a = np.minimum(x, 1 - x) * 2
    elif variant == 4:
        # Cosine window
        x = np.linspace(0.5 / n, 1.0 - 0.5 / n, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif variant == 5:
        # Flat with tapered edges (Hamming-like)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(n * 0.1)
        if taper > 1:
            for i in range(taper):
                w = 0.5 * (1 - np.cos(np.pi * i / taper))
                a[i] = w
                a[n - 1 - i] = w
    elif variant == 6:
        # Wide Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-1.0 * x ** 2)
    elif variant == 7:
        # "Ideal" shape: try to flatten auto-conv
        # The auto-conv of a truncated sinc is close to rectangular
        x = np.linspace(-1, 1, n)
        a = np.sinc(x * 3) ** 2  # sinc^2 has triangular auto-conv
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Quick LSE-LBFGS from parent
    Phase 2: Nesterov subgradient (fast, many iterations)
    Phase 3: LSE-LBFGS with very small epsilon
    Phase 4: Coordinate refinement
    Phase 5: Multi-length exploration
    Phase 6: Multi-init restarts
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    n_best = len(best_state['seq']) if best_state['seq'] is not None else 960

    # === Phase 1: LSE-LBFGS (quick, from parent) ===
    print(f"\n=== Phase 1: LSE-LBFGS n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn = lse_lbfgs_refine(best_state['seq'], t_alloc, eps_start=0.05, eps_end=0.005, n_stages=8)
        try_update(best_state, a_pn, "lse_lbfgs_1")

    # === Phase 2: Nesterov subgradient (fast, many iterations) ===
    print(f"\n=== Phase 2: Nesterov Subgrad (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ns, _ = nesterov_subgrad_refine(best_state['seq'], t_alloc, n_max_iters=300000)
        try_update(best_state, a_ns, "nesterov_sg")

    # === Phase 3: LSE-LBFGS with very small epsilon ===
    print(f"\n=== Phase 3: LSE-LBFGS fine (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, eps: 0.005 -> 0.001")
        a_pn2 = lse_lbfgs_refine(best_state['seq'], t_alloc, eps_start=0.005, eps_end=0.001, n_stages=6)
        try_update(best_state, a_pn2, "lse_lbfgs_fine")

    # === Phase 4: Coordinate refinement ===
    print(f"\n=== Phase 4: Coordinate Refine (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        a_cr, _ = coordinate_refine(best_state['seq'], t_alloc, n_steps=100000)
        try_update(best_state, a_cr, "coord_refine")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-128, -96, -64, -48, -32, 32, 48, 64, 96, 128, 192, 256]:
        alt_n = base + delta
        if 384 <= alt_n <= 2560:
            candidate_lengths.add(alt_n)
    # Round numbers
    for rn in [512, 640, 768, 896, 1024, 1152, 1280, 1408, 1536]:
        if 384 <= rn <= 2560 and abs(rn - base) > 48:
            candidate_lengths.add(rn)

    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - base))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 80 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 150.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # Quick LSE-LBFGS
        t_pn = t_total * 0.55
        a_alt = lse_lbfgs_refine(init_alt, t_pn, eps_start=0.03, eps_end=0.003, n_stages=5)
        if try_update(best_state, a_alt, f"lse_n{alt_n}"):
            # If improved, do Nesterov
            rem = deadline - time.monotonic()
            if rem > 40:
                t_ns = min(rem * 0.12, 100.0)
                a_alt2, _ = nesterov_subgrad_refine(best_state['seq'], t_ns, n_max_iters=100000)
                try_update(best_state, a_alt2, f"nesterov_n{alt_n}")

    # === Phase 6: Multi-init restarts ===
    print(f"\n=== Phase 6: Multi-Init Restarts (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        inits = []

        # Different spectral initializations
        for v in [0, 1, 2, 4, 5, 6, 7]:
            inits.append(spectral_init(n_cur, variant=v))

        # Perturbed versions of best
        for scale in [0.05, 0.10, 0.15]:
            pert = best_state['seq'] + scale * rng.standard_normal(n_cur)
            pert = np.maximum(pert, 0)
            if pert.sum() > 0.01:
                pert /= pert.sum()
                inits.append(pert)

        # Try each init with quick LSE-LBFGS
        for idx, init in enumerate(inits[:5]):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.15, 60.0)
            if t_sub < 10:
                break
            a_init = lse_lbfgs_refine(init, t_sub, eps_start=0.05, eps_end=0.005, n_stages=4)
            try_update(best_state, a_init, f"init_{idx}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = lse_lbfgs_refine(best_state['seq'], t_fin, eps_start=0.003, eps_end=0.0005, n_stages=4)
        try_update(best_state, a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ns = min(rem * 0.25, 50.0)
        a_ns2, _ = nesterov_subgrad_refine(best_state['seq'], t_ns, n_max_iters=100000)
        try_update(best_state, a_ns2, "final_nesterov")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cr = min(rem * 0.30, 40.0)
        a_cr2, _ = coordinate_refine(best_state['seq'], t_cr, n_steps=50000)
        try_update(best_state, a_cr2, "final_coord")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - (deadline - float(budget_s) + 20.0)
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
