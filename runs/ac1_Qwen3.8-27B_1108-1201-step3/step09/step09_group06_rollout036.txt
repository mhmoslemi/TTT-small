def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def spectral_shape(a, gamma=0.5, alpha=0.3):
    """Apply spectral shaping: flatten the power spectrum and project back."""
    n = len(a)
    # Use full FFT to handle the phase correctly
    F = fft(a.astype(np.float64))
    # Flatten the magnitude
    mag = np.abs(F)
    phase = np.angle(F)
    # Apply gamma compression to flatten
    mag_flat = mag ** (1 - gamma)
    # Reconstruct with original phase
    F_new = mag_flat * np.exp(1j * phase)
    # Inverse FFT
    a_new = np.real(ifft(F_new))
    # Project onto non-negative, sum=1
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    # Blend
    return (1 - alpha) * a + alpha * a_new

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _lp_solver_cache = None


def get_lp_solver():
    global _lp_solver_cache
    if _lp_solver_cache is not None:
        return _lp_solver_cache
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cache = cp.ECOS
    return _lp_solver_cache


def score_of(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def spectral_shape(a, gamma=0.4, blend=0.3):
    """Flatten power spectrum and project back to non-negative simplex."""
    n = len(a)
    F = fft(a.astype(np.float64))
    mag = np.abs(F)
    mag[mag < 1e-300] = 1e-300
    phase = np.angle(F)
    # Compress dynamic range
    log_mag = np.log(mag)
    log_mag_centered = log_mag - log_mag.mean()
    # Soft clipping / compression
    compressed = log_mag_centered / (1.0 + gamma * np.abs(log_mag_centered))
    mag_new = np.exp(log_mag.mean() + compressed)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(F_new))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return (1.0 - blend) * a + blend * a_new


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000, use_spectral=True):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    spectral_cooldown = 0
    gamma = 0.3
    rng = np.random.default_rng(999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = score_of(a)
        score_y = score_of(y)
        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            improved = True

        if improved:
            stall_count = 0

        if it % 150 == 0:
            if not improved:
                stall_count += 1

            if use_spectral and stall_count >= 4:
                if spectral_cooldown <= 0:
                    # Try spectral shaping
                    a_shaped = spectral_shape(a, gamma=gamma, blend=0.25)
                    score_shaped = score_of(a_shaped)
                    # Accept if not significantly worse
                    if score_shaped < best_score + 0.1:
                        a = a_shaped
                        y = a_shaped.copy()
                    spectral_cooldown = 15
                    gamma = min(gamma + 0.08, 0.65)
                else:
                    spectral_cooldown -= 1
                stall_count = max(0, stall_count - 1)
            elif stall_count >= 12:
                # Random perturbation escape
                noise = rng.random(n) * 0.04
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall_count = 0

            gamma = max(gamma - 0.015, 0.2)

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} stall={stall_count}")

    return best_a, best_score


def _lbgfs_func(z_in):
    T_val = _lbgfs_state['T']
    n = _lbgfs_state['n']
    if time.monotonic() - _lbgfs_state['t0'] > _lbgfs_state['time_limit']:
        return 1e15
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum()) + np.log(len(exp_Tb))
    return (2.0 * n / T_val) * log_sum_exp - 2.0 * n * np.log(len(b))


def _lbgfs_jac(z_in):
    T_val = _lbgfs_state['T']
    n = _lbgfs_state['n']
    if time.monotonic() - _lbgfs_state['t0'] > _lbgfs_state['time_limit']:
        return np.zeros_like(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.2, maxiter=12000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    n_stages = 7
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        _lbgfs_state['T'] = stage_T
        _lbgfs_state['n'] = n
        _lbgfs_state['t0'] = t0
        _lbgfs_state['time_limit'] = time_limit
        try:
            res = minimize(
                fun=_lbgfs_func,
                x0=z,
                jac=_lbgfs_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    result = softmax_from_z(z)
    return normalize_nonneg(result)


def compute_subgradient_a(a, n_active_frac):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n)
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g


def mirror_descent_refine(a0, time_limit, n_active_frac=0.05):
    n = len(a0)
    a = normalize_nonneg(a0)
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_of(a)
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.4
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        g = compute_subgradient_a(a, n_active_frac)
        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        polyak_lr = (best_score - f_lb) / (2.0 * g_norm * best_score + 1e-15)
        alpha = max(min(lr, polyak_lr), 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        score_new = score_of(a_new)

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.75
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def local_swap_search(a0, time_limit, n_steps=40000):
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Focus on the top few peaks
        top_peaks = np.argsort(b)[::-1][:3]

        improved = False
        for pk in top_peaks:
            contrib = np.zeros(n)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contrib[idx] = a[idx] * a[pk - idx]

            # Find high contributors at this peak
            src_candidates = np.argsort(contrib)[::-1][:4]
            # Find low positions (least impact on this peak)
            dst_candidates = np.argsort(contrib)[:4]

            for si in src_candidates:
                for di in dst_candidates:
                    if si == di or a[si] < 1e-14:
                        continue
                    for frac in [0.01, 0.02, 0.04]:
                        amount = a[si] * frac
                        a_trial = a.copy()
                        a_trial[si] -= amount
                        a_trial[di] += amount
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = score_of(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * 0.02
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = score_of(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and step > 15000:
            # Early stop if stuck for too long
            if step % 20000 == 0:
                break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 0.6
    else:
        a = np.ones(n) * 0.6
        edge = max(1, int(n * 0.08))
        a[:edge] = 1.0
        a[-edge:] = 1.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
