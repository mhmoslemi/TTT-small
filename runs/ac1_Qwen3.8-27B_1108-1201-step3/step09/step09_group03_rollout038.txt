"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses mirror descent with multi-peak subgradients, coordinate descent,
and asymmetric initializations. Key insight: for non-negative sequences,
the convolution peak is typically in the middle (k~n-1), and asymmetric
sequences can reduce this peak below the uniform value.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute the objective score for sequence a (assumed sum=1, non-negative)."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    return 2.0 * n * M


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def init_ramp(n, rng):
    """Asymmetric linear ramp (higher at one end)."""
    a = np.arange(1, n + 1, dtype=np.float64)
    if rng.random() < 0.5:
        a = a[::-1]
    a = a / a.sum()
    return a


def init_exp(n, rng):
    """Asymmetric exponential decay."""
    rate = rng.uniform(0.01, 0.1)
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    if rng.random() < 0.5:
        a = a[::-1]
    a = a / a.sum()
    return a


def init_raised_cosine_asym(n, rng):
    """Asymmetric raised cosine window."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    # Asymmetric: shift the center
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.3, 0.8)
    x = (t - center) / width
    a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_two_hump_asym(n, rng):
    """Two humps with different heights (asymmetric)."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.4)
    c2 = n * rng.uniform(0.5, 0.8)
    w1 = n * rng.uniform(0.05, 0.2)
    w2 = n * rng.uniform(0.05, 0.2)
    h1 = rng.uniform(0.5, 1.0)
    h2 = rng.uniform(0.5, 1.0)
    a = h1 * np.exp(-0.5 * ((t - c1) / w1) ** 2) + h2 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def mirror_descent(a0, time_limit, lr_init=0.1, top_k=20, verbose=True, label=""):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Uses multi-peak subgradient for the max-convolution objective.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.01 * M
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient: sg[i] = sum_k w[k] * (a[k-i] + a[i-(k-n+1+... )]) 
        # Actually: d(conv[k])/da[i] = a[k-i] if 0<=k-i<n, plus a[i-(k-n+1)]... 
        # More precisely: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] (from j=i term) + a[i-(k-n+1)]... 
        # Wait: conv[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j]*a[k-j]
        # d/d(a[i]) conv[k] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, from the other factor)
        # = 2*a[k-i] if 0<=k-i<n and k-i != i
        # = a[k-i] + a[k-i] + ... if k-i = i (i.e., k=2i), it's 2*a[i] + derivative of a[i]*a[i] = 2*a[i]
        # Actually d/d(a[i]) (a[i]*a[k-i]) when k-i=i: d/d(a[i])(a[i]*a[i]) = 2*a[i]
        # And d/d(a[i]) (a[j]*a[k-j]) for j != i and k-j = i: = a[j] = a[k-i]
        # So total: if k-i = i (k=2i): 2*a[i]
        #           if k-i != i (and 0<=k-i<n): a[k-i] (from j=i) + a[k-i] (from k-j=i, j=k-i) = 2*a[k-i]
        #           If k-i = i and also k-j = i for j != i: not possible since j=k-i=i
        # So: d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n
        # (The factor of 2 comes from the two ways a[i] appears in the sum)
        # Unless k-i = i, in which case it's d(a[i]^2)/da[i] = 2*a[i] (same formula!)
        # So in all cases: d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            # conv[k] depends on a[i] for i in [max(0, k-n+1), min(k, n-1)]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # d(conv[k])/da[i] = 2*a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] = a[k-i_start : k-i_end+1][::-1]
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Running average
        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        conv = fast_convolve(a)

        # Periodically check running average
        if it % 100 == 0:
            avg_score = compute_score(avg_a)
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f} M={M:.8f}")

    return best_a, best_score

                delta = 1e-6
            else:

                base = a[i]
                
            k_start = i

def fw_step(a, b, conv_ab, step, use_away=True):
    """
    One Frank-Wolfe step: update b to minimize max(conv(a,b)) on the simplex.
    Returns updated b and conv_ab.
    """
    n = len(a)
    k_star = int(np.argmax(conv_ab))
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    # Gradient of conv_ab[k_star] w.r.t. b[i] = a[k_star - i] for i in [i_start, i_end]
    grad = a[k_star - i_end:k_star - i_start + 1][::-1]
    j_star = i_start + int(np.argmin(grad))

    t = 2.0 / (step + 2.0)

    if use_away and step >= 50:
        # Away-step
        grad_full = np.zeros(n)
        grad_full[i_start:i_end + 1] = grad
        away_scores = b * grad_full
        i_away = int(np.argmax(away_scores))
        if b[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
            t_away = min(t * 0.5, b[i_away] * 0.9)
            # Remove contribution of i_away
            ks_s = i_away
            ks_e = min(i_away + n - 1, 2 * n - 2)
            conv_ab[ks_s:ks_e + 1] -= t_away * a
            b[i_away] -= t_away
            b[i_away] = max(b[i_away], 0.0)
            s_b = b.sum()
            if s_b < 1e-15:
                b = np.ones(n) / n
                conv_ab = convolve_ab(a, b)
            else:
                b = b / s_b
                conv_ab = convolve_ab(a, b)
            return b, conv_ab

    # Standard FW step
    b_new = (1.0 - t) * b
    b_new[j_star] += t
    conv_new = (1.0 - t) * conv_ab
    ks_s = j_star
    ks_e = min(j_star + n - 1, 2 * n - 2)
    conv_new[ks_s:ks_e + 1] += t * a
    return b_new, conv_new


def fw_alternating(a0, time_limit, n_steps=100, verbose=True, label=""):
    """
    Frank-Wolfe alternating minimization.
    Alternate: optimize b given a (min max conv(a,b)), then optimize a given b.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    conv_ab = convolve_ab(a, b)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    total_steps = 0

    while time.monotonic() - t0 < time_limit:
        # Optimize b given a
        for step in range(n_steps):
            b, conv_ab = fw_step(a, b, conv_ab, total_steps, use_away=True)
            total_steps += 1

        # Evaluate b
        b_score = compute_score(b)
        if b_score < best_score:
            best_score = b_score
            best_a = b.copy()

        # Optimize a given b
        conv_ba = convolve_ab(b, a)
        for step in range(n_steps):
            a, conv_ba = fw_step(b, a, conv_ba, total_steps, use_away=True)
            total_steps += 1

        # Evaluate a
        a_score = compute_score(a)
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

        # Also try the average
        avg = 0.5 * (a + b)
        avg = avg / avg.sum()
        avg_score = compute_score(avg)
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg.copy()

        if verbose and (total_steps // n_steps) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW {label} steps={total_steps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses mirror descent with multi-peak subgradients, coordinate descent with
prefix/suffix max for O(1) boundary queries, and diverse asymmetric

peak is in the middle (k~n-1), and asymmetric sequences reduce it below

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    B = rfft(b.astype(np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute score assuming a is non-negative with sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    return 2.0 * n * M


def init_ramp(n, rng):
    """Asymmetric linear ramp."""
    a = np.arange(1, n + 1, dtype=np.float64)
    if rng.random() < 0.5:
        a = a[::-1]
    a = a / a.sum()
    return a

    """Asymmetric exponential."""
    rate = rng.uniform(0.02, 0.15)
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    if rng.random() < 0.5:

    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    """
    Mirror descent (exponentiated gradient) on the simplex.
    Uses multi-peak softmax-weighted subgradient of max(conv(a,a)).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * M + 1e-15
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # sg[i] = sum_k w_k * 2*a[k-i] for 0 <= k-i < n
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # a[k-i] for i in [i_start, i_end]
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        avg_a = (it * avg_a + a_new) / (it + 1)
        a = a_new
        conv = fast_convolve(a)

        if it % 50 == 0:
            avg_score = compute_score(avg_a)
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with prefix/suffix max for O(1) boundary queries.
    Changing a[i] affects conv[k] for k in [i, i+n-1].
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    order = np.arange(n)

    # Precompute prefix and suffix max
    prefix_max = np.maximum.accumulate(conv)
    suffix_max = np.maximum.accumulate(conv[::-1])[::-1]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        if sweeps % 5 == 1:
            np.random.shuffle(order)

        for i in order:
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max using prefix/suffix
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(prefix_max[k_start - 1]))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(suffix_max[k_end + 1]))

            # Try multiplier factors
            base = a[i]
            factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
            best_factor = 1.0
            best_new_M = M

            affected = conv[k_start:k_end + 1]
            # a_shifted = a (as proven: the shift always gives the full vector)

            for f in factors:
                delta = base * (f - 1.0)
                new_val = base + delta
                if new_val < -1e-15:
                    continue
                # new_affected[k] = affected[k] + 2*delta*a[k-i]
                # a[k-i] for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]
                new_affected = affected + 2.0 * delta * a[:n]
                # Quadratic correction at k=2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_factor = f

            if best_factor != 1.0:
                delta = base * (best_factor - 1.0)
                a[i] = max(a[i] + delta, 0.0)
                conv[k_start:k_end + 1] = conv[k_start:k_end + 1] + 2.0 * delta * a[:n]
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    conv[2 * i] += delta * delta
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                # Update prefix/suffix (expensive, do less often)
                improved = True

        # Rebuild prefix/suffix periodically
        if sweeps % 10 == 0:
            prefix_max = np.maximum.accumulate(conv)
            suffix_max = np.maximum.accumulate(conv[::-1])[::-1]

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score
