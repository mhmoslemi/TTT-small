"""

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv)/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def p_norm_obj(a, p):
    """Compute (sum c_k^p)^(1/p) for c = convolve(a,a), a normalized sum=1."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    return (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_grad_z(z, p):
    """Gradient of 2n*(sum c^p)^(1/p) w.r.t. log-domain z (softmax param)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=7):
    """L-BFGS with adaptive p-norm smoothing."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    z = np.log(a0) - np.log(a0).max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(100, 6000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            return 2.0 * len(a) * p_norm_obj(a, stage_p)

        def jac(z_val):
            return p_norm_grad_z(z_val, stage_p)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = compute_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"      p-norm L-BFGS p={stage_p:.0f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def multiplicative_weights_optimize(a0, time_limit, K_start=1, K_end=10,
                                     eta_start=0.5, momentum=0.85):
    """Mirror descent (multiplicative weights) with adaptive top-K peak subgradient."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum buffer
    it = 0
    stall = 0
    prev_score = best_score

    total_iters_expected = 5000
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 20000:
            break

        # Adaptive K
        frac = min(1.0, it / total_iters_expected)
        K = max(1, int(K_start + frac * (K_end - K_start)))

        # Compute convolution and find top-K peaks
        conv = fast_convolve(a)
        # Ignore first and last (they have fewer terms)
        conv_work = conv.copy()
        if len(conv_work) > 2:
            conv_work[0] = -np.inf
            conv_work[-1] = -np.inf

        top_K_idx = np.argsort(conv_work)[::-1][:K]
        top_K_vals = conv_work[top_K_idx]

        # Weighted subgradient: g_j = sum_k w_k * a_{k-j}
        conv_max = top_K_vals[0]
        if conv_max < 1e-15:
            break
        # Weights proportional to peak height (with floor)
        w = top_K_vals / top_K_vals.sum()

        g = np.zeros(n)
        for k_idx, w_k in zip(top_K_idx, w):
            # g_j += w_k * a_{k-j} (with zero padding)
            # a_{k-j} is valid when 0 <= k-j < n, i.e., j in [k-n+1, k]
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            for j in range(js, je + 1):
                g[j] += w_k * a[k_idx - j]

        # Momentum
        v = momentum * v + (1 - momentum) * g

        # Adaptive step size with backtracking
        eta = eta_start / (1.0 + 0.05 * it)
        retries = 0
        while retries < 8:
            # Multiplicative weights update
            log_mult = -eta * v
            log_mult -= log_mult.max()  # numerical stability
            mult = np.exp(log_mult)
            a_new = a * mult
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s

            score_new = compute_score(a_new)
            if score_new < best_score - 1e-12:
                a = a_new
                best_score = score_new
                best_a = a.copy()
                stall = 0
                break
            else:
                eta *= 0.5
                retries += 1
                if eta < 1e-8:
                    break
        else:
            # Stall
            stall += 1
            if stall > 50:
                # Perturbation
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.05
                a = a + noise
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                v = np.zeros(n)
                stall = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            if best_score < prev_score - 1e-7:
                prev_score = best_score
                stall = 0
            print(f"      MW iter={it} ({elapsed:.0f}s): score={best_score:.6f}, K={K}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max conv(a,b) subject to b>=0, sum b=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()

        s = 1.0

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_stall = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_stall - 1e-7:

                rng = np.random.default_rng(it + 123)
                noise = rng.standard_normal(n) * 0.03

                y = np.maximum(y, 0)

def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:

        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        # Flattened with soft edges
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = max(2, n // 16)
        for i in range(edge):
            t = (i + 1) / edge
            a[i] = t ** 1.5
            a[n - 1 - i] = t ** 1.5
    elif variant == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={len(parent)}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: Multiplicative Weights (primary optimizer) ===
    print(f"\n=== Phase 1: Multiplicative Weights n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, K: 1->10, eta_start=0.4, momentum=0.85")
        a_mw, _ = multiplicative_weights_optimize(
            best_state['seq'], t_alloc, K_start=1, K_end=10,
            eta_start=0.4, momentum=0.85)
        try_update(best_state, a_mw, "mul_weights_1")

    # === Phase 2: p-norm L-BFGS (refinement) ===
    print(f"\n=== Phase 2: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, p: 10 -> 600")
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_alloc,
                                    p_start=10.0, p_end=600.0, n_stages=8)
        try_update(best_state, a_pn, "pnorm_lbfgs_1")

    # === Phase 3: Nesterov LP ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver,
                                      n_max_iters=80000)
        try_update(best_state, a_lp, "nesterov_lp_1")

    # === Phase 4: Local Refinement v2 ===
    print(f"\n=== Phase 4: Local Refinement v2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 10:
        a_lr, _ = local_refinement_v2(best_state['seq'], t_alloc, n_steps=40000)
        try_update(best_state, a_lr, "local_v2_1")

    # === Phase 5: Multiplicative Weights round 2 (higher K) ===
    print(f"\n=== Phase 5: MW r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        a_mw2, _ = multiplicative_weights_optimize(
            best_state['seq'], t_alloc, K_start=3, K_end=15,
            eta_start=0.2, momentum=0.9)
        try_update(best_state, a_mw2, "mul_weights_2")

    # === Phase 6: Length Exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-64, -32, 32, 64, 96, 128, 160]:
        alt_n = base + delta
        if 512 <= alt_n <= 2048:
            candidate_lengths.add(alt_n)
    # Try round numbers good for FFT
    for rn in [768, 896, 1024, 1152, 1280, 1408]:
        if 512 <= rn <= 2048 and abs(rn - base) > 16:
            candidate_lengths.add(rn)

    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 150.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # Quick p-norm L-BFGS
        t_pn = t_total * 0.55
        a_alt = p_norm_lbfgs_refine(init_alt, t_pn, p_start=30.0, p_end=300.0, n_stages=5)
        improved = try_update(best_state, a_alt, f"pn_n{alt_n}")

        # If improved, do local refinement
        if improved:
            rem = deadline - time.monotonic()
            if rem > 20:
                t_lr = min(rem * 0.12, 60.0)
                a_alt2, _ = local_refinement_v2(best_state['seq'], t_lr, n_steps=10000)
                try_update(best_state, a_alt2, f"lr_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.15, 120.0)
        n_cur = len(best_state['seq'])

        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 1e-15:
                perturbed = perturbed / ps
            t_sub = min(rem * 0.35, 50.0)
            a_pert = p_norm_lbfgs_refine(perturbed, t_sub, p_start=50.0, p_end=400.0, n_stages=5)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.30, 60.0)
        a_fin = p_norm_lbfgs_refine(best_state['seq'], t_fin,
                                     p_start=200.0, p_end=1000.0, n_stages=4)
        try_update(best_state, a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.25, 50.0)
        a_lr2, _ = local_refinement_v2(best_state['seq'], t_lr, n_steps=15000)
        try_update(best_state, a_lr2, "final_local")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_mw = min(rem * 0.30, 40.0)
        a_mw3, _ = multiplicative_weights_optimize(
            best_state['seq'], t_mw, K_start=5, K_end=20,
            eta_start=0.1, momentum=0.92)
        try_update(best_state, a_mw3, "final_mw")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
