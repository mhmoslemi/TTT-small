def subgradient_polish(a0, time_limit, n_steps=200000):
    """Fine subgradient polishing with adaptive step sizes."""
    n = len(a0)

    lr = 0.1
    no_improve = 0

        if time.monotonic() - t0 > time_limit:

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: positions within 0.01% of max
        threshold = max_b * (1 - 1e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: 2 * sum of a_{k-j} over active k, averaged
        g = np.zeros(n)
        for k in active:
            idx = np.arange(n)
            km = k - idx
            valid = (km >= 0) & (km < n)
            g[valid] += 2.0 * a[km[valid]]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = compute_score(a_trial)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive lr
        if no_improve >= 15:
            lr *= 0.8
            no_improve = 0
        lr = max(1e-10, lr)

        if step % 50000 == 0:
            print(f"    SubGrad {time.monotonic()-t0:.0f}s: score={best_score:.10f} lr={lr:.2e}")

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    candidates = [solver]
    for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if s not in candidates:
            candidates.append(s)

    for sol in candidates:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

        a_new = np.maximum(np.asarray(a_new, dtype=np.float64), 0)
        s = a_new.sum()
        if s < 1e-10:
            return
        a_new = a_new / s
        sc = compute_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            print(f"  [*] NEW BEST [{label}]: {sc:.10f} (n={len(a_new)})")

    # === Phase 1: Nesterov LP (primary) ===
    print(f"\n--- Phase 1: Nesterov LP n={len(best_seq)} ---")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.42, 750.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t1, solver=solver, n_max_iters=500000)
        try_update(a_lp, "LP1")

    # === Phase 2: Subgradient ===
    print(f"\n--- Phase 2: Subgradient (score={best_score:.10f}) ---")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 20:
        a_sg, _ = subgradient_polish(best_seq, t2, n_steps=300000)
        try_update(a_sg, "SG1")

    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n--- Phase 3: L-BFGS (score={best_score:.10f}) ---")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 15:
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
        a_lb, _ = lbgfs_polish(best_seq, t3, T_stages=T_stages)
        try_update(a_lb, "LBFGS1")

    # === Phase 4: Peak coordinate descent ===
    print(f"\n--- Phase 4: Peak CD (score={best_score:.10f}) ---")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)
    if t4 > 15:
        a_cd, _ = peak_coordinate_descent(best_seq, t4, n_steps=50000)
        try_update(a_cd, "CD1")

    # === Phase 5: Multi-length exploration ===
    print(f"\n--- Phase 5: Multi-length (score={best_score:.10f}) ---")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 0.8, 1.5, 0.7]:
        alt_n = int(n_cur * factor)
        if 400 <= alt_n <= 3000 and alt_n != n_cur and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)
    alt_lengths = alt_lengths[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alt = min(rem * 0.35, 250.0)
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        t_lp = t_alt * 0.55
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=100000)
        try_update(a_alt, f"LP_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sg = min(rem * 0.3, 60.0)
            a_alt2, _ = subgradient_polish(best_seq, t_sg, n_steps=50000)
            try_update(a_alt2, f"SG_n{alt_n}")

    # === Phase 6: Second LP pass on best ===
    print(f"\n--- Phase 6: Second LP (score={best_score:.10f}) ---")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.25, 200.0)
    if t6 > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t6, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "LP2")

    # === Phase 7: Final polish ===
    print(f"\n--- Phase 7: Final polish (score={best_score:.10f}) ---")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.4, 60.0)
        a_f1, _ = subgradient_polish(best_seq, t7, n_steps=50000)
        try_update(a_f1, "SG_final")

    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.4, 40.0)
        a_f2, _ = peak_coordinate_descent(best_seq, t8, n_steps=20000)
        try_update(a_f2, "CD_final")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f} n={len(best_seq)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
