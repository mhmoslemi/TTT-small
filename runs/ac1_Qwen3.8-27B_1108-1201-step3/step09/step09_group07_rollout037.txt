    # Softmax to get a

import numpy as np

def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a, via FFT."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (2 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_stages, steps_per_stage, lr_init=0.05):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    step_count = 0
    
    for T in T_stages:
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)
            
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr
            
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    elapsed = time.monotonic() - t0
                    print(f"    NG step={step_count} ({elapsed:.0f}s): score={best_score:.8f} T={T:.1f}")
    
    return best_a, best_score


def softmax_lfgs(a_init, T, maxiter=5000):
    """L-BFGS optimization in log-space with softmax parameterization."""
    n = len(a_init)
    
    def func_and_grad(x):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Nesterov accelerated projected gradient with log-sum-exp smoothing and adaptive temperature
2. L-BFGS refinement in centered log-space (softmax parameterization)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a via FFT."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (2 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_stages, steps_per_stage, lr_init=0.05):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    step_count = 0

    for T in T_stages:
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)

            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr

            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 200 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    elapsed = time.monotonic() - t0
                    print(f"    NG step={step_count} ({elapsed:.0f}s): score={best_score:.8f} T={T:.1f}")

    # Final check
    sc = score_normalized(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()

    return best_a, best_score

    """L-BFGS optimization in centered log-space with softmax parameterization."""
    n = len(a_init)
    x0 = np.log(a_init + 1e-30)

    t0 = time.monotonic()
    call_count = [0]

    def func_and_grad(x):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            # Return a large value to signal termination
            return 1e10, np.zeros(n)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            if rem < 50:
                break
            t_sub = min(rem * 0.18, 100.0)
            if t_sub < 25:
                break
            a_init = make_init(n_best, style, rng)
            print(f"  style={style}, budget={t_sub:.0f}s")
            T_stages3 = [5.0, 15.0, 40.0, 100.0, 250.0]
            a_sub, _ = nesterov_pg(a_init, t_sub * 0.55, T_stages3, steps_per_stage=2000, lr_init=0.08)
            try_update(a_sub, f"start_{style}")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_sub2 = softmax_lbfgs(a_sub, 300.0, maxiter=1500, time_limit=min(rem*0.3, 25.0))
                try_update(a_sub2, f"start_{style}_lbfgs")

            rem = deadline - time.monotonic()
            if rem > 10:
                a_sub3, _ = peak_shave(a_sub, min(rem*0.3, 15.0), n_steps=10000)
                try_update(a_sub3, f"start_{style}_ps")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for delta in [-48, -24, 24, 48, 96, 128, -96]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        alt_n = n_best + delta
        if alt_n < 256 or alt_n > 2048:
            continue
        t_total = min(rem * 0.14, 80.0)
        if t_total < 20:
            continue
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        T_stages4 = [10.0, 30.0, 80.0, 200.0, 400.0]
        a_alt, _ = nesterov_pg(init_alt, t_total * 0.5, T_stages4, steps_per_stage=1500, lr_init=0.06)
        try_update(a_alt, f"n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2 = softmax_lbfgs(a_alt, 400.0, maxiter=1500, time_limit=min(rem*0.3, 20.0))
            try_update(a_alt2, f"n{alt_n}_lbfgs")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_alt3, _ = peak_shave(a_alt, min(rem*0.3, 10.0), n_steps=5000)
            try_update(a_alt3, f"n{alt_n}_ps")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 60.0)
        a_fin, _ = peak_shave(best_seq, t_fin, n_steps=30000)
        try_update(a_fin, "final_polish")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin2 = min(rem * 0.4, 30.0)
        a_fin2 = softmax_lbfgs(best_seq, 1000.0, maxiter=1000, time_limit=t_fin2)
        try_update(a_fin2, "final_lbfgs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
