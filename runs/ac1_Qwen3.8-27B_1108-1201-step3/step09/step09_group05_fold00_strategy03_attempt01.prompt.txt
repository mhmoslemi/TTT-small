def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):

    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000, n_stages=None):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def frank_wolfe_refine(a0, time_limit, n_steps=50000):
    """Frank-Wolfe conditional gradient with away steps for local optimization."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient: g[i] = a[k_star - i] valid
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = a[k_star - i]

        # FW toward step: move toward e_i where i = argmin(g)
        i_toward = int(np.argmin(g))

        # Try multiple step sizes
        best_trial = None
        best_trial_score = best_score

        for t_val in [0.5, 0.3, 0.15, 0.08, 0.04]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score_trial = 2.0 * n * fast_convolve(a_trial).max()
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()

        # Away step: move away from e_j where j = argmax(g) in support
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.3, 0.2, 0.1, 0.05]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:

        a = np.exp(-x ** 2 * 3)
    elif mode == 2:

        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_shave_local(a0, time_limit, n_steps=15000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        # Find top contributors
        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                if contribs[ci] > 0:
                    contrib_count[idx[ci]] += contribs[ci]

        top_coords = np.argsort(contrib_count)[::-1][:15]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                # Redistribute to low-contrib positions
                low_coords = np.argsort(contrib_count)[:n // 4]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False

                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                else:
                    a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

                break

            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

                        if score < best_score - 1e-15:

                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:

            if not improved:
