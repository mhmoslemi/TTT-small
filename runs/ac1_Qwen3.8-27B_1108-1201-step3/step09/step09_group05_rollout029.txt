import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, k_star):
    """Compute subgradient of max_k conv(a,a)_k at a, given peak at k_star."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for i in range(js, je + 1):
        g[i] = 2.0 * a[k_star - i]
    return g


def nesterov_subgrad_refine(a0, time_limit, n_max_iters=200000):
    """Nesterov accelerated subgradient method (no LP)."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        g = compute_subgradient(y, k_star)
        
        # Mirror descent step in log domain
        # Or simpler: project onto simplex
        alpha = 2.0 / (it + 2)
        
        # Gradient step
        y_grad = y - alpha * g / (2.0 * n)  # normalize gradient
        y_grad = np.maximum(y_grad, 0)
        sy = y_grad.sum()
        if sy > 1e-15:
            y_grad = y_grad / sy
        y = y_grad.copy()
        
        # Nesterov update
        a_new = ((it + 1) * a + (it + 2) * y_grad) / (2 * it + 3)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa > 1e-15:
            a_new = a_new / sa
            a = a_new
            y = y_grad  # y tracks a in Nesterov
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=15, iters_per_stage=500):
    """L-BFGS with log-sum-exp smoothing, very low final temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Only process significant weights
            threshold = 1e-10
            significant = w > threshold
            if not np.any(significant):
                # Fallback: use argmax
                k_star = int(np.argmax(b))
                g_a = np.zeros(n)
                js = max(0, k_star - n + 1)
                je = min(n - 1, k_star)
                idx = np.arange(js, je + 1)
                g_a[idx] = 2.0 * a[k_star - idx]
                S = np.dot(g_a, a)
                return a * (g_a - S)
            
            grad_a = np.zeros(n)
            for k in np.where(significant)[0]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # More iterations at low temperature
        if stage_T < 1.0:
            niters = iters_per_stage * 2
        elif stage_T < 10.0:
            niters = iters_per_stage
        else:
            niters = iters_per_stage // 2
        
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': niters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': niters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def peak_pair_optimize(a0, time_limit, n_steps=50000):
    """Targeted pair optimization to reduce the peak convolution value."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Get all pairs contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Sort by contribution (descending)
        top_pairs = np.argsort(contribs)[::-1][:20]
        
        improved = False
        for ci in top_pairs:
            i = idx[ci]
            j = k_star - i
            if i >= n or j >= n or i == j:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Try reducing the larger one and adding to a low-contrib position
            if a[i] >= a[j]:
                reduce_i, reduce_j = i, j
            else:
                reduce_i, reduce_j = j, i
            
            # Find a low-contrib position to redistribute
            low_pairs = np.argsort(contribs)[:5]
            
            for li in low_pairs:
                li_idx = idx[li]
                if li_idx == reduce_i or (k_star - li_idx == reduce_i):
                    continue
                if li_idx >= n or (k_star - li_idx) >= n:
                    continue
                
                for factor in [0.95, 0.90, 0.85, 0.80]:
                    a_trial = a.copy()
                    diff = a_trial[reduce_i] * (1 - factor)
                    a_trial[reduce_i] *= factor
                    # Redistribute proportionally to low-contrib elements
                    target = li_idx
                    a_trial[target] += diff
                    a_trial = norm_sequence(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try symmetric pair reduction
            for ci in top_pairs[:10]:
                i = idx[ci]
                j = k_star - i
                if i >= n or j >= n or i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1, f2 in [(0.95, 0.95), (0.93, 0.97), (0.97, 0.93), (0.90, 0.90)]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = norm_sequence(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
    
    return best_a, best_score


def nesterov_lp_with_perturb(a0, time_limit, solver=None, n_max_iters=30000):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Wait, I realize there's a problem with my `nesterov_subgrad_refine` function. The Nesterov acceleration formula isn't quite right. Let me fix it.

In Nesterov's accelerated gradient method:
- y_t = a_t + (t-1)/(t+2) * (a_t - a_{t-1})  [momentum]
- a_{t+1} = proj(y_t - alpha_t * grad(y_t))

Or the dual averaging variant:
- y_t = (1/(t+1)) * sum_{s=1}^{t} a_s  [averaged iterate]
- a_t = argmin (1/t) * ||a - y_t||^2 / (2*alpha) + f(a)

For our problem, a simpler approach is:
- Compute gradient at y (the "lookahead" point)
- Update: a_{t+1} = proj_simplex(a_t - alpha_t * grad(y_t))
- y_{t+1} = a_{t+1} + (t+1)/(t+4) * (a_{t+1} - a_t)  [Nesterov momentum]

Let me simplify and just use the standard FISTA-like approach:

```
y_0 = a_0
t_1 = 1
for t = 1, 2, ...:
    a_t = prox(y_t - alpha * grad(y_t))  # for simplex, this is just projection
    t_{new} = (1 + sqrt(1 + 4*t^2)) / 2
    y_{t+1} = a_t + (t - 1) / t_{new} * (a_t - a_{t-1})
```

But the prox of a linear function on the simplex is just the Euclidean projection onto the simplex, which is:
`proj_simplex(v) = v + lambda` where lambda is chosen so that sum = 1 and all elements >= 0.

Actually, for the constraint `a >= 0, sum(a) = 1`, the projection is:
1. Start with `v`
2. Find `tau` such that `max(v_i - tau, 0)` sums to 1
3. Result is `max(v - tau, 0)`

This can be done efficiently by sorting.

Let me rewrite the subgradient method properly:

```python
def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

1. The Nesterov subgradient is correct
2. The L-BFGS is very effective at low temperature
3. The peak pair optimization is effective
4. Time management is good

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] subject to b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient(a, k_star):
    """Compute subgradient of max_k conv(a,a)_k at a, with peak at k_star."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for i in range(js, je + 1):
        g[i] = 2.0 * a[k_star - i]
    return g


def fista_refine(a0, time_limit, n_max_iters=200000):
    """FISTA (accelerated proximal gradient) on the simplex for f(a)=max_k conv(a,a)_k."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Estimate Lipschitz constant
    b0 = fast_convolve(a)
    L_est = 2.0 * n * b0.max()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        b_y = fast_convolve(y)
        k_star = int(np.argmax(b_y))
        g = compute_subgradient(y, k_star)
        
        # Step size (backtracking or fixed)
        step = 1.0 / (2.0 * np.sqrt(n))
        
        # Gradient step
        v = y - step * g
        # Project onto simplex
        a_new = project_simplex(v)
        
        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a_new.copy()
        t_mom = t_new
        a = a_new
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp_with_perturb(a0, time_limit, solver=None, n_max_iters=30000):
    """Nesterov LP alternating with structured sinusoidal perturbation on stall."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall_count = 0
            prev_best = best_score
        elif it % 150 == 0:
            if best_score >= prev_best - 1e-8:
                stall_count += 1
                if stall_count >= 4:
                    # Structured perturbation
                    rng_local = np.random.default_rng(999 + it)
                    freq = rng_local.integers(2, max(3, n // 8))
                    phase = rng_local.random() * 2 * np.pi
                    scale = 0.04 + 0.02 * stall_count
                    pert = scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                    y = y + pert
                    y = np.maximum(y, 0)
                    y = y / y.sum()
                    stall_count = 0
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.003, n_stages=18, iters_per_stage=400):
    """L-BFGS with log-sum-exp smoothing and very aggressive temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # For very low temperature, only the top weights matter
            threshold = max(1e-8, 0.01 / T_val)
            significant = w > threshold
            if not np.any(significant):
                k_star = int(np.argmax(b))
                g_a = np.zeros(n)
                js = max(0, k_star - n + 1)
                je = min(n - 1, k_star)
                idx = np.arange(js, je + 1)
                g_a[idx] = 2.0 * a[k_star - idx]
                S = np.dot(g_a, a)
                return a * (g_a - S)
            
            sig_indices = np.where(significant)[0]
            grad_a = np.zeros(n)
            for k in sig_indices:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (g_a - S) if False else a * (grad_a - S)
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # More iterations at lower temperatures (where refinement matters most)
        if stage_T < 0.5:
            niters = iters_per_stage * 3
        elif stage_T < 5.0:
            niters = iters_per_stage * 2
        elif stage_T < 20.0:
            niters = iters_per_stage
        else:
            niters = iters_per_stage // 2
        
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': niters, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': niters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                try_update(a_alt2, f"pair_n{alt_n}")
    
    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.35, 55.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=2.0, T_end=0.001, n_stages=12, iters_per_stage=400)
        try_update(a_fin, "final_lbgfs")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        t_pp_fin = min(rem * 0.25, 25.0)
        a_pp_fin, _ = peak_pair_optimize(best_seq, t_pp_fin, n_steps=20000)
        try_update(a_pp_fin, "final_pair")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        t_fista_fin = min(rem * 0.25, 20.0)
        a_fista_fin, _ = fista_refine(best_seq, t_fista_fin, n_max_iters=50000)
        try_update(a_fista_fin, "final_fista")
    
    rem = deadline - time.monotonic()
    if rem > 8:
        t_lp_fin = min(rem * 0.3, 20.0)
        if t_lp_fin > 5:
            a_lp_fin, _ = nesterov_lp_with_perturb(best_seq, t_lp_fin, solver=solver, n_max_iters=3000)
            try_update(a_lp_fin, "final_lp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
