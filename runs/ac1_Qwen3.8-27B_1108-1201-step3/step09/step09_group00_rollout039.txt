import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = n + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_rows(a, k_list):
    """Build rows of conv matrix for specific output positions."""
    n = len(a)
    rows = []
    for k in k_list:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)


def simplex_project(v, target_sum=1.0):
    """Project onto the probability simplex {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if target_sum <= 0:
        return np.ones(n) / n
    
    # Scale to sum=1 first, then we can scale back
    v = v / max(v.sum(), 1e-300)
    target = 1.0  # project to sum=1
    
    # Standard simplex projection
    if v.min() >= 0 and abs(v.sum() - 1) < 1e-15:
        return v
    
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    csv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (csv - 1))[0][-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


_lp_solver_cached = None


def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix T such that T @ b = conv(a, b)
    # T has shape (2n-1, n), T[k, j] = a[k-j] if 0 <= k-j < n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
                if alt == solver:
                    continue
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        pass
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None or not np.all(np.isfinite(b_val)):
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value) if M_var.value is not None else None


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=500000, verbose=True):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                # Add noise to escape
                noise = rng.standard_normal(n) * (0.05 / np.sqrt(stall_count))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            if verbose and (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

def mirror_descent_refine(a0, time_limit, n_steps=200000, verbose=True):
    """Mirror descent (multiplicative weights) with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Subgradient of max at peak_k: gradient of conv(a,a)_{peak_k}
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for j in range(js, je + 1):
            g[j] = 2.0 * a[peak_k - j]  # d/da_j of sum a_i a_{k-i}
        
        # Mirror descent: a_i <- a_i * exp(-lr * g_i)
        a_trial = a * np.exp(-lr * g)
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-300:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.8
                no_improve = 0
        
        lr = max(1e-6, lr)
        
        if verbose and step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000, verbose=True):
    """L-BFGS on LogSumExp-smoothed objective in log-domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
    
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z, T=stage_T):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_local = softmax_from_z(z)
            b = fast_convolve(a_local)
            Tb = b / T
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n) * log_sum_exp
        
        def jac(z, T=stage_T):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a_local = softmax_from_z(z)
            b = fast_convolve(a_local)
            Tb = b / T
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_local[k - idx]
            
            # Convert to z-gradient
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 100, 'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass
        
        if verbose:
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    L-BFGS T={stage_T}: score={sc:.8f} ({time.monotonic()-t0:.0f}s)")
    
    return softmax_from_z(z)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def peak_shave(a0, time_limit, n_steps=50000, verbose=True):
    """Directly shave the peak of the self-convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find peak contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Reduce top contributors, increase others
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        # Multiplicative update: reduce peak contributors
        a_trial = a.copy()
        for pos in top_positions:
            a_trial[pos] *= (1.0 - lr)
        
        mask = np.ones(n, dtype=bool)
        mask[top_positions] = False
        # Redistribute to non-contributing positions
        removed = a[top_positions].sum() - a_trial[top_positions].sum()
        if removed > 0:
            other_mass = a_trial[mask].sum()
            if other_mass > 1e-15:
                a_trial[mask] += removed * a_trial[mask] / other_mass
        
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
                # Try a different strategy: increase non-contributors
                a_trial2 = a.copy()
                a_trial2[top_positions] *= (1.0 - lr * 0.5)
                mask2 = np.ones(n, dtype=bool)
                mask2[top_positions] = False
                removed2 = a[top_positions].sum() - a_trial2[top_positions].sum()
                if removed2 > 0:
                    other_mass2 = a_trial2[mask2].sum()
                    if other_mass2 > 1e-15:
                        a_trial2[mask2] += removed2 * a_trial2[mask2] / other_mass2
                a_trial2 = np.maximum(a_trial2, 0)
                st2 = a_trial2.sum()
                if st2 > 1e-15:
                    a_trial2 = a_trial2 / st2
                score2 = 2.0 * n * fast_convolve(a_trial2).max()
                if score2 < best_score - 1e-15:
                    best_score = score2
                    best_a = a_trial2.copy()
                    a = a_trial2
                    no_improve = 0
        
        lr = max(1e-6, lr)
        
        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.maximum(0, np.cos(np.pi * x)) ** 3 + 0.05
    else:
        a = 1.0 / (1.0 + 5.0 * np.exp(-10 * (x - 0.5)))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)

def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, method='random'):
    """Create a perturbed version of the sequence."""
    n = len(a)
    a = a.copy()
    
    if method == 'random':
        scale = rng.uniform(0.05, 0.2)
        noise = rng.standard_normal(n) * scale
        a = a + noise
    elif method == 'lowfreq':
        freq = rng.integers(1, 5)
        amp = rng.uniform(0.03, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        a = a * (1.0 + amp * np.cos(2 * np.pi * freq * x + phase))
    elif method == 'anti_peak':
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_pos = idx[np.argsort(contribs)[::-1][:max(1, n // 10)]]
        a[top_pos] *= 0.7
    elif method == 'inversion':
        a = 1.0 - a
    elif method == 'blur':
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    elif method == 'sharpen':
        p = rng.uniform(1.2, 2.0)
        a = a ** p
        s = a.sum()
        if s > 1e-15:
            a = a / s
    else:
        a = a + rng.standard_normal(n) * 0.1
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v, target_sum=1.0):
    """Project vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    v = v / max(v.sum(), 1e-300)
    if v.min() >= 0 and abs(v.sum() - 1) < 1e-14:
        return v
    sv = np.sort(v)[::-1]
    csv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (csv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                noise_scale = 0.04 / max(1, np.sqrt(stall_count))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if verbose and (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain vector to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000, verbose=True):
    """L-BFGS on LogSumExp-smoothed objective in log-domain with adaptive T schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z, T_val=stage_T):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_loc = softmax_from_z(z)
            b = fast_convolve(a_loc)
            Tb = b / T_val
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n) * log_sum_exp

        def jac(z, T_val=stage_T):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a_loc = softmax_from_z(z)
            b = fast_convolve(a_loc)
            Tb = b / T_val
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_loc[k - idx]

            S = np.dot(grad_a, a_loc)
            return a_loc * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 50, 'disp': False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x
        except Exception:
            pass

        if verbose and (stage_i % 3 == 0 or stage_i == n_stages - 1):
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    LBFGS T={stage_T}: score={sc:.8f} ({time.monotonic()-t0:.0f}s)")

    return softmax_from_z(z)


def mirror_descent_refine(a0, time_limit, n_steps=200000, verbose=True):
    """Mirror descent (multiplicative weights) with adaptive step size and anti-peak boosting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.03
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Subgradient of max at peak
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        for j in range(js, je + 1):
            g[j] = 2.0 * a[peak_k - j]

        # Mirror descent step with small uniform regularization
        a_trial = a * np.exp(-lr * g + lr * 0.005 / n)
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-300:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 40:
                lr *= 0.85
                no_improve = 0

        lr = max(1e-7, lr)

        if verbose and step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=50000, verbose=True):
    """Directly reduce the maximum convolution peak by adjusting contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top_m = min(12, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        a_trial = a.copy()
        reduction = 1.0 - lr
        a_trial[top_positions] *= reduction

        mask = np.ones(n, dtype=bool)
        mask[top_positions] = False
        removed = a[top_positions].sum() - a_trial[top_positions].sum()
        if removed > 1e-15:
            other_mass = a_trial[mask].sum()
            if other_mass > 1e-15:
                a_trial[mask] += removed * a_trial[mask] / other_mass

        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 25:
                # Try broader reduction on top 20% of contributors
                a_trial2 = a.copy()
                n_top = max(1, len(idx) // 5)
                top20 = idx[np.argsort(contribs)[::-1][:n_top]]
                a_trial2[top20] *= (1.0 - lr * 0.5)
                mask2 = np.ones(n, dtype=bool)
                mask2[top20] = False
                removed2 = a[top20].sum() - a_trial2[top20].sum()
                if removed2 > 1e-15:
                    other_mass2 = a_trial2[mask2].sum()
                    if other_mass2 > 1e-15:
                        a_trial2[mask2] += removed2 * a_trial2[mask2] / other_mass2
                a_trial2 = np.maximum(a_trial2, 0)
                st2 = a_trial2.sum()
                if st2 > 1e-15:
                    a_trial2 = a_trial2 / st2
                score2 = 2.0 * n * fast_convolve(a_trial2).max()
                if score2 < best_score - 1e-15:
                    best_score = score2
                    best_a = a_trial2.copy()
                    a = a_trial2
                lr *= 0.8
                no_improve = 0

        lr = max(1e-7, lr)

        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence from a diverse set of templates."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.maximum(0, np.cos(np.pi * x)) ** 3 + 0.05
    elif choice == 7:
        a = 1.0 / (1.0 + 5.0 * np.exp(-10 * (x - 0.5)))
    elif choice == 8:
        a = 0.4 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    else:
        a = np.abs(np.sin(np.pi * x)) ** 0.7 + 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence using local computation (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, method='random'):
    """Create a perturbed version of the sequence using different strategies."""
    n = len(a)
    a = a.copy()

    if method == 'random':
        scale = rng.uniform(0.05, 0.15)
        a = a + rng.standard_normal(n) * scale
    elif method == 'lowfreq':
        freq = int(rng.integers(1, 6))
        amp = rng.uniform(0.03, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        a = a * (1.0 + amp * np.cos(2 * np.pi * freq * x + phase))
    elif method == 'anti_peak':
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_pos = idx[np.argsort(contribs)[::-1][:max(1, n // 8)]]
        a[top_pos] *= 0.6
    elif method == 'inversion':
        a = 1.0 - a
    elif method == 'blur':
        kernel_size = int(rng.integers(3, 9))
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif method == 'sharpen':
        p = rng.uniform(1.3, 2.5)
        a = a ** p
    elif method == 'two_bump':
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        x = np.linspace(0, 1, n)
        a = 0.5 * a + 0.3 * np.exp(-50 * (x - c1) ** 2) + 0.3 * np.exp(-50 * (x - c2) ** 2)
    else:
        a = a + rng.standard_normal(n) * 0.1

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov LP, mirror descent, L-BFGS LogSumExp,
    peak shaving, multi-restart with diverse perturbations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP on best sequence ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nlp_p1")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_md, _ = mirror_descent_refine(best_seq, t_alloc, n_steps=300000)
        try_update(a_md, "mirror_p2")

    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 30:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "lbgfs_p3")

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=80000)
        try_update(a_ps, "peak_p4")

    # === Phase 5: Try n=2000 ===
    print(f"\n=== Phase 5: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 250:
        t_alt = min(rem * 0.28, 380.0)
        n_alt = 2000
        print(f"  Trying n={n_alt}, budget={t_alt:.0f}s")
        init_alt = interpolate_sequence(best_seq, n_alt)
        sc_init = evaluate_local(init_alt)
        print(f"  Init score: {sc_init:.8f}")
        try_update(init_alt, "interp_n2000")

        t_lp_alt = t_alt * 0.55
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp_alt, solver=solver, n_max_iters=200000)
        try_update(a_alt, "lp_n2000")

        rem = deadline - time.monotonic()
        if rem > 60:
            t_lb_alt = min(rem * 0.35, 120.0)
            T_stages_alt = [15.0, 7.0, 3.0, 1.5, 0.7, 0.3, 0.15, 0.07, 0.03]
            a_alt2 = logsumexp_lbgfs(best_seq, t_lb_alt, T_stages=T_stages_alt,
                                      maxiter_per_stage=3000)
            try_update(a_alt2, "lbgfs_n2000")

    # === Phase 6: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 6: Multi-restart (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 180:
        methods = ['random', 'lowfreq', 'anti_peak', 'inversion', 'blur', 'sharpen', 'two_bump']
        n_restarts = min(len(methods), max(3, int(rem / 50)))
        t_per = max(25.0, rem * 0.30 / n_restarts)
        print(f"  {n_restarts} restarts, ~{t_per:.0f}s each")

        for i in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 35:
                break
            method = methods[i % len(methods)]
            perturbed = perturb_sequence(best_seq, rng, method=method)

            t_lp_r = min(t_per * 0.6, rem * 0.45)
            a_r, _ = nesterov_lp_refine(perturbed, t_lp_r, solver=solver, n_max_iters=40000)
            try_update(a_r, f"restart_{method}")

            rem = deadline - time.monotonic()
            if rem > 25:
                t_lb_r = min(t_per * 0.3, rem * 0.3)
                a_r2 = logsumexp_lbgfs(best_seq, t_lb_r,
                                        T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                        maxiter_per_stage=2000)
                try_update(a_r2, f"r{method}_lb")

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Quick mirror descent
        t_md = min(rem * 0.15, 50.0)
        if t_md > 10:
            a_md2, _ = mirror_descent_refine(best_seq, t_md, n_steps=50000)
            try_update(a_md2, "mirror_final")

        # Peak shave
        rem = deadline - time.monotonic()
        t_ps = min(rem * 0.20, 60.0)
        if t_ps > 10:
            a_ps2, _ = peak_shave(best_seq, t_ps, n_steps=50000)
            try_update(a_ps2, "peak_final")

        # Very low T L-BFGS
        rem = deadline - time.monotonic()
        t_fin = min(rem * 0.35, 80.0)
        if t_fin > 10:
            a_fin = logsumexp_lbgfs(best_seq, t_fin,
                                     T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                                     maxiter_per_stage=4000)
            try_update(a_fin, "lbgfs_final")

    # === Phase 8: Last chance peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        a_ls, _ = peak_shave(best_seq, min(rem * 0.4, 20.0), n_steps=15000)
        try_update(a_ls, "last_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    actual = evaluate_sequence(result)
    print(f"Verify with evaluate_sequence: {actual:.8f}")
    return result
