"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Direct LP (HiGHS) with Nesterov acceleration for global structure
2. Multi-stage L-BFGS with log-sum-exp smoothing and aggressive annealing
3. Enhanced peak shaving with water-filling and pair moves
4. Multi-length and multi-initialization exploration
5. Adaptive perturbation to escape local minima
Key improvements: faster LP via scipy HiGHS, better temperature schedules,
pair-swap moves, and more diverse exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed):
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    M_val = float(res.x[n])
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def nesterov_lp(a0, time_limit, n_max_iters=200000):
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if it % 500 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(99999 + it)
                noise = rng_local.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(np.clip(z_shifted, -100, 100))
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=None):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if n_stages is None:
        n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # grad_a[i] = sum_j a[j] * w[i+j]
            grad_a = np.zeros(n)
            for i in range(n):
                grad_a[i] = np.dot(a, w[i:i + n])
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        iters_per_stage = max(50, 20000 // n_stages)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def peak_shave_local(a0, time_limit, n_steps=20000):
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:10]
        sorted_contrib = np.argsort(contribs)

        improved = False
        # Try reducing top contributors
        for ci in top_contrib[:6]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.78, 0.70]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to low-contrib positions
                low_coords = idx[sorted_contrib[:n // 3]]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-12
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                else:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_trial[mask] += diff * a_trial[mask] / max(a_trial[mask].sum(), 1e-30)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pair moves
        if not improved:
            found = False
            for ci in top_contrib[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.93, 0.88, 0.82, 0.75]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = norm_sequence(a_trial)
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            if not improved:
                # Try single coordinate increase on low positions
                b_val = fast_convolve(a)
                min_k = int(np.argmin(b[1:-1])) + 1
                js2 = max(0, min_k - n + 1)
                je2 = min(n - 1, min_k)
                for i in range(js2, je2 + 1):
                    if a[i] > 0.95:
                        continue
                    for delta in [0.001, 0.005, 0.01, 0.02]:
                        a_trial = a.copy()
                        a_trial[i] += delta
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if not improved:
                no_improve += 1
                if no_improve > 300:
                    break
        else:
            no_improve = 0

    return best_a, best_score


def frank_wolfe_refine(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = a[k_star - idx]
        i_toward = int(np.argmin(g))
        best_trial = None
        best_trial_score = best_score
        for t_val in [0.4, 0.25, 0.12, 0.06, 0.03]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score_trial = 2.0 * n * fast_convolve(a_trial).max()
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()
        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.4, 0.25, 0.15, 0.08]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()
        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
    return best_a, best_score


def structured_init(n, rng, mode=0):
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * x * (1 - x))
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x)) * (1 + 0.2 * np.sin(6 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 4 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            if rem < 15:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)
            t_sub = min(rem * 0.25, 60.0)
            t_lp_pert = t_sub * 0.5
            a_pert, _ = nesterov_lp(perturbed, t_lp_pert, n_max_iters=20000)
            try_update(a_pert, f"pert_{scale}")
            rem = deadline - time.monotonic()
            if rem > 10 and best_seq is not None and len(best_seq) == n_cur:
                t_lb_pert = min(rem * 0.3, 30.0)
                if t_lb_pert > 8:
                    a_pert2 = lbgfs_refine(best_seq, t_lb_pert, T_start=10.0, T_end=0.1, n_stages=6)
                    try_update(a_pert2, f"pert_lb_{scale}")

    # === Phase 10: Final polish cascade ===
    print(f"\n=== Phase 10: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.3, 50.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.005, n_stages=10)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_fw = min(rem * 0.2, 30.0)
        a_fw2, _ = frank_wolfe_refine(best_seq, t_fw, n_steps=30000)
        try_update(a_fw2, "final_fw")

    rem = deadline - time.monotonic()
    if rem > 8:
        t_ps = min(rem * 0.2, 25.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=15000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 8:
        t_lp_fin = min(rem * 0.3, 20.0)
        a_lp_fin, _ = nesterov_lp(best_seq, t_lp_fin, n_max_iters=10000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
