import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def compute_grad_smooth(a, T):
    """Gradient of smoothed max: (1/T)*log(sum exp(T*b_k))."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad_i = 2 * sum_j a[j] * w[j+i]
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def compute_grad_topk(a, K=5):
    """Gradient using top-K peaks (subgradient-like)."""
    n = len(a)
    b = fast_convolve(a)
    # Get top K peak indices
    top_idx = np.argsort(b)[::-1][:K]
    # Uniform weights on top K
    w = np.zeros(len(b))
    for k in top_idx:
        w[k] = 1.0 / K
    # Same formula as before
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr

def coordinate_descent(a0, time_limit, n_rounds=100):
    """Exact 1D coordinate descent on peak-contributing coordinates."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find coordinates that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        contribs = a[contrib_idx] * a[peak_k - contrib_idx]
        top_contrib = contrib_idx[np.argsort(contribs)[::-1][:10]]
        
        improved = False
        for i in top_contrib:
            if a[i] < 1e-15:
                continue
            # Try reducing a[i] and redistributing
            for factor in [0.98, 0.96, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                delta = a_trial[i] * (1 - factor)
                a_trial[i] *= factor
                # Redistribute delta uniformly to other coordinates
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] += delta / mask.sum()
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        # Also try small random perturbations
        if not improved and rnd % 5 == 0:
            rng_local = np.random.default_rng(rnd * 1000)
            for _ in range(5):
                a_trial = a.copy()
                pert = rng_local.standard_normal(n) * 0.01
                a_trial = project_simplex(a_trial + pert)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    break
        
        if not improved:
            # Check if we're stuck - try a more aggressive perturbation
            if rnd % 10 == 0 and rnd > 0:
                rng_local = np.random.default_rng(rnd * 7)
                a_trial = a + rng_local.standard_normal(n) * 0.05
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                else:
                    a = a_trial  # Accept anyway to escape local min
            else:
                break

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):

def make_bimodal_init(n, rng):
    """Create a bimodal initialization with mass at both ends."""
    x = np.linspace(0, 1, n)
    # Mass concentrated near 0 and 1
    sigma = rng.uniform(0.1, 0.3)
    a = np.exp(-((x) / sigma)**2) + np.exp(-((x - 1) / sigma)**2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_trapped_init(n, rng):
    """Create initialization with trapped mass (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    depth = rng.uniform(0.3, 0.8)
    a = 1.0 - depth * np.sin(np.pi * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_random_smooth_init(n, rng):
    """Random smooth initialization."""
    # Generate random frequencies and phases
    n_freq = rng.integers(2, 6)
    a = np.ones(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_freq):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a = a * (1.0 + amp * np.sin(2 * np.pi * freq * x + phase))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
