def fast_convolve(a):

    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)

    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_subgradient(a, conv, n_active=10):
    """Compute subgradient of max(conv(a,a)) at point a.
    Uses top n_active peaks for a smoothed subgradient."""
    n = len(a)
    # Get top peaks
    k_act = min(n_active, 2 * n - 1)
    if k_act == 2 * n - 1:
        peaks = np.arange(2 * n - 1)
        weights = np.ones(k_act) / k_act
    else:
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        # Soft-max weights
        temp = 0.001 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        weights = w / w.sum()
    
    # Compute subgradient: for each active lag k, grad += w_k * 2 * a_reflected_shifted
    grad = np.zeros(n)
    for idx in range(k_act):
        if weights[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        w = 2.0 * weights[idx]
        # grad[j] += w * a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        # a[k-j] for j in [j_min, j_max]
        # k-j ranges from k-j_max to k-j_min
        idx_a = k - np.arange(j_min, j_max + 1)
        grad[j_min:j_max+1] += w * a[idx_a]
    
    return grad

def fista_optimize(a0, time_limit, verbose=False, label=""):
    """Accelerated subgradient method (FISTA-like) for the convex problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    iters = 0
    
    # Adaptive step size
    step = 1.0 / n
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Extrapolation (Nesterov)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a + (t / t_new - 1.0) * (a - y)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 0:
            y = y / ys
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, n_active=15)
        
        # Gradient step + projection
        a_new = simplex_project(y - step * grad)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive step size
        if patience > max_patience:
            step *= 0.8
            patience = 0
        
        a = a_new
        t = t_new
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, step={step:.6f}")
    
    return best_a, best_score, iters

def spectral_peak_reduction(a0, n_iters=50, verbose=False):
    """Iterative spectral method to reduce autocorrelation peaks.
    Based on the idea of making |A(k)|^2 more flat."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    
    for it in range(n_iters):
        # Compute spectrum
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: make power as flat as possible (but keep DC)
        # Use a soft "compression" of the power spectrum
        avg_power = power.sum() / fft_size
        # Reduce peaks in power spectrum
        gamma = 0.9  # compression factor
        power_new = np.where(power > avg_power, 
                           np.sqrt(gamma * power + (1-gamma) * avg_power),
                           power)
        # Preserve DC
        power_new[0] = power[0]
        
        # Take square root and inverse FFT
        mag_new = np.sqrt(power_new)
        # Preserve phases
        phases = np.angle(A)
        A_new = mag_new * np.exp(1j * phases)
        a_new = irfft(A_new, fft_size)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        # Check improvement
        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent: for each coordinate, use golden section search
    to find the value that minimizes the full score function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(k_hi - k_lo + 1)
            for idx in range(len(c_vals)):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            # Search range
            upper = max(2.0 * v, 5.0 / n)
            
            # Golden section search
            best_x = _golden_section(a, conv, i, v, upper, c_vals, a_partners,
                                     has_quad, quad_idx, unaff_max, n)

            if best_x is not None and abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def _golden_section(a, conv, i, v, upper, c_vals, a_partners, has_quad, quad_idx, unaff_max, n):
    """Golden section search for coordinate i."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    lo = 0.0
    hi = upper
    
    def f(x):
        ns = 1.0 + (x - v)
        if ns < 0.01:
            return 1e10
        nv = c_vals + 2.0 * x * a_partners
        if has_quad:
            nv[quad_idx] = c_vals[quad_idx] + x * x
        nm = max(unaff_max, float(nv.max()))
        return 2.0 * n * nm / (ns * ns)
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    x = (lo + hi) * 0.5
    if abs(x - v) < 1e-12:
        return None
    return x

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.85, 1.15, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated subgradient (FISTA-like) with active-set detection
2. Multi-scale warm starts (interpolation from smaller sizes)
3. Exact coordinate descent with golden section search
4. Spectral peak-reduction preprocessing
5. Adaptive time allocation across phases
6. Active-set LP refinement when solver available
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

DEADLINE = 0.0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_subgradient(a, conv, n_active=15):
    """Compute subgradient of max(conv(a,a)) at point a using top active peaks."""
    n = len(a)
    k_act = min(n_active, 2 * n - 1)
    if k_act == 2 * n - 1:
        peaks = np.arange(2 * n - 1)
        weights = np.ones(k_act) / k_act
    else:
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.001 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        weights = w / w.sum()

    grad = np.zeros(n)
    for idx in range(k_act):
        if weights[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        w = 2.0 * weights[idx]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        idx_a = k - np.arange(j_min, j_max + 1)
        grad[j_min:j_max + 1] += w * a[idx_a]

    return grad

def _score_at_x(x, v, c_vals, a_partners, has_quad, quad_idx, unaff_max, n):
    """Compute the score function value for a given coordinate value x."""
    ns = 1.0 + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)

def golden_section_coords(c_vals, a_partners, has_quad, quad_idx, unaff_max, n, v, upper, max_iter=40):
    """Golden section search for optimal value of a single coordinate. Returns x or None."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = upper

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_at_x(c, v, c_vals, a_partners, has_quad, quad_idx, unaff_max, n)
    fd = _score_at_x(d, v, c_vals, a_partners, has_quad, quad_idx, unaff_max, n)

    for _ in range(max_iter):
        if hi - lo < 1e-13:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_at_x(c, v, c_vals, a_partners, has_quad, quad_idx, unaff_max, n)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_at_x(d, v, c_vals, a_partners, has_quad, quad_idx, unaff_max, n)

    x = (lo + hi) * 0.5
    if abs(x - v) < 1e-12:
        return None
    return x

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section search.
    For each coordinate, find the exact minimizer of the score function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            n_aff = k_hi - k_lo + 1
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(2.5 * v, 5.0 / n)
            best_x = golden_section_coords(c_vals, a_partners, has_quad, quad_idx,
                                           unaff_max, n, v, upper)

            if best_x is not None:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD [{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def fista_optimize(a0, time_limit, verbose=False, label=""):
    """Accelerated subgradient method (FISTA-like) for the convex problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_nest = 1.0
    best_a = a.copy()
    conv_best = fast_convolve(a)
    best_score = 2.0 * n * float(conv_best.max())
    t0 = time.monotonic()
    iters = 0

    step = 1.0 / n
    patience = 0
    max_patience = 300
    best_step = step
    consecutive_bad = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        momentum = (t_nest / t_new - 1.0)
        y = a + momentum * (a - y)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, n_active=15)

        a_new = simplex_project(y - step * grad)

        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            conv_best = conv_new
            patience = 0
            consecutive_bad = 0
        else:
            patience += 1

        if patience > max_patience:
            step *= 0.75
            best_step = step
            patience = 0
            consecutive_bad += 1
            if consecutive_bad > 5:
                # Reset momentum to avoid oscillation
                y = a.copy()
                t_nest = 1.0
                step = max(step * 0.5, 1.0 / (10.0 * n))
                consecutive_bad = 0

        # Occasionally try increasing step if we're making progress
        if patience == 0 and step < 5.0 / n:
            step *= 1.05

        a = a_new
        t_nest = t_new

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA [{label}] it={iters} ({elapsed:.0f}s): best={best_score:.8f}, step={step:.6f}")

    return best_a, best_score, iters

def spectral_peak_reduction(a0, n_iters=80):
    """Iterative spectral method to reduce autocorrelation peaks."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())

    for it in range(n_iters):
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2

        avg_power = power[1:].mean()
        if avg_power < 1e-15:
            break

        gamma = 0.85 + 0.1 * (it / n_iters)
        power_new = power.copy()
        mask = power[1:] > avg_power
        power_new[1:][mask] = np.sqrt(gamma * power[1:][mask] + (1 - gamma) * avg_power)

        mag_new = np.sqrt(power_new)
        phases = np.angle(A)
        A_new = mag_new * np.exp(1j * phases)
        a_new = irfft(A_new, fft_size)[:n]

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        score = 2.0 * n * float(fast_convolve(a_new).max())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.55)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_bimodal(n, rng):
    """Bimodal initialization - two Gaussian bumps."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.25, 0.4)
    c2 = n * rng.uniform(0.6, 0.75)
    w1 = n * rng.uniform(0.08, 0.15)
    w2 = n * rng.uniform(0.08, 0.15)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturbed_restart(a, rng, sigma=0.02):
    """Create a perturbed version of a for basin hopping."""
    n = len(a)

            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat1', init_flat_spectrum(n, rng)))
        inits.append(('cos', init_cosine(n, rng)))
        inits.append(('bimodal', init_bimodal(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))

        for itype, cand in inits:
            t_left = 100.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(2.0, min(6.0, t_left / 6.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_{itype}")
            candidates.append((s_opt, a_opt, itype, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best internal: {candidates[0][0]:.8f}")

    warm_start = candidates[0][1].copy()
    for s_c, a_c, it_c, n_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: FISTA (accelerated subgradient) ===
    print(f"\n=== Phase 2: FISTA (best={best_score:.8f}) ===")
    rem = DEADLINE - time.monotonic() - 900
    if rem > 30:
        t_fista = min(rem * 0.7, 250.0)
        n_fista = len(warm_start)
        print(f"  FISTA n={n_fista}, budget={t_fista:.0f}s")
        a_fista, s_fista, iters = fista_optimize(warm_start, t_fista, verbose=True, label="fista")
        best_seq, best_score = check_and_update(a_fista, best_seq, best_score, f"fista_n{n_fista}")
        if s_fista < 2.0 * n_fista * float(fast_convolve(warm_start).max()) - 1e-6:
            warm_start = a_fista.copy()
            print(f"  FISTA improved warm start to {s_fista:.8f}")

    # === Phase 3: Spectral peak reduction ===
    print(f"\n=== Phase 3: Spectral Peak Reduction (best={best_score:.8f}) ===")
    rem = DEADLINE - time.monotonic() - 800
    if rem > 15:
        a_spec, s_spec = spectral_peak_reduction(warm_start, n_iters=60)
        best_seq, best_score = check_and_update(a_spec, best_seq, best_score, "spectral")
        if s_spec < 2.0 * len(warm_start) * float(fast_convolve(warm_start).max()) - 1e-6:
            warm_start = a_spec.copy()
            print(f"  Spectral improved to {s_spec:.8f}")
        else:
            print(f"  Spectral: {s_spec:.8f} (no improvement)")

    # === Phase 4: Intensive coordinate descent ===
    print(f"\n=== Phase 4: Intensive CD (best={best_score:.8f}) ===")
    n_cd = len(warm_start)
    for rnd in range(6):
        rem = DEADLINE - time.monotonic() - 500
        if rem < 20:
            break
        t_alloc = min(rem * 0.4, 180.0)
        print(f"  CD round {rnd + 1} n={n_cd}, budget={t_alloc:.0f}s")
        a_opt, s_opt, sweeps = exact_coord_descent(warm_start, t_alloc,
                                                   verbose=True, label=f"cd_r{rnd}")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"cd_r{rnd}")
        if s_opt < 2.0 * n_cd * float(fast_convolve(warm_start).max()) - 1e-8:
            warm_start = a_opt.copy()
        else:
            # Stuck - try a perturbation
            if rnd > 0:
                warm_start = perturbed_restart(warm_start, rng, sigma=0.01)
                print(f"  Perturbing to escape plateau (round {rnd})")

    # === Phase 5: Multi-scale exploration ===
    print(f"\n=== Phase 5: Multi-scale (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_base = len(best_seq)

        # Try 2x size
        rem = DEADLINE - time.monotonic() - 250
        if rem > 30:
            n_try = n_base * 2
            if n_try <= 4096:
                t_ms = min(rem * 0.5, 100.0)
                print(f"  Multi-scale n={n_try}, budget={t_ms:.0f}s")
                a_interp = interpolate_seq(best_seq, n_try)
                a_ms, s_ms, _ = exact_coord_descent(a_interp, t_ms, label=f"ms_{n_try}")
                best_seq, best_score = check_and_update(a_ms, best_seq, best_score, f"ms_n{n_try}")

        # Try 3/2 size
        rem = DEADLINE - time.monotonic() - 150
        if rem > 20:
            n_try2 = int(n_base * 1.5)
            if abs(n_try2 - n_base) > 64:
                t_ms2 = min(rem * 0.5, 80.0)
                print(f"  Multi-scale n={n_try2}, budget={t_ms2:.0f}s")
                a_interp2 = interpolate_seq(best_seq, n_try2)
                a_ms2, s_ms2, _ = exact_coord_descent(a_interp2, t_ms2, label=f"ms_{n_try2}")
                best_seq, best_score = check_and_update(a_ms2, best_seq, best_score, f"ms_n{n_try2}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # First: another FISTA pass on the best found
        rem = DEADLINE - time.monotonic() - 80
        if rem > 30:
            t_f2 = min(rem * 0.4, 80.0)
            print(f"  FISTA polish, budget={t_f2:.0f}s")
            a_f2, s_f2, _ = fista_optimize(best_seq, t_f2, label="fista_polish")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fista_polish")

        # Coordinate descent polish
        rem = DEADLINE - time.monotonic() - 40
        if rem > 15:
            t_cd = min(rem * 0.6, 80.0)
            print(f"  CD polish, budget={t_cd:.0f}s")
            a_cd, s_cd, _ = exact_coord_descent(best_seq, t_cd, verbose=True, label="cd_polish")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_polish")

        # Final fine pass
        rem = DEADLINE - time.monotonic() - 15
        if rem > 8:
            a_fin, s_fin, _ = exact_coord_descent(best_seq, min(rem * 0.8, 25.0), label="final")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
