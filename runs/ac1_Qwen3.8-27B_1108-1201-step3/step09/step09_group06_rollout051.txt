"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage L-BFGS with LogSumExp smoothing (FFT-accelerated gradients)
2. LP-based best response (Frank-Wolfe / conditional gradient)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_lse_objective(z, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    n = len(a)
    return (2.0 * n / T) * log_sum_exp


def compute_lse_gradient(z, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    n = len(a)
    size = 2 * n - 1
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # grad_a = 2 * correlate(w, a)
    # correlate(w, a)[j] = sum_k w[k] * a[k-j]
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 2.0 * corr
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbgfs_multistage(a0, time_limit, T_start=300.0, T_end=0.05, n_stages=12, maxiter_per_stage=5000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iters = min(maxiter_per_stage, int(remaining * 50))
        max_iters = max(max_iters, 50)

        try:
            res = minimize(
                fun=lambda z_in: compute_lse_objective(z_in, stage_T),
                x0=z,
                jac=lambda z_in: compute_lse_gradient(z_in, stage_T),
                method='L-BFGS-B',
                options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        a_cur = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if si % 3 == 0 or si == n_stages - 1:
            print(f"    L-BFGS stage {si+1}/{n_stages} T={stage_T:.2f} ({elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def lbgfs_adaptive(a0, time_limit):
    """Adaptive L-BFGS that monitors progress and adjusts temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    T = 200.0
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
    stall_rounds = 0

    while time.monotonic() - t0 < time_limit - 10:
        max_T = time_limit - (time.monotonic() - t0) - 15
        if max_T < 10:
            break
        prev_score = best_score
        try:
            res = minimize(
                fun=lambda z_in: compute_lse_objective(z_in, T),
                x0=z,
                jac=lambda z_in: compute_lse_gradient(z_in, T),
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 900000, 'disp': False}
            )
            z = res.x
        except Exception:
            break

        a_cur = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if sc < best_score:
            best_score = sc
            best_z = z.copy()
            stall_rounds = 0
        else:
            stall_rounds += 1

        elapsed = time.monotonic() - t0
        print(f"    L-BFGS-Adapt T={T:.2f} ({elapsed:.0f}s): score={best_score:.6f} stall={stall_rounds}")

        if stall_rounds >= 3:
            break
        if T > 0.05:
            T = max(T * 0.4, 0.05)
        else:
            if stall_rounds >= 1:
                break

    return softmax_from_z(best_z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    if s < 1e-15:

    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_score - 1e-8:

                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0

def simulated_annealing_search(a0, time_limit, T_start=0.01, T_end=0.0001, n_iter=500000):
    """Local search with simulated annealing to escape local minima."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    cur_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    T = T_start

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iter))

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.08 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:4]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Choose perturbation type
        ptype = rng.integers(0, 4)
        a_trial = a.copy()

        if ptype == 0:
            # Move mass from high contrib to low contrib
            top_src = np.argsort(contrib)[::-1][:3]
            low_dst = np.argsort(contrib)[:3]
            si = top_src[rng.integers(len(top_src))]
            di = low_dst[rng.integers(len(low_dst))]
            if si == di or a[si] < 1e-12:
                continue
            amount = a[si] * rng.uniform(0.01, 0.15)
            a_trial[si] -= amount
            a_trial[di] += amount
        elif ptype == 1:
            # Random swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-12:
                continue
            amount = a[i] * rng.uniform(0.01, 0.1)
            a_trial[i] -= amount
            a_trial[j] += amount
        elif ptype == 2:
            # Spread: take from one, add to two neighbors
            i = rng.integers(n)
            if a[i] < 1e-12:
                continue
            amount = a[i] * rng.uniform(0.05, 0.2)
            a_trial[i] -= amount
            j1 = (i + 1) % n
            j2 = (i + n - 1) % n
            a_trial[j1] += amount * rng.uniform(0.3, 0.7)
            a_trial[j2] += a_trial[j1] * 0  # placeholder
            # Redo properly
            a_trial = a.copy()
            w1 = rng.uniform(0.3, 0.7)
            a_trial[i] -= amount
            a_trial[j1] += amount * w1
            a_trial[j2] += amount * (1 - w1)
        else:
            # Gaussian perturbation
            noise = rng.normal(0, 0.05, n)
            a_trial = a + noise

        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        new_score = 2.0 * n * fast_convolve(a_trial).max()

        # Metropolis acceptance
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score

def parametric_init(n, rng, seed_type=None):
    """Generate diverse parametric initializations."""
    if seed_type is None:
        seed_type = rng.integers(0, 10)

    x = np.linspace(0, 1, n)
    xc = x - 0.5  # centered in [-0.5, 0.5]

    if seed_type == 0:
        # Generalized cosine
        p = rng.uniform(0.5, 3.0)
        a = np.cos(np.pi * xc) ** (2 * p)
    elif seed_type == 1:
        # Gaussian
        sigma = rng.uniform(0.15, 0.45)
        a = np.exp(-xc ** 2 / (2 * sigma ** 2))
    elif seed_type == 2:
        # Super-ellipse
        p = rng.uniform(1.0, 4.0)
        a = np.maximum(1 - np.abs(xc) * 2, 0) ** (1.0 / p)
    elif seed_type == 3:
        # Lorentzian
        gamma = rng.uniform(0.1, 0.4)
        a = 1.0 / (1.0 + (xc / gamma) ** 2)
    elif seed_type == 4:
        # Raised cosine with power
        p = rng.uniform(0.5, 2.0)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * xc)) ** p
    elif seed_type == 5:
        # Two-peak
        sep = rng.uniform(0.15, 0.35)
        a = np.exp(-(xc + sep) ** 2 / 0.02) + np.exp(-(xc - sep) ** 2 / 0.02)
        a += 0.3 * np.exp(-xc ** 2 / 0.05)
    elif seed_type == 6:
        # Uniform with cosine edge taper
        taper = rng.uniform(0.3, 1.0)
        a = np.ones(n)
        edge = int(n * 0.15)
        for i in range(edge):
            f = 1 - taper * np.cos(np.pi * i / (2 * edge))
            a[i] *= f
            a[n - 1 - i] *= f
    elif seed_type == 7:
        # Power law
        alpha = rng.uniform(0.3, 2.0)
        a = (1 - np.abs(xc) * 2) ** alpha
        a = np.maximum(a, 0)
    elif seed_type == 8:
        # Bump + small oscillation
        a = np.exp(-xc ** 2 / 0.05)
        freq = rng.integers(3, 10)
        amp = rng.uniform(0.05, 0.2)
        a *= (1 + amp * np.cos(2 * np.pi * freq * x))
        a = np.maximum(a, 0)
    else:
        # Random smooth (via polynomial coefficients)
        deg = rng.integers(3, 8)
        coeffs = rng.uniform(-1, 1, deg + 1)
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * x ** i
        a = np.maximum(a, 0)
        if a.max() < 1e-10:
            a = np.exp(-xc ** 2 / 0.05)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    best_n = 960

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            best_n = len(parent)
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score, best_n
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = len(a)
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}) [{time.monotonic()-t_start:.0f}s] ***")

    # === Phase 1: Multi-start screening ===
    print(f"\n=== Phase 1: Multi-Start Screening (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.12, 200.0)
    if t_phase1 > 20:
        print(f"  Budget: {t_phase1:.0f}s")
        candidates = []

        # Add parent (if exists)
        if best_seq is not None:
            candidates.append(best_seq.copy())

        # Generate diverse parametric inits at best_n
        for i in range(8):
            a_init = parametric_init(best_n, rng, seed_type=i)
            candidates.append(a_init)

        # Generate at a couple other lengths
        for alt_n in [int(best_n * 0.85), int(best_n * 1.15)]:
            if 200 <= alt_n <= 3000:
                a_init = parametric_init(alt_n, rng, seed_type=rng.integers(0, 10))
                candidates.append(a_init)

        # Quick screening: a few L-BFGS iterations on each
        per_candidate_time = max(5.0, t_phase1 / len(candidates))
        for ci, cand in enumerate(candidates):
            if time.monotonic() > deadline - 30:
                break
            cand_time = min(per_candidate_time, deadline - time.monotonic() - 20)
            if cand_time < 3:
                break
            a_screened = lbgfs_multistage(cand, cand_time, T_start=100.0, T_end=0.5, n_stages=4, maxiter_per_stage=1000)
            try_update(a_screened, f"screen_{ci}")

    # === Phase 2: Deep L-BFGS on best ===
    print(f"\n=== Phase 2: Deep L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.30, 500.0)
    if t_phase2 > 30:
        print(f"  Budget: {t_phase2:.0f}s")
        a_deep = lbgfs_multistage(best_seq, t_phase2, T_start=500.0, T_end=0.02, n_stages=15, maxiter_per_stage=5000)
        try_update(a_deep, "deep_lbgfs")

    # === Phase 3: Adaptive L-BFGS ===
    print(f"\n=== Phase 3: Adaptive L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.20, 350.0)
    if t_phase3 > 30:
        print(f"  Budget: {t_phase3:.0f}s")
        a_adapt = lbgfs_adaptive(best_seq, t_phase3)
        try_update(a_adapt, "adapt_lbgfs")

    # === Phase 4: LP-based refinement ===
    print(f"\n=== Phase 4: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.15, 250.0)
    if t_phase4 > 20:
        print(f"  Budget: {t_phase4:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_phase4, solver=solver, n_max_iters=200000)
        try_update(a_lp, "lp_refine")

    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.15, 250.0)
    if t_phase5 > 20:
        print(f"  Budget: {t_phase5:.0f}s")
        a_sa, _ = simulated_annealing_search(best_seq, t_phase5, T_start=0.005, T_end=0.00001, n_iter=300000)
        try_update(a_sa, "sim_anneal")

        # Polish the SA result
        rem = deadline - time.monotonic()
        if rem > 30:
            t_polish = min(rem * 0.3, 60.0)
            a_pol = lbgfs_multistage(a_sa, t_polish, T_start=50.0, T_end=0.1, n_stages=6, maxiter_per_stage=2000)
            try_update(a_pol, "sa_polish")

    # === Phase 6: Deterministic local search ===
    print(f"\n=== Phase 6: Local Deterministic Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_phase6 = min(rem * 0.10, 150.0)
    if t_phase6 > 15:
        print(f"  Budget: {t_phase6:.0f}s")
        a_ls, _ = local_deterministic_search(best_seq, t_phase6, n_steps=200000)
        try_update(a_ls, "local_det")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [0.9, 1.1, 1.2, 0.8, 1.3]:
        alt_n = int(best_n * mult)
        if 400 <= alt_n <= 2500 and alt_n != best_n:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alt = min(rem * 0.30, 180.0)
        if t_alt < 20:
            break
        print(f"  n={alt_n}, budget={t_alt:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            best_n = alt_n
            print(f"  Interpolation improved: {sc_alt:.6f}")

        # L-BFGS on this length
        t_lb = t_alt * 0.6
        a_alt = lbgfs_multistage(init_alt, t_lb, T_start=200.0, T_end=0.1, n_stages=8, maxiter_per_stage=3000)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        # Quick LP
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lp_alt = min(rem * 0.3, 50.0)
            if t_lp_alt > 10:
                a_lp_alt, _ = nesterov_lp_refine(best_seq, t_lp_alt, solver=solver, n_max_iters=30000)
                try_update(a_lp_alt, f"lp_n{alt_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.4, 100.0)
        print(f"  Final L-BFGS budget: {t_fin:.0f}s")
        a_fin = lbgfs_multistage(best_seq, t_fin, T_start=30.0, T_end=0.01, n_stages=8, maxiter_per_stage=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ls = min(rem * 0.3, 40.0)
        a_ls2, _ = local_deterministic_search(best_seq, t_ls, n_steps=50000)
        try_update(a_ls2, "final_local")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_sa2 = min(rem * 0.4, 30.0)
        a_sa2, _ = simulated_annealing_search(best_seq, t_sa2, T_start=0.001, T_end=0.00001, n_iter=100000)
        try_update(a_sa2, "final_sa")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}) [{elapsed:.0f}s]")
    return [float(x) for x in best_seq]
