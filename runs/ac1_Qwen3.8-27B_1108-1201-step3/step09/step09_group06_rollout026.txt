def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / (1.0 - cssv[-1] / n * (n - 1)) < 0) if n > 1 else 0
    # Standard simplex projection
    theta = (v.sum() - s) / n if v.sum() > s else 0
    # Use the standard algorithm
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cumsum - s) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) * s / n
    rho_star = idx[-1] + 1
    theta = (cumsum[rho_star - 1] - s) / rho_star
    return np.maximum(v - theta, 0)

    # Build Hankel matrix H[i,j] = alpha[i+j]
    H = np.zeros((n, n))
    for i in range(n):
        H[i, i:i + n] = alpha[i:i + n]
    
    # Solve H @ a_new = lambda * ones
    # Use least squares for robustness
    rhs = np.ones(n)
    try:
        a_unscaled, residuals, rank, sv = np.linalg.lstsq(H, rhs, rcond=1e-12)
    except Exception:
        return a.copy()
    
    s = a_unscaled.sum()
    if abs(s) < 1e-15:
        return a.copy()
    a_new = a_unscaled / s
    
    # Ensure non-negativity
    if (a_new < -1e-10).any():
        # Project onto simplex
        a_new = project_simplex(a_new, 1.0)
    
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss < 1e-15:
        return a.copy()
    a_new = a_new / ss
    
    # Damped update
    a_result = (1 - damping) * a + damping * a_new
    a_result = np.maximum(a_result, 0)
    ss = a_result.sum()
    if ss < 1e-15:
        return a.copy()
    return a_result / ss


def kkt_iterative_refine(a0, time_limit, solver=None, n_max_iters=50000, damping_schedule=None):
    """
    KKT-based iterative refinement:
    1. Solve LP to get dual alpha
    2. Solve Hankel system for new a
    3. Damped update
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val, alpha = solve_lp_and_get_dual(a, solver)
        if b_new is None:
            stall += 1
            if stall > 20:
                break
            continue
        
        # KKT step
        damping = 0.3 if it < 100 else (0.1 if it < 500 else 0.05)
        a_kkt = kkt_hankel_step(a, alpha, damping=damping)
        
        # Also try the Nesterov-style update
        alpha_nest = 2.0 / (it + 2)
        a_nest = (1 - alpha_nest) * a + alpha_nest * b_new
        a_nest = np.maximum(a_nest, 0)
        a_nest = a_nest / a_nest.sum()
        
        # Pick the better one
        score_kkt = 2.0 * n * fast_convolve(a_kkt).max()
        score_nest = 2.0 * n * fast_convolve(a_nest).max()
        
        if score_kkt <= score_nest:
            a = a_kkt
            score = score_kkt
        else:
            a = a_nest
            score = score_nest
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Try a random perturbation
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b, alpha = solve_lp_and_get_dual(y, solver)
        if b_new is None:
            continue
        
        # Nesterov acceleration
        alpha_step = 2.0 / (it + 2)
        y = (1 - alpha_step) * y + alpha_step * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
            continue
        y = y / sy
        
        # Heavy ball / Nesterov momentum for a
        momentum = (it + 1) / (it + 3)
        a_new = momentum * a + (1 - momentum) * b_new
        # Actually standard Nesterov: a_{k+1} = ((k+1)*a_k + (beta_k*(k+2))*b) / (k+1 + beta_k*(k+2))
        a = ((it + 1) * a + (alpha_step * (it + 2)) * b_new) / (it + 1 + alpha_step * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Also check y
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if (it + 1) % 500 == 0:
            if stall_count >= 20:
                rng = np.random.default_rng(it * 13)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = y.copy()
                stall_count = 0
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=15000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def func_factory(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def jac_factory(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=func_factory(stage_T),
                x0=z,
                jac=jac_factory(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def local_swap_search(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find indices that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Also consider near-peaks
        threshold = peak_val * 0.95
        near_peaks = np.where(b >= threshold)[0]
        
        total_contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js_k = max(0, k - n + 1)
            je_k = min(n - 1, k)
            idx_k = np.arange(js_k, je_k + 1)
            total_contrib[idx_k] += a[idx_k] * a[k - idx_k]
        
        # Source: high contribution, Destination: low contribution
        top_src = np.argsort(total_contrib)[::-1][:3]
        low_dst = np.argsort(total_contrib)[:5]
        
        improved = False
        for si in top_src:
            if a[si] < 1e-14:
                continue
            for di in low_dst:
                if si == di:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Random perturbation every so often
            if no_improve % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-14:
                    amount = a[i] * 0.02
                    a[i] -= amount
                    a[j] += amount
                    a = np.maximum(a, 0)
                    a = a / a.sum()
            
            if no_improve > 2000:
                break
    
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        # Flat-ish initialization (aiming for uniform convolution)
        a = np.ones(n) / n
        # Add small perturbation
        a += rng.random(n) * 0.01
    else:
        # Parabolic
        x = np.linspace(-1, 1, n)
        a = 1 - x ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
