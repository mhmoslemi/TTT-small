    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1, M_var >= 0]

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(b_fixed):
    """Build the convolution matrix T where T[k,:] gives the coefficients for (b_fixed * a)[k]."""
    n = len(b_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        T[k, js:je+1] = b_fixed[k - (je:js-1:-1)]  # reversed
