        score_y = score_normalized(y)
        cur_best = min(score_a, score_y)

        if cur_best < best_score - 1e-12:
            if score_a <= score_y:
                best_a = a.copy()
            else:
                best_a = y.copy()
            best_score = cur_best
            stall_count = 0
            step_scale = min(step_scale * 1.02, 1.5)
        else:
            stall_count += 1

        if it % 300 == 0:
            if best_score < prev_score - 1e-9:
                stall_count = max(0, stall_count - 3)
            prev_score = best_score

            if stall_count >= 15 and restart_count < max_restarts:
                restart_count += 1
                b_curr = fast_convolve(y)
                peak_val = b_curr.max()
                threshold = peak_val - 0.03 * peak_val
                active_peaks = np.where(b_curr >= threshold)[0]

                pressure = np.zeros(n)
                for k in active_peaks:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    pressure[idx] += y[k - idx]

                pressure = np.maximum(pressure, 1e-15)
                target = y.sum() / n
                eta = 0.1
                y_perturbed = y * (target / pressure) ** eta
                y_perturbed = np.maximum(y_perturbed, 0)
                sp = y_perturbed.sum()
                if sp > 1e-15:
                    y = y_perturbed / sp
                a = y.copy()
                stall_count = 0
                step_scale = max(0.5, step_scale * 0.7)

        if (it + 1) % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} stall={stall_count} rs={restart_count}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=12000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, n_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.04 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        if len(near_peaks) == 0:
            near_peaks = np.array([peak_k])

        pressure = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += a[idx] * a[k - idx]

        total_pressure = pressure.sum()
        if total_pressure < 1e-30:
            break

        target = a.sum() / n
        high_pressure = np.where(pressure > 1.2 * target * total_pressure / n)[0]
        low_pressure = np.where(pressure < 0.8 * target * total_pressure / n)[0]

        improved = False

        if len(high_pressure) > 0 and len(low_pressure) > 0:
            si = high_pressure[np.argmax(pressure[high_pressure])]
            di = low_pressure[np.argmin(pressure[low_pressure])]

            if a[si] > 1e-14:
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = score_normalized(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break

        if not improved:
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * 0.02
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = score_normalized(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Add small noise to escape
                noise = rng.random(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
            if no_improve > 5000:
                break

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def mirror_descent_v2(a0, time_limit, n_active_frac=0.05):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.01)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def dual_guided_update(a0, time_limit, solver=None):
    """Use LP dual variables to guide direct updates to a."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        # Use the LP solution as a direction
        # The optimal b tells us where to "move mass"
        # Update a toward b with adaptive step
        b_curr = fast_convolve(a)
        max_b_curr = b_curr.max()
        b_b = fast_convolve(b_new)
        max_b_b = b_b.max()

        # Weighted average based on which is better
        w = 0.3
        a_trial = (1 - w) * a + w * b_new
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            break
        a_trial = a_trial / st
        score_trial = score_normalized(a_trial)

        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a.copy()
        else:
            # Try smaller step
            a_trial = (1 - 0.1) * a + 0.1 * b_new
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score_trial = score_normalized(a_trial)
            if score_trial < best_score - 1e-14:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
            else:
                if it > 100:
                    break

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DUAL iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        parent = best_seq.copy()
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")

    # === Phase 1: Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_v2(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 30:
        a_md, _ = mirror_descent_v2(best_seq, t_alloc, n_active_frac=0.04)
        try_update(a_md, "mirror_desc")

    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=100.0, T_end=0.1, maxiter=12000)
        try_update(a_lb, "lbgfs_1")

    # === Phase 4: Nesterov LP Round 2 ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_v2(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_steps=50000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Alternative Lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        alt_lengths = []
        for n_alt in [int(n_best * 0.85), int(n_best * 1.15), int(n_best * 1.3)]:
            if 400 <= n_alt <= 3000 and n_alt != n_best:
                alt_lengths.append(n_alt)
        alt_lengths = sorted(set(alt_lengths))[:2]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_total = min(rem * 0.4, 180.0)
            if t_total < 30:
                break
            print(f"  n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_seq, alt_n)
            sc_alt = evaluate_safe(init_alt)
            cur_best_alt = init_alt.copy()
            if sc_alt < best_score:
                best_score = sc_alt
                best_seq = init_alt.copy()

            t_lp = t_total * 0.7
            a_alt, _ = nesterov_lp_v2(cur_best_alt, t_lp, solver=solver, n_max_iters=200000)
            try_update(a_alt, f"lp_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
                t_md = min(rem * 0.3, 40.0)
                if t_md > 10:
                    a_alt2, _ = mirror_descent_v2(best_seq, t_md, n_active_frac=0.05)
                    try_update(a_alt2, f"md_n{alt_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.05, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.35, 50.0)
        a_cd2, _ = coordinate_descent(best_seq, t_cd, n_steps=20000)
        try_update(a_cd2, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.4, 40.0)
        a_lp_fin, _ = nesterov_lp_v2(best_seq, t_lp_fin, solver=solver, n_max_iters=50000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
