"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return solver_name


def solve_lp_best_response(a_fixed, solver=None):
    """Find b >= 0, sum(b)=1 that minimizes max_k (a*b)_k."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternating_nesterov(a0, time_limit, solver=None, max_iters=200000):
    """LP alternating with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 5:
                # Perturbation to escape stall
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, max_steps=100000):
    """Direct peak-shaving: find the peak lag and move mass to reduce it."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find which indices contribute to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributors = np.arange(js, je + 1)
        contrib_vals = a[contributors] * a[peak_k - contributors]
        
        # Move mass from high-contributor to random low-index position
        if len(contributors) > 0:
            # Pick the largest contributor
            max_contrib_idx = contributors[np.argmax(contrib_vals)]
            # Move mass to a random position that doesn't contribute to this peak
            safe_positions = [i for i in range(n) if i < js or i > je]
            if safe_positions:
                target = safe_positions[rng.integers(len(safe_positions))]
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amount = a[max_contrib_idx] * frac
                    a_trial = a.copy()
                    a_trial[max_contrib_idx] -= amount
                    a_trial[target] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        break
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amount = a[i] * 0.05
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
            
            if no_improve > 2000:
                break
    
    return best_a, best_score


def mirror_descent(a0, time_limit, max_steps=200000):
    """Mirror descent with adaptive step size targeting peak reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Compute subgradient: gradient of max through active set
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        gap = best_score - 2.0 * n / (2.0 * n - 1.0)
        alpha = min(lr, gap / (2.0 * g_norm * best_score + 1e-15))
        alpha = max(alpha, 0.005)
        
        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)
        
        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000):
    """L-BFGS with LogSumExp approximation."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def asymmetric_bump_init(n, center_frac, width, rng):
    """Create an asymmetric bump initialization."""
    x = np.linspace(0, 1, n)
    center = center_frac
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    # Add slight asymmetry
    a *= (1 + 0.3 * np.sin(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_asymmetric_init(n, rng):
    """Random asymmetric initialization."""
    x = np.linspace(0, 1, n)
    a = rng.random(n)
    # Add a smooth component
    a += 0.5 * np.exp(-((x - rng.random()) ** 2) / (2 * (0.1 + 0.3 * rng.random()) ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return solver_name


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def mirror_descent_adaptive(a0, time_limit, max_steps=300000, active_frac=0.05):
    """Mirror descent with Polyak step and adaptive active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    lr = 0.4
    no_improve = 0
    f_lb = 2.0 * n / (2.0 * n - 1.0)

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b - active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-12:
            break

        polyak_lr = gap / (2.0 * g_norm * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.75
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.05, maxiter=10000):
    """L-BFGS with multi-stage LogSumExp annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 7
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(25, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave(a0, time_limit, max_steps=50000):
    """Direct peak-shaving: target the highest peak and redistribute mass."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributors = np.arange(js, je + 1)

        if len(contributors) == 0:
            no_improve += 1
            continue

        contrib_vals = a[contributors] * a[peak_k - contributors]
        max_idx = int(contributors[np.argmax(contrib_vals)])

        safe_positions = np.concatenate([
            np.arange(0, js),
            np.arange(je + 1, n)
        ])

        improved = False
        if len(safe_positions) > 0:
            for _ in range(3):
                target = int(safe_positions[rng.integers(len(safe_positions))])
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amount = a[max_idx] * frac
                    a_trial = a.copy()
                    a_trial[max_idx] -= amount
                    a_trial[target] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            for _ in range(2):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                amount = a[i] * 0.04
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 300:
                break

    return best_a, best_score


def asymmetric_bump(n, center, width):
    """Create an asymmetric bump sequence centered at 'center' fraction."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a *= (1 + 0.2 * np.cos(2 * np.pi * (x - center)))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_bump(n, c1, c2, w1, w2):
    """Two-bump initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + 0.8 * np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_smooth_asymmetric(n, rng):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    n_best = len(best_seq) if best_seq is not None else 960

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 0: Generate and evaluate diverse initializations ===
    print(f"\n=== Phase 0: Initialization (current best={best_score:.6f}) ===")
    candidates = []

    if parent is not None:
        candidates.append((best_score, parent.copy(), "parent"))

    for center in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.55, 0.6, 0.65, 0.7, 0.8]:
        for width in [0.1, 0.15, 0.2, 0.25, 0.3]:
            a_init = asymmetric_bump(n_best, center, width)
            sc = evaluate_safe(a_init)
            candidates.append((sc, a_init, f"bump_{center}_{width}"))

    for c1 in [0.2, 0.3, 0.7]:
        for c2 in [0.5, 0.6, 0.8]:
            a_init = two_bump(n_best, c1, c2, 0.15, 0.15)
            sc = evaluate_safe(a_init)
            candidates.append((sc, a_init, f"2bump_{c1}_{c2}"))

    for _ in range(8):
        a_init = random_smooth_asymmetric(n_best, rng)
        sc = evaluate_safe(a_init)
        candidates.append((sc, a_init, "random"))

    candidates.sort(key=lambda x: x[0])
    print(f"  Best candidate: {candidates[0][0]:.6f} ({candidates[0][2]})")
    print(f"  Top 5: {[f'{c[0]:.4f}' for c in candidates[:5]]}")

    # Start from the best candidate
    if candidates[0][0] < best_score:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.50, 800.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating_nesterov(best_seq, t_alloc, solver=solver, max_iters=500000)
        try_update(a_lp, "lp_phase1")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        a_md, _ = mirror_descent_adaptive(best_seq, t_alloc, active_frac=0.04)
        try_update(a_md, "mirror_desc")

    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=60.0, T_end=0.05, maxiter=12000)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave(best_seq, t_alloc, max_steps=80000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Second LP round from perturbed start ===
    print(f"\n=== Phase 5: LP Alt r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 40:
        # Try from second-best initialization too
        second_best = candidates[0][1].copy()
        if candidates[0][2] != "parent" and parent is not None:
            second_best = parent.copy()
        a_pert = second_best + rng.random(len(second_best)) * 0.02
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_lp2, _ = lp_alternating_nesterov(a_pert, t_alloc, solver=solver, max_iters=300000)
        try_update(a_lp2, "lp_phase2")

    # === Phase 6: Alternate length ===
    print(f"\n=== Phase 6: Alt Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    for mult in [1.25, 0.75, 1.5]:
        alt_n = int(n_best * mult)
        if not (400 <= alt_n <= 3000) or alt_n == n_best:
            continue
        if rem < 80:
            break
        t_total = min(rem * 0.30, 150.0)
        if t_total < 40:
            break
        print(f"  Trying n={alt_n} (budget={t_total:.0f}s)")
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = cur.copy()

        a_alt, _ = lp_alternating_nesterov(cur, t_total * 0.6, solver=solver, max_iters=100000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            a_md_alt, _ = mirror_descent_adaptive(best_seq, min(rem * 0.3, 50.0), active_frac=0.05)
            try_update(a_md_alt, f"md_n{alt_n}")
        break  # Only try one alternate length

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        a_fin = lbgfs_refine(best_seq, min(rem * 0.35, 50.0), T_start=8.0, T_end=0.03, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_ps2, _ = peak_shave(best_seq, min(rem * 0.30, 30.0), max_steps=20000)
        try_update(a_ps2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_md3, _ = mirror_descent_adaptive(best_seq, min(rem * 0.4, 20.0), active_frac=0.02)
        try_update(a_md3, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
