Actually, in the game formulation:
- State: probability `b` on [n] (the "strategy" we're optimizing)
- Opponent: probability `u` on [2n-1] (the "environment")
- Payoff: `u^T * (T*b)` = `(T*b)^T * u`

The best response of the minimizer (us) to a fixed `u` is: `b* = e_{argmin_i (T^T*u)_i}`
The best response of the maximizer (env) to a fixed `b` is: `u* = e_{argmax_k (T*b)_k}`

In the Nesterov/FTRL approach:
- We maintain a "virtual" strategy `y` (the extrapolated point)
- The opponent plays the best response to `y`: `u* = e_{argmax_k (T*y)_k}`
- We update our strategy based on the loss from `u*`

The loss for strategy `b_i` when the opponent plays `u*` is: `(T*b)_i * u*_i`... no wait.

Let me re-derive. In the FTRL/Online Mirror Descent framework for zero-sum games:

At round t:
1. Player 1 plays `a_t` (based on past losses)
2. Nature (Player 2) plays `e_{k_t}` where `k_t = argmax_k (T*a_t)_k` (worst case)
3. Loss for Player 1: `l_t(b) = (T*b)_{k_t}` (linear in b)
4. Update: `a_{t+1} = argmin_b sum_{s<=t} l_s(b) + R(b)` where R is a regularizer

For FTRL with entropy regularizer:
`a_{t+1}(i) = exp(-sum_{s<=t} (T^T e_{k_s})_i) / Z`

This is the Multiplicative Weights (MW) algorithm. The Nesterov-accelerated version uses extrapolation.

The key advantage of this approach: no LP needed! Each iteration just requires:
1. Compute `T*a` (FFT): O(n log n)
2. Find argmax: O(n)
3. Update `a` using the new loss: O(n)

Total per iteration: O(n log n), which for n=960 is very fast (~microseconds).

The disadvantage: MW converges as O(1/sqrt(t)) in regret, while the LP-based Nesterov converges as O(1/t). So we need more iterations to get the same accuracy.

                    best_a = y.copy()
                    stall_count = 0

                y = y + noise
                y = np.maximum(y, 0)

            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, n_max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-np.inf, None)]
    b_ub = np.zeros(2 * n - 1)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        T_y = build_conv_matrix(y)
        A_ub = np.hstack([T_y, -np.ones((2 * n - 1, 1))])

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                continue
            b_new = np.maximum(res.x[:n], 0)
            s_b = b_new.sum()
            if s_b < 1e-15:
                continue
            b_new = b_new / s_b
        except Exception:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        denom = (it + 1.0) + alpha * (it + 2.0)
        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new) / denom
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        if it % 50 == 0:
            score_a = 2.0 * n * fast_convolve(a).max()
            if score_a < best_score - 1e-12:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            else:
                score_y = 2.0 * n * fast_convolve(y).max()
                if score_y < best_score - 1e-12:
                    best_score = score_y
                    best_a = y.copy()
                    stall_count = 0
                else:
                    stall_count += 1

            if stall_count >= 80:
                rng = np.random.default_rng(it * 17 + 3)
                noise_scale = 0.005 + 0.005 * (stall_count // 80)
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            score_now = 2.0 * n * fast_convolve(a).max()
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f} cur={score_now:.6f}")

    return best_a, best_score


def mw_nesterov(a0, time_limit, n_max_iters=50000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    regret = np.zeros(n)
    eta = 1.0 / np.sqrt(np.log(n))

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_curr = softmax_from_z(z - eta * regret)
        b = fast_convolve(a_curr)
        k_star = int(np.argmax(b))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = a_curr[k_star - idx]
        regret += grad

        alpha = 2.0 / (it + 2.0)
        z = z - eta * regret  # implicit in the softmax

        if it % 10000 == 0:
            score = 2.0 * n * b.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_curr.copy()

        if (it + 1) % 500000 == 0:
            elapsed = time.monotonic() - t0
            a_check = softmax_from_z(z - eta * regret)
            score_check = 2.0 * n * fast_convolve(a_check).max()
            if score_check < best_score:
                best_score = score_check
                best_a = a_check.copy()
            print(f"    MW iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_outer=50):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1, -0.03, -0.05]:
                amount = a[i] * frac
                if a[i] + amount < 0:
                    continue
                j = rng.integers(n)
                if j == i:
                    continue
                a_trial = a.copy()
                a_trial[i] += amount
                a_trial[j] -= amount
                if a_trial[j] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and outer > 5:
            break

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = (1 - x) ** 0.5
    else:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_dual(a0, time_limit, n_max_iters=200000):
    """Nesterov LP using the dual formulation for potentially faster solves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0

    # Dual: maximize lambda s.t. T_y^T u >= lambda*1, sum(u)=1, u>=0
    # Variables: u (2n-1), lambda (1) -> 2n variables
    # Constraints: T_y^T u - lambda >= 0 (n), sum(u) = 1 (1)
    # In min form: minimize -lambda s.t. -T_y^T u + lambda <= 0, sum(u)=1, u>=0, lambda free
    m = 2 * n - 1
    c_dual = np.zeros(m + 1)
    c_dual[-1] = -1.0  # minimize -lambda
    A_eq_dual = np.zeros((1, m + 1))
    A_eq_dual[0, :m] = 1.0
    b_eq_dual = np.array([1.0])
    bounds_dual = [(0.0, None)] * m + [(-np.inf, None)]

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Build T_y^T: shape (n, 2n-1)
        T_y = build_conv_matrix(y)  # (2n-1, n)
        T_yT = T_y.T  # (n, 2n-1)
        A_ub_dual = np.hstack([-T_yT, np.ones((n, 1))])
        b_ub_dual = np.zeros(n)

        try:
            res = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual, A_eq=A_eq_dual,
                          b_eq=b_eq_dual, bounds=bounds_dual, method='highs')
            if res.status != 0:
                continue
            u_vals = np.maximum(res.x[:m], 0)
            su = u_vals.sum()
            if su < 1e-15:
                continue
            u_vals = u_vals / su
            lambda_val = res.x[-1]
        except Exception:
            continue

        # Recover b from complementary slackness or just use the dual info
        # Actually, we need b for the Nesterov update. Let's get it from the primal.
        # The primal optimal b can be extracted from the dual solution via
        # complementary slackness, but it's easier to just solve the primal.
        # Alternative: use the dual solution to guide the search.
        # For now, let's use the argmin of T_y^T u as a direction.
        v = T_yT @ u_vals  # v_i = (T_y^T u)_i
        i_min = int(np.argmin(v))

        # Create a sparse b concentrated near i_min
        b_new = np.zeros(n)
        width = max(1, n // 20)
        start = max(0, i_min - width)
        end = min(n, i_min + width + 1)
        b_new[start:end] = 1.0
        b_new = b_new / b_new.sum()

        # Mix with the LP primal solution for better quality
        T_y_prim = build_conv_matrix(y)
        A_ub_prim = np.hstack([T_y_prim, -np.ones((2 * n - 1, 1))])
        c_prim = np.zeros(n + 1)
        c_prim[-1] = 1.0
        A_eq_prim = np.zeros((1, n + 1))
        A_eq_prim[0, :n] = 1.0
        b_eq_prim = np.array([1.0])
        bounds_prim = [(0.0, None)] * n + [(-np.inf, None)]
        b_ub_prim = np.zeros(2 * n - 1)
        try:
            res2 = linprog(c_prim, A_ub=A_ub_prim, b_ub=b_ub_prim, A_eq=A_eq_prim,
                           b_eq=b_eq_prim, bounds=bounds_prim, method='highs')
            if res2.status == 0:
                b_lp = np.maximum(res2.x[:n], 0)
                sl = b_lp.sum()
                if sl > 1e-15:
                    b_lp = b_lp / sl
                    b_new = 0.7 * b_lp + 0.3 * b_new
                    b_new = b_new / b_new.sum()
        except Exception:
            pass

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        denom = (it + 1.0) + alpha * (it + 2.0)
        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new) / denom
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        if it % 50 == 0:
            score_a = 2.0 * n * fast_convolve(a).max()
            if score_a < best_score - 1e-12:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            if stall_count >= 60:
                rng = np.random.default_rng(it * 13 + 5)
                y = y + rng.random(n) * 0.01
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP-dual iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_base = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: L-BFGS smooth warm-up
    print(f"\n=== Phase 1: L-BFGS Smooth (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_smooth = lbgfs_smooth(best_seq, t_alloc, T_start=80.0, T_end=0.5, n_stages=7)
        try_update(a_smooth, "lbgfs_smooth")

    # Phase 2: Nesterov LP (main optimization)
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, n_max_iters=500000)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 3: Second Nesterov LP round (tighter, from current best)
    print(f"\n=== Phase 3: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.2, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, n_max_iters=300000)
        try_update(a_lp2, "nesterov_lp_2")

    # Phase 4: MW with Nesterov (fast iterations)
    print(f"\n=== Phase 4: MW Nesterov (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.1, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, _ = mw_nesterov(best_seq, t_alloc, n_max_iters=50000000)
        try_update(a_mw, "mw_nesterov")

    # Phase 5: Coordinate descent
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 60.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0, 0.75, 1.333]:
        alt_n = int(n_base * mult)
        if 512 <= alt_n <= 2500 and alt_n != n_base:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        t_total = min(rem * 0.3, 200.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interp improved: {sc_alt:.6f}")

        t_lp = t_total * 0.7
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, n_max_iters=200000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_smooth = min(rem * 0.3, 40.0)
            a_alt_s = lbgfs_smooth(best_seq, t_smooth, T_start=20.0, T_end=0.3, n_stages=5)
            try_update(a_alt_s, f"smooth_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.3, 60.0)
        a_fin = lbgfs_smooth(best_seq, t_fin, T_start=15.0, T_end=0.1, n_stages=5)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_lp_fin = min(rem * 0.4, 60.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, n_max_iters=50000)
        try_update(a_lp_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.3, 30.0)
        a_cd_fin, _ = coordinate_descent(best_seq, t_cd, n_outer=30)
        try_update(a_cd_fin, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
