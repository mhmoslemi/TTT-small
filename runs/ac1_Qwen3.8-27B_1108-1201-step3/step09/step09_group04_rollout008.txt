                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall_count = 0

            if it % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    AS-ALT iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}, |active|={len(active_set)}")

    return best_a, best_score


def pnorm_gradient_z(z, p, n_coeffs):
    """Gradient of 2n*(sum c^p)^(1/p) w.r.t. log-domain z where a=softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = n_coeffs
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    # correlate(w, a, 'valid') gives length n
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_value_z(z, p, n_coeffs):
    """Value of 2n*(sum c^p)^(1/p) where a=softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = n_coeffs
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def pnorm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=6):
    """L-BFGS with p-norm smoothing in log domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, 3000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_val):
            return pnorm_value_z(z_val, stage_p, n)

        def jac(z_val):
            return pnorm_gradient_z(z_val, stage_p, n)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    PNorm p={stage_p:.0f} ({elapsed:.0f}s): score={sc:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def peak_shaving(a0, time_limit, n_steps=50000):
    """Targeted peak shaving: reduce the maximum convolution value directly."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peaks (exclude the very first and last which are a[0]^2 and a[-1]^2)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_idx = np.argmax(b_work)
        peak_val = b[peak_idx]

        # Find contributors to this peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_idx - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Strategy: reduce top contributors and redistribute to low-contribution positions
        top_contrib = np.argsort(contribs)[::-1][:8]
        low_contrib = np.argsort(contribs)[:8]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_idx - i
            if i == j or j >= n or j < 0:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            # Try reducing this pair
            for f1, f2 in [(0.97, 0.97), (0.95, 0.95), (0.93, 0.93), (0.97, 0.95)]:
                a_trial = a.copy()
                a_trial[i] *= f1
                a_trial[j] *= f2
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 0.01:
                    continue
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try single coordinate reduction
            for ci in top_contrib:
                i = idx[ci]
                if a[i] < 1e-14:
                    continue
                for factor in [0.96, 0.93, 0.90]:
                    diff = a[i] * (1 - factor)
                    a_trial = a.copy()
                    a_trial[i] -= diff
                    if a_trial[i] < 0:
                        a_trial[i] = 0
                        diff = a[i]
                    # Redistribute to low-contributing positions
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and step % 50 == 0:
            # Random perturbation
            rng = np.random.default_rng(step + 999)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.08, 0.08))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step % 100 != 0:
            # No improvement, keep trying with smaller steps
            pass

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def bimodal_init(n, alpha=0.5, rng=None):
    """Create bimodal initialization: mass concentrated at edges."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)
    # Two Gaussians at edges
    a = np.exp(-((x - 0.15) ** 2) / (2 * (alpha * 0.15) ** 2)) + \
        np.exp(-((x - 0.85) ** 2) / (2 * (alpha * 0.15) ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate():
    """
    Main search function combining active-set alternating LP, p-norm gradient
    polishing, peak shaving, and multi-length exploration.
    """
    t_start = time.monotonic()
    total_budget = 1800.0
    deadline = t_start + total_budget - 30.0  # 30s safety margin

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_score = float('inf')
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 960
        best_seq = bimodal_init(n_init)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Active-Set Alternating LP (main workhorse) ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"\n=== Phase 1: AS-Alt LP (budget={t_alloc:.0f}s) ===")
        a_lp, sc_lp = active_set_alternating(best_seq, t_alloc, solver, n_max_iters=200000)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            n_best = len(best_seq)
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 2: P-norm gradient polish ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"\n=== Phase 2: PNorm L-BFGS (budget={t_alloc:.0f}s) ===")
        a_pn = pnorm_lbfgs(best_seq, t_alloc, p_start=15.0, p_end=300.0, n_stages=5)
        sc_pn = compute_score(a_pn)
        if sc_pn < best_score:
            best_score = sc_pn
            best_seq = a_pn.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 3: Peak shaving ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"\n=== Phase 3: Peak Shaving (budget={t_alloc:.0f}s) ===")
        a_ps, sc_ps = peak_shaving(best_seq, t_alloc, n_steps=80000)
        if sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 4: Multi-length (budget={rem:.0f}s) ===")
    candidate_lengths = sorted(set([
        n_best - 64, n_best - 32, n_best + 32, n_best + 64,
        n_best - 128, n_best + 128, n_best + 192,
        768, 896, 1024, 1152, 1280, 1408
    ]))
    # Filter valid lengths
    candidate_lengths = [c for c in candidate_lengths if 400 <= c <= 2560 and c != n_best]
    # Sort by proximity to current best
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.20, 200.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n} (budget={t_total:.0f}s)")

        # Interpolate
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = compute_score(init_alt)

        # Quick p-norm optimization
        t_pn = t_total * 0.4
        a_alt = pnorm_lbfgs(init_alt, t_pn, p_start=20.0, p_end=200.0, n_stages=4)
        sc_alt = compute_score(a_alt)

        # Quick AS-Alt
        rem_sub = deadline - time.monotonic()
        if rem_sub > 30:
            t_as = min(rem_sub * 0.3, 80.0)
            a_alt2, sc_alt2 = active_set_alternating(a_alt, t_as, solver, n_max_iters=50000)
            if sc_alt2 < sc_alt:
                a_alt = a_alt2
                sc_alt = sc_alt2

        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()
            n_best = alt_n
            print(f"  NEW BEST at n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final alternating + polish ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 5: Final Polish (budget={rem:.0f}s) ===")
        # One more round of AS-Alt
        t_as = min(rem * 0.4, 150.0)
        if t_as > 30:
            a_final, sc_final = active_set_alternating(best_seq, t_as, solver, n_max_iters=100000)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  NEW BEST: {best_score:.6f}")

        # Peak shaving
        rem = deadline - time.monotonic()
        if rem > 20:
            t_ps = min(rem * 0.4, 80.0)
            a_ps2, sc_ps2 = peak_shaving(best_seq, t_ps, n_steps=50000)
            if sc_ps2 < best_score:
                best_score = sc_ps2
                best_seq = a_ps2.copy()
                print(f"  NEW BEST: {best_score:.6f}")

        # Final p-norm
        rem = deadline - time.monotonic()
        if rem > 20:
            t_pn = min(rem * 0.5, 60.0)
            a_fin = pnorm_lbfgs(best_seq, t_pn, p_start=100.0, p_end=500.0, n_stages=3)
            sc_fin = compute_score(a_fin)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  NEW BEST: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
        elapsed = time.monotonic() - t_start
        print(f"Total time: {elapsed:.0f}s")

    return [float(x) for x in best_seq]
