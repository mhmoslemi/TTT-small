def fast_convolve(a):

    a = np.maximum(a, 0)

def lse_smoothed_obj(a, T):
    """Log-sum-exp smoothed objective."""
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    return (2.0 * len(a) / T) * (m + np.log(np.exp(Tb - m).sum()))

def lse_smoothed_grad(a, T):
    """Gradient of log-sum-exp smoothed objective in 'a' space (not log-a space)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    exp_Tb = np.exp(Tb - m)
    p = exp_Tb / exp_Tb.sum()
    # grad_a[i] = 2 * sum_k p[k] * a[k-i]
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * p[k] * a[k - idx]
    return grad_a

def mirror_descent_step(a, subgrad, eta):

                    break

            continue

        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            continue
        b_val = b_val / s

        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:

            best_score = score_a

    return best_a, best_score

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:

    old_x = np.linspace(0, 1, old_n)

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def structured_init(n, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x)) * (1 + 0.2 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
