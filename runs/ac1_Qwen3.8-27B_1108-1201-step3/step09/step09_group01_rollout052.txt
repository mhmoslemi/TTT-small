                          -0.02, -0.05, -0.1, -0.15, -0.25, -0.4, -0.6, -0.8])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_min > 0:
                unaff_left = float(conv[:k_min].max())
            else:
                unaff_left = -1e30
            if k_max < 2 * n - 2:
                unaff_right = float(conv[k_max + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Affected convolution values
            affected = conv[k_min:k_max + 1]
            # Coefficients: for k in affected range, the linear coefficient is 2*a[k-i]
            ks = np.arange(k_min, k_max + 1)
            js = ks - i
            valid = (js >= 0) & (js < n)
            v = np.zeros(len(ks))
            v[valid] = 2.0 * a[js[valid]]

            has_quad = (2 * i >= k_min) and (2 * i <= k_max)
            quad_idx = 2 * i - k_min if has_quad else -1
            quad_base = affected[quad_idx] if has_quad else 0.0

            cur_score = M / (S * S)

            best_delta = 0.0
            best_new_score = cur_score

            deltas = step_fracs * a[i]

            for delta in deltas:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                if new_val > 1.0:
                    continue

                new_sum = S + delta
                if new_sum < 1e-10:
                    continue

                # Compute new max
                new_affected = affected + delta * v
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] = quad_base + new_val * new_val - a[i] * a[i]
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < 0:
                    new_max = 0

                new_score = new_max / (new_sum * new_sum)
                if new_score < best_new_score - 1e-16:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < cur_score - 1e-15:
                # Apply the change
                a[i] = max(a[i] + best_delta, 0.0)
                S += best_delta

                # Update convolution
                changes = best_delta * v
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = a[i] * a[i] - (a[i] - best_delta) * (a[i] - best_delta)
                conv[k_min:k_max + 1] += changes
                M = max(unaff_max, float(conv[k_min:k_max + 1].max()))

                score = 2.0 * n * M / (S * S)
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def coord_descent_fine(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    score = 2.0 * n * M / (S * S)
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    sweeps = 0

    fine_fracs = np.array([0.995, 0.99, 0.985, 0.98, 0.97,
                          1.005, 1.01, 1.015, 1.02, 1.03, 1.05])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            if k_min > 0:
                unaff_left = float(conv[:k_min].max())
            else:
                unaff_left = -1e30
            if k_max < 2 * n - 2:
                unaff_right = float(conv[k_max + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_min:k_max + 1]
            ks = np.arange(k_min, k_max + 1)
            js = ks - i
            valid = (js >= 0) & (js < n)
            v = np.zeros(len(ks))
            v[valid] = 2.0 * a[js[valid]]

            has_quad = (2 * i >= k_min) and (2 * i <= k_max)
            quad_idx = 2 * i - k_min if has_quad else -1
            quad_base = affected[quad_idx] if has_quad else 0.0

            cur_score = M / (S * S)
            best_delta = 0.0
            best_new_score = cur_score

            for frac in fine_fracs:
                delta = a[i] * (frac - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_sum = S + delta
                if new_sum < 1e-10:
                    continue

                new_affected = affected + delta * v
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] = quad_base + new_val * new_val - a[i] * a[i]
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < 0:
                    new_max = 0

                new_score = new_max / (new_sum * new_sum)
                if new_score < best_new_score - 1e-17:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < cur_score - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                S += best_delta

                changes = best_delta * v
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = a[i] * a[i] - (a[i] - best_delta) * (a[i] - best_delta)
                conv[k_min:k_max + 1] += changes
                M = max(unaff_max, float(conv[k_min:k_max + 1].max()))

                score = 2.0 * n * M / (S * S)
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    Fictitious play with Nesterov-style averaging.
    Alternates LP best responses and tracks the running average.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma_it = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma_it) * a_sequent + gamma_it * a
        b_sequent = (1.0 - gamma_it) * b_sequent + gamma_it * b

        candidates = [a, b, a_sequent, b_sequent, 0.5 * (a_sequent + b_sequent)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None, verbose=False):
    """Damped LP refinement with adaptive step."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 3.0 / (it + 1.0))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DLP it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 7))
    profile = np.ones(half) * 0.8
    for j in range(n_comp):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)

    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Initialize with two bumps."""
    gap = rng.uniform(0.15, 0.35)
    center1 = n * (0.5 - gap)
    center2 = n * (0.5 + gap)
    width = n * rng.uniform(0.08, 0.2)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width) ** 2) + np.exp(-0.5 * ((t - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_ridges(n, rng):
    """Flat base with periodic ridges."""
    period = int(n / rng.integers(3, 10))
    t = np.arange(n)
    a = np.ones(n) * 0.6
    ripple_amp = rng.uniform(0.15, 0.35)
    a += ripple_amp * np.cos(2 * np.pi * t / period)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structure_preserving_perturb(a0, rng, strength=0.05):
    """Perturb while preserving the overall shape."""
    n = len(a0)
    a = a0.copy()
    # Gaussian perturbation with decay at boundaries
    boundary_decay = np.linspace(0.5, 1.0, n // 4)
    decay = np.concatenate([boundary_decay, np.ones(n - 2 * (n // 4)), boundary_decay[::-1]])
    noise = rng.standard_normal(n) * decay * strength
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def sparse_restructure(a0, rng, frac_keep=0.7):
    """Keep a random subset of coordinates, zero the rest, add noise."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) < frac_keep
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.15)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + fast CD
    Phase 2: Intensive objective-aware CD
    Phase 3: LP fictitious play with Nesterov averaging
    Phase 4: CD polishing after LP
    Phase 5: Shake & re-optimize (global exploration)
    Phase 6: Final ultra-fine polishing
    """
    deadline = time.monotonic() + 1788.0
    rng = np.random.default_rng(77)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using default uniform: n={len(parent)}, score={best_score:.6f}")

    # === Phase 1: Diverse initialization + fast CD ===
    print(f"\n=== Phase 1: Diverse Init + CD (budget 80s) ===")
    t1_start = time.monotonic()
    t1_budget = 80.0

    n_values = [len(parent)]
    for alt in [768, 896, 1152, 1280]:
        if alt != len(parent):
            n_values.append(alt)
    n_values = n_values[:4]

    candidates = []
    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > t1_budget - 5:
            break

        inits = []
        inits.append(("parent", interpolate_seq(parent, n)))
        inits.append(("sym", init_symmetric(n, rng)))
        inits.append(("bimodal", init_bimodal(n, rng)))
        inits.append(("ridges", init_ridges(n, rng)))

        for name, init in inits:
            t_left = t1_budget - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(2.0, min(8.0, t_left / 2.0))
            a_opt, s_opt, _ = coord_descent_full(init, t_alloc, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Candidates: " + ", ".join([f"{c[0]:.6f}(n={c[2]},{c[3]})" for c in candidates[:4]]))

    for s_c, a_c, n_c, it_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Intensive objective-aware CD ===
    print(f"\n=== Phase 2: Intensive CD (budget 300s) ===")
    t2_budget = 300.0
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(4):
            rem = deadline - time.monotonic() - 800
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  CD round {rnd+1}/{4}, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")
            a_opt, s_opt, sweeps = coord_descent_full(current, t_alloc, verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_cd_r{rnd}")

            # Perturb for next round
            if rnd < 2:
                current = structure_preserving_perturb(best_seq, rng, strength=0.03 + 0.02 * rnd)
            else:
                current = best_seq.copy()

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: NFP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem * 0.5, 250.0)
            print(f"  NFP n={n_best}, budget={t_alloc:.0f}s")
            initial = best_seq.copy()
            a_fp, score_fp, iters = nesterov_fictitious_play(
                n_best, t_alloc, initial=initial, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "nfp_main")

            # Damped LP polish
            rem = deadline - time.monotonic() - 350
            if rem > 30:
                t_dl = min(rem * 0.4, 100.0)
                a_dl, _ = damped_lp_refine(a_fp, t_dl, solver=solver, verbose=True)
                best_seq, best_score = check_and_update(a_dl, best_seq, best_score, "dlp_main")

    # === Phase 4: CD polishing after LP ===
    print(f"\n=== Phase 4: CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  CD polish, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_opt, s_opt, _ = coord_descent_full(best_seq, t_alloc, verbose=True, label="p4")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "p4_cd")

    # === Phase 5: Shake & Re-optimize (global exploration) ===
    print(f"\n=== Phase 5: Shake & Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for shake_round in range(5):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break

            # Apply perturbation
            if shake_round % 2 == 0:
                perturbed = structure_preserving_perturb(best_seq, rng, strength=0.04 + 0.02 * shake_round)
            else:
                perturbed = sparse_restructure(best_seq, rng, frac_keep=0.75 - 0.05 * shake_round)

            t_alloc = min(rem * 0.25, 50.0)
            a_opt, s_opt, _ = coord_descent_full(perturbed, t_alloc, label=f"shake{shake_round}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"shake_{shake_round}")

    # === Phase 6: Try alternative n ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [n_best - 128, n_best + 128, n_best + 256]:
            if alt_n < 256 or alt_n > 2500:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 40.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = coord_descent_full(init_alt, t_alloc, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 7: Final ultra-fine polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Fine CD
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_fine = min(rem * 0.5, 100.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = coord_descent_fine(best_seq, t_fine, verbose=True, label="final1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_1")

        # Another fine pass
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_f2, s_f2, _ = coord_descent_fine(best_seq, min(rem * 0.6, 40.0), label="final2")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_2")

        # One more CD pass with standard steps
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_c3, s_c3, _ = coord_descent_full(best_seq, min(rem * 0.7, 25.0), label="final3")
            best_seq, best_score = check_and_update(a_c3, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
