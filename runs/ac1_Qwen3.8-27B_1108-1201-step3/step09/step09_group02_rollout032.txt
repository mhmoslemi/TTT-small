    # Define q = p (length 2n-1)
    # c = sum_m a[m] * q[m+j]
    # This is the "cross-correlation" of a and q, but with q indexed forward.
    # 
    # Using FFT: IFFT(FFT(a) * conj(FFT(q))) gives the circular cross-correlation
    # r[j] = sum_m a[m] * q[m-j] (mod size)
    # We want s[j] = sum_m a[m] * q[m+j] = r[-j] (in circular sense)
    # So s[j] = r[(size - j) % size]
    
    size = 1 << (3 * n - 1).bit_length()  # enough for linear
    A = rfft(a, size)
    Q = rfft(p, size)
    r = irfft(np.conj(A) * Q, size)  # r[j] = sum_m a[m] * q[m-j] (circular -> linear with padding)
    # We want s[j] = sum_m a[m] * q[m+j]
    # s[j] = r[-j mod size] = r[size-j]
    # For j = 0: s[0] = r[0] (since -0 mod size = 0)
    # For j > 0: s[j] = r[size-j]
    s = np.zeros(n)
    s[0] = r[0]
    if n > 1:
        s[1:] = r[size - np.arange(1, n)]
    return 2.0 * s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth mirror descent (exponentiated gradient) with adaptive temperature annealing on active lags
2. Fast coordinate descent with partial golden section for polishing
3. LP fictitious play with line search for global improvements

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.SCS, cp.ECOS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section 1D search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            unaff_max_f = unaff_max
            c_vals_f = c_vals
            a_partners_f = a_partners
            has_quad_f = has_quad
            quad_idx_f = quad_idx
            current_sum_f = current_sum
            v_f = v
            n_f = n

            def compute_score(x, _c=c_vals_f, _ap=a_partners_f, _qm=unaff_max_f,
                              _hq=has_quad_f, _qi=quad_idx_f, _cs=current_sum_f,
                              _v=v_f, _n=n_f):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

    """Fast coordinate descent with trial steps (no golden section)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    trial_factors = [0.9, 0.8, 1.2, 1.5, 0.7, 1.3]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            best_new_score = 2.0 * n * max(unaff_max, float(conv[k_lo:k_hi + 1].max()))

            for factor in trial_factors:
                x = v * factor
                if x < 0 or x > 10.0 / n:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                ns = current_sum + (x - v)
                if ns < 0.01:
                    continue
                new_max = max(unaff_max, float(nv.max()))
                sc = 2.0 * n * new_max / (ns * ns)
                if sc < best_new_score:
                    best_new_score = sc
                    best_delta = x - v

            if best_delta != 0.0:
                a[i] = v + best_delta
                current_sum += best_delta
                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps


def compute_smoothed_gradient(a, p):
    """
    Compute grad[j] = 2 * sum_k p[k] * a[k-j] for j=0..n-1.
    Uses a loop (O(n^2) but fast for n~1024-2048).
    """
    n = len(a)
    result = np.zeros(n)
    for j in range(n):
        result[j] = 2.0 * np.dot(a, p[j:j + n])
    return result


def mirror_descent_optimize(a0, time_limit, verbose=False, label=""):
    """
    Mirror descent (exponentiated gradient) with smooth max approximation
    and adaptive temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()

    # Temperature: starts low (broad focus), increases (sharper focus on peaks)
    T = 5.0 / max(conv.max(), 1e-10)
    T_max = 50.0 / max(conv.max(), 1e-10)
    lr = 0.05 * n
    no_improve = 0
    it = 0
    T_increase = 1.002

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax weights over lags
        Tconv = T * conv
        Tconv_shifted = Tconv - Tconv.max()
        exp_val = np.exp(Tconv_shifted)
        p = exp_val / exp_val.sum()

        # Compute gradient
        grad = compute_smoothed_gradient(a, p)

        # Mirror descent update: a[i] *= exp(-lr * grad[i])
        log_update = -lr * grad
        # Shift for numerical stability
        log_update_shifted = log_update - log_update.max()
        weights = a * np.exp(log_update_shifted)
        Z = weights.sum()
        if Z < 1e-30:
            break
        a_new = weights / Z

        # Compute score
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
        else:
            no_improve += 1
            if no_improve >= 50:
                # Increase temperature (focus more on peaks)
                T = min(T * T_increase, T_max)
                # Slightly reduce learning rate
                lr *= 0.99
                no_improve = 0
                # Try a small random perturbation to escape
                if no_improve > 100:
                    a = a + np.random.default_rng(it).standard_normal(n) * 0.005
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    no_improve = 0

        # Periodically do a few fast CD sweeps
        if it % 500 == 0 and time.monotonic() - t0 < time_limit - 10:
            t_cd_start = time.monotonic()
            a_cd, s_cd, _ = fast_coord_descent(a, min(5.0, time_limit - (time.monotonic() - t0) - 5), label=label + "_cd")
            if s_cd < best_score:
                best_score = s_cd
                best_a = a_cd.copy()
            a = a_cd
            conv = fast_convolve(a)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} T={T:.1f} lr={lr:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, it


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fictitious_play_with_linesearch(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with line search along the best response direction."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                continue
            a_new = a_new / cs
            score_new = 2.0 * n * float(fast_convolve(a_new).max())
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                continue
            a = a / cs

        for cand in [a, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters


def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=10, verbose=False):
    """Basin hopping: perturb the sequence and re-optimize."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 60.0)

        sigma = 0.015 + 0.03 * rng.random()
        perturbed = a + rng.standard_normal(n) * sigma
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Peak-breaking perturbation
        conv_p = fast_convolve(a)
        top_k = min(10, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx) // 10)
            picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
            for ii in picks:
                perturbed[ii] *= (1.0 + rng.standard_normal() * 0.03)
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
