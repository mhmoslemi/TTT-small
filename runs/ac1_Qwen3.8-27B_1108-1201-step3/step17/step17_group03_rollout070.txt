    margin_frac = 5e-3

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        c = _fast_conv(y)
        M_curr = c.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - c
        threshold = margin_frac * M_curr
        active_idx = np.where(margins <= threshold)[0]

        if len(active_idx) < 3:
            active_idx = np.argsort(margins)[:50]
        elif len(active_idx) > 300:
            active_idx = np.argsort(margins)[:300]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_s = max(0, int(k_int) - n + 1)
            i_e = min(n - 1, int(k_int))
            for i in range(i_s, i_e + 1):
                T_sel[row, i] = y[int(k_int) - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        my = _fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.85, 1e-5)

        if verbose and t % 20 == 0:
            print(f"    LP it={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Phase 0: Grid search over (n, p) for initial candidates
    print("\n=== Phase 0: Grid search over (n, p) ===")
    candidate_list = []
    n_values = [2000, 3000, 4000, 5000, 6000, 8000]
    p_values = [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 6.0, 8.0, 10.0]

    for n in n_values:
        for p in p_values:
            a = _power_law(n, p)
            c = _fast_conv(a)
            M = c.max()
            J = 2 * n * M
            candidate_list.append((J, n, p, a.copy()))

    candidate_list.sort(key=lambda x: x[0])
    print(f"  Best from grid: J={candidate_list[0][0]:.6f} (n={candidate_list[0][1]}, p={candidate_list[0][2]:.1f})")
    for i in range(min(10, len(candidate_list))):
        print(f"    #{i}: J={candidate_list[i][0]:.6f} (n={candidate_list[i][1]}, p={candidate_list[i][2]:.1f})")

    best_seq = candidate_list[0][3].copy()
    best_score = candidate_list[0][0]
    n_best = len(best_seq)

    # Also check if parent is good
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 1e-10:
            p_score = _eval(parent)
            if np.isfinite(p_score) and p_score < best_score:
                best_seq = _norm(parent)
                best_score = p_score
                n_best = len(best_seq)
                print(f"  Parent is better: J={best_score:.6f} (n={n_best})")
    except Exception:
        pass

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = _norm(a)
        sc = _eval(a)
        if np.isfinite(sc) and sc < best_score:
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")
            best_seq = a.copy()
            best_score = sc

    # Phase 1: Smooth max descent on best candidate
    print(f"\n=== Phase 1: Smooth Max Descent (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_out, _ = _smooth_max_descent(best_seq, t1, beta=30.0, top_k=40, verbose=True)
        try_update(a_out, "smd_main")

    # Phase 2: Peak shaving
    print(f"\n=== Phase 2: Peak Shaving (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.2, 350.0)
    if t2 > 40:
        a_out, _ = _peak_shave(best_seq, t2, verbose=True)
        try_update(a_out, "peak_shave1")

    # Phase 3: Second SMD round with different beta
    print(f"\n=== Phase 3: SMD round 2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 40:
        a_out, _ = _smooth_max_descent(best_seq, t3, beta=15.0, top_k=60, verbose=True)
        try_update(a_out, "smd2")

    # Phase 4: LP Frank-Wolfe
    print(f"\n=== Phase 4: LP Frank-Wolfe (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_cur <= 6000:
        a_out, _ = _lp_frank_wolfe(best_seq, t4, verbose=True)
        try_update(a_out, "lp_fw")

    # Phase 5: Try second-best initial candidates
    print(f"\n=== Phase 5: Alternative starts ===")
    remaining = time_budget - (time.monotonic() - t_global)
    for i in range(1, min(4, len(candidate_list))):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        J_init, n_init, p_init, a_init = candidate_list[i]
        print(f"  Trying n={n_init}, p={p_init:.1f} (init J={J_init:.6f})")
        t5 = min(remaining * 0.15, 150.0)
        if t5 > 30:
            a_out, _ = _smooth_max_descent(a_init, t5, beta=25.0, top_k=40, verbose=False)
            try_update(a_out, f"alt_n{n_init}_p{p_init:.0f}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60:
                a_out2, _ = _peak_shave(a_out, min(remaining * 0.5, 100.0), verbose=False)
                try_update(a_out2, f"alt_ps_n{n_init}_p{p_init:.0f}")

    # Phase 6: Final peak shaving on absolute best
    print(f"\n=== Phase 6: Final peak shave (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t6 = min(remaining * 0.4, 200.0)
        if t6 > 20:
            a_out, _ = _peak_shave(best_seq, t6, verbose=True)
            try_update(a_out, "final_ps")

    # Phase 7: Final SMD polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final SMD polish (n={len(best_seq)}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 20:
            a_out, _ = _smooth_max_descent(best_seq, t7, beta=50.0, top_k=20, verbose=False)
            try_update(a_out, "final_smd")

    # Final validation
    if best_seq is not None:
        fs = _eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        best_seq = _power_law(3600, 1.0)
        best_score = _eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
