def fast_autocorr(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # current iterates
    x = a.copy()  # Nesterov extrapolation
    
    gamma = 3.0  # step size parameter
    
    for t in range(1, n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive margin: start loose, tighten over time
        progress = min(t / (0.7 * n_iters_max), 1.0)
        margin_frac = margin_start * (margin_end / margin_start) ** progress
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select active constraints
        sorted_idx = np.argsort(ac)[::-1]
        
        # Strategy 1: Top K by value
        K1 = min(max(50, int(top_K_max * 0.3)), len(ac))
        
        # Strategy 2: Based on margin
        n_active = np.sum(ac >= M_curr - margin_frac * M_curr)
        K2 = min(max(n_active, 30), top_K_max)
        
        K = max(K1, min(K2, top_K_max))
        active_idx = sorted_idx[:K]
        
        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                idx = k_int - j
                T_sel[row, j] = 2.0 * y[idx]  # Linearized: conv ≈ grad * (b - y) + curr
        
        # Actually, for LP-FW, we want:
        # conv(a,a)[k] ≈ conv(y,y)[k] + sum_j grad[k][j] * (b_j - y_j)
        # = ac[k] + sum_j 2*y[k-j]*(b_j - y_j)
        # We want this ≤ M_curr
        # sum_j 2*y[k-j]*b_j - sum_j 2*y[k-j]*y_j ≤ M_curr - ac[k]
        # sum_j grad[k][j] * b_j ≤ M_curr - ac[k] + 2*ac[k]
        # sum_j grad[k][j] * b_j ≤ M_curr + ac[k]
        
        # Wait, let me reconsider. The standard LP-FW for minimizing max_k f_k(a) where f_k is convex:
        # At point y, the linearization of f_k is: f_k(y) + ∇f_k(y)·(b-y)
        # We want: f_k(y) + ∇f_k(y)·(b-y) ≤ M for all k in active set
        # i.e., ∇f_k(y)·b ≤ M - f_k(y) + ∇f_k(y)·y
        # Since f_k is the k-th convolution (quadratic), its gradient at y is:
        # ∇f_k(y)[j] = 2*y[k-j] (as computed above)
        # And f_k(y) = ac[k]
        # ∇f_k(y)·y = sum_j 2*y[k-j]*y[j] = 2*ac[k] (since ac[k] = sum_j y[j]*y[k-j])
        
        # So the constraint becomes:
        # sum_j 2*y[k-j]*b_j ≤ M - ac[k] + 2*ac[k] = M + ac[k]
        
        # This makes sense: we're asking for b to be a direction that, when we move from y,
        # keeps the linearized convolution values bounded.
        
        # Actually, the standard FW is:
        # minimize t
        # subject to: ∇f_k(y)·b ≤ t - f_k(y) + ∇f_k(y)·y  for active k
        #              b ≥ 0, sum(b) = 1
        # Then move: y_new = (1-α)y + α*b
        
        # But the previous code uses a simpler formulation:
        # maximize sum(b) subject to ∇f_k(y)·b ≤ M_curr for active k, b ≥ 0
        # Then normalize b to sum 1.
        
        # Let me use the simpler formulation (maximize sum(b) with constraints)
        # since it's faster and worked well before.
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # FW step with Nesterov
        alpha = 2.0 / (t + 2.0)
        
        # Try quadratic line search on J((1-α)y + α*g)
        # J(α) = 2n * max_k conv((1-α)y + αg, (1-α)y + αg)
        # Each conv[k](α) = conv((1-α)y + αg, (1-α)y + αg)[k]
        # = (1-α)²*conv(y,y)[k] + 2α(1-α)*conv(y,g)[k] + α²*conv(g,g)[k]
        # This is quadratic in α!
        
        # For the top few constraints, compute the quadratic coefficients
        y_g_conv = np.convolve(y, g_dir, mode='full')  # conv(y,g)
        g_g_conv = np.convolve(g_dir, g_dir, mode='full')  # conv(g,g)
        
        # For each active constraint k, the value at parameter α is:
        # f_k(α) = (1-α)²*ac[k] + 2α(1-α)*yg[k] + α²*gg[k]
        # = ac[k] - 2α*ac[k] + α²*ac[k] + 2α*yg[k] - 2α²*yg[k] + α²*gg[k]
        # = ac[k] + α*(2*yg[k] - 2*ac[k]) + α²*(ac[k] - 2*yg[k] + gg[k])
        
        # We want max_k f_k(α) to be minimized.
        # This is a min-max of quadratics problem.
        
        # Simple approach: try a few α values
        best_alpha = alpha
        best_trial_m = M_curr
        
        # Candidate alphas
        candidate_alphas = [alpha]
        # Add some around the FW step
        for frac in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            candidate_alphas.append(alpha * frac)
        # Also try smaller steps
        for frac in [0.3, 0.1, 0.05]:
            candidate_alphas.append(alpha * frac)
        
        for ca in candidate_alphas:
            if ca < 0.001 or ca > 2.0:
                continue
            trial = (1.0 - ca) * y + ca * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            tm = fast_autocorr(trial).max()
            if tm < best_trial_m:
                best_trial_m = tm
                best_alpha = ca
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov update
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Update best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPv2 iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")
    
    return best_a, best_m


def refined_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Refined coordinate descent that:
    1. Focuses on the peak lag
    2. Uses analytical 2-move optimization
    3. Tries multi-move (block) adjustments
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        top_peaks = np.argsort(ac)[::-1][:10]
        
        improved = False
        for peak_k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(peak_k)
            
            # Get contributors to this peak
            j_start = max(0, peak_k - n + 1)
            j_end = min(n, peak_k + 1)
            
            # Contribution of pair (i, k-i) to conv[k] is a[i]*a[k-i]
            # Total = sum of all a[i]*a[k-i] for valid i
            # Gradient wrt a[j] is 2*a[k-j]
            
            # Strategy: find a[j] with large a[k-j] (high contributor)
            # and move mass from j to a random/different position
            
            # Find top contributors
            contribs = {}
            for j in range(j_start, j_end):
                idx = peak_k - j
                if 0 <= idx < n:
                    c = a[j] * a[idx]
                    contribs[j] = c
            
            if not contribs:
                continue
            
            # Sort by contribution
            sorted_contribs = sorted(contribs.items(), key=lambda x: -x[1])
            n_top = min(20, len(sorted_contribs))
            
            for src_j, src_contrib in sorted_contribs[:n_top]:
                src_j = int(src_j)
                src_val = a[src_j]
                if src_val < 1e-12:
                    continue
                
                partner_idx = peak_k - src_j
                if partner_idx < 0 or partner_idx >= n:
                    continue
                
                # Try moving from src_j to various destinations
                # Optimal: minimize the increase in max_conv
                # The decrease in conv[peak_k] from removing amount δ from a[src_j]:
                # Δconv[peak_k] = -2*δ*a[partner_idx] (approximately, for small δ)
                # But this might increase other convolutions.
                
                best_dst = -1
                best_new_m = M_curr
                best_delta = 0
                
                # Try a few fraction amounts
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    delta = src_val * frac
                    if delta < 1e-14:
                        continue
                    
                    # Try several destinations
                    for _ in range(40):
                        dst = int(rng.integers(n))
                        if dst == src_j:
                            continue
                        
                        a_trial = a.copy()
                        a_trial[src_j] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        
                        # Quick check: compute only the peak and a few neighbors
                        # For speed, compute full autocorr
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m - 1e-15:
                            best_new_m = M_trial
                            best_dst = dst
                            best_delta = delta
                
                if best_dst >= 0 and best_new_m < M_curr - 1e-15:
                    a[src_j] -= best_delta
                    a[best_dst] += best_delta
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-15:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break  # Try next peak
            if improved:
                continue
        
        if not improved:
            stall += 1
            # Random restart perturbation
            if stall % 50 == 0:
                # Small perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def amd_optimize_v2(a0, time_limit, top_k=60, beta=4.0, c0=3.5,
                    restart_every=5000, verbose=False, use_backtrack=True):
    """Nesterov mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    
    def compute_subgradient(y, ac, top_k_val, beta_val):
        k_max = min(top_k_val, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_val
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                idx = k_int - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * wi * y[idx]
        return grad
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, cur_top_k, cur_beta)
        
        if use_backtrack:
            eta = gamma / (np.sqrt(t) + 1.0)
            base_m = fast_autocorr(y).max()
            accepted = False
            for _ in range(6):
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new / s_new
                m_trial = fast_autocorr(a_new).max()
                if m_trial < base_m - 1e-18:
                    x_prev = x
                    x = a_new
                    accepted = True
                    break
                else:
                    eta *= 0.5
            if not accepted:
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = y.copy()
                else:
                    a_new = a_new / s_new
                x_prev = x
                x = a_new
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)
    
    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    """Perturb in frequency domain for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'tanh':
        x = np.linspace(-3, 3, n)
        return norm_sum1(1.0 / (1.0 + np.exp(-2*x)))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses improved LP-FW, refined coordinate descent,
    AMD, and multi-length exploration to minimize J = 2n*max(conv(a,a))/sum(a)^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Quick AMD to get a good starting point ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD warmup (n={n0}) ===")
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize_v2(a_in, t1, top_k=60, beta=4.0, c0=3.5,
                                    restart_every=3000, verbose=False, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_warm")
    
    # === Phase 2: Main LP-FW with adaptive margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True,
                                      top_K_max=400, margin_start=2e-3, margin_end=5e-5,
                                      n_iters_max=100000)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")
    
    # === Phase 3: Refined coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coordinate descent (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = refined_coordinate_descent(a_in, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd1")
    
    # === Phase 4: LP-FW with tighter margins (refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True,
                                      top_K_max=500, margin_start=1e-3, margin_end=1e-5,
                                      n_iters_max=100000)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")
    
    # === Phase 5: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    length_tries = sorted(set([n_best - 200, n_best + 200, n_best + 400]))
    length_tries = [l for l in length_tries if 1000 <= l <= 5500]
    
    for new_n in length_tries[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"\n=== Phase 5: Extend to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_max=500,
                       margin_start=2e-3, margin_end=5e-5,
                       n_iters_max=50000, per_iter_timeout=10.0):

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    gamma = 3.5

    for t in range(1, n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(t / max(0.5 * n_iters_max, 100), 1.0)
        margin_frac = margin_start * (margin_end / margin_start) ** progress

        ac = fast_autocorr(y)
        M_curr = ac.max()

        sorted_idx = np.argsort(ac)[::-1]

        # Adaptive K: start moderate, increase over time
        K = min(int(50 + top_K_max * progress), top_K_max, len(ac))

        # Also consider margin-based selection
        n_active = np.sum(ac >= M_curr - margin_frac * M_curr)
        K = max(K, min(int(n_active), top_K_max))
        K = max(K, 30)

        active_idx = sorted_idx[:K]

        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)

            prob.solve(solver=solver, verbose=False)

            continue

            continue

        best_trial_m = M_curr

                continue

            if tm < best_trial_m:
                best_trial_m = tm
                best_alpha = ca

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            x_new = y_new + beta_t * (y_new - y)

            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()

def refined_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):

        improved = False
        for peak_k in top_peaks:
            if time.monotonic() - t0 > time_limit:

            j_end = min(n, peak_k + 1)

                idx = peak_k - j
                if 0 <= idx < n:
                    c = a[j] * a[idx]
                    if c > 1e-15:
                        contribs.append((j, c))
            contribs.sort(key=lambda x: -x[1])

            n_top = min(15, len(contribs))
            for src_j, _ in contribs[:n_top]:
                src_j = int(src_j)
                src_val = a[src_j]
                if src_val < 1e-12:
                    continue

                best_dst = -1
                best_new_m = M_curr
                best_delta = 0.0

                for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                    delta = src_val * frac
                    if delta < 1e-14:
                        continue

                    for _ in range(30):
                        dst = int(rng.integers(n))
                        if dst == src_j:
                            continue
                        a_trial = a.copy()
                        a_trial[src_j] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:

                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m - 1e-15:

                if best_dst >= 0 and best_new_m < M_curr - 1e-12:
                    a[src_j] -= best_delta
                    a[best_dst] += best_delta

                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-12:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:

                i = int(rng.integers(n))

                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 80 == 0:

            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: AMD warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 1: AMD warmup (n={n0}) ===")
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize_v2(a_in, t1, top_k=50, beta=4.0, c0=3.0,
                                    restart_every=3000, verbose=False, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_warm")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True,
                                      top_K_max=350, margin_start=2e-3, margin_end=1e-4,
                                      n_iters_max=80000, per_iter_timeout=8.0)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: CD (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = refined_coordinate_descent(a_in, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd1")

    # === Phase 4: LP-FW refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW refine (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True,
                                      top_K_max=450, margin_start=1e-3, margin_end=5e-5,
                                      n_iters_max=80000, per_iter_timeout=8.0)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_refine")

    # === Phase 5: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CD2 (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = refined_coordinate_descent(a_in, t5, verbose=True, rng_seed=99)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd2")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    length_tries = sorted(set([n_best - 200, n_best + 200, n_best + 500]))
    length_tries = [l for l in length_tries if 1500 <= l <= 5500]

    for new_n in length_tries[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.30, 180.0)
        if t6a > 25:
            a_out, _ = amd_optimize_v2(a_long, t6a, top_k=50, beta=4.0, c0=3.0,
                                        restart_every=3000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4500:
            t6b = min(remaining * 0.30, 120.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe_v2(a_in, t6b, solver, verbose=False,
                                              top_K_max=300, margin_start=2e-3, margin_end=1e-4,
                                              n_iters_max=40000, per_iter_timeout=5.0)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 7: Freq diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Freq-div (n={n_best}) ===")
    rng = np.random.default_rng(200)
    if remaining > 80:
        t7_per = min(remaining * 0.08, 80.0)
        if t7_per > 20:
            for trial in range(3):
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining < t7_per + 20:
                    break
                a_pert = freq_perturb(best_seq, strength=0.008, rng=rng)
                a_out, _ = amd_optimize_v2(a_pert, t7_per, top_k=50, beta=5.0, c0=4.0,
                                            restart_every=2000, verbose=False, use_backtrack=True)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 8: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.20, 180.0)
    if t8 > 25 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t8, solver, verbose=True,
                                      top_K_max=500, margin_start=5e-4, margin_end=3e-5,
                                      n_iters_max=60000, per_iter_timeout=5.0)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final CD (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.35, 120.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = refined_coordinate_descent(a_in, t9, verbose=True, rng_seed=314)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
