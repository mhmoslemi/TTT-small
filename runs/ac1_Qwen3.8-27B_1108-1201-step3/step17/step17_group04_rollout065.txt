def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def get_subgradient_adaptive(a, ac, top_k=None, beta=3.0):
    n = len(a)
    grad = np.zeros(n)
    M = ac.max()
    if M < 1e-30:
        return grad
    if top_k is None:
        peak_idx = np.where(ac >= 0.8 * M)[0]
    else:
        peak_idx = np.argsort(ac)[::-1][:top_k]
    if len(peak_idx) == 0:
        peak_idx = np.array([int(np.argmax(ac))])
    peak_vals = np.maximum(ac[peak_idx], 1e-30)
    w = peak_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(peak_idx)) / len(peak_idx)
    else:
        w = w / w_sum
    for wi, k in zip(w, peak_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def spectral_shape_init(n, method='highpass', beta_boost=0.3, seed=42):
    rng = np.random.default_rng(seed)
    L = 1 << (2 * n - 1 - 1).bit_length()
    if method == 'highpass':
        a = np.ones(n) / n
        A = np.fft.rfft(a, L)
        half_len = A.shape[0]
        freqs = np.arange(half_len)
        max_f = max(1, half_len - 1)
        boost = (freqs / max_f) ** beta_boost + 0.1
        A_new = (np.abs(A) * boost) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, L)[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    elif method == 'random_phase':
        best_a = np.ones(n) / n
        best_J = compute_J(best_a)
        half_len = L // 2 + 1
        for trial in range(60):
            h_phases = rng.uniform(0, 2 * np.pi, half_len)
            mag = np.ones(half_len)
            A = mag * np.exp(1j * h_phases)
            a_new = np.fft.irfft(A, L)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            J = compute_J(a_new)
            if J < best_J:
                best_J = J
                best_a = a_new.copy()
        return best_a
    elif method == 'tukey_window':
        r = 0.3
        w = np.ones(n)
        edge = int(r * n / 2)
        if edge > 0:
            ar = np.arange(edge) / edge - 1
            ramp = 0.5 * (1 + np.cos(np.pi * ar))
            w[:edge] = ramp
            w[-edge:] = ramp
        w = np.maximum(w, 0)
        s = w.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return w / s
    return np.ones(n) / n

        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, L)

            a_new = a_new / s

            best_J = J_new

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = T_full[binding_idx, :]
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=%d (%.0fs): J=%.6f" % (t, elapsed, 2.0 * n * best_m))
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)

        a_out, _ = peak_reduction_v2(norm_sum1(best_seq), t3, verbose=True, seed=77)
        _try_update(a_out, "peakred2_n%d" % len(best_seq))

    # Phase 4: subgradient round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: SGD round 2 (n=%d) ===" % n_best)
    t4 = min(remaining * 0.16, 280.0)
    if t4 > 20:
        a_out, _ = nesterov_subgradient_v2(norm_sum1(best_seq), t4, verbose=True, seed=555)
        _try_update(a_out, "sgd2b_n%d" % n_best)

    # Phase 5: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW (n=%d) ===" % n_best)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20 and n_best <= 4000:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver, verbose=True, top_K=300)
        _try_update(a_out, "lpfw_n%d" % n_best)

    # Phase 6: peak reduction round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Peak reduction round 2 (n=%d) ===" % n_best)
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 20:
        a_out, _ = peak_reduction_v2(norm_sum1(best_seq), t6, verbose=True, seed=222)
        _try_update(a_out, "peakred2b_n%d" % n_best)

    # Phase 7: length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Length exploration ===")
    lengths_to_try = []
    for delta in [400, -400, 800, -800]:
        l = n_best + delta
        if 1000 <= l <= 6000:
            lengths_to_try.append(l)
    for new_n in lengths_to_try[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  Trying n=%d" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.2, 150.0)
        if t7a > 20:
            a_out, _ = nesterov_subgradient_v2(a_long, t7a, verbose=False, seed=777)
            _try_update(a_out, "sgd_n%d" % new_n)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = peak_reduction_v2(norm_sum1(best_seq), min(remaining * 0.15, 60.0),
                                             verbose=False, seed=333)
                _try_update(a_out, "peakred_n%d" % len(best_seq))

    # Phase 8: final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final polish (n=%d) ===" % n_best)
    t8 = min(remaining * 0.5, 200.0)
    if t8 > 15:
        a_out, _ = nesterov_subgradient_v2(norm_sum1(best_seq), t8, verbose=True, seed=4321)
        _try_update(a_out, "sgd_final_n%d" % n_best)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, _ = peak_reduction_v2(norm_sum1(best_seq), min(remaining * 0.5, 100.0),
                                     verbose=True, seed=1234)
        _try_update(a_out, "peakred_final_n%d" % n_best)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (best_score, len(best_seq),
                                                         time.monotonic() - t_global))
    else:
        print("\nERROR: no valid sequence found, using fallback")
        best_seq = make_symmetric_init(3200)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=%.6f" % best_score)

    return [float(x) for x in best_seq]
