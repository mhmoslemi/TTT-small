def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_autocorr(a, ac):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * ac.max() / (s * s))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_search(n, time_limit, best_init, best_score_init):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = norm_sum1(best_init)
    best_val = best_score_init
    i_arr = np.arange(n, dtype=np.float64)

    def try_update(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
            elapsed = time.monotonic() - t0
            print(f"    Parametric [{label}]: J={val:.6f} (t={elapsed:.0f}s)")

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        try_update((i_arr + 1.0) ** p, f"pow_{p}")
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        try_update((n - i_arr) ** p, f"ipow_{p}")

    # Beta-like
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            try_update(a, f"beta_{alpha}_{beta}")

    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_update(a, f"gc{c:0.1f}s{sigma:0.1f}")

    # Bimodal
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.05, 0.08, 0.12, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                try_update(a, f"bi_{sigma:0.2f}")

    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            try_update(1.0 / (1.0 + np.exp(-s * (x - c))), f"sg{c:0.1f}_{s}")

    # Sinusoidal
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            try_update(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01), f"sn{freq}_{ph:0.2f}")

    # Exponential
    for rate in [1, 2, 3, 5, 10]:
        try_update(np.exp(-rate * x), f"er{rate}")
        try_update(np.exp(rate * x), f"enr{rate}")

    # Cosh
    for c in [0.0, 0.1, 0.2]:
        for s in [2, 5, 10]:
            try_update(np.cosh(s * (x - c)), f"ch{c:0.1f}_{s}")

    if time.monotonic() - t0 < time_limit * 0.8:
        # Blend with best
        for w in [0.2, 0.4, 0.6]:

            a_lin = (i_arr + 1.0) / (i_arr + 1.0).sum()
            a_blend = w * a_base + (1.0 - w) * a_lin
            try_update(a_blend, f"bl{w:0.1f}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_val:.6f}")

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    # Initialize constraint set with top lags
    init_K = min(initial_K, 2 * n - 1)
    init_idx = np.argsort(ac_y)[::-1][:init_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0
    failed_solves = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x.copy()

        # Manage constraint set size
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep constraints that are most "binding" (smallest margin)
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        # Solve LP: minimize M s.t. A*g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            failed_solves += 1
            if failed_solves > 5:
                # Reduce constraints and retry
                constraint_list = sorted(constraint_set)[:min(initial_K, len(constraint_set))]
                constraint_set = set(constraint_list)
                failed_solves = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def compute_soft_grad(y, ac, top_k, beta):
    n = len(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * y[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k, beta, c0, restart_every, verbose):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_grad(y, ac, cur_top_k, cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Structured perturbation: mix with a parametric family
            pert_type = int(rng.integers(3))
            if pert_type == 0:
                i_arr = np.arange(n, dtype=np.float64)
                p = float(rng.uniform(0.5, 2.0))
                pert_seq = norm_sum1((i_arr + 1.0) ** p)
            elif pert_type == 1:
                x_arr = np.linspace(0, 1, n)
                c = float(rng.uniform(0.3, 0.7))
                sig = float(rng.uniform(0.1, 0.4))
                pert_seq = norm_sum1(np.exp(-((x_arr - c) ** 2) / (2 * sig * sig)))
            else:
                pert = rng.normal(0, 0.03, n) * best_a
                pert_seq = norm_sum1(best_a + pert)

            mix_w = float(rng.uniform(0.1, 0.4))
            x_r = norm_sum1((1 - mix_w) * best_a + mix_w * pert_seq)
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 30, 150)
            cur_beta = max(cur_beta - 0.3, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")

    return best_a, best_m


def batch_coordinate_descent(a0, time_limit, rng_seed, verbose=True):
    """Aggressive batch coordinate descent with lookahead."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        improved = False
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        # For each top lag, try to reduce it
        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            # Find top contributing pairs
            n_pairs = min(8, len(idx))
            pair_indices = np.argsort(contrib)[::-1][:n_pairs]

            for pi in pair_indices:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[pi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score all positions for receiving mass
                # A position j is good if it doesn't contribute much to any top lag
                scores = np.zeros(n)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2] * ac[k2i]
                        scores[k2i - idx2] += a[idx2] * ac[k2i]

                # Positions that are in the peak region get high scores (bad to move to)
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                # Among non-peak positions, find the one with lowest score
                candidate_mask = ~in_peak
                if candidate_mask.sum() < 10:
                    candidate_mask = np.ones(n, dtype=bool)
                candidate_mask[src] = False

                cand_scores = scores.copy()
                cand_scores[~candidate_mask] = np.inf
                dst = np.argmin(cand_scores)

                if dst == src or a[dst] > 0.5 * src_val:
                    continue

                # Try multiple transfer fractions
                found = False
                for frac in [0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

                if not found:
                    break  # Try next pair

        if not improved:
            stall += 1
            if stall > max_stall:
                break

            # Random perturbation when stuck
            if stall % 10 == 0:
                n_moves = int(rng.integers(3, 15))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = float(rng.uniform(0.005, 0.03))
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    BatchCoord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def spectral_flattening(a0, time_limit, verbose=True):
    """Try to flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    fft_size = 1 << (2 * n - 2).bit_length()
    learning_rate = 0.5
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Target: flat power spectrum
        P_mean = P.mean()
        # Gradient of (max deviation from flat) in time domain
        # Use soft-max of power spectrum deviations
        P_dev = P - P_mean
        beta_spec = 5.0 + t * 0.001
        w_spec = np.exp(beta_spec * np.abs(P_dev) / P_mean)
        w_spec = w_spec / w_spec.sum()

        # Gradient in frequency domain, project to time
        grad_freq = 2.0 * A * w_spec / max(P.max(), 1e-30)
        grad_time = np.fft.irfft(grad_freq, fft_size)[:n]

        # Mirror descent step
        lr = learning_rate / (1.0 + np.sqrt(t))
        a_new = a * np.exp(-lr * grad_time)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                # Restart with perturbation
                a = norm_sum1(best_a + rng.normal(0, 0.01, n) * best_a)
                stall = 0
                learning_rate *= 0.7

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multiple phases."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent
                best_score = p_score
                parent_loaded = True
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        i_arr = np.arange(n_base, dtype=np.float64)
        # Start with a power-law shape
        best_seq = norm_sum1((i_arr + 1.0) ** 0.7)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search for better initialization ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 80.0)
    if t1 > 10:
        a_param, val_param = parametric_search(n0, t1, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        check_and_update(a_param, "parametric")

    # === Phase 2: LP-FW main optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW cutting plane (n={n_best}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, initial_K=400, max_K=3000,
                                   add_per_iter=50, verbose=True)
        check_and_update(a_out, f"lp_main_n{n_best}")

    # === Phase 3: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Spectral flattening (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_flattening(a_in, t3, verbose=False)
        check_and_update(a_out, f"spectral_n{n_best}")

    # === Phase 4: Batch coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Batch coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = batch_coordinate_descent(a_in, t4, rng_seed=77, verbose=True)
        check_and_update(a_out, f"batchcoord_n{n_best}")

    # === Phase 5: AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: AMD polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t5, top_k=80, beta=3.5, c0=4.0,
                                restart_every=4000, verbose=False)
        check_and_update(a_out, f"amd_n{n_best}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    # Try slightly longer and shorter
    length_candidates = []
    for delta in [500, -500, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            length_candidates.append(new_n)

    # Also try some "round" numbers
    for rn in [3000, 4000, 5000, 6000]:
        if rn != n_best and 1000 <= rn <= 7000:
            length_candidates.append(rn)

    # Remove duplicates, sort
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick AMD on new length
        t6a = min(remaining * 0.4, 120.0)
        if t6a > 10:
            a_out, _ = amd_optimize(a_long, t6a, top_k=60, beta=4.0, c0=3.0,
                                    restart_every=3000, verbose=False)
            check_and_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5500:
            t6b = min(remaining * 0.5, 150.0)
            if t6b > 20:
                a_in = norm_sum1(a_long)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, initial_K=300,
                                           max_K=2000, add_per_iter=40, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 6000:
        t7 = min(remaining * 0.40, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=500,
                                       max_K=3500, add_per_iter=60, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final batch coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final batch coord (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.50, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = batch_coordinate_descent(a_in, t8, rng_seed=314, verbose=True)
            check_and_update(a_out, f"batchcoord_final_n{n_best}")

    # === Phase 9: Final quick AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 5:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t9, top_k=100, beta=3.0, c0=5.0,
                                    restart_every=2000, verbose=False)
            check_and_update(a_out, f"amd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
