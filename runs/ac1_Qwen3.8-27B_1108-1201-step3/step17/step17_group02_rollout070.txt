```
def project_simplex(v):
    n=len(v)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    rho_candidates = 1+np.arange(n)
    # Find largest rho such that u[rho-1] + 1/rho*(css[rho-1]-1) >0
    vals = u + (css-1)/rho_candidates
    idx = np.where(vals>0)[0]
    if len(idx)==0:
        return np.ones(n)/n
    rho = idx[-1]+1
    theta = (css[rho-1]-1)/rho
    y=np.maximum(v-theta,0.0)
    # normalize due to numerics
    s=y.sum()
    if s<=0: return np.ones(n)/n
    return y/s
```

```python
"""
Search for non-negative step-function coefficients minimizing
    2 * n * max(convolve(a, a)) / (sum(a))^2.

The method is a tailored tangent-space under-projection:
1. Normalize the current vector to the simplex.
2. Use FFT to compute the full self-convolution c = a*a.
3. Build a subgradient of the smoothed maximum max(c) from active lags.
4. Form candidate tangent directions d = P - Q, where P is a sink
   distribution on low-gradient coordinates and Q is a source distribution
   on high-gradient / high-mass coordinates.
5. For each direction, exactly line-search in 1D using the identity
       conv(a + f d, a + f d) = conv(a,a) + 2f conv(a,d) + f^2 conv(d,d).
6. Accept improving moves; use adaptive peak weighting, periodic
   re-openings, random escaping moves, and multi-length interpolation.
"""

import numpy as np
import time

# ---------------------------------------------------------------------------
# Constants used by the tangent-direction line search
# ---------------------------------------------------------------------------

FRACTIONS_ARRAY = np.array(
    (
        1e-9, 1e-8, 1e-7, 1e-6, 5e-6,
        1e-5, 5e-5, 1e-4, 5e-4, 1e-3,
        2e-3, 5e-3, 1e-2, 2e-2, 5e-2,
        1e-1, 2e-1, 0.5, 0.8, 0.999
    ),
    dtype=np.float64,
)

SOFT_BETAS = (1.0, 4.0, 15.0)
SPARSE_KS = (1, 3, 8)
PEAK_TAUS = (20.0, 200.0, 2000.0, 20000.0, 100000.0)


# ---------------------------------------------------------------------------
# Basic utility / FFT helpers
# ---------------------------------------------------------------------------

def clean_array(x):
    """Return a finite float64 array, replacing non-finite entries by 0."""
    y = np.asarray(x, dtype=np.float64)
    return np.where(np.isfinite(y), y, 0.0)


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = clean_array(a)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or a.size == 0:
        n = max(1, int(a.size))
        return np.ones(n, dtype=np.float64)
    return a / s


def fast_conv(a):
    """Full linear self-convolution a * a via FFT."""
    a = clean_array(a).ravel()
    n = a.shape[0]
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_conv_full(a, b):
    """Full linear convolution a * b via FFT."""
    a = clean_array(a).ravel()
    b = clean_array(b).ravel()
    n = a.shape[0]
    m = b.shape[0]
    if n == 0 or m == 0:
        return np.zeros(1, dtype=np.float64)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def grad_from_weights(a, w):
    """
    Compute g[j] = 2 * sum_k w[k] * a[k-j] for j = 0..n-1.

    This is the subgradient of sum_k w[k] * conv(a,a)[k] with respect to a.
    Expects len(a)=n and len(w)=2n-1.
    """
    a = clean_array(a).ravel()
    w = clean_array(w).ravel()
    n = a.shape[0]
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    fa = a[::-1]
    conv_len = n + w.shape[0] - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    FA = np.fft.rfft(fa, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(FA * W, fft_size)[:conv_len]

    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = clean_array(v).ravel()
    n = v.shape[0]
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    vals = u + (css - 1.0) / rho_candidates
    idx = np.nonzero(vals > 0.0)[0]

    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rho = int(idx[-1] + 1)
    theta = (css[rho - 1] - 1.0) / float(rho)
    y = np.maximum(v - theta, 0.0)
    s = float(y.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return y / s


def evaluate_safe(a):
    """Safe wrapper around the external evaluate_sequence function."""
    try:
        x = clean_array(a).ravel()
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s <= 1e-12 or x.size == 0:
            return float("inf")
        x = x / s
        # The evaluator clips anyway, but keep values sane.
        x = np.minimum(x, 1000.0)
        seq = [float(t) for t in x.tolist()]
        val = float(evaluate_sequence(seq))
        return val if np.isfinite(val) else float("inf")
    except Exception:
        return float("inf")


def update_best(a, best_seq, best_score, label, t_start):
    """Evaluate a candidate and update the best sequence/score if improved."""
    if a is None:
        return best_seq, best_score

    x = clean_array(a).ravel()
    if x.size == 0:
        return best_seq, best_score

    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-12:
        return best_seq, best_score
    x = x / s

    sc = evaluate_safe(x)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = x.copy()
        elapsed = time.monotonic() - t_start
        print(f"  NEW BEST [{label}] score={best_score:.6f} n={len(x)} t={elapsed:.0f}s")

    return best_seq, best_score


# ---------------------------------------------------------------------------
# One-direction exact 1D line search
# ---------------------------------------------------------------------------

def evaluate_direction(a, c, d, max_step_frac):
    """
    For a zero-sum tangent direction d, line-search f in [0, fmax] such that
    a_f = a + f d stays non-negative, and minimize max(conv(a_f, a_f)).

    Returns (f, m_new) if an improvement over c.max() is found, else None.
    """
    a = clean_array(a).ravel()
    c = clean_array(c).ravel()
    d = clean_array(d).ravel()

    if a.size == 0 or c.size == 0 or d.size != a.size:
        return None

    neg = np.nonzero(d < -1e-15)[0]
    if neg.size == 0:
        return None

    fmax = float(np.min(a[neg] / (-d[neg])))
    if not np.isfinite(fmax) or fmax <= 1e-15:
        return None

    if max_step_frac is not None and max_step_frac > 0:
        fmax = min(fmax, float(max_step_frac))

    base_m = float(c.max())
    if not np.isfinite(base_m):
        return None

    A = 2.0 * cross_conv_full(a, d)
    B = fast_conv(d)

    if not np.all(np.isfinite(A)) or not np.all(np.isfinite(B)):
        return None

    fs = FRACTIONS_ARRAY * fmax
    fs = fs[(fs > 0.0) & (fs < fmax + 1e-15)]
    if fs.size == 0:
        return None

    f_col = fs[:, None]
    vals = c[None, :] + f_col * A + (f_col * f_col) * B
    ms = vals.max(axis=1)
    i = int(np.argmin(ms))

    best_f = float(fs[i])
    best_m = float(ms[i])

    width = max(fmax * 1e-5, abs(best_f) * 0.4)
    for _ in range(4):
        lo = max(0.0, best_f - width)
        hi = min(fmax, best_f + width)
        if hi - lo <= max(fmax * 1e-14, 1e-16):
            break

        grid = np.linspace(lo, hi, 7)
        g2 = grid[:, None]
        vals = c[None, :] + g2 * A + (g2 * g2) * B
        ms = vals.max(axis=1)
        j = int(np.argmin(ms))

        if ms[j] < best_m:
            best_m = float(ms[j])
            best_f = float(grid[j])
            width = (hi - lo) * 0.25
        else:
            break

    if best_f > 0.0 and best_m < base_m:
        return best_f, best_m

    return None


# ---------------------------------------------------------------------------
# Main tangent-space subgradient search
# ---------------------------------------------------------------------------

def tangent_search(a0, time_limit, verbose=True, rng_seed=1, tau_stage0=0):
    """
    Local tangent-space search.  Returns (a, M) where M = max(conv(a,a))
    and a is normalized to sum 1.
    """
    if time_limit <= 2.0:
        a = norm_sum1(a0)
        c = fast_conv(a)
        return a, float(c.max())

    a = norm_sum1(a0)
    n = a.shape[0]
    c = fast_conv(a)
    M = float(c.max())

    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    it = 0
    tau_stage = int(tau_stage0) % len(PEAK_TAUS)
    max_step_soft = 0.35
    max_step_sparse = 1.0
    last_out = 0.0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        it += 1

        # Periodically re-normalize and recompute exactly to remove drift.
        if it % 150 == 0:
            a = norm_sum1(a)
            c = fast_conv(a)
            M = float(c.max())
            if M < best_M:
                best_M = M
                best_a = a.copy()

        # --- Peak weighting for the smooth maximum subgradient ----------
        peak_tau = float(PEAK_TAUS[tau_stage])
        shifted = c - M
        floor = -50.0 / peak_tau
        shifted = np.maximum(shifted, floor)
        w = np.exp(peak_tau * shifted)
        sw = float(w.sum())

        if not np.isfinite(sw) or sw <= 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / sw

        g = grad_from_weights(a, w)
        g = np.where(np.isfinite(g), g, 0.0)

        gm = float(g.mean())
        gs_std = float(g.std())
        if gs_std < 1e-12:
            gs_std = 1e-12
        gs = (g - gm) / gs_std
        gs = np.clip(gs, -10.0, 10.0)

        order = np.argsort(g)
        current_M = M

        best_f = 0.0
        best_m = current_M + 1.0
        best_d = None

        def consider(d):
            nonlocal best_f, best_m, best_d
            if d is None:
                return
            d = clean_array(d)
            if not np.any(d != 0.0):
                return
            # Quick linear screen with the current subgradient.
            lin = float(np.dot(g, d))
            if not np.isfinite(lin) or lin > -1e-14:
                # Still try if stall is large? For speed, skip usually.
                if stall < 200:
                    return
            res = evaluate_direction(a, c, d, max_step_soft)
            if res is not None and res[1] < best_m:
                best_f = res[0]
                best_m = res[1]
                best_d = d

        # --- Soft sink/source directions --------------------------------
        for beta in SOFT_BETAS:
            P = np.exp(-beta * gs)
            P = np.where(np.isfinite(P), P, 0.0)
            sp = float(P.sum())
            if sp <= 1e-300:
                continue
            P = P / sp

            Q = np.exp(beta * gs) * (a + 1e-15)
            Q = np.where(np.isfinite(Q), Q, 0.0)
            sq = float(Q.sum())
            if sq <= 1e-300:
                continue
            Q = Q / sq

            d = P - Q
            d = d - float(d.sum()) / float(n)
            consider(d)

        # --- Sparse sink/source directions ------------------------------
        for k in SPARSE_KS:
            p_idx = order[:k]
            P = np.zeros(n, dtype=np.float64)
            P[p_idx] = 1.0 / float(max(1, len(p_idx)))
            sp = float(P.sum())
            if sp <= 1e-300:
                continue
            P = P / sp

            q_cand = order[::-1][:max(2 * k, 4)]
            q_cand = q_cand[a[q_cand] > 1e-12]
            if q_cand.size == 0:
                continue

            Q = np.zeros(n, dtype=np.float64)
            qmass = a[q_cand]
            sm = float(qmass.sum())
            if sm <= 1e-300:
                continue
            Q[q_cand] = qmass / sm

            d = P - Q
            d = d - float(d.sum()) / float(n)

            cap = 1.0 if k == 1 else max_step_sparse
            # evaluate_direction's max_step_frac is hard-coded to soft cap in
            # consider(); therefore call it directly with sparse cap.
            res = evaluate_direction(a, c, d, cap)
            if res is not None and res[1] < best_m:
                best_f = res[0]
                best_m = res[1]
                best_d = d

        # --- Peak-specific single-layer pair moves ----------------------
        top_peaks = np.argsort(c)[::-1][:3]
        for kk in top_peaks:
            ki = int(kk)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            if s > e:
                continue
            idx = np.arange(s, e + 1, dtype=np.int64)
            partners = ki - idx

            p_pos = int(idx[np.argmin(a[partners])])
            q_pos = int(idx[np.argmax(a[partners])])

            if p_pos == q_pos or a[q_pos] <= 1e-14:
                continue

            P = np.zeros(n, dtype=np.float64)
            Q = np.zeros(n, dtype=np.float64)
            P[p_pos] = 1.0
            Q[q_pos] = 1.0
            d = P - Q
            res = evaluate_direction(a, c, d, 1.0)
            if res is not None and res[1] < best_m:
                best_f = res[0]
                best_m = res[1]
                best_d = d

        # --- Random exploration directions ------------------------------
        if best_m > current_M and (stall > 0 or it % 20 == 0):
            pos_idx = np.nonzero(a > 1e-12)[0]
            max_rand = 6 if stall < 50 else 14
            for rr in range(max_rand):
                if pos_idx.size == 0:
                    break
                k = 1 + (rr % 4)

                if rr % 2 == 0:
                    p_idx = order[:min(k, n)]
                else:
                    p_idx = rng.choice(n, size=min(k, n), replace=False)

                q_idx = rng.choice(pos_idx, size=min(k, len(pos_idx)), replace=False)

                P = np.zeros(n, dtype=np.float64)
                P[p_idx] = 1.0 / float(max(1, len(p_idx)))
                sp = float(P.sum())
                if sp <= 1e-300:
                    continue
                P = P / sp

                Q = np.zeros(n, dtype=np.float64)
                qmass = a[q_idx]
                sm = float(qmass.sum())
                if sm <= 1e-300:
                    continue
                Q[q_idx] = qmass / sm

                d = P - Q
                d = d - float(d.sum()) / float(n)
                res = evaluate_direction(a, c, d, 0.8)
                if res is not None and res[1] < best_m:
                    best_f = res[0]
                    best_m = res[1]
                    best_d = d

        # --- Acceptance -------------------------------------------------
        if best_m < current_M - 1e-15 and best_f > 0.0 and best_d is not None:
            a_new = a + best_f * best_d
            a_new = np.maximum(a_new, 0.0)
            S = float(a_new.sum())

            if S <= 1e-12:
                stall += 1
            else:
                a_new = a_new / S
                c_new = fast_conv(a_new)
                M_new = float(c_new.max())

                if np.isfinite(M_new) and M_new < current_M - 1e-15:
                    a = a_new
                    c = c_new
                    M = M_new
                    stall = 0

                if M < best_M:
                    best_M = M
                    best_a = a.copy()
        else:
            stall += 1

            if stall > 100:
                sigma = min(2.0, 0.2 + 1e-4 * float(stall))
                noise = rng.normal(0.0, sigma, n) * (a + 1e-6)
                a_pert = project_simplex(a + noise)
                c_pert = fast_conv(a_pert)
                M_pert = float(c_pert.max())

                # Accept if it is a new best, or if stuck very long.
                if np.isfinite(M_pert) and (M_pert < best_M or stall > 500):
                    a = a_pert
                    c = c_pert
                    M = M_pert
                    stall = 0
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()

            if stall > 1200:
                # Hard restart from best with larger perturbation.
                sigma = 0.6
                noise = rng.normal(0.0, sigma, n) * (best_a + 1e-6)
                a = project_simplex(best_a + noise)
                c = fast_conv(a)
                M = float(c.max())
                stall = 0
                tau_stage = 0
                max_step_soft = 0.35
                max_step_sparse = 1.0

        # --- Adaptive parameters -----------------------------------------
        if it % 250 == 0:
            tau_stage = (tau_stage + 1) % len(PEAK_TAUS)

        if stall > 200:
            max_step_soft = max(max_step_soft, 0.6)
            max_step_sparse = max(max_step_sparse, 1.0)

        # --- Progress output ---------------------------------------------
        if verbose and (time.monotonic() - last_out) > 100.0:
            curJ = 2.0 * n * M
            bestJ = 2.0 * n * best_M
            print(
                f"    tangent t={elapsed:.0f}s it={it} "
                f"J={curJ:.6f} bestJ={bestJ:.6f} stall={stall} tau={PEAK_TAUS[tau_stage]:.0f}"
            )
            last_out = time.monotonic()

    # Final exact cleanup.
    best_a = norm_sum1(best_a)
    best_c = fast_conv(best_a)
    best_M = float(best_c.max())
    return best_a, best_M


# ---------------------------------------------------------------------------
# Initializers and interpolation
# ---------------------------------------------------------------------------

def make_u_shape(n, sharpness=2.0):
    """U-shaped initializer, heavier in the tails."""
    n = int(max(1, n))
    x = np.linspace(-1.0, 1.0, n)
    a = 1.0 + sharpness * np.abs(x) ** 1.5
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    """Two Gaussian bumps near the ends."""
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    a = (
        np.exp(-((x - 0.12) / width) ** 2)
        + np.exp(-((x - 0.88) / width) ** 2)
        + 1e-4
    )
    return norm_sum1(a)


def make_spectral(n, n_modes=20, seed=0):
    """Smooth non-negative random Fourier initializer."""
    n = int(max(1, n))
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.5

    for freq in range(1, int(n_modes) + 1):
        coeff = rng.normal(0.0, 0.18 / np.sqrt(float(freq)))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += coeff * np.sin(2.0 * np.pi * freq * x + phase)

    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    f = norm_sum1(f)
    old_n = f.shape[0]
    new_n = int(max(1, new_n))

    if old_n == new_n:
        return f.copy()
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, f)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return y / s

def propose_candidate(budget_s=1800.0):
    """
    Return the best coefficient list found within the time budget.
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    if time_budget < 30.0:
        time_budget = max(30.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Start from the parent sequence when available.
    # -----------------------------------------------------------------
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = clean_array(p).ravel()
        p = np.maximum(p, 0.0)
        if p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 1e-10:
            parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is not None:
        start_seq = parent.copy()
        print(
            f"Parent loaded: n={len(start_seq)}, "
            f"score={evaluate_safe(start_seq):.6f}"
        )
    else:
        start_seq = make_u_shape(4200, 2.0)
        print(f"No parent; using U-shape start: n={len(start_seq)}")

    best_seq, best_score = update_best(start_seq, best_seq, best_score, "start", t_start)

    # If parent is too large for FFT-friendly optimization, search on a
    # shorter interpolated copy while keeping the original best if better.
    search_seq = best_seq.copy() if best_seq is not None else start_seq
    if search_seq.size > 6000:
        search_seq = interpolate_sequence(search_seq, 5000)
        best_seq, best_score = update_best(search_seq, best_seq, best_score, "interp_toparent", t_start)

    n_main = int(len(search_seq))
    if n_main > 6000:
        n_main = 5000
        search_seq = interpolate_sequence(search_seq, n_main)

    # -----------------------------------------------------------------
    # Phase 1: Strong tangent search on the parent / best sequence.
    # -----------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_start)
    t1 = min(rem - 10.0, 720.0)
    if t1 > 60.0 and search_seq.size <= 6000:
        a1, _ = tangent_search(search_seq, t1, verbose=True, rng_seed=11, tau_stage0=0)
        best_seq, best_score = update_best(a1, best_seq, best_score, "tangent_parent", t_start)

    # -----------------------------------------------------------------
    # Phase 2: Continue from the current best with a different tau schedule.
    # -----------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_start)
    t2 = min(rem - 10.0, 450.0)
    if t2 > 50.0 and best_seq is not None and len(best_seq) <= 6000:
        a2, _ = tangent_search(best_seq.copy(), t2, verbose=True, rng_seed=22, tau_stage0=2)
        best_seq, best_score = update_best(a2, best_seq, best_score, "tangent_best2", t_start)

    # -----------------------------------------------------------------
    # Phase 3: Explore different low-dimensional shapes at the main length.
    # -----------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_start)
    t3 = min(rem - 10.0, 280.0)
    if t3 > 70.0:
        # U-shape variant
        au = make_u_shape(n_main, 2.3)
        best_seq, best_score = update_best(au, best_seq, best_score, "u_shape", t_start)

        t3a = min(t3 * 0.45, 160.0)
        if t3a > 40.0:
            au2, _ = tangent_search(au, t3a, verbose=False, rng_seed=33, tau_stage0=1)
            best_seq, best_score = update_best(au2, best_seq, best_score, "u_tangent", t_start)

        rem = time_budget - (time.monotonic() - t_start)
        if rem > 80.0:
            ab = make_bimodal(n_main, 0.14)
            best_seq, best_score = update_best(ab, best_seq, best_score, "bimodal", t_start)

            t3b = min(rem * 0.35, 110.0)
            if t3b > 35.0:
                ab2, _ = tangent_search(ab, t3b, verbose=False, rng_seed=34, tau_stage0=0)
                best_seq, best_score = update_best(ab2, best_seq, best_score, "bimodal_tangent", t_start)

    # -----------------------------------------------------------------
    # Phase 4: Length exploration.
    # -----------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_start)
    if best_seq is not None:
        base_n = int(len(best_seq))
        trial_lens = []
        for delta in (400, 800, -400):
            nn = base_n + delta
            if 3000 <= nn <= 5600:
                trial_lens.append(nn)

        # Deduplicate preserving order.
        seen = set()
        unique_lens = []
        for nn in trial_lens:
            if nn not in seen:
                seen.add(nn)
                unique_lens.append(nn)

        for nn in unique_lens[:2]:
            rem = time_budget - (time.monotonic() - t_start)
            if rem < 180.0:
                break

            a_interp = interpolate_sequence(best_seq, nn)
            best_seq, best_score = update_best(a_interp, best_seq, best_score, f"interp_n{nn}", t_start)

            t4 = min(rem * 0.42, 250.0)
            if t4 > 60.0:
                a4, _ = tangent_search(a_interp, t4, verbose=False, rng_seed=44 + (nn % 53), tau_stage0=1)
                best_seq, best_score = update_best(a4, best_seq, best_score, f"tangent_n{nn}", t_start)

    # -----------------------------------------------------------------
    # Phase 5: Final focused polish on the best sequence.
    # -----------------------------------------------------------------
    rem = time_budget - (time.monotonic() - t_start)
    if best_seq is not None and len(best_seq) <= 6000:
        t5 = min(rem - 15.0, 420.0)
        if t5 > 40.0:
            a5, _ = tangent_search(best_seq.copy(), t5, verbose=True, rng_seed=77, tau_stage0=3)
            best_seq, best_score = update_best(a5, best_seq, best_score, "final_tangent", t_start)

    # -----------------------------------------------------------------
    # Final safety cleanup.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = make_u_shape(3000, 1.8)

    best_seq = norm_sum1(best_seq)
    best_seq = np.maximum(best_seq, 0.0)

    final_score = evaluate_safe(best_seq)
    if not np.isfinite(final_score):
        best_seq = make_u_shape(2500, 1.6)
        final_score = evaluate_safe(best_seq)

    print(
        f"FINAL score={final_score:.6f} n={len(best_seq)} "
        f"time={time.monotonic() - t_start:.0f}s"
    )

    return [float(x) for x in best_seq.tolist()]
```
