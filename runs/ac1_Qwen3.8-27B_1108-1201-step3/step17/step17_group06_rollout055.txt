def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    conv = fast_conv(a)
    return 2.0 * n * conv.max()

    n = len(a)

    conv_scaled -= conv_scaled.max()  # numerical stability

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with exact soft-max subgradients (all lags)
2. Adaptive temperature schedule for subgradient sharpness
3. Multi-phase strategy: MD -> length sweep -> extended MD -> local search
4. Parametric shape search (power-law, skewed Gaussian) for diverse initializations
5. Pair-wise local search for final polishing
"""

import numpy as np
import time

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """J = 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def soft_subgrad(a, conv, tau):
    """Exact soft-max subgradient of max(conv) using all lags."""
    n = len(a)
    c = conv / tau
    c -= c.max()
    w = np.exp(c)
    w /= w.sum()
    # g_j = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    a_rev = a[::-1].copy()
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(Ar * W, fft_size)[:conv_len]
    g = 2.0 * cross[n - 1:2 * n - 1]
    return g

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def md_soft(a0, time_limit, tau0=0.01, tau_min=0.001, c0=5.0,
            restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent with soft-max subgradient."""
    n = len(a0)
    x = normalize(np.asarray(a0, dtype=np.float64))
    x_prev = x.copy()
    best_a = x.copy()
    best_val = eval_J(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(1.0, t / 20000.0)
        tau = tau0 * (tau_min / tau0) ** progress
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            y = y / sy
        else:
            y = x.copy()
        
        # Compute subgradient
        conv = fast_conv(y)
        g = soft_subgrad(y, conv, tau)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror step (KL projection onto simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * np.sqrt(best_a)
            x_r = best_a + pert
            x_r = normalize(x_r)
            x = x_r.copy()
            x_prev = x_r.copy()
            c0 = max(c0 * 0.92, 0.5)
            if verbose:
                print(f"    MD restart t={t}: J={best_val:.6f}, c0={c0:.2f}")
        
        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, tau={tau:.5f}")
    
    return best_a, best_val

def local_search(a0, time_limit, verbose=False, seed=123):
    """Local search: try pair-wise mass transfers guided by subgradient."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_conv(a)
        tau = 0.005
        g = soft_subgrad(a, conv, tau)
        
        # Strategy: try to move mass from high g to low g
        # (where increasing a_j most increases the peak)
        # Source: positions where g is high and a is significant
        # Dest: positions where g is low
        
        # Sample candidate (source, dest) pairs
        improved = False
        n_tries = min(50, max(10, n // 50))
        
        for _ in range(n_tries):
            # Weighted sampling: source from high g*a, dest from low g
            src_weights = np.maximum(g * a, 0)
            src_weights /= src_weights.sum() + 1e-30
            dst_weights = np.maximum(1.0 + g, 0)  # prefer low g
            dst_weights /= dst_weights.sum() + 1e-30
            
            src = rng.choice(n, p=src_weights)
            dst = rng.choice(n, p=dst_weights)
            
            if src == dst or a[src] < 1e-12:
                continue
            
            # Try several transfer fractions
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                if a_t[src] < -1e-14:
                    continue
                a_t = np.maximum(a_t, 0)
                a_t /= a_t.sum()
                J_t = eval_J(a_t)
                if J_t < best_J - 1e-15:
                    a = a_t
                    best_J = J_t
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 20 == 0:
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.02)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            no_improve = 0
                            break
                        else:
                            a = best_a.copy()
        
        if verbose and no_improve % 50 == 0:
            print(f"    LS ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, no_improve={no_improve}")
    
    return best_a, best_J

def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_skew_gauss(n, mu=0.3, alpha=5.0, beta=2.0):
    """Skewed Gaussian: exp(-alpha*(x-mu)^2 + beta*(x-mu))."""
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * (x - mu) ** 2 + beta * (x - mu))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_power_volume(n, p=1.0, q=0.5):
    """a_i ∝ (i+1)^p * (n-i+1)^q"""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p * (n - i + 1) ** q
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with exact soft-max subgradients over ALL lags
2. Adaptive temperature annealing for subgradient sharpness
3. Multi-phase strategy: MD → multi-init → length sweep → extended MD → local search
4. Parametric shape search for diverse initializations
5. Subgradient-guided local search for final polishing
"""

import numpy as np
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """J = 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def soft_subgrad(a, conv, tau):
    """Exact soft-max subgradient of max(conv) using all lags via FFT."""
    n = len(a)
    c = conv / tau
    c -= c.max()
    w = np.exp(c)
    w /= w.sum()
    # g_j = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # = 2 * (a_rev * w)[n-1+j]  via convolution
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(Ar * W, fft_size)[:conv_len]
    g = 2.0 * cross[n - 1:2 * n - 1]
    return g


def normalize(a):
    """Project to simplex: non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def md_soft(a0, time_limit, tau0=0.01, tau_min=0.001, c0=5.0,
            restart_every=5000, verbose=False, seed=42):
    """Nesterov-accelerated mirror descent with soft-max subgradient."""
    n = len(a0)
    x = normalize(np.asarray(a0, dtype=np.float64))
    x_prev = x.copy()
    best_a = x.copy()
    best_val = eval_J(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: geometric decay
        progress = min(1.0, t / 30000.0)
        tau = tau0 * (tau_min / tau0) ** progress

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
        else:
            y = x.copy()

        # Compute soft-max subgradient (all lags)
        conv = fast_conv(y)
        g = soft_subgrad(y, conv, tau)

        # Step size (Polylak-type)
        eta = c0 / (np.sqrt(t) + 1.0)

        # Mirror step (KL projection onto simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.04)
            pert = rng.normal(0, scale, n) * np.sqrt(np.maximum(best_a, 0))
            x_r = normalize(best_a + pert)
            x = x_r.copy()
            x_prev = x_r.copy()
            c0 = max(c0 * 0.93, 0.5)
            if verbose:
                print(f"    MD restart t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, c0={c0:.2f}")

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, tau={tau:.5f}")

    return best_a, best_val


def local_search(a0, time_limit, verbose=False, seed=123):
    """Subgradient-guided local search with pair-wise mass transfers."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_conv(a)
        g = soft_subgrad(a, conv, 0.005)

        improved = False
        n_tries = min(80, max(20, n // 40))

        for _ in range(n_tries):
            # Source: high g (contributes most to peak)
            src_w = np.maximum(g * a, 0)
            src_w_sum = src_w.sum()
            if src_w_sum < 1e-30:
                break
            src_w /= src_w_sum

            # Destination: low g (contributes least)
            g_max = g.max() + 1e-15
            dst_w = g_max - g + 0.01
            dst_w_sum = dst_w.sum()
            dst_w /= dst_w_sum

            src = rng.choice(n, p=src_w)
            dst = rng.choice(n, p=dst_w)

            if src == dst or a[src] < 1e-12:
                continue

            for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.4]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                if a_t[src] < -1e-14:
                    continue
                a_t = np.maximum(a_t, 0)
                a_t /= a_t.sum()
                J_t = eval_J(a_t)
                if J_t < best_J - 1e-15:
                    a = a_t
                    best_J = J_t
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                # Random kick
                for _ in range(8):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            no_improve = 0
                            improved = True
                            break
                        else:
                            a = best_a.copy()

        if verbose and t0 and (time.monotonic() - t0) % 30 < 0.1:
            print(f"    LS ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, no_imp={no_improve}")

    return best_a, best_J


def sweep_local(a0, time_limit, verbose=False):
    """Sequential sweep: for each position, try optimal transfer to a few neighbors."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    order = list(range(n))
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        np.random.seed(pass_count)
        rng_pass = np.random.default_rng(pass_count)
        rng_pass.shuffle(order)

        pass_improved = False
        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = order[idx]
            if a[j] < 1e-12:
                continue

            # Try moving to a few nearby and random positions
            candidates = set()
            for d in [-3, -2, -1, 1, 2, 3, -5, 5, -10, 10]:
                k = j + d
                if 0 <= k < n and k != j:
                    candidates.add(k)
            # Add a few random targets
            for _ in range(5):
                candidates.add(int(rng_pass.integers(n)))

            for k in candidates:
                if k == j:
                    continue
                for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
                    amt = a[j] * frac
                    if amt < 1e-14:
                        continue
                    a_t = a.copy()
                    a_t[j] -= amt
                    a_t[k] += amt
                    if a_t[j] < -1e-14:
                        continue
                    a_t = np.maximum(a_t, 0)
                    a_t /= a_t.sum()
                    J_t = eval_J(a_t)
                    if J_t < best_J - 1e-15:
                        a = a_t
                        best_J = J_t
                        best_a = a.copy()
                        pass_improved = True
                        break

        pass_count += 1
        if not pass_improved and pass_count > 3:
            break
        if verbose and pass_count % 2 == 0:
            print(f"    Sweep pass {pass_count} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_power_volume(n, p=1.0, q=0.5):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p * (n - i + 1) ** q
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_skew_gauss(n, mu=0.3, alpha=5.0, beta=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * (x - mu) ** 2 + beta * (x - mu))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
