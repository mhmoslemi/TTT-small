Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with annealed soft-max
2. Exact subgradient bundle method
3. LP-based Frank-Wolfe for final polishing
4. Coordinate descent for last-mile improvements

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad_softmax(a, p):
    """
    Compute gradient of soft-max objective.
    grad[j] = sum_k p_k * a[k-j] for valid k,j
    = convolve(a[::-1], p)[j + n - 1]
    """
    n = len(a)
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * P, fft_size)[:conv_len]
    return conv[n-1:2*n-1]

def softmax(x, max_val=None):
    if max_val is None:
        max_val = x.max()
    exp_x = np.exp(x - max_val)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return exp_x / s

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad(a, p):
    """
    Compute grad[j] = sum_k p[k] * a[k-j] for valid indices.
    Equivalently: convolve(a[::-1], p)[j + n - 1] for j=0..n-1.
    """
    n = len(a)
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * P, fft_size)[:conv_len]
    return conv[n - 1:2 * n - 1]


def softmax(x, max_val=None):

    exp_x = np.exp(x - max_val)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return exp_x / s

def nesterov_softmax(a0, time_limit, t_start_ratio=0.05, t_end_ratio=1e-4,
                      decay=0.93, inner_iters=80, gamma=3.0,
                      verbose=True, rng_seed=42):
    """
    Nesterov-accelerated mirror descent with annealed soft-max.
    Minimizes f_t(a) = t * log(sum_k exp(c_k/t)) which approximates max(c_k) as t->0.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    c_init = fast_autocorr(a)
    M_init = c_init.max()
    t = M_init * t_start_ratio
    t_end = M_init * t_end_ratio

    x = a.copy()
    x_prev = a.copy()
    global_iter = 0

    while t > t_end:
        # Adaptive inner iterations based on remaining time
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        est_iters_remaining = int(np.log(t / t_end) / np.log(1.0 / decay))
        avail_per_phase = remaining / max(1, est_iters_remaining)
        n_inner = max(10, min(inner_iters, int(avail_per_phase / 0.05)))

        for _ in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break
            global_iter += 1

            beta_t = (global_iter - 1.0) / (global_iter + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            c = fast_autocorr(y)
            p = softmax(c / t)
            grad = compute_grad(y, p)

            # Adaptive learning rate
            eta = gamma / (np.sqrt(global_iter) + 1.0)

            log_a = np.log(y) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()

            x_prev = x
            x = a_new

            M_new = fast_autocorr(x).max()
            if M_new < best_m - 1e-18:
                best_m = M_new
                best_a = x.copy()

        # Also check Nesterov extrapolate

            y_ext = np.maximum(y_ext, 1e-15)

            M_ext = fast_autocorr(y_ext).max()

                best_a = y_ext.copy()

            elapsed = time.monotonic() - t0

            print(f"    SoftMax t_idx={int(np.log(M_init/t)/np.log(1/decay))} ({elapsed:.0f}s): J={J:.8f}")

def nesterov_subgrad(a0, time_limit, gamma=2.0, top_k=1,
                     restart_every=5000, verbose=False, rng_seed=99):
    """
    Nesterov-accelerated mirror descent with exact (or top-k) subgradient.
    Used in the cold regime where soft-max gradient ~= exact subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_autocorr(y)
        # Top-k subgradient: average of top-k peak subgradients
        top_idx = np.argsort(c)[::-1][:top_k]
        if len(top_idx) == 1:
            grad = compute_grad(y, np.eye(len(c))[top_idx[0]])
        else:
            # Equal-weight top-k
            p = np.zeros(len(c))
            for k in top_idx:
                p[k] = 1.0 / len(top_idx)
            grad = compute_grad(y, p)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SubGrad restart t={t} ({elapsed:.0f}s): J={J:.8f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=None):
    """LP-based Frank-Wolfe with adaptive constraint selection and Nesterov."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if top_K is None:
        top_K = min(300, 2 * n - 1)

    K_sel = min(top_K, 2 * n - 1)
    use_all = (n <= 1500)

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            constraint_vals = np.zeros(n)
            for k in range(2 * n - 1):
                ki = int(k)
                s = max(0, ki - n + 1)
                e = min(n, ki + 1)
                idx = np.arange(s, e)
                constraint_vals[idx] += y[ki - idx]
            # Actually compute T @ y properly
            T_y = np.zeros(2 * n - 1)
            for j in range(n):
                T_y[j:j + n] += y[j] * y  # this is wrong, let me fix
            # Use FFT instead
            # constraint_vals = ac (which is T_full @ y = ac)
            constraint_vals = ac
            margins = M_curr - constraint_vals
            binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix rows for selected peaks
        rows = []
        rhs = []
        for k in binding_idx:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            partner = ki - idx
            row = np.zeros(n)
            row[idx] = y[partner]
            rows.append(row)
            rhs.append(M_curr)

        if not rows:
            break

        T_sel = np.array(rows)
        rhs_arr = np.array(rhs)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= rhs_arr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.8f}")

        if not use_all and t % 10 == 0:
            K_sel = min(K_sel + 30, 2 * n - 1)
            if K_sel >= 2 * n - 1:
                use_all = True

    return best_a, best_m


def pairwise_coord_descent(a0, time_limit, verbose=True, rng_seed=123):
    """
    Pairwise coordinate descent: for each pair (i,j), find optimal mass transfer.
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:10]

        # Build sensitivity
        sens = np.zeros(n)
        for k in top_peaks:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            sens[idx] += a[ki - idx]

        improved = False

        # For peaks, try reducing the most responsible coordinate
        for k in top_peaks[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination (low sensitivity)
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(sens[candidates])]

            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.8f}, stall={stall}")

    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_cosine(n, power=1.0):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** (2 * power)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_triangle(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - 2.0 * np.abs(x - 0.5)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Annealed soft-max (primary method) ===
    print(f"\n=== Phase 1: Annealed SoftMax (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 60:
        # Try from parent
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_softmax(a_in, t_phase1 * 0.5,
                                     t_start_ratio=0.05, t_end_ratio=1e-5,
                                     decay=0.92, inner_iters=100, gamma=3.0,
                                     verbose=True, rng_seed=42)
        try_update(a_out, "softmax_parent")

    # === Phase 2: Soft-max with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SoftMax variant (n={len(best_seq)}) ===")
    t_phase2 = min(remaining * 0.15, 250.0)

    if t_phase2 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_softmax(a_in, t_phase2,
                                     t_start_ratio=0.02, t_end_ratio=1e-6,
                                     decay=0.90, inner_iters=60, gamma=2.0,
                                     verbose=True, rng_seed=77)
        try_update(a_out, "softmax_cold")

    # === Phase 3: Multi-init soft-max ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multi-init (n={n0}) ===")
    t_phase3 = min(remaining * 0.15, 250.0)

    if t_phase3 > 60:
        inits = [
            (make_u_shape(n0, 1.0), "u1.0"),
            (make_u_shape(n0, 3.0), "u3.0"),
            (make_cosine(n0, 1.5), "cos1.5"),
            (make_bimodal(n0, 0.2), "bimodal"),
            (spectral_init(n0, 25, 1), "spec1"),
            (spectral_init(n0, 15, 42), "spec2"),
        ]
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.0)
            inits.append((norm_sum1(blend), "blend"))

        t_per = t_phase3 / len(inits)
        for idx, (init, lbl) in enumerate(inits):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 20:
                break
            print(f"  Multi-init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = nesterov_softmax(init, t_avail,
                                         t_start_ratio=0.05, t_end_ratio=1e-4,
                                         decay=0.93, inner_iters=80, gamma=3.0,
                                         verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"multi_{lbl}")

    # === Phase 4: Exact subgradient Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: SubGrad Nesterov (n={len(best_seq)}) ===")
    t_phase4 = min(remaining * 0.10, 150.0)

    if t_phase4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgrad(a_in, t_phase4, gamma=2.5, top_k=1,
                                     restart_every=4000, verbose=False, rng_seed=55)
        try_update(a_out, "subgrad1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out, _ = nesterov_subgrad(a_in, remaining * 0.5, gamma=2.0, top_k=3,
                                         restart_every=3000, verbose=False, rng_seed=66)
            try_update(a_out, "subgrad3")

    # === Phase 5: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW (n={n_best}) ===")
    t_phase5 = min(remaining * 0.15, 250.0)

    if t_phase5 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K = min(500, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t_phase5, solver, verbose=True, top_K=top_K)
        try_update(a_out, "lp_main")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord Descent (n={n_best}) ===")
    t_phase6 = min(remaining * 0.08, 100.0)

    if t_phase6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_coord_descent(a_in, t_phase6, verbose=True, rng_seed=321)
        try_update(a_out, "coord")

    # === Phase 7: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP round 2 (n={n_best}) ===")
    t_phase7 = min(remaining * 0.10, 150.0)

    if t_phase7 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K = min(800, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t_phase7, solver, verbose=True, top_K=top_K)
        try_update(a_out, "lp_r2")

    # === Phase 8: Final soft-max polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final SoftMax polish (n={n_best}) ===")
    t_phase8 = min(remaining * 0.15, 200.0)

    if t_phase8 > 30:
        a_in = norm_sum1(best_seq)
        # Very cold soft-max for final polish
        a_out, _ = nesterov_softmax(a_in, t_phase8,
                                     t_start_ratio=0.01, t_end_ratio=1e-7,
                                     decay=0.88, inner_iters=50, gamma=1.5,
                                     verbose=True, rng_seed=999)
        try_update(a_out, "softmax_final")

    # === Phase 9: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Length exploration ===")

    if remaining > 120:
        trial_lengths = [n_best + 300, n_best - 300, n_best + 600]
        trial_lengths = [l for l in trial_lengths if 2000 <= l <= 6000]
        trial_lengths = list(dict.fromkeys(trial_lengths))

        for new_n in trial_lengths[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={new_n}...")
            a_interp = interpolate_sequence(best_seq, new_n)
            t_avail = min(remaining * 0.4, 100.0)
            if t_avail > 30:
                a_out, _ = nesterov_softmax(a_interp, t_avail,
                                             t_start_ratio=0.05, t_end_ratio=1e-5,
                                             decay=0.93, inner_iters=80, gamma=3.0,
                                             verbose=False, rng_seed=new_n % 500)
                try_update(a_out, f"len_{new_n}")

    # === Phase 10: Final coordination ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 10: Final coordination ===")
        a_in = norm_sum1(best_seq)
        t_avail = min(remaining * 0.5, 60.0)
        a_out, _ = pairwise_coord_descent(a_in, t_avail, verbose=False, rng_seed=777)
        try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    return [float(x) for x in best_seq]
