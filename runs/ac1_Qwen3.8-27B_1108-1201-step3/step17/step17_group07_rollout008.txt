        # This is: minimize 0 subject to sum(a')=1, a'>=0, (conv(a',a'))_k <= M_mid for all k
        # But this is NONLINEAR (conv is quadratic)!

        # So we need to linearize: use the current a to linearize
        # (conv(a', a'))_k ≈ 2 * sum_j a'_{k-j} * a_j - b_k
        # So the constraint becomes: 2 * sum_j a'_{k-j} * a_j <= b_k + M_mid
        # Which is: sum_j a'_{k-j} * (2*a_j) <= b_k + M_mid
        
        # This is a linear program!
        a_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(a_var) == 1]
        for k in range(2*n-1):
            # (conv(a', a'))_k ≈ 2 * sum_j a'_{k-j} * a_j - b_k
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)

            coeffs = 2.0 * a[k - idx]

            a = normalize(a_var.value)  # update linearization point
        else:
            M_low = M_mid
        
        if M_high - M_low < 1e-8:

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def fast_conv(a, fft_size=None):
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_gradient(a, weights, fft_size):
    """Compute h_j = sum_k weights[k] * a[k-j] where 0<=k-j<n (cross-correlation)."""
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    h = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    return h

def nesterov_smooth_mirror_descent(a0, time_limit, beta_init=30.0, beta_end=300.0,
                                    verbose=False):
    """Nesterov-accelerated mirror descent with softmax-weighted smooth gradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    fft_size = 1 << (2 * n).bit_length()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    beta = beta_init
    stall = 0
    max_stall = 800
    rng = np.random.default_rng(42 + n)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_nest = (t - 1.0) / (t + 1.5)
        y = x + beta_nest * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute convolution and softmax weights
        b = fast_conv(y, fft_size)
        M = b.max()

        # Softmax weights (numerically stable)
        log_w = beta * (b - M)
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
            w_sum = 1.0
        w /= w_sum

        # Smooth gradient: h_j = sum_k w_k * y_{k-j} [valid]
        h = smooth_gradient(y, w, fft_size)

        # Mirror descent step
        eta = 2.0 * w_sum / (beta * (np.sqrt(t + 1.0) + 1.0))
        eta = min(eta, 2.0)

        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_conv(x, fft_size).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if t % 200 == 0:

            beta = beta_init * (beta_end / beta_init) ** progress

        if stall > max_stall:
            stall = 0
            max_stall = max(300, int(max_stall * 0.7))
            strategy = int(rng.integers(4))
            if strategy == 0:

            elif strategy == 1:

                idx = rng.choice(n, size=min(n // 3, 800), replace=False)
                x_r[idx] *= rng.uniform(0.6, 1.4, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            else:
                # Move mass toward low-convolution regions
                b_best = fast_conv(best_a, fft_size)
                low_regions = np.argsort(b_best)[:n // 4]
                x_r = best_a.copy()
                x_r[low_regions] *= 1.1
                x_r = np.maximum(x_r, 1e-15)
                x_r /= x_r.sum()
            x_r = np.maximum(x_r, 1e-15)
            x_r /= x_r.sum()
            x = x_r
            x_prev = x_r

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0

            print(f"  NSMD t={t} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, stall={stall}")

def lp_remez_step(a, fft_size, max_active, r, solver):
    """Single LP Remez step with given active set size and trust region."""
    n = len(a)
    b = fast_conv(a, fft_size)
    max_b = b.max()

    # Adaptive active set: include lags close to max
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(b[active])[::-1]][:max_active]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    # Trust region: allow a to deviate from current by factor (1+r)
    constraints.append(a_var <= a * (1.0 + r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return a.copy(), False

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), False

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False
    return a_new / s, True

def remez_lp_method(a0, time_limit, solver, max_active=60, r_init=0.15, verbose=False):
    """LP Remez with adaptive trust region and line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    fft_size = 1 << (2 * n).bit_length()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(777)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        a_cand, success = lp_remez_step(a, fft_size, max_active, r, solver)
        if not success:
            no_improve += 5
            r = max(0.01, r * 0.8)
            continue

        # Line search
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial, fft_size).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.4, r * 1.15)
                break

        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.85)
            if no_improve > 15:
                # Perturb to escape local structure
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve = 0

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, no_imp={no_improve}")

    return best_a, best_m

def fwf_smooth_direction(a0, time_limit, beta=100.0, verbose=False):
    """Frank-Wolfe using smooth direction from softmax weights."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    fft_size = 1 << (2 * n).bit_length()

        b = fast_conv(a, fft_size)
        M = b.max()

        # Softmax weights
        log_w = beta * (b - M)
        log_w -= log_w.max()
        w = np.exp(log_w)
        w /= w.sum()

        # Smooth gradient
        h = smooth_gradient(a, w, fft_size)

        # FWF direction: e_i where i = argmin h (minimize <h, g> over simplex = min h_i)
        i = int(np.argmin(h))

        # Line search
        best_t = 0.0
        best_trial_m = M

            a[i] += best_t

        if best_trial_m < best_m - 1e-16:
            best_m = best_trial_m
            best_a = a.copy()

        else:

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves and bisection."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    fft_size = 1 << (2 * n).bit_length()

    stall = 0
    max_stall = max(30, n // 80)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                m_trial = fast_conv(a_trial, fft_size).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':

    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Nesterov Smooth Mirror Descent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 1: NSMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_smooth_mirror_descent(norm_sum1_safe(best_seq), t1,
                                                   beta_init=30.0, beta_end=400.0, verbose=True)
        try_update(a_out, f"NSMD_n{n0}")

    # === Phase 2: LP Remez (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_lp_method(norm_sum1_safe(best_seq), t2, solver=solver,
                                    max_active=70, r_init=0.2, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")

    # === Phase 3: FWF smooth direction (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: FWF smooth n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = fwf_smooth_direction(norm_sum1_safe(best_seq), t3, beta=150.0, verbose=False)
        try_update(a_out, f"FWF_n{len(best_seq)}")

    # === Phase 4: NSMD polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: NSMD polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_smooth_mirror_descent(norm_sum1_safe(best_seq), t4,
                                                   beta_init=200.0, beta_end=600.0, verbose=True)
        try_update(a_out, f"NSMD_polish_n{len(best_seq)}")

    # === Phase 5: LP Remez polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: LP Remez polish (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = remez_lp_method(norm_sum1_safe(best_seq), t5, solver=solver,
                                    max_active=40, r_init=0.05, verbose=False)
        try_update(a_out, f"Remez_polish_{len(best_seq)}")

    # === Phase 6: Try different lengths (remaining - 60s for final polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        t6_total = remaining - 60
        print(f"\n=== Phase 6: Multi-length exploration (budget={t6_total:.0f}s) ===")

        # Try a few nearby lengths
        n_candidates = []
        for delta in [200, -200, 400, -400]:
            n_cand = n0 + delta
            if 1500 <= n_cand <= 4000 and n_cand != n0:
                n_candidates.append(n_cand)

        n_candidates = n_candidates[:3]  # Limit to 3 different lengths

        for n_cand in n_candidates:
            remaining = time_budget - (time.monotonic() - t_global) - 60
            if remaining < 40:
                break
            print(f"\n  Trying n={n_cand} (remaining={remaining:.0f}s)")
            a_alt = interpolate_sequence(norm_sum1_safe(best_seq), n_cand)

            t6a = min(remaining * 0.5, 100.0)
            if t6a > 20:
                a_out, _ = nesterov_smooth_mirror_descent(a_alt, t6a,
                                                           beta_init=50.0, beta_end=300.0, verbose=False)
                try_update(a_out, f"NSMD_n{n_cand}")

            remaining = time_budget - (time.monotonic() - t_global) - 60
            if remaining > 30:
                t6b = min(remaining * 0.4, 60.0)
                if t6b > 15:
                    a_out, _ = remez_lp_method(norm_sum1_safe(best_seq), t6b, solver=solver,
                                                max_active=50, r_init=0.1, verbose=False)
                    try_update(a_out, f"Remez_n{n_cand}")

    # === Phase 7: CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.5, 80.0)
    if t7 > 10:
        print(f"\n=== Phase 7: CD polish n={len(best_seq)} (budget={t7:.0f}s) ===")
        a_out, _ = coordinate_descent_polish(norm_sum1_safe(best_seq), t7, verbose=False, rng_seed=456)
        try_update(a_out, f"CD_{len(best_seq)}")

    # === Phase 8: Final LP Remez ===
    remaining = time_budget - (time.monotonic() - t_global) - 5
    if remaining > 15:
        print(f"\n=== Phase 8: Final LP Remez (budget={remaining:.0f}s) ===")
        a_out, _ = remez_lp_method(norm_sum1_safe(best_seq), remaining, solver=solver,
                                    max_active=50, r_init=0.03, verbose=False)
        try_update(a_out, f"Remez_final_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]

def norm_sum1_safe(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
