def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_excess_subgradient(a, ac, top_k=80, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    mean_ac = ac.mean()
    excess = np.maximum(ac - mean_ac, 0.0)
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_excess = np.maximum(ac[top_idx] - mean_ac, 1e-30)
    w = top_excess ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_l2_gradient(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = A * A
    grad_full = np.fft.irfft(C * A, fft_size)[:n]
    grad_full2 = np.fft.irfft(C * np.conj(A) * fft_size / fft_size, fft_size)[:n]
    grad = 4.0 * (grad_full + grad_full2)
    return np.real(grad)

    gamma = c0
    stall_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        grad = compute_l2_gradient(y)
        grad = np.maximum(grad, 0.0)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()

            stall_count = 0

def coordinate_descent(a0, time_limit, n_coords=50, verbose=False, rng_seed=42):

        ac = fast_autocorr(a)

        improved = False

        coord_order = np.argsort(a)[::-1][:n_coords]

            if time.monotonic() - t0 > time_limit:
                break
            i_int = int(i)
            a_i = a[i_int]
            if a_i < 1e-12:
                continue

            other = a.copy()
            other[i_int] = 0.0
            base_ac = fast_autocorr(other)

            base_at_i = np.zeros(len(ac))
            for k in range(len(ac)):
                j1 = k - i_int
                j2 = i_int - k
                val = 0.0
                if 0 <= j1 < n:
                    val += a[j1]
                if 0 <= j2 < n:
                    val += a[j2]
                base_at_i[k] = val

            full_ac = base_ac + a_i * base_at_i

            lo = 0.0
            hi = a_i * 5.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                trial_ac = base_ac + mid * base_at_i
                if trial_ac.max() <= full_ac.max() + 1e-15:
                    lo = mid
                else:
                    hi = mid
            optimal_val = lo

            a_trial = a.copy()
            a_trial[i_int] = optimal_val
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0

        for _ in range(n_coords):

                break
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst or a[src] < 1e-10:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)

                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t

                    best_m = M_trial

                    improved = True

                    break

            stall += 1

        if verbose and stall % 20 == 0:

            print(f"    CoordDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Transport ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=200):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp

    y = a.copy()
    x = a.copy()

    T_full = np.zeros((2 * n - 1, n))

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_power_law(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_sinc_like(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_multihump(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(a)


def make_asymmetric(n, skew=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-skew * x * (1 - x) * 4.0) * (1.0 + skew * x)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(aa, label):
        nonlocal best_seq, best_score
        if aa is None:
            return
        aa = np.maximum(np.asarray(aa, dtype=np.float64), 0.0)
        s = aa.sum()
        if s < 0.01:
            return
        aa_norm = aa / s
        sc = evaluate_safe(aa_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = aa_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(aa)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: Multi-init spectral L2 pre-processing
    print(f"\n=== Phase 1: Spectral L2 optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)

    if t_phase1 > 30:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_sinc_like(n0, bandwidth=4.0))
        labels.append("sinc4")

        inits.append(make_sinc_like(n0, bandwidth=6.0))
        labels.append("sinc6")

        inits.append(make_asymmetric(n0, skew=1.5))
        labels.append("asym1.5")

        inits.append(make_asymmetric(n0, skew=3.0))
        labels.append("asym3")

        inits.append(make_multihump(n0, n_humps=4, spread=0.15))
        labels.append("4hump")

        inits.append(make_cosine(n0, mode=1))
        labels.append("cos1")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  L2 '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = spectral_l2_optimize(init, t_avail, c0=3.0, verbose=False)
            try_update(a_out, f"l2_{lbl}")

    # Phase 2: Adaptive-beta AMD (main optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Adaptive AMD (n={n_best}) ===")
    t2 = min(remaining * 0.45, 700.0)

    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = adaptive_amd(a_in, t2, top_k=60, beta_start=3.0, beta_end=20.0,
                                c0=4.0, restart_every=5000, verbose=False)
        try_update(a_out, f"amd_adaptive_n{n_best}")

        # Second AMD run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.10, 120.0)
        if t2b > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = adaptive_amd(a_in, t2b, top_k=100, beta_start=5.0, beta_end=30.0,
                                    c0=5.0, restart_every=3000, verbose=False)
            try_update(a_out, f"amd_hard_n{n_best}")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coordinate descent (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t3, n_coords=50, verbose=False, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # Phase 4: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=300)
        try_update(a_out, f"lp_n{n_best}")

    # Phase 5: Mass transport
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Mass transport (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = mass_transport_descent(a_in, t5, verbose=True, rng_seed=777)
        try_update(a_out, f"transport_n{n_best}")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length ({n_best}) ===")
    if remaining > 60:
        for delta in [500, -500, 1000]:
            new_n = n_best + delta
            if new_n < 1000 or new_n > 8000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={new_n}...")
            a_long = interpolate_sequence(best_seq, new_n)
            try_update(a_long, f"interp_n{new_n}")

            t6 = min(remaining * 0.3, 100.0)
            if t6 > 15:
                a_out, _ = adaptive_amd(a_long, t6, top_k=50, beta_start=3.0,
                                        beta_end=15.0, c0=4.0, restart_every=4000, verbose=False)
                try_update(a_out, f"amd_n{new_n}")

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = adaptive_amd(a_in, t7, top_k=80, beta_start=10.0, beta_end=40.0,
                                    c0=3.0, restart_every=8000, verbose=False)
            try_update(a_out, f"amd_polish_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.4, 60.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_descent(a_in, t7b, n_coords=30, verbose=False, rng_seed=123)
                try_update(a_out, f"coord_final_n{n_best}")

    # Phase 8: One more AMD with very hard beta if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.6, 60.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = adaptive_amd(a_in, t8, top_k=120, beta_start=15.0, beta_end=60.0,
                                    c0=2.5, restart_every=10000, verbose=False)
            try_update(a_out, f"amd_veryhard_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
