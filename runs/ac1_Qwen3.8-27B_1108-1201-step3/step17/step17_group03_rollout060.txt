For the line search, I can use the fact that f(α) = max_k ((1-α)² conv(x,x)[k] + 2α(1-α) conv(x,v)[k] + α² conv(v,v)[k])

This is the max of a set of quadratics in α. The minimum of this can be found by checking the breakpoints where different quadratics are maximized.

A simpler approach: use bisection on the derivative, or just evaluate at a grid of α values (e.g., 20 points) and take the best.

def fast_conv(a, b=None):
    """Compute full convolution a*b (or a*a) using FFT."""
    if b is None:
        b = a
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_conv_aa(a):
    """Compute conv(a,a) using FFT (optimized for same input)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, ensure non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = fast_conv_aa(a)
    return 2.0 * n * c.max()


def lp_direction(a, M, top_k=2000, solver=None):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for top-k active k, b >= 0.
    Returns the normalized direction b/sum(b).
    """
    n = len(a)
    c = fast_conv(aa)  # autocorrelation of a
    # Select top-k constraints (those with highest conv(a,a)[k])
    k_idx = np.argsort(c)[::-1][:min(top_k, 2*n-1)]
    
    # Build constraint matrix rows
    # conv(a,b)[k] = sum_i a[i] * b[k-i] for valid i
    # This is a row of the constraint matrix
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in k_idx:
        # conv(a,b)[k] = sum_{i} a[i] * b[k-i] where 0<=i<n and 0<=k-i<n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        # The row is: for each j (index of b), coefficient is a[k-j] if 0<=k-j<n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        # coeff[j] = a[k-j]
        row = np.zeros(n)
        for j in range(i_min, i_max + 1):
            row[j] = a[k - j]
        constraints.append(row @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    if solver is None:
        solver = get_solver()
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum


def line_search(x, v, c_x, c_xv, c_vv, n_evals=20):
    """
    Find best alpha in [0,1] to minimize max of conv((1-a)x+av, (1-a)x+av).
    c_x = conv(x,x), c_xv = conv(x,v), c_vv = conv(v,v)
    """
    best_alpha = 0.0
    best_obj = float('inf')
    
    for alpha in np.linspace(0, 1, n_evals):
        # conv((1-a)x+av, (1-a)x+av) = (1-a)^2 * c_x + 2a(1-a) * c_xv + a^2 * c_vv
        obj = (1-alpha)**2 * c_x + 2*alpha*(1-alpha) * c_xv + alpha**2 * c_vv
        m = obj.max()
        if m < best_obj:
            best_obj = m
            best_alpha = alpha
    
    # Refine around best
    if 0 < best_alpha < 1:
        lo = max(0, best_alpha - 0.05)
        hi = min(1, best_alpha + 0.05)
        for alpha in np.linspace(lo, hi, 20):
            obj = (1-alpha)**2 * c_x + 2*alpha*(1-alpha) * c_xv + alpha**2 * c_vv
            m = obj.max()
            if m < best_obj:
                best_obj = m
                best_alpha = alpha
    
    return best_alpha, best_obj


def fw_optimize(a0, time_limit, solver, verbose=True, top_k=2000, 
                min_improve=1e-10, max_no_improve=50):
    """
    LP-based Frank-Wolfe with exact line search.
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_conv_aa(x).max()
    t0 = time.monotonic()
    
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        c_x = fast_conv_aa(x)
        M = c_x.max()
        
        if M < best_m - min_improve:
            best_m = M
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Get LP direction
        v = lp_direction(x, M, top_k=top_k, solver=solver)
        if v is None:
            # Try with fewer constraints
            v = lp_direction(x, M, top_k=min(500, top_k//2), solver=solver)
            if v is None:
                break
        
        # Compute cross terms for line search
        c_xv = fast_conv(x, v)
        c_vv = fast_conv_aa(v)
        
        # Line search
        alpha, obj_val = line_search(x, v, c_x, c_xv, c_vv)
        
        if alpha < 1e-6:
            break
        
        # Update
        x_new = (1 - alpha) * x + alpha * v
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            break
        x = x_new / s
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            M_new = fast_conv_aa(x).max()
            J = 2.0 * n * M_new
            print(f"    FW iter={iteration} ({elapsed:.0f}s): J={J:.6f}, M={M_new:.8f}, alpha={alpha:.4f}")
    
    return best_x, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """
    Coordinate descent that targets autocorrelation peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv_aa(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv_aa(a)
        M = c.max()
        
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find the peak
        peak_k = int(np.argmax(c))
        
        # Find contributing pairs
        i_min = max(0, peak_k - n + 1)
        i_max = min(n - 1, peak_k)
        pairs = []
        for i in range(i_min, i_max + 1):
            j = peak_k - i
            pairs.append((i, j, a[i] * a[j]))
        
        if not pairs:
            break
        
        # Sort by contribution (descending)
        pairs.sort(key=lambda x: x[2], reverse=True)
        
        improved = False
        for i, j, contrib in pairs[:10]:
            if time.monotonic() - t0 > time_limit:
                    break
            
            if contrib < 1e-15:
                continue
            
            # Try moving mass from the larger of a[i], a[j] to a random position
            src = i if a[i] >= a[j] else j
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            best_alpha = 0
            best_M = M
            
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try several destinations
                for _ in range(20):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv_aa(a_trial).max()
                    if M_trial < best_M:
                        best_M = M_trial
                        best_alpha = (src, dst, amt)
            
            if best_alpha is not None and best_alpha != 0:
                src_i, dst_i, amt = best_alpha
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                improved = True
                break
        
        if not improved:
            # Random perturbation
            src = int(rng.integers(n))
            if a[src] > 1e-10:
                dst = int(rng.integers(n))
                if dst != src:
                    amt = a[src] * 0.01
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    CoordPolish ({elapsed:.0f}s): J={J:.6f}, best_M={best_m:.8f}")
    
    return best_a, best_m


def spectral_perturb(a, strength=0.01, rng=None):
    """Perturb in frequency domain."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = np.abs(A) * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv_aa(a):
    """Compute full convolution conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute full convolution conv(a,b) using FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, ensure non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:

def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = fast_conv_aa(a)
    return 2.0 * n * c.max()


def build_constraint_rows(a, k_indices, n):
    """Build the constraint matrix rows for conv(a,b)[k] <= M."""
    n_rows = len(k_indices)
    rows = np.zeros((n_rows, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        j_range = np.arange(j_min, j_max + 1)
        rows[idx, j_range] = a[k - j_range]
    return rows


def lp_direction(a, M, top_k, solver):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for top-k k, b >= 0.
    Returns (normalized_direction, optimal_value) or (None, None).
    """
    n = len(a)
    c_aa = fast_conv_aa(a)
    k_idx = np.argsort(c_aa)[::-1][:min(top_k, 2 * n - 1)]
    
    rows = build_constraint_rows(a, k_idx, n)
    # Remove zero rows
    nonzero = np.any(rows != 0, axis=1)
    rows = rows[nonzero]
    if rows.shape[0] == 0:
        return None, None
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [rows @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None
    
    return b_val / b_sum, b_sum


def line_search_quadratic(c_x, c_xv, c_vv, coarse=30, fine=30):
    """
    Find best alpha in [0,1] minimizing max_k [(1-a)^2 c_x[k] + 2a(1-a) c_xv[k] + a^2 c_vv[k]].
    """
    best_alpha = 0.0
    best_obj = float('inf')
    
    alphas = np.linspace(0, 1, coarse)
    for alpha in alphas:
        obj = (1 - alpha) ** 2 * c_x + 2 * alpha * (1 - alpha) * c_xv + alpha ** 2 * c_vv
        m = obj.max()
        if m < best_obj:
            best_obj = m
            best_alpha = alpha
    
    # Fine search around best
    if 0 < best_alpha < 1:
        lo = max(0.0, best_alpha - 1.0 / coarse)
        hi = min(1.0, best_alpha + 1.0 / coarse)
        alphas_fine = np.linspace(lo, hi, fine)
        for alpha in alphas_fine:
            obj = (1 - alpha) ** 2 * c_x + 2 * alpha * (1 - alpha) * c_xv + alpha ** 2 * c_vv
            m = obj.max()
            if m < best_obj:
                best_obj = m
                best_alpha = alpha
    
    return best_alpha, best_obj


def fw_optimize(a0, time_limit, solver, verbose=True, top_k=2000, max_iter=200):
    """

    max_no_improve = 80
    iteration = 0
    
    while time.monotonic() - t0 < time_limit and iteration < max_iter:
        iteration += 1
        c_x = fast_conv_aa(x)
        M = c_x.max()
        
        if M < best_m - 1e-12:
            best_m = M
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Get LP direction
        v, lp_val = lp_direction(x, M, top_k, solver)
        if v is None:
            v, lp_val = lp_direction(x, M, max(200, top_k // 3), solver)
            if v is None:
                break
        
        # If LP value is close to 1, we're near a local optimum
        if lp_val is not None and lp_val < 1.0 + 1e-6:

        # Line search
        c_xv = fast_conv_ab(x, v)
        c_vv = fast_conv_aa(v)
        alpha, obj_val = line_search_quadratic(c_x, c_xv, c_vv)

        if s < 1e-15:

            best_x = x.copy()

def mirror_descent(a0, time_limit, verbose=True, eta0=0.5, decay=0.9995,
                   top_k_peaks=10, restart_every=3000):
    """
    Mirror descent in log-space with analytical subgradient.
    Subgradient of max_k conv(a,a)[k] at a is 2*a[k*-i] (reversed/shifted).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv_aa(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    # Log-space parameterization
    w = np.log(np.maximum(a, 1e-30))
    eta = eta0
    stall = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Recover a from w
        a = np.exp(w - w.max())
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        c = fast_conv_aa(a)
        M = c.max()
        
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            # Perturb and restart
            pert = rng.normal(0, 0.01, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            w = np.log(a_r)
            stall = 0
            eta = eta0
            continue
        
        # Compute subgradient: for top peaks, weighted combination
        peak_indices = np.argsort(c)[::-1][:top_k_peaks]
        peak_vals = c[peak_indices]
        weights = peak_vals ** 8  # Heavy weighting on top peaks
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(top_k_peaks) / top_k_peaks
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(weights, peak_indices):
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * a[j]
        
        # Mirror descent step in log-space
        # w_new = w - eta * grad / a (then renormalize)
        ratio = grad / np.maximum(a, 1e-30)
        w = w - eta * ratio
        w = w - w.max()  # numerical stability
        
        eta *= decay
        
        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={iteration} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):

    """
    n = len(a0)
    a = norm_sum1(a0)

    best_m = fast_conv_aa(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    fractions = [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv_aa(a)
        M = c.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        top_peaks = np.argsort(c)[::-1][:5]
        
        improved = False
        for peak_k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break
            
            peak_k = int(peak_k)
            i_min = max(0, peak_k - n + 1)
            i_max = min(n - 1, peak_k)
            
            # Compute contributions
            contribs = []
            for i in range(i_min, i_max + 1):
