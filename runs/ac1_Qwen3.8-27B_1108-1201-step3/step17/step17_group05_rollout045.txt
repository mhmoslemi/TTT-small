```python
def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_grad_at_peak(a, top_k=5, beta=20.0):
    """Compute subgradient focusing on top-k peak lags.
    Returns (gradient, max_value)."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    
    # Find top-k lags
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    
    # Soft weights
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_top = w_top / w_top.sum()
    
    # Build gradient
    # grad[j] = 2 * sum over top lags k of w_k * a[k-j] (where valid)
    grad = np.zeros(n)
    for wi, ki in zip(w_top, top_idx):
        ki = int(ki)
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            grad[j_start:j_end+1] += 2.0 * wi * a[ki - np.arange(j_start, j_end+1)]
    
    return grad, M
```

```python
def targeted_redistribute(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find peak lags
        peak_lags = np.where(ac > 0.9 * M)[0]
        
        # For each peak lag, find the position with maximum contribution
        # and move some mass away
        improved = False
        
        for k in peak_lags[:5]:
            k = int(k)
            # positions j where a[j] * a[k-j] is the autocorrelation contribution
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k - js]
            idx = np.argmax(contrib)
            src = js[idx]
            
            # Find destination: position with low contribution to all peaks
            # and far from src
            dst_scores = np.zeros(n)
            for k2 in peak_lags[:3]:
                k2 = int(k2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                for d in range(s2, e2 + 1):
                    partner = k2 - d
                    if 0 <= partner < n:
                        dst_scores[d] += a[partner]
            
            # Add distance penalty (prefer far positions)
            dist = np.abs(np.arange(n) - src).astype(float) / n
            dst_scores = dst_scores / (1.0 + dist)
            
            # Find best destination
            dst = np.argmin(dst_scores)
            if dst == src:
                continue
            
            # Try moving mass
            for frac in [0.01, 0.005, 0.02, 0.003, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_new = a.copy()
                a_new[src] -= amt
                a_new[dst] += amt
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new / s
                m_new = fast_autocorr(a_new).max()
                if m_new < best_m - 1e-14:
                    a = a_new
                    best_m = m_new
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:

                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
    
    return best_a, best_m
```

Let me also think about a **frequency-domain** initialization:
- Choose a target autocorrelation shape (e.g., flat with a slight dip at the center)
- The power spectrum is `|FFT(a)|^2`
- If I want `conv(a,a) ≈ target`, then `FFT(conv(a,a)) ≈ FFT(target)`
- And `FFT(conv(a,a)) ≈ FFT(a) * conj(FFT(a)) = |FFT(a)|^2` (for the circular case)
- So `|FFT(a)| ≈ sqrt(FFT(target))`
- Then `a = IFFT(exp(i*phase) * sqrt(FFT(target)))` where phase is random

```python
def freq_domain_init(n, target_peak_ratio=0.75, seed=0):
    """Initialize using frequency domain shaping."""
    rng = np.random.default_rng(seed)
    
    # Target autocorrelation: flat with ratio to uniform's peak
    # Uniform autocorrelation is triangular: (n - |k - (n-1)|) / n^2
    # We want peak = target_peak_ratio / n
    # So target[k] = target_peak_ratio / n for all k (roughly)
    # But we need sum(target) = 1 (since sum(conv(a,a)) = (sum a)^2 = 1)
    # So target[k] = 1/(2n-1) for all k (perfectly flat)
    
    # Let's aim for a "damped" version
    center = n - 1
    k = np.arange(2 * n - 1)
    # Triangular shape (uniform's autocorrelation)
    tri = (n - np.abs(k - center)).clip(0) / n**2
    # Flat target
    flat = np.ones(2 * n - 1) / (2 * n - 1)
    # Blend
    target = 0.5 * tri + 0.5 * flat
    
    # Normalize target so sum = 1
    target = target / target.sum()
    
    # Get power spectrum
    fft_target = np.fft.rfft(target, 2 * n)
    # Power spectrum of a
    power = np.real(fft_target)
    power = np.maximum(power, 0)  # ensure non-negative
    # Magnitude of FFT(a)
    mag = np.sqrt(power / n)  # rough scaling
    
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, len(mag))
    phase[0] = 0  # ensure real
    
    # Build FFT(a)
    A = mag * np.exp(1j * phase)
    
    # Inverse FFT
    a = np.fft.irfft(A, 2 * n)[:n]
    
    # Clip and normalize
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    
    return a
```

Let me think about a **pair-based optimization**:
- The autocorrelation at lag k is `sum_j a[j] * a[k-j]`
- If we fix all other positions, the autocorrelation is a function of the pair (a[j], a[k-j])
- We can optimize the pair: given `a[j] + a[k-j] = c` (fixed sum), the product `a[j] * a[k-j]` is maximized when `a[j] = a[k-j] = c/2`
- To minimize the product (and thus the autocorrelation), we should make `a[j]` and `a[k-j]` as unequal as possible

This suggests: for each peak lag k, make the contributing pairs as "unequal" as possible.

This is an interesting direction! Let me think about how to implement it:

For each peak lag k, the pairs are (j, k-j) for valid j. To minimize `sum a[j]*a[k-j]`:
- For each pair, make one entry large and the other small
- But entries appear in multiple pairs (for different lags), so there's a tradeoff

This is like a constraint satisfaction problem. A simple greedy approach:
1. Find the peak lag k*
2. For the pair (j, k*-j) with the largest product, try to make them unequal
3. Move mass from the smaller to the larger (or vice versa) - actually, to reduce the product, move mass FROM the larger TO the smaller? No, that makes them more equal.
   
Wait: `x * y` with `x + y = c` is maximized at `x = y = c/2` and minimized at `x = 0, y = c` or `x = c, y = 0`.
