"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Classical window functions as initializations (Hann, Blackman-Harris, etc.)
2. FFT-accelerated gradient computation via cross-correlation
3. L-BFGS optimization on log-parameterized simplex with progressive tau smoothing
4. Multi-length search
5. Final coordinate descent polish
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2. Assumes a is nonnegative."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * conv.max()

def compute_gradient_from_weights(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation.
    a has length n, w has length 2n-1 (weights on each convolution lag).
    Returns gradient g of length n where g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j].
    """
    n = len(a)
    conv_len = 2 * n - 1
    # Need N >= n + conv_len - 1 for linear cross-correlation
    N = 1 << (n + conv_len).bit_length()
    
    a_pad = np.zeros(N)
    a_pad[:n] = a
    
    w_pad = np.zeros(N)
    w_pad[:conv_len] = w
    
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    
    # Cross-correlation: (a ⊛ w)[i] = sum_j a[j] * w[i+j]
    # = ifft(conj(fft(a)) * fft(w))[i]
    g_half = np.fft.irfft(np.conj(A) * W, N)[:n]
    
    return 2.0 * g_half

def soft_max_weights(conv, tau):
    """Compute soft-max weights on convolution values."""
    c_max = conv.max()
    if tau < 1e-12:
        # Hard max
        w = np.zeros_like(conv)
        w[conv == c_max] = 1.0 / np.sum(conv == c_max)
        return w
    w = np.exp((conv - c_max) / tau)
    w = w / w.sum()
    return w

def smooth_max(conv, tau):
    """Smooth approximation of max using log-sum-exp."""
    c_max = conv.max()
    return c_max + tau * np.log(np.sum(np.exp((conv - c_max) / tau)))

def objective_and_grad(x, tau):
    """Objective and gradient for log-parameterized optimization.
    x is the log-parameter, a = softmax(x) (or more precisely, a = exp(x)/sum(exp(x))).
    We optimize 2n * smooth_max(conv(a), tau).
    """
    n = len(x)
    # Numerically stable softmax
    x_shift = x - x.max()
    e = np.exp(x_shift)
    a = e / e.sum()
    
    conv = fast_conv(a)
    sm = smooth_max(conv, tau)
    obj = 2.0 * n * sm
    
    # Gradient of sm w.r.t. a
    w = soft_max_weights(conv, tau)
    g_a = compute_gradient_from_weights(a, w)
    
    # Chain rule: da_i/dx_j = a_i*(delta_ij - a_j)
    # d(obj)/dx_j = sum_i d(obj)/da_i * da_i/dx_j
    # = sum_i (2n * g_a[i]) * a_i * (delta_ij - a_j)
    # = 2n * (a_j * sum_i g_a[i]*a_i - g_a[j] * ... )
    # Wait: sum_i g_a[i] * a_i * (delta_ij - a_j) = g_a[j]*a_j - a_j*sum_i g_a[i]*a_i
    
    dot = np.dot(g_a, a)
    g_x = 2.0 * n * a * (dot - g_a)
    
    return obj, g_x

def make_hann(n):
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s

def optimize_lbfgs(a0, time_budget, n_max_iter=200, tau_schedule=None):
    """Optimize using L-BFGS on log-parameterized simplex with progressive tau."""
    n = len(a0)
    x0 = np.log(a0 + 1e-30)
    
    t_start = time.monotonic()
    best_a = a0.copy()
    best_J = eval_J(a0)
    
    if tau_schedule is None:
        tau_schedule = [0.5, 0.1, 0.01, 0.001, 0.0001]
    
    for tau in tau_schedule:
        if time.monotonic() - t_start > time_budget * 0.9:
            break
        
        def obj(x):
            o, _ = objective_and_grad(x, tau)
            return o
        
        def grad(x):
            _, g = objective_and_grad(x, tau)
            return g
        
        try:
            result = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                            options={'maxiter': n_max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
            
            # Convert back
            x_opt = result.x
            x_shift = x_opt - x_opt.max()
            e = np.exp(x_shift)
            a_new = e / e.sum()
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            J_new = eval_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                print(f"    L-BFGS tau={tau}: J={J_new:.6f} (from {best_J:.6f})")
            else:
                print(f"    L-BFGS tau={tau}: J={J_new:.6f} (no improvement, best={best_J:.6f})")
            
            # Update x0 for next tau
            x0 = np.log(best_a + 1e-30)
            
        except Exception as e:
            print(f"    L-BFGS error at tau={tau}: {e}")
            x0 = np.log(best_a + 1e-30)
    
    return best_a, best_J

def coordinate_descent_polish(a0, time_budget, seed=42):
    """Final polish via coordinate descent."""
    n = len(a0)
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_budget and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        
        # Find the lag with max value
        k_max = int(np.argmax(conv))
        
        # Compute "pressure" at each position
        # pressure[i] = 2 * a[k_max - i] if 0 <= k_max - i < n
        pressure = np.zeros(n)
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            pressure[idx] = 2.0 * a[k_max - idx]
        
        # Find the position with highest pressure and significant mass
        src_score = pressure * a
        src = int(np.argmax(src_score))
        
        if a[src] < 1e-15:
            stall += 10
            continue
        
        # Find destination: minimize increase in max conv
        # A position j increases conv[k] by 2*a[k-j] for relevant k
        # We want to minimize the max over k of the increase
        dest_score = np.zeros(n)
        # For the current max lag:
        j_min = max(0, k_max - n + 1)
        j_max = min(n - 1, k_max)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            dest_score[idx] = 2.0 * a[k_max - idx]
        
        # Also consider second and third max lags
        top_lags = np.argsort(conv)[::-1][:5]
        for k2 in top_lags[1:]:
            k2_i = int(k2)
            j2_min = max(0, k2_i - n + 1)
            j2_max = min(n - 1, k2_i)
            if j2_min <= j2_max:
                idx = np.arange(j2_min, j2_max + 1)
                dest_score[idx] += 0.5 * a[k2_i - idx]
        
        dest_score[src] = np.inf
        dst = int(np.argmin(dest_score))
        
        # Transfer mass
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4, 0.7]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-15:
                continue
            a_t /= s
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-12:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.005, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        s = a.sum()
                        if s > 1e-15:
                            a /= s
                            J_new = eval_J(a)
                            if J_new < best_J:
                                best_J = J_new
                                best_a = a.copy()
                                stall = 0
                                break
        
        if time.monotonic() - t0 > time_budget:
            break
    
    return best_a, best_J

def adam_optimize(a0, time_budget, lr=0.1, tau=0.01, beta1=0.9, beta2=0.999):
    """Adam optimization with FFT gradient."""
    n = len(a0)
    x = np.log(a0 + 1e-30)  # log parameterization
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0
    
    best_a = a0.copy()
    best_J = eval_J(a0)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        t += 1
        obj, g = objective_and_grad(x, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Convert back and check
        x_shift = x - x.max()
        e = np.exp(x_shift)
        a_new = e / e.sum()

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Classical window functions (Hann, Blackman-Harris, etc.) as initializations
2. FFT-accelerated gradient via cross-correlation (O(n log n) per step)

import numpy as np

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2. Assumes a is nonnegative."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_conv(a_norm)
    return float(2.0 * n * conv.max())


def compute_gradient_from_weights(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation.
    a has length n, w has length 2n-1.
    Returns g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j].
    """
    n = len(a)
    conv_len = 2 * n - 1
    N = 1 << (2 * n).bit_length()

    a_pad = np.zeros(N)
    a_pad[:n] = a

    w_pad = np.zeros(N)
    w_pad[:conv_len] = w

    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)

    g_half = np.fft.irfft(np.conj(A) * W, N)[:n]
    return 2.0 * g_half


def soft_max_weights(conv, tau):
    """Compute soft-max weights on convolution values."""
    c_max = conv.max()
    if tau < 1e-10:
        w = np.zeros_like(conv)
        mask = (conv == c_max)
        w[mask] = 1.0 / np.sum(mask)
        return w
    w = np.exp((conv - c_max) / tau)
    w = w / w.sum()
    return w


def smooth_max(conv, tau):
    """Smooth approximation of max using log-sum-exp."""
    c_max = conv.max()
    if tau < 1e-10:
        return c_max
    return c_max + tau * np.log(np.sum(np.exp((conv - c_max) / tau)))


def objective_and_grad(x, tau):
    """Objective and gradient for log-parameterized optimization on softmax simplex."""
    n = len(x)
    x_shift = x - x.max()
    e = np.exp(x_shift)
    a = e / e.sum()

    conv = fast_conv(a)
    sm = smooth_max(conv, tau)
    obj = 2.0 * n * sm

    w = soft_max_weights(conv, tau)
    g_a = compute_gradient_from_weights(a, w)

    dot = np.dot(g_a, a)
    g_x = 2.0 * n * a * (dot - g_a)

    return obj, g_x

def make_flat_cos(n, beta):
    """Cosine power window."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    t = np.pi * (x - center) / (n - 1)
    a = np.cos(t) ** beta
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    a = np.exp(-((x - 0.5) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s

    a = np.exp(-((x - p1) ** 2) / (2.0 * width ** 2)) + np.exp(-((x - p2) ** 2) / (2.0 * width ** 2))
    s = a.sum()
    return a / s
