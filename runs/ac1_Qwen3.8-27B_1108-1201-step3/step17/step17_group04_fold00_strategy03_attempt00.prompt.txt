def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    return a / s

def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    """Compute subgradient of max(conv(a,a))."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def peak_reduction_moves(a0, time_limit, verbose=False, seed=77):
    """
    Targeted coordinate moves: identify peak contributors and relocate mass to safe zones.
    Uses sensitivity analysis to find optimal transfer destinations.
    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top peaks
        n_peaks = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity: for each position i, how much does it contribute to peaks?
        sensitivity = np.zeros(n)
        for k in top_lags[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            # Add to sensitivity of both idx and j_idx
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc

        # Find high-sensitivity positions with substantial mass
        high_sens = np.where((sensitivity > sensitivity.mean() + sensitivity.std()) & (a > 0.001))[0]
        low_sens = np.where((sensitivity < sensitivity.mean() - 0.5 * sensitivity.std()) | (sensitivity == 0))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False

        for _ in range(20):  # Try up to 20 moves per outer iteration
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            # Try multiple transfer fractions
            fracs = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]

                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0
            else:
                # Try random move as escape
                if stall > 100:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.05)

                        a[i] -= amt
                        a[j] += amt

                        J_new = compute_J(a)
                        if J_new < best_J:

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K=500):
    """Improved LP Frank-Wolfe with more constraints and better step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K constraints (most binding)
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def block_optimize(a0, time_limit, verbose=False, seed=200):
    """
    Block coordinate optimization: pick a block of consecutive coordinates,
    optimize them via local quadratic approximation or small LP.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    block_sizes = [4, 8, 16, 32, 64]

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:5]

        # Pick a block near a peak-contributing region
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose block size
            bs = block_sizes[rng.integers(len(block_sizes))]
            # Choose block start (biased toward peak regions)
            if rng.random() < 0.6 and len(top_lags) > 0:
                k = int(top_lags[0])
                center = k // 2
                start = max(0, min(n - bs, center + rng.integers(-n//4, n//4 - bs)))
            else:
                start = int(rng.integers(max(1, n - bs)))

            end = min(n, start + bs)
            if end - start < 2:
                continue

            # Try adjusting the block
            block = a[start:end].copy()
            block_sum = block.sum()
            block_sum_rest = a.sum() - block_sum

            if block_sum < 1e-10:
                continue

            # Local optimization: try to redistribute mass within block
            # to reduce peak autocorrelation
            best_block = block.copy()
            best_local_J = compute_J(a)

            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                # Random perturbation of block
                perturb = rng.normal(0, 0.1 * block_sum, end - start)
                new_block = block + perturb
                new_block = np.maximum(new_block, 0)
                nb_sum = new_block.sum()
                if nb_sum < 1e-10:
                    continue
                # Scale to preserve total sum
                new_block = new_block * (block_sum / nb_sum)

                a_trial = a.copy()
                a_trial[start:end] = new_block
                J_trial = compute_J(a_trial)
                if J_trial < best_local_J - 1e-14:
                    best_local_J = J_trial
                    best_block = new_block.copy()

            if best_local_J < best_J - 1e-14:
                a[start:end] = best_block
                best_J = best_local_J
                best_a = a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    BlockOpt ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Create a symmetric initialization (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)
