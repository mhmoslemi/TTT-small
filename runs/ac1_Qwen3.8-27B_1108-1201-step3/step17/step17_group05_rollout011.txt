def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_mv(a, v):
    """Compute conv(a, v) using FFT. Efficient for computing T_k @ v."""
    n = len(a)
    nv = len(v)
    size = n + nv - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    V = np.fft.rfft(np.asarray(v, dtype=np.float64), fft_size)
    return np.fft.irfft(A * V, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, top_k=20, soft_beta=10.0):
    """
    Compute (soft) subgradient of max_k conv(a,a)[k].
    Uses exp-softmax over top-k peaks.
    Returns gradient vector g of length n.
    """
    n = len(a)
    ac = fast_autocorr(a)
    # Select top-k lags
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Soft-max weights
    shifted = top_vals - top_vals.max()
    w = np.exp(soft_beta * shifted)
    w = w / w.sum()
    # Build weight vector over all lags
    w_full = np.zeros(len(ac))
    for i, idx in enumerate(top_idx):
        w_full[int(idx)] = w[i]
    # Gradient: g[j] = 2 * sum_k w_full[k] * a[k-j] (for valid k-j)
    # This is 2 * conv(a, w_full) appropriately indexed
    # conv(a, w_full)[j + k] involves a[j] * w_full[k]
    # We want: g[j] = 2 * sum_k w_full[k] * a[k-j] for 0 <= k-j < n
    # = 2 * sum_k w_full[k] * a[k-j] * 1[0<=k-j<n]
    # Let m = k-j, so k = m+j: g[j] = 2 * sum_m w_full[m+j] * a[m] * 1[0<=m<n]
    # = 2 * (cross-correlation of a and w_full, shifted)
    # Actually: g[j] = 2 * sum_m a[m] * w_full[m+j]
    # This is 2 * (a cross-correlated with w_full) at lag j
    # = 2 * conv(a, w_full_reversed)[n-1+j] ... let me just use direct FFT
    
    # Simpler: use the formula g[j] = 2 * sum_{k in support} w[k] * a[k-j] * 1[0<=k-j<n]
    # For each j: g[j] = 2 * sum over k where 0 <= k-j < n of w[k] * a[k-j]
    # = 2 * sum over m (=k-j) where 0<=m<n of w[j+m] * a[m]
    # = 2 * <a, w[j:j+n]> (inner product of a with the slice of w starting at j)
    
    # This is a cross-correlation. Can be computed as:
    # g[j] = 2 * (a * w_full_flipped)[(n-1) + j] ... 
    
    # Actually let me just use: g = 2 * ifft(fft(a) * conj(fft(w_full_padded))) ... 
    
    # Simplest correct approach:
    # g[j] = 2 * sum_{m=0}^{n-1} a[m] * w_full[j+m] (if j+m < len(w_full))
    # This is the cross-correlation of a and w_full at lag j: (a ⋆ w_full)[j]
    # = conv(a, w_full_reversed)[len(w_full_reversed) - 1 + j]
    
    # Let w_rev = w_full[::-1], length = 2n-1
    # conv(a, w_rev)[j + (2n-2) - (n-1)] ... this is getting confusing.
    
    # Let me just use direct FFT approach:
    # g[j] = 2 * sum_m a[m] * w_full[j+m] for m = 0..n-1, j+m valid
    # = 2 * (circular cross-correlation if we pad properly)
    
    # Use: define v = w_full reversed, length 2n-1
    # Then conv(a, v) has length (2n-1) + n - 1 = 3n - 2
    # conv(a,v)[j + (2n-2)] = sum_m a[m] * v[2n-2-j] ... no this is wrong.
    
    # OK let me just do it the straightforward way with the relation:
    # g[j] = 2 * sum_k w_full[k] * a[k-j] * 1[0 <= k-j < n]
    # For a given j, the valid k range is: max(0, j) <= k <= min(2n-2, j+n-1)
    # And a[k-j] where m = k-j ranges from max(0,j) - j = max(0) = 0 to min(2n-2, j+n-1) - j
    
    # This is exactly: g[j] = 2 * (a cross-correlated with w_full)[j]
    # where cross-correlation at lag j means sum_m a[m] * w_full[m+j]
    
    # In terms of convolution: (a ⋆ w)[j] = sum_m a[m] * w[m+j] = conv(a, w_rev)[len(w_rev)-1+j]
    # where w_rev = w[::-1]
    
    w_rev = w_full[::-1]  # length 2n-1
    total_len = n + len(w_rev) - 1  # = n + 2n-1 - 1 = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_rev_pad = np.zeros(fft_size)
    w_rev_pad[:len(w_rev)] = w_rev
    conv_result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_rev_pad), fft_size)
    # conv_result[j + (2n-2)] = sum_m a[m] * w_rev[2n-2-j+m]...

    # g[j] = 2 * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w_full[k] * a[k-j]
    
    # This is a sliding dot product. For efficiency, use FFT:
    # g[j] = 2 * sum_m a[m] * w_full[j+m] (where m = k-j, so k = j+m)
    # Valid m: 0 <= m < n and 0 <= j+m < 2n-1
    # So m ranges from max(0, -(j)) = 0 to min(n-1, 2n-2-j)
    
    # g[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} a[m] * w_full[j+m]

    # = 2 * ifft(fft(a) * conj(fft(w_full))) ... with appropriate padding
    
    # Actually the simplest: use np.correlate or just FFT
    # correlation[j] = sum_m a[m] * w_full[j+m]
    # This is the same as conv(a, w_full_reversed) at index (len(w_full_reversed)-1+j)
    
    # Let me just compute it directly for now (it's O(n * top_k) which is fine for n=3200, top_k=20)
    g = np.zeros(n)
    for k_idx in top_idx:
        k_i = int(k_idx)
        # Contribution from lag k_i: 2 * w[k_i] * a[k_i - j] for valid j
        # a[k_i - j] is valid when 0 <= k_i - j < n, i.e., k_i - (n-1) <= j <= k_i
        j_start = max(0, k_i - n + 1)
        j_end = min(n - 1, k_i)
        if j_start <= j_end:
            g[j_start:j_end+1] += 2.0 * w_full[k_i] * a[k_i - np.arange(j_start, j_end+1)]
    
    return g, ac.max()


def lp_frank_wolfe(a0, time_limit, solver, top_k=300, verbose=True, seed=42):
    """
    LP-based Frank-Wolfe with adaptive top-K constraint selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Nesterov momentum
    y = a.copy()
    x_mom = a.copy()
    
    iterations = 0
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iterations += 1
        t = iterations
        
        # Compute current autocorrelation and select constraints
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select top-K constraints (most binding)
        K = min(top_k, 2 * n - 1)
        # Use a mix: top peaks + some random for diversity
        top_idx = np.argsort(ac)[::-1][:K]
        if rng.random() < 0.1 and K > 50:
            # Occasionally add random constraints for diversity
            random_idx = rng.integers(0, 2 * n - 1, size=K // 10)
            top_idx = np.concatenate([top_idx, random_idx])
            top_idx = np.unique(top_idx)[:K]
        
        # Build constraint matrix (sparse-friendly)
        # Constraint m: sum_j y[m-j] * b[j] <= M_curr for valid j
        # T[m, j] = y[m-j] if 0 <= m-j < n, else 0
        T = np.zeros((len(top_idx), n))
        for col, m in enumerate(top_idx):
            m_i = int(m)
            j_start = max(0, m_i - n + 1)
            j_end = min(n - 1, m_i)
            if j_start <= j_end:
                T[col, j_start:j_end+1] = y[m_i - np.arange(j_start, j_end+1)]
        
        # Solve LP: max sum(b) s.t. b >= 0, T @ b <= M_curr, sum(b) = 1
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(len(top_idx)),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Infeasible means we're already good, or need to handle differently
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Check if LP direction improves (FW gap)
        # Gap = (f(y) - f(g_dir)) / gradient-related
        # Simple check: if y == g_dir, we're at a FW stationary point
        if np.allclose(y, g_dir, atol=1e-8):
            break
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_mom = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_candidate = y_new + beta_t * (y_new - y)
            x_candidate = np.maximum(x_candidate, 0.0)
            sx = x_candidate.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_candidate / sx
        
        prev_y = y.copy()
        y = y_new
        
        # Evaluate both y and momentum point
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()
            stall = 0
        
        # Restart if stalled
        if stall > max_stall:
            # Perturb best
            pert = rng.normal(0, 0.02, n) * best_a
            y = best_a + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
            x_mom = y.copy()
            stall = 0
            if verbose:
                print(f"    LP-FW restart at iter {t}, J={2*n*best_m:.6f}")
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, M={M_curr:.2e}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP-FW done: {iterations} iters, {elapsed:.0f}s, J={2*n*best_m:.6f}")
    
    return best_a, best_m


def projected_gradient_optimize(a0, time_limit, verbose=True, seed=42):
    """
    Projected gradient descent with backtracking on the simplex.
    Uses the subgradient structure grad_k[j] = 2*a[k-j].
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = 0.2  # Initial step size
    iterations = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        iterations += 1
        
        # Compute subgradient
        g, M_curr = compute_subgrad(a, top_k=30, soft_beta=8.0)
        
        # Projected gradient step with backtracking
        improved = False
        for _ in range(10):  # Backtracking
            a_trial = a - eta * g
            a_trial = project_simplex(a_trial)
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M_curr:  # Descent (simplified)
                a = a_trial
                improved = True
                break
            eta *= 0.5
        
        if not improved:
            # Try a smaller step or restart
            eta = max(eta * 0.5, 1e-6)
            if eta < 1e-8:
                # Random restart
                pert = rng.normal(0, 0.01, n) * a
                a = project_simplex(a + pert)
                eta = 0.2
                stall += 1
                continue
        
        # Evaluate
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Increase step size if we made good progress
        if improved and M_curr - M_new > 1e-6:
            eta = min(eta * 1.1, 1.0)
        
        if verbose and iterations % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PGD iter={iterations} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    PGD done: {iterations} iters, {elapsed:.0f}s, J={2*n*best_m:.6f}")
    
    return best_a, best_m


def coordinate_descent_pairs(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent using pair moves that directly target peak reduction.
    Key insight: moving mass from position p to position q changes
    conv(a,a)[k] by 2*a[k-p]*delta + ... for all k.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 100
    peels = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find the top peak lags
        n_top = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improvement_found = False
        
        for k in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k)
            
            # Find the pair (p, q) with p+q=k_i that has max contribution
            p_start = max(0, k_i - n + 1)
            p_end = min(n - 1, k_i)
            if p_start > p_end:
                continue
            
            ps = np.arange(p_start, p_end + 1)
            qs = k_i - ps
            contrib = a[ps] * a[qs]
            max_c_idx = np.argmax(contrib)
            p_src = int(ps[max_c_idx])
            q_src = int(qs[max_c_idx])
            src_val = a[p_src]
            
            if src_val < 1e-12:
                continue
            
            # Find a good destination: a position that minimizes the increase in all top peaks
            # For destination d, the change in conv(a,a)[k] is approximately:
            # -2*a[d]*a[k-d] (new terms) + 2*a[src]*a[k-src] (removed terms)
            # We want to minimize the max over top peaks
            
            # Compute "danger" for each destination
            # danger[d] = max over top peaks k of (a[d] * a[k-d] + a[src] * a[k-d] for the new position)
            # Simpler: just compute the sum over top peaks of the new contribution
            danger = np.zeros(n)
            for k2 in top_lags[:8]:
                k2i = int(k2)
                # If we put mass at position d, it contributes a[d]*a[k2-d] to conv[k2]
                # (plus the symmetric a[k2-d]*a[d])
                d_start = max(0, k2i - n + 1)
                d_end = min(n - 1, k2i)
                ds = np.arange(d_start, d_end + 1)
                danger[ds] += a[k2i - ds]
            
            # We want to minimize danger[d] while being far from the source
            # Also avoid positions that are partners of the source in top peaks
            for k2 in top_lags[:5]:
                k2i = int(k2)
                partner = k2i - p_src
                if 0 <= partner < n:
                    danger[partner] += 10 * a[partner]  # Penalty for being a partner
        
        # Select best destination
        # Exclude positions too close to source
        dist = np.abs(np.arange(n) - p_src).astype(float)
