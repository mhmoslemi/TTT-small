def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv_pair(a, b):
    """Compute full linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def softmax_full(a, alpha):
    """Compute softmax of conv(a,a) with temperature alpha."""
    b = fast_conv(a)
    b_max = b.max()
    w = np.exp(alpha * (b - b_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(b)) / len(b)
    else:
        w = w / w_sum
    return w, b

def compute_grad(a, w):
    """Compute gradient: grad_j = 2 * sum_m w_{j+m} * a_m"""
    n = len(a)
    a_rev = a[::-1]
    # conv(w, a_rev)[j + n - 1] for j = 0, ..., n-1
    C = fast_conv_pair(w, a_rev)
    return 2.0 * C[n-1 : 2*n-1]

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
