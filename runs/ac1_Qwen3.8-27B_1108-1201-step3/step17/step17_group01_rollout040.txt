"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with annealed log-sum-exp (fast, many iterations)
2. LP-based Frank-Wolfe for precise constraint handling
3. Multi-start with diverse initializations
4. Adaptive length exploration

import numpy as np
import time

# --- Helper functions ---

def fast_autocorr(a):
    """Compute linear autocorrelation (a * a) via FFT."""
    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]

def compute_smooth_gradient(a, mu, top_k=None):
    """Compute gradient of smooth approximation of max(autocorr)."""
    n = len(a)
    r = fast_autocorr(a)
    
    if top_k is not None and top_k < len(r):
        # Sparse gradient: only use top-k peaks
        idx = np.argsort(r)[::-1][:top_k]
        r_top = r[idx]
        r_masked = np.full_like(r, -np.inf)
        r_masked[idx] = r_top
    else:
        r_masked = r
    
    # Stable log-sum-exp
    r_max = r_masked.max()
    exp_r = np.exp(mu * (r_masked - r_max))
    w = exp_r / exp_r.sum()
    
    # Cross-correlation c_i = sum_k w_k * a_{k-i}
    # c = conv(w, a_flipped)[n-1 : 2n-1]
    a_flipped = a[::-1]
    N2 = 1 << (len(w) + len(a_flipped) - 1 - 1).bit_length()
    W = np.fft.rfft(w, N2)
    Af = np.fft.rfft(a_flipped, N2)
    C = np.fft.irfft(W * Af, N2)[:len(w) + len(a_flipped) - 1]
    c = C[n-1:2*n-1]  # c[i] for i = 0, ..., n-1
    
    # Gradient: g_i = 2*c_i - w[2i] * a_i
    g = 2.0 * c
    w_2i = w[0:2*n-1:2]  # w[0], w[2], w[4], ..., w[2n-2]
    g -= w_2i * a
    
    return g, r

def smooth_optimize(a0, time_limit, mu_start=10.0, mu_end=1000.0, 
                    eta=0.1, verbose=False, top_k=None):
    """Smooth gradient descent with annealing on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_r_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Use log-parameterization b = log(a)
    b = np.log(a)
    
    # Adam optimizer state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = eta
    
    total_iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        t_adam += 1
        
        # Anneal mu
        progress = min(1.0, total_iters / 5000)
        mu = mu_start * (mu_end / mu_start) ** progress
        
        # Compute gradient
        g, r = compute_smooth_gradient(a, mu, top_k=top_k)
        
        # Transform to b-space (softmax gradient)
        dot_ag = np.dot(a, g)
        grad_b = a * (g - dot_ag)
        
        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad_b
        v_adam = beta2 * v_adam + (1 - beta2) * grad_b**2
        m_hat = m_adam / (1 - beta1**t_adam)
        v_hat = v_adam / (1 - beta2**t_adam)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a-space
        # Use stable softmax
        b_shifted = b - b.max()
        a_new = np.exp(b_shifted)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Check improvement
        r_max = r.max()
        if r_max < best_r_max - 1e-12:
            best_r_max = r_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Learning rate schedule
        if stall > 2000:
            lr *= 0.5
            stall = 0
            # Optional: random perturbation
            if lr < 1e-6:
                break
        
        if verbose and total_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_r_max
            print(f"    SmoothOpt it={total_iters} ({elapsed:.0f}s): J={J:.6f}, mu={mu:.1f}, lr={lr:.6f}")
    
    return best_a, best_r_max

def fast_autocorr(a):
    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]

def compute_gradient(a, mu):
    """Compute gradient of smooth max approximation of autocorrelation."""
    n = len(a)
    r = fast_autocorr(a)
    r_max = r.max()
    exp_r = np.exp(mu * (r - r_max))
    w = exp_r / exp_r.sum()
    
    # Cross-correlation c_i = sum_k w_k * a_{k-i}
    a_rev = a[::-1]
    N2 = 1 << (3 * n - 3 - 1).bit_length()
    W = np.fft.rfft(w, N2)
    Ar = np.fft.rfft(a_rev, N2)
    C = np.fft.irfft(W * Ar, N2)
    c = C[n-1:2*n-1]
    
    # Gradient: g_i = 2*c_i - w[2i]*a_i
    g = 2.0 * c - w[0:2*n-1:2] * a
    return g, r

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return norm_sum1(f_interp)

    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2*sigma**2))
