"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp formulation with L-BFGS on softmax parameterization (key innovation)
2. Temperature annealing for progressively sharper peak approximation
3. LP Frank-Wolfe for polynomial-time refinement
4. Nesterov subgradient for global exploration
5. Parametric sweeps (power law, exponential) for diverse initializations
"""

import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cvx

# --- Global solver setup ---
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is None:
        for s in [cvx.CLARABEL, cvx.ECOS, cvx.PDLP, cvx.SCS]:
            try:
                x = cvx.Variable(10, nonneg=True)
                prob = cvx.Problem(cvx.Minimize(cvx.sum(x)), [cvx.sum(x) == 1, x <= 0.4])
                prob.solve(solver=s, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _global_solver = s
                    print(f"  LP solver: {s}")
                    return s
            except Exception:
                continue
        _global_solver = cvx.ECOS
        print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    ac = fast_convolve(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return exp_x / s


def compute_c_vector(a, p):
    """
    Compute c[j] = sum_m a[m] * p[j+m] for j=0,...,n-1 using FFT.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    a_rev = a[::-1]
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A_rev * P, fft_size)
    return conv_result[n - 1: 2 * n - 1]


def G_and_grad_softmax(x, T):
    """
    Objective and gradient for softmax-parameterized optimization.
    G(x) = (1/T) * logsumexp(conv(softmax(x), softmax(x)))
    """
    a = softmax_fn(x)
    n = len(a)
    conv = fast_convolve(a)
    conv_T = conv / T
    cmax = conv_T.max()
    exp_shifted = np.exp(conv_T - cmax)
    exp_sum = exp_sum_val = exp_shifted.sum()
    if exp_sum_val < 1e-300:
        return 1e10, np.zeros_like(x)
    logsumexp_val = cmax + np.log(exp_sum_val)
    obj = logsumexp_val

    # Softmax weights
    p = exp_shifted / exp_sum_val

    # Compute c[j] = sum_m a[m] * p[j+m]
    c = compute_c_vector(a, p)

    # dG/da = c / T
    dG_da = c / T

    # dG/dx = a * (dG_da - <dG_da, a>)
    offset = np.dot(dG_da, a)
    dG_dx = a * (dG_da - offset)

    return obj, dG_dx


def lbfgs_optimize(a_init, T, time_limit, max_iter=5000, verbose=False, seed=42):
    """
    Run L-BFGS on softmax-parameterized objective.
    a_init: initial normalized sequence (sum=1, non-negative)
    T: temperature for log-sum-exp
    """
    n = len(a_init)
    t0 = time.monotonic()

    # Initialize x from a_init
    a_clipped = np.maximum(a_init, 1e-15)
    x0 = np.log(a_clipped)  # softmax(x0) ≈ a_clipped (up to normalization)

    best_obj = compute_obj_normalized(a_init)
    best_a = a_init.copy()

    a_max_iterations = max_iter

    # Use scipy L-BFGS-B (unconstrained on x)
    result = minimize(
        G_and_grad_softmax,
        x0,
        args=(T,),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': a_max_iterations, 'ftol': 1e-12, 'gtol': 1e-8,
                 'maxls': 50, 'maxfun': a_max_iterations * 3}
    )

    if result.success or hasattr(result, 'fun'):
        x_best = result.x
        a_result = softmax_fn(x_best)
        a_result = norm_sum1(a_result)
        obj_result = compute_obj_normalized(a_result)
        if obj_result < best_obj:
            best_obj = obj_result
            best_a = a_result

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    L-BFGS (T={T:.2e}): obj={best_obj:.6f}, time={elapsed:.1f}s, "
              f"status={result.get('status', 'N/A')}")

    return best_a, best_obj


def lbfgs_optimize_with_progress(a_init, T, time_limit, max_iter=5000, verbose=False):
    """L-BFGS with time monitoring via callback."""
    n = len(a_init)
    t0 = time.monotonic()
    a_clipped = np.maximum(a_init, 1e-15)
    x0 = np.log(a_clipped)

    state = {'best_obj': compute_obj_normalized(a_init), 'best_a': a_init.copy(),
             'stop': False, 'count': 0}

    def callback(xk):
        state['count'] += 1
        if state['stop']:
            return True
        if time.monotonic() - t0 > time_limit:
            state['stop'] = True
            return True
        # Track best
        a_k = softmax_fn(xk)
        a_k = norm_sum1(a_k)
        obj_k = compute_obj_normalized(a_k)
        if obj_k < state['best_obj']:
            state['best_obj'] = obj_k
            state['best_a'] = a_k.copy()
        if verbose and state['count'] % 100 == 0:
            print(f"      iter={state['count']}, obj={state['best_obj']:.6f}, t={time.monotonic()-t0:.0f}s")
        return False

    result = minimize(
        G_and_grad_softmax,
        x0,
        args=(T,),
        method='L-BFGS-B',
        jac=True,
        callback=callback,
        options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                 'maxls': 50, 'maxfun': max_iter * 3}
    )

    # Final check
    x_final = result.x
    a_final = softmax_fn(x_final)
    a_final = norm_sum1(a_final)
    obj_final = compute_obj_normalized(a_final)
    if obj_final < state['best_obj']:
        state['best_obj'] = obj_final
        state['best_a'] = a_final

    if verbose:
        print(f"    L-BFGS+ (T={T:.2e}): obj={state['best_obj']:.6f}, "
              f"iters={state['count']}, time={time.monotonic()-t0:.1f}s")

    return state['best_a'], state['best_obj']


def temperature_annealing(a_init, time_limit, T_start=None, T_end=None, n_restarts=5, verbose=True):
    """
    Run L-BFGS with decreasing temperature.
    """
    n = len(a_init)
    a_current = a_init.copy()
    best_a = a_init.copy()
    best_obj = compute_obj_normalized(a_init)
    t0 = time.monotonic()

    if T_start is None:
        conv_init = fast_convolve(a_current)
        T_start = max(conv_init.max() / 20.0, 1e-8)
    if T_end is None:
        T_end = T_start / (10.0 ** n_restarts)

    T_values = np.geomspace(T_start, T_end, n_restarts + 1)

    for i, T in enumerate(T_values):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_alloc = remaining / (n_restarts - i) * 0.7
        t_alloc = max(t_alloc, 15.0)

        if verbose:
            print(f"  T-anneal step {i+1}/{n_restarts+1}: T={T:.2e}, time={t_alloc:.0f}s")

        a_out, obj_out = lbfgs_optimize_with_progress(a_current, T, t_alloc,
                                                       max_iter=3000, verbose=False)
        if obj_out < best_obj:
            best_obj = obj_out
            best_a = a_out.copy()
            a_current = a_out.copy()
            if verbose:
                print(f"    Improved: obj={best_obj:.6f}")
        else:
            # Perturb to escape
            rng = np.random.default_rng(i * 100 + 7)
            sigma = 0.01
            pert = rng.normal(0, sigma, n) * a_current
            a_pert = np.maximum(a_current + pert, 1e-10)
            a_current = a_pert / a_pert.sum()
            a_out, obj_out = lbfgs_optimize_with_progress(a_current, T, t_alloc,
                                                           max_iter=2000, verbose=False)
            if obj_out < best_obj:
                best_obj = obj_out
                best_a = a_out.copy()
                a_current = a_out.copy()
                if verbose:
                    print(f"    Improved after perturb: obj={best_obj:.6f}")

    if verbose:
        print(f"  T-anneal done: best_obj={best_obj:.6f}, time={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution
        ac = fast_convolve(y)
        M_curr = ac.max()

        # Select top-K binding constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix for selected rows
        # T[k, j] = y[k-j] if 0 <= k-j < n
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = y[k_int - j]

        b_var = cvx.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cvx.Problem(cvx.Maximize(cvx.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-style step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_mom = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_mom = y_new + beta_t * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx

        y = y_new

        # Check both y and x_mom
        my = fast_convolve(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if t % 5 == 0:
            mx = fast_convolve(x_mom).max()
            if mx < best_m:
                best_m = mx
                best_a = x_mom.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def nesterov_subgradient(a0, time_limit, top_k=15, beta=3.0, c=3.0,
                          exact_frac=0.5, restart_every=2000, seed=42, verbose=False):
    """Nesterov-accelerated subgradient descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_convolve(y)
        M = ac.max()

        # Adaptive subgradient
        use_exact = rng.random() < exact_frac
        grad = np.zeros(n)
        if use_exact:
            k = int(np.argmax(ac))
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            grad[idx] = 2.0 * a[j_idx]
        else:
            k_max = min(cur_top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** beta
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * wi * a[j_idx]

        # Step size
        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = c * gap / (np.dot(grad, grad) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        obj_new = compute_obj_normalized(x)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.01 + 0.005 * min(stall, 10)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 60)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): obj={best_obj:.6f}")

    return best_a, best_obj


def peak_reduction(a0, time_limit, seed=77, verbose=False):
    """Targeted peak reduction via mass relocation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_convolve(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]

        # Sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii, jj in zip(idx, j_idx):
                sensitivity[ii] += a[ii] * a[jj]
                sensitivity[jj] += a[jj] * a[ii]

        high_sens = np.where(sensitivity > sensitivity.mean())[0]
        low_sens = np.where(sensitivity < sensitivity.mean())[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst or a[src] < 1e-8:
                continue

            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                J_trial = compute_obj_normalized(a_trial)
                if J_trial < best_obj - 1e-12:
                    a = a_trial
                    best_obj = J_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int
