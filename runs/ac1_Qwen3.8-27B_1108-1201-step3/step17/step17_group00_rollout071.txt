"""
Hybrid search for non-negative step-function heights minimizing

    J(a) = 2 * n * max_k (a * a)_k / (sum_j a_j)^2

using:
  * FFT-based full self-convolution evaluation
  * smooth log-sum-exp (soft max) polishing with L-BFGS on softmax parameters
  * cutting-plane Frank-Wolfe/LP refinement for bottleneck self-convolution
  * randomized gradient-guided mass-move polishing
  * length interpolation exploration

The goal is to beat 1.5027 from a parent near 1.504077 by aggressively
polishing with smoother nonlinear optimization and smaller, faster LP cutting
planes.
"""

except Exception:

def next_pow2(m):
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_sum(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.reshape(-1)
    if arr.size == 0:
        return np.empty(0, dtype=np.float64)
    arr = arr.copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def fast_conv_self(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(arr, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(c, 0.0)


def fast_convolve_two(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    nx = x.size
    ny = y.size
    if nx == 0 or ny == 0:
        return np.empty(0, dtype=np.float64)
    size = nx + ny - 1
    fft_size = next_pow2(size)
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    c = np.fft.irfft(X * Y, fft_size)[:size]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(c, 0.0)


def score_fast(a):
    arr = normalize_sum(a)
    n = arr.size
    if n == 0:
        return float("inf")
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    c = fast_conv_self(arr)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * c.max() / (s * s))


def try_update_best(best_a, best_score, a_new, label):
    if a_new is None:
        return best_a, best_score
    arr = normalize_sum(a_new)
    if arr.size == 0:
        return best_a, best_score
    s = score_fast(arr)
    if np.isfinite(s) and s < best_score - 1e-12:
        print(f"  NEW BEST [{label}]: J={s:.8f}, n={len(arr)}")
        return arr, s
    return best_a, best_score


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    if f.ndim != 1:
        f = f.reshape(-1)
    old_n = f.size
    new_n = int(new_n)
    if new_n <= 0:
        return np.empty(0, dtype=np.float64)
    if old_n == new_n:
        return normalize_sum(f)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)

    s = float(f.sum())
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    f = f / s

    old_x = (np.arange(old_n, dtype=np.float64) + 0.5) / float(old_n)
    new_x = (np.arange(new_n, dtype=np.float64) + 0.5) / float(new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    return normalize_sum(out)


def lse_weights_and_value(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.empty(0, dtype=np.float64)
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    c = np.maximum(c, 0.0)
    cmax = float(c.max())

    if beta <= 1e-8:
        p = np.ones(c.size, dtype=np.float64) / float(c.size)
        return float(np.mean(c)), p

    scaled = beta * (c - cmax)
    ex = np.exp(scaled)
    ex = np.nan_to_num(ex, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(ex.sum())
    if s <= 1e-300:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return cmax, p
    p = ex / s
    F = cmax + float(np.log(s)) / float(beta)
    return float(F), p


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.empty(0, dtype=np.float64)
    z = z - float(np.mean(z))
    z = z - float(z.max())
    a = np.exp(z)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return a / s


def z_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.empty(0, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    z = np.log(a)
    return z - float(np.mean(z))


def lse_function_from_a(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.empty(0, dtype=np.float64)

    # Use a slightly larger FFT so that both self-convolution and gradient
    # can share the same rfft(a).
    L = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, L)

    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    c = np.maximum(c, 0.0)

    F, p = lse_weights_and_value(c, float(beta))

    if n == 0 or p.size == 0:
        return F, np.zeros(n, dtype=np.float64)

    # h_j = sum_k p_k (a_{k-j} + a_{j-k}).
    P = np.fft.rfft(p, L)
    P_rev = np.fft.rfft(p[::-1], L)
    h1 = np.fft.irfft(A * P, L)[:n]
    h2full = np.fft.irfft(A * P_rev, L)
    start = 2 * n - 2
    h = h1 + h2full[start:start + n]
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    h = np.maximum(h, 0.0)

    s = float(np.dot(a, h))
    return F, h - s


def lse_objective(x, beta):
    a = softmax_from_z(x)
    F, _ = lse_function_from_a(a, beta)
    if not np.isfinite(F):
        return 1e10
    return float(F)


def lse_grad(x, beta):
    a = softmax_from_z(x)
    _, grad_a = lse_function_from_a(a, beta)
    g = a * grad_a
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return g - float(np.mean(g))


def lse_callback(ckpt):
    global _LSE_LAST_X
    xv = ckpt.x
    if xv is not None:
        xv = np.asarray(xv, dtype=np.float64)
        if not np.all(np.isfinite(xv)):
            xv = np.where(np.isfinite(xv), xv, 0.0)
        xv = xv - float(np.mean(xv))
        ckpt.x = xv
        _LSE_LAST_X = xv.copy()
    if time.monotonic() > _LSE_DEADLINE:
        raise TimeBudgetExceeded


def lse_bfgs_polish(a0, total_time, beta_stages=None, verbose=True, seed=123):
    global _LSE_DEADLINE, _LSE_LAST_X

    a = normalize_sum(a0)
    n = a.size
    if n == 0:
        return a, float("inf")
    if not _HAVE_SCIPY:
        return a, score_fast(a)

    t0 = time.monotonic()
    deadline = t0 + float(total_time)
    c = fast_conv_self(a)
    best_m = float(c.max())
    best_score = float(2.0 * n * best_m)
    best = a.copy()

    if beta_stages is None:
        base = max(1000.0, 14.0 / (best_m + 1e-15))
        beta_stages = [base / 50.0, base / 5.0, base, base * 4.0, base * 12.0]

    rng = np.random.default_rng(seed)

    for si, beta in enumerate(beta_stages):
        remaining = deadline - time.monotonic()
        if remaining <= 1.0:
            break

        n_stages_left = max(1, len(beta_stages) - si)
        stage_time = max(2.0, remaining / float(n_stages_left))
        stage_time = min(stage_time, remaining)
        if remaining < 5.0:
            stage_time = max(0.2, remaining)

        _LSE_DEADLINE = time.monotonic() + stage_time
        _LSE_LAST_X = None

        x0 = z_from_a(best)
        noise_scale = 1e-4 if si == 0 else 1e-5
        x0 = x0 + noise_scale * rng.normal(0.0, 1.0, size=n)
        x0 = x0 - float(np.mean(x0))
        x0 = np.nan_to_num(x0, nan=0.0, posinf=0.0, neginf=0.0)

        stage_maxfun = int(min(4000, max(200, stage_time * 80.0)))

        x_final = None
        try:
            res = sciopt.minimize(
                lse_objective,
                x0,
                jac=lse_grad,
                args=(float(beta),),
                method="L-BFGS-B",
                options={
                    "maxfun": stage_maxfun,
                    "maxiter": stage_maxfun,
                    "ftol": 1e-15,
                    "gtol": 1e-8,
                    "disp": False,
                },
                callback=lse_callback,
            )
            x_final = res.x
        except TimeBudgetExceeded:
            x_final = _LSE_LAST_X
        except Exception:
            x_final = _LSE_LAST_X

        if x_final is not None:
            a_new = softmax_from_z(np.asarray(x_final, dtype=np.float64))
            a_new = normalize_sum(a_new)
            m_new = float(fast_conv_self(a_new).max())
            if np.isfinite(m_new) and m_new < best_m - 1e-15:
                best_m = m_new
                best_score = float(2.0 * n * best_m)
                best = a_new.copy()
                if verbose:
                    print(f"    LSE-BFGS stage {si + 1}/{len(beta_stages)} beta={float(beta):.0f}: J={best_score:.8f}")

    return best, best_score


def build_lp_matrix(y, klist):
    y = np.asarray(y, dtype=np.float64)
    n = y.size
    K = len(klist)
    if K == 0:
        return csr_matrix((0, n), dtype=np.float64)

    klist = np.asarray(klist, dtype=np.int64)
    starts = np.maximum(0, klist - n + 1)
    ends = np.minimum(n, klist + 1)
    lens = ends - starts
    total = int(lens.sum())

    if total == 0:
        return csr_matrix((K, n), dtype=np.float64)

    rows = np.repeat(np.arange(K, dtype=np.int32), lens)
    col_arrays = []
    data_arrays = []

    for r in range(K):
        s = int(starts[r])
        e = int(ends[r])
        if s >= e:
            continue
        k = int(klist[r])
        idx = np.arange(s, e, dtype=np.int32)
        col_arrays.append(idx)
        data_arrays.append(np.asarray(y[k - idx], dtype=np.float64))

    if col_arrays:
        cols = np.concatenate(col_arrays)
        data = np.concatenate(data_arrays)
    else:
        cols = np.empty(0, dtype=np.int32)
        data = np.empty(0, dtype=np.float64)

    return csr_matrix((data, (rows, cols)), shape=(K, n), dtype=np.float64)

        if stall > 1800:

            cyy = fast_conv_self(y)
            active = set(int(k) for k in np.argsort(cyy)[::-1][:initial_K])

                    klist.append(ik)
            klist.sort()

        A = build_lp_matrix(y, klist)
        g_var = cp.Variable(n, nonneg=True)
        m_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, A @ g_var <= m_var]
        prob = cp.Problem(cp.Minimize(m_var), cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fail_count += 1
            if fail_count > 25:
                break
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            fail_count += 1
            if fail_count > 25:
                break
            if len(klist) > initial_K:
                active = set(int(k) for k in np.argsort(cyy)[::-1][:initial_K])
            continue

        gval = None
        if g_var.value is not None:
            gval = np.asarray(g_var.value, dtype=np.float64).reshape(n)
            gval = np.maximum(gval, 0.0)
            gs = float(gval.sum())
            if gs > 1e-12:
                gval = gval / gs
            else:
                gval = None

        if gval is None:
            fail_count += 1
            if fail_count > 25:
                break
            continue

        fail_count = 0

        y_old = y.copy()
        alpha = 2.0 / float(it + 2)
        y_new = (1.0 - alpha) * y_old + alpha * gval
        y_new = normalize_sum(y_new)

        if it == 1:
            x_new = y_new.copy()
        else:
            beta_t = float(it - 1) / float(it + 2)
            x_new = y_new + beta_t * (y_new - y_old)
            x_new = np.maximum(x_new, 0.0)
            sx = float(x_new.sum())
            if sx <= 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        yg = fast_convolve_two(y_old, gval)
        if yg.size > 0:
            viol = np.where(yg > Mcur + 1e-9)[0]
            if viol.size > 0:
                topv = viol[np.argsort(yg[viol])[::-1][:add_per_iter]]
                for k in topv:
                    active.add(int(k))

        if verbose and it % 25 == 1:
            cur_score = float(2.0 * n * best_m)
            print(f"    LP-FW iter={it}, score={cur_score:.8f}, |K|={len(active)}, stall={stall}")

        it += 1

    return best, float(2.0 * n * best_m)


def random_polish(a0, time_limit, rng_seed=2, verbose=True):
    a = normalize_sum(a0)
    n = a.size
    if n < 2:
        return a, score_fast(a)

    c = fast_conv_self(a)
    best_m = float(c.max())
    best = a.copy()
    current_m = best_m

    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 <= float(time_limit) and stall < 2500:
        c = fast_conv_self(a)
        current_m = float(c.max())

        if current_m < best_m - 1e-15:
            best_m = current_m
            best = a.copy()
            stall = 0

        top_k = int(min(16, c.size))
        idx = np.argsort(c)[::-1][:top_k]
        p = np.zeros(c.size, dtype=np.float64)
        p[idx] = 1.0 / float(top_k)

        L = next_pow2(3 * n - 2)
        A = np.fft.rfft(a, L)
        P = np.fft.rfft(p, L)
        P_rev = np.fft.rfft(p[::-1], L)
        h1 = np.fft.irfft(A * P, L)[:n]
        h2 = np.fft.irfft(A * P_rev, L)
        start = 2 * n - 2
        h = h1 + h2[start:start + n]
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
        h = np.maximum(h, 0.0)
        g = h - float(np.mean(h))

        src_w = np.maximum(g, 0.0) * a + 1e-15
        dst_w = np.maximum(-g, 0.0) + 1e-15
        src_sum = float(src_w.sum())
        dst_sum = float(dst_w.sum())
        if src_sum > 0:
            src_w = src_w / src_sum
        else:
            src_w = np.ones(n, dtype=np.float64) / float(n)
        if dst_sum > 0:
            dst_w = dst_w / dst_sum
        else:
            dst_w = np.ones(n, dtype=np.float64) / float(n)

        src_top = np.argsort(src_w)[::-1][:80]
        dst_top = np.argsort(dst_w)[:80]

        improved = False
        for _ in range(40):
            if time.monotonic() - t0 > float(time_limit):
                break
            src = int(src_top[rng.integers(len(src_top))])
            dst = int(dst_top[rng.integers(len(dst_top))])
            if src == dst:
                continue
            frac = _MOVE_FRACTIONS[int(rng.integers(len(_MOVE_FRACTIONS)))]
            amt = a[src] * frac
            if amt < 1e-14:
                continue

            trial = a.copy()
            trial[src] -= amt
            trial[dst] += amt
            trial = np.maximum(trial, 0.0)
            trial = normalize_sum(trial)
            m = float(fast_conv_self(trial).max())

            if m < current_m - 1e-15:
                a = trial
                current_m = m
            if m < best_m - 1e-15:
                best_m = m
                best = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 40 and stall % 10 == 0:
                noise = rng.normal(0.0, 1e-4, size=n)
                a = normalize_sum(np.maximum(best * (1.0 + noise), 0.0))
                current_m = float(fast_conv_self(a).max())
                stall = 0
            if stall > 500:
                heavy = rng.normal(0.0, 3e-3, size=n)
                a = normalize_sum(np.maximum(best * (1.0 + heavy), 0.0))
                current_m = float(fast_conv_self(a).max())
                stall = 0

        if verbose and stall % 300 == 0:
            print(f"    Random polish stall={stall}, best J={2.0 * n * best_m:.8f}")

    return best, float(2.0 * n * best_m)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    hard_limit = min(float(budget_s) - 60.0, 1740.0)

    best_a = None
    try:
        parent_obj = globals().get("height_sequence_1", None)
        if parent_obj is not None:
            p = np.asarray(parent_obj, dtype=np.float64)
            if p.ndim != 1:
                p = p.reshape(-1)
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 0.01:
                best_a = normalize_sum(p)
    except Exception:
        best_a = None

    if best_a is None or best_a.size == 0:
        n_fb = 4000
        x = np.linspace(0.0, 1.0, n_fb)
        best_a = 0.5 * (1.0 - np.cos(np.pi * x)) ** 1.2 + 0.05 * np.sin(np.pi * x) ** 2 + 1e-3
        best_a = normalize_sum(best_a)

    best_score = score_fast(best_a)
    if not np.isfinite(best_score):
        n_fb = 4000
        x = np.linspace(0.0, 1.0, n_fb)
        best_a = normalize_sum(np.sin(np.pi * x) ** 2)
        best_score = score_fast(best_a)

    recent_candidates = [best_a.copy()]
    print(f"Initial: n={len(best_a)}, J_fast={best_score:.8f}")

    try:
        off = evaluate_sequence([float(v) for v in best_a])
        if np.isfinite(off):
            print(f"Initial official score: {off:.8f}")
    except Exception:
        pass

    # Phase 1: smooth LSE polish on the supplied parent.
    rem = hard_limit - (time.monotonic() - t_start)
    t1 = min(rem * 0.16, 260.0)
    if t1 > 45:
        a1, _ = lse_bfgs_polish(best_a, t1, verbose=True, seed=17)
        prev = best_score
        best_a, best_score = try_update_best(best_a, best_score, a1, "lse1")
        if best_score < prev and len(recent_candidates) < 8:
            recent_candidates.append(best_a.copy())
        if len(recent_candidates) > 8:
            recent_candidates.pop(0)

    # Phase 2: LP / cutting-plane Frank-Wolfe.
    rem = hard_limit - (time.monotonic() - t_start)
    n_cur = len(best_a)
    t2 = min(rem * 0.34, 650.0)
    if t2 > 100 and n_cur <= 7000:
        solver = get_lp_solver()
        if solver is not None:
            a2, _ = lp_frank_wolfe(
                best_a,
                t2,
                solver,
                initial_K=64,
                max_K=320,
                add_per_iter=24,
                verbose=True
            )
            prev = best_score
            best_a, best_score = try_update_best(best_a, best_score, a2, f"lp_n{len(a2)}")
            if best_score < prev:
                recent_candidates.append(best_a.copy())
                if len(recent_candidates) > 8:
                    recent_candidates.pop(0)

    # Phase 3: LSE polish after LP.
    rem = hard_limit - (time.monotonic() - t_start)
    t3 = min(rem * 0.22, 330.0)
    if t3 > 45:
        a3, _ = lse_bfgs_polish(best_a, t3, verbose=True, seed=31)
        prev = best_score
        best_a, best_score = try_update_best(best_a, best_score, a3, "lse2")
        if best_score < prev:
            recent_candidates.append(best_a.copy())
            if len(recent_candidates) > 8:
                recent_candidates.pop(0)

    # Phase 4: fast randomized local moves.
    rem = hard_limit - (time.monotonic() - t_start)
    t4 = min(rem * 0.08, 140.0)
    if t4 > 30:
        a4, _ = random_polish(best_a, t4, rng_seed=5, verbose=True)
        prev = best_score
        best_a, best_score = try_update_best(best_a, best_score, a4, "random")
        if best_score < prev:
            recent_candidates.append(best_a.copy())
            if len(recent_candidates) > 8:
                recent_candidates.pop(0)

    # Phase 5: length exploration around the best length found so far.
    rem = hard_limit - (time.monotonic() - t_start)
    base_n = len(best_a)
    tried = set()
    tried.add(base_n)
    quick = []

    for delta in (-1000, -500, 500, 1000, 1500, 2000):
        nn = base_n + delta
        if nn in tried:
            continue
        if 1500 <= nn <= 9000:
            tried.add(nn)
            candid = interpolate_sequence(best_a, nn)
            scand = score_fast(candid)
            quick.append((scand, nn, candid))
        if len(quick) >= 6:
            break

    quick.sort()
    explored_this_phase = 0
    for item in quick:
        if time.monotonic() - t_start > hard_limit - 90:
            break
        scand, nn, candid = item
        if not np.isfinite(scand):
            continue
        # Skip clearly hopeless extensions, but still allow a few borderline ones.
        if scand > best_score * 1.08 + 0.01:
            continue

        rem = hard_limit - (time.monotonic() - t_start)
        stage_time = min(rem * 0.42, 230.0)
        if stage_time < 35:
            break

        a_long, _ = lse_bfgs_polish(candid, stage_time, verbose=False, seed=101)
        prev = best_score
        best_a, best_score = try_update_best(best_a, best_score, a_long, f"len_tune_n{nn}")
        if best_score < prev:
            recent_candidates.append(best_a.copy())
            if len(recent_candidates) > 8:
                recent_candidates.pop(0)

        # A short random polish can sometimes unlock the length change.
        rem = hard_limit - (time.monotonic() - t_start)
        if rem > 45:
            a_long2, _ = random_polish(best_a, min(rem * 0.18, 80.0), rng_seed=55, verbose=False)
            prev = best_score
            best_a, best_score = try_update_best(best_a, best_score, a_long2, f"len_rand_n{nn}")
            if best_score < prev:
                recent_candidates.append(best_a.copy())
                if len(recent_candidates) > 8:
                    recent_candidates.pop(0)

        explored_this_phase += 1
        if explored_this_phase >= 3:
            break

    # Phase 6: final high-smoothness LSE squeeze.
    rem = hard_limit - (time.monotonic() - t_start)
    t6 = min(rem * 0.55, 260.0)
    if t6 > 20:
        best_m_now = float(fast_conv_self(best_a).max())
        base_beta = max(1000.0, 20.0 / (best_m_now + 1e-15))
        high_stages = [base_beta, base_beta * 5.0, base_beta * 20.0, base_beta * 60.0]
        a6, _ = lse_bfgs_polish(
            best_a,
            t6,
            beta_stages=high_stages,
            verbose=True,
            seed=211
        )
        prev = best_score
        best_a, best_score = try_update_best(best_a, best_score, a6, "lse_final")
        if best_score < prev:
            recent_candidates.append(best_a.copy())
            if len(recent_candidates) > 8:
                recent_candidates.pop(0)

    # Final official validation among recent strong candidates.
    rem = hard_limit - (time.monotonic() - t_start)
    if rem > 15:
        final_seq = best_a
        final_official = None
        try:
            cand_list = [best_a]
            for cand in recent_candidates:
                if cand is not best_a and len(cand) == len(best_a):
                    cand_list.append(cand)
                elif cand is not best_a:
                    cand_list.append(cand)

            seen = set()
            for cand in cand_list:
                key = (len(cand), float(cand.sum()), float(cand.max()))
                if key in seen:
                    continue
                seen.add(key)
                val = evaluate_sequence([float(v) for v in cand])
                if np.isfinite(val):
                    if final_official is None or val < final_official:
                        final_official = val
                        final_seq = cand
        except Exception:
            final_seq = best_a

        if final_seq is not None and final_seq.size == best_a.size:
            best_a = finalize = normalize_sum(final_seq)
        elif final_seq is not None:
            best_a = normalize_sum(final_seq)
        else:
            best_a = best_a
        best_score = score_fast(best_a)

        try:
            off_final = evaluate_sequence([float(v) for v in best_a])
            if np.isfinite(off_final):
                print(f"FINAL official score: {off_final:.8f}")
            else:
                print(f"FINAL fast score: {best_score:.8f}")
        except Exception:
            print(f"FINAL fast score: {best_score:.8f}")
    else:
        print(f"FINAL fast score: {best_score:.8f}")

    # Absolute fallback, should almost never trigger.
    if best_a is None or best_a.size == 0 or not np.isfinite(best_score):
        x = np.linspace(0.0, 1.0, 3200)
        best_a = normalize_sum(np.sin(np.pi * x) ** 2)
        best_score = score_fast(best_a)

    return [float(v) for v in np.asarray(best_a, dtype=np.float64)]
