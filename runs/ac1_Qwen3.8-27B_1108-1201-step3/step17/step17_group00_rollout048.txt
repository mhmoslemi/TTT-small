def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    if not HAS_CP:
        return None
    for s in [cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  Solver: {s}")

        except Exception:

    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient_at_peak(a, ac, active_lags):
    """Compute gradient of max autocorrelation using active set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * a[j_idx]
    return grad / len(active_lags)


def lp_frank_wolfe_improved(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    """LP-FW with adaptive constraint management and line search."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_val = 2.0 * n * fast_autocorr(y).max()
    t0 = time.monotonic()
    
    if solver is None:
        return best_a, best_val

    # Initialize constraint set: top autocorrelation lags
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    # Use top K lags plus any lags with value > 0.95*M
    threshold = 0.95 * M_curr
    init_set = set(int(k) for k in np.where(ac_y >= threshold)[0])
    if len(init_set) < initial_K:
        top_idx = np.argsort(ac_y)[::-1][:initial_K]
        init_set = set(int(k) for k in top_idx)
    constraint_set = init_set
    
    iter_count = 0
    stall = 0
    max_stall = 2000
    best_stall_reset = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        curr_val = 2.0 * n * M_curr

        if curr_val < best_val - 1e-10:
            best_val = curr_val
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        # Periodically reset to top lags of current y
        if t % 50 == 0 or len(constraint_list) > max_K:
            new_set = set()
            # Always keep lags where autocorr is close to peak
            close_threshold = M_curr * (1.0 - 0.05)
            close_lags = np.where(ac >= close_threshold)[0]
            new_set = set(int(k) for k in close_lags)
            # Also keep top initial_K
            top = np.argsort(ac)[::-1][:initial_K]
            new_set |= set(int(k) for k in top)
            constraint_set = new_set if len(new_set) <= max_K else set(int(k) for k in top[:max_K])
            constraint_list = sorted(constraint_set)

        if not constraint_list:
            break

        K = len(constraint_list)
        # Build LP matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > 200:
                constraint_set = set(int(k) for k in constraint_list[:200])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violated constraints
        conv_yg = fast_conv_ab(y, g_dir)
        violations = np.where(conv_yg > M_curr + 1e-7)[0]
        if len(violations) > 0:
            # Add most violated
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                constraint_set.add(int(k))

        # Line search: try multiple alpha values
        best_alpha = 0.1
        best_trial_val = float('inf')
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_tr = y_trial.sum()
            if s_tr < 1e-15:
                continue
            y_trial = y_trial / s_tr
            ac_tr = fast_autocorr(y_trial)
            val_tr = 2.0 * n * ac_tr.max()
            if val_tr < best_trial_val:
                best_trial_val = val_tr
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y = y / sy

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={best_val:.6f}, |S|={len(constraint_set)}, stall={stall}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={best_val:.6f}, iters={iter_count}")
    return best_a, best_val


def kkt_mirror_descent(a0, time_limit, verbose=True):
    """Fixpoint mirror descent based on KKT characterization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 0.5
    smooth_beta = 20.0  # Temperature for smooth max
    stall = 0
    max_stall = 300

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Smooth max: weight lags by exp(beta * ac[k] / M_curr)
        scaled = ac / max(M_curr, 1e-15)
        weights = np.exp(smooth_beta * (scaled - 1.0))  # Peak gets weight 1, others < 1
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros_like(ac)
            top_k = np.argmax(ac)
            weights[top_k] = 1.0
            w_sum = 1.0
        else:
            weights = weights / w_sum

        # Compute gradient with soft active set
        n_active = min(50, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_active]
        # Weight by exp(beta*(ac[k]-M)/M)
        active_weights = np.exp(smooth_beta * (ac[top_lags] - M_curr) / max(M_curr, 1e-15))
        aw_sum = active_weights.sum()
        if aw_sum < 1e-15:
            active_weights = np.ones(n_active) / n_active
        else:
            active_weights = active_weights / aw_sum

        grad = np.zeros(n)
        for wi, k in zip(active_weights, top_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        # Mirror descent step: a_new = a * exp(-eta * grad) / sum
        # Adaptive eta
        eta_t = eta / (1.0 + 0.01 * t)
        log_a = np.log(np.maximum(a, 1e-30)) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        ac_new = fast_autocorr(a_new)
        val_new = 2.0 * n * ac_new.max()

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Optionally increase beta to sharpen active set
            if t % 500 == 0:
                smooth_beta = min(smooth_beta + 5.0, 100.0)
        else:
            stall += 1
            # Occasionally try a larger step
            if stall > 50 and stall % 10 == 0:
                eta_t2 = eta * 2.0
                log_a2 = np.log(np.maximum(a, 1e-30)) - eta_t2 * grad
                max_log2 = log_a2.max()
                a_try = np.exp(log_a2 - max_log2)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    val_try = 2.0 * n * fast_autocorr(a_try).max()
                    if val_try < val_new:
                        a = a_try
                        if val_try < best_val:
                            best_val = val_try
                            best_a = a.try.copy() if hasattr(a, 'try') else a.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT-MD iter={t} ({elapsed:.0f}s): J={best_val:.6f}, beta={smooth_beta:.0f}, stall={stall}")

    return best_a, best_val


def coordinate_polish_v2(a0, time_limit, rng_seed=42, verbose=True):
    """Improved coordinate descent with better move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top contributing lags
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the pair with max contribution
            pair_idx = np.argmax(contrib)
            src = idx[pair_idx]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute "impact score" for each destination
            # Moving mass from src to dst changes autocorrelation at all lags
            # that involve src or dst
            # Approximate: look at lags where dst has low contribution
            dst_scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    # Contribution from position i
                    dst_scores[idx2] += a[k2i - idx2]
                    mirror = k2i - idx2
                    valid_m = (mirror >= 0) & (mirror < n)
                    dst_scores[reverse_indices(idx2, k2i, n)] += a[mirror[valid_m]]

            # Find best destination (lowest score = least impact on peak)
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            # Also mask positions that are in the peak region
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                mask[s2:e2 + 1] = False
            
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.where(mask | np.ones(n, dtype=bool))[0]
                mask[src] = False
                candidates = np.where(mask)[0]
            
            if len(candidates) == 0:
                continue

            dst = candidates[np.argmin(dst_scores[candidates])]
            if dst == src:
                continue

            # Try multiple fractions
            found = False
            for frac in [0.01, 0.02, 0.04, 0.06, 0.10, 0.15]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_val / (2.0 * n) - 1e-14:
                    a = a_trial
                    curr_v = 2.0 * n * fast_autocorr(a).max()
                    if curr_v < best_val:
                        best_val = curr_v
                        best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and src_val > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / max(a.sum(), 1e-15)
                    M_new = fast_autocorr(a).max()
                    if 2.0 * n * M_new < best_val:
                        best_val = 2.0 * n * M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if 2.0 * n * M_new < best_val:
                        best_val = 2.0 * n * M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): J={best_val:.6f}, stall={stall}")

    return best_a, best_val


def reverse_indices(idx, k, n):
    """For convolution at lag k, the mirror index of position i is k-i."""
    mirror = k - idx
    # Clip to valid range
    mirror = np.clip(mirror, 0, n - 1)
    return mirror


def symmetric_optimization(a0, time_limit, verbose=True):
    """Exploit potential structure: optimize in the space of log-symmetric sequences."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Work with log(a) parameterization for natural simplex movement
    log_a = np.log(np.maximum(a, 1e-30))
    half_n = n // 2

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Reconstruct a from log_a (renormalize)
        a_cur = np.exp(log_a - log_a.max())
        a_cur = a_cur / a_cur.sum()

        ac = fast_autocorr(a_cur)
        M = ac.max()
        val = 2.0 * n * M
        if val < best_val - 1e-10:
            best_val = val
            best_a = a_cur.copy()

        # Get active lags (top 20)
        top_k = np.argsort(ac)[::-1][:20]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in top_k:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a_cur[k_int - idx]
        grad /= len(top_k)

        # Mirror descent step with adaptation
        eta = 0.3 / (1.0 + 0.05 * t)
        log_a = log_a - eta * grad
        # Renormalize in log-space
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            log_a = np.log(np.maximum(a_cur, 1e-30))
            continue
        a_new = a_new / s_new

        # Check improvement
        ac_new = fast_autocorr(a_new)
        val_new = 2.0 * n * ac_new.max()
        if val_new < val:
            a_cur = a_new
            log_a = np.log(np.maximum(a_new, 1e-30))
        else:
            # Revert and reduce step
            eta = eta * 0.5
            log_a = np.log(np.maximum(a_cur, 1e-30))

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SymMD iter={t} ({elapsed:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick KKT mirror descent on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: KKT Mirror Descent (n={n0}) ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = kkt_mirror_descent(a_in, t1, verbose=True)
        check_and_update(a_out, "kkt_md")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t2, solver, initial_K=300, max_K=3500, verbose=True)
        check_and_update(a_out, f"lp_main_n{n_best}")

    # === Phase 3: KKT MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: KKT MD polish (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = kkt_mirror_descent(a_in, t3, verbose=True)
        check_and_update(a_out, "kkt_polish")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent (n={n_best}) ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish_v2(a_in, t4, rng_seed=1, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Second LP-FW round (with fresh constraint set) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}) ===")
    if remaining > 120 and n_best <= 6000:
        t5 = min(remaining * 0.25, 350.0)
        if t5 > 60:
            a_in = norm_sum1(best_seq)
            # Perturb slightly for fresh start
            rng = np.random.default_rng(123)
            perturbation = rng.normal(0, 0.005, n_best) * a_in
            a_pert = a_in + perturbation
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            a_out, _ = lp_frank_wolfe_improved(a_pert, t5, solver, initial_K=400, max_K=4000, verbose=True)
            check_and_update(a_out, f"lp_r2_n{n_best}")

    # === Phase 6: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    # Try a shorter length (sometimes helps)
    shorter_lengths = [n_best - 500, n_best - 1000]
    shorter_lengths = [l for l in shorter_lengths if l >= 3000]
    
    # Try a longer length
    longer_lengths = [n_best + 500, n_best + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 7000]
    
    for new_n in (shorter_lengths[:1] + longer_lengths[:2]):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Different length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick KKT MD
        t6a = min(remaining * 0.30, 120.0)
        if t6a > 20:
            a_out, _ = kkt_mirror_descent(a_long, t6a, verbose=False)
            check_and_update(a_out, f"kkt_n{new_n}")
        
        # LP-FW if time permits
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 6000:
            t6b = min(remaining * 0.40, 200.0)
            if t6b > 40:
                a_out, _ = lp_frank_wolfe_improved(norm_sum1(best_seq), t6b, solver,
                                                     initial_K=250, max_K=2500, verbose=False)
                # Interpolate result back to new_n if needed
                if len(a_out) != new_n:
                    a_out = interpolate_sequence(a_out, new_n)
                check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 20:
            # Run both KKT and coordinate
            t7a = t7 * 0.5
            t7b = t7 * 0.5
            
            a_in = norm_sum1(best_seq)
            a_out1, _ = kkt_mirror_descent(a_in, t7a, verbose=False)
            check_and_update(a_out1, "final_kkt")
            
            a_in = norm_sum1(best_seq)
            a_out2, _ = coordinate_polish_v2(a_in, t7b, rng_seed=777, verbose=False)
            check_and_update(a_out2, "final_coord")

    # === Phase 8: If time remains, another LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60 and n_best <= 6000:
        print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
        t8 = remaining - 30.0
        if t8 > 40:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_improved(a_in, t8, solver, initial_K=500, max_K=5000,
                                                verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 5200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
