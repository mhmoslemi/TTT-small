    # g = np.convolve(W, a[::-1], 'full') ... let me verify
    # np.convolve(W, a_rev, 'full')[m] = sum_k W[k] * a_rev[m-k] = sum_k W[k] * a[n-1-(m-k)]
    # = sum_k W[k] * a[n-1-m+k]
    # We want g[i] = sum_k W[k] * a[k-i]
    # Set n-1-m+k = k-i => n-1-m = -i => m = n-1+i
    # So g[i] = np.convolve(W, a[::-1], 'full')[n-1+i]
    
    W_full = np.zeros(2*n-1)
    W_full[idx] = w
    
    a_rev = a[::-1]
    conv_W_arev = np.convolve(W_full, a_rev, 'full')
    # g[i] = conv_W_arev[n-1+i] for i in [0, n-1]
    g1 = conv_W_arev[n-1:n-1+n]
    
    # Similarly for the second part: sum_k W[k] * a[i-k]
    # This is just np.convolve(W, a, 'full')[i + something]
    # np.convolve(W, a, 'full')[m] = sum_k W[k] * a[m-k]
    # We want sum_k W[k] * a[i-k], which is np.convolve(W, a, 'full')[i]
    # But W has length 2n-1 and a has length n, so the full convolution has length 3n-2
    # g2[i] = np.convolve(W_full, a, 'full')[i] ... wait, need to check indexing
    # np.convolve(v, w, 'full')[n] = sum_{k} v[k] * w[n-k]
    # So g2[i] = sum_k W[k] * a[i-k] = np.convolve(W_full, a, 'full')[i]
    # But we need i-k to be in [0, n-1], which is automatically handled.
    
    # Hmm, but this is getting complicated. Let me just use the FFT approach for the whole thing.
    
    # The subgradient of max_k (a*a)_k is:
    # grad = sum_k alpha_k * d(a*a)_k/da
    # where alpha_k are the subgradient weights ( convex combo, concentrated on argmax)
    # d(a*a)_k/da is the vector with:
    #   component i = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n and i-k != k-i)
    # Actually I realize d(a*a)_k/da_i = a[k-i] + a[i-k] when both are in range, 
    # but when k-i = i (i.e., i=k/2), it's just a[i] (not 2*a[i]).
    # 
    # Wait, let me re-derive. (a*a)_k = sum_j a_j a_{k-j}
    # d/dx_i: 
    #   The term with j=i: d(a_i * a_{k-i})/dx_i = a_{k-i}
    #   The term with j=k-i (if k-i != i and 0<=k-i<n): d(a_{k-i} * a_i)/dx_i = a_{k-i}
    #   So total: a_{k-i} + a_{k-i} = 2*a_{k-i} when k-i != i and both in range
    #   Or just a_{k-i} when k-i = i (i.e., k=2i)
    #   Or just a_{k-i} when only one of i, k-i is in range
    #
    # So it's approximately 2 * a[k-i] (with a correction at the midpoint).
    
    # For a practical approximation, I'll use:
    # grad_i ≈ 2 * sum_k W[k] * a[k-i] (for k-i in [0,n-1])
    # This overcounts by factor 2 at the midpoint, but that's a minor issue.
    
    # Using FFT:
    fft_size = 1 << (3*n).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(W_full, fft_size)
    # cross-correlation: ifft(conj(A_fft) * W_fft) gives sum_j a[j] * W[i+j]
    # which is what we want (up to indexing)
    
    # Actually, let me just use np.correlate or a simpler approach for moderate n.
    
    # For n up to ~5000, direct computation with vectorization should be fine.
    grad = np.zeros(n)
    for k_int in idx:
        wi = w[np.where(idx == k_int)[0][0]]
        # Valid i range: max(0, k_int-n+1) to min(n-1, k_int)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n - 1, k_int)
        i_vals = np.arange(i_lo, i_hi + 1)
        j1 = k_int - i_vals  # = k - i
        # contrib = a[j1] + a[i-k] where valid
        contrib = a[j1].copy()
        j2 = i_vals - k_int
        valid_j2 = (j2 >= 0) & (j2 < n)
        if valid_j2.any():
            contrib += a[j2[valid_j2]]
        # Correct for double counting where j1 == j2
        same = (j1 == j2)
        if same.any():
            contrib[same] -= a[j1[same]]
        grad[i_vals] += wi * contrib
    
    return grad

Let me also think about whether there's a theoretical insight I'm missing.

The problem is equivalent to: minimize $\|a*a\|_{L^\infty}$ subject to $a \geq 0$, $\|a\|_1 = 1$.

In the frequency domain, this is: minimize the peak of the inverse DFT of $|A|^2$ subject to the constraint that the inverse DFT of $A$ is non-negative.

This is related to the "autocorrelation minimization" problem in signal processing, but with the twist that we're looking at the linear (not circular) convolution and the non-negativity constraint.

One key insight: if $a$ is a probability density (continuous), then $a*a$ is the density of the sum of two independent random variables. We want this sum density to be as flat as possible. The most "flat" sum density comes from a uniform distribution (the sum of two uniforms is triangular), but we can do better with other distributions.

Actually, for the discrete case with non-negativity, the optimal sequence might resemble a "band-limited" function - one whose Fourier transform is concentrated in a certain band. This is because we want $|A(\omega)|^2$ to be flat, which means $A(\omega)$ should have constant magnitude (but varying phase).

But $A(\omega)$ is the DFT of a non-negative sequence, which severely constrains its phase.

Also, I want to make sure the mirror descent step is correct. The objective in the normalized setting (sum=1) is:
$J(a) = 2n \cdot M(a)$ where $M(a) = \max_k (a*a)_k$

The mirror descent uses the KL divergence as the Bregman distance, which leads to multiplicative updates:
$a_i \leftarrow a_i \cdot \exp(-\eta \cdot g_i)$

where $g_i$ is the subgradient of $M(a)$ at $a$.

The step size $\eta$ should be chosen carefully. Too large and we oscillate; too small and we converge slowly.

A common choice is $\eta_t = c / \sqrt{t}$ (Nesterov) or adaptive based on the observed change.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive mirror descent with broad-peak subgradients,
frequency-domain projection, and local coordinate polish.
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_subgradient(a, ac, frac=0.90, beta=2.0):
    """Subgradient of max(ac) using all peaks above frac*max."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    mask = ac >= frac * M
    idx = np.where(mask)[0]
    vals = ac[idx]
    w = (vals / M) ** beta  # normalized weights
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k_int in zip(w, idx):
        k_int = int(k_int)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n - 1, k_int)
        if i_lo > i_hi:
            continue
        i_vals = np.arange(i_lo, i_hi + 1)
        j1 = k_int - i_vals
        contrib = a[j1].copy()
        j2 = i_vals - k_int
        valid2 = (j2 >= 0) & (j2 < n)
        if valid2.any():
            contrib = contrib.copy()
            contrib[valid2] += a[j2[valid2]]
        same = (j1 == j2)
        if same.any():
            contrib[same] -= a[j1[same]]
        grad[i_vals] += wi * contrib
    return grad

def amd_optimize(a0, time_limit, frac=0.90, beta=2.0, c0=3.0, 
                 restart_every=5000, verbose=False):
    """Adaptive mirror descent with Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    restart_count = 0
    
    for t in range(1, 10_000_000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, frac=frac, beta=beta)
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            restart_count += 1
            # Structured perturbation
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            # Add a low-frequency perturbation
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2*np.pi)
            pert = strength * np.sin(2*np.pi*freq*np.arange(n)/n + phase) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adjust parameters
            frac = max(0.80, frac - 0.02)
            beta = max(1.0, beta - 0.2)
            gamma = c0 * (1 + 0.1 * min(restart_count, 5))
            if verbose:
                print(f"    AMD restart #{restart_count}: J={2*n*best_m:.6f}")
        
        if verbose and t % 10000 == 0:
            print(f"    AMD t={t}: J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m

def freq_flatten(a0, time_limit, n_iters=5000, alpha_start=0.3, verbose=False):
    """Frequency-domain spectrum flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n-1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        alpha = alpha_start * (1.0 + 0.5 * it / max(1, n_iters))
        # Compress dynamic range
        P_new = P_mean * (P / P_mean) ** (1.0 - alpha)
        P_new = np.maximum(P_new, 1e-30)
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass from high-contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Top peaks
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            if i_lo > i_hi:
                continue
            i_vals = np.arange(i_lo, i_hi + 1)
            j1 = k_int - i_vals
            contrib = a[i_vals] * a[j1]
            
            # Find the pair with maximum contribution
            max_idx = np.argmax(contrib)
            src = i_vals[max_idx]
            src_j = j1[max_idx]
            
            if a[src] < 1e-15:
                continue
            
            # Try moving mass from src to a random other position
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    break
                # Try several destinations
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initial(n, kind='cos'):
    x = np.linspace(0, 1, n)
    if kind == 'cos':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif kind == 'sinc':
        t = (x - 0.5) * 4.0
        a = np.sinc(t) ** 2
    elif kind == 'gauss':
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'pow':
        a = (x + 1e-6) ** 0.5
    elif kind == 'multihump':
        a = np.zeros(n)
        for c in [0.25, 0.5, 0.75]:
            a += np.exp(-30*(x-c)**2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the objective."""

def compute_subgradient(a, ac, frac=0.90, beta=2.0):
    """Subgradient of max(ac) using all peaks above frac*max with exponential weights."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    mask = ac >= frac * M
    idx = np.where(mask)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(ac))])
    vals = ac[idx]
    w = (vals / M) ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k_int in zip(w, idx):
        k_int = int(k_int)
        i_lo = max(0, k_int - n + 1)
        i_hi = min(n - 1, k_int)
        if i_lo > i_hi:
            continue
        i_vals = np.arange(i_lo, i_hi + 1)
        j1 = k_int - i_vals
        contrib = a[j1].copy()
        j2 = i_vals - k_int
        valid2 = (j2 >= 0) & (j2 < n)
        if valid2.any():
            contrib[valid2] += a[j2[valid2]]
        same = (j1 == j2)
        if same.any():
            contrib[same] -= a[j1[same]]
        grad[i_vals] += wi * contrib
    return grad


def amd_optimize(a0, time_limit, frac=0.90, beta=2.0, c0=3.0,
                 restart_every=5000, verbose=False, seed=123):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    restart_count = 0
    cur_frac = frac
    cur_beta = beta

    for t in range(1, 20_000_000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, frac=cur_frac, beta=cur_beta)

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))

            x_prev = x_r.copy()
            # Adapt parameters
