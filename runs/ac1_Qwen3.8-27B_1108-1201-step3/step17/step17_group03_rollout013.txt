    """Compute linear autocorrelation using FFT."""

This means the k-th constraint is:

Which is: (x * b)[k] <= M

This is correct: it's the k-th element of the convolution of x and b.

But wait, x is the current iterate (y_t in FW terminology), and we're computing conv(x, b). The FW method requires that we linearize f at the current point. The gradient of f(x) = max_k conv(x,x)[k] at the point x is a subgradient:
∂f(x) = {sum_{k in active} w_k * ∇conv(x,x)[k] : w_k >= 0, sum w_k = 1}

where ∇conv(x,x)[k] is the gradient of the k-th autocorrelation component with respect to x.

The LP in the previous code is NOT solving the standard FW linearization. Instead, it's solving:
max sum(b) s.t. b >= 0, conv(x, b) <= M

This is a different (but related) formulation. The key difference is:
- Standard FW: minimize <g, v> over v in simplex, where g is a subgradient
- LP-FW variant: maximize sum(v) subject to conv(x, v) <= M

The LP-FW variant is actually a valid descent method because:
- If conv(x, v) <= M (where M = max conv(x,x)), then moving toward v reduces the maximum autocorrelation.
- Maximizing sum(v) gives the "most spread out" feasible direction.

This is essentially the Frank-Wolfe method for the problem:
minimize max_k conv(x,x)[k]
over x in simplex

using a non-standard oracle. The oracle finds the point in the "polar" of the feasible set that is farthest from the origin (in the sum norm).

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.005, 0.01, 0.02, 0.04, 0.06, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall % 30 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def multi_peak_cd(a0, time_limit, verbose=True, rng_seed=200):
    """Coordinate descent that attacks multiple peaks simultaneously."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_vals = np.sort(ac)[::-1][:5]
        peak_indices = np.argsort(ac)[::-1][:5]

        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            i_start = max(0, pk - n + 1)
            i_end = min(n, pk + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[pk - idx]

            if contrib.sum() < 1e-20:
                continue

            top_n = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_n]

            for li in top_local:
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                best_move = None
                best_new_m = ac.max()
                for frac in [0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    for _ in range(20):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)
                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
            if improved:
                break

        if not improved:
            stall += 1

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiPeakCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Generate diverse inits and pick best ===
    print(f"\n=== Phase 1: Diverse initialization (n={n0}) ===")
    inits = []
    inits.append(('parent', norm_sum1(best_seq)))
    inits.append(('sine', make_parametric(n0, 'sine')))
    inits.append(('beta_2_2', make_parametric(n0, 'beta_2_2')))
    inits.append(('anti_sym', make_parametric(n0, 'anti_sym')))
    rng_init = np.random.default_rng(0)
    rand_seq = np.abs(rng_init.normal(1.0, 0.5, n0))
    rand_seq = norm_sum1(rand_seq)
    # Smooth the random sequence
    kernel = np.ones(11) / 11.0
    rand_smooth = np.convolve(rand_seq, kernel, mode='same')
    inits.append(('rand_smooth', norm_sum1(rand_smooth)))

    for name, seq in inits:
        sc = evaluate_safe(seq)
        print(f"  Init '{name}': score={sc:.6f}")
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc

    # === Phase 2: LP-FW with line search (main, K=400) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW v2 main (n={len(best_seq)}, K=400) ===")
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 40 and len(best_seq) <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True, top_K=400,
                                      margin_frac=2e-3, use_line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Mirror descent for diverse exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror descent (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t3, top_k=80, beta_pow=4.0,
                                           c0=4.0, restart_every=2000, verbose=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "mirror")

    # === Phase 4: LP-FW with tighter constraints (K=800) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW v2 tight (n={n_best}, K=800) ===")
    t4 = min(remaining * 0.22, 350.0)
    if t4 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True, top_K=800,
                                      margin_frac=5e-4, use_line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: CD polish (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t5, verbose=True, rng_seed=42,
                                             n_top_contrib=25, n_trials=50)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_polish")

    # === Phase 6: Multi-peak CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-peak CD (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_peak_cd(a_in, t6, verbose=True, rng_seed=300)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "multi_cd")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    new_n = n_best + 400
    if remaining > 250 and new_n <= 5500:
        print(f"\n=== Phase 7: Extend to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        sc_long = evaluate_safe(a_long)
        print(f"  Interpolated score: {sc_long:.6f}")

        t7a = min(remaining * 0.20, 200.0)
        if t7a > 40:
            a_out, _ = lp_frank_wolfe_v2(a_long, t7a, solver, verbose=True, top_K=400,
                                          margin_frac=2e-3, use_line_search=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.10, 80.0)
            if t7b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_descent_polish(a_in, t7b, verbose=False, rng_seed=77)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"cd_n{new_n}")

    # === Phase 8: Final LP on absolute best with very tight margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP tight (n={n_best}, K=600) ===")
    t8 = min(remaining * 0.18, 250.0)
    if t8 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t8, solver, verbose=True, top_K=600,
                                      margin_frac=1e-4, use_line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final CD (n={len(best_seq)}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.35, 200.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_polish(a_in, t9, verbose=True, rng_seed=999,
                                                 n_top_contrib=30, n_trials=60)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
