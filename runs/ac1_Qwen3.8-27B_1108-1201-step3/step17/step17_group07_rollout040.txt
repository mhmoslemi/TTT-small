Let me define w[k] = weights[k] for k = 0, ..., 2n-2.
Then sum_{m=0}^{n-1} w[j+m] * a[m] is a cross-correlation of w and a.

This can be computed via FFT! So the full gradient computation is O(n log n).

Let me reformulate:
grad[j] = 2*tau * (w * a)[j] + correction_term

where (w * a)[j] = sum_{m=0}^{n-1} w[j+m] * a[m] (treating w as a "reversed" filter).

Actually, let me define it more carefully. Let's use FFT:
- Compute A = FFT(a, size=2n)
- Compute W = FFT(w, size=2n) where w[k] = weights[k] for k=0,...,2n-2 and w[2n-1]=0
- The cross-correlation c[j] = sum_m w[j+m]*a[m] can be computed as IFFT(W * conj(A))[2n-1-j] or something... 

This is getting complicated. Let me just use the direct O(n*k) approach with top-k weights (since the softmax concentrates mass on a few lags).

Actually, for the log-sum-exp with large tau, the weights concentrate on the few lags with the largest convolution values. So using top-k lags is a good approximation, and the computation is O(n*k).

Let me design the full algorithm:

**Algorithm: Adam + LogSumExp + Mirror Descent Hybrid**

1. Normalize a to sum 1.
2. At each step:
   a. Compute conv = fast_conv(a)
   b. Compute softmax weights over conv (with temperature tau)
   c. Compute gradient using top-k weighted lags
   d. Update using Adam (or mirror descent)
   e. Clip to non-negative, renormalize
3. Adaptively increase tau over time (from 10 to 1000)
4. Periodically restart from best with perturbation

Let me also think about the **shape** of the optimal solution. Based on the theory:
- The optimal sequence likely has a U-shape or arch shape (high at edges, low in middle, or vice versa)
- The exact shape depends on n

For the initialization, I should try:
- Triangle (current best shape)
- U-shape: a[i] = 1 + c*(2*x-1)^2 where x = i/(n-1)
- Arch: a[i] = 1/(1 + c*(2*x-1)^2)
- Cosine: a[i] = cos(pi*x)^2 (U-shaped, 0 at edges)
- Secant: a[i] = 1/cos(pi*x/2)^2 (peaked at edges)

Let me now write the full code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with log-sum-exp smoothed objective
2. Mirror descent with entropy regularization as alternative
3. LP-based Remez for polishing
4. Adaptive temperature scheduling
5. Multi-initialization and multi-length exploration
6. Coordinate descent for final polish
"""

import numpy as np
import cvxpy as cp
import time

# Global solver cache
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_smoothed_grad(a, conv, tau, top_k=50):
    """Compute gradient of (1/tau)*log(sum(exp(tau*conv))) w.r.t. a.
    Uses top-k approximation for efficiency. Returns (lse_value, gradient)."""
    n = len(a)
    max_c = conv.max()
    shifted = tau * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    weights = exp_shifted / sum_exp
    lse_val = (1.0/tau) * (np.log(sum_exp) + tau * max_c)
    
    # Use top-k weights for gradient computation
    k = min(top_k, len(weights))
    top_idx = np.argpartition(weights, -k)[-k:]
    top_weights = weights[top_idx]
    top_idx_sorted = np.sort(top_idx)
    
    grad = np.zeros(n)
    for ki in range(len(top_idx_sorted)):
        k_int = int(top_idx_sorted[ki])
        w = top_weights[ki]
        if w < 1e-12:
            continue
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        idx = np.arange(j_start, j_end + 1)
        kj = k_int - idx
        # d(conv[k])/da[j] = 2*a[k-j] normally, a[j] if k=2j
        contrib = 2.0 * w * a[kj]
        # Fix diagonal terms where 2*j == k
        diag_mask = (2 * idx == k_int)
        if np.any(diag_mask):
            contrib[diag_mask] = w * a[idx[diag_mask]]
        grad[idx] += contrib
    grad *= tau
    return lse_val, grad

def adam_logsumexp(a0, time_limit, tau_init=20.0, tau_final=200.0, 
                   top_k=50, lr=0.01, beta1=0.9, beta2=0.999, 
                   epsilon=1e-8, verbose=False, rng_seed=42):
    """Adam optimizer with log-sum-exp smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    stall = 0
    max_stall = 20000
    
    total_iters = 500000
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(t / (total_iters * 0.8), 1.0)
        tau = tau_init * (tau_final / tau_init) ** progress
        
        conv = fast_conv(a)
        lse_val, grad = compute_smoothed_grad(a, conv, tau, top_k)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to simplex (non-negative, sum=1) using mirror descent step
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        conv_new = fast_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = max(5000, int(max_stall * 0.7))
            pert = rng.normal(0, 0.01, n)
            a = best_a * (1 + pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE t={t} ({elapsed:.1f}s): obj={best_obj:.8f}, tau={tau:.1f}, stall={stall}")
    
    return best_a, best_obj / (2.0 * n)  # Return best_a and best_max_b

def nesterov_mirror_descent_v2(a0, time_limit, top_k=30, c0=3.0, verbose=False, rng_seed=42):
    """Improved Nesterov mirror descent with adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 5000
    rng = np.random.default_rng(rng_seed)
    c = c0
    step_mult = 1.0  # Adaptive multiplier
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_conv(y)
        M = ac.max()
        
        # Find active lags (near-max)
        threshold = M * 0.99
        active = np.where(ac >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(ac)[::-1][:top_k]
        active = np.sort(active)
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            idx = np.arange(j_start, j_end + 1)
            kj = k_int - idx
            contrib = 2.0 * y[kj]
            diag_mask = (2 * idx == k_int)
            if np.any(diag_mask):
                contrib[diag_mask] = y[idx[diag_mask]]
            g[idx] += contrib
        g /= max(1, len(active))
        
        # Mirror descent step
        eta = c * step_mult / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            step_mult = min(2.0, step_mult * 1.01)
        else:
            stall_count += 1
            step_mult = max(0.3, step_mult * 0.999)
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(2000, int(max_stall * 0.6))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.02, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 800), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                # Edge enhancement
                x_r = best_a.copy()
                edge_size = n // 5
                x_r[:edge_size] *= 1.2
                x_r[-edge_size:] *= 1.2
                mid_start = n // 3
                mid_end = 2 * n // 3
                x_r[mid_start:mid_end] *= 0.85
            else:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.5, c * 0.6)
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD2 t={t} ({elapsed:.1f}s): J={J:.8f}, c={c:.3f}, mult={step_mult:.3f}")
    
    return best_a, best_m

def lp_remez_single(a, max_cons=50, r=0.1, solver=None):
    """ LP Remez: minimize max of convolution via linearization."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Active set: lags where b is close to max
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            kj = k_int - idx
            # coeff[j] = 2*a[k-j] normally, a[j] if 2j=k
            coeffs = 2.0 * a[kj]
            diag = (2 * idx == k_int)
            if np.any(diag):
                coeffs = coeffs.copy()
                coeffs[diag] = a[idx[diag]]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Trust region
    constraints.append(a_var >= a * (1.0 - r))
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_method(a0, time_limit, solver=None, r_init=0.1, verbose=False):
    """LP Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_single(a, max_cons=50, r=r, solver=solver)
        
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.5, r * 1.1)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.7)
            if no_improve > 30:
                rng = np.random.default_rng(t)
                idx = rng.choice(n, size=min(n//4, 500), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = r_init
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=123):
    """Randomized coordinate descent with bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_trials = min(200, n // 2)
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            # Try to move mass from i to j (or reduce peak at i)
            for frac in [0.001, 0.005, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    break
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0  # in [-1, 1]
    if kind == 'triangle':
        a = np.maximum(1.0 - np.abs(t), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(np.pi * t)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-2.0 * t ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'U_shape':
        a = 1.0 + 2.0 * t ** 2
    elif kind == 'arch':
        a = 1.0 / (1.0 + 3.0 * t ** 2)
    elif kind == 'quartic':
        a = 1.0 - t ** 4
    elif kind == 'cos_sq':
        a = np.cos(np.pi * x) ** 2  # U-shaped, 0 at edges
    elif kind == 'secant':
        a = 1.0 / np.cos(np.pi * x / 2.0) ** 2  # peaked at edges
        a = np.minimum(a, 10.0)  # Cap to avoid infinity at edges
    elif kind == 'edge_peaked':
        a = 1.0 + 3.0 * t ** 4
    else:
        a = np.maximum(1.0 - np.abs(t), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

Another idea: **Use the fact that the optimal solution likely satisfies certain KKT conditions**. At the optimum, the gradient of the objective with respect to a[j] should be zero (for a[j] > 0) or negative (for a[j] = 0).

The gradient of max(conv) w.r.t. a[j] is:

For boundary points (a[j] = 0), we need g[j] <= lambda.

This suggests that at the optimum, all active coordinates have the same "gradient" value. This is similar to the water-filling principle.

1. Computing the gradient g at the current point
2. Identifying coordinates where a[j] > 0 (active set)
3. For active coordinates, check if g[j] is all approximately equal
4. If not, adjust a to equalize the gradients

def fast_conv(a):

def norm_sum1(a):

def compute_subgrad_active(a, conv, top_k=30):

            break

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        g, active = compute_subgrad_active(y, ac, top_k)

        M_val = ac.max()

        adapt = 1.0 + 0.3 * np.log1p(sharpness) / np.log1p(10.0)

def adam_logsumexp(a0, time_limit, tau_init=15.0, tau_final=200.0,
                   top_k=50, lr=0.005, beta1=0.9, beta2=0.999,
                   epsilon=1e-8, verbose=False, rng_seed=7):
    """Adam optimizer with log-sum-exp smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    stall = 0
    max_stall = 30000

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(t / (300000.0), 1.0)
        tau = tau_init * (tau_final / tau_init) ** progress

        conv = fast_conv(a)
        max_c = conv.max()
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -50.0, 50.0)
        exp_s = np.exp(shifted)
        sum_exp = exp_s.sum()
        weights = exp_s / sum_exp

        # Top-k gradient
        k = min(top_k, len(weights))
        top_idx = np.argpartition(weights, -k)[-k:]
        top_idx_sorted = np.sort(top_idx)

        grad = np.zeros(n)
        for ki in range(len(top_idx_sorted)):
            k_int = int(top_idx_sorted[ki])
            w = weights[k_int]
            if w < 1e-10:
                continue
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            idx = np.arange(j_start, j_end + 1)
            kj = k_int - idx
            contrib = 2.0 * w * a[kj]
            diag = (2 * idx == k_int)
            if np.any(diag):
                contrib = contrib.copy()
                contrib[diag] = w * a[idx[diag]]
            grad[idx] += tau * contrib

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad ** 2)
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        conv_new = fast_conv(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > max_stall:
            stall = 0
            max_stall = max(5000, int(max_stall * 0.7))
            pert = rng.normal(0, 0.015, n)
            a = best_a * (1 + pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE t={t} ({elapsed:.1f}s): J={best_obj:.8f}, tau={tau:.1f}")

    return best_a, best_obj / (2.0 * n)

def lp_remez_single(a, max_cons=50, r=0.1, solver=None):
    """LP Remez: minimize max convolution via linearization with trust region."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            kj = k_int - idx
            coeffs = 2.0 * a[kj]
            diag = (2 * idx == k_int)
            if np.any(diag):
                coeffs = coeffs.copy()
                coeffs[diag] = a[idx[diag]]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    constraints.append(a_var >= a * (1.0 - r))
    constraints.append(a_var <= a * (1.0 + r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_method(a0, time_limit, solver=None, r_init=0.08, verbose=False):
    """LP Remez with adaptive trust region and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        a_cand = lp_remez_single(a, max_cons=50, r=r, solver=solver)

        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.4, r * 1.08)
                break

        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.7)
            if no_improve > 25:
                rng = np.random.default_rng(t)
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = r_init

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")

    return best_a, best_m

def gradient_equalize(a0, time_limit, verbose=False, rng_seed=55):
    """Heuristic: try to equalize subgradient across active coordinates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv(a)
        M = conv.max()
        g, active = compute_subgrad_active(a, conv, top_k=20)

        active_coords = np.where(a > 1e-10)[0]
        if len(active_coords) < 10:
            break

        g_active = g[active_coords]
        g_mean = np.mean(g_active)
        g_std = np.std(g_active)

        if g_std < 1e-12:
            break

        # Proportional adjustment to equalize gradients
        for step_size in [0.02, 0.01, 0.005, 0.001]:
            delta = step_size * (g_mean - g)
            a_new = a + delta * a
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            m_new = fast_conv(a_new).max()
            if m_new < best_m - 1e-15:
                a = a_new
                best_m = m_new
                best_a = a.copy()
                stall = 0
                break
        else:
            stall += 1
            if stall > 100:
                a = best_a * (1 + rng.normal(0, 0.005, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GradEq t={t} ({elapsed:.1f}s): J={J:.8f}, g_std={g_std:.2e}")

    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=123):
    """Randomized coordinate descent with multiple step sizes."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 40)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_trials = min(200, max(50, n // 5))

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            for frac in [0.002, 0.005, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    break
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    if kind == 'triangle':
        a = np.maximum(1.0 - np.abs(t), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(np.pi * t)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-2.0 * t ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'U_shape':
        a = 1.0 + 2.0 * t ** 2
    elif kind == 'arch':
        a = 1.0 / (1.0 + 3.0 * t ** 2)
    elif kind == 'quartic':
        a = 1.0 - t ** 4
    elif kind == 'cos_sq':
        a = np.cos(np.pi * x) ** 2
    elif kind == 'edge_peaked':
        a = 1.0 + 4.0 * t ** 4
    elif kind == 'secant_clip':
        a = 1.0 / (1.0 + 0.1 * np.cos(np.pi * x) ** 2)
    else:
        a = np.maximum(1.0 - np.abs(t), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update_seq(a, best_seq, best_score, t_global, label):
    """Attempt to update the best sequence. Returns (best_seq, best_score, was_updated)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score, False
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
