def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, top_k=60, beta=4.0):
    """Compute soft subgradient of max(ac) with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    idx = np.argsort(ac)[::-1][:k_max]
    vals = np.maximum(ac[idx], 1e-30)
    w = vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        pos_idx = np.arange(i_start, i_end)
        other_idx = k_int - pos_idx
        grad[pos_idx] += 2.0 * wi * a[other_idx]
    return grad


def compute_pressure(a, ac, peak_lag):
    """Compute pressure: how much each element contributes to the peak lag."""
    n = len(a)
    k = int(peak_lag)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    pos_idx = np.arange(i_start, i_end)
    other_idx = k - pos_idx
    pressure = a[pos_idx] * a[other_idx]
    return pos_idx, pressure


def greedy_peak_elimination(a, time_limit, verbose=True, rng_seed=77, max_iter=50000):
    """
    Greedy coordinate descent that redistributes mass from high-pressure
    elements to low-pressure elements to reduce the peak autocorrelation.
    Uses a priority-queue-like approach for efficiency.
    """
    n = len(a)
    a = norm_sum1(a.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 200:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        pos_idx, pressure = compute_pressure(a, ac, peak_k)
        total_p = pressure.sum()
        if total_p < 1e-20:
            stall += 1
            continue

        # Find the highest-pressure source
        src_local = int(np.argmax(pressure))
        src = int(pos_idx[src_local])
        src_val = a[src]
        if src_val < 1e-12:
            stall += 1
            continue

        # Compute destination pressure (how much each element would ADD to the peak if we put mass there)
        # Moving mass from src to dst changes b_k by: -a[other_src]*amt + a[other_dst]*amt (approximately)
        # We want to minimize the increase, so pick dst where a[other_dst] is small
        dst_candidates = np.arange(n)
        # Exclude src
        dst_candidates = dst_candidates[dst_candidates != src]

        # Compute "cost" of putting mass at each dst: a[peak_k - dst] (the partner value)
        partner = np.zeros(n)
        for dst in dst_candidates:
            other = peak_k - dst
            if 0 <= other < n:
                partner[dst] = a[other]
            else:
                partner[dst] = 0.0

        # Try a few fractions of the source mass
        best_move = None
        best_new_m = M_curr

        fracs = [0.002, 0.005, 0.01, 0.02, 0.04]
        # Pick top candidates by lowest partner value
        n_cand = min(20, len(dst_candidates))
        low_partner_idx = np.argsort(partner[dst_candidates])[:n_cand]
        candidate_dsts = dst_candidates[low_partner_idx]

        for frac in fracs:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            for dst in candidate_dsts:
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_new_m - 1e-15:
                    best_new_m = M_trial
                    best_move = (src, int(dst), amt)
            if best_move is not None:
                break  # Found a good move at this fraction, don't try larger

        if best_move is not None:
            src_i, dst_i, amt = best_move
            a[src_i] -= amt
            a[dst_i] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                # Revert
                a[src_i] += amt
                a[dst_i] -= amt
                a = a / a.sum()
                stall += 1
        else:
            # No improvement found; try a random perturbation
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
            if verbose and stall % 50 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    GreedyPE ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    GreedyPE done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=60, beta=4.0, c0=3.5,
                 restart_every=3000, verbose=True):
    """
    Nesterov Mirror Descent with backtracking line search
    and adaptive subgradient parameters.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    step_scale = 1.0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Backtracking on the mirror map step
        eta = gamma * step_scale / (np.sqrt(t) + 1.0)
        base_m = ac.max()
        accepted = False
        for _ in range(8):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            step_scale = min(step_scale * 1.02, 3.0)  # slightly increase step
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Structured restart: perturb in the direction opposite to the subgradient
            ac_best = fast_autocorr(best_a)
            g_best = compute_subgradient(best_a, ac_best, top_k=cur_top_k, beta=cur_beta)
            log_pert = np.log(np.maximum(best_a, 1e-15)) + 0.02 * rng.standard_normal(n) - 0.01 * g_best / (g_best.max() + 1e-15)
            x_r = np.exp(log_pert - log_pert.max())
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 100)
            cur_beta = max(cur_beta - 0.3, 2.0)
            step_scale = max(step_scale * 0.8, 0.3)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, topk={cur_top_k}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=2e-3, nesterov=True):
    """
    LP Frank-Wolfe with adaptive margin and Nesterov acceleration.
    The LP maximizes sum(b) subject to convolution constraints <= M.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    lp_fail_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active constraints adaptively
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, int(k_int) - n + 1)
            i_end = min(n, int(k_int) + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[int(k_int) - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 10:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 10:
                break
            continue

        lp_fail_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.9, 1e-5)

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def freq_perturb(a, strength=0.02, rng=None):
    """Perturb in the frequency domain to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    # Perturb phases more at high frequencies
    freq_weights = np.linspace(0.5, 1.5, len(phase))
    phase_pert = phase + rng.normal(0, strength / freq_weights)
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'gaussian':
        x = np.linspace(0, 1, n)
        mu = 0.5
        sigma = 0.25
        return norm_sum1(np.exp(-0.5 * ((x - mu) / sigma) ** 2))
    elif family == 'uniform':
        return np.ones(n) / n
    elif family == 'beta_sym':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x * (1 - x) + 0.05)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 6: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 300, n_best + 600, n_best + 1000]))
    longer_lengths = [l for l in longer_lengths if l <= 5500 and remaining > 150]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 6: Extend to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.15, 150.0)
        if t6a > 30:
            a_out, _ = amd_optimize(a_long, t6a, top_k=50, beta=4.0, c0=3.0,
                                     restart_every=2000, verbose=False)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4500:
            t6b = min(remaining * 0.10, 100.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=True, top_K=200,
                                           margin_frac=2e-3)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 7: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final AMD polish (n={n_best}) ===")
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t7, top_k=80, beta=3.0, c0=2.5,
                                restart_every=5000, verbose=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_final")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.12, 150.0)
    if t8 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=350,
                                   margin_frac=1e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final greedy polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final greedy (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.25, 200.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = greedy_peak_elimination(a_in, t9, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "greedy_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
