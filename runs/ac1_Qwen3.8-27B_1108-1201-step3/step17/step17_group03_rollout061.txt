    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'bell':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.exp(-4.0 * (x - 0.5) ** 2))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def compute_smooth_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    Uses FFT for O(n log n) computation."""
    n = len(a)
    c_max = c.max()
    log_weights = beta * (c - c_max)
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = weights / w_sum

    # g[i] = (2/beta) * sum_k p[k] * a[k-i] where 0<=k-i<n
    # This is (p * a_reflected) extracted appropriately
    # Equivalent to cross-correlation: g[i] = (2/beta) * sum_m p[m+i] * a[m]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # corr[i] = sum_m p[m+i] * a[m]
    # = IFFT(conj(P) * A) ... need to be careful
    # Actually: correlation(p, a)[i] = sum_j p[j+i]*a[j]
    # FFT method: IFFT(conj(FFT(p_padded)) * FFT(a_padded))[:n]
    corr = np.fft.irfft(np.conj(P) * A, fft_size)[:n]
    g = (2.0 / beta) * corr
    return g


def smoothed_mirror_descent(a0, time_limit, verbose=True, beta_start=5.0,
                            beta_end=30.0, lr=2.0, momentum_start=0.0,
                            momentum_end=0.7, max_iter=50000,
                            restart_every=3000, rng_seed=42):
    """Smoothed mirror descent with Nesterov momentum on log-domain."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Schedule
        frac = min(t / max_iter, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        momentum = momentum_start + (momentum_end - momentum_start) * frac
        eta = lr / (np.sqrt(t) + 1.0) * (beta_start / max(beta, 1.0))

        # Nesterov: y = x + momentum * (x - x_prev)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and gradient
        c = fast_autocorr(y)
        g = compute_smooth_gradient(y, c, beta)

        # Mirror descent step in log domain
        log_a = np.log(np.maximum(y, 1e-300)) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            # Perturb in log domain
            log_pert = rng.normal(0, 0.01, n) * np.log(np.maximum(best_a, 1e-15))
            x_r = np.exp(np.log(np.maximum(best_a, 1e-15)) + log_pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SaMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=300,
                   margin_frac=2e-3, validate=True, max_inner=5):
    """LP Frank-Wolfe with optional constraint validation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active constraints
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix (sparse-equivalent)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with validation
        alpha = 2.0 / (t + 2.0)
        accepted = False

        for inner in range(max_inner):
            a_trial = (1.0 - alpha) * y + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial / s_t

            if validate:
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial <= M_curr + 1e-12:
                    accepted = True
                    break
                else:
                    # Add violated constraints
                    new_violated = np.where(ac_trial > M_curr)[0]
                    if len(new_violated) > 0:
                        # Add some of them
                        to_add = new_violated[:min(50, len(new_violated))]
                        extra_T = np.zeros((len(to_add), n))
                        for row, k_int in enumerate(to_add):
                            i_start = max(0, k_int - n + 1)
                            i_end = min(n, k_int + 1)
                            for i in range(i_start, i_end):
                                extra_T[row, i] = y[k_int - i]
                        # Re-solve with additional constraints
                        b_var2 = cp.Variable(n, nonneg=True)
                        constraints2 = [T_sel @ b_var2 <= M_curr,
                                        extra_T @ b_var2 <= M_curr]
                        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                        try:
                            prob2.solve(solver=solver, verbose=False)
                        except Exception:
                            break
                        if prob2.status in ('optimal', 'optimal_inaccurate'):
                            b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                            b_sum2 = b_val.sum()
                            if b_sum2 < 1e-15:
                                break
                            g_dir = b_val / b_sum2
                            continue
                    # If validation keeps failing, accept anyway
                    break
            else:
                accepted = True
                break

        if not accepted:
            a_trial = (1.0 - alpha) * y + alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                a_trial = y.copy()
            else:
                a_trial = a_trial / s_t

        # Nesterov update for x
        y_new = a_trial
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten margin adaptively
            margin_frac = max(margin_frac * 0.85, 1e-5)

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def directed_peak_crush(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-crushing with directed moves based on autocorrelation structure."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Find source: index i that contributes most to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        # For each source candidate, find best destination
        n_src = min(10, len(contrib))
        top_src = np.argsort(contrib)[::-1][:n_src]

        # Precompute the "cost" of moving mass to each position
        # Moving from src to dst changes c[peak_k] by:
        # delta = amt * (a[peak_k - dst] + a[dst - peak_k] - a[peak_k - src] - a[src - peak_k])
        # We want delta < 0, so we want a[peak_k-dst] + a[dst-peak_k] < a[peak_k-src] + a[src-peak_k]

        best_move = None
        best_delta = 0

        for sli in top_src:
            src = int(idx[sli])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            src_cost = a[peak_k - src] if 0 <= peak_k - src < n else 0.0
            src_cost2 = a[src - peak_k] if 0 <= src - peak_k < n else 0.0
            src_total_cost = src_cost + src_cost2

            # Find best destination: minimize dst_cost
            dst_cost_arr = np.zeros(n)
            for d in range(n):
                c1 = a[peak_k - d] if 0 <= peak_k - d < n else 0.0
                c2 = a[d - peak_k] if 0 <= d - peak_k < n else 0.0
                dst_cost_arr[d] = c1 + c2

            # Best destinations are those with lowest cost
            dst_candidates = np.argsort(dst_cost_arr)[:20]
            for dst in dst_candidates:
                dst = int(dst)
                if dst == src:
                    continue
                dst_total_cost = dst_cost_arr[dst]
                delta_coeff = dst_total_cost - src_total_cost
                if delta_coeff >= 0:
                    continue
                # Try a few fractions
                for frac in [0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    # Quick estimate of change in M
                    est_delta = amt * delta_coeff
                    if est_delta < best_delta:
                        # Verify with actual computation (expensive but accurate)
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        actual_delta = M_trial - M_curr
                        if actual_delta < best_delta:
                            best_delta = actual_delta
                            best_move = (src, dst, amt)

        if best_move is not None:
            src_i, dst_i, amt = best_move
            a[src_i] -= amt
            a[dst_i] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.005, 0.03))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

                                                   lr=1.5, momentum_start=0.0,
                                                   momentum_end=0.5, max_iter=40000,
                                                   restart_every=2500, rng_seed=300+trial)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 8: Final peak-crush on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Peak-Crush (n={n_best}) ===")
    if remaining > 40:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = directed_peak_crush(a_in, t8, verbose=True, rng_seed=555)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final: One more SaMD polish ===
    remaining = time_budget - (time.monotonic() - t0_global) if False else time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Final polish: SaMD (n={n_best}) ===")
        t_final = min(remaining * 0.4, 100.0)
        if t_final > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = smoothed_mirror_descent(a_in, t_final, verbose=False, beta_start=30.0,
                                               beta_end=60.0, lr=0.8, momentum_start=0.1,
                                               momentum_end=0.4, max_iter=30000,
                                               restart_every=2000, rng_seed=777)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "smd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
