```python
import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def lse_objective(a, beta):
    n = len(a)
    conv = fast_conv(a)
    bc = beta * conv
    bc -= bc.max()
    return float(np.log(np.exp(bc).sum()))

def lse_gradient(a, beta):
    n = len(a)
    conv = fast_conv(a)
    bc = beta * conv
    bc -= bc.max()
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    
    # FFT size for cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(w, N)
    
    # grad = beta * (ifft(conj(A)*P)[:n] + ifft(A*P)[:n])
    #      = beta * ifft((A + conj(A)) * P)[:n]
    #      = beta * ifft(2*Re(A) * P)[:n]
    # But with rfft, A is complex. Let me just compute both halves.
    
    corr1 = np.fft.irfft(np.conj(A) * P, N)[:n]
    corr2 = np.fft.irfft(A * P, N)[:n]
    
    return beta * (corr1 + corr2)

def optimize_lse(a0, time_limit, beta_start=10.0, beta_end=500.0, 
                 beta_schedule='linear', c0=1.0, verbose=False):
    """Minimize log-sum-exp smoothed objective using mirror descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    
    # Beta schedule: gradually increase beta
    n_steps = 5000  # total iterations
    current_beta = beta_start
    
    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update beta (log-linear schedule)
        frac = t / n_steps
        log_beta = np.log(beta_start) + frac * (np.log(beta_end) - np.log(beta_start))
        current_beta = np.exp(log_beta)
        
        # Compute gradient
        g = lse_gradient(a, current_beta)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent update (KL projection onto simplex)
        log_r = np.log(a + 1e-30) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()
        
        a = a_new
        
        # Track best
        val = eval_J(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    return best_a, best_val
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (LSE) smoothed mirror descent with continuation method
2. FFT-based smooth gradients (exact, not top-K approximation)
3. Multi-strategy initialization including flat-top window shapes
4. Adaptive step sizes with Nesterov acceleration
5. Coordinate descent fine-tuning with pressure-based transfers
6. Length sweeping with interpolation
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming a is normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def lse_gradient(a, beta):
    """
    Compute exact gradient of log(sum(exp(beta * conv_k))) with respect to a.
    Uses FFT for efficiency. O(n log n) complexity.
    """
    n = len(a)
    conv = fast_conv(a)
    # Softmax weights
    bc = beta * conv
    bc -= bc.max()
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()

    # FFT size: need N >= 3n-2 to avoid wrap-around
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(w, N)

    # corr1[i] = sum_j a[j] * w[i+j]  (cross-correlation)
    corr1 = np.fft.irfft(np.conj(A) * P, N)[:n]
    # corr2[i] = sum_j a[j] * w[i-j]  (cross-correlation, other direction)
    corr2 = np.fft.irfft(A * P, N)[:n]

    return beta * (corr1 + corr2)


def topk_subgradient(a, conv, top_k=100, beta_w=10.0):
    """Compute soft subgradient using top-K lags (from previous code, refined)."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent_lse(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                       n_steps=10000, c0=2.0, verbose=False):
    """
    Mirror descent with LSE-smoothed objective and continuation method.
    Gradually increases beta from beta_start to beta_end.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0

    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Beta schedule: log-linear
        frac = min(t / n_steps, 1.0)
        log_beta = np.log(beta_start) + frac * (np.log(beta_end) - np.log(beta_start))
        current_beta = np.exp(log_beta)

        # Nesterov extrapolation
        if t > 1:
            momentum = (t - 1.0) / (t + 1.0)
            y = x + momentum * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute LSE gradient
        g = lse_gradient(y, current_beta)

        # Adaptive step size: scale by gradient norm
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            eta = c0 / (np.sqrt(t) + 1.0)
        else:
            eta = c0 * y.max() / (g_norm + 1e-10)
            eta = min(eta, c0 / (np.sqrt(t) * 0.1 + 1.0))

        # Mirror descent update
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > 3000:
            stall = 0
            restart_count += 1
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            if verbose:
                print(f"    LSE-MD restart {restart_count} at t={t}: J={best_val:.6f}")

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-MD t={t} ({elapsed:.0f}s): J={best_val:.6f}, beta={current_beta:.0f}")

    return best_a, best_val


def mirror_descent_topk(a0, time_limit, top_k=80, beta_w=10.0, c0=4.0,

        # Subgradient
        conv = fast_conv(y)
        g = topk_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        # Step size
        eta = cur_c0 / (np.sqrt(t) + 1.0)

        # Mirror update
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        val = eval_J(a_new)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.03)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 0.5, 3.0)
            cur_c0 = max(cur_c0 * 0.92, 1.0)
            if verbose:
                print(f"    TK-MD restart t={t}: J={best_val:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TK-MD t={t} ({elapsed:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with pressure-based mass transfer and random restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)

        # Find top lags
        n_lags = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        # Compute pressure at each position
        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        # Source: highest pressure * mass
        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-14:
            stall += 1
            continue

        # Destination: minimize impact
        dest_scores = np.zeros(n)
        for k in top_lags[:6]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_scores[idx] += a[k_i - idx]
        dest_scores[src] = np.inf
        dest_scores += 0.05 * pressure
        dst = int(np.argmin(dest_scores))

        # Try transfer fractions
        improved = False
        for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.1, 0.2, 0.35]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-9:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall > 100:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    CD ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_flat_top_window(n):
    """Flat-top window (flat frequency response, non-negative)."""
    x = np.arange(n)
    w = np.zeros(n)
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    for k in range(n):
        w[k] = a0 - a1 * np.cos(2 * np.pi * k / (n - 1)) \
               + a2 * np.cos(4 * np.pi * k / (n - 1)) \
               - a3 * np.cos(6 * np.pi * k / (n - 1)) \
               + a4 * np.cos(8 * np.pi * k / (n - 1))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def make_cosine_ushape(n, power=1.0):
    """U-shaped cosine distribution: high at edges, low in center."""
    x = np.linspace(0, 1, n)
    a = (2 * x - 1) ** 2  # U-shape
    a = a ** power + 0.01
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal: two Gaussians near edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.08) ** 2) + np.exp(-width * (x - 0.92) ** 2) + 0.003
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_nuttall(n):
    """Nuttall window."""
    x = np.arange(n)
    a0, a1, a2, a3 = 0.3635819, 0.4891775, 0.1365995, 0.0106412
    w = a0 - a1 * np.cos(2 * np.pi * x / (n - 1)) \
        + a2 * np.cos(4 * np.pi * x / (n - 1)) \
        - a3 * np.cos(6 * np.pi * x / (n - 1))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def make_gaussian_dip(n, depth=0.5, width=0.2):
    """Gaussian dip in the center."""
    x = np.linspace(0, 1, n)
    a = 1.0 - depth * np.exp(-((x - 0.5) / width) ** 2)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_blackman(n):
    """Blackman window."""
    x = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """
    Main optimization driver.
    Strategy:
      Phase 1: Multi-init LSE mirror descent (continuation method)
      Phase 2: Multi-init top-K mirror descent
      Phase 3: Coordinate descent
      Phase 4: Length sweep
      Phase 5: Final polish (alternating LSE + CD)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent loaded: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_flat_top_window(n0)
        best_score = eval_J(best_a)
        print(f"Initial (flat-top): n={len(best_a)}, J={best_score:.6f}")

    n_best = len(best_a)

    # Generate diverse initializations
    def get_inits(n):
        inits = []
        labels = []
        if parent is not None and len(parent) == n:
            inits.append(parent.copy())
            labels.append("parent")
        inits.append(make_flat_top_window(n))
        labels.append("flattop")
        inits.append(make_nuttall(n))
        labels.append("nuttall")
        inits.append(make_blackman(n))
        labels.append("blackman")
        inits.append(make_bimodal(n, width=60))
        labels.append("bimodal60")
        inits.append(make_bimodal(n, width=120))
        labels.append("bimodal120")
        inits.append(make_cosine_ushape(n, power=1.0))
        labels.append("cosin

eU1")
        inits.append(make_cosine_ushape(n, power=2.0))
        labels.append("cosineU2")
        inits.append(make_gaussian_dip(n, depth=0.4, width=0.15))
        labels.append("gauss_dip")
        inits.append(make_gaussian_dip(n, depth=0.6, width=0.25))
        labels.append("gauss_dip2")
        inits.append(make_smooth_random(n, seed=7))
        labels.append("rand7")
        inits.append(make_smooth_random(n, seed=13))
        labels.append("rand13")
        # Blends
        if parent is not None and len(parent) == n:
            b1 = 0.5 * parent + 0.5 * make_flat_top_window(n)
            b1 = np.maximum(b1, 0) / np.maximum(b1.sum(), 1e-15)
            inits.append(b1)
            labels.append("blend_parent_ftw")
            b2 = 0.3 * parent + 0.7 * make_bimodal(n, 80)
            b2 = np.maximum(b2, 0) / np.maximum(b2.sum(), 1e-15)
            inits.append(b2)
            labels.append("blend_parent_bim")
        return inits, labels

    # === Phase 1: LSE Mirror Descent (multi-init) ===
    print(f"\n{'='*50}")
    print(f"Phase 1: LSE Mirror Descent (n={n_best})")
    print(f"{'='*50}")

    inits, labels = get_inits(n_best)
    t_per_init = max(20.0, (time_budget * 0.25) / len(inits))

    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 120:
            break
        t_avail = min(t_per_init, time_budget - 120 - elapsed)
        if t_avail < 15:
            break
        print(f"  LSE-MD [{lbl}] budget={t_avail:.0f}s ...", end=" ")
        a_out, v = mirror_descent_lse(init, t_avail, beta_start=50, beta_end=3000,
                                       n_steps=5000, c0=2.0, verbose=False)
        print(f"J={v:.6f}")
        try_update(a_out, f"lse_{lbl}")

    # === Phase 2: Top-K Mirror Descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n{'='*50}")
    print(f"Phase 2: Top-K MD (n={len(best_a)})")
    print(f"{'='*50}")

    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        print(f"  TK-MD on current best, budget={t2:.0f}s")
        a_out, v = mirror_descent_topk(best_a.copy(), t2, top_k=80, beta_w=10.0,
                                        c0=4.0, restart_every=5000, verbose=True)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "tk_md_1")

    # Also try TK-MD with different params
    remaining = time_budget - (time.monotonic() - t_global)
    t2b = min(remaining * 0.10, 120.0)
    if t2b > 15:
        print(f"  TK-MD (wider) on current best, budget={t2b:.0f}s")
        a_out, v = mirror_descent_topk(best_a.copy(), t2b, top_k=150, beta_w=6.0,
                                        c0=3.0, restart_every=4000, verbose=False)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "tk_md_wide")

    # === Phase 3: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n{'='*50}")
    print(f"Phase 3: Coordinate Descent (n={len(best_a)})")
    print(f"{'='*50}")

    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        print(f"  CD budget={t3:.0f}s")
        a_out, v = coordinate_descent(best_a.copy(), t3, verbose=True, seed=99)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "cd_1")

    # === Phase 4: LSE again after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n{'='*50}")
    print(f"Phase 4: LSE after CD (n={len(best_a)})")
    print(f"{'='*50}")

    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        print(f"  LSE-MD budget={t4:.0f}s")
        a_out, v = mirror_descent_lse(best_a.copy(), t4, beta_start=200,
                                       beta_end=8000, n_steps=5000, c0=1.5, verbose=False)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "lse_after_cd")

    # === Phase 5: Length Sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n{'='*50}")
    print(f"Phase 5: Length Sweep (best n={len(best_a)})")
    print(f"{'='*50}")

    n_best = len(best_a)
    lengths_to_try = []
    for d in [-500, -250, -100, 100, 250, 500, 750, 1000]:
        nl = n_best + d
        if 1000 <= nl <= 6000:
            lengths_to_try.append(nl)
    for nl in [2000, 2500, 3000, 3500, 4000, 4500, 5000]:
        if nl not in lengths_to_try and 1000 <= nl <= 6000:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  n={nl}...", end=" ")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.20, 80.0)
        if t_md > 12:
            a_out, v = mirror_descent_lse(a_int, t_md, beta_start=100,
                                           beta_end=5000, n_steps=3000, c0=2.0, verbose=False)
            print(f"LSE J={v:.6f}", end=" ")
            try_update(a_out, f"len{nl}_lse")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                t_cd = min(remaining * 0.10, 40.0)
                if t_cd > 10:
                    a_out2, v2 = coordinate_descent(interpolate_seq(best_a, nl),
                                                     t_cd, verbose=False, seed=nl)
                    print(f"CD J={v2:.6f}")
                    try_update(a_out2, f"len{nl}_cd")
                    continue
        print()

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n{'='*50}")
    print(f"Phase 6: Final Polish (n={n_cur})")
    print(f"{'='*50}")

    if remaining > 30:
        t6a = min(remaining * 0.35, 180.0)
        print(f"  LSE-MD (high beta) budget={t6a:.0f}s")
        a_out, v = mirror_descent_lse(best_a.copy(), t6a, beta_start=500,
                                       beta_end=10000, n_steps=10000, c0=1.0, verbose=True)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "final_lse")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t6b = min(remaining * 0.35, 150.0)
        print(f"  CD final budget={t6b:.0f}s")
        a_out, v = coordinate_descent(best_a.copy(), t6b, verbose=True, seed=555)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "final_cd")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t6c = min(remaining - 15, 200.0)
        print(f"  TK-MD final budget={t6c:.0f}s")
        a_out, v = mirror_descent_topk(best_a.copy(), t6c, top_k=200, beta_w=5.0,
                                        c0=2.5, restart_every=10000, verbose=True)
        print(f"  Result: J={v:.6f}")
        try_update(a_out, "final_tkmd")

    # Final validation
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\n{'='*50}")
        print(f"FINAL: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        print(f"{'='*50}")
        return [float(x) for x in best_a]
    else:
        n_fb = 3000
        fb = make_flat_top_window(n_fb)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
```
