Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP-based Frank-Wolfe with Nesterov acceleration
2. Proximal FW variant with simplex projection
3. Targeted peak-shaving local search
4. Multi-length exploration with interpolation
5. Asymmetric initialization exploiting left-right anti-correlation
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_subproblem_all(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Uses ALL constraints."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    # Build constraint matrix (sparse-friendly)
    # Row k: sum_j a[k-j] * b[j] <= M for valid j
    # We build this as a dense matrix product for small n
    if n <= 5000:
        # Build the constraint matrix
        # Row k, col j: coefficient is a[k-j] if 0 <= k-j < n, else 0
        # Number of rows: 2n-1
        num_rows = 2 * n - 1
        T = np.zeros((num_rows, n), dtype=np.float64)
        for k in range(num_rows):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                T[k, j] = a[k - j]
        constraints.append(T @ b_var <= M)
    else:
        # For very large n, use top constraints only
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(3000, num_rows)]
        T = np.zeros((len(top_k), n), dtype=np.float64)
        for idx, k in enumerate(top_k):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T[idx, j] = a[ki - j]
        constraints.append(T @ b_var <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_subproblem_topk(a, M, solver, top_K=1000):
    """Solve LP with only top-K most binding constraints."""
    n = len(a)
    num_rows = 2 * n - 1
    top_K = min(top_K, num_rows)

    ac = fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:top_K]

    T = np.zeros((top_K, n), dtype=np.float64)
    for idx, k in enumerate(top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        for j in range(j_start, j_end):
            T[idx, j] = a[ki - j]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def afw_optimize(a0, time_limit, solver, top_K=None, use_all=False,
                 nesterov=True, verbose=True, rng_seed=42):
    """Accelerated Frank-Wolfe with optional all-constraint LP."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    y = a.copy()
    iter_count = 0
    stall = 0
    max_stall = 500

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Solve LP subproblem
        if use_all and n <= 4000:
            b_dir = lp_subproblem_all(y, M_curr, solver)
        else:
            b_dir = lp_subproblem_topk(y, M_curr, solver, top_K=top_K or min(2000, 2*n-1))

        if b_dir is None:
            stall += 1
            continue

        iter_count += 1

        if nesterov and t > 1:
            alpha = 2.0 / (t + 2.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if nesterov:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        else:
            x_new = y_new.copy()

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            stall = 0

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AFW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

        if stall > max_stall:
            # Restart with perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            y_r = best_a + pert
            y_r = np.maximum(y_r, 1e-12)
            y_r = y_r / y_r.sum()
            y = y_r
            x = y_r.copy()
            stall = 0
            max_stall = min(max_stall + 100, 2000)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    AFW final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def proximal_fw(a0, time_limit, verbose=True, lr_schedule='exp', rng_seed=13):
    """Proximal FW: gradient step + simplex projection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    lr = 1.0 / np.sqrt(n)
    stall = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Subgradient: average of top-20 peaks' gradients
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        weights = np.maximum(ac[top_k], 1e-30) ** 8
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            j = np.arange(j_start, j_end)
            grad[j] += 2.0 * w * a[ki - j]

        # Gradient step + projection
        if lr_schedule == 'exp':
            lr_t = lr * np.exp(-0.0001 * t)
        else:
            lr_t = lr / (1 + 0.01 * t)

        a_new = a - lr_t * grad
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                pert = rng.normal(0, 0.01, n) * best_a
                a_p = best_a + pert
                a_p = np.maximum(a_p, 0)
                a_p = a_p / a_p.sum()
                M_p = fast_autocorr(a_p).max()
                if M_p < best_m:
                    a = a_p
                    best_m = M_p
                    best_a = a_p.copy()
                    stall = 0
                else:
                    stall = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ProxFW t={t} ({elapsed:.0f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    ProxFW final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, rng_seed=55):
    """Targeted peak-shaving: identify responsible pairs and redistribute mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    n_peaks = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        improved = False

        # Build sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:6]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            # Find top contributing pairs
            order = np.argsort(contrib)[::-1][:5]
            for oi in order:
                src = idx[oi]
                src_partner = partner[oi]

                # Determine which to reduce
                imp_src = sens[src]
                imp_prt = sens[src_partner]
                if imp_src >= imp_prt:
                    reduce_idx = src
                    keep_idx = src_partner
                else:
                    reduce_idx = src_partner
                    keep_idx = src

                if a[reduce_idx] < 1e-12:
                    continue

                # Find best destination (low sensitivity, not in peak)
                exclude = {reduce_idx, keep_idx}
                mask = np.ones(n, dtype=bool)
                mask[list(exclude)] = False
                candidates = np.where(mask & ~in_peak)[0]
                if len(candidates) == 0:
                    candidates = np.where(mask)[0]
                if len(candidates) == 0:
                    continue

                cand_sens = sens[candidates]
                # Try a few destinations
                dest_order = np.argsort(cand_sens)[:10]

                for di in dest_order[:5]:
                    dst = candidates[di]
                    found = False
                    for frac in [0.005, 0.01, 0.02, 0.003, 0.03, 0.008]:
                        amt = a[reduce_idx] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_idx] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if improved:
                    break

            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                n_pert = min(10, n)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                for pi in idx_pert:
                    if a[pi] > 1e-10:
                        j = int(rng.integers(n))
                        if j != pi:
                            frac = rng.uniform(0.005, 0.03)
                            amt = a[pi] * frac
                            a[pi] -= amt
                            a[j] += amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakShave final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_asymmetric(n, concentration=0.3, seed=0):
    """Create an asymmetric sequence with mass concentrated toward one side."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Mass concentrated toward x=concentration
    a = np.exp(-((x - concentration) / 0.15) ** 2)
    # Add asymmetry
    a *= (1.0 + 0.5 * np.sin(2 * np.pi * x))
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_ramp(n, decay=0.5):
    """Geometric ramp (asymmetric)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-decay * n * x)
    return norm_sum1(a)


def make_dual_peak(n, pos1=0.2, pos2=0.7, width=0.1):
    """Two peaks at asymmetric positions."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + 0.8 * np.exp(-((x - pos2) / width) ** 2)
    return norm_sum1(a)


def subgradient_mirror_descent(a0, time_limit, top_k=30, beta_exp=6.0,
                                c0=5.0, verbose=False, rng_seed=99):
    """Mirror descent with soft subgradient (exponential weighting of peaks)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_exp
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            j = np.arange(j_start, j_end)
            grad[j] += 2.0 * wi * y[ki - j]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: AFW with all constraints on parent (if feasible) ===
    print(f"\n=== Phase 1: AFW all-constraints (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 60 and n0 <= 4500:
        a_in = norm_sum1(best_seq)
        # Use all constraints for smaller n, top-K for larger
        use_all = (n0 <= 3500)
        top_K = min(3000, 2 * n0 - 1)
        a_out, _ = afw_optimize(a_in, t1, solver, top_K=top_K, use_all=use_all,
                                 nesterov=True, verbose=True, rng_seed=42)
        try_update(a_out, "afw_parent")

    # === Phase 2: AFW with diverse initializations ===
    print(f"\n=== Phase 2: Multi-init AFW ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = min(remaining * 0.20, 280.0)

    if t2_total > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_asymmetric(n0, concentration=0.25, seed=1))
        labels.append("asym_025")

        inits.append(make_asymmetric(n0, concentration=0.75, seed=2))
        labels.append("asym_075")

        inits.append(make_dual_peak(n0, pos1=0.15, pos2=0.70, width=0.12))
        labels.append("dualpeak")

        inits.append(make_ramp(n0, decay=0.8))
        labels.append("ramp")

        # Blend parent with asymmetric
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_asymmetric(n0, 0.3, seed=3)
            inits.append(norm_sum1(blend))
            labels.append("blend_asym")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t2_total / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  AFW init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = afw_optimize(init, t_avail, solver,
                                     top_K=min(2000, 2 * n0 - 1), use_all=False,
                                     nesterov=True, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"afw_{lbl}")

    # === Phase 3: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak shaving (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "peakshave")

    # === Phase 4: AFW round 2 on updated best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: AFW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30 and n_best <= 4500:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 3500)
        top_K = min(4000, 2 * n_best - 1)
        a_out, _ = afw_optimize(a_in, t4, solver, top_K=top_K, use_all=use_all,
                                 nesterov=True, verbose=True, rng_seed=200)
        try_update(a_out, "afw_r2")

    # === Phase 5: Proximal FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Proximal FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = proximal_fw(a_in, t5, verbose=False, lr_schedule='exp', rng_seed=31)
        try_update(a_out, "prox_fw")

    # === Phase 6: Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Mirror Descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_mirror_descent(a_in, t6, top_k=40, beta_exp=8.0,
                                                c0=4.0, verbose=False, rng_seed=55)
        try_update(a_out, "mirror_descent")

    # === Phase 7: Peak shaving again ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Peak shaving round 2 ===")
    n_best = len(best_seq)
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t7, verbose=True, rng_seed=99)
        try_update(a_out, "peakshave2")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")

    trial_lengths = []
    for delta in [300, 600, -300, 900, 1200]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.4, 200.0)
        if t8a > 30:
            use_all = (new_n <= 3500)
            top_K = min(3000, 2 * new_n - 1)
            a_out, _ = afw_optimize(a_interp, t8a, solver, top_K=top_K, use_all=use_all,
                                     nesterov=True, verbose=False, rng_seed=new_n % 997)
            try_update(a_out, f"afw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.3, 80.0)
            if t8b > 20:
                a_out, _ = peak_shaving(norm_sum1(best_seq), t8b, verbose=False, rng_seed=new_n % 500)
                try_update(a_out, f"peak_n{new_n}")

    # === Phase 9: Final AFW with all constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final AFW (n={n_best}) ===")
    if remaining > 40 and n_best <= 4500:
        t9 = min(remaining * 0.4, 200.0)
        if t9 > 30:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 3500)
            top_K = min(5000, 2 * n_best - 1)
            a_out, _ = afw_optimize(a_in, t9, solver, top_K=top_K, use_all=use_all,
                                     nesterov=True, verbose=True, rng_seed=777)
            try_update(a_out, "afw_final")

    # === Phase 10: Final peak shaving + coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final polish (n={n_best}) ===")
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shaving(a_in, t10, verbose=False, rng_seed=1234)
            try_update(a_out, "peak_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
