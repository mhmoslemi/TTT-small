    rng = np.random.default_rng(seed)
    # Generate random phases
    L = 1 << (2*n - 2).bit_length()  # FFT size
    phases = rng.uniform(0, 2*np.pi, L)

    A = np.exp(1j * phases)
    # Make it Hermitian (for real output)
    A[:L//2] = np.exp(1j * rng.uniform(0, 2*np.pi, L//2))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    J_new = eval_J(a)
                    if J_new < best_J - 1e-15:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if verbose and stall % 80 == 0:
            print(f"    PCD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_u_shape(n, p=1.0, sharpness=1.0):
    """U-shaped (bathtub) distribution."""
    x = np.linspace(0, 1, n)
    a = x ** p + (1.0 - x) ** p
    a = a ** sharpness
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=60.0):
    """Bimodal distribution with two peaks."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.08) ** 2) + np.exp(-width * (x - 0.92) ** 2) + 0.003
    return a / a.sum()


def make_asymmetric(n, p=1.2, shift=0.25):
    """Asymmetric distribution."""
    x = np.linspace(0, 1, n)
    a = ((x + shift) ** p) * ((1.0 - x + 0.01) ** 0.4)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_spectral_init(n, seed=0, alpha=0.5):
    """Initialize by flattening spectrum of a random signal."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    # Random signal
    a = rng.uniform(0.1, 1.0, n)
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    mag_mean = mag[1:].mean()
    if mag_mean > 1e-15:
        new_mag = mag ** alpha * (mag_mean ** (1.0 - alpha))
        new_mag[0] = mag[0]
        phase = np.angle(A)
        a = np.fft.irfft(new_mag * np.exp(1j * phase), fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_u_shape(n0, p=1.0, sharpness=1.0)
        best_score = eval_J(best_a)
        print(f"Initial U-shape: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)

    # === Phase 1: Multi-init mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD Multi-Init (n={n0}), budget={remaining:.0f}s ===")

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_u_shape(n0, p=0.8, sharpness=1.0))
    labels.append("u0.8")
    inits.append(make_u_shape(n0, p=1.2, sharpness=1.0))
    labels.append("u1.2")
    inits.append(make_u_shape(n0, p=1.0, sharpness=2.0))
    labels.append("u_sharp")
    inits.append(make_power_ramp(n0, 0.5))
    labels.append("ramp0.5")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp1.5")
    inits.append(make_bimodal(n0, width=50.0))
    labels.append("bimodal")
    inits.append(make_spectral_init(n0, seed=7, alpha=0.4))
    labels.append("spec7")
    inits.append(make_spectral_init(n0, seed=42, alpha=0.6))
    labels.append("spec42")
    inits.append(make_smooth_random(n0, seed=13))
    labels.append("smooth13")
    inits.append(make_asymmetric(n0, p=1.0, shift=0.2))
    labels.append("asym")

    if parent is not None:
        b1 = 0.5 * parent + 0.5 * make_u_shape(n0, 1.0, 1.0)
        inits.append(b1 / b1.sum())
        labels.append("blend_u")
        b2 = 0.7 * parent + 0.3 * make_spectral_init(n0, seed=99, alpha=0.5)
        inits.append(b2 / b2.sum())
        labels.append("blend_s")

    # Deduplicate
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 80)], 4))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    # Allocate time: give more time to parent and best candidates
    t_per_base = max(12.0, remaining * 0.32 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 90:
            break
        # Give parent and U-shape more time
        multiplier = 1.5 if lbl in ("parent", "u1.0", "u_sharp") else 1.0
        t_avail = min(t_per_base * multiplier, time_budget - 90 - elapsed)
        if t_avail < 8:
            break
        print(f"  MD [{lbl}] ({t_avail:.0f}s)...")
        seed_val = 100 * idx + 7
        a_out, _ = mirror_descent_v2(init, t_avail, top_k=100, beta_w=8.0,
                                      c0=3.5, restart_every=4000, seed=seed_val)
        try_update(a_out, f"md_{lbl}")

    # === Phase 2: Extended MD on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended MD (n={len(best_a)}), budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 25:
        a_out, _ = mirror_descent_v2(best_a.copy(), t2, top_k=120, beta_w=7.0,
                                      c0=3.0, restart_every=6000, verbose=True, seed=200)
        try_update(a_out, "md_ext")

    # === Phase 3: Paired coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: PCD (n={len(best_a)}), budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.12, 160.0)
    if t3 > 15:
        a_out, _ = paired_coord_descent(best_a.copy(), t3, verbose=True, seed=77)
        try_update(a_out, "pcd1")

    # === Phase 4: MD after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MD after CD, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.10, 130.0)
    if t4 > 15:
        a_out, _ = mirror_descent_v2(best_a.copy(), t4, top_k=150, beta_w=6.0,
                                      c0=2.5, restart_every=8000, seed=300)
        try_update(a_out, "md_after_cd")

    # === Phase 5: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length Sweep, budget={remaining:.0f}s ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-300, -150, 150, 300, 500, 700]:
        nl = n_best + d
        if 1500 <= nl <= 5500:
            lengths_to_try.append(nl)
    for nl in [2000, 2500, 3000, 3500, 4000, 4500]:
        if nl not in lengths_to_try and 1500 <= nl <= 5500:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length sweep: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        # Quick MD
        t_md = min(remaining * 0.30, 90.0)
        if t_md > 12:
            a_out, _ = mirror_descent_v2(a_int, t_md, top_k=100, beta_w=8.0,
                                          c0=3.0, restart_every=4000, seed=nl)
            try_update(a_out, f"len{nl}_md")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and len(best_a) != nl:
            t_cd = min(remaining * 0.12, 45.0)
            if t_cd > 10:
                a_cur = interpolate_seq(best_a, nl)
                a_out, _ = paired_coord_descent(a_cur, t_cd, verbose=False, seed=nl + 500)
                try_update(a_out, f"len{nl}_cd")

    # === Phase 6: Final polish MD with high top-K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 6: Final MD Polish (n={n_cur}), budget={remaining:.0f}s ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 20:
        a_out, _ = mirror_descent_v2(best_a.copy(), t6, top_k=200, beta_w=5.0,
                                      c0=2.0, restart_every=10000, verbose=True, seed=500)
        try_update(a_out, "final_md")

    # === Phase 7: Final PCD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final PCD, budget={remaining:.0f}s ===")
    t7 = min(remaining * 0.25, 140.0)
    if t7 > 12:
        a_out, _ = paired_coord_descent(best_a.copy(), t7, verbose=True, seed=314)
        try_update(a_out, "final_pcd")

    # === Phase 8: Extra MD if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 8: Extra MD, budget={remaining:.0f}s ===")
        a_out, _ = mirror_descent_v2(best_a.copy(), remaining - 20,
                                      top_k=200, beta_w=4.0, c0=1.5,
                                      restart_every=12000, verbose=False, seed=999)
        try_update(a_out, "extra_md")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_u_shape(n_fb, p=1.0, sharpness=1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
