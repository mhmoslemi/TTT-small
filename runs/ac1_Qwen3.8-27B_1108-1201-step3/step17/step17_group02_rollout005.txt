def fast_autocorr(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K_init=400, 
                      constraint_growth=30, max_constraints=None, rng_seed=99):
    """

    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

        max_constraints = min(2 * n - 1, 5000)

    x = a.copy()
    
    # Active constraint set

    def update_active_set(y_curr, top_K):
        """Update the active constraint set based on current margins."""
        ac = fast_autocorr(y_curr)
        M_curr = ac.max()
        if M_curr < 1e-16:
            return set()
        # Select constraints that are within a margin of the peak

            cand_sorted = candidates

            y = y / sy

        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w_exp = top_vals ** cur_beta
        w_exp = w_exp / w_exp.sum()
        w_rank = 1.0 / (np.arange(k_max) + 1.0) ** 0.7
        w_rank = w_rank / w_rank.sum()
        w = 0.6 * w_exp + 0.4 * w_rank

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = np.max(log_a)
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.93, 3.0)
            cur_top_k = min(cur_top_k + 3, 200)

        if stall_count > restart_every:
            stall_count = 0
            scale = 0.04 + 0.02 * np.sqrt(min(cur_top_k / 100.0, 3.0))
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 250)
            cur_beta = max(cur_beta - 1.0, 2.5)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD2 t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    n_top = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            else:
                candidates = candidates[np.argsort(sens[candidates])[:min(50, len(candidates))]]

            found = False
            for dst in candidates[:20]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.003, 0.015]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.015)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            n_pert = min(3, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord2 ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord2 final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def peak_equalization_lp(a0, time_limit, solver, verbose=True, n_peaks=8, rng_seed=77):
    """Try to equalize the top peaks by solving LPs that target specific peak reductions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for outer_iter in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        peak_vals = ac[top_peaks]
        
        # Target: reduce the maximum peak by solving LP with tightened constraint
        target_M = M_curr * 0.995  # Try to reduce by 0.5%
        
        # Include all top peaks as constraints
        direction = solve_lp_direction(a, top_peaks.tolist(), target_M, solver)
        
        if direction is None:
            # Try slightly less aggressive
            target_M = M_curr * 0.998
            direction = solve_lp_direction(a, top_peaks.tolist(), target_M, solver)
        
        if direction is None:
            # Fallback: use all constraints within 20% of peak
            margins = M_curr - ac
            thresh = M_curr * 0.8
            candidates = np.where(ac >= thresh)[0]
            if len(candidates) > 2000:
                candidates = candidates[np.argsort(ac[candidates])[::-1]][:2000]
            direction = solve_lp_direction(a, candidates.tolist(), M_curr * 0.999, solver)
        
        if direction is None:
            # Small random perturbation
            pert = rng.normal(0, 0.01, n) * a
            a_new = a + pert
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
            continue
        
        alpha = exact_line_search(a, direction, max_iter=15)
        a_new = (1.0 - alpha) * a + alpha * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        M_new = fast_autocorr(a_new).max()
        a = a_new
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        if verbose and outer_iter % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PEQ iter={outer_iter} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_u_shape(n_base, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick AMD warm-up ===
    print(f"\n=== Phase 1: AMD warm-up (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)
    if t_phase1 > 30:
        # Try AMD from parent and a U-shape
        a_in = norm_sum1(best_seq)
        t_half = t_phase1 / 2.0
        a_out, _ = amd_optimize_v2(a_in, t_half, top_k=80, beta=7.0, c0=5.0,
                                   restart_every=4000, verbose=False, rng_seed=42)
        try_update(a_out, "amd_parent")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_u = make_u_shape(n0, sharpness=2.0)
            a_out, _ = amd_optimize_v2(a_u, min(t_half, remaining * 0.8),
                                        top_k=80, beta=6.0, c0=4.5,
                                        restart_every=4000, verbose=False, rng_seed=7)
            try_update(a_out, "amd_u")

    # === Phase 2: Main LP-FW ===
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60 and n_best <= 6000:
        max_K = min(8000, 2 * n_best - 1)
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True,
                                      top_K_init=min(600, 2 * n_best - 1),
                                      constraint_growth=30, max_K=max_K,
                                      margin_frac=0.12, reselect_every=25,
                                      use_line_search=True, rng_seed=55)
        try_update(a_out, "lp_main")

    # === Phase 3: Peak equalization LP ===
    print(f"\n=== Phase 3: Peak equalization (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_equalization_lp(a_in, t3, solver, verbose=True,
                                         n_peaks=10, rng_seed=77)
        try_update(a_out, "peq")

    # === Phase 4: LP-FW round 2 with more constraints ===
    print(f"\n=== Phase 4: LP-FW round 2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 60 and n_best <= 6000:
        max_K = min(10000, 2 * n_best - 1)
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True,
                                      top_K_init=min(1000, 2 * n_best - 1),
                                      constraint_growth=40, max_K=max_K,
                                      margin_frac=0.08, reselect_every=20,
                                      use_line_search=True, rng_seed=111)
        try_update(a_out, "lp_r2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (base n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    trial_lengths = []
    for delta in [-400, -200, 200, 400, 600]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")
        
        t7a = min(remaining * 0.30, 150.0)
        if t7a > 20:
            a_out, _ = amd_optimize_v2(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                        restart_every=4000, verbose=False, rng_seed=new_n % 997)
            try_update(a_out, f"amd_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t7b = min(remaining * 0.40, 180.0)
            if t7b > 30:
                max_K = min(6000, 2 * new_n - 1)
                a_in = norm_sum1(best_seq)
                # Interpolate a_in to new_n for LP
                a_in_new = interpolate_sequence(a_in, new_n)
                a_out, _ = lp_frank_wolfe_v2(a_in_new, t7b, solver, verbose=False,
                                              top_K_init=min(500, 2 * new_n - 1),
                                              constraint_growth=30, max_K=max_K,
                                              margin_frac=0.12, reselect_every=25,
                                              use_line_search=True, rng_seed=new_n % 777)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP-FW on best ===
    print(f"\n=== Phase 6: Final LP-FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 6000:
        t6 = min(remaining * 0.45, 300.0)
        if t6 > 30:
            max_K = min(12000, 2 * n_best - 1)
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t6, solver, verbose=True,
                                          top_K_init=min(1500, 2 * n_best - 1),
                                          constraint_growth=50, max_K=max_K,
                                          margin_frac=0.05, reselect_every=15,
                                          use_line_search=True, rng_seed=222)
            try_update(a_out, "lp_final")

    # === Phase 7: Coordinate polish ===
    print(f"\n=== Phase 7: Coordinate polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20 and n_best <= 8000:
        t7 = min(remaining * 0.7, 150.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish_v2(a_in, t7, verbose=True, rng_seed=456)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_u_shape(n_fb, sharpness=2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
