"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with soft-max smoothing and softmax parameterization
2. Annealed soft-max parameter for progressive refinement
3. Multiple asymmetric initializations
4. Coordinate descent polishing with exact 1D optimization
Key insight: np.convolve(a,a) is NOT the autocorrelation; it's the cross-correlation
of a with its reverse. Non-symmetric sequences can have lower peak convolution
than sum(a^2), enabling objectives below 2.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Top-level helper functions

def fft_convolve(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_stable(x):
    """Stable softmax."""
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def compute_smoothed_obj_grad(x, tau):
    """
    Compute f(x) = 2n * soft_max(conv(a,a), tau) and gradient.
    a = softmax(x).
    """
    n = len(x)
    a = softmax_stable(x)
    c = fft_convolve(a)
    
    # Soft-max
    c_over_tau = c / tau
    c_max = np.max(c_over_tau)
    exp_c = np.exp(c_over_tau - c_max)
    exp_sum = exp_c.sum()
    S = tau * (np.log(exp_sum) + c_max)
    p = exp_c / exp_sum
    
    # g_a[i] = 2 * sum_m a[m] * p[i+m]
    # = 2 * cross_corr(a, p)[i] where cross_corr(a,p)[i] = sum_m a[m]*p[i+m]
    # Compute via: conv(a[::-1], p)[n-1:i+n-1]
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1  # = 3n - 2
    fft_size2 = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size2)
    P = np.fft.rfft(p, fft_size2)
    cross_full = np.fft.irfft(A * P, fft_size2)
    g_a = 2.0 * cross_full[n-1 : 2*n - 1]
    
    # Gradient w.r.t. x: g_x[i] = a[i] * (g_a[i] - dot(a, g_a))
    d = np.dot(a, g_a)
    g_x = a * (g_a - d)
    
    f = 2.0 * n * S
    return f, g_x


def nesterov_subgradient(a0, time_limit, top_k=15, beta=3.0, c_param=3.0,
                          exact_frac=0.4, restart_every=3000, seed=42):

    n = len(a0)

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fft_convolve(y)
        M = ac.max()
        
        use_exact = rng.random() < exact_frac
        if use_exact:
            k = int(np.argmax(ac))
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            g = np.zeros(n)
            g[idx] = 2.0 * y[j_idx]
        else:
            k_max = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** beta
            w = w / w.sum()
            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * y[j_idx]
        
        eta = c_param * (M - 1.0/n) / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)
        
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new

def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fft_convolve(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):

def make_ramp_init(n, alpha=1.0):
    """Linear ramp initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * x / (n - 1)
    return a / a.sum()


def make_exp_ramp_init(n, alpha=2.0):
    """Exponential ramp initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * x / (n - 1))
    return a / a.sum()


def make_skewed_gauss_init(n, center=0.7, sigma=0.2):
    """Skewed Gaussian (off-center)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-((x - center) / sigma) ** 2)
    return a / a.sum()


def make_random_smooth_init(n, seed=0):
    """Random smooth initialization (low-pass filtered white noise)."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Simple box filter for smoothing
    window = 10
    kernel = np.ones(window) / window
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / a.sum()


def lbfgs_optimize(a_init, tau_schedule, max_iter=500, tol=1e-8, time_limit=300):
    """
    Run L-BFGS with a schedule of tau values.
    a_init: initial sequence (will be converted to log-space).
    tau_schedule: list of tau values (decreasing).
    """
    n = len(a_init)
    # Initialize in log-space
    a_init = np.maximum(a_init, 1e-10)
    x = np.log(a_init / a_init.sum())
    x = x - x.mean()  # center to prevent drift
    
    best_a = a_init.copy()
    best_score = compute_J(a_init)
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def func_and_grad(x):
            f, g = compute_smoothed_obj_grad(x, tau)
            return f, g
        
        result = minimize(
            func_and_grad,
            x,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter,
                'ftol': tol,
                'gtol': 1e-6,
                'maxls': 50,
            }
        )
        
        if time.monotonic() - t0 > time_limit:
            break
        
        # Convert back
        a_new = softmax_stable(result.x)
        J_new = compute_J(a_new)
        
        if J_new < best_score:
            best_score = J_new
            best_a = a_new.copy()
        
        # Update x for next tau
        x = result.x - result.x.mean()
        
        print(f"    tau={tau:.6f}: J={best_score:.8f}, n_iter={result.nit}")
    
    return best_a, best_score
