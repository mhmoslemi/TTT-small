Let me now think about a **dual approach**. The primal problem is:

The dual problem is:
$$\max_{w \geq 0, \sum w_k = 1} \min_{a: \sum a_i = 1, a_i \geq 0} a^T (\sum_k w_k B_k) a$$

By strong duality (if it holds), the optimal value of the primal equals the optimal value of the dual. But since the problem is non-convex (as we discussed), strong duality might not hold exactly. """
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth mirror descent with adaptive softmax-temperature subgradient
2. LP-based Frank-Wolfe with adaptive constraint selection and line search
3. 1D coordinate descent (mass transfer) with golden section search

import numpy as np
import time

def _get_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def _autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _eval(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _smooth_subgrad(a, c, tau):
    """Compute smooth subgradient using softmax weights."""
    n = len(a)
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    
    # Weighted sum of shifted a
    grad = np.zeros(n)
    for k in range(len(c)):
        if p[k] < 1e-15:
            continue
        # grad[i] += 2 * p[k] * a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * p[k] * a[j_idx]
    return grad

def _mirror_descent_smooth(a0, time_limit, tau0=10.0, tau_max=500.0, c0=3.0, verbose=False):
    """Smooth mirror descent with adaptive temperature."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    tau = tau0
    gamma = c0
    stall = 0
    max_stall = 5000
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        tau = min(tau_max, tau0 * (1.0 + 0.001 * t))
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute smooth gradient
        c = _autocorr(y)
        g = _smooth_subgrad(y, c, tau)
        
        # Mirror descent step

def _coord_descent(a0, time_limit, verbose=False, rng_seed=42):
    """1D coordinate descent via mass transfer with golden section."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    golden = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = _autocorr(a)
        M_curr = c.max()
        peaks = np.argsort(c)[::-1][:min(20, len(c))]
        
        improved = False
        
        for k_peak in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            kp = int(k_peak)
            
            # For this peak, the contributing pairs are (i, kp-i)
            i_start = max(0, kp - n + 1)
            i_end = min(n - 1, kp)
            
            if i_start > i_end:
                continue
            
            # Find the pair with maximum contribution
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[kp - idx]
            max_i = idx[np.argmax(contrib)]
            
            if a[max_i] < 1e-10:
                continue
            
            # Try transferring mass from max_i to various destinations
            # Golden section over transfer fraction
            lo = 0.0
            hi = min(1.0, a[max_i] / (a.sum() * 0.5))  # Can't transfer more than half
            
            if hi < 1e-6:
                continue
            
            # Evaluate at golden section points
            c1 = lo + (1 - golden) * (hi - lo)
            c2 = lo + golden * (hi - lo)
            
            best_frac = 0
            best_trial_m = M_curr
            
            for _ in range(8):  # 8 golden section steps
                a_c1 = a.copy()
                a_c1[max_i] -= a[max_i] * c1
                a_c1[np.argmax(a)] += a[max_i] * c1  # Transfer to max position
                a_c1 = _normalize(a_c1)
                m1 = _autocorr(a_c1).max()
                
                a_c2 = a.copy()
                a_c2[max_i] -= a[max_i] * c2
                a_c2[np.argmax(a)] += a[max_i] * c2
                a_c2 = _normalize(a_c2)
                m2 = _autocorr(a_c2).max()
                
                if m1 < m2:
                    hi = c2
                    c2 = c1
                    _c2 = _c1
                    c1 = lo + (1 - golden) * (hi - lo)
                else:
                    lo = c1
                    c1 = c2
                    c2 = lo + golden * (hi - lo)
                
                if m1 < best_trial_m:
                    best_trial_m = m1
                    best_frac = c1
                if m2 < best_trial_m:
                    best_trial_m = m2
                    best_frac = c2
            
            if best_frac > 0 and best_trial_m < M_curr - 1e-12:
                a_new = a.copy()
                a_new[max_i] -= a[max_i] * best_frac
                # Distribute to multiple low positions
                dsts = np.argsort(a)[:5]
                for d in dsts:
                    if d != max_i:
                        a_new[d] += a[max_i] * best_frac / len(dsts)
                a_new = _normalize(a_new)
                M_new = _autocorr(a_new).max()
                if M_new < best_m:
                    a = a_new
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.01)
                a[i] -= a[i] * frac
                a[j] += a[i] * frac
                a = _normalize(a)
                M_new = _autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
            else:
                if stall > 50:
                    break
    
    return best_a, best_m

def _autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _eval(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_subgrad(a, c, tau, top_k=None):
    """Compute smooth subgradient using softmax weights."""
    n = len(a)
    if top_k is not None and top_k < len(c):
        # Only use top-k for efficiency
        top_idx = np.argsort(c)[::-1][:top_k]
        c_work = c[top_idx]
        others = c.max() - c_work.min()
    else:
        top_idx = np.arange(len(c))
        c_work = c
    
    c_max = c_work.max()
    exp_c = np.exp(tau * (c_work - c_max))
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for ki, k in enumerate(top_idx):
        if p[ki] < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * p[ki] * a[j_idx]
    return grad

def _mirror_descent_smooth(a0, time_limit, tau0=5.0, tau_max=500.0, c0=3.0, top_k=200, verbose=False):
    """Smooth mirror descent with adaptive temperature and Nesterov momentum."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    tau = tau0
    gamma = c0
    stall = 0
    max_stall = 6000
    
    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature: slowly increase
        tau = min(tau_max, tau0 * (1.0 + 0.0005 * t))
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute smooth gradient
        c = _autocorr(y)
        g = _smooth_subgrad(y, c, tau, top_k=top_k)
        
        # Mirror descent step with clipping for stability
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        # Clip to prevent overflow/underflow
        log_a = np.clip(log_a - log_a.max(), -50, 50)
        a_new = np.exp(log_a)
        a_new = _normalize(a_new)
        
        x_prev = x
        x = a_new
        
        M_new = _autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Restart with structural perturbation
            pert = rng.normal(0, 0.03, n) * best_a
            x = _normalize(best_a + pert)
            x_prev = x.copy()
            stall = 0
            tau = max(tau0, tau * 0.5)
            gamma = c0 * (1.0 + 0.1 * min(stall // max_stall, 5))
            max_stall = int(max_stall * 1.3)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, stall={stall}")
    
    return best_a, best_m

def _fista_projected(a0, time_limit, c0=2.0, top_k=100, verbose=False):
    """FISTA with Euclidean projection onto simplex."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    L = 8.0  # Lipschitz constant estimate
    
    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = _autocorr(y)
        g = _smooth_subgrad(y, c, tau=20.0, top_k=top_k)
        
        # Gradient step + projection
        eta = 1.0 / L
        z = y - eta * g
        x_new = _proj_simplex(z)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y_new = _normalize(y_new)
        
        x_prev = x
        x = x_new
        y = y_new
        t_k = t_new
        
        # Adaptive L
        M_new = _autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            L = max(0.5, L * 0.98)
        elif t % 500 == 0:
            L = min(50.0, L * 1.05)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.0f}s): J={J:.6f}, L={L:.2f}")
    
    return best_a, best_m

def _lp_fw(a0, time_limit, solver, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp
    
    y = a.copy()
    x = a.copy()
    
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    
    K = 80
    max_K = min(400, 2 * n - 1)
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Fill constraint matrix
        np.fill_diagonal(T, 0)  # Ensure clean state (T is reused)
        for j in range(n):
            T[j:j + n, j] = y
        
        c = _autocorr(y)
        M_curr = c.max()
        if M_curr < 1e-16:
            break
        
        # Select binding constraints: tightest + some random
        margins = M_curr - c
        K_cur = min(K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_cur]
        
        # Add random constraints for diversity
        n_rand = max(10, K_cur // 8)
        rng = np.random.default_rng(t * 7 + 13)
        rand_idx = rng.choice(2 * n - 1, size=n_rand, replace=False)
        binding_idx = np.unique(np.concatenate([binding_idx, rand_idx]))[:K_cur]
        
        T_sel = T[binding_idx, :]
        M_sel = M_curr * np.ones(len(binding_idx))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe step with line search
        best_alpha = 2.0 / (t + 2.0)
        best_obj = M_curr
        for alpha in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1]:
            y_trial = (1 - alpha) * y + alpha * g_dir
            y_trial = _normalize(y_trial)
            M_trial = _autocorr(y_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = _normalize(y_new)
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = _normalize(x_new)
        else:
            x_new = y_new.copy()
        
        y = y_new
        x = x_new
        
        # Check both
        M_y = _autocorr(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
        M_x = _autocorr(x).max()
        if M_x < best_m:
            best_m = M_x
            best_a = x.copy()
        
        # Adaptive K
        if t % 50 == 0:
            K = min(max_K, K + 15)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")
    
    return best_a, best_m

def _coord_descent(a0, time_limit, verbose=False, rng_seed=42):
    """1D coordinate descent via mass transfer."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = _autocorr(a)
        M_curr = c.max()
        n_peaks = min(12, len(c))
        peaks = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        
        for kp in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(kp)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            
            # Try transferring to several low-cost destinations
            # Cost: how much this position contributes to peak lags
            costs = np.zeros(n)
            for k2 in peaks[:6]:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                if js2 > je2:
                    continue
                idx2 = np.arange(js2, je2 + 1)
                costs[idx2] += a[k2i - idx2]
            
            # Destination: low cost, low value
            candidates = np.argsort(costs + a)[:10]
            
            for dst in candidates:
                if dst == src:
                    continue
                
                # Try several transfer fractions
                for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = _normalize(a_trial)
                    M_trial = _autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration when stuck
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = _normalize(a)
                M_new = _autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def _make_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return _normalize(a)

def _make_beta_dist(n, alpha=2.0, beta=1.0):
    x = np.linspace(0.001, 0.999, n)
    a = x ** (alpha - 1) * (1.0 - x) ** (beta - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def _make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.001)
    return _normalize(a)

def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return _normalize(f_interp)
