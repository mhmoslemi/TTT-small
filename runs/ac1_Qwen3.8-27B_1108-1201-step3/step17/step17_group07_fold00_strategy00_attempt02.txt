    if new_length > n:
        try:
            interpolate(a_norm, new_length)
        except:
            # Fallback to longer sequence if interpolation fails
            pass
    return remez_poly(a_norm, degree=3)

def coordinate_descend_from_conv(a, mode='initial'):
    """

    :param a: Sequence to improve
    :param mode: either 'initial' or 'final' determines strategy
    :return: Improved sequence
    """
    n = len(a)
    a_norm = safe_norm(a)
    if mode == 'initial':
        sorted_convention = np.argsort(fast_conv(a))[-5:][::-1]  # top 5 convention indices
    else:
        sorted_convention = np.argsort(fast_conv(a))[:n//5]     # least important convention

    # Initialize a descent sequence with all zeros except for top convention points
    descent_sequence = np.zeros(n)
    descent_sequence[sorted_convention[:min(20, len(sorted_convention))]] = 1 / min(20, n)

    # Coordinate descent using local only violations
    # Instead of global variables, use local iteration
    for i in range(n):
        # Skip positions already active in a
        if i == -1 or i >= len(a) or a[i] < 0.01:
            continue
        # Create local bump allowed by convention
        a_local = a_norm.copy()
        a_local[i] = max(0.01, a_norm[i])  # allow 1% of other a to be here

        # Update convention-based on the new bump
        conv_updated = fast_conv(a_local)
        if conv_updated > 0.99 * max(fast_conv(a_norm)):
            # This violation is too bad - try next
            continue
        a_local_sum = sum(a_local)
        a_new = safe_norm(a_local)
        return a_new

    # If no improvement, return the original sequence
    return a_norm
