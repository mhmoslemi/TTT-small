def _fast_autocorr(a):

def _compute_subgrad(a, beta=10.0):
    """

    Uses exp(beta*(ac - max)) weighting over all lags.
    Returns (gradient, max_value).

    w = np.exp(beta * (ac - M))

        w = top_mask.astype(np.float64)

        if w_sum < 1:
            w = np.ones_like(ac)
            w_sum = len(ac)
        w /= w_sum

def _interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _make_smooth_init(n, seed=0, n_freqs=5):
    """Create a smooth non-negative initialization via random Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freqs + 1):
        coeff = rng.normal(0, 0.05 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _fast_evaluate(a):
    """Quick evaluation using FFT (avoids calling evaluate_sequence for speed)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = _fast_autocorr(a)
    M = ac.max()
    return float(2.0 * n * M / (s * s))


def _final_polish(a0, time_limit, verbose=True):
    """
    Final polishing: alternate between short MD runs and coordinate descent
    with decreasing perturbation scales.
    """
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()

    phases = [
        ("md", 0.4),
        ("cd", 0.3),
        ("md", 0.15),
        ("cd", 0.15),
    ]

    for phase_type, frac in phases:
        if time.monotonic() - t0 > time_limit:
            break
        t_avail = frac * time_limit - (time.monotonic() - t0)
        if t_avail < 5:
            break

        if phase_type == "md":
            a_out, M_out = _nesterov_mirror_descent(
                a, min(t_avail, 120.0), beta_init=15.0, c_step=2.0,
                restart_every=2000, seed=777, verbose=False)
        else:
            a_out, M_out = _coordinate_descent(
                a, min(t_avail, 120.0), n_top_peaks=15, seed=555, verbose=False)

        if M_out < best_M:
            best_M = M_out
            best_a = a_out.copy()
            a = a_out

    return best_a, best_M


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization: subgradient descent -> coordinate descent ->
    multi-scale exploration -> length interpolation -> final polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = _fast_evaluate(a)
        # Cross-check with actual evaluator
        try:
            sc_real = evaluate_sequence([float(x) for x in a])
            if np.isfinite(sc_real):
                sc = sc_real
        except Exception:
            pass
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _fast_evaluate(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = _make_smooth_init(n_base, seed=1, n_freqs=7)
        best_score = _fast_evaluate(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Primary Nesterov mirror descent (40% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov MD (n={n0}) ===")
    t1 = remaining * 0.40
    if t1 > 60:
        a_out, _ = _nesterov_mirror_descent(
            best_seq, t1, beta_init=12.0, c_step=5.0,
            restart_every=5000, seed=42, verbose=True)
        try_update(a_out, f"nmd_main_n{n0}")

        # Second run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.15, t1 * 0.4)
        if t1b > 30:
            a_out2, _ = _nesterov_mirror_descent(
                best_seq, t1b, beta_init=20.0, c_step=3.0,
                restart_every=3000, seed=137, verbose=False)
            try_update(a_out2, f"nmd_sharp_n{n0}")

    # === Phase 2: Multi-beta exploration (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-beta (n={n0}) ===")
    t2 = remaining * 0.15
    if t2 > 60:
        a_out, _ = _multi_beta_descent(best_seq, t2, seeds=[7, 42, 99], verbose=False)
        try_update(a_out, f"multibeta_n{n0}")

    # === Phase 3: Coordinate descent (20% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate Descent (n={n0}) ===")
    t3 = remaining * 0.20
    if t3 > 30:
        a_out, _ = _coordinate_descent(
            best_seq, t3, n_top_peaks=25, seed=123, verbose=True)
        try_update(a_out, f"cd_n{n0}")

    # === Phase 4: Final polish (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Final Polish (n={n0}) ===")
    t4 = remaining * 0.10
    if t4 > 20:
        a_out, _ = _final_polish(best_seq, t4, verbose=True)
        try_update(a_out, f"polish_n{n0}")

    # === Phase 5: Try nearby lengths (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 100, n_best + 100, n_best - 200, n_best + 200,
                         n_best - 400, n_best + 400]
    candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000]
    # Remove duplicates and sort by distance from n_best
    seen = set()
    unique_lengths = []
    for l in sorted(candidate_lengths, key=lambda x: abs(x - n_best)):
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)

    t_per_length = max(20.0, remaining * 0.15 / len(unique_lengths[:3]))
    for new_n in unique_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"\n=== Phase 5: Length n={new_n} ===")
        a_interp = _interpolate_seq(best_seq, new_n)
        t_avail = min(t_per_length, remaining * 0.4)
        if t_avail > 15:
            a_out, _ = _nesterov_mirror_descent(
                a_interp, t_avail, beta_init=10.0, c_step=4.0,
                restart_every=3000, seed=777, verbose=False)
            try_update(a_out, f"nmd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_cd = min(remaining * 0.3, 60.0)
            if t_cd > 10:
                a_out, _ = _coordinate_descent(
                    best_seq if len(best_seq) == new_n else a_interp,
                    t_cd, n_top_peaks=15, seed=333, verbose=False)
                try_update(a_out, f"cd_n{new_n}")

    # === Phase 6: Final short MD on best (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 6: Final MD (n={n_best}) ===")
        t6 = remaining * 0.6
        a_out, _ = _nesterov_mirror_descent(
            best_seq, t6, beta_init=15.0, c_step=2.0,
            restart_every=2000, seed=2024, verbose=False)
        try_update(a_out, f"final_md_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        # Final cross-check with actual evaluator
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = _make_smooth_init(n_fb, seed=1)
        best_score = _fast_evaluate(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
