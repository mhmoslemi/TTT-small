For the quadratic form: a^T Q_k a = sum_{i,j} Q_k[i,j] a[i] a[j]

The matrix Q_k is a symmetric matrix with 1/2 on a specific anti-diagonal and possibly 1 on the diagonal.

Is Q_k PSD? The eigenvalues of a matrix with 1/2 on one anti-diagonal are... For a 2×2 example: Q = [[0, 1/2], [1/2, 0]], eigenvalues ±1/2. NOT PSD!

import time
import cvxpy as cp
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_global_solver = None

def get_cvxpy_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  CVXPY solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_lp_matrix(a, active_idx, n):
    """Build the constraint matrix A where A[k_idx, j] = a[active_val - j] if valid."""
    K = len(active_idx)
    A = np.zeros((K, n))
    for row in range(K):
        k = int(active_idx[row])
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        # A[row, j] = a[k - j] for j in [j_start, j_end)
        # a[k-j] where j goes from j_start to j_end-1
        # k-j goes from k-j_start down to k-(j_end-1)
        idx = np.arange(j_start, j_end)
        A[row, idx] = a[k - idx]
    return A


def solve_lp(a, active_idx, M, n):
    """Solve: maximize sum(b) s.t. (a*b)[k] <= M for k in active_idx, b >= 0."""
    K = len(active_idx)
    if K == 0:
        return np.ones(n) / n
    
    A = build_lp_matrix(a, active_idx, n)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            b_val = np.maximum(res.x, 0)
            if b_val.sum() > 1e-10:
                return b_val / b_val.sum()
    except Exception:
        pass
    
    # Fallback: uniform
    return np.ones(n) / n


def line_search(a, b_dir, n, n_grid=25):
    """1D line search: minimize max_conv((1-t)*a + t*b_dir) over t in [0,1]."""
    best_t = 0.0
    best_m = fast_conv(a).max()
    
    # Coarse grid
    alphas = np.linspace(0, 1, n_grid)
    vals = []
    for alpha in alphas:
        a_trial = (1 - alpha) * a + alpha * b_dir
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            vals.append(np.inf)
            continue
        a_trial = a_trial / s
        vals.append(fast_conv(a_trial).max())
    
    vals = np.array(vals)
    best_i = np.argmin(vals)
    best_m = vals[best_i]
    best_t = alphas[best_i]
    
    # Refine with parabolic interpolation
    if 0 < best_i < n_grid - 1:
        # Parabolic interpolation
        y0, y1, y2 = vals[best_i-1], vals[best_i], vals[best_i+1]
        denom = (y0 - 2*y1 + y2)
        if abs(denom) > 1e-15:
            delta = 0.5 * (y0 - y2) / denom
            if -0.5 < delta < 0.5:
                t_par = alphas[best_i] + delta * (alphas[1] - alphas[0])
                if 0 <= t_par <= 1:
                    a_trial = (1 - t_par) * a + t_par * b_dir
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s > 1e-15:
                        a_trial = a_trial / s
                        m_par = fast_conv(a_trial).max()
                        if m_par < best_m:
                            best_m = m_par
                            best_t = t_par
    
    return best_t, best_m


def lp_fw_optimize(a0, time_limit, top_K=300, init_margin=2e-3, 
                   margin_decay=0.97, min_margin=1e-5, verbose=True):
    """LP Frank-Wolfe with line search."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    margin = init_margin
    stall = 0
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        ac = fast_conv(a)
        M = ac.max()
        
        # Select active constraints: all above threshold
        threshold = M * (1 - margin)
        active_mask = ac >= threshold
        if active_mask.sum() > top_K:
            # Take top K
            active_idx = np.argsort(ac)[::-1][:top_K]
        else:
            active_idx = np.where(active_mask)[0]
            if len(active_idx) < 5:
                active_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        
        # Solve LP
        b_dir = solve_lp(a, active_idx, M, n)
        
        # Line search
        t_best, m_best = line_search(a, b_dir, n)
        
        if t_best > 1e-8:
            a_new = (1 - t_best) * a + t_best * b_dir
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            else:
                a_new = a.copy()
        else:
            # Try tiny step
            a_new = a + 0.01 * (b_dir - a)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            else:
                a_new = a.copy()
        
        # Evaluate
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            a = a_new
            best_m = min(best_m, m_new)
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
            stall = 0
        else:
            # Try smaller step
            a_test = a + 0.1 * (b_dir - a)
            a_test = np.maximum(a_test, 0)
            s = a_test.sum()
            if s > 1e-15:
                a_test = a_test / s
                m_test = fast_conv(a_test).max()
                if m_test < best_m:
                    best_m = m_test
                    best_a = a_test.copy()
                    a = a_test
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        
        # Adaptive margin
        if stall == 0:
            margin = max(margin * margin_decay, min_margin)
        elif stall > 50:
            # Restart with perturbation
            rng = np.random.default_rng(iteration)
            pert = rng.normal(0, 0.01, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            margin = max(margin * 0.8, min_margin)
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW it={iteration} ({elapsed:.0f}s): J={J:.6f}, K={len(active_idx)}, margin={margin:.1e}")
    
    return best_a, best_m
