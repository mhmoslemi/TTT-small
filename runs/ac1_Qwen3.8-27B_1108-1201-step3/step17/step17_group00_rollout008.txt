def fast_autocorr(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def objective_simple(a):
    """Compute 2*n*sum(a^2)/(sum a)^2 - the simplified objective for non-neg a."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ss = np.sum(a * a)
    return float(2.0 * len(a) * ss / (s * s))


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def projection_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def projected_gradient_simplex(a0, time_limit, n_iter_max=500000, verbose=True):
    """
    Nesterov-accelerated projected gradient descent on 2*n*sum(a^2)/(sum a)^2.
    Since we normalize to sum=1, this becomes minimizing 2*n*sum(a^2),
    which is simply driving a toward uniformity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective_simple(a)
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, n_iter_max + 1):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Gradient of sum(a^2) w.r.t. a is 2*a
        # We want to minimize sum(a^2) subject to a in simplex
        # Gradient step: y - eta * 2*y, then project to simplex
        eta = 2.0 / (t + 2.0)
        step = y - eta * 2.0 * y
        x_new = projection_simplex(step)

        M_new = objective_simple(x_new)
        if M_new < best_val - 1e-15:
            best_val = M_new
            best_a = x_new.copy()

        x_prev = x
        x = x_new

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            # Check convergence: how close to uniform?
            uniformity = np.abs(x - 1.0 / n).max()
            print(f"    PG iter={t} ({elapsed:.0f}s): obj={best_val:.8f}, max_dev={uniformity:.2e}")

            if uniformity < 1e-10:
                print("    PG: converged to uniform!")
                break

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    PG done ({elapsed:.0f}s): obj={best_val:.8f}")
    return best_a, best_val


def flatten_heuristic(a0, time_limit, verbose=True):
    """
    Direct flattening: iteratively reduce the max element by spreading
    its excess to the minimum elements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective_simple(a)
    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        target = a.sum() / n  # = 1/n since sum=1
        max_idx = np.argmax(a)
        min_idx = np.argmin(a)

        if a[max_idx] - a[min_idx] < 1e-12:
            break

        # Move half the excess
        amt = (a[max_idx] - a[min_idx]) * 0.4
        a[max_idx] -= amt
        a[min_idx] += amt
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        val = objective_simple(a)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a.copy()

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Flatten iter={it} ({elapsed:.0f}s): obj={best_val:.8f}")

    return best_a, best_val


def lp_frank_wolfe_cutting(a0, time_limit, solver, initial_K=200, max_K=2000, add_per_iter=30, verbose=True):
    """LP-based Frank-Wolfe with cutting-plane constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective_simple(a)
    # Also track the true max of autocorrelation
    ac_best = fast_autocorr(a)
    best_m_true = ac_best.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    g_dir = a.copy()

    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()

    # Initialize constraint set with top K lags
    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    iter_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Update best
        val_y = 2.0 * n * M_curr
        if val_y < best_val - 1e-15:
            best_val = val_y
            best_a = y.copy()
            best_m_true = M_curr

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        val_x = 2.0 * n * M_x
        if val_x < best_val - 1e-15:
            best_val = val_x
            best_a = x.copy()
            best_m_true = M_x

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build LP constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        # Solve LP: min M s.t. sum(g)=1, g>=0, coeffs_matrix @ g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        conv_ys = fast_xcorr(y, g_dir)
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): obj={best_val:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): obj={best_val:.6f}, iters={iter_count}")
    return best_a, best_val


def soft_subgradient_optimize(a0, time_limit, top_k=50, beta=3.0, c0=3.0, verbose=True):
    """
    Mirror descent with soft subgradient for the max-autocorrelation objective.
    Uses the fact that for non-neg sequences, the max is at the center.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective_simple(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    restart_every = 3000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # For non-neg sequences, the main constraint is at the center (lag 0)
        # Gradient of sum(y^2) is 2*y
        # But we also need to worry about other lags potentially becoming max
        # Use soft subgradient: weight the top-k autocorrelation values
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        val_new = objective_simple(x)
        if val_new < best_val - 1e-15:
            best_val = val_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Restart with perturbation toward uniform
            uniform = np.ones(n) / n
            perturb = 0.5 * best_a + 0.5 * uniform
            x = perturb / perturb.sum()
            x_prev = x.copy()

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftSG iter={t} ({elapsed:.0f}s): obj={best_val:.8f}")

    return best_a, best_val


def parametric_search(n, time_limit, best_init, best_score):
    """Search over parametric families for a good initial sequence."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = best_init.copy()
    best_val = best_score
    i_arr = np.arange(n, dtype=np.float64)

    def try_update_local(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        val = objective_simple(a_n)
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
            if time.monotonic() - t0 < time_limit * 0.8:
                print(f"    Parametric [{label}]: obj={val:.6f}")

    # Powers
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        try_update_local(a, f"pow_{p}")

    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = (n - i_arr) ** p
        try_update_local(a, f"ipow_{p}")

    # Beta-like
    for alpha in [0.5, 1.0, 2.0]:
        for bbeta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** bbeta
            try_update_local(a, f"beta_{alpha}_{bbeta}")

    # Gaussians
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_update_local(a, f"gc{c:0.1f}s{sigma:0.1f}")

    # Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            a = 1.0 / (1.0 + np.exp(-s * (x - c)))
            try_update_local(a, f"sg{c:0.1f}_{s}")

    # Sinusoidal
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01)
            try_update_local(a, f"sn{freq}_{ph:0.2f}")

    # Exponential
    for rate in [1, 2, 5, 10]:
        a = np.exp(-rate * x)
        try_update_local(a, f"er{rate}")
        a = np.exp(rate * x)
        try_update_local(a, f"enr{rate}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.1f}s): best obj={best_val:.6f}")
    return best_a, best_val


def coordinate_polish(a0, time_limit, rng_seed, verbose=True):
    """
    Coordinate descent: move mass from peak-contributing positions to low-impact ones.
    Optimized for the simplified objective 2*n*sum(a^2)/(sum a)^2.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective_simple(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False

        # For the simplified objective, the best move is to reduce variance
        # Move from max to min
        max_i = np.argmax(a)
        min_i = np.argmin(a)

        if a[max_i] - a[min_i] > 1e-14:
            # Try several fractions
            for frac in [0.3, 0.5, 0.7, 0.9]:
                amt = a[max_i] * frac
                a_trial = a.copy()
                a_trial[max_i] -= amt
                a_trial[min_i] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                val = objective_simple(a_trial)
                if val < best_val - 1e-15:
                    a = a_trial
                    best_val = val
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try random pair moves
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                if a[i] < a[j]:
                    i, j = j, i
                if a[i] - a[j] < 1e-14:
                    continue
                amt = (a[i] - a[j]) * 0.5
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                val = objective_simple(a_trial)
                if val < best_val - 1e-15:
                    a = a_trial
                    best_val = val
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                val = objective_simple(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): obj={best_val:.8f}, stall={stall}")

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5200
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init (uniform): n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Update helper
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Try uniform sequence of various lengths ===
    print(f"\n=== Phase 0: Uniform sequences ===")
    for nn in [1000, 2000, 3000, 4000, 5000, 5200, 6000, 8000, 10000]:
        if time.monotonic() - t_global > time_budget * 0.05:
            break
        a_unif = np.ones(nn) / nn
        check_and_update(a_unif, f"uniform_n{nn}")

    # === Phase 1: Parametric search on best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Parametric search (n={n_best}) ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_param, _ = parametric_search(n_best, t1, best_seq, objective_simple(best_seq))
        check_and_update(a_param, "parametric")

    # === Phase 2: Projected gradient (fast, drives toward uniform) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Projected gradient (n={n_best}) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = projected_gradient_simplex(a_in, t2, verbose=False)
        check_and_update(a_out, f"projgrad_n{n_best}")

    # === Phase 3: LP-FW with cutting planes ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW cutting plane (n={n_best}) ===")
    t3 = min(remaining * 0.30, 450.0)
    if t3 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_cutting(a_in, t3, solver, initial_K=300, max_K=2500,
                                           add_per_iter=30, verbose=True)
        check_and_update(a_out, f"lp_main_n{n_best}")

    # === Phase 4: Soft subgradient optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Soft subgradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = soft_subgradient_optimize(a_in, t4, top_k=50, beta=4.0, c0=3.5, verbose=False)
        check_and_update(a_out, f"softsg_n{n_best}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate polish (n={n_best}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, rng_seed=77, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length exploration ===")

    candidate_lengths = [n_best // 2, n_best + 1000, n_best + 2000, n_best + 4000]
    candidate_lengths = [l for l in candidate_lengths if 500 <= l <= 12000 and l != n_best]

    for new_n in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  Exploring n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        check_and_update(a_long, f"interp_n{new_n}")

        # Quick projected gradient on new length
        t6 = min(remaining * 0.15, 60.0)
        if t6 > 10:
            a_out, _ = projected_gradient_simplex(a_long, t6, verbose=False)
            check_and_update(a_out, f"projgrad_n{new_n}")

        # LP-FW on new length if small enough
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5000:
            t6b = min(remaining * 0.25, 120.0)
            if t6b > 20:
                a_out, _ = lp_frank_wolfe_cutting(a_long, t6b, solver, initial_K=200,
                                                   max_K=1500, add_per_iter=20, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 5500:
        t7 = min(remaining * 0.45, 250.0)
        if t7 > 25:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_cutting(a_in, t7, solver, initial_K=400,
                                               max_K=3000, add_per_iter=40, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, rng_seed=314, verbose=True)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5200
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback (uniform): score={best_score:.6f}")

    return [float(x) for x in best_seq]
