        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j1 = k - idx
        j2 = idx - k
        contrib = np.zeros_like(idx, dtype=np.float64)
        m1 = (j1 >= 0) & (j1 < n)
        contrib[m1] += a[j1[m1]]
        m2 = (j2 >= 0) & (j2 < n)
        contrib[m2] += a[j2[m2]]
        grad[idx] += w * contrib

    return grad


def smoothed_subgrad_descent(a0, time_limit, beta_start=10.0, beta_end=500.0,
                             top_k=100, c0=3.0, restart_every=2000,
                             verbose=False, rng_seed=42):
    """
    Nesterov-accelerated subgradient descent on the log-sum-exp smoothed objective.
    Beta anneals from beta_start to beta_end for increasing exactness.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        frac = min(1.0, t / (time_limit * 30))
        beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        g = lse_subgradient(y, ac, beta=beta, top_k=top_k)

        # Mirror descent step (multiplicative)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > restart_every:
            stall = 0
            strength = min(0.15, 0.03 + 0.02 * (restart_every // 500))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            gamma *= 1.1
            top_k = min(top_k + 5, 200)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SGD restart t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_m


def coordinate_refine(a0, time_limit, n_moves=50, rng_seed=42, verbose=False):
    """
    Local refinement via coordinate descent.
    At each step, find the lag with maximum autocorrelation,
    then move mass from the contributing pair to reduce it.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))

        # Find contributing pairs for k_star
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        j1 = k_star - idx
        j2 = idx - k_star

        # Contribution of each pair
        contrib = np.zeros(len(idx))
        m1 = (j1 >= 0) & (j1 < n)
        contrib[m1] += a[idx[m1]] * a[j1[m1]]
        m2 = (j2 >= 0) & (j2 < n) & (j2 != j1)
        contrib[m2] += a[idx[m2]] * a[j2[m2]]

        # Find the largest contributing pair
        if len(contrib) == 0 or contrib.max() < 1e-20:
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            continue

        # Try moving mass from the worst contributor
        worst_local = np.argmax(contrib)
        src_i = int(idx[worst_local])
        src_j1 = int(j1[worst_local]) if m1[worst_local] else -1
        src_j2 = int(j2[worst_local]) if m2[worst_local] else -1

        # Determine source for mass removal
        if src_j1 >= 0 and src_j1 != src_j2:
            # Move from src_j1 or src_i
            sources = [src_i]
            if src_j1 != src_i:
                sources.append(src_j1)
        elif src_j2 >= 0 and src_j2 != src_j1:
            sources = [src_i]
            if src_j2 != src_i:
                sources.append(src_j2)
        else:
            sources = [src_i]

        improved = False
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            # Find best destination: position with low autocorrelation impact
            # Heuristic: positions where the sequence is already low
            cand = np.argsort(a)[:max(5, n // 10)]
            for dst in cand:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
            stall += 1

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def spectral_shape_optimize(a0, time_limit, n_iters=3000, verbose=False):
    """
    Direct spectral shaping: iteratively modify the power spectrum
    to reduce the time-domain peak while maintaining non-negativity.
    Uses a soft constraint approach.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    N = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(77)

    # Current spectrum
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    P_target = P.copy()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute time domain from current spectrum
        A_cur = np.sqrt(P_target) * np.exp(1j * np.angle(A))
        a_time = np.fft.irfft(A_cur, N)[:n]

        # Soft non-negativity: use rectified linear + exponential
        a_time = np.maximum(a_time, 0.0)
        s = a_time.sum()
        if s < 1e-15:
            # Reset
            a_time = a.copy()
        else:
            a_time = a_time / s

        # Update target spectrum: blend current with flat
        A_new = np.fft.rfft(a_time, N)
        P_new = np.abs(A_new) ** 2

        # Blend towards flat spectrum
        P_mean = P_new.mean()
        alpha = 0.3 + 0.3 * (it / max(1, n_iters))
        P_target = np.maximum(P_target ** (1 - alpha) * (P_mean) ** alpha, 1e-30)

        # Add small random perturbation to avoid cycles
        P_target *= (1.0 + 0.01 * rng.standard_normal(len(P_target)))
        P_target = np.maximum(P_target, 1e-30)

        A = np.sqrt(P_target) * np.exp(1j * np.angle(A_new))

        # Check improvement
        M_new = fast_autocorr(a_time).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_time.copy()
            a = a_time

    return best_a, best_m


def make_init_power_law(n, power=1.0, center=True):
    """Generate a power-law initialization."""
    x = np.arange(1, n + 1, dtype=np.float64) / n
    if center:
        a = np.abs(x - 0.5) ** power
    else:
        a = x ** power
    a = np.maximum(a, 0.001)
    return normalize(a)


def make_init_cosine(n, mode=1):
    """Generate a cosine initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.001)
    return normalize(a)


def make_init_triangular(n, asym=0.3):
    """Generate an asymmetric triangular initialization."""
    x = np.linspace(0, 1, n)
    peak = 0.5 + asym
    a = 1.0 - np.abs(x - peak) / max(peak, 1.0 - peak)
    a = np.maximum(a, 0.001)
    return normalize(a)


def make_init_bimodal(n, sep=0.3):
    """Generate a bimodal initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5 + sep) ** 2) / 0.02) + np.exp(-((x - 0.5 - sep) ** 2) / 0.02)
    a = np.maximum(a, 0.001)
    return normalize(a)


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == new_n:
        return a.copy()
    a = normalize(a)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(x_new, x_old, a)
    return normalize(a_interp)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining spectral methods, smoothed subgradient
    descent, and coordinate refinement across multiple scales.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Spectral initialization via GS ===
    print(f"\n=== Phase 1: Spectral GS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p1 = min(remaining * 0.15, 250.0)
    if t_p1 > 30:
        # Try from parent
        a_gs, _ = gs_spectral_flat(best_seq, t_p1 * 0.5, n_iter_max=30000,
                                    eta=1.0, heat=0.05, verbose=False)
        try_update(a_gs, "gs_parent")

        # Try from various inits
        remaining = time_budget - (time.monotonic() - t_global)
        t_p1b = min(remaining * 0.10, 150.0)
        if t_p1b > 20:
            inits = [
                make_init_triangular(n0, asym=0.2),
                make_init_bimodal(n0, sep=0.2),
                make_init_power_law(n0, power=1.5, center=True),
            ]
            t_per = t_p1b / len(inits)
            for i, init in enumerate(inits):
                if time.monotonic() - t_global > time_budget - 60:
                    break
                a_gs, _ = gs_spectral_flat(init, t_per, n_iter_max=20000,
                                            eta=1.0, heat=0.1, verbose=False)
                try_update(a_gs, f"gs_init{i}")

    # === Phase 2: Smoothed subgradient descent ===
    print(f"\n=== Phase 2: Smoothed SGD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t_p2 = min(remaining * 0.30, 500.0)
    if t_p2 > 30:
        # Multiple runs with different parameters
        configs = [
            {'beta_start': 5.0, 'beta_end': 200.0, 'top_k': 80, 'c0': 3.0, 'restart_every': 1500},
            {'beta_start': 20.0, 'beta_end': 1000.0, 'top_k': 150, 'c0': 5.0, 'restart_every': 3000},
            {'beta_start': 10.0, 'beta_end': 500.0, 'top_k': 120, 'c0': 4.0, 'restart_every': 2000},
        ]
        t_per = t_p2 / len(configs)
        for i, cfg in enumerate(configs):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  SGD config {i} (t={t_avail:.0f}s)...")
            a_out, _ = smoothed_subgrad_descent(best_seq, t_avail,
                                                 beta_start=cfg['beta_start'],
                                                 beta_end=cfg['beta_end'],
                                                 top_k=cfg['top_k'],
                                                 c0=cfg['c0'],
                                                 restart_every=cfg['restart_every'],
                                                 verbose=False,
                                                 rng_seed=42 + i)
            try_update(a_out, f"sgd_{i}")

    # === Phase 3: Spectral shape optimization ===
    print(f"\n=== Phase 3: Spectral shape (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p3 = min(remaining * 0.10, 150.0)
    if t_p3 > 20:
        a_out, _ = spectral_shape_optimize(best_seq, t_p3, n_iters=2000, verbose=False)
        try_update(a_out, "spectral_shape")
        # Then SGD polish
        remaining = time_budget - (time.monotonic() - t_global)
        t_p3b = min(remaining * 0.08, 100.0)
        if t_p3b > 15:
            a_out, _ = smoothed_subgrad_descent(best_seq, t_p3b,
                                                 beta_start=50.0, beta_end=500.0,
                                                 top_k=150, c0=4.0,
                                                 restart_every=3000, verbose=False)
            try_update(a_out, "sgd_after_spectral")

    # === Phase 4: Coordinate refinement ===
    print(f"\n=== Phase 4: Coordinate refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p4 = min(remaining * 0.10, 120.0)
    if t_p4 > 15:
        a_out, _ = coordinate_refine(best_seq, t_p4, rng_seed=42, verbose=False)
        try_update(a_out, "coord_n0")
        a_out, _ = coordinate_refine(best_seq, t_p4, rng_seed=999, verbose=False)
        try_update(a_out, "coord_n0b")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)

    # Try slightly shorter
    shorter_n = max(1000, n_cur - 800)
    if remaining > 60 and shorter_n != n_cur:
        print(f"  Trying n={shorter_n}...")
        a_short = interpolate_seq(best_seq, shorter_n)
        try_update(a_short, f"interp_n{shorter_n}")
        t_s = min(remaining * 0.15, 120.0)
        if t_s > 15:
            a_out, _ = smoothed_subgrad_descent(a_short, t_s,
                                                 beta_start=20.0, beta_end=500.0,
                                                 top_k=100, c0=4.0,
                                                 restart_every=2000, verbose=False)
            try_update(a_out, f"sgd_n{shorter_n}")

    # Try slightly longer
    remaining = time_budget - (time.monotonic() - t_global)
    longer_n = min(6000, n_cur + 1000)
    if remaining > 60 and longer_n != n_cur:
        print(f"  Trying n={longer_n}...")
        a_long = interpolate_seq(best_seq, longer_n)
        try_update(a_long, f"interp_n{longer_n}")
        t_l = min(remaining * 0.15, 120.0)
        if t_l > 15:
            a_out, _ = smoothed_subgrad_descent(a_long, t_l,
                                                 beta_start=20.0, beta_end=500.0,
                                                 top_k=100, c0=4.0,
                                                 restart_every=2000, verbose=False)
            try_update(a_out, f"sgd_n{longer_n}")

    # === Phase 6: Final intensive optimization on best ===
    print(f"\n=== Phase 6: Final optimization (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)

    if remaining > 60:
        # Intensive SGD with aggressive parameters
        t_f1 = min(remaining * 0.30, 200.0)
        if t_f1 > 20:
            a_out, _ = smoothed_subgrad_descent(best_seq, t_f1,
                                                 beta_start=100.0, beta_end=2000.0,
                                                 top_k=200, c0=5.0,
                                                 restart_every=5000, verbose=True,
                                                 rng_seed=777)
            try_update(a_out, "sgd_final_agg")

        # GS refinement
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.20, 100.0)
        if t_f2 > 15:
            a_out, _ = gs_spectral_flat(best_seq, t_f2, n_iter_max=20000,
                                         eta=1.0, heat=0.02, verbose=False)
            try_update(a_out, "gs_final")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.25, 100.0)
        if t_f3 > 10:
            a_out, _ = coordinate_refine(best_seq, t_f3, rng_seed=12345, verbose=True)
            try_update(a_out, "coord_final")

        # One more SGD
        remaining = time_budget - (time.monotonic() - t_global)
        t_f4 = min(remaining * 0.30, 80.0)
        if t_f4 > 10:
            a_out, _ = smoothed_subgrad_descent(best_seq, t_f4,
                                                 beta_start=200.0, beta_end=3000.0,
                                                 top_k=250, c0=6.0,
                                                 restart_every=8000, verbose=False,
                                                 rng_seed=55555)
            try_update(a_out, "sgd_final2")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = normalize(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
