"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with EXACT subgradients (O(n) per step)
2. Multi-lag subgradient combination for stability
3. LP-based Frank-Wolfe refinement (periodic)
4. Adaptive step size with backtracking
5. Length sweeping and symmetric/non-symmetric variants
Key improvements over previous: exact subgradient (vs soft-max), 
faster iterations, better step size adaptation, multi-lag weighting.

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the objective using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def exact_subgradient(a, top_k=20, beta=5.0):
    """
    Compute weighted combination of exact subgradients for top-k peaking lags.
    Returns (gradient, max_value, top_lags).
    
    For lag k, the subgradient of C(k) w.r.t. a[j] is 2*a[k-j] (if valid).
    We combine top-k lags with exponential weights.
    """
    n = len(a)
    C = fast_autocorr(a)
    M = C.max()
    
    # Find top-k lags
    k = min(top_k, len(C))
    # Use soft-max weights for stability
    shifted = C - M
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # All equal, just use the argmax
        top_idx = np.argmax(C)
        g = np.zeros(n)
        for j in range(n):
            diff = top_idx - j
            if 0 <= diff < n:
                g[j] = 2.0 * a[diff]
        return g, M, [top_idx]
    
    w = w / w_sum
    # Only use lags with significant weight
    threshold = 1e-8
    active = w > threshold
    if not active.any():
        top_idx = np.argmax(C)
        g = np.zeros(n)
        for j in range(n):
            diff = top_idx - j
            if 0 <= diff < n:
                g[j] = 2.0 * a[diff]
        return g, M, [top_idx]
    
    active_idx = np.where(active)[0]
    
    # Compute gradient: g[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is: g = 2 * convolve(a_reversed, w_active, mode=...)
    # More precisely: g[j] = 2 * sum_{k in active} w[k] * a[k-j]
    # Let's use FFT
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    g = 2.0 * conv[n-1:2*n-1]
    
    return g, M, active_idx[:top_k]

def nesterov_mirror_descent(a0, time_limit, verbose=True, top_k=30, beta=5.0,
                            lr=0.05, momentum_init=0.9, seed=42):
    """
    Nesterov-accelerated mirror descent with exact subgradients.
    Works on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    C = fast_autocorr(a)
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    momentum = momentum_init
    iterations = 0
    stall = 0
    max_stall = 5000
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        # Nesterov extrapolation
        if iterations == 1:
            y = x.copy()
        else:
            t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * iterations * iterations))
            y = x + (1.0 - t_k) / t_k * (x - x_prev)
            # Clip to non-negative
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute exact subgradient
        g, M_curr, _ = exact_subgradient(y, top_k=top_k, beta=beta)
        
        # Mirror descent step (in log-space)
        # Adaptive learning rate
        eta = lr / (1.0 + 0.01 * iterations)
        
        # Update: x_new = y * exp(-eta * g) / sum(y * exp(-eta * g))
        log_update = np.log(y) - eta * g
        log_update -= log_update.max()  # numerical stability
        x_new = np.exp(log_update)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        x_prev = x.copy()
        x = x_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Perturb around best
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Random restart with some probability
            if rng.random() < 0.2:
                x = rng.dirichlet(np.ones(n) * 0.3)
                x_prev = x.copy()
        
        if verbose and iterations % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  NMD iter={iterations} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def lp_direction(a, M, top_K=200, solver=None):
    """
    Solve LP to find a feasible direction b that satisfies conv(a, b)[k] <= M.
    Returns the LP solution (normalized to sum 1) or None if infeasible.
    """
    n = len(a)
    if solver is None:
        solver = get_solver()
    
    # Build constraint matrix: T[m, j] = a[m-j] if 0 <= m-j < n
    # conv(a, b)[m] = sum_j a[m-j] * b[j] <= M
    K_sel = min(top_K, 2 * n - 1)
    
    # Select most binding constraints (where conv(a,a) is largest)
    C = fast_autocorr(a)
    top_idx = np.argsort(C)[::-1][:K_sel]
    
    T = np.zeros((K_sel, n))
    for col_idx, m in enumerate(top_idx):
        m_int = int(m)
        j_start = max(0, m_int - n + 1)
        j_end = min(n - 1, m_int)
        js = np.arange(j_start, j_end + 1)
        T[col_idx, js] = a[m_int - js]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(K_sel),
                   cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def fw_refine(a0, time_limit, solver, verbose=True, top_K=300):
    """
    Frank-Wolfe refinement using LP directions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        C = fast_autocorr(y)
        M_curr = C.max()
        
        # Get LP direction
        b_dir = lp_direction(y, M_curr, top_K=top_K, solver=solver)
        if b_dir is None:
            # Try with slightly relaxed M
            b_dir = lp_direction(y, M_curr * 1.01, top_K=top_K, solver=solver)
            if b_dir is None:
                break
        
        # FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def get_solver():
    """Find the best available LP solver."""
    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: ECOS (fallback)")
    return cp.ECOS

def make_gaussian_init(n, center=0.5, sigma=0.2, seed=0):

def coordinate_refine(a0, time_limit, verbose=True, seed=42):
    """
    Fast coordinate descent: move mass from high-contribution positions to low ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_autocorr(a)
        M = C.max()
        
        # Find the peak lag
        k_peak = int(np.argmax(C))
        
        # Find positions with highest contribution to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        
        if contrib.max() < 1e-20:
            break
        
        # Source: position with highest contribution
        src_pos = idx[np.argmax(contrib)]
        
        # Destination: position with lowest "impact" on all top lags
        top_lags = np.argsort(C)[::-1][:min(10, 2*n-1)]
        impact = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            for j in range(s, e + 1):
                impact[j] += a[ki - j]
        
        # Choose destination far from source with low impact
        dist = np.abs(np.arange(n) - src_pos).astype(float)
        scores = impact + 0.1 * (1.0 - dist / n)  # prefer far, low impact
        
        best_dst = int(np.argmin(scores))
        if best_dst == src_pos:
            # Pick the next best
            scores[src_pos] = np.inf
            best_dst = int(np.argmin(scores))
        
        # Transfer mass
        for frac in [0.01, 0.005, 0.02, 0.001, 0.05, 0.002]:
            amt = a[src_pos] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src_pos] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                break
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))

        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def exact_subgradient(a, top_k=15, beta=10.0):
    """
    Compute weighted exact subgradient for top-k peaking lags. O(K*n).
    For lag k: d C(k)/d a[j] = 2*a[k-j] (if 0 <= k-j < n).
    """
    n = len(a)
    C = fast_autocorr(a)
    M = C.max()
    
    k = min(top_k, len(C))
    sorted_idx = np.argsort(C)[::-1][:k]
    sorted_vals = C[sorted_idx]
    
    weights = np.exp(beta * (sorted_vals - M))
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(k)
        weights[0] = 1.0
    else:
        weights = weights / w_sum
    
    g = np.zeros(n)
    for idx_i in range(k):
        w_i = weights[idx_i]
        if w_i < 1e-12:
            continue
        ki = int(sorted_idx[idx_i])
        j_min = max(0, ki - n + 1)
        j_max = min(n - 1, ki)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            g[js] += w_i * 2.0 * a[ki - js]
    
    return g, M

            t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * iterations * iterations))
            momentum = (1.0 - t_k) / t_k
            y = x + momentum * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
            if symmetric:
                y = 0.5 * (y + y[::-1])
                y = y / y.sum()
        
        # Compute exact subgradient
        if symmetric:
            g, M_curr = exact_subgradient_symmetric(y, top_k=top_k, beta=beta)
        else:
            g, M_curr = exact_subgradient(y, top_k=top_k, beta=beta)
        
        # Adaptive learning rate
        if adapt_lr:
            current_lr = lr / (1.0 + 0.005 * iterations)
        eta = current_lr
        
        # Mirror descent step (exponentiated gradient on simplex)
        log_update = np.log(y + 1e-30) - eta * g
        log_update -= log_update.max()
        x_new = np.exp(log_update)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        if symmetric:
            x_new = 0.5 * (x_new + x_new[::-1])
            x_new = x_new / x_new.sum()
        
        x_prev = x.copy()
        x = x_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            r = rng.random()
            if r < 0.5:
                # Perturb around best
                scale = rng.uniform(0.005, 0.03)
                pert = rng.normal(0, scale, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                if symmetric:
                    x_r = 0.5 * (x_r + x_r[::-1])
                    x_r = x_r / x_r.sum()
                x = x_r
            elif r < 0.8:
                # Dirichlet restart with concentration
                conc = rng.uniform(0.2, 1.0)
                x = rng.dirichlet(np.ones(n) * conc)
                if symmetric:
                    x = 0.5 * (x + x[::-1])
                    x = x / x.sum()
            else:
                # Blend with smooth
                smooth = np.ones(n) / n
                blend = 0.5 * best_a + 0.5 * smooth
                x = blend / blend.sum()
                if symmetric:
                    x = 0.5 * (x + x[::-1])
                    x = x / x.sum()
            x_prev = x.copy()
        
        if verbose and iterations % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD{'_SYM' if symmetric else ''} iter={iterations} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def get_solver():
    """Find the best available LP solver."""
    solvers = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for s in solvers:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: ECOS (fallback)")
    return cp.ECOS


def fw_refine(a0, time_limit, solver, verbose=True, top_K=200, max_iters=5000):
    """
    Frank-Wolfe refinement using LP directions with constraint selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        C = fast_autocorr(y)
        M_curr = C.max()
        
        # Build constraint matrix for top-K most binding lags
        K_sel = min(top_K, 2 * n - 1)
        top_idx = np.argsort(C)[::-1][:K_sel]
        
        T = np.zeros((K_sel, n))
        for col_idx in range(K_sel):
            m = int(top_idx[col_idx])
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            T[col_idx, js] = y[m - js]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try relaxed constraint
            try:
                constraints = [T @ b_var <= M_curr * 1.05 * np.ones(K_sel),
                               cp.sum(b_var) == 1]
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        prev_y = y.copy()
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
