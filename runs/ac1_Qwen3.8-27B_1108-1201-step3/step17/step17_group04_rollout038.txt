with backtracking line search, multi-scale optimization, and LP Frank-Wolfe.

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS


def get_subgradient_topk(a, ac, top_k, beta):
    """Compute weighted subgradient using top-K peaks with temperature."""
    n = len(a)
    grad = np.zeros(n)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def get_subgradient_exact(a, ac):
    """Compute exact subgradient at the argmax peak."""
    n = len(a)
    k = int(np.argmax(ac))
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad = np.zeros(n)
    grad[idx] = 2.0 * a[j_idx]
    return grad

    Uses mirror descent (exponential) steps on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k_init
    cur_beta = beta_init
    cur_c = c_init
    cur_exact_frac = exact_frac
    max_its = 5000000

    for t in range(1, max_its):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()

        # Adaptive subgradient selection
        if rng.random() < cur_exact_frac:
            g = get_subgradient_exact(y, ac)
        else:
            g = get_subgradient_topk(y, ac, cur_top_k, cur_beta)

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break

        # Backtracking line search
        eta = cur_c * (M - 1.0 / n) / max(g_norm_sq, 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        best_eta = 0.0
        best_a_candidate = None
        for _bt in range(12):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                eta *= 0.5
                continue
            a_new = a_new / s_new

            J_new = compute_J(a_new)
            # Armijo-like condition: J should decrease
            if J_new < M * 2.0 * n - 1e-12:
                best_eta = eta
                best_a_candidate = a_new
                break
            eta *= 0.5
            if eta < 1e-12:
                break

        if best_a_candidate is not None:
            x_prev = x
            x = best_a_candidate
            J_new = compute_J(x)
            if J_new < best_J - 1e-14:
                best_J = J_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
        else:
            # No improvement from subgradient; try a small random step
            x_prev = x
            perturb = rng.normal(0, 0.005, n) * x
            x = x + perturb
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            stall += 1

        # Adaptive restart
        if stall > 3000:
            stall = 0
            sigma = 0.01 + 0.005 * min(stall, 20)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_c = min(cur_c + 0.3, 10.0)
            cur_exact_frac = min(cur_exact_frac + 0.03, 0.7)

        # Gradually increase top_k and decrease beta (flatter targeting over time)
        frac = min(1.0, t / 100000.0)
        cur_top_k = max(cur_top_k, int(top_k_init + frac * 30))
        cur_beta = min(cur_beta, beta_init - frac * 1.0)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD-BD t={t} ({elapsed:.0f}s): J={best_J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")

    return best_a, best_J


def coordinate_refinement(a0, time_limit, seed=99):
    """
    Coordinate refinement: for each position, try moving mass to other positions.
    Uses the sensitivity of the max peak to guide which moves to try.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))

        # Compute sensitivity: which positions contribute most to the peak?
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n, peak_lag + 1)
        idx = np.arange(i_start, i_end)
        j_idx = peak_lag - idx
        contribution = a[idx] * a[j_idx]

        # Sort by contribution (descending)
        order = np.argsort(contribution)[::-1]
        top_positions = idx[order[:min(50, len(idx))]]

        # Find "safe" positions (low contribution to peak)
        all_contrib = np.zeros(n)
        all_contrib[idx] = contribution
        safe_positions = np.where(all_contrib < 0.1 * max(all_contrib.max(), 1e-30))[0]

        if len(safe_positions) < 5:
            safe_positions = np.where(a < np.percentile(a, 20))[0]

        improved = False
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            src = top_positions[rng.integers(len(top_positions))]
            dst = safe_positions[rng.integers(len(safe_positions))]
            if src == dst or a[src] < 1e-10:
                continue

            # Binary search on transfer amount
            lo, hi = 0.0, a[src] * 0.5
            for _bs in range(8):
                mid = (lo + hi) / 2.0
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    lo = mid
                    continue
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    lo = mid
                else:
                    hi = mid

            if lo > 1e-12:
                a[src] -= lo
                a[dst] += lo
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J - 1e-14:
                    best_J = J_new
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Random perturbation to escape
            n_pert = max(3, n // 200)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.001, 0.02)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, top_K=300, verbose=False):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = T_full[binding_idx, :]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_cosine_init(n):
    """Cosine bell initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.7):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.12):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def spectral_refinement(a0, time_limit, seed=555):
    """
    Spectral domain refinement: adjust the log-periodogram to flatten autocorrelation.
    Uses gradient of max(IFFT(|A|^2)) with respect to log|A|.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Work with real, even spectrum for real symmetric a
    # Use full FFT
    N = 1 << (2 * n - 1).bit_length()

    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, N)
        log_mag = np.log(np.abs(A) + 1e-30)
        phase = np.angle(A)

        ac = fast_autocorr(a)
        M = ac.max()
        peak_lag = int(np.argmax(ac))

        # Gradient of R(peak_lag) with respect to log|A[k]|
        # dR(m)/d log|A[k]| = 2|A[k]| * Re(A[k] * conj(A[k]) * exp(2*pi*i*k*m/N)) / |A[k]|
        # = 2|A[k]| * cos(2*pi*k*peak_lag/N) ... simplified
        # Actually: R(m) = (1/N) sum_k |A[k]|^2 exp(2*pi*i*k*m/N)
        # dR(m)/d log|A[k]| = (1/N) * 2|A[k]|^2 * cos(2*pi*k*m/N) (for real case)
        # But we need the real sequence, so let's use a simpler approach:
        # Perturb log_mag and check if it helps

        # Use top-K peaks for a softer objective
        top_k = 5
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]

        # Compute gradient as sum over top peaks
        k_freq = np.arange(len(A))
        grad_log = np.zeros(len(A))
        for kk in top_idx:
            kk_int = int(kk)
            cos_val = np.cos(2.0 * np.pi * k_freq * kk_int / N)
            grad_log += 2.0 * np.abs(A) ** 2 * cos_val

        # Step in log_mag direction (negative gradient to reduce peaks)
        lr = 0.1 / (it + 1.0)
        lr = max(lr, 0.001)
        log_mag -= lr * grad_log / (np.abs(grad_log).max() + 1e-30)

        # Reconstruct
        A_new = np.exp(log_mag) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Occasional random re-initialization of spectrum
            if it % 500 == 499:
                log_mag = log_mag + rng.normal(0, 0.02, len(log_mag))

    return best_a, best_J


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with adaptive multi-method optimization.
    Uses subgradient descent with backtracking, coordinate refinement,
    LP Frank-Wolfe, and spectral refinement across multiple sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_cosine_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Subgradient from parent with backtracking ===
    print(f"\n=== Phase 1: SGD-BD on parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_out, _ = subgradient_bd(norm_sum1(best_seq), t1,
                                  top_k_init=10, beta_init=3.0, c_init=3.0,
                                  exact_frac=0.3, seed=42, verbose=True)
        _try_update(a_out, "sgd_bd_parent")

        # Second run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.10, 150.0)
        if t1b > 20:
            a_out2, _ = subgradient_bd(norm_sum1(best_seq), t1b,
                                       top_k_init=20, beta_init=2.0, c_init=5.0,
                                       exact_frac=0.5, seed=123, verbose=False)
            _try_update(a_out2, "sgd_bd_parent_v2")

    # === Phase 2: Try different lengths ===
    print(f"\n=== Phase 2: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    # Try slightly larger
    for delta in [300, 600]:
        new_n = n_best + delta
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80 or new_n > 5500:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t2a = min(remaining * 0.12, 180.0)
        if t2a > 20:
            a_out, _ = subgradient_bd(a_long, t2a,
                                      top_k_init=10, beta_init=3.0, c_init=3.0,
                                      exact_frac=0.3, seed=555 + delta, verbose=False)
            _try_update(a_out, f"sgd_n{new_n}")

    # Try slightly smaller
    for delta in [300, 600]:
        new_n = n_best - delta
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60 or new_n < 2000:
            break
        print(f"  Trying shorter n={new_n}")
        a_short = interpolate_sequence(best_seq, new_n)
        t2b = min(remaining * 0.08, 120.0)
        if t2b > 15:
            a_out, _ = subgradient_bd(a_short, t2b,
                                      top_k_init=10, beta_init=3.0, c_init=3.0,
                                      exact_frac=0.3, seed=777 + delta, verbose=False)
            _try_update(a_out, f"sgd_n{new_n}")

    # === Phase 3: Intensive optimization on current best ===
    print(f"\n=== Phase 3: Intensive SGD on best (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        # Run with progressively more aggressive settings
        a_out, _ = subgradient_bd(a_in, t3,
                                  top_k_init=15, beta_init=2.5, c_init=4.0,
                                  exact_frac=0.4, seed=999, verbose=True)
        _try_update(a_out, "sgd_intensive")

        # Another variant
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.10, 150.0)
        if t3b > 20:
            a_in2 = norm_sum1(best_seq)
            a_out2, _ = subgradient_bd(a_in2, t3b,
                                       top_k_init=30, beta_init=1.5, c_init=6.0,
                                       exact_frac=0.2, seed=2024, verbose=False)
            _try_update(a_out2, "sgd_intensive_flat")

    # === Phase 4: Coordinate refinement ===
    print(f"\n=== Phase 4: Coordinate refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_refinement(a_in, t4, seed=314)
        _try_update(a_out, "coord_refine")

    # === Phase 5: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP Frank-Wolfe (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30 and n_best <= 4500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, top_K=300, verbose=True)
        _try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 6: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Spectral refinement (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_refinement(a_in, t6, seed=555)
        _try_update(a_out, "spectral")

    # === Phase 7: Final combined pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final optimization (n={n_best}) ===")
    if remaining > 40:
        # Quick SGD + coord refinement
        t7a = min(remaining * 0.4, 120.0)
        t7b = min(remaining * 0.3, 80.0)
        if t7a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_bd(a_in, t7a,
                                      top_k_init=25, beta_init=2.0, c_init=5.0,
                                      exact_frac=0.4, seed=4321, verbose=False)
            _try_update(a_out, "final_sgd")
        remaining = time_budget - (time.monotonic() - t_global)
        if t7b > 15 and remaining > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_refinement(a_in, min(t7b, remaining - 10), seed=777)
            _try_update(a_out, "final_coord")

    # === Phase 8: Try one more approach - blend with structured init ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Blend + optimize (n={n_best}) ===")
        a_p = norm_sum1(best_seq)
        a_struct = make_cosine_init(n_best)
        for blend in [0.3, 0.5, 0.7]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            a_blend = (1 - blend) * a_p + blend * a_struct
            a_blend = norm_sum1(a_blend)
            t8 = min(remaining * 0.25, 100.0)
            if t8 > 15:
                a_out, _ = subgradient_bd(a_blend, t8,
                                          top_k_init=15, beta_init=2.5, c_init=4.0,
                                          exact_frac=0.4, seed=8888, verbose=False)
                _try_update(a_out, f"blend_{blend}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_cosine_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
