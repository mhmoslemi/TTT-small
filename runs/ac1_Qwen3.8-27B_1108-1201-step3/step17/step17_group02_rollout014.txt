def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_solve_frank_wolfe(a, M, solver, top_K=None):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    
    Returns the normalized solution b/sum(b), or None if infeasible.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    if top_K is not None and top_K < num_constraints:
        # Use only the top_K tightest constraints
        ac = fast_autocorr(a)
        margins = M - ac
        binding_idx = np.argsort(margins)[:top_K]
    else:
        binding_idx = np.arange(num_constraints)
    
    n_bc = len(binding_idx)
    
    # Build constraint matrix: T[k,j] = a[k-j] if 0 <= k-j < n
    T = np.zeros((n_bc, n))
    for ii in range(n_bc):
        k = int(binding_idx[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            T[ii, j_start:j_end] = a[k - j_end + 1 : k - j_start + 1][::-1]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum


def fw_optimize(a0, time_limit, solver, top_K=None, verbose=True, rng_seed=42):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    y = a.copy()  # Main iterate
    x = a.copy()  # Nesterov momentum
    iter_count = 0
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y_new = x + beta_t * (x - y)
        y_new = np.maximum(y_new, 1e-15)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = x.copy()
        else:
            y_new = y_new / sy
        
        # Solve LP for Frank-Wolfe direction
        M_curr = fast_autocorr(y_new).max()
        g = lp_solve_frank_wolfe(y_new, M_curr, solver, top_K=top_K)
        
        if g is None:
            # LP failed, try with slightly relaxed M
            g = lp_solve_frank_wolfe(y_new, M_curr * 1.01, solver, top_K=top_K)
            if g is None:
                y = y_new
                x = y_new
                stall += 1
                continue
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Update
        y = (1.0 - alpha) * y_new + alpha * g
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_new.copy()
        else:
            y = y / sy
        
        # Update Nesterov
        x = y.copy()
        iter_count += 1
        
        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m - 1e-16:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 500:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            y_r = best_a + pert
            y_r = np.maximum(y_r, 1e-10)
            y_r = y_r / y_r.sum()
            y = y_r
            x = y_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FW restart iter={iter_count} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}, stall={stall}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FW final: J={J:.6f}, iters={iter_count}, time={elapsed:.0f}s")
    return best_a, best_m
