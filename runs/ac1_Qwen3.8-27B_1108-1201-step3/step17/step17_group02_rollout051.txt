    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size, fft_size)[:size]

def autocorr_gradient(a, p):
    """
    Compute g[j] = 2 * sum_{k=j}^{min(j+n-1, 2n-2)} p[k] * a[k-j]
    using FFT-based convolution.
    """
    n = len(a)
    # g[j]/2 = sum_m p[j+m] * a[m] for m = 0..min(n-1, 2n-2-j)
    # = conv(p, a_reversed)[j + n - 1]
    a_rev = a[::-1].copy()
    conv_len = (2*n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(P * AR, fft_size, fft_size)[:conv_len]
    return 2.0 * conv[n-1 : n-1+n]

def logsumexp_value(ac, beta):
    """Compute smooth max approximation: (1/beta)*log(sum(exp(beta*ac)))"""
    m = ac.max()
    if m < 1e-30:
        return 0.0
    shifted = beta * (ac - m)
    s = np.sum(np.exp(shifted))
    return m + np.log(s) / beta

def logsumexp_weights(ac, beta):
    """Compute the weights p[k] = exp(beta*ac[k]) / sum exp(beta*ac[k])"""
    m = ac.max()
    shifted = beta * (ac - m)
    e = np.exp(shifted)
    return e / e.sum()

def objective_softmax(z, beta):
    """Objective and gradient in softmax parameterization."""
    n = len(z)
    # Compute a = softmax(z)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    
    # Compute autocorrelation
    ac = fast_autocorr(a)
    
    # Compute weights and value
    m_ac = ac.max()
    shifted = beta * (ac - m_ac)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    L = m_ac + np.log(exp_shifted.sum()) / beta
    
    # Compute gradient w.r.t. a
    g_a = autocorr_gradient(a, p)
    
    # Chain rule for softmax
    S = np.sum(g_a * a)
    g_z = a * (g_a - S)
    
    obj = 2.0 * n * L
    grad = 2.0 * n * g_z
    
    return obj, grad

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

def lbfgs_run(a0, time_limit, beta_init=10.0, beta_max=2000.0, n_betas=5, verbose=True):
    """Run L-BFGS with increasing beta (temperature annealing)."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0 / a0.sum()
    
    # Initialize z from a (inverse softmax)
    z = np.log(a + 1e-30)
    z = z - z.mean()  # center for numerical stability
    
    best_a = a.copy()
    best_J = 2.0 * n * fast_autocorr(a).max()
    
    betas = np.geomspace(beta_init, beta_max, n_betas)
    
    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Run L-BFGS for this beta
        def fun(z):
            obj, grad = objective_softmax(z, beta)
            return obj, grad
        
        t_start = time.monotonic()
        max_time_for_this = (time_limit - (time.monotonic() - t0)) * 0.7
        t_start_abs = time.monotonic()
        
        # Use callback to monitor time
        stop_flag = [False]
        def callback(k):
            if time.monotonic() - t_start_abs > max_time_for_this:
                stop_flag[0] = True
            if verbose and k % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"      L-BFGS beta={beta:.1f} iter={k} t={elapsed:.0f}s")
            return stop_flag[0]
        
        try:
            res = minimize(fun, z, method='L-BFGS-B', jac=True, 
                          options={'maxiter': 500, 'maxfun': 500,
                                   'ftol': 1e-12, 'gtol': 1e-8,
                                   'callback': callback})
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at beta={beta}: {e}")
            continue
        
        if res.fun and res.x is not None:
            # Convert back to a
            z_res = res.x
            z_max = z_res.max()
            a_new = np.exp(z_res - z_max)
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)
            
            J_new = 2.0 * n * fast_autocorr(a_new).max()
            if np.isfinite(J_new) and J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"      ** New best at beta={beta:.1f}: J={best_J:.6f} (t={elapsed:.0f}s)")
            
            # Update z for next iteration
            z = np.log(a_new + 1e-30)
            z = z - z.mean()
    
    return best_a, best_J

def nesterov_pg(a0, time_limit, beta=50.0, c0=1.0, verbose=False):
    """Nesterov-accelerated projected gradient on the simplex."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    max_restart = 10000000
    cur_beta = beta
    
    rng = np.random.default_rng(42)
    
    for t in range(1, max_restart):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        x = y + beta_t * (y - x_prev)
        x = project_simplex(x)
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        
        # Compute gradient
        ac = fast_autocorr(x)
        p = logsumexp_weights(ac, cur_beta)
        g = autocorr_gradient(x, p)
        
        # Step
        eta = c0 / (np.sqrt(t) + 1.0)
        x_new = x - eta * g
        x_new = project_simplex(x_new)
        
        x_prev = y
        y = x_new
        
        # Check improvement
        M = fast_autocorr(y).max()
        J = 2.0 * n * M
        if J < best_M * 2.0 * n - 1e-12:  # comparing M directly
            if M < best_M:
                best_M = M
                best_a = y.copy()
                stall = 0
        else:
            stall += 1
        
        # Adaptive beta
        if stall > 2000 and cur_beta < 500:
            cur_beta = min(cur_beta * 1.5, 500.0)
            stall = 0
        
        # Restart
        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = project_simplex(x_r)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            y = x_r
            x_prev = x_r.copy()
            cur_beta = max(cur_beta * 0.8, 10.0)
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Nesterov restart: J={2*n*best_M:.6f} t={elapsed:.0f}s beta={cur_beta:.0f}")
    
    return best_a, best_M

def peak_shave(a0, time_limit, verbose=False, rng_seed=42):
    """Iterative peak shaving: identify top peaks and redistribute mass."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:10]
        
        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_k[:5]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_p = partner[np.argmax(contrib)]
            
            if a[src] < 1e-12:
                continue
            
            # Choose which to reduce
            if sens[src] >= sens[src_p]:
                reduce_idx = src
            else:
                reduce_idx = src_p
            
            # Find best destination (lowest sensitivity)
            dest = np.argmin(sens)
            if dest == reduce_idx:
                dest = (dest + 1) % n
            
            # Try to move mass
            found = False
            for frac in [0.005, 0.01, 0.02, 0.003, 0.03, 0.001, 0.05]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dest] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != reduce_idx:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            stall += 1
            if stall > 200:
                # Perturbation
                pert = rng.normal(0, 0.01, n) * best_a
                a_new = best_a + pert
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                a = a_new
                best_a = a_new.copy()
                best_M = fast_autocorr(a).max()
                stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave: J={2*n*best_M:.6f} stall={stall} t={elapsed:.0f}s")
    
    return best_a, best_M

def make_chirp(n, starts_before=False):
    """Generate a chirp-like sequence."""
    x = np.arange(n) / n
    a = np.abs(np.exp(1j * np.pi * x * x * n))  # This is just 1...
    # Let me use a different "chirp" - a sequence with linearly varying frequency
    # Actually for real non-negative, let's use a modulated sinusoid
    freq = np.linspace(1, 20, n)
    a = np.abs(np.sin(2 * np.pi * np.cumsum(freq) / n)) + 0.1
    return a / a.sum()

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return a / a.sum()

def make_bimodal(n, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.01
    return a / a.sum()

def spectral_init(n, seed=0, n_modes=20):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.2 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
