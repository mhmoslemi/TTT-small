    except Exception:
        return float('inf')


def _lp_solve(y, M, n, solver, K=None):
    """
    Solve the LP: maximize sum(b) subject to c_k(b; y) <= M for selected k, b >= 0.
    If K is None, use all constraints.
    """
    # Determine which constraints to use
    total_k = 2 * n - 1
    if K is None or K >= total_k:
        k_indices = np.arange(total_k)
    else:
        # Use the K constraints with highest c_k values
        ac = fast_autocorr(y)
        k_indices = np.argsort(ac)[::-1][:K]
    
    num_k = len(k_indices)
    
    # Build constraint matrix
    # T[k_idx, j] = y[k - j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    for ii, k in enumerate(k_indices):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        for j in range(j_start, j_end + 1):
            rows.append(ii)
            cols.append(j)
    
    rows = np.array(rows)
    cols = np.array(cols)
    
    if len(rows) == 0:
        return None
    
    k_vals = k_indices[rows]
    vals = y[k_vals - cols]
    
    T = sparse.csr_matrix((vals, (rows, cols)), shape=(num_k, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(num_k)]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s  # Normalize to simplex


def lp_frank_wolfe(a0, time_limit, solver, K=None, nesterov=True, verbose=True):
    """LP-based Frank-Wolfe with optional Nesterov acceleration."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            y = y + beta_t * (y - y_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        
        y_prev = y.copy()
        
        # Solve LP
        ac = fast_autocorr(y)
        M = ac.max()
        if M < 1e-16:
            break
        
        b = _lp_solve(y, M, n, solver, K=K)
        if b is None:
            # Fallback: use uniform direction
            b = np.ones(n) / n
        
        # Line search: try several step sizes
        best_alpha = 2.0 / (t + 2.0)  # Default FW step size
        best_val = fast_autocorr(y + best_alpha * (b - y)).max()
        
        for alpha in [1.0, 0.8, 0.5, 0.3, 0.1]:
            y_trial = (1.0 - alpha) * y + alpha * b
            y_trial = np.maximum(y_trial, 0.0)
            s_t = y_trial.sum()
            if s_t < 1e-15:
                continue
            y_trial = y_trial / s_t
            val = fast_autocorr(y_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        a = y_new
        my = fast_autocorr(a).max()
        if my < best_m:
            best_m = my
            best_a = a.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """Coordinate descent: move mass from peak-contributing to low-sensitivity positions."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_peaks = min(20, 2*n-1)
        top_k = np.argsort(ac)[::-1][:n_peaks]
        
        # Build sensitivity: how much does each position contribute to top peaks
        sens = np.zeros(n)
        peak_mask = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            peak_mask[idx] = True
            partner = ki - idx
            sens[idx] += a[partner]
        
        improved = False
        
        # Try moving mass from each peak-contributing position
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (low sensitivity, not in peak region)
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Score destinations by sensitivity
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]
            
            if dst == src:
                continue
            
            # Try different fractions
            found = False
            for frac in [0.005, 0.01, 0.02, 0.003, 0.05, 0.001]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found and stall > 20:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    a[src] -= a[src] * frac
                    a[j] += a[src] * frac  # Wait, this is wrong after the subtraction
                    # Let me redo:
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            # Global random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
