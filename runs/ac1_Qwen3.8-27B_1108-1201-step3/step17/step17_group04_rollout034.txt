    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def gs_spectral_equalization(a0, n_iter=500, time_limit=60.0, verbose=False, seed=42):
    """
    Gerchberg-Saxton style spectral equalization.
    Alternates between:
    - Time domain: enforce non-negativity and normalization
    - Frequency domain: flatten the power spectrum (minimize peak autocorrelation)
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # FFT size >= 2n-1 for linear convolution
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Frequency domain step: flatten power spectrum
        A = np.fft.rfft(a, N)
        power = A.real**2 + A.imag**2  # |A|^2
        
        # Target: flat power spectrum
        avg_power = power.sum() / len(power)
        
        # Gentle flattening (to avoid instability)
        # Use a "soft" version: new |A| = avg_power * (|A|/avg_power)^alpha
        # alpha < 1 flattens, alpha > 1 sharpens
        # We want to flatten, so use the inverse: where power is high, reduce it
        ratio = power / (avg_power + 1e-30)
        # New power: geometric mean of current and target (soft flattening)
        # new_power = avg_power * ratio^0.5 (this is a gentle step)
        new_ratio = ratio ** 0.5  # Square root = gentle flattening
        new_power = avg_power * new_ratio
        new_power = np.maximum(new_power, 0)
        
        # Update magnitude, keep phase
        phase = np.angle(A)
        new_mag = np.sqrt(new_power)
        A_new = new_mag * (np.cos(phase) + 1j * np.sin(phase))
        
        # Time domain: inverse FFT, clip, normalize
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter={it} ({elapsed:.1f}s): J={best_J:.6f}")
    
    if verbose:
        print(f"    GS final: J={best_J:.6f}")
    return best_a, best_J


def gs_with_ov(a0, time_limit=60.0, verbose=False, seed=42, ov=0.5):
    """
    GS with over-relaxation parameter.
    ov < 1: under-relaxation (more stable)
    ov > 1: over-relaxation (faster but less stable)
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    # Start with gentle flattening, gradually increase aggressiveness
    alpha = 0.3  # flattening exponent (0 = no change, 1 = full flatten)
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Gradually increase aggressiveness
        alpha = min(0.3 + it * 0.0002, 0.9)
        
        A = np.fft.rfft(a, N)
        power = A.real**2 + A.imag**2
        avg_power = power.sum() / len(power)
        
        ratio = power / (avg_power + 1e-30)
        new_ratio = ratio ** alpha
        new_power = avg_power * new_ratio
        new_power = np.maximum(new_power, 0)
        
        phase = np.angle(A)
        new_mag = np.sqrt(new_power)
        A_new = new_mag * (np.cos(phase) + 1j * np.sin(phase))
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        # Over/under-relaxation
        a = (1 - ov) * a + ov * a_new
        
        J_curr = compute_J(a)
        if J_curr < best_J:
            best_J = J_curr
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS-OV iter={it} ({elapsed:.1f}s): J={best_J:.6f}, alpha={alpha:.3f}")
    
    return best_a, best_J


def nesterov_subgradient(a0, time_limit, verbose=False,
                         top_k=15, beta=3.0, c=3.0,
                         exact_frac=0.4, restart_every=3000,
                         nesterov=True, seed=42):
    """Nesterov-accelerated mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        use_exact = rng.random() < exact_frac
        if use_exact:
            g = get_subgradient(y, ac, exact=True)
        else:
            g = get_subgradient(y, ac, exact=False, top_k=cur_top_k, beta=cur_beta)

        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.01 + 0.01 * (stall // restart_every)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 80)
            cur_beta = max(cur_beta - 0.3, 1.5)
            cur_c = min(cur_c + 0.5, 8.0)
            exact_frac = min(exact_frac + 0.05, 0.7)

    return best_a, best_J


def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    """Compute subgradient of max(conv(a,a))."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix (sparse structure)
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build T matrix for selected constraints
        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T_sel[row, idx] = y[j_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()

        y = y_new

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit=60.0, verbose=False, seed=42):
    """
    Final micro-optimization via coordinate descent.
    For each coordinate, try to adjust it optimally given all others fixed.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute the autocorrelation
    ac = fast_autocorr(a)
    
    # Order coordinates by sensitivity (how much they contribute to peaks)
    peak_idx = int(np.argmax(ac))
    # Coordinates that contribute to the peak
    i_start = max(0, peak_idx - n + 1)
    i_end = min(n, peak_idx + 1)
    peak_coords = np.arange(i_start, i_end)
    
    # Do multiple passes
    for pass_num in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random order for diversity
        order = rng.permutation(n)
        
        improved_this_pass = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try to optimize coordinate i
            # If we change a[i] by delta, how does the objective change?
            # a[i] appears in conv[k] for k in [i, i+n-1] as a[i]*a[k-i] (and symmetrically)
            
            # Simple: try a few values
            current_val = a[i]
            if current_val < 1e-10:
                # Try adding some mass
                candidates = [0, 0.001/n, 0.01/n, 0.05/n, 0.1/n]
            else:
                # Try reducing or slightly adjusting
                candidates = [0, current_val * 0.5, current_val * 0.8, current_val * 1.2, current_val * 2]
            
            best_candidate = current_val
            best_local_J = best_J
            
            for cand in candidates:
                cand = max(0, cand)
                a_trial = a.copy()
                a_trial[i] = cand
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_t = compute_J(a_trial)
                if J_t < best_local_J:
                    best_local_J = J_t
                    best_candidate = cand
            
            if best_candidate != current_val and best_local_J < best_J - 1e-14:
                a[i] = best_candidate
                a = a / a.sum()
                best_J = best_local_J
                best_a = a.copy()
                improved_this_pass = True
        
        if not improved_this_pass:
            break
    
    return best_a, best_J


def make_symmetric_init(n):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

1. Gerchberg-Saxton spectral equalization (iterative time/frequency projection)
   - Key innovation: directly targets flat power spectrum to equalize autocorrelation

4. Coordinate descent for final micro-optimization

import numpy as np

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:

    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def gs_spectral_equalization(a0, time_limit=120.0, verbose=False, seed=42, ov=0.6, alpha_start=0.2, alpha_end=0.85):
    """
    Gerchberg-Saxton spectral equalization.
    Alternates between frequency-domain flattening and time-domain non-negativity projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    max_iter = 100000
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive alpha: start gentle, become more aggressive
        frac = min(it / (max_iter * 0.7), 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Frequency domain: flatten power spectrum
        A = np.fft.rfft(a, N)
        power = A.real * A.real + A.imag * A.imag
        avg_power = power.sum() / len(power)

        ratio = power / (avg_power + 1e-30)
        new_ratio = ratio ** alpha
        new_power = avg_power * new_ratio
        new_power = np.maximum(new_power, 1e-30)

        # Keep phase, update magnitude
        mag = np.sqrt(A.real * A.real + A.imag * A.imag)
        phase = np.arctan2(A.imag, A.real)
        new_mag = np.sqrt(new_power)
        A_new = new_mag * np.cos(phase) + 1j * new_mag * np.sin(phase)

        # Time domain: inverse FFT, project onto non-negative, normalize
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        # Over/under-relaxation
        a = (1.0 - ov) * a + ov * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = a_new
        else:
            a = a / s

        J_curr = compute_J(a)
        if J_curr < best_J:
            best_J = J_curr
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter={it} ({elapsed:.1f}s): J={best_J:.6f}, alpha={alpha:.3f}")

    if verbose:
        print(f"    GS done: J={best_J:.6f}")
    return best_a, best_J


def gs_targeted(a0, time_limit=120.0, verbose=False, seed=42):
    """
    Targeted GS: only flatten power spectrum at frequencies that contribute to the peak autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    half_N = N // 2

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Find peak lag
        ac = fast_autocorr(a)
        peak_k = int(np.argmax(ac))

        # In frequency domain, autocorr[k] = (1/N) sum_m |A[m]|^2 exp(2*pi*i*m*k/N)
        # To reduce autocorr at peak_k, reduce |A[m]|^2 where exp term aligns with current value
        A = np.fft.rfft(a, N)
        power = A.real * A.real + A.imag * A.imag

        # Compute "contribution" of each frequency to the peak
        m = np.arange(half_N + 1)
        phase_contrib = np.cos(2.0 * np.pi * m * peak_k / N)
        # Frequencies with positive contribution to peak should be reduced
        weight = np.maximum(phase_contrib, 0.0)
        weight = weight / (weight.sum() + 1e-30)

        # Apply targeted reduction
        avg_power = power.sum() / len(power)
        reduction = 0.5  # How aggressively to reduce (0=no reduction, 1=full)
        new_power = power * (1.0 - reduction * weight * 0.5)
        new_power = np.maximum(new_power, 1e-30)

        # Also apply gentle global flattening
        ratio = new_power / (avg_power + 1e-30)
        new_power = avg_power * ratio ** 0.3
        new_power = np.maximum(new_power, 1e-30)

        mag = np.sqrt(A.real * A.real + A.imag * A.imag)
        phase = np.arctan2(A.imag, A.real)
        new_mag = np.sqrt(new_power)
        A_new = new_mag * np.cos(phase) + 1j * new_mag * np.sin(phase)

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        # Relaxation
        a = 0.4 * a + 0.6 * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = a_new
        else:
            a = a / s

        J_curr = compute_J(a)
        if J_curr < best_J:
            best_J = J_curr
            best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS-T iter={it} ({elapsed:.1f}s): J={best_J:.6f}")

    return best_a, best_J


def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_subgradient(a0, time_limit, verbose=False,
                          top_k=15, beta=3.0, c=3.0,
                          exact_frac=0.4, restart_every=3000,
                          seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.0) if t > 1 else 0.0
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        use_exact = rng.random() < exact_frac
        if use_exact:
            g = get_subgradient(y, ac, exact=True)
        else:
            g = get_subgradient(y, ac, exact=False, top_k=cur_top_k, beta=cur_beta)

        gap = M - 1.0 / n
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.01 + 0.005 * (t // (restart_every * 10))
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 80)
            cur_beta = max(cur_beta - 0.3, 1.5)
            cur_c = min(cur_c + 0.5, 8.0)
            exact_frac = min(exact_frac + 0.05, 0.7)

    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=400):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = np.zeros((K_sel, n))
        for row in range(K_sel):
            k_int = int(binding_idx[row])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T_sel[row, idx] = y[j_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()

        y = y_new

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit=60.0, verbose=False, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for pass_num in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        peak_idx = int(np.argmax(ac))
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n, peak_idx + 1)
        peak_coords = np.arange(i_start, i_end)

        # Focus on peak-contributing coordinates first
        other_coords = np.setdiff1d(np.arange(n), peak_coords)
        order = np.concatenate([rng.permutation(peak_coords), rng.permutation(other_coords)])

        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            cur = a[i]
            fracs = [0.0, 0.3, 0.6, 0.8, 1.2, 1.5, 2.0]
            candidates = [cur * f for f in fracs]
            if cur < 1e-8:
                candidates = [0.0, 0.001/n, 0.005/n, 0.02/n]

            best_cand = cur
            best_local = best_J

            for cand in candidates:
                cand = max(0.0, cand)
                if cand == cur:
                    continue
                a_trial = a.copy()
                a_trial[i] = cand
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_t = compute_J(a_trial)
                if J_t < best_local:
                    best_local = J_t
                    best_cand = cand

            if best_cand != cur and best_local < best_J - 1e-15:
                a[i] = best_cand
                a = a / a.sum()
                best_J = best_local
                best_a = a.copy()
                improved = True

        if not improved and pass_num > 10:
            break

    return best_a, best_J


def make_symmetric_init(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.7):
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_asymmetric_init(n):
    """Asymmetric: high on left, low on right."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.3) / 0.2) ** 2) + np.exp(-((x - 0.7) / 0.15) ** 2) * 0.5 + 0.005
    return norm_sum1(a)


def make_bimodal_init(n):
    """Two well-separated bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.25) / 0.1) ** 2) + np.exp(-((x - 0.75) / 0.1) ** 2) + 0.005
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Multi-stage optimization combining GS spectral equalization, LP Frank-Wolfe,
    Nesterov subgradient, and coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: GS Spectral Equalization (primary method) ===
    print(f"\n=== Phase 1: GS Spectral Equalization (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)

    if t1 > 30:
        # GS from parent with moderate over-relaxation
        a_in = norm_sum1(best_seq)
        a_out, _ = gs_spectral_equalization(a_in, time_limit=t1 * 0.5, verbose=True, seed=42, ov=0.6, alpha_start=0.2, alpha_end=0.85)
        try_update(a_out, "gs_parent_ov06")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # GS with higher over-relaxation from best
            a_in = norm_sum1(best_seq)
            a_out2, _ = gs_spectral_equalization(a_in, time_limit=min(remaining * 0.35, 150.0), verbose=False, seed=123, ov=0.8, alpha_start=0.3, alpha_end=0.9)
            try_update(a_out2, "gs_parent_ov08")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            # Targeted GS
            a_in = norm_sum1(best_seq)
            a_out3, _ = gs_targeted(a_in, time_limit=min(remaining * 0.35, 120.0), verbose=False, seed=456)
            try_update(a_out3, "gs_targeted")

    # === Phase 2: Nesterov SGD ===
    print(f"\n=== Phase 2: Nesterov SGD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t2, verbose=False,
                                        top_k=15, beta=3.0, c=3.0,
                                        exact_frac=0.4, restart_every=2000,
                                        seed=42)
        try_update(a_out, "sgd_phase2")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n_best}) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=400)
        try_update(a_out, "lpfw_phase3")

    # === Phase 4: GS polish from LP result ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: GS Polish (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = gs_spectral_equalization(a_in, time_limit=t4, verbose=False, seed=789, ov=0.5, alpha_start=0.4, alpha_end=0.9)
        try_update(a_out, "gs_polish")

    # === Phase 5: SGD v2 with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: SGD v2 (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t5, verbose=False,
                                        top_k=30, beta=2.0, c=5.0,
                                        exact_frac=0.6, restart_every=5000,
                                        seed=999)
        try_update(a_out, "sgd_v2")

    # === Phase 6: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coordinate Descent (n={n_best}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, time_limit=t6, verbose=True, seed=42)
        try_update(a_out, "coord_descent")

    # === Phase 7: Second LP pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP Pass 2 (n={n_best}) ===")
    if remaining > 40 and n_best <= 3500:
        t7 = min(remaining * 0.30, 250.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=500)
            try_update(a_out, "lpfw_pass2")

    # === Phase 8: Final GS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final GS (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = gs_spectral_equalization(a_in, time_limit=t8, verbose=False, seed=3333, ov=0.4, alpha_start=0.5, alpha_end=0.95)
            try_update(a_out, "gs_final")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Coord (n={n_best}) ===")
    if remaining > 10:
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 5:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent(a_in, time_limit=t9, seed=777)
            try_update(a_out, "coord_final")

    # === Phase 10: One more SGD if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final SGD (n={n_best}) ===")
        t10 = min(remaining * 0.6, 100.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_subgradient(a_in, t10, verbose=False,
                                            top_k=40, beta=2.0, c=5.0,
                                            exact_frac=0.6, restart_every=8000,
                                            seed=4321)
            try_update(a_out, "sgd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
