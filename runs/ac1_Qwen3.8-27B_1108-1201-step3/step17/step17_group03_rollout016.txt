```python
def compute_logsumexp_grad(a, beta=100):
    """Compute gradient of logsumexp(beta * autocorr(a)) w.r.t. a"""
    n = len(a)
    c = fast_autocorr(a)  # shape (2n-1,)
    log_max = np.max(beta * c)
    weights = np.exp(beta * c - log_max)
    weights = weights / weights.sum()  # softmax(beta * c)
    
    # Gradient w.r.t. a_i: sum_k weights[k] * (a[i+k-(n-1)] + a[i-k+(n-1)])
    # c[k] = sum_i a[i] * a[i+k-(n-1)] for valid i
    # dc[lag]/da[i] = a[i+lag] (if valid) + a[i-lag] (if valid)
    # where lag = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        w = weights[k]
        if w < 1e-10:
            continue
        # c[k] = sum_i a[i] * a[i+lag] where both indices valid
        # For lag >= 0: i from 0 to n-1-lag, c[k] += a[i]*a[i+lag]
        # dc[k]/da[i] = a[i+lag] for i in [0, n-1-lag] + a[i-lag] for i in [lag, n-1]
        if lag >= 0:
            i_vals = np.arange(n - lag)
            grad[i_vals] += w * a[i_vals + lag]
            if lag > 0:
                i_vals2 = np.arange(lag, n)
                grad[i_vals2] += w * a[i_vals2 - lag]
        else:
            lag_abs = -lag
            i_vals = np.arange(n - lag_abs)
            grad[i_vals] += w * a[i_vals + lag_abs]
    
    return grad
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-domain Adam with soft-max (logsumexp) gradient for smooth optimization
2. Adaptive beta scheduling for gradual peak sharpening
3. Peak-crushing coordinate descent refinement
4. Multi-length exploration with interpolation
5. Spectral diversification for escape from local minima
"""

import numpy as np
import time

_global_t0 = None

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_smooth_grad(a, beta=50.0, top_k=300):
    """
    Compute gradient of logsumexp(beta * autocorr(a)) w.r.t. a.
    Uses top-k lags for efficiency.
    """
    n = len(a)
    c = fast_autocorr(a)
    
    # Get top-k autocorrelation values
    k_eff = min(top_k, len(c))
    top_idx = np.argpartition(c, -k_eff)[-k_eff:]
    top_c = c[top_idx]
    
    # Softmax weights on top-k
    exp_beta_c = np.exp(beta * (top_c - top_c.max()))
    weights = exp_beta_c / exp_beta_c.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # dc_k/da[m] includes a[m+lag] for m in [0, n-1-lag]
            valid_m = np.arange(0, n - lag)
            grad[valid_m] += w * a[valid_m + lag]
            if lag > 0:
                # and a[m-lag] for m in [lag, n-1]
                valid_m2 = np.arange(lag, n)
                grad[valid_m2] += w * a[valid_m2 - lag]
        else:
            la = -lag
            valid_m = np.arange(0, n - la)
            grad[valid_m] += w * a[valid_m + la]
    
    return grad

def softmax_params_to_a(z):
    """Convert log-parameters to probability vector."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def a_to_softmax_grad(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. log-params (softmax)."""
    # dJ/dz_j = a_j * (dJ/da_j - sum_i dJ/da_i * a_i)
    inner = np.dot(grad_a, a)
    return a * (grad_a - inner)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def adam_step(z, grad_z, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step in log-domain."""
    m = beta1 * m + (1 - beta1) * grad_z
    v = beta2 * v + (1 - beta2) * (grad_z ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

def peak_crush(a, n_rounds=50, n_trials=10, rng=None):
    """Aggressive peak-crushing via mass redistribution."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for _ in range(n_rounds):
        ac = fast_autocorr(a)
        M = ac.max()
        peak_k = int(np.argmax(ac))
        lag = peak_k - (n - 1)
        
        # Find the top contributors to this peak
        if lag >= 0:
            valid = np.arange(0, n - lag)
            contrib = a[valid] * a[valid + lag]
        else:
            la = -lag
            valid = np.arange(0, n - la)
            contrib = a[valid] * a[valid + la]
        
        if contrib.sum() < 1e-20:
            break
        
        # Find the element with max contribution
        top_idx = np.argsort(contrib)[-min(5, len(contrib)):]
        
        improved = False
        for ti in top_idx:
            src = int(valid[ti])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            best_new_m = M
            best_move = None
            for frac in [0.01, 0.03, 0.07, 0.15]:
                amt = src_val * frac
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)
            
            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-15:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
    
    return best_a

def spectral_perturb(a, strength=0.02, rng=None):
    """Perturb in frequency domain for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def logsumexp_optimize(a0, time_limit, lr=0.05, beta_start=20.0, beta_end=200.0,
                       top_k=300, verbose=True, rng_seed=42):
    """
    Main optimizer using log-domain Adam with adaptive logsumexp gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall_count = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Adaptive beta
        progress = min(1.0, elapsed / time_limit)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
        
        # Compute gradient
        a_current = softmax_params_to_a(z)
        grad_a = compute_smooth_grad(a_current, beta=beta, top_k=top_k)
        grad_z = a_to_softmax_grad(grad_a, a_current)
        
        # Adam step
        z, m, v = adam_step(z, grad_z, m, v, t, lr=lr)
        
        # Convert back and normalize
        a_new = softmax_params_to_a(z)
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count > max_stall:
            stall_count = 0
            # Perturb from best
            pert = rng.normal(0, 0.01, n) * best_a
            z = np.log(np.maximum(best_a + pert, 1e-10))
            z = z - z.mean()  # Center for numerical stability
            m = np.zeros(n)
            v = np.zeros(n)
            if verbose:
                print(f"      Restart at t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
        
        if verbose and t % 500 == 0:
            print(f"      Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.1f}")
    
    return best_a

def make_parametric(n, family='sine'):
    """Create parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'sech':
        x = np.linspace(-2, 2, n)
        return norm_sum1(1.0 / np.cosh(x))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update best sequence/score."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-domain Adam with adaptive logsumexp (smooth max) gradient
2. Adaptive beta scheduling for gradual peak sharpening
3. Aggressive peak-crushing coordinate descent refinement
4. Multi-length exploration with interpolation
5. Spectral diversification for escape from local minima
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_smooth_grad(a, beta=50.0, top_k=300):
    """
    Compute gradient of logsumexp(beta * autocorr(a)) w.r.t. a.
    Uses top-k lags for efficiency.
    """
    n = len(a)
    c = fast_autocorr(a)

    k_eff = min(top_k, len(c))
    top_idx = np.argpartition(c, -k_eff)[-k_eff:]
    top_c = c[top_idx]

    exp_beta_c = np.exp(beta * (top_c - top_c.max()))
    w_sum = exp_beta_c.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_eff) / k_eff
    else:
        weights = exp_beta_c / w_sum

    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            valid_m = np.arange(0, n - lag)
            grad[valid_m] += w * a[valid_m + lag]
            if lag > 0:
                valid_m2 = np.arange(lag, n)
                grad[valid_m2] += w * a[valid_m2 - lag]
        else:
            la = -lag
            valid_m = np.arange(0, n - la)
            grad[valid_m] += w * a[valid_m + la]

    return grad


def softmax_params_to_a(z):
    """Convert log-parameters to probability vector."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def a_to_softmax_grad(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. log-params (softmax)."""
    inner = np.dot(grad_a, a)
    return a * (grad_a - inner)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def peak_crush(a, n_rounds=100, n_trials=8, rng=None, verbose=False):
    """Aggressive peak-crushing via mass redistribution."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > 1740:
            break
        ac = fast_autocorr(a)
        M = ac.max()
        peak_k = int(np.argmax(ac))
        lag = peak_k - (n - 1)

        if lag >= 0:
            valid = np.arange(0, n - lag)
            contrib = a[valid] * a[valid + lag]
        else:
            la = -lag
            valid = np.arange(0, n - la)
            contrib = a[valid] * a[valid + la]

        if contrib.sum() < 1e-20:
            break

        top_count = min(5, len(contrib))
        top_idx = np.argsort(contrib)[-top_count:]

        improved = False
        for ti in top_idx:
            src = int(valid[ti])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M
            best_move = None
            for frac in [0.005, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-15:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            i_idx = int(rng.integers(n))
            j_idx = int(rng.integers(n))
            if i_idx != j_idx and a[i_idx] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i_idx] * frac
                a[i_idx] -= amt
                a[j_idx] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-15:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True

        if verbose and rnd % 20 == 0:
            print(f"        PeakCrush round {rnd}: J={2*n*best_m:.6f}")

    return best_a


def spectral_perturb(a, strength=0.02, rng=None):
    """Perturb in frequency domain for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def logsumexp_optimize(a0, time_limit, lr=0.03, beta_start=10.0, beta_end=200.0,
                       top_k=300, verbose=True, rng_seed=42, max_stall=400):
    """
    Main optimizer using log-domain Adam with adaptive logsumexp gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall_count = 0
    lr_current = lr
    patience_counter = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0

        # Adaptive beta: start smooth, end sharp
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)

        # Adaptive top_k: start with fewer, increase over time
        cur_top_k = int(top_k * (0.5 + 0.5 * progress))

        # Compute gradient
        a_current = softmax_params_to_a(z)
        grad_a = compute_smooth_grad(a_current, beta=beta, top_k=cur_top_k)
        grad_z = a_to_softmax_grad(grad_a, a_current)

        # Adam step with decaying learning rate
        lr_t = lr_current / (1.0 + 0.01 * progress)
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Convert back
        a_new = softmax_params_to_a(z)

        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
            patience_counter = 0
        else:
            stall_count += 1
            patience_counter += 1

        # Reduce learning rate on plateau
        if patience_counter > 200:
            lr_current *= 0.8
            patience_counter = 0

        # Restart if stalled too long
        if stall_count > max_stall:
            stall_count = 0
            pert = rng.normal(0, 0.01 * (1 + 0.5 * progress), n) * best_a
            z_restart = np.log(np.maximum(best_a + pert, 1e-10))
            z_restart = z_restart - z_restart.mean()
            z = z_restart
            m = np.zeros(n)
            v = np.zeros(n)
            lr_current = lr
            if verbose:
                print(f"      Adam restart t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

        if verbose and t % 1000 == 0:
            print(f"      Adam t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.0f}, lr={lr_current:.4f}")

    return best_a


def make_parametric(n, family='sine'):
    """Create parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'sech':
        x = np.linspace(-2.5, 2.5, n)
        return norm_sum1(1.0 / np.cosh(x))
    elif family == 'raised_cos':
        x = np.linspace(-1, 1, n)
        return norm_sum1(0.5 + 0.5 * np.cos(np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update best sequence/score."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function combining log-domain Adam with logsumexp gradient,
    peak-crushing refinement, multi-length exploration, and spectral diversification.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Adam logsumexp optimization on parent ===
    print(f"\n=== Phase 1: LogAdam main (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out = logsumexp_optimize(a_in, t1, lr=0.04, beta_start=15.0,
                                    beta_end=150.0, top_k=250, verbose=True,
                                    rng_seed=42, max_stall=300)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "logadam1")

    # === Phase 2: Peak crushing refinement ===
    print(f"\n=== Phase 2: Peak Crush (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 120.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        rng = np.random.default_rng(123)
        a_out = peak_crush(a_in, n_rounds=int(min(300, t2 * 3)), n_trials=8,
                           rng=rng, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "peakcrush1")

    # === Phase 3: Second Adam with perturbation and higher beta ===
    print(f"\n=== Phase 3: LogAdam sharp (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        rng = np.random.default_rng(77)
        a_pert = a_in + rng.normal(0, 0.003, len(a_in)) * a_in
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out = logsumexp_optimize(a_pert, t3, lr=0.02, beta_start=50.0,
                                    beta_end=350.0, top_k=400, verbose=True,
                                    rng_seed=77, max_stall=250)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "logadam2")

    # === Phase 4: Peak crush again ===
    print(f"\n=== Phase 4: Peak Crush 2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.07, 100.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        rng = np.random.default_rng(456)
        a_out = peak_crush(a_in, n_rounds=int(min(200, t4 * 2)), n_trials=10,
                           rng=rng, verbose=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "peakcrush2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    lengths_to_try = []
    for delta in [300, -300, 600, -600, 1000, -1000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 5500:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - n_best))

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining: {remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t_try = min(remaining * 0.12, 150.0)
        if t_try > 30:
            a_out = logsumexp_optimize(a_long, t_try, lr=0.03, beta_start=15.0,
                                        beta_end=120.0, top_k=200, verbose=False,
                                        rng_seed=new_n % 1000, max_stall=200)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out = peak_crush(a_out, n_rounds=50, n_trials=5,
                                   rng=np.random.default_rng(new_n), verbose=False)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                              t_global, f"len_{new_n}")

    # === Phase 6: Spectral diversification ===
    print(f"\n=== Phase 6: Spectral diversify ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    rng = np.random.default_rng(99)

    if remaining > 80:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_pert = spectral_perturb(best_seq, strength=0.025, rng=rng)
            t_trial = min(remaining * 0.10, 120.0)
            if t_trial > 25:
                a_out = logsumexp_optimize(a_pert, t_trial, lr=0.03,
                                            beta_start=30.0, beta_end=200.0,
                                            top_k=250, verbose=False,
                                            rng_seed=100 + trial, max_stall=200)
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 15:
                    a_out = peak_crush(a_out, n_rounds=40, n_trials=5,
                                       rng=np.random.default_rng(200 + trial), verbose=False)
                best_seq, best_score = try_update(a_out, best_seq, best_score,
                                                  t_global, f"spectral{trial}")

    # === Phase 7: Final Adam with very high beta ===
    print(f"\n=== Phase 7: Final sharp Adam (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t7 = min(remaining * 0.25, 200.0)
        if t7 > 25:
            a_in = norm_sum1(best_seq)
            a_out = logsumexp_optimize(a_in, t7, lr=0.015, beta_start=100.0,
                                        beta_end=500.0, top_k=500, verbose=True,
                                        rng_seed=4242, max_stall=200)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "final_adam")

    # === Phase 8: Final peak crush ===
    print(f"\n=== Phase 8: Final peak crush (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.35, 150.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            rng = np.random.default_rng(555)
            a_out = peak_crush(a_in, n_rounds=int(min(400, t8 * 4)), n_trials=12,
                               rng=rng, verbose=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "final_pc")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
```
