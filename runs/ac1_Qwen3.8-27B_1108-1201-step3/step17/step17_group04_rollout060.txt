def fast_autocorr(a, fft_size=None):

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_subgradient(a, ac, exact=False, top_k=10, beta_soft=3.0, fft_size=None):
    """Compute subgradient of max(conv(a,a)) or soft approximation."""
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    xcorr_size = 2 * fft_size
    grad = np.zeros(n)

    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_soft
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.0)
            cur_c = min(cur_c + 0.3, 10.0)
            cur_exact_frac = min(cur_exact_frac + 0.03, 0.8)

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


# ========== Spectral Shaping ==========

def spectral_shaping(a0, time_limit, alpha=0.7, n_iter=1000, verbose=True, seed=123, fft_size=None):
    """
    Iterative spectral shaping: modify the power spectrum to be flatter.
    |A_new(f)| = |A(f)|^alpha compresses dynamic range toward flat.
    """
    n = len(a0)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    cur_alpha = alpha
    stall = 0

    for iteration in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Modify magnitude: compress toward flat
        mag_max = mag.max()
        if mag_max < 1e-15:
            break

        # Normalize magnitude, then compress
        mag_norm = mag / mag_max
        mag_new = mag_norm ** cur_alpha * mag_max

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project onto simplex (clip negatives, normalize)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Fallback: use original
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try with slightly different alpha
            if iteration % 50 == 0:
                cur_alpha = max(0.3, min(0.95, cur_alpha + rng.uniform(-0.05, 0.05)))
            stall += 1
            if stall > 200:
                # Random perturbation to escape
                n_pert = max(10, n // 50)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    amt = a[p] * rng.uniform(0.01, 0.1)
                    a[p] -= amt
                    dst = int(rng.integers(n))
                    a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_test = compute_J(a)
                if J_test < best_J:
                    best_J = J_test
                    best_a = a.copy()
                    stall = 0
                else:
                    a = best_a.copy()
                stall = 0

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={iteration} ({elapsed:.0f}s): J={best_J:.6f}, alpha={cur_alpha:.3f}")

    return best_a, best_J


# ========== Simulated Annealing Polish ==========

def simulated_annealing_polish(a0, time_limit, verbose=False, seed=999, fft_size=None):
    """
    Simulated annealing: try random perturbations, accept with probability exp(-dJ/T).
    Good for fine-tuning after main optimization.
    """
    n = len(a0)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_start = (best_J - 1.0) * 0.01  # Initial temperature
    T_end = T_start * 0.001
    n_total = 50000

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break

        # Annealing schedule
        progress = it / n_total
        T = T_start * (T_end / T_start) ** progress

        # Random perturbation: move mass from random source to random destination
        n_moves = rng.integers(1, 5)
        a_trial = a.copy()
        for _ in range(n_moves):
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst:
                continue
            frac = rng.uniform(0.001, 0.05)
            amt = a_trial[src] * frac
            if amt < 1e-12:
                continue
            a_trial[src] -= amt
            a_trial[dst] += amt

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        J_trial = compute_J(a_trial)
        dJ = J_trial - best_J

        if dJ < 0 or rng.random() < np.exp(-dJ / max(T, 1e-15)):
            a = a_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={it} ({elapsed:.0f}s): J={best_J:.6f}, T={T:.6f}")

    return best_a, best_J


# ========== Peak Reduction Moves ==========

def peak_reduction_moves(a0, time_limit, verbose=False, seed=77, fft_size=None):
    """Targeted coordinate moves: relocate mass from peak contributors to safe zones."""
    n = len(a0)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a, fft_size)
        M_curr = ac.max()

        n_peaks = min(10, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc

        sens_mean = sensitivity.mean()
        sens_std = sensitivity.std()
        high_sens = np.where((sensitivity > sens_mean + sens_std) & (a > 0.0005))[0]
        low_sens = np.where((sensitivity < sens_mean - 0.3 * sens_std))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            fracs = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Larger perturbation
                n_pert = max(5, n // 100)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


# ========== LP Frank-Wolfe (Simplified) ==========

def lp_frank_wolfe(a0, time_limit, verbose=False, top_K=300, fft_size=None):
    """Simplified LP Frank-Wolfe using cvxpy."""
    try:
        import cvxpy as cp
    except ImportError:
        if verbose:
            print("    cvxpy not available, skipping LP-FW")
        return a0, compute_J(a0)

    n = len(a0)
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a, fft_size).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix for top-K constraints
        ac = fast_autocorr(y, fft_size)
        M_curr = ac.max()

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build T matrix (sparse-ish)
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T_sel[ii, idx] = y[j_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Minimize(cp.max(cp.norm(M_curr * cp.ones(K_sel) - T_sel @ b_var, np.inf))), constraints)

        # Actually, maximize sum(b_var) subject to T_sel @ b_var <= M_curr
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        my = fast_autocorr(y, fft_size).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


# ========== Main Search Orchestrator ==========

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining L-BFGS, Nesterov, spectral shaping,
    simulated annealing, peak reduction, and LP Frank-Wolfe.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    fft_size = 1 << (2 * n0 - 1).bit_length()
    print(f"Starting: n={n0}, score={best_score:.6f}, fft_size={fft_size}")

    # === Phase 1: L-BFGS Soft-Max (main innovation) ===
    print(f"\n=== Phase 1: L-BFGS Soft-Max (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)

    if t1 > 60:
        # Try from multiple initializations
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Slightly perturbed parent
        if parent is not None:
            rng_init = np.random.default_rng(42)
            pert = rng_init.normal(0, 0.01, n0) * norm_sum1(parent)
            inits.append(norm_sum1(parent + pert))
            labels.append("parent_pert")

        inits.append(make_symmetric_init(n0))
        labels.append("symmetric")
        inits.append(make_gaussian_edges(n0, sigma=0.12))
        labels.append("gauss")

        # Allocate time among initializations
        n_trials = min(len(inits), 3)  # Limit to 3 L-BFGS trials
        t_per = t1 / n_trials

        for idx in range(n_trials):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 30:
                break
            print(f"  L-BFGS init '{labels[idx]}' (t={t_avail:.0f}s)...")
            a_out, _ = lbfgs_soft_max(inits[idx], t_avail,
                                       beta_start=n0, beta_end=5 * n0,
                                       n_betas=4, verbose=True, fft_size=fft_size)
            try_update(a_out, f"lbfgs_{labels[idx]}")

    # === Phase 2: Nesterov Subgradient Polish ===
    print(f"\n=== Phase 2: Nesterov SGD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    fft_best = 1 << (2 * n_best - 1).bit_length()
    t2 = min(remaining * 0.20, 350.0)

    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t2, verbose=True,
                                        top_k=20, beta_soft=2.5, c_step=4.0,
                                        exact_frac=0.5, restart_every=2000,
                                        seed=42, fft_size=fft_best)
        try_update(a_out, f"sgd_n{n_best}")

    # === Phase 3: Spectral Shaping ===
    print(f"\n=== Phase 3: Spectral Shaping (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    fft_best = 1 << (2 * n_best - 1).bit_length()
    t3 = min(remaining * 0.15, 250.0)

    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_shaping(a_in, t3, alpha=0.7, n_iter=2000,
                                    verbose=True, seed=123, fft_size=fft_best)
        try_update(a_out, f"spectral_n{n_best}")

    # === Phase 4: L-BFGS again on best (with different beta range) ===
    print(f"\n=== Phase 4: L-BFGS refine (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    fft_best = 1 << (2 * n_best - 1).bit_length()
    t4 = min(remaining * 0.15, 250.0)

    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_soft_max(a_in, t4,
                                   beta_start=2 * n_best, beta_end=10 * n_best,
                                   n_betas=3, verbose=True, fft_size=fft_best)
        try_update(a_out, f"lbfgs_refine_n{n_best}")

    # === Phase 5: Peak Reduction ===
    print(f"\n=== Phase 5: Peak Reduction (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    fft_best = 1 << (2 * n_best - 1).bit_length()
    t5 = min(remaining * 0.12, 200.0)

    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_moves(a_in, t5, verbose=True,
                                        seed=77, fft_size=fft_best)
        try_update(a_out, f"peakred_n{n_best}")

    # === Phase 6: Simulated Annealing ===
    print(f"\n=== Phase 6: Simulated Annealing (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    fft_best = 1 << (2 * n_best - 1).bit_length()
    t6 = min(remaining * 0.10, 150.0)

    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = simulated_annealing_polish(a_in, t6, verbose=True,
                                              seed=999, fft_size=fft_best)
        try_update(a_out, f"sa_n{n_best}")

    # === Phase 7: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP-FW (n={n_best}) ===")
    t7 = min(remaining * 0.10, 200.0)

    if t7 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, verbose=True, top_K=400)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 8: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")

    for new_n in [n_best + 200, n_best - 200]:
        if new_n < 1000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        fft_new = 1 << (2 * new_n - 1).bit_length()
        a_long = interpolate_sequence(best_seq, new_n)
        t8a = min(remaining * 0.3, 120.0)
        if t8a > 30:
            print(f"  Trying n={new_n}...")
            a_out, _ = lbfgs_soft_max(a_long, t8a,
                                       beta_start=new_n, beta_end=4 * new_n,
                                       n_betas=3, verbose=False, fft_size=fft_new)
            try_update(a_out, f"lbfgs_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = nesterov_subgradient(norm_sum1(best_seq) if len(best_seq) == new_n else a_out,
                                                  min(remaining * 0.4, 60.0),
                                                  verbose=False, top_k=20,
                                                  fft_size=fft_new)
                try_update(a_out2, f"sgd_n{new_n}")

    # === Phase 9: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    fft_best = 1 << (2 * n_best - 1).bit_length()
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")

    if remaining > 20:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_subgradient(a_in, t9, verbose=False,
                                            top_k=30, beta_soft=2.0, c_step=5.0,
                                            exact_frac=0.7, restart_every=5000,
                                            seed=4321, fft_size=fft_best)
            try_update(a_out, f"sgd_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
