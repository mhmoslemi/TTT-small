For the active peak k*, the function (locally) is a^T B_{k*} a. The Hessian is 2*B_{k*}. We can use this to compute a Newton-like step:
a_new = a - eta * (H + lambda*I)^{-1} * g

But this is expensive for large n. Instead, we can use a first-order approximation with a better step size.

**Step size selection**: Use backtracking line search to find the optimal step size.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient mirror descent with adaptive step size
2. LP-based Frank-Wolfe for global optimization
3. Multi-peak subgradient (average of top-M peak gradients)
4. Smart restarts with perturbation
5. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def exact_subgradient(a, ac, M=5):
    """Compute exact subgradient using top-M peaks."""
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:min(M, len(ac))]
    # Subgradient is convex combo of 2*B_k*a for k in top_idx
    g = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        # B_k * a: (B_k * a)[i] = a[ki - i] if 0 <= ki-i < n, else 0
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        partner = ki - idx
        g[idx] += 2.0 * a[partner]
    g /= len(top_idx)  # average
    return g


def mirror_descent_optimize(a0, time_limit, M=5, c0=3.0, verbose=False, rng_seed=42):
    """Mirror descent with exact subgradient and adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    t = 0
    stall = 0
    eta = c0 / np.sqrt(1.0)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(a)
        g = exact_subgradient(a, ac, M=M)
        
        # Mirror descent update (exponential family / entropic mirror)
        eta_t = eta / np.sqrt(t)
        log_a = np.log(np.maximum(a, 1e-30)) - eta_t * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive: if stalled, perturb
        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            eta = max(c0 * 0.8, eta * 0.7)
            
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=None):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    if top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(500, 2 * n - 1)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix for top-K constraints
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Select binding constraints (closest to M_curr)
            constraint_vals = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                ki = int(k)
                s = max(0, ki - n + 1)
                e = min(n - 1, ki)
                idx = np.arange(s, e + 1)
                constraint_vals[k] = np.dot(y[idx], y[ki - idx])
            binding_idx = np.argsort(constraint_vals)[::-1][:K_sel]
        
        # Build LP: max sum(b) s.t. b >= 0, T_sel @ b <= M_curr
        T_sel = np.zeros((len(binding_idx), n))
        for row, k in enumerate(binding_idx):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            T_sel[row, idx] = y[ki - idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(len(binding_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated FW
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
        
        # Adaptively increase K
        if t % 30 == 0:
            K_sel = min(K_sel + 50, 2 * n - 1)
    
    return best_a, best_m


def targeted_gradient_descent(a0, time_limit, M=10, verbose=False, rng_seed=42):
    """
    Targeted gradient descent: at each step, compute the gradient of the
    top-M peaks (as a subgradient of the max), and take a mirror descent step.
    Uses backtracking line search for step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = 0.1
    stall = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit and stall < 20000:
        t += 1
        ac = fast_autocorr(a)
        g = exact_subgradient(a, ac, M=M)
        
        # Mirror descent step with backtracking
        eta_t = eta
        for _ in range(10):
            log_a = np.log(np.maximum(a, 1e-30)) - eta_t * g
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            s = a_trial.sum()
            if s < 1e-15:
                break
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m:
                a = a_trial
                if M_trial < best_m - 1e-16:
                    best_m = M_trial
                    best_a = a_trial.copy()
                stall = 0
                break
            else:
                eta_t *= 0.5
                if eta_t < 1e-10:
                    break
        else:
            # No improvement found, try smaller or perturb
            stall += 1
            if stall % 100 == 0:
                eta = max(0.01, eta * 0.8)
                if stall % 500 == 0:
                    pert = rng.normal(0, 0.01, n) * a
                    a = a + pert
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    stall = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    TGD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def spectral_init(n, n_modes=30, seed=0):
    """Generate smooth non-negative sequence via Fourier synthesis."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):

def make_bimodal(n, width=0.15):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Wait, I realize there might be an issue. The `evaluate_sequence` function clips to [0, 1000] and requires sum >= 0.01. Since we're normalizing to sum=1, that's fine.

Let me also think about whether there's a fundamentally better approach.

Actually, I just realized something important. The previous algorithm achieved 1.504120 with n=4200. The theoretical lower bound for this problem (with non-negative sequences) is actually related to the "optimal aperiodic autocorrelation" problem.

For the specific formulation J(a) = 2n * max_k R(k) / (sum a)^2, the minimum over all non-negative sequences of length n... let me think about what happens as n → ∞.

For large n, if we think of a as a density function on [0,1], then:
- R(k/n) ≈ n * (f*f)(k/n) where (f*f) is the convolution of f with itself
- sum a ≈ n * ∫f
- J ≈ 2n * n * max(f*f) / (n * ∫f)^2 = 2 * max(f*f) / (∫f)^2

For f = 1 (uniform): max(f*f) = 1 (at the center), ∫f = 1, so J → 2.

Can we do better in the continuous limit? We need to minimize max(f*f)(x) / (∫f)^2 for f >= 0 on [0,1].

If f is supported on a subset of [0,1] of measure α < 1, then (f*f)(x) is supported on [0, 2α] (or [0, 2] if the support is [0, α]). The peak of f*f at x = α is ∫ f(t)f(α-t)dt. If f is uniform on [0, α], then f = 1/α on [0,α], and (f*f)(x) = (min(x, α, 2α-x, ...)) / α^2 for x in [0, 2α]. The peak is at x = α: (f*f)(α) = α/α^2 = 1/α. And ∫f = 1. So J_cont = 2*(1/α)/1 = 2/α > 2.

So concentrating the function makes it worse. Spreading it out (uniform) is better in the continuous limit.

But we're already at 1.5 which is LESS than 2! How is that possible?

Oh wait, I think I made an error in the continuous limit calculation. Let me redo it.

In the discrete case, J = 2n * max_k R(k) / (sum a)^2.

If a_i ≈ f(i/n) / n (so that sum a_i ≈ ∫f = 1), then:
R(k) = sum_i a_i * a_{k-i} ≈ sum_i (f(i/n)/n) * (f((k-i)/n)/n) ≈ (1/n) * ∫ f(t) f(k/n - t) dt = (1/n) * (f*f)(k/n)

So max_k R(k) ≈ (1/n) * max (f*f)

J = 2n * (1/n) * max(f*f) / 1^2 = 2 * max(f*f)

For uniform f = 1 on [0,1]: max(f*f) = 1 (the peak of the triangular function at x=1 is 1).
So J → 2.

For J < 2, we need max(f*f) < 1. Since (f*f)(1) = ∫_0^1 f(t)f(1-t)dt, we need this to be < 1.

By Cauchy-Schwarz: (∫_0^1 f(t)dt)^2 <= ∫_0^1 f(t)^2 dt * 1 = ∫ f^2

And (f*f)(1) = ∫_0^1 f(t)f(1-t)dt. By Cauchy-Schwarz: (f*f)(1)^2 <= (∫f^2)(∫f^2) = (∫f^2)^2 (since f(1-t) has the same L2 norm as f(t)).

So (f*f)(1) <= ∫f^2.

Also, by the same identity, (f*f)(1) = ∫_0^1 f(t)f(1-t)dt.

            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            no_improve = 0

        # Adaptive K increase
        if t % 15 == 0:
            K_sel = min(K_sel + 50, K_max)

        # If no improvement for a while, try tighter M
        if use_tighten and no_improve > 50 and no_improve % 50 == 0:
            # Try to find a direction under a tighter peak
            M_tight = M_curr * tighten_factor
            T_sel2 = np.zeros((len(binding_idx), n))
            for row_i, k in enumerate(binding_idx):
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n - 1, ki)
                if i_start <= i_end:
                    idx = np.arange(i_start, i_end + 1)
                    T_sel2[row_i, idx] = y[ki - idx]
            b_var2 = cp.Variable(n, nonneg=True)
            constraints2 = [T_sel2 @ b_var2 <= M_tight * np.ones(len(binding_idx))]
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
            try:
                prob2.solve(solver=solver, verbose=False)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b2 = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    if b2.sum() > 1e-10:
                        g2 = b2 / b2.sum()
                        y_try = (1 - 0.3) * y + 0.3 * g2
                        y_try = np.maximum(y_try, 0.0)
                        y_try = y_try / y_try.sum()
                        m_try = fast_autocorr(y_try).max()
                        if m_try < best_m:
                            best_m = m_try
                            best_a = y_try.copy()
                            y = y_try
                            no_improve = 0
            except Exception:
                pass

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def peak_reduction_search(a0, time_limit, verbose=True, rng_seed=42, n_top=20):
    """
    Targeted peak reduction: identify top contributing pairs to peaks and
    redistribute mass to reduce the maximum peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity: for each position, how much it contributes to top peaks
        sens = np.zeros(n)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Find top contributing pairs
            top_pairs = np.argsort(contrib)[::-1][:3]
            for pi in top_pairs:
                src = idx[pi]
                partner = ki - src
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Determine which to reduce based on sensitivity
                imp_src = sens[src]
                imp_partner = sens[partner] if 0 <= partner < n else 0

                if imp_src >= imp_partner:
                    reduce_idx = src
                else:
                    reduce_idx = partner

                if a[reduce_idx] < 1e-12:
                    continue

                # Find best destination (low sensitivity, not in any top peak)
                in_peaks = np.zeros(n, dtype=bool)
                for kk in top_k[:5]:
                    kk_i = int(kk)
                    s2 = max(0, kk_i - n + 1)
                    e2 = min(n - 1, kk_i)
                    in_peaks[np.arange(s2, e2 + 1)] = True

                candidates = np.where(~in_peaks)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                cand_sens = sens[candidates]
                dst = candidates[np.argmin(cand_sens)]

                if dst == reduce_idx:
                    dst = (dst + 1) % n

                # Try multiple fractions
                found = False
                for frac in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.12]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

                if found:
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 20 == 0:
                n_pert = min(3, n)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakRed ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakRed final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def subgradient_mirror_descent(a0, time_limit, M_peaks=5, verbose=False, rng_seed=42):
    """Mirror descent using exact subgradient of the max autocorrelation function."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    t = 0
    stall = 0
    eta_base = 2.0

    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Compute exact subgradient: average of 2*B_k*a for top M peaks
        top_idx = np.argsort(ac)[::-1][:min(M_peaks, len(ac))]
        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner = ki - idx
            g[idx] += 2.0 * a[partner]
        g /= len(top_idx)

        # Mirror descent step
        eta_t = eta_base / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta_t * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Restart strategy
        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            eta_base = max(1.0, eta_base * 0.85)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def fine_local_search(a0, time_limit, verbose=False, rng_seed=0):
    """
    Very fine-grained local search: try moving tiny amounts of mass between
    pairs of positions, accepting only strict improvements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Precompute top peaks
    ac = fast_autocorr(a)
    top_k = np.argsort(ac)[::-1][:15]

    # Build sensitivity map
    sens = np.zeros(n)
    for k in top_k:
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n - 1, ki)
        idx = np.arange(s, e + 1)
        sens[idx] += a[ki - idx]

    # Sort positions by sensitivity (high-sens first for source, low-sens for dest)
    sens_order_desc = np.argsort(sens)[::-1]
    sens_order_asc = np.argsort(sens)

    stall = 0
    max_stall = 5000
    n_tries_per_round = min(200, 2 * n)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        round_improved = False
        n_checked = 0

        for _ in range(n_tries_per_round):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source from high-sensitivity, dest from low-sensitivity
            src = sens_order_desc[int(rng.integers(min(50, n)))]
            dst = sens_order_asc[int(rng.integers(min(50, n)))]

            if src == dst or a[src] < 1e-12:
                continue

            n_checked += 1
            for frac in [0.005, 0.01, 0.002, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    round_improved = True
                    # Update sensitivity occasionally
                    if n_checked % 50 == 0:
                        ac = fast_autocorr(a)
                        top_k = np.argsort(ac)[::-1][:15]
                        sens = np.zeros(n)
                        for k in top_k:
                            ki = int(k)
                            s2 = max(0, ki - n + 1)
                            e2 = min(n - 1, ki)
                            idx = np.arange(s2, e2 + 1)
                            sens[idx] += a[ki - idx]
                        sens_order_desc = np.argsort(sens)[::-1]
                        sens_order_asc = np.argsort(sens)
                    break

        if round_improved:
            stall = 0
        else:
            stall += 1

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FineLS ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    - LP-FW with adaptive K and peak tightening
    - Targeted peak reduction local search
    - Subgradient mirror descent
    - Fine-grained local search
    - Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_u_shape(3200, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW on parent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW (n={n0}, remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 450.0)
    if t1 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True,
                                   K_start=100, K_max=min(3000, 2 * n0 - 1),
                                   use_tighten=True, tighten_factor=0.998)
        try_update(a_out, "lp_p1")

    # === Phase 2: Peak reduction local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Peak reduction (n={len(best_seq)}, remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_search(a_in, t2, verbose=True, rng_seed=42, n_top=25)
        try_update(a_out, "peak_p2")

    # === Phase 3: Fine local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Fine local search (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_local_search(a_in, t3, verbose=False, rng_seed=77)
        try_update(a_out, "fine_p3")

    # === Phase 4: LP-FW round 2 with larger K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}, remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   K_start=min(500, 2 * n_best - 1),
                                   K_max=min(4000, 2 * n_best - 1),
                                   use_tighten=True, tighten_factor=0.997)
        try_update(a_out, "lp_p4")

    # === Phase 5: Subgradient mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Subgradient MD (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_mirror_descent(a_in, t5, M_peaks=8,
                                               verbose=False, rng_seed=123)
        try_update(a_out, "md_p5")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [-200, 200, -400, 400, 600, -600, 100, -100]:
        new_n = n_best + delta
        if 2000 <= new_n <= 5500:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = trial_lengths[:4]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.3, 180.0)
        if t6a > 30 and new
