"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed L-BFGS optimization with temperature annealing
2. Multi-strategy initialization (uniform, power ramps, bimodal, parent)
3. Subgradient mirror descent for final polish
4. Coarse-to-fine length sweeping
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv2(a, b):
    """Compute full linear convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def lse(x, T):
    """Log-sum-exp approximation of max."""
    x_max = x.max()
    return x_max + T * np.log(np.sum(np.exp((x - x_max) / T)))

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def compute_lse_gradient(a, T):
    """Compute gradient of 2*n*LSE(T, conv(a)) w.r.t. a."""
    n = len(a)
    conv = fast_conv(a)
    # p = softmax(conv / T)
    conv_scaled = conv / T
    conv_scaled -= conv_scaled.max()
    p = np.exp(conv_scaled)
    p /= p.sum()
    # g[i] = 2 * sum_j p[i+j] * a[j] = 2 * conv(a, p_rev)[2n-2-i]
    conv_result = fast_conv2(a, p[::-1])
    g = 2.0 * conv_result[n-1:2*n-1][::-1]
    return g, conv

def lse_lbfgs_optimize(a_init, time_limit, n_t_levels=10, maxiter=300, verbose=False):
    """Optimize using LSE-smoothed L-BFGS with temperature annealing."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    
    # Initial temperature
    conv0 = fast_conv(a)
    T_start = max(conv0.max() / 5.0, 1e-6)
    
    # Initialize z from a: a ≈ softmax(z) => z ≈ log(a)
    z = np.log(a + 1e-30)
    
    T = T_start
    for level in range(n_t_levels):
        if time.monotonic() - t0 > time_limit:
            break
        
        T_min_iter = maxiter  # iterations for this level
        
        def obj_fn(z):
            a_local = softmax(z)
            conv_local = fast_conv(a_local)
            return 2.0 * n * lse(conv_local, T)
        
        def grad_fn(z):
            a_local = softmax(z)
            g, conv_local = compute_lse_gradient(a_local, T)
            grad_z = a_local * (g - np.dot(g, a_local))
            return grad_z
        
        if verbose:
            print(f"    LSE level {level}: T={T:.6f}")
        
        try:
            result = minimize(
                obj_fn, grad_fn, z,
                method='L-BFGS-B',
                options={'maxiter': T_min_iter, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            z = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")
            break
        
        a_new = softmax(z)
        val = eval_J(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            if verbose:
                print(f"    Level {level}: J={best_val:.6f}")
        
        # Anneal T
        T = T / 2.5
        
        if T < 1e-10:
            break
    
    return best_a, best_val
