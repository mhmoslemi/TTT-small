**Peak-aware coordinate descent:**
- Find the peak lag k*
- The peak value is `ac[k*] = sum_i a[i] * a[k*-i]`
- The "contribution" of pair (i, k*-i) to the peak is `a[i] * a[k*-i]`
- To reduce the peak, we want to reduce the high-contribution pairs
- We can do this by:
  a) Reducing a[i] and giving the mass to j where a[j] * a[k*-j] is small
  b) Or by "spreading" the mass more evenly

The key is to find a good target j for the mass. The ideal j is one where:
- a[k*-j] is small (so giving mass to j doesn't increase the peak)
- a[j] is currently small (so adding mass doesn't create a new peak elsewhere)

def subgradient_method(a0, time_limit, verbose=True, step_decay=1.0, momentum=0.9):
    n = len(a0)
    a = norm_sum1(a0)
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Subgradient: gradient of ac[peak_k] w.r.t. a
        grad = np.zeros(n)
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        for i in range(i_start, i_end):
            grad[i] = a[peak_k - i]
        
        # Use top-K peaks for a smoother gradient
        top_k = min(20, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n, int(k) + 1)
            for i in range(i_start, i_end):
                grad[i] += a[int(k) - i]
        
        # Step size
        eta = step_decay / (np.sqrt(t) + 1.0)
        
        # Update with momentum
        v = momentum * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s

        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'sech':
        x = np.linspace(-3, 3, n)
        return norm_sum1(1.0 / np.cosh(x))
    elif family == 'gaussian':
        x = np.linspace(-3, 3, n)
        return norm_sum1(np.exp(-x ** 2))
    elif family == 'raised_cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(0.5 * (1 + np.cos(2 * np.pi * x)))
    elif family == 'tent':
        x = np.linspace(0, 1, n)
        return norm_sum1(1.0 - 2.0 * np.abs(x - 0.5))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe_enhanced(a0, time_limit, solver, verbose=True,
                             top_K_init=80, top_K_max=500,
                             margin_init=5e-3, margin_min=1e-5,
                             max_iters=200000):
    """
    Enhanced LP Frank-Wolfe with:
    - Line search (try multiple step sizes)
    - Adaptive top-K (start small, grow over time)
    - Geometric margin schedule
    - Periodic momentum restart
    """
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    margin = margin_init
    top_K = top_K_init
    stall_count = 0

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < 1e-16:
            break

        # Adaptive top-K: grow slowly
        top_K = min(top_K_init + (t // 100) * 5, top_K_max)

        # Select active constraints
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:min(top_K, len(margins))]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        if K == 0:
            continue

        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row in range(K):
            k_int = int(active_idx[row])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx_arr = np.arange(i_start, i_end)
            T_sel[row, idx_arr] = y[k_int - idx_arr]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y_nest = y + beta_t * (y - x)
        y_nest = np.maximum(y_nest, 0.0)
        sy = y_nest.sum()
        if sy < 1e-15:
            y_nest = y.copy()
        else:
            y_nest = y_nest / sy

        # Line search: try multiple step sizes
        base_alpha = 2.0 / (t + 2.0)
        alphas = [base_alpha * 0.25, base_alpha * 0.5, base_alpha,
                  base_alpha * 1.5, base_alpha * 2.0, base_alpha * 3.0]
        best_alpha = base_alpha
        best_trial_m = M_curr

        for alpha in alphas:
            candidate = (1.0 - alpha) * y_nest + alpha * g_dir
            candidate = np.maximum(candidate, 0.0)
            s_c = candidate.sum()
            if s_c < 1e-15:
                continue
            candidate = candidate / s_c
            m_trial = fast_autocorr(candidate).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        # Compute new x
        x_new = (1.0 - best_alpha) * y_nest + best_alpha * g_dir
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_nest.copy()
        else:
            x_new = x_new / sx

        y = y_nest
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()
            stall_count = 0
            # Tighten margin and grow K on improvement
            margin = max(margin * 0.92, margin_min)
            top_K = min(top_K + 2, top_K_max)
        else:
            stall_count += 1

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
            stall_count = 0

        # Periodic restart: reset momentum from best with perturbation
        if stall_count > 2000:
            stall_count = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            margin = max(margin * 1.5, margin_init)

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW+ iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, "
                  f"margin={margin:.2e}, top_K={top_K}, stall={stall_count}")

    return best_a, best_m


def coordinate_descent_enhanced(a0, time_limit, verbose=True, rng_seed=42):
    """
    Peak-aware coordinate descent with gradient-guided target selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Compute contributions to peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 10
            continue

        # Get top peak lags for target scoring
        n_peaks = min(15, len(ac))
        top_peak_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute badness for each target position
        badness = np.zeros(n)
        for pk in top_peak_lags:
            pk_int = int(pk)
            j_start = max(0, pk_int - n + 1)
            j_end = min(n, pk_int + 1)
            j_arr = np.arange(j_start, j_end)
            badness[j_arr] += a[pk_int - j_arr] ** 2

        # Small regularization to avoid concentrating
        badness += a ** 2 * 0.05

        # Try reducing top contributing sources
        n_top = min(12, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            badness[src] = np.inf
            # Get candidate targets (lowest badness)
            n_targets = min(25, n)
            targets = np.argsort(badness)[:n_targets]

            best_new_m = M_curr
            best_move = None

            fracs = [0.003, 0.008, 0.015, 0.03, 0.05, 0.08]
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for dst_idx in targets[:15]:
                    dst = int(dst_idx)
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m - 1e-15:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random exploration
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDesc final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def spectral_perturb(a, strength=0.01, rng=None):
    """Perturb the phase of the DFT to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

    return [float(x) for x in best_seq]
