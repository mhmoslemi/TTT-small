def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(y, ac, n_lags=None):
    """Compute subgradient of max_k b_k at point y.
    Uses the lags where b_k is near maximum.
    grad b_k(a)[j] = 2*a[k-j] if 0<=k-j<n else 0
    """
    n = len(y)
    M = ac.max()
    if n_lags is None:
        # Use all lags within 99.5% of max
        active = np.where(ac >= M * 0.995)[0]
    else:
        idx = np.argpartition(-ac, min(n_lags, len(ac)-1))[:n_lags]
        active = np.sort(idx)
    
    g = np.zeros(n)
    count = 0
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
            count += 1
    
    if count > 0:
        g /= count
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=30, c0=3.0, verbose=False):
    """Nesterov mirror descent with subgradient from near-max lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 5000
    rng = np.random.default_rng(42 + n)
    c = c0
    
    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_conv(y)
        M = ac.max()
        
        # Adaptive: use more lags when max is spread out

        g = compute_subgradient(y, ac, n_lags=k_use)
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0

            strategy = int(rng.integers(4))
            if strategy == 0:

                idx = rng.choice(n, size=min(n//3, 800), replace=False)

            elif strategy == 2:

                idx1 = rng.choice(n, size=n//4, replace=False)

                    amt = x_r[i1] * 0.3
                    x_r[i1] -= amt
                    x_r[i2] += amt

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.7)
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, c={c:.3f}")

def lp_remez_step(a, active_lags, M_target, r, solver=None):
    """Single LP Remez step with specified active lags and trust region radius r."""
    n = len(a)
    b = fast_conv(a)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            # Linearized constraint: b[k] + 2 * sum(coeffs * (a_var - a))[idx] <= M_var
            # = b[k] + 2 * sum(coeffs * a_var[idx]) - 2 * sum(coeffs * a[idx]) <= M_var
            # = 2 * sum(coeffs * a_var[idx]) - 2 * sum(coeffs * a[idx]) <= M_var - b[k]
            # Note: 2 * sum(coeffs * a[idx]) = 2 * sum(a[k-idx] * a[idx]) = 2 * b[k] (approximately)
            # Actually: coeffs = 2*a[k-idx], so sum(coeffs*a[idx]) = 2*sum(a[k-idx]*a[idx]) = 2*b[k]
            # So: 2*sum(coeffs*a_var[idx]) - 4*b[k] <= M_var - b[k]
            # I.e.: 2*sum(coeffs*a_var[idx]) <= M_var + 3*b[k]
            # Hmm, let me redo this more carefully.
            # b_k(a') ≈ b_k(a) + grad_k . (a' - a) = b_k(a) + sum_j 2*a[k-j]*(a'_j - a_j)
            # We want b_k(a') <= M, so: b_k(a) + sum_j 2*a[k-j]*(a'_j - a_j) <= M
            # i.e.: sum_j 2*a[k-j]*a'_j <= M - b_k(a) + sum_j 2*a[k-j]*a_j
            # Note sum_j 2*a[k-j]*a_j = 2*b_k(a) (since b_k = sum_j a_j * a[k-j])
            # So: sum_j 2*a[k-j]*a'_j <= M + b_k(a)
            lhs = cp.sum(cp.multiply(2.0 * a[k_int - idx], a_var[idx]))
            constraints.append(lhs <= M_var + b[k_int])
    
    # Trust region: L1 distance bounded
    # sum |a_var - a| <= r (approximated as: -r <= sum(a_var - a)... no)
    # Use: a_var >= a*(1-r) and a_var <= a*(1+r) (box trust region)
    # Or better: use L2 trust region
    # For efficiency, use a relaxed box: a_var <= a*(1+r) and a_var >= 0 (already nonneg)
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), M_target
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), M_target
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), M_target
    a_new = a_new / s
    return a_new, float(M_var.value)

            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[k_int - idx]
            count += 1
    if count > 0:
        g /= count
    return g

def solve_1d_max_linear(c, d, t_lo, t_hi, n_iter=50):
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    a = float(t_lo)
    b = float(t_hi)
    if b - a < 1e-14:
        return (a + b) / 2.0
    c1 = b - inv_phi * (b - a)
    c2 = a + inv_phi * (b - a)
    f1 = float(np.max(c + c1 * d))
    f2 = float(np.max(c + c2 * d))
    for _ in range(n_iter):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - inv_phi * (b - a)
            f1 = float(np.max(c + c1 * d))
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + inv_phi * (b - a)
            f2 = float(np.max(c + c2 * d))
    return (a + b) / 2.0

def lp_remez_step(a, active_lags, r, solver=None):
    n = len(a)
    b = fast_conv(a)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            lhs = cp.sum(cp.multiply(2.0 * a[k_int - idx], a_var[idx]))
            constraints.append(lhs <= M_var + b[k_int])
    constraints.append(a_var <= a * (1.0 + r))
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), float(b.max())
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float(b.max())
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), float(b.max())
    a_new = a_new / s
    return a_new, float(M_var.value)

def remez_lp_method(a0, time_limit, solver=None, r_init=0.2, verbose=False, tag=""):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_conv(a).max())
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_conv(a)
        M = float(b.max())
        threshold = M * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) > 80:
            top_idx = np.argsort(b)[::-1][:80]
            active = np.sort(top_idx)
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        a_cand, M_cand = lp_remez_step(a, active, r, solver=solver)
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = float(fast_conv(a_trial).max())
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.5, r * 1.03)
                break
        if not accepted:
            no_improve += 1
            r = max(0.002, r * 0.75)
            if no_improve > 12:
                rng = np.random.default_rng(iteration * 13 + 7)
                strategy = int(rng.integers(3))
                if strategy == 0:
                    idx = rng.choice(n, size=min(n // 4, 600), replace=False)
                    a[idx] *= rng.uniform(0.4, 2.0, len(idx))
                elif strategy == 1:
                    a = a * (1.0 + rng.normal(0, 0.05, n))
                else:
                    idx1 = rng.choice(n, size=n // 5, replace=False)
                    idx2 = rng.choice(n, size=n // 5, replace=False)
                    for i1, i2 in zip(idx1, idx2):
                        amt = a[i1] * 0.4
                        a[i1] -= amt
                        a[i2] += amt
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                no_improve = 0
                r = 0.15
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez{tag} iter={iteration} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, act={len(active)}, stall={no_improve}")
    return best_a, best_m

def nesterov_mirror_descent(a0, time_limit, top_k=30, c0=3.0, verbose=False, tag=""):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_conv(a).max())
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 8000
    rng = np.random.default_rng(42 + n)
    c = c0
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        ac = fast_conv(y)
        M = ac.max()
        near_frac = 0.999
        active_count = int(np.sum(ac >= M * near_frac))
        k_use = min(max(top_k, active_count + 5), min(150, len(ac)))
        g = compute_subgradient(y, ac, n_lags=k_use)
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        x_prev = x
        x = a_new
        M_new = float(fast_conv(x).max())
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(2000, int(max_stall * 0.6))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.04, n)
                x_r = best_a * (1.0 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 3, 800), replace=False)
                x_r[idx] *= rng.uniform(0.3, 2.5, len(idx))
            elif strategy == 2:
                x_r = best_a.copy()
                idx1 = rng.choice(n, size=n // 4, replace=False)
                idx2 = rng.choice(n, size=n // 4, replace=False)
                for i1, i2 in zip(idx1, idx2):
                    amt = x_r[i1] * 0.5
                    x_r[i1] -= amt
                    x_r[i2] += amt
            else:
                x_r = best_a * (1.0 + rng.normal(0, 0.015, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.6)
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD{tag} t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, c={c:.3f}")
    return best_a, best_m

def coordinate_descent_2d(a0, time_limit, n_pairs=150, verbose=False, tag=""):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = float(fast_conv(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    sweep = 0
    stall = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        ac = fast_conv(a)
        g = compute_subgradient(a, ac, n_lags=60, frac=0.999)
        g_mean = g.mean()
        high_mask = g > g_mean
        low_mask = g < g_mean
        high_idx = np.where(high_mask)[0]
        low_idx = np.where(low_mask)[0]
        if len(high_idx) < 10:
            high_idx = np.argpartition(g, -max(10, n // 5))[::-1][:n // 5]
            high_idx = np.sort(high_idx)
        if len(low_idx) < 10:
            low_idx = np.argpartition(g, max(9, n // 5))[:n // 5]
            low_idx = np.sort(low_idx)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(high_idx[rng.integers(len(high_idx))])
            j = int(low_idx[rng.integers(len(low_idx))])
            if i == j or a[i] < 1e-12:
                continue
            d = np.zeros(2 * n - 1)
            k1s = i
            k1e = min(i + n, 2 * n - 1)
            d[k1s:k1e] += a[:k1e - k1s]
            k2s = j
            k2e = min(j + n, 2 * n - 1)
            d[k2s:k2e] -= a[:k2e - k2s]
            c_vec = ac - a[i] * d
            t_lo = 0.0
            t_hi = a[i] + a[j]
            t_opt = solve_1d_max_linear(c_vec, d, t_lo, t_hi)
            a_trial = a.copy()
            a_trial[i] = max(0.0, t_opt)
            a_trial[j] = max(0.0, a[i] + a[j] - t_opt)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = float(fast_conv(a_trial).max())
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 15:
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                a[idx] *= rng.uniform(0.5, 1.5, len(idx))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                stall = 0
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD2D{tag} sweep={sweep} ({elapsed:.1f}s): J={J:.8f}, stall={stall}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'flat':
        a = np.ones(n)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2400
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting with n={n0}, score={best_score:.8f}, target=1.5027")

    # === Phase 1: LP Remez aggressive (large trust region) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Remez aggressive n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t1, solver=solver, r_init=0.25, verbose=True, tag="_P1")
        try_update(a_out, f"Remez_P1_n{n0}")

    # === Phase 2: 2D Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: 2D CD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = coordinate_descent_2d(norm_sum1(best_seq), t2, n_pairs=200, verbose=True, tag="_P2")
        try_update(a_out, f"CD2D_P2_n{len(best_seq)}")

    # === Phase 3: LP Remez fine (small trust region) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP Remez fine n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t3, solver=solver, r_init=0.08, verbose=True, tag="_P3")
        try_update(a_out, f"Remez_P3_n{len(best_seq)}")

    # === Phase 4: NMD exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: NMD exploration n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t4, top_k=40, c0=4.0, verbose=True, tag="_P4")
        try_update(a_out, f"NMD_P4_n{len(best_seq)}")

    # === Phase 5: 2D CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 200.0)
    print(f"\n=== Phase 5: 2D CD polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = coordinate_descent_2d(norm_sum1(best_seq), t5, n_pairs=300, verbose=True, tag="_P5")
        try_update(a_out, f"CD2D_P5_n{len(best_seq)}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        n_alts = [n0 + 200, n0 - 200, n0 + 400]
        n_alts = [x for x in n_alts if 1500 <= x <= 3500]
        t_per = remaining / (len(n_alts) + 1)
        print(f"\n=== Phase 6: Multi-length (budget={remaining:.0f}s, lengths={n_alts}) ===")
        for n_alt in n_alts:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"  Trying n={n_alt}...")
            a_alt = interpolate_sequence(best_seq, n_alt)
            t_sub = min(t_per * 0.7, remaining - 10)
            if t_sub > 20:
                a_out, _ = remez_lp_method(a_alt, t_sub, solver=solver, r_init=0.15, verbose=False, tag=f"_n{n_alt}")
                try_update(a_out, f"Remez_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = coordinate_descent_2d(norm_sum1(best_seq if len(best_seq) == n_alt else a_alt),
                                                  min(remaining * 0.5, 60), n_pairs=100, verbose=False, tag=f"_n{n_alt}")
                try_update(a_out, f"CD2D_n{n_alt}")
        # Final length polish on best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            n_best = len(best_seq)
            a_out, _ = remez_lp_method(norm_sum1(best_seq), min(remaining - 10, 100), solver=solver, r_init=0.05, verbose=False, tag="_final")
            try_update(a_out, f"Remez_final_n{n_best}")

    # === Phase 7: Final NMD + CD push ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = remaining - 10
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final push n={n_best} (budget={t7:.0f}s) ===")
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t7 * 0.5, top_k=20, c0=2.0, verbose=False, tag="_final")
        try_update(a_out, f"NMD_final_{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = coordinate_descent_2d(norm_sum1(best_seq), remaining - 5, n_pairs=500, verbose=False, tag="_last")
            try_update(a_out, f"CD2D_last_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
