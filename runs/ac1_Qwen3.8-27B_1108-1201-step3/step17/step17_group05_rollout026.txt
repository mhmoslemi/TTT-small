def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(v):
    n = len(v)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.maximum(v - mid, 0).sum() > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2, 0.0)


def compute_smooth_grad(a, beta):
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    shifted = ac - M
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / w_sum
    a_rev = a[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A, fft_size)[:total_len]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, M


def proximal_gradient_optimize(a0, time_limit, verbose=True):

    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    total_iters = 100000
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, t / 50000)
        beta = 2.0 + 38.0 * progress
        
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x.copy()
        
        g, M_curr = compute_smooth_grad(y, beta)
        eta = 1.0 / (beta * np.sqrt(n))
        v = y - eta * g

        x = x_new

                       cp.sum(b_var) == 1]

        try:
            prob.solve(solver=solver, verbose=False)
