def fast_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_obj(a):

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        return np.ones_like(z) / len(z)
    return e / s


def lse_obj_grad(z, beta):
    """
    Compute LSE-smoothed objective and gradient w.r.t. z (log-space parameters).
    a = softmax(z), objective = (1/beta)*log(sum_k exp(beta * C(k)))
    """
    n = len(z)
    a = softmax(z)
    c = fast_conv(a)  # length 2n-1
    # LSE
    shifted = beta * (c - c.max())
    max_shifted = 0.0  # since we shifted by c.max()*beta
    exp_vals = np.exp(shifted)
    lse_val = (beta * c.max() + np.log(exp_vals.sum())) / beta
    # Weights w(k) = exp(beta*C(k)) / sum(exp(beta*C(m)))

    # Gradient of LSE w.r.t. a: grad_a[j] = sum_k w(k) * dC(k)/da[j]

    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv_wa = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_pad), fft_size)
    # conv_wa[k] = sum_j a[j] * w[k-j]
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] = 2 * sum_k' w[k'+j] * a[k'] where k' = k-j
    # = 2 * conv(w, a_rev) ... let me just do it directly
    # Actually: grad_a[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # Let m = k-j, so k = m+j: grad_a[j] = 2 * sum_m w[m+j] * a[m]
    # = 2 * (a * w_shifted)[j] where w_shifted is w shifted... 
    # Simpler: grad_a = 2 * convolve(w, a, mode='valid') padded... 
    # Let me use the direct relationship:
    # grad_a[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j] (if m+j < len(w))
    # This is the cross-correlation of a and w.
    # corr[m] = sum_j a[j] * w[j+m]
    # grad_a[j] = 2 * corr[j]
    # corr can be computed as: conv(a, w_flipped) where w_flipped[m] = w[len(w)-1-m]
    
    # Actually the simplest: grad_a[j] = 2 * sum_k w[k]*a[k-j] for valid k-j
    # = 2 * (a * w)[j] in the cross-correlation sense
    # Let's just compute it: for each j, sum over k where 0<=k-j<n
    # This is equivalent to: for each m in [0,n-1] (letting m=k-j), add a[m]*w[m+j] to grad_a[j]
    # So grad_a[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # = 2 * cross_correlation(a, w, lag=j)
    
    # FFT approach: pad both to appropriate size
    # corr[j] = sum_{m=0}^{n-1} a[m] * w[m+j]
    # = sum_m a[m] * w[m+j]
    # If we define a_rev[m'] = a[n-1-m'], then:
    # corr[j] = sum_m a[m] * w[m+j]
    # Let n1 = n (length of a), n2 = len(w) = 2n-1
    # This is the (j-th element of) the cross-correlation
    # cross_correlation(a, w)[j] = sum_m a[m] * w[m+j]
    # = conv(a, w_flipped)[n-1+j] where w_flipped[i] = w[n2-1-i]
    
    w_flipped = w[::-1]
    total_len2 = n + len(w_flipped) - 1
    fft_size2 = 1 << (total_len2 - 1).bit_length()
    a_pad2 = np.zeros(fft_size2)
    a_pad2[:n] = a
    wf_pad2 = np.zeros(fft_size2)
    wf_pad2[:len(w_flipped)] = w_flipped
    conv_full = np.fft.irfft(np.fft.rfft(a_pad2) * np.fft.rfft(wf_pad2), fft_size2)[:total_len2]
    # conv_full[k] = sum_i a[i] * w_flipped[k-i] = sum_i a[i] * w[len(w)-1-(k-i)]
    # = sum_i a[i] * w[len(w)-1-k+i]
    # Let k = n-1+j: conv_full[n-1+j] = sum_i a[i] * w[len(w)-n+j+i]
    # len(w) = 2n-1, so: = sum_i a[i] * w[n-1+j+i]
    # Hmm this is getting complicated. Let me just use a simpler approach.
    
    # Direct approach using the fact that grad_a[j] = 2 * sum_m a[m] * w[m+j]
    # This is just a "sliding dot product" - can be done with FFT
    # Define: for the sum over m of a[m]*w[m+j], this is the cross-correlation at lag j
    # cross_corr(a, w)[j] = sum_m a[m] * conj(w[m+j]) (for real: sum_m a[m]*w[m+j])
    
    # numpy doesn't have a direct cross-correlation, but we can use:
    # cross_corr(a, w) = conv(a, w[::-1])[n-1:] (for the "same" part)
    # Actually: cross_corr[j] = sum_i a[i] * w[i+j]

    # conv(a,w)[k] = sum_i a[i]*w[k-i] for i=0..n-1, k-i=0..len(w)-1
    # xcorr[j] = sum_i a[i]*w[i+j] = sum_i a[i]*w[(i+j)]

    # = sum_i a[i]*w[k] evaluated at k = i+j... this IS conv(a_rev, w) or something.

    # Define u of length n: u[i] = a[i]
    # Define v of length 2n-1: v[i] = w[i]

    # This is the "valid" cross-correlation.
    # g = ifft(fft(u, N) * conj(fft(v, N))) ... no, that gives circular correlation.
    # For linear: pad to length >= n + (2n-1) - 1 = 3n-2
    N_fft = 1 << (3 * n - 1).bit_length()
    U = np.fft.rfft(a, N_fft)
    V = np.fft.rfft(w, N_fft)
    # cross-correlation: ifft(U * conj(V)) gives sum_i a[i] * w[i+j] (circular)
    # For linear, we need the "same" part
    xcorr_circ = np.fft.irfft(U * np.conj(V), N_fft)
    # xcorr_circ[j] = sum_i a[i] * w[(i+j) mod N_fft]
    # For j < n and i < n, (i+j) < 2n < N_fft (for n >= 2), so no wraparound
    # So xcorr_circ[j] = sum_i a[i] * w[i+j] for j = 0, ..., n-1 (valid part)
    grad_a = 2.0 * xcorr_circ[:n]
    
    # Gradient w.r.t. z: df/dz[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return lse_val, grad_z


def lse_optimize(a0, time_limit, beta_start=5.0, beta_end=500.0, verbose=True, seed=42):
    """
    Optimize using LSE-smoothed objective with L-BFGS-B and adaptive beta annealing.
    Works in log-space (z) so a = softmax(z) always satisfies simplex constraints.
    """
    n = len(a0)
    a0 = norm_sum1(a0)
    # Initialize z from a0
    z = np.log(a0 + 1e-30)
    z -= z.mean()  # center for numerical stability
    
    best_a = a0.copy()
    best_score = eval_obj(a0)
    t0 = time.monotonic()
    
    # Adaptive beta schedule: geometric from beta_start to beta_end
    n_stages = 8
    betas = np.geomspace(beta_start, beta_end, n_stages)
    
    time_per_stage = time_limit / n_stages
    
    for stage_idx, beta in enumerate(betas):
        t_stage_start = time.monotonic()
        t_avail = max(5.0, time_limit - (t_stage_start - t0))
        
        # Define objective and gradient for scipy
        def func_and_grad(z_val):
            val, grad = lse_obj_grad(z_val, beta)
            return val, grad
        
        # Run L-BFGS-B
        try:
            result = scipy_opt.minimize(
                func_and_grad,
                z,
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': 500, 'maxfun': 2000, 'maxls': 50,
                         'ftol': 1e-12, 'gtol': 1e-8, 'disp': False}
            )
            z_new = result.x
            a_new = softmax(z_new)
            
            # Evaluate actual (non-smoothed) objective
            score = eval_obj(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            z = z_new
        except Exception as e:
            if verbose:
                print(f"    LSE stage {stage_idx} (beta={beta:.1f}) failed: {e}")
        
        # Also try with current z but evaluate actual objective
        a_curr = softmax(z)
        score_curr = eval_obj(a_curr)
        if score_curr < best_score:
            best_score = score_curr
            best_a = a_curr.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {stage_idx}/{n_stages} (beta={beta:.1f}, {elapsed:.0f}s): "
                  f"obj={best_score:.8f}")
        
        # Check time
        if time.monotonic() - t0 >= time_limit - 5:
            break
    
    return best_a, best_score


def cutting_plane_fw(a0, time_limit, solver, verbose=True, max_iter=200):
    """
    LP Frank-Wolfe with cutting planes (only active/violated constraints) and away steps.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    
    # Constraint tracking: we maintain a list of lag indices
    active_lags = []
    
    for iteration in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M_curr = ac.max()
        
        # Select constraints: top lags by value + any previously active
        n_sel = min(200, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:n_sel]
        # Union with previous active lags
        all_lags = list(set(list(top_idx[:50].astype(int)) + [int(x) for x in active_lags]))
        all_lags = all_lags[:400]  # cap
        
        # Build constraint matrix T: T[k, j] = a[k-j] if 0 <= k-j < n
        K = len(all_lags)
        T = np.zeros((K, n))
        for row_idx, k in enumerate(all_lags):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            js = np.arange(j_start, j_end + 1)
            T[row_idx, js] = a[k_int - js]
        
        # Solve LP: maximize sum(b) s.t. T@b <= M_curr, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            # Infeasible - all lags at capacity, try to find violated ones
            violated = np.where(ac > M_curr * (1 + 1e-6))[0]
            if len(violated) == 0:
                break
            active_lags = [int(v) for v in violated[:50]]
            continue
        b_val = b_val / b_sum
        
        # Standard FW step
        gamma = 2.0 / (iteration + 2.0)
        a_new = (1.0 - gamma) * a + gamma * b_val
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        # Away step: find the vertex with most negative "gradient" dot product
        # Approximate: find the coordinate where b_val is lowest relative to a
        # (simplified away step)
        if iteration > 5:
            # Simple away step: reduce the smallest component of a
            min_idx = np.argmin(a)
            if a[min_idx] > 1e-8:
                # Try moving away from e_min_idx
                a_away = a - 0.1 * a[min_idx] * np.eye(n)[min_idx]
                a_away = a - 0.05 * a[min_idx]
                a_away[min_idx] = 0
                a_away = np.maximum(a_away, 0)
                a_away = a_away / a_away.sum()
                # Blend
                a_new = 0.8 * a_new + 0.2 * a_away
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
        
        score = eval_obj(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.3, 0.1]:
                a_try = (1.0 - gamma * frac) * a + gamma * frac * b_val
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    score_try = eval_obj(a_try)
                    if score_try < best_score:
                        best_score = score_try
                        best_a = a_try.copy()
                        a = a_try
                        break
        
        # Update active lags: add newly violated
        ac_new = fast_conv(a)
        violated = np.where(ac_new > ac_new.max() * 0.95)[0]
        active_lags = [int(v) for v in violated[:50]]
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CP-FW iter={iteration} ({elapsed:.0f}s): obj={best_score:.8f}, K={K}")
    
    return best_a, best_score


def mirror_descent_quick(a0, time_limit, verbose=True, seed=42):
    """
    Quick mirror descent with FFT subgradient for initial progress.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    beta = 6.0
    stall = 0
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        bt = (it - 1.0) / (it + 2.0)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Subgradient using all lags (soft max)
        ac = fast_conv(y)
        M_curr = ac.max()
        shifted = ac - M_curr
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / ws
        w[w < 1e-10] = 0
        
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        a_rev = np.zeros(fft_size)
        a_rev[:n] = y[::-1]
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w
        conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Mirror descent step
        eta = 5.0 / (np.sqrt(it) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x.copy()
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
        
        # Adaptive beta
        beta = max(3.0, 6.0 - 0.0005 * it)
    
    score = eval_obj(best_a)
    return best_a, score


def coordinate_polish(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent: try moving mass from high-contribution positions to low ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    improvements = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_k = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:top_k]
        
        found = False
        for k in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find best destination (low sensitivity)
            sens = np.zeros(n)
            for k2 in top_lags[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]
            
            candidates = np.where(sens < np.median(sens))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Sort by distance from src (prefer far)
            dist = np.abs(candidates.astype(float) - src)
            order = np.argsort(-dist)
            best_dst = candidates[order[0]]
            
            if best_dst == src:
                continue
            
            for frac in [0.03, 0.01, 0.05, 0.02, 0.005]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[src] -= amt
                a_try[best_dst] += amt
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc = eval_obj(a_try)
                if sc < best_score - 1e-12:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    found = True
                    stall = 0
                    improvements += 1
                    break
            
            if found:
                break
        
        if not found:
            stall += 1
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    a[i] -= a[i] * 0.02
                    a[j] += a[i] * 0.02
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    sc = eval_obj(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord done ({elapsed:.0f}s): obj={best_score:.8f}, imps={improvements}")
    
    return best_a, best_score


def make_smooth_init(n, seed=0, symmetrize=True):
    """Create smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    freqs = [1, 2, 3, 4, 5, 7, 11]
    for f in freqs:
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    if symmetrize:
        a = 0.5 * (a + a[::-1])
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining LSE optimization, cutting-plane LP-FW,
    mirror descent, and coordinate polishing across multiple sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        best_seq = make_smooth_init(n_base, seed=1)
        best_score = eval_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.8f}")
    print(f"Time budget: {time_budget:.0f}s")
    
    # === Phase 1: LSE optimization on parent (main driver) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LSE Optimization (t={t1:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, sc_out = lse_optimize(a_in, t1, beta_start=5.0, beta_end=500.0,
                                      verbose=True, seed=42)
        try_update(a_out, "lse_parent")
    
    # === Phase 2: Cutting-plane LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        print(f"\n=== Phase 2: CP-FW (t={t2:.0f}s, n={len(best_seq)}) ===")
        a_in = norm_sum1(best_seq)
        a_out, sc_out = cutting_plane_fw(a_in, t2, solver, verbose=True, max_iter=100)
        try_update(a_out, "cpfw")
    
    # === Phase 3: LSE polish with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LSE Polish (t={t3:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, sc_out = lse_optimize(a_in, t3, beta_start=50.0, beta_end=2000.0,
                                      verbose=True, seed=123)
        try_update(a_out, "lse_polish")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Coordinate Descent (t={t4:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, sc_out = coordinate_polish(a_in, t4, verbose=True, seed=42)
        try_update(a_out, "coord")
    
    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else 3200
    candidate_lengths = [n_best - 200, n_best + 200, n_best + 500, n_best - 500]
    candidate_lengths = sorted(set([l for l in candidate_lengths if 1500 <= l <= 6000]))
    
    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: Length n={new_n} (remaining={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t5a = min(remaining * 0.30, 120.0)
        if t5a > 30:
            a_out, sc_out = lse_optimize(a_long, t5a, beta_start=10.0, beta_end=300.0,
                                          verbose=False, seed=777)
            try_update(a_out, f"lse_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4500:
            t5b = min(remaining * 0.25, 100.0)
            if t5b > 20:
                a_out, sc_out = cutting_plane_fw(a_long, t5b, solver, verbose=False, max_iter=50)
                try_update(a_out, f"cpfw_n{new_n}")
    
    # === Phase 6: Second LSE round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 6: LSE Round 2 (t={remaining:.0f}s) ===")
        t6 = min(remaining * 0.40, 250.0)
        if t6 > 40:
            a_in = norm_sum1(best_seq)
            a_out, sc_out = lse_optimize(a_in, t6, beta_start=30.0, beta_end=1000.0,
                                          verbose=True, seed=555)
            try_update(a_out, "lse_r2")
    
    # === Phase 7: Final CP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else 3200
    if remaining > 50 and n_best <= 5000:
        print(f"\n=== Phase 7: Final CP-FW (t={remaining:.0f}s) ===")
        t7 = min(remaining * 0.50, 200.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, sc_out = cutting_plane_fw(a_in, t7, solver, verbose=True, max_iter=80)
            try_update(a_out, "cpfw_final")
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final Coord (t={remaining:.0f}s) ===")
        t8 = min(remaining * 0.7, 120.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, sc_out = coordinate_polish(a_in, t8, verbose=True, seed=999)
            try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_smooth_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")
    
    return [float(x) for x in best_seq]
