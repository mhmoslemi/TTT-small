        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1 vector."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def max_gradient(a, k):
    """Compute gradient of conv(a,a)_k w.r.t. a.
    Result: grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # Valid i: 0 <= i < n and 0 <= k-i < n
    # => max(0, k-n+1) <= i <= min(n-1, k)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def max_subgradient(a, ac, top_k=10):
    """Compute subgradient of max_k conv(a,a)_k.
    Uses average of gradients of top-k peaks for robustness.
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Weight by how close each peak is to the max
    max_val = top_vals[0]
    if max_val < 1e-20:
        return np.zeros(n)
    # Exponential weights: peaks close to max get more weight
    w = np.exp(-10.0 * (max_val - top_vals) / max_val)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        grad += wi * max_gradient(a, k_int)
    return grad


def mirror_descent_optimize(a0, time_limit, eta0=1.0, nesterov=True,
                            top_k=10, verbose=True, restart_every=2000):
    """Mirror descent with exact max-subgradient and backtracking line search."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    ac = fast_autocorr(x)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if nesterov:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        ac_y = fast_autocorr(y)
        g = max_subgradient(y, ac_y, top_k=top_k)

        # Backtracking line search
        eta = eta0 / (1.0 + 0.5 * np.sqrt(t))
        base_m = ac_y.max()
        accepted = False

        for _ in range(8):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            eta *= 0.5

        if not accepted:
            # Take a small step anyway to make progress
            eta = eta * 0.1
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        # Track best
        m_new = fast_autocorr(x).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_fw_optimize(a0, time_limit, solver, top_K=300, margin_init=2e-3,
                   use_line_search=True, verbose=True, nesterov=True,
                   decay_margin=True):
    """LP Frank-Wolfe with adaptive active set and line search."""
    n = len(a0)
    y = norm_sum1(a0)
    x_mom = y.copy() if nesterov else None
    best_a = y.copy()
    ac = fast_autocorr(y)
    best_m = ac.max()
    t0 = time.monotonic()
    margin = margin_init

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Adaptive active set
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = int(active_mask.sum())

        if n_active > top_K:
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:min(top_K, 200)]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row in range(K):
            k_int = int(active_idx[row])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                T_sel[row, i_start:i_end] = y[k_int - np.arange(i_start, i_end)]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            margin = max(margin * 1.5, 1e-2)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            margin = max(margin * 1.5, 1e-2)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        if use_line_search:
            default_alpha = 2.0 / (t + 2.0)
            candidates = [default_alpha, min(1.0, 2.0 * default_alpha),
                          max(0.01, default_alpha * 0.5), 0.1, 0.05, 0.02, 0.01]
            best_alpha = default_alpha
            best_obj = M_curr
            for alpha in candidates:
                if alpha <= 0 or alpha > 1:
                    continue
                y_trial = (1.0 - alpha) * y + alpha * g_dir
                y_trial = np.maximum(y_trial, 0.0)
                sy = y_trial.sum()
                if sy < 1e-15:
                    continue
                y_trial = y_trial / sy
                m_trial = fast_autocorr(y_trial).max()
                if m_trial < best_obj:
                    best_obj = m_trial
                    best_alpha = alpha
            alpha = best_alpha
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov
        if nesterov:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
            y = y_new
            x_mom = x_new
            # Check both
            m_y = fast_autocorr(y).max()
            if m_y < best_m:
                best_m = m_y
                best_a = y.copy()
                if decay_margin:
                    margin = max(margin * 0.97, 1e-5)
            m_x = fast_autocorr(x_mom).max()
            if m_x < best_m:
                best_m = m_x
                best_a = x_mom.copy()
                if decay_margin:
                    margin = max(margin * 0.97, 1e-5)
        else:
            y = y_new
            m_y = fast_autocorr(y).max()
            if m_y < best_m:
                best_m = m_y
                best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin:.1e}")

    return best_a, best_m


def coordinate_descent_optimize(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent using exact gradient structure.
    
    Key insight: to reduce peak k, move mass from i (where a[k-i] is large)
    to j (where a[k-j] is small for all active peaks).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top peaks
        n_peaks = min(20, len(ac))
        top_peak_idx = np.argsort(ac)[::-1][:n_peaks]

        # For each top peak, find the best (src, dst) move
        best_improvement = 0
        best_move = None

        for pk in top_peak_idx[:5]:
            pk = int(pk)
            # Compute "sensitivity" of peak pk to each position
            # grad[i] = 2*a[pk-i] if 0<=pk-i<n
            i_start = max(0, pk - n + 1)
            i_end = min(n, pk + 1)
            sens = np.zeros(n)
            if i_start < i_end:
                sens[i_start:i_end] = a[pk - np.arange(i_start, i_end)]

            # Source: position with highest a[i]*sens[i] (largest contribution)
            contrib = a * sens
            # Destination: position with low sensitivity across all top peaks
            # Compute combined "cost" of placing mass at each position
            combined_cost = np.zeros(n)
            for pk2 in top_peak_idx[:8]:
                pk2 = int(pk2)
                i_s2 = max(0, pk2 - n + 1)
                i_e2 = min(n, pk2 + 1)
                if i_s2 < i_e2:
                    combined_cost[i_s2:i_e2] += a[pk2 - np.arange(i_s2, i_e2)]

            # Try top sources
            top_src = np.argsort(contrib)[::-1][:10]
            for si in top_src:
                si = int(si)
                if a[si] < 1e-12:
                    continue
                # Find best destination (low combined cost, not same as source)
                cost_masked = combined_cost.copy()
                cost_masked[si] = np.inf
                # Also prefer positions with low value (to spread mass)
                top_dst = np.argsort(cost_masked)[:20]
                for di in top_dst[:5]:
                    di = int(di)
                    if di == si:
                        continue
                    # Try a small transfer
                    frac = min(0.05, a[si] / max(a.sum(), 1e-15) * 0.5)
                    amt = a[si] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[si] -= amt
                    a_trial[di] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    improvement = M_curr - m_trial
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (si, di, amt)

        if best_move is not None and best_improvement > 1e-15:
            si, di, amt = best_move
            a[si] -= amt
            a[di] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDesc final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def multi_peak_lp(a0, time_limit, solver, n_peaks=5, verbose=True):
    """LP that targets multiple peaks simultaneously with tighter formulation."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac = fast_autocorr(y)
    best_m = ac.max()
    t0 = time.monotonic()

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Use top n_peaks constraints
        top_idx = np.argsort(ac)[::-1][:n_peaks]

        K = len(top_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(top_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                T_sel[row, i_start:i_end] = y[k_int - np.arange(i_start, i_end)]

        # LP: max sum(b) s.t. T_sel @ b <= M_curr * (1 - 1/t), b >= 0
        # Tighter bound forces more reduction
        bound = M_curr * (1.0 - 0.5 / (t + 1.0))
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= bound]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        candidates = [default_alpha, 0.1, 0.05, 0.02, 0.01, 0.005]
        best_alpha = default_alpha
        best_obj = M_curr
        for alpha in candidates:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            m_trial = fast_autocorr(y_trial).max()
            if m_trial < best_obj:
                best_obj = m_trial
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = y.copy()
        else:
            y = y / sy

        m_new = fast_autocorr(y).max()
        if m_new < best_m:
            best_m = m_new
            best_a = y.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiPeakLP t={t} ({elapsed:.0f}s): J={J:.6f}, alpha={best_alpha:.4f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    """Generate a parametric initial sequence."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'raised_cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(0.5 * (1 + np.cos(2 * np.pi * x)))
    elif family == 'hann':
        x = np.linspace(0, 1, n)
        return norm_sum1(0.5 * (1 + np.cos(2 * np.pi * (x - 0.5))))
    elif family == 'flatop':
        x = np.linspace(0, 1, n)
        a = 0.128
        return norm_sum1(a + (1-a) * np.cos(2*np.pi*x/(1+2*a)))
    elif family == 'gaussian':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.exp(-8 * (x - 0.5) ** 2))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def freq_perturb(a, strength=0.01, rng=None):
    """Perturb sequence in frequency domain for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update the best sequence if a is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function. Returns the best sequence found within budget."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Mirror descent with exact subgradient ===
    print(f"\n=== Phase 1: Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 200.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t1, eta0=2.0, nesterov=True,
                                           top_k=10, verbose=True, restart_every=3000)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "md1")

    # === Phase 2: LP-Frank-Wolfe (main) ===
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 400.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t2, solver, top_K=300, margin_init=3e-3,
                                  use_line_search=True, verbose=True, nesterov=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp1")

    # === Phase 3: Coordinate descent refinement ===
    print(f"\n=== Phase 3: Coord Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_optimize(a_in, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd1")

    # === Phase 4: LP-FW with tighter margins ===
    print(f"\n=== Phase 4: LP-FW tight (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t4, solver, top_K=400, margin_init=1e-3,
                                  use_line_search=True, verbose=True, nesterov=True,
                                  decay_margin=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp2")

    # === Phase 5: Multi-peak LP ===
    print(f"\n=== Phase 5: Multi-Peak LP (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_peak_lp(a_in, t5, solver, n_peaks=5, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "mpl")

    # === Phase 6: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Freq Diversify (n={n_best}) ===")
    rng = np.random.default_rng(123)
    t6_per = min(remaining * 0.06, 80.0)
    if t6_per > 25:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 30:
                break
            a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
            a_out, _ = mirror_descent_optimize(a_pert, t6_per, eta0=3.0,
                                               nesterov=True, top_k=10,
                                               verbose=False, restart_every=1500)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 7: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    candidate_lengths = []
    for delta in [200, -200, 400, -400]:
        new_n = n_best + delta
        if 2000 <= new_n <= 5000 and new_n != n_best:
            candidate_lengths.append(new_n)
    candidate_lengths = list(set(sorted(candidate_lengths)))[:2]

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.25, 150.0)
        if t7a > 30:
            a_out, _ = lp_fw_optimize(a_long, t7a, solver, top_K=200,
                                      margin_init=2e-3, use_line_search=True,
                                      verbose=False, nesterov=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"n{new_n}")

    # === Phase 8: Final LP polish on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t8, solver, top_K=500, margin_init=5e-4,
                                  use_line_search=True, verbose=True, nesterov=True,
                                  decay_margin=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Coord Desc (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.30, 150.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_optimize(a_in, t9, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety: ensure we return before budget
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
