        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [max(0.0, float(x)) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_smooth_subgrad(y, beta):
    """Compute smooth subgradient using softmax temperature beta. O(n*K) where K is effective active set size."""
    n = len(y)
    b = fast_conv(y)
    b_max = b.max()
    b_shifted = b - b_max
    # Softmax probabilities
    exp_b = np.exp(beta * b_shifted)
    p = exp_b / exp_b.sum()
    
    # Find effective active set (where p is significant)
    threshold = max(1e-8, 1.0 / (len(b) * 10))
    active = np.where(p > threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * p[k] * y[k_int - idx]
    return g, b_max

def nesterov_smooth(a0, time_limit, beta_start=5.0, beta_end=50.0, c0=3.0, verbose=False):
    """Nesterov accelerated mirror descent with smooth subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Dual variable (log-space)
    w = np.zeros(n)
    w_prev = np.zeros(n)
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(12345 + n)
    c = c0
    
    max_iters = 5000000
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta: increase over time to focus on peaks
        frac = min(1.0, t / (max_iters * 0.3))
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        # Compute y in primal from dual variables
        w_y = w + beta_t * (w - w_prev)
        w_y_shifted = w_y - w_y.max()
        y = np.exp(w_y_shifted)
        y = y / y.sum()
        
        # Compute smooth subgradient
        g, b_max = compute_smooth_subgrad(y, beta)
        
        # Step size
        eta = c / (np.sqrt(t) + 1.0)
        
        # Dual update
        w_new = w - eta * g
        w_prev = w
        w = w_new
        
        # Convert back to primal
        w_shifted = w - w.max()
        a_new = np.exp(w_shifted)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, int(max_stall * 0.65))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 3, 400), replace=False)
                x_r[idx] *= rng.uniform(0.6, 1.4, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.003, n))
            else:
                # Blend with parabolic
                x_par = make_init(n, 'parabolic')
                x_r = 0.7 * best_a + 0.3 * x_par
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            # Reset dual variable
            w = np.log(x_r + 1e-30)
            w = w - w.max()
            w_prev = w.copy()
            c = max(0.3, c * 0.5)
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSMD t={t} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, stall={stall_count}")
    
    return best_a, best_m

def lp_remez_step(a, active_lags, r, solver=None):
    """Single LP Remez step with given active set and trust region."""
    n = len(a)
    b = fast_conv(a)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            # Linearized constraint: b_k(a_new) <= b_k(a) + 2*(a_new-a)^T shift_k(a)
            # = 2*sum(a_new_j * a_{k-j}) <= M_var
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var)
    
    if r < 1.0:
        constraints.append(a_var >= a * (1.0 - r))
        constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), b.max()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), b.max()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), b.max()
    return a_new / s, b.max()

def remez_lp_method(a0, time_limit, solver=None, r_init=0.1, verbose=False, max_active=50):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute active set: all lags within 0.5% of max
        b = fast_conv(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > max_active:
            active = np.argsort(b)[::-1][:max_active]
        
        # LP step
        a_cand, _ = lp_remez_step(a, active, r, solver=solver)
        
        # Line search with acceptance
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-15:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.2, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.002, r * 0.7)
            if no_improve > 30:
                # Perturb to escape
                rng = np.random.default_rng(777 + t)
                idx = rng.choice(n, size=min(n // 4, 300), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = r_init
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, |S|={len(active)}")
    
    return best_a, best_m

def remez_tight_polish(a0, time_limit, solver=None, verbose=False):
    """Very tight Remez polish with tiny trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = 0.02
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Use tighter threshold for active set
        threshold = max_b * 0.9999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            active = np.argsort(b)[::-1][:30]
        
        a_cand, _ = lp_remez_step(a, active, r, solver=solver)
        
        accepted = False
        for t_frac in [1.0, 0.5, 0.25, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                break
        
        if accepted:
            r = min(r, 0.1)
        else:
            r = max(0.001, r * 0.8)
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    TightRemez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.5f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent with random pair transfers."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Try larger moves
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                for frac in [0.1, 0.2, 0.35]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stall = 0
    
    return best_a, best_m

def make_init(n, kind='parabolic'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'parabolic':
        a = 4.0 * x * (1.0 - x)
    elif kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'cubic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 3 * (2.0 * x - 1.0), 0.0)
    else:
        a = 4.0 * x * (1.0 - x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2500

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'parabolic')

    if t2 > 30:

    # === Phase 3: NSMD from parabolic init (15% time) ===
