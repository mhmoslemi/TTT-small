```
def simplex_project(v):
    v=np.asarray(v,dtype=np.float64).ravel()
    v=np.where(np.isfinite(v),v,0.0)
    n=v.size
    if n==0: return v
    mu=np.sort(v)[::-1]
    cssv=np.cumsum(mu)
    cs=np.arange(1,n+1)
    tmp=mu*cs > (cssv-1.0)
    if not np.any(tmp): return np.ones(n)/n
    rho=np.nonzero(tmp)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
    p=np.maximum(v-theta,0.0)
    ss=p.sum()
    if ss<1e-15: return np.ones(n)/n
    return p/ss
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
- peak-weighted mirror descent with Nesterov extrapolation and line search,
- exact/subgrid peak shaving coordinate descent,
- corrected LP Frank-Wolfe (max total mass under linearized convolution caps),
- multi-resolution length interpolation and polish.

All helpers are top-level. No nested functions or lambdas are used.
"""

import numpy as np
import time

try:
    import cvxpy as cp
except Exception:
    cp = None

_global_solver = None


def get_solver():
    """Find a usable cvxpy LP/conic solver, or return None."""
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    if cp is None:
        _global_solver = None
        return None

    candidates = ["PDLP", "GUROBI", "MOSEK", "ECOS", "SCS", "XPRESS",
                  "CBC", "GLOP", "GLPK", "SCIP"]
    for name in candidates:
        if not hasattr(cp, name):
            continue
        solver = getattr(cp, name, name)
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)),
                              [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _global_solver = solver
                print(f"  LP solver: {name}")
                return _global_solver
        except Exception:
            continue

    _global_solver = None
    return None


def fast_autocorr(a):
    """Full linear convolution conv(a, a) via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.asarray(out, dtype=np.float64)


def norm_sum1(a):
    """Clip to nonnegative and normalize so that sum(a) == 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    v = np.where(np.isfinite(v), v, 0.0)
    n = v.size
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho_mask = mu * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(rho_mask):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(rho_mask)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    p = np.maximum(v - theta, 0.0)
    s = float(p.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return p / s


def evaluate_safe(a):
    """Call the external evaluator safely."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        vals = [float(x) for x in arr]
        sc = float(evaluate_sequence(vals))
        if np.isfinite(sc):
            return sc
        return float("inf")
    except Exception:
        return float("inf")


def weighted_subgradient(a, beta, topk, exact_prob, rng):
    """
    Compute a weighted subgradient of max(conv(a,a)).

    If exact_prob triggers, use equal weights on lags very close to the maximum.
    Otherwise use a softmax over the top-k lags.
    """
    n = len(a)
    ac = fast_autocorr(a)
    if ac.size == 0:
        return np.zeros(n, dtype=np.float64), 0.0
    maxv = float(np.max(ac))
    if maxv <= 1e-30:
        return np.zeros(n, dtype=np.float64), maxv

    L = ac.size
    if exact_prob is not None and rng is not None and rng.random() < exact_prob:
        thresh = maxv - 1e-5 * max(maxv, 1e-30)
        idx = np.where(ac >= thresh)[0]
        if idx.size == 0:
            idx = np.argmax(ac, keepdims=True)
        if idx.size > 8:
            idx = np.argsort(ac)[::-1][:8]
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0 / max(1, idx.size)
    else:
        k = int(min(max(1, int(topk)), L))
        if k >= L:
            idx = np.arange(L)
        else:
            idx = np.argsort(ac)[::-1][:k]
        vals = ac[idx]
        shifted = vals - float(np.max(vals))
        wtop = np.exp(beta * shifted)
        s = float(np.sum(wtop))
        if not np.isfinite(s) or s <= 1e-300:
            wtop = np.ones(len(idx), dtype=np.float64) / max(1, len(idx))
        else:
            wtop = wtop / s
        w = np.zeros(L, dtype=np.float64)
        w[idx] = wtop

    w = np.where(np.isfinite(w), w, 0.0)
    sw = float(w.sum())
    if sw < 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[0] = 1.0
    else:
        w = w / sw

    total_len = n + L - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size, dtype=np.float64)
    a_rev[:n] = a[::-1]
    wp = np.zeros(fft_size, dtype=np.float64)
    wp[:L] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(wp), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad, maxv


def mirror_optimize(a0, time_limit, beta=20.0, topk=40, step_init=0.5,
                    step_max=1.2, exact_prob=0.3, seed=0, verbose=False,
                    restart_every=4000):
    """
    Exponentiated-gradient / mirror descent on the simplex with Nesterov
    extrapolation and adaptive line search.
    """
    n = len(a0)
    x = norm_sum1(a0)
    best = x.copy()
    cur_m = float(fast_autocorr(x).max())
    best_m = cur_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x_prev = x.copy()
    step = float(step_init)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it == 1:
            y = x.copy()
        else:
            beta_t = (it - 1.0) / (it + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-12)
            y = norm_sum1(y)

        g, maxv = weighted_subgradient(y, beta, topk, exact_prob, rng)
        if maxv <= 1e-30:
            break

        d = g / maxv
        d = d - np.mean(d)
        dnrm = float(np.max(np.abs(d))) if d.size else 0.0
        if dnrm > 1e-15:
            d = d / dnrm
        else:
            d = np.zeros(n, dtype=np.float64)

        base = step / max(1.0, float(np.sqrt(float(it))))
        best_trial = x.copy()
        best_trial_m = cur_m

        for mult in (1.0, 0.4, 0.16):
            eta = base * mult
            if eta < 1e-12:
                continue
            logy = np.log(y) - eta * d
            logy = logy - np.max(logy)
            nt = np.exp(logy)
            nt = np.maximum(nt, 1e-30)
            nt = norm_sum1(nt)
            m = float(fast_autocorr(nt).max())
            if m < best_trial_m:
                best_trial_m = m
                best_trial = nt

        x_prev = x
        x = best_trial
        cur_m = best_trial_m

        if cur_m < best_m - 1e-12:
            best_m = cur_m
            best = x.copy()
            stall = 0
            if step < step_max:
                step = min(step_max, step * 1.03)
        else:
            stall += 1
            step = max(0.05, step * 0.985)

            if cur_m - best_m > 1e-6 * max(1.0, best_m) and stall > 120:
                x = best.copy()
                x_prev = x.copy()
                cur_m = best_m
                stall = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    mirror it={it} ({elapsed:.0f}s): J={J:.6f} stall={stall}")

        if stall > restart_every:
            noise_scale = 0.02 + 0.03 * min(1.0, stall / (2.0 * restart_every))
            xr = norm_sum1(best + rng.normal(0.0, noise_scale, n) * best)
            x = xr
            x_prev = xr.copy()
            cur_m = float(fast_autocorr(x).max())
            stall = 0
            step = step_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    mirror restart it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best, float(best_m)


def peak_shaving_descent(a0, time_limit, seed=0, active_count=28,
                         source_count=36, verbose=False):
    """
    Coordinate descent specifically structured to shave the current peak
    convolution lag(s). Candidate destinations are scored using a linearized
    increment model over active lags, then accepted only after exact FFT
    evaluation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best = a.copy()
    cur_m = float(fast_autocorr(a).max())
    best_m = cur_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    d_arr = np.arange(n, dtype=np.int64)
    fractions = (0.002, 0.005, 0.01, 0.02, 0.05, 0.008, 0.015, 0.08)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 150:
            if stall % 25 == 0:
                a = best.copy()
            else:
                a = norm_sum1(best + rng.normal(0.0, 0.03, n) * best)
            cur_m = float(fast_autocorr(a).max())
            stall = 0

        c = fast_autocorr(a)
        M = float(c.max())
        cur_m = M
        L = 2 * n - 1
        acnt = int(min(max(1, active_count), L))
        active = np.argsort(c)[::-1][:acnt]

        score = np.zeros(n, dtype=np.float64)
        for lagv in active:
            lag = int(lagv)
            js = max(0, lag - n + 1)
            je = min(n - 1, lag)
            if js <= je:
                idx = np.arange(js, je + 1)
                score[idx] += a[idx] * a[lag - idx]

        src_count = int(min(max(1, source_count), n))
        top_s = np.argsort(score)[::-1][:src_count]
        top_a = np.argsort(a)[::-1][:src_count]
        sources = np.unique(np.concatenate((top_s, top_a)))

        improved_this_sweep = False

        for sv in sources:
            s = int(sv)
            if a[s] < 1e-12:
                continue

            us = np.empty(acnt, dtype=np.float64)
            for j, lagv in enumerate(active):
                lag = int(lagv)
                val = 0.0
                p = lag - s
                if 0 <= p < n:
                    val += float(a[p])
                q = s - lag
                if 0 <= q < n:
                    val += float(a[q])
                us[j] = val

            delta = max(float(a[s]) * 0.02, 1e-9)
            approx = np.full(n, -np.inf, dtype=np.float64)

            for j, lagv in enumerate(active):
                lag = int(lagv)
                u_d = np.zeros(n, dtype=np.float64)
                m1 = (lag - d_arr >= 0) & (lag - d_arr < n)
                if np.any(m1):
                    u_d[m1] += a[lag - d_arr[m1]]
                m2 = (d_arr - lag >= 0) & (d_arr - lag < n)
                if np.any(m2):
                    u_d[m2] += a[d_arr - lag[m2]]
                vals = c[lag] + delta * (u_d - us[j])
                approx = np.maximum(approx, vals)

            approx[~np.isfinite(approx)] = 1e100
            order = np.argsort(approx)

            tries = 0
            for dv in order:
                d = int(dv)
                if d == s:
                    continue
                tries += 1
                if tries > 14:
                    break
                if approx[d] > M + 1e-10:
                    break

                for frac in fractions:
                    amt = float(a[s]) * float(frac)
                    if amt <= a[s] * 0.0 + 1e-18 or amt > a[s]:
                        continue
                    trial = a.copy()
                    trial[s] -= amt
                    trial[d] += amt
                    trial = np.maximum(trial, 0.0)
                    ts = float(trial.sum())
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    mtr = float(fast_autocorr(trial).max())

                    if mtr < best_m - 1e-14:
                        a = trial
                        best = a.copy()
                        best_m = mtr
                        cur_m = mtr
                        stall = 0
                        improved_this_sweep = True
                        break

                    if mtr < cur_m - 1e-13 and mtr < best_m * (1.0005):
                        a = trial
                        cur_m = mtr
                        stall = 0
                        improved_this_sweep = True
                        break

                    if rng.random() < 0.03 and mtr < best_m * (1.002):
                        a = trial
                        cur_m = mtr
                        improved_this_sweep = True
                        break

                if improved_this_sweep:
                    break

            if improved_this_sweep:
                break

        if not improved_this_sweep:
            stall += 1
            for _ in range(40):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.04))
                amt = float(a[i]) * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                trial = np.maximum(trial, 0.0)
                ts = float(trial.sum())
                if ts < 1e-15:
                    continue
                trial = trial / ts
                mtr = float(fast_autocorr(trial).max())

                if mtr < best_m - 1e-14:
                    a = trial
                    best = a.copy()
                    best_m = mtr
                    cur_m = mtr
                    stall = 0
                    improved_this_sweep = True
                    break

                if mtr < cur_m - 1e-13 and mtr < best_m * (1.0005):
                    a = trial
                    cur_m = mtr
                    stall = 0
                    improved_this_sweep = True
                    break

                if rng.random() < 0.02 and mtr < best_m * (1.002):
                    a = trial
                    cur_m = mtr
                    improved_this_sweep = True
                    break

            if verbose and (stall % 50 == 0 or improved_this_sweep):
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    peak-shave stall={stall} ({elapsed:.0f}s): J={J:.6f}")

    return best, float(best_m)


def lp_frank_wolfe_optimize(a0, time_limit, solver, topK=80, verbose=False):
    """
    Corrected Frank-Wolfe step using the LP from the reference idea:
    maximize total mass b >= 0 subject to linearized convolution caps
    conv(y, b) <= M, where y is the current normalized sequence and
    M = max conv(y,y). The LP solution is normalized and used as a descent
    direction.
    """
    n = len(a0)
    if solver is None:
        return norm_sum1(a0), float(fast_autocorr(norm_sum1(a0)).max())

    y = norm_sum1(a0)
    best = y.copy()
    cur_m = float(fast_autocorr(y).max())
    best_m = cur_m
    t0 = time.monotonic()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M = float(np.max(ac))
        if M <= 1e-30:
            break

        K = int(min(max(1, topK), 2 * n - 1))
        idx = np.argsort(ac)[::-1][:K]
        T = np.zeros((K, n), dtype=np.float64)
        for r, lagv in enumerate(idx):
            lag = int(lagv)
            js = max(0, lag - n + 1)
            je = min(n - 1, lag)
            if js <= je:
                jsel = np.arange(js, je + 1)
                T[r, jsel] = y[lag - jsel]

        b = cp.Variable(n, nonneg=True)
        constraints = [
            T @ b <= M * np.ones(K, dtype=np.float64),
            b <= 10.0
        ]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue

        bval = b.value
        if bval is None:
            continue
        bval = np.maximum(np.asarray(bval, dtype=np.float64), 0.0)
        bsum = float(bval.sum())
        if bsum <= 1e-12:
            continue
        g = norm_sum1(bval)

        alphas = (2.0 / (t + 2.0), 1.0 / (t + 1.0), 0.2, 0.08)
        best_new = y.copy()
        best_new_m = cur_m

        for al in alphas:
            al = float(al)
            if al <= 0.0 or al >= 1.0:
                continue
            new = (1.0 - al) * y + al * g
            new = np.maximum(new, 0.0)
            new = norm_sum1(new)
            mnew = float(fast_autocorr(new).max())
            if mnew < best_new_m:
                best_new_m = mnew
                best_new = new

        if best_new_m < best_m - 1e-14:
            best_m = best_new_m
            best = best_new.copy()

        if best_new_m <= M + 1e-13:
            y = best_new
            cur_m = best_new_m

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    lp-fw it={t} ({elapsed:.0f}s): J={J:.6f}")

    return best, float(best_m)


def interpolate_sequence(f, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    f = norm_sum1(f)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    old_x = (np.arange(old_n, dtype=np.float64) - 0.5) / float(old_n)
    new_x = (np.arange(new_n, dtype=np.float64) - 0.5) / float(new_n)
    vals = np.interp(new_x, old_x, f)
    vals = np.maximum(vals, 0.0)
    s = float(vals.sum())
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return vals / s


def make_power_law_init(n, exponent=1.0, side="right"):
    """Asymmetric power-law initialization."""
    i = np.arange(n, dtype=np.float64)
    if side == "right":
        base = 1.0 + i / float(n)
    else:
        base = 1.0 + (float(n) - 1.0 - i) / float(n)
    a = np.power(base, float(exponent))
    return norm_sum1(a)


def make_exp_decay_init(n, rate=2.0, side="right"):
    """Asymmetric exponential initialization."""
    i = np.arange(n, dtype=np.float64)
    if side == "right":
        a = np.exp(float(rate) * i / float(n))
    else:
        a = np.exp(float(rate) * (float(n) - 1.0 - i) / float(n))
    return norm_sum1(a)


def make_shifted_gaussian_init(n, center=0.35, width=0.20):
    """Shifted Gaussian initialization."""
    i = np.arange(n, dtype=np.float64)
    x = i / float(n)
    a = np.exp(-0.5 * ((x - float(center)) / float(width)) ** 2)
    return norm_sum1(a)


def blend_sequence(a, b, t):
    """Mix two sequences: t*a + (1-t)*b, normalized."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.size != b.size:
        b = interpolate_sequence(b, a.size)
    out = float(t) * a + (1.0 - float(t)) * b
    return norm_sum1(out)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator. Returns the best float list found."""
    t_start = time.monotonic()
    allowed = min(float(budget_s), 1800.0)
    deadline = t_start + max(60.0, allowed - 35.0)

    print(f"Starting search. budget={allowed:.0f}s effective_deadline=+{deadline - t_start:.0f}s")

    best_seq = None
    best_score = float("inf")
    best_n = 0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and p.sum() > 1e-10:
            parent = norm_sum1(p)
    except Exception:
        parent = None

    if parent is not None:
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)
        best_n = len(best_seq)
        print(f"Loaded parent: n={best_n}, score={best_score:.6f}")
    else:
        best_n = 3200
        best_seq = make_exp_decay_init(best_n, 1.8, "right")
        best_score = evaluate_safe(best_seq)
        print(f"No parent; using default n={best_n}, score={best_score:.6f}")

    # Quick diverse initialization screening at the current length.
    init_candidates = [
        make_power_law_init(best_n, 1.0, "right"),
        make_power_law_init(best_n, 2.0, "right"),
        make_power_law_init(best_n, 3.0, "right"),
        make_exp_decay_init(best_n, 1.2, "right"),
        make_exp_decay_init(best_n, 2.6, "right"),
        make_exp_decay_init(best_n, 5.0, "right"),
        make_shifted_gaussian_init(best_n, 0.30, 0.18),
        make_shifted_gaussian_init(best_n, 0.40, 0.25),
    ]
    if parent is not None:
        init_candidates.append(blend_sequence(parent, make_exp_decay_init(best_n, 2.4, "right"), 0.6))
        init_candidates.append(blend_sequence(parent, make_power_law_init(best_n, 1.8, "right"), 0.5))
        init_candidates.append(blend_sequence(parent, make_shifted_gaussian_init(best_n, 0.32, 0.22), 0.55))

    for init in init_candidates:
        sc = evaluate_safe(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = init
            best_n = len(best_seq)
            print(f"Quick init improvement: n={best_n}, score={best_score:.6f}")

    # Short initial polish from the best known point.
    rem = deadline - time.monotonic()
    if rem > 60:
        t_use = min(55.0, rem * 0.25)
        if t_use > 20:
            a_out, _ = mirror_optimize(
                best_seq, t_use, beta=35.0, topk=24,
                step_init=0.45, step_max=1.0, exact_prob=0.45,
                seed=17, verbose=False
            )
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                best_n = len(best_seq)
                print(f"Early mirror improvement: n={best_n}, score={best_score:.6f}")

    rem = deadline - time.monotonic()
    if rem > 90:
        t_use = min(90.0, rem * 0.30)
        if t_use > 25:
            a_out, _ = peak_shaving_descent(
                best_seq, t_use, seed=31,
                active_count=24, source_count=32, verbose=False
            )
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                best_n = len(best_seq)
                print(f"Early peak-shave improvement: n={best_n}, score={best_score:.6f}")

    # Main mirror-descent sweep with varied hyperparameters.
    mirror_params = [
        (28.0, 30, 0.52, 1.25, 0.35, 101),
        (55.0, 14, 0.38, 0.85, 0.62, 102),

        rem = deadline - time.monotonic()

            break

    # LP Frank-Wolfe phase 2 with more constraints.
    rem = deadline - time.monotonic()
    if solver is not None and best_n <= 4200 and rem > 120:
        t_use = min(180.0, rem * 0.28)
        if t_use > 40:
            print(f"LP-FW phase 2 t={t_use:.0f}s topK=120")

            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                best_n = len(best_seq)
                print(f"LP-FW phase2 improvement: n={best_n}, score={best_score:.6f}")

    # Length variants. Longer discretizations can approach the continuous limit.
    offsets = (200, 400, 800, -200, -400)
    tried_lengths = set()
    for off in offsets:
        rem = deadline - time.monotonic()
        if rem < 160:
            break
        new_n = best_n + int(off)

            print(f"Final LP-FW polish t={t_use:.0f}s")
            a_out, _ = lp_frank_wolfe_optimize(best_seq, t_use, solver, topK=70, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                best_n = len(best_seq)
                print(f"Final LP-FW improvement: n={best_n}, score={best_score:.6f}")

    # Final validation and fallback.
    if best_seq is None:
        best_seq = make_exp_decay_init(3200, 1.8, "right")
    final_score = evaluate_safe(best_seq)
    if not np.isfinite(final_score):
        best_seq = make_power_law_init(3200, 1.2, "right")
        final_score = evaluate_safe(best_seq)
    else:
        best_score = final_score

    elapsed = time.monotonic() - t_start
    print(f"FINAL SEARCH done: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")

    if best_seq is None or len(best_seq) == 0:
        fallback = make_exp_decay_init(3200, 1.8, "right")
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
```
