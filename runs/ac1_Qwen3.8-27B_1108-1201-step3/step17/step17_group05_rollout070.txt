def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_grad_at_peak(a, top_k=5, beta=20.0):
    """Compute subgradient focusing on top-k peak lags."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_top = w_top / w_top.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w_top, top_idx):
        ki = int(ki)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad[js] += 2.0 * wi * a[ki - js]

def compute_grad_full(a, beta=8.0):
    """Compute soft subgradient using all lags with exp weighting."""
    n = len(a)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_noise_init(n, seed=0, n_freqs=5):
    """Create a noise-like initialization with multiple frequencies."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for i in range(n_freqs):
        freq = rng.uniform(1, 10)
        coeff = rng.normal(0, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_asymmetric_init(n, seed=0):
    """Create an asymmetric initialization (geometric-like)."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    decay = rng.uniform(0.5, 2.0)
    a = np.exp(-decay * x) + 0.05
    a += 0.1 * rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_symmetric_init(n, seed=0):
    """Create a symmetric smooth initialization."""
    a = make_noise_init(n, seed=seed, n_freqs=4)
    a_sym = 0.5 * (a + a[::-1])
    return norm_sum1(a_sym)


def make_bipolar_init(n, seed=0):
    """Create a bipolar (U-shaped) initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    alpha = rng.uniform(0.3, 0.7)
    a = 1.0 + alpha * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.05)
    a += 0.05 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
