import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def subgradient_at_lag(a, k):
    """Exact subgradient of c_k(a) = sum_i a_i * a_{i+k}."""
    n = len(a)
    g = np.zeros(n)
    # c_k = sum_{i=0}^{n-1-k} a_i * a_{i+k}
    # dc_k/da_j = a_{j+k} (if j+k < n) + a_{j-k} (if j-k >= 0)
    if k == 0:
        g[:] = 2 * a
    else:
        g[:n-k] += a[k:]
        g[k:] += a[:n-k]
    return g

def md_optimize(a0, time_limit, verbose=False):

    a = norm_sum1(a0)

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # FW iterate
    x = a.copy()  # Nesterov point
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Select top K constraints
        K_sel = min(K, 2*n-1)
        top_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix (sparse)
        T_rows = []
        T_cols = []
        T_vals = []
        for ci, k in enumerate(top_idx):
            k_i = int(k)
            for j in range(max(0, k_i - n + 1), min(n, k_i + 1)):
                m = k_i - j
                if 0 <= m < n:
                    T_rows.append(ci)
                    T_cols.append(m)
                    T_vals.append(y[j])
        
        from scipy.sparse import csr_matrix
        T = csr_matrix((T_vals, (T_rows, T_cols)), shape=(K_sel, n))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M * np.ones(K_sel),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def subgradient_at_lag(a, k):
    n = len(a)
    g = np.zeros(n)
    k = int(k)
    if k == 0:
        g[:] = 2.0 * a
    elif k < n:
        g[:n - k] += a[k:]
        g[k:] += a[:n - k]
    return g


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def make_smooth_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss


def build_constraint_matrix(y, top_idx, n):
    K = len(top_idx)
    rows = []
    cols = []
    vals = []
    for ci, k in enumerate(top_idx):
        k_i = int(k)
        j_start = max(0, k_i - n + 1)
        j_end = min(n - 1, k_i)
        js = np.arange(j_start, j_end + 1)
        rows.extend([ci] * len(js))
        cols.extend(js.tolist())
        vals.extend(y[js].tolist())
    if len(vals) == 0:
        return csr_matrix((K, n))
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def mirror_descent(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    rng = np.random.default_rng(42)

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()

        eps = max(0.02 * M, 1e-8)
        active = np.where(ac >= M - eps)[0]
        n_active = min(len(active), 15)
        active_sel = active[:n_active]

        g = np.zeros(n)
        for k in active_sel:
            g += subgradient_at_lag(y, int(k))
        g /= n_active

        eta = 4.0 / (np.sqrt(t) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x.copy()
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, K_init=150, K_max=400, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    K_cur = K_init
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M = ac.max()

        K_sel = min(K_cur, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]

        T = build_constraint_matrix(y, top_idx, n)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M * np.ones(K_sel),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-10:
            continue

        d = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * d
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        improved = False
        mc_y = fast_autocorr(y).max()
        if mc_y < best_m - 1e-15:
            best_m = mc_y
            best_a = y.copy()
            improved = True

        mc_x = fast_autocorr(x).max()
        if mc_x < best_m - 1e-15:
            best_m = mc_x
            best_a = x.copy()
            improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        # Adapt K: increase if stalled (more constraints = better direction)
        if stall > 200 and K_cur < K_max:
            K_cur = min(K_cur + 50, K_max)
            stall = 0

        # Periodic restart if very stalled
        if stall > 1000:
            stall = 0
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            pert = rng.normal(0, 0.01, n) * best_a
            y = best_a + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
            x = y.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_cur}")

    return best_a, best_m


def projected_gradient_optimize(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 0.1 / n

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        # Multi-lag subgradient
        eps = max(0.05 * M, 1e-10)
        active = np.where(ac >= M - eps)[0]
        n_act = min(len(active), 10)
        g = np.zeros(n)
        for k in active[:n_act]:
            g += subgradient_at_lag(a, int(k))
        g /= n_act

        # Backtracking line search
        improved = False
        for trial in range(20):
            eta_try = eta * (0.7 ** trial)
            a_trial = a - eta_try * g
            a_trial = simplex_project(a_trial)
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                # Increase eta if we used a smaller one
                if trial > 0:
                    eta = max(eta * 1.1, eta_try * 1.5)
                break
        if not improved:
            eta *= 0.5
            if eta < 1e-10:
                break

    return best_a, best_m


def coordinate_descent_optimize(a0, time_limit, verbose=True, seed=123):

            if t_avail < 10:
                break
            print(f"  MD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent(init, t_avail, verbose=False)
            try_update(a_out, f"md_{lbl}")

    # === Phase 2: Main LP Frank-Wolfe (55% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW Main (n={n_best}, remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.55, 900.0)

    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, K_init=150, K_max=400, verbose=True)
        try_update(a_out, f"lp_fw_main_n{n_best}")

    # === Phase 3: Projected gradient refinement (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: ProjGrad (n={n_best}, remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 170.0)

    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = projected_gradient_optimize(a_in, t3, verbose=False)
        try_update(a_out, f"projgrad_n{n_best}")

    # === Phase 4: Second LP-FW round (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW Round 2 (n={n_best}, remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)

    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, K_init=200, K_max=500, verbose=True)
        try_update(a_out, f"lp_fw_r2_n{n_best}")

    # === Phase 5: Coordinate descent polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Descent (n={n_best}, remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 170.0)

    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_optimize(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Final projected gradient + LP (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}, remaining={remaining:.0f}s) ===")

    if remaining > 60:
        t6a = min(remaining * 0.4, 100.0)
        if t6a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = projected_gradient_optimize(a_in, t6a, verbose=False)
            try_update(a_out, f"projgrad_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.6, 100.0)
            if t6b > 20 and n_best <= 4000:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, K_init=200, K_max=400, verbose=False)
                try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Try nearby lengths (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 100, n_best + 100]
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 4500]

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 7: Length n={new_n} (remaining={remaining:.0f}s) ===")
        a_long = interpolate_seq(best_seq, new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 20:
            a_out, _ = mirror_descent(a_long, t7a, verbose=False)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4000:
            t7b = min(remaining * 0.5, 150.0)
            if t7b > 20:
                a_out, _ = lp_frank_wolfe(a_long, t7b, solver, K_init=150, K_max=300, verbose=False)
                try_update(a_out, f"lp_n{new_n}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_smooth(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
