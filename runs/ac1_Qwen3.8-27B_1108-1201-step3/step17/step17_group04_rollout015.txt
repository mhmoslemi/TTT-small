def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lse_objective_and_grad(a, beta):
    """
    Compute log-sum-exp of autocorrelation and its gradient using FFT.
    Returns (lse_value, gradient_vector).
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Numerically stable LSE
    ac_max = ac.max()
    ac_shifted = ac - ac_max
    exp_ac = np.exp(beta * ac_shifted)
    lse_val = (ac_max + np.log(exp_ac.sum())) / beta
    
    # Softmax weights
    p = exp_ac / exp_ac.sum()
    
    # Gradient: grad[m] = sum_j a[j] * (p[m+j] + p[m-j]) with zero-padding
    # Using FFT for efficiency
    a_rev = a[::-1]
    
    # conv1_full = convolve(a_rev, p, 'full'), take indices [n-1 : 2n-1]
    # conv2_full = convolve(a, p, 'full'), take indices [0 : n]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv2_full = np.fft.irfft(A * P, fft_size)[:n]
    
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv1_full = np.fft.irfft(A_rev * P, fft_size)[n - 1:2 * n - 1]
    
    grad = conv1_full + conv2_full
    return lse_val, grad


def lse_smoothed_optimize(a0, time_limit, verbose=False, seed=0,
                          beta_start=5.0, beta_end=300.0,
                          lr_start=0.5, lr_end=0.01,
                          momentum=0.9, n_steps_max=500000):
    """
    LSE-smoothed gradient optimization with adaptive temperature and momentum.
    Uses mirror descent (exponential update) with Nesterov-like momentum.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Momentum buffer
    v = np.zeros(n)
    
    # Track progress
    stall = 0
    max_stall = 5000
    
    for step in range(1, n_steps_max + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta (logarithmic annealing)
        progress = min(step / (n_steps_max * 0.8), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        # Adaptive learning rate (decreasing)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute LSE gradient
        lse_val, grad = lse_objective_and_grad(a, beta)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        
        # Mirror descent (exponential) update with momentum
        # a_new = a * exp(-lr * v) / sum(a * exp(-lr * v))
        log_step = -lr * v
        # Clip for numerical stability
        log_step = np.clip(log_step, -50, 50)
        a_new = a * np.exp(log_step)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Periodic restart with perturbation
        if stall > max_stall:
            stall = 0
            sigma = 0.005 + 0.01 * (step // (2 * max_stall))
            pert = rng.normal(0, sigma, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE restart ({elapsed:.0f}s): J={best_J:.6f}, beta={beta:.1f}")
        
        # Adaptive momentum (reduce when oscillating)
        if stall > max_stall // 2:
            momentum = max(0.7, momentum - 0.001)
        else:
            momentum = min(0.95, momentum + 0.0005)
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={step} ({elapsed:.0f}s): J={best_J:.6f}, beta={beta:.1f}, lr={lr:.4f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    LSE done ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def nesterov_subgradient(a0, time_limit, verbose=False,
                         top_k=15, beta=3.0, c=3.0,
                         exact_frac=0.4, restart_every=3000,
                         nesterov=True, seed=42):
    """Nesterov-accelerated mirror descent with adaptive exact/soft subgradient mixing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        use_exact = rng.random() < exact_frac
        if use_exact:
            k = int(np.argmax(ac))
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            g = np.zeros(n)
            g[idx] = 2.0 * a[j_idx]
        else:
            k_max = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** cur_beta
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
            g = np.zeros(n)
            for wi, kk in zip(w, top_idx):
                kk = int(kk)
                i_start = max(0, kk - n + 1)
                i_end = min(n, kk + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = kk - idx
                g[idx] += 2.0 * wi * a[j_idx]

        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.01 + 0.01 * (stall // restart_every)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 80)
            cur_beta = max(cur_beta - 0.3, 1.5)
            cur_c = min(cur_c + 0.5, 8.0)
            exact_frac = min(exact_frac + 0.05, 0.7)

    return best_a, best_J


def peak_reduction_moves(a0, time_limit, verbose=False, seed=77):
    """
    Targeted coordinate moves: identify peak contributors and relocate mass to safe zones.
    Uses sensitivity analysis to find optimal transfer destinations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_peaks = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc

        high_sens = np.where((sensitivity > sensitivity.mean() + 0.5 * sensitivity.std()) & (a > 0.0005))[0]
        low_sens = np.where((sensitivity < sensitivity.mean() - 0.3 * sensitivity.std()) | (sensitivity == 0))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False

        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            fracs = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0

        if not improved:
            stall += 1
            if stall % 30 == 0:
                n_pert = max(5, n // 200)
                perturb_positions = rng.choice(n, size=n_pert, replace=False)
                for p in perturb_positions:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=500):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, seed=123):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed, using the fact that the objective is
    piecewise quadratic in each coordinate.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # For the peak lag, the contribution is sum_i a[i]*a[i+peak_k]
        # If we change a[m] to a[m] + delta (compensated elsewhere),
        # the change in r(peak_k) is approximately 2 * delta * (a[m+peak_k] + a[m-peak_k])
        
        # Try moving mass from high-contributing positions to low-contributing ones
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = peak_k - idx
        
        contrib = a[idx] * a[j_idx]
        
        if len(idx) == 0:
            stall += 1
            continue
        
        # Find the position with highest contribution
        max_c_idx = np.argmax(contrib)
        src = idx[max_c_idx]
        
        # Find destination: position that contributes least to top peaks
        top_lags = np.argsort(ac)[::-1][:5]
        sens = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            s0 = max(0, ki - n + 1)
            s1 = min(n, ki + 1)
            ii = np.arange(s0, s1)
            jj = ki - ii
            cc = a[ii] * a[jj]
            sens[ii] += cc
            sens[jj] += cc
        
        candidates = np.where(sens < sens.mean())[0]
        if len(candidates) == 0:
            stall += 1
            continue
        
        # Try several transfer fractions
        improved = False
        src_val = a[src]
        if src_val > 1e-10:
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = src_val * frac
                for dst in rng.choice(candidates, size=min(5, len(candidates)), replace=False):
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J - 1e-14:
                        a = a_trial
                        best_J = J_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    J_new = compute_J(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDescent final ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Create a symmetric initialization (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_weighted_cosine_init(n, freq=2.0):
    """Weighted cosine with different frequency."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * freq * x) ** 2 * np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_multi_peak_init(n, n_peaks=4):
    """Multiple Gaussian peaks."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_peaks):
        center = (i + 0.5) / n_peaks
        sigma = 1.0 / (n_peaks * 4)
        a += np.exp(-((x - center) / sigma) ** 2)
    a += 0.005
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with LSE-smoothed optimization as the primary method,
    combined with peak reduction, LP Frank-Wolfe, and coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LSE-smoothed optimization (MAIN) ===
    print(f"\n=== Phase 1: LSE Smoothed Optimization (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_symmetric_init(n0))
        labels.append("symmetric")

        inits.append(make_power_law_init(n0, power=0.8))
        labels.append("power0.8")

        inits.append(make_gaussian_edges(n0, sigma=0.10))
        labels.append("gauss_edges")

        inits.append(make_weighted_cosine_init(n0, freq=2.0))
        labels.append("cosine2")

        # Blend parent with symmetric
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_symmetric_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(30.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 15:
                break
            print(f"  LSE init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = lse_smoothed_optimize(init, t_avail, verbose=False,
                                              beta_start=5.0, beta_end=200.0,
                                              lr_start=0.3, lr_end=0.005,
                                              momentum=0.9, seed=100 + idx)
            try_update(a_out, f"lse_{lbl}")

    # === Phase 2: Nesterov subgradient refinement ===
    print(f"\n=== Phase 2: Nesterov SGD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t2, verbose=False,
                                        top_k=20, beta=3.0, c=3.0,
                                        exact_frac=0.5, restart_every=3000,
                                        seed=42)
        try_update(a_out, f"sgd_n{len(a_in)}")

    # === Phase 3: Peak reduction moves ===
    print(f"\n=== Phase 3: Peak reduction (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_moves(a_in, t3, verbose=True, seed=77)
        try_update(a_out, f"peakred_n{len(a_in)}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=400)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 5: LSE with higher beta (focus on top peak) ===
    print(f"\n=== Phase 5: LSE high-beta (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lse_smoothed_optimize(a_in, t5, verbose=False,
                                         beta_start=100.0, beta_end=500.0,
                                         lr_start=0.1, lr_end=0.001,
                                         momentum=0.85, seed=500)
        try_update(a_out, f"lse_highbeta_n{len(a_in)}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t6, verbose=True, seed=123)
        try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 7: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    for new_n in [n_best + 300, n_best - 300]:
        if new_n < 500 or new_n > 6000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t7 = min(remaining * 0.20, 120.0)
        if t7 > 20:
            a_out, _ = lse_smoothed_optimize(a_long, t7, verbose=False,
                                             beta_start=5.0, beta_end=200.0,
                                             lr_start=0.3, lr_end=0.005,
                                             momentum=0.9, seed=555)
            try_update(a_out, f"lse_n{new_n}")

    # === Phase 8: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 40:
        a_in = norm_sum1(best_seq)
        # Short LSE
        t8a = min(remaining * 0.3, 100.0)
        if t8a > 15:
            a_out, _ = lse_smoothed_optimize(a_in, t8a, verbose=False,
                                             beta_start=50.0, beta_end=300.0,
                                             lr_start=0.1, lr_end=0.002,
                                             momentum=0.9, seed=777)
            try_update(a_out, f"lse_polish_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        # Short peak reduction
        if remaining > 20:
            t8b = min(remaining * 0.3, 80.0)
            if t8b > 10:
                a_out, _ = peak_reduction_moves(norm_sum1(best_seq), t8b, verbose=False, seed=999)
                try_update(a_out, f"peakred_polish_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        # Short coordinate descent
        if remaining > 15:
            t8c = min(remaining * 0.3, 60.0)
            if t8c > 10:
                a_out, _ = coordinate_descent(norm_sum1(best_seq), t8c, verbose=False, seed=2000)
                try_update(a_out, f"coord_polish_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
