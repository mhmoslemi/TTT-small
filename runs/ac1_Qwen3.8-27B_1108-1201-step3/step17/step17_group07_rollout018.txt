        if no_improve > max_stall:
            no_improve = 0
            max_stall = max(1000, int(max_stall * 0.7))
            # Perturb
            strategy = int(rng.integers(3))
            if strategy == 0:
                a = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                a = best_a.copy()
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                a[idx] *= rng.uniform(0.5, 1.5, len(idx))
            else:
                a = best_a * (1 + rng.normal(0, 0.003, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            # Reset Adam
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            t = 0
            lr = max(0.01, lr * 0.7)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_b
            print(f"    SMD t={t} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, no_improve={no_improve}")
    
    return best_a, best_b

def lp_remez(a, max_iter=100, time_limit=100.0, solver=None, r=0.15, verbose=False):
    """LP-based Remez iteration without trust region."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    active_set = [int(np.argmax(fast_conv(a)))]
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        
        # Update active set: add lags where b[k] > current max of active
        max_active = max(b[k] for k in active_set)
        new_active = [k for k in range(len(b)) if b[k] > max_active * 0.999]
        if len(new_active) > 30:
            # Keep top 30
            new_active = sorted(new_active, key=lambda k: -b[k])[:30]
        active_set = new_active
        
        # Solve LP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        
        for k in active_set:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                # Linearization: sum coeffs * a_var <= M + b[k] - sum coeffs * a
                # i.e., sum coeffs * a_var - M <= b[k]
                offset = np.sum(coeffs * a[idx])
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Line search
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                break
        
        if not accepted:
            # Try without line search (full step)
            m_full = fast_conv(a_new).max()
            if m_full < best_m - 1e-18:
                a = a_new
                best_m = m_full
                best_a = a.copy()
            else:
                break  # No improvement, stop
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez it={it} ({elapsed:.1f}s): J={J:.8f}, |K|={len(active_set)}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit=30.0, verbose=False):
    """Random pair coordinate descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = max(20, n // 100)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 5)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.001, 0.005, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sinc_sqr':
        t = np.pi * (x - 0.5)
        a = np.where(np.abs(t) > 1e-8, (np.sin(t) / t) ** 2, 1.0)
    else:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to_n(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
