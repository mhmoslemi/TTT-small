def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_subgrad(a0, time_limit, top_k=20, beta=3.0, c=3.0,
                     exact_frac=0.5, restart_every=3000, seed=42):
    """Nesterov-accelerated mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c
    cur_exact_frac = exact_frac

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        g = get_subgradient_multi(y, ac, top_k=cur_top_k, beta=cur_beta,
                                  exact_frac=cur_exact_frac, rng=rng)

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break

        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = cur_c * gap / (g_norm_sq + 1e-20)
        eta = min(eta, 3.0 / (t + 1.0))
        eta = max(eta, 1e-9)

        log_y = np.log(y)
        log_a_new = log_y - eta * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.02 + 0.005 * (stall // restart_every)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_c = min(cur_c + 0.3, 8.0)
            cur_exact_frac = min(cur_exact_frac + 0.03, 0.8)

    return best_a, best_J


def safe_transfer_optimize(a0, time_limit, seed=77, n_sources=30, n_sinks=30,
                           n_top_peaks=15, frac_candidates=None):
    """
    Smart coordinate descent: identify safe transfers from peak contributors to safe zones.
    Uses all top peaks simultaneously to determine safe destinations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    if frac_candidates is None:
        frac_candidates = [0.02, 0.05, 0.08, 0.12, 0.2, 0.35, 0.5, 0.7]

    stall = 0
    max_stall = 200
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iter_count += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top peaks
        n_peaks = min(n_top_peaks, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute interaction vectors for top peaks
        # c_i(k) = a_{k-i} + a_{i-k} (interaction of position i with lag k)
        # For efficiency, compute for all i and top peaks
        c_matrix = np.zeros((n, n_peaks))
        for pi, k in enumerate(top_lags):
            k_int = int(k)
            for i in range(n):
                v1 = k_int - i
                v2 = i - k_int
                val = 0.0
                if 0 <= v1 < n:
                    val += a[v1]
                if 0 <= v2 < n:
                    val += a[v2]
                c_matrix[i, pi] = val

        # Weight peaks by their values (higher peaks get more weight)
        peak_vals = ac[top_lags]
        weights = peak_vals / peak_vals.sum()

        # Weighted danger: D_i = sum_p weights[p] * c_matrix[i,p]
        danger = c_matrix @ weights

        # Also compute "peak-focused" danger (only top 3 peaks)
        top3_weights = np.zeros(n_peaks)
        top3_weights[:3] = 1.0 / 3.0
        danger_peak3 = c_matrix @ top3_weights

        # Find sources: high danger, substantial mass
        threshold_src = danger_peak3.mean() + 0.3 * danger_peak3.std()
        sources = np.where((danger_peak3 > threshold_src) & (a > 0.0005))[0]

        # Find sinks: low danger
        threshold_sink = danger.mean() - 0.2 * danger.std()
        sinks = np.where(danger < threshold_sink)[0]

        if len(sources) == 0 or len(sinks) == 0:
            # Fallback: use rank-based selection
            src_rank = np.argsort(danger_peak3)[::-1]
            sink_rank = np.argsort(danger)
            sources = src_rank[:n_sources]
            sources = sources[a[sources] > 0.0003]
            sinks = sink_rank[:n_sinks]
            if len(sources) == 0 or len(sinks) == 0:
                stall += 1
                continue

        sources = sources[:n_sources]
        sinks = sinks[:n_sinks]

        best_move_J = best_J
        best_move = None

        # Try all source-sink pairs
        n_tried = 0
        for i in sources:
            if time.monotonic() - t0 > time_limit:
                break
            for j in sinks:
                if i == j:
                    continue
                n_tried += 1
                src_val = a[i]
                if src_val < 1e-8:
                    continue

                for frac in frac_candidates:
                    amt = src_val * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    J_trial = compute_J(a_trial)
                    if J_trial < best_move_J:
                        best_move_J = J_trial
                        best_move = (i, j, frac, a_trial.copy())

        if best_move is not None and best_move_J < best_J - 1e-14:
            i, j, frac, a_new = best_move
            a = a_new
            best_J = best_move_J
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            if stall % 30 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    p = int(rng.integers(n))
                    if a[p] > 1e-6:
                        dst = int(rng.integers(n))
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[p] * frac
                        a[p] -= amt
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0

        if iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SafeTransfer ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def local_search_fine(a0, time_limit, seed=123, n_pairs=200):
    """
    Fine local search: try many random pairs with optimal line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]

        # Compute danger for source/sink selection
        danger = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            for i in range(n):
                v1 = k_int - i
                v2 = i - k_int
                if 0 <= v1 < n:
                    danger[i] += a[v1]
                if 0 <= v2 < n:
                    danger[i] += a[v2]

        improved = False
        for attempt in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick source weighted by danger
            src_probs = np.maximum(danger * a, 0)
            src_probs = src_probs / (src_probs.sum() + 1e-30)
            i = int(rng.choice(n, p=src_probs))

            # Pick sink weighted by low danger
            sink_probs = np.maximum(1.0 - danger / (danger.max() + 1e-30), 0.01)
            sink_probs = sink_probs / sink_probs.sum()
            j = int(rng.choice(n, p=sink_probs))

            if i == j or a[i] < 1e-8:
                continue

            # Binary search for best fraction
            lo, hi = 0.01, 0.9
            best_frac = 0.1
            best_trial_J = compute_J(a)

            for _ in range(8):
                mid = (lo + hi) / 2.0
                amt = a[i] * mid
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid

            if best_trial_J < best_J - 1e-14:
                amt = a[i] * best_frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1

    return best_a, best_J


def parametric_shapes(n, n_params=200, rng=None):
    """Generate various parametric shapes for initialization."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    xc = x - 0.5  # centered

    shapes = []

    # U-shaped: high at edges
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]:
        a = 1.0 + alpha * xc**2
        a = np.maximum(a, 0.01)
        shapes.append(norm_sum1(a))

    # Inverted U: high in middle
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 - alpha * xc**2
        a = np.maximum(a, 0.001)
        shapes.append(norm_sum1(a))

    # Gaussian center
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
        a = np.exp(-xc**2 / (2 * sigma**2))
        shapes.append(norm_sum1(a))

    # Two bumps
    for d in [0.2, 0.25, 0.3, 0.35, 0.4]:
        for sigma in [0.05, 0.08, 0.12, 0.15]:
            a = np.exp(-(x - (0.5 - d))**2 / (2*sigma**2)) + np.exp(-(x - (0.5 + d))**2 / (2*sigma**2))
            shapes.append(norm_sum1(a))

    # Power law edges
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = x**p + (1-x)**p + 0.01
        shapes.append(norm_sum1(a))

    # Exponential edges
    for alpha in [1, 2, 4, 8, 15]:
        a = np.exp(alpha * (x*(1-x)))  # high in middle, low at edges... wait
        # Actually x*(1-x) is high in middle. Let's use 1-4*x*(1-x) = (2x-1)^2
        a = np.exp(-alpha * (1 - 4*x*(1-x)))  # = exp(-alpha*(2x-1)^2) high in middle
        shapes.append(norm_sum1(a))

    # Asymmetric
    for skew in [0.1, 0.2, 0.3]:
        a = np.exp(-((x - 0.5 + skew)**2) / (2*0.15**2)) + np.exp(-((1-x - 0.5 + skew)**2) / (2*0.15**2))
        shapes.append(norm_sum1(a))

    # Random smooth
    for _ in range(30):
        freqs = rng.uniform(0.5, 5.0, 5)
        amps = rng.uniform(0.1, 1.0, 5)
        phases = rng.uniform(0, 2*np.pi, 5)
        a = np.ones(n)
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.sin(2*np.pi*f*x + ph)
        a = np.maximum(a, 0.01)
        shapes.append(norm_sum1(a))

    return shapes


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def anneal_polish(a0, time_limit, seed=999):
    """
    Simulated-annealing-style polish: try random perturbations,
    accept if improvement, with decreasing perturbation size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    temp = 1.0
    decay = 0.995

    while time.monotonic() - t0 < time_limit:
        temp *= decay
        # Number of positions to perturb
        n_pert = max(1, int(n * temp * 0.01))
        positions = rng.choice(n, size=n_pert, replace=False)

        a_trial = a.copy()
        for p in positions:
            if a[p] > 1e-8:
                frac = rng.uniform(0.005, 0.05 * temp + 0.005)
                amt = a[p] * frac
                dst = int(rng.integers(n))
                a_trial[p] -= amt
                a_trial[dst] += amt

        a_trial = np.maximum(a_trial, 0.0)
        a_trial = a_trial / a_trial.sum()
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
        elif J_trial < best_J + 0.01 * temp:
            # Accept worse with annealing probability
            a = a_trial
        # else: keep a unchanged

    return best_a, best_J


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(1.0 + 2.0 * (x - 0.5)**2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Parametric exploration ===
    print(f"\n=== Phase 1: Parametric shapes (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 30:
        rng = np.random.default_rng(0)
        shapes = parametric_shapes(n0, rng=rng)
        t_per_shape = t1 / len(shapes)
        best_shape_J = float('inf')
        best_shape = None

        for idx, shape in enumerate(shapes):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per_shape, remaining - 30)
            if t_avail < 5:
                # Just evaluate directly
                J_direct = compute_J(shape)
                if J_direct < best_shape_J:
                    best_shape_J = J_direct
                    best_shape = shape.copy()
                continue

            a_out, J_out = nesterov_subgrad(shape, t_avail, top_k=15, beta=3.0,
                                            c=3.0, exact_frac=0.5,
                                            restart_every=2000, seed=42+idx)
            if J_out < best_shape_J:
                best_shape_J = J_out
                best_shape = a_out

        if best_shape is not None:
            try_update(best_shape, "parametric")

        # Also try short SGD from parent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_short = min(remaining * 0.1, 100.0)
            a_out, _ = nesterov_subgrad(norm_sum1(parent), t_short, top_k=20,
                                        beta=3.0, c=4.0, exact_frac=0.5,
                                        restart_every=2000, seed=111)
            try_update(a_out, "sgd_parent")

    # === Phase 2: Safe transfer optimization ===
    print(f"\n=== Phase 2: Safe transfers (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)

                                          n_sources=40, n_sinks=40,
                                          n_top_peaks=15)
        try_update(a_out, "safe_transfer")

    # === Phase 3: Fine local search ===
    print(f"\n=== Phase 3: Fine local search (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_search_fine(a_in, t3, seed=123, n_pairs=300)
        try_update(a_out, "local_fine")

    # === Phase 4: SGD polish ===
    print(f"\n=== Phase 4: SGD polish (n={len(best_seq)}) ===")

        try_update(a_long, f"interp_n{new_n}")

        t6a = min(remaining * 0.25, 150.0)
        if t6a > 20:
            a_out, _ = nesterov_subgrad(a_long, t6a, top_k=20, beta=3.0,
                                        c=3.5, exact_frac=0.5,
                                        restart_every=2000, seed=555+new_n%100)
            try_update(a_out, f"sgd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = safe_transfer_optimize(norm_sum1(best_seq),
                                              min(remaining * 0.2, 80.0),
                                              seed=333, n_sources=30, n_sinks=30)
            try_update(a_out, f"safe2_n{len(a_in) if 'a_in' in dir() else n_best}")

    # === Phase 7: Second round of safe transfers on best ===
