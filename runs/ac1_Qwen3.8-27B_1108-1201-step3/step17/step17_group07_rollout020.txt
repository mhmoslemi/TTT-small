b'[k] = b[k] + 2*delta*(a[k-i]-a[k-j]) + delta^2*(I[2i=k]+I[2j=k]-2*I[i+j=k]),
enabling O(n) evaluation without full FFT per candidate.
"""

import numpy as np
import time

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1 and clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_topk_subgrad(y, ac, top_k):
    """Compute subgradient using top-k lags. O(n*k)."""
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=25, c0=5.0, verbose=False):
    """Nesterov mirror descent with top-k subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        ac = fast_conv(y)
        M = ac.max()
        near_max = int(np.sum(ac >= M * 0.995))
        k_use = min(max(top_k, near_max + 5), min(150, len(ac)))
        g = compute_topk_subgrad(y, ac, k_use)
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        x_prev = x
        x = a_new
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.65))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.03, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 3, 800), replace=False)
                x_r[idx] *= rng.uniform(0.4, 1.6, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.008, n))
            else:
                # Antisymmetric perturbation
                half = n // 2
                x_r = best_a.copy()
                pert_left = rng.normal(0, 0.02, half)
                x_r[:half] *= (1 + pert_left)
                x_r[half:] *= (1 - pert_left[:min(half, n - half)])
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.5)
        if verbose and t % 30000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}")
    return best_a, best_m

def pair_cd_search(a, time_limit, verbose=False, seed=123, greedy_fraction=0.4, triple_search=True):
    """
    Pair-coordinate descent with exact ternary/bisection search.
    Uses O(n) delta-effect evaluation (no FFT per candidate).
    """
    n = len(a)
    a = norm_sum1(a).copy()
    b = fast_conv(a)
    best_a = a.copy()
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iterations = 0
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Choose pair: greedy or random
        if rng.random() < greedy_fraction:
            # Greedy: target the max lag
            k_star = int(np.argmax(b))
            # Compute subgradient direction
            sg = np.zeros(n)
            for kk in [k_star]:
                kk = int(kk)
                j_start = max(0, kk - n + 1)
                j_end = min(n - 1, kk)
                if j_start <= j_end:
                    idx = np.arange(j_start, j_end + 1)
                    sg[idx] += 2.0 * a[kk - idx]
            # Best i: largest sg (most to decrease)
            # Best j: smallest sg (most to increase)
            i = int(np.argmax(sg))
            j = int(np.argmin(sg))
            if sg[i] - sg[j] < 1e-10:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    j = (j + 1) % n
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                j = (j + 1) % n
        
        if a[i] < 1e-12 and a[j] < 1e-12:
            stall += 1
            continue
        
        # Try both directions
        best_delta = 0.0
        best_m_trial = best_m
        
        for direction in [1, -1]:
            # direction=1: a[i]-=delta, a[j]+=delta
            # direction=-1: a[j]-=delta, a[i]+=delta
            if direction == 1:
                src, dst = i, j
                max_delta = a[src] * 0.9
            else:
                src, dst = j, i
                max_delta = a[src] * 0.9
            
            if max_delta < 1e-14:
                continue
            
            # Compute effect vector: effect[k] = 2*(a[k-src] - a[k-dst])
            effect = np.zeros(2 * n - 1)
            # a[k-src] is nonzero for k in [src, src+n-1]
            k_lo1 = src
            k_hi1 = min(src + n - 1, 2 * n - 2)
            if k_lo1 <= k_hi1:
                effect[k_lo1:k_hi1 + 1] += 2.0 * a[0:k_hi1 - k_lo1 + 1]
            k_lo2 = dst
            k_hi2 = min(dst + n - 1, 2 * n - 2)
            if k_lo2 <= k_hi2:
                effect[k_lo2:k_hi2 + 1] -= 2.0 * a[0:k_hi2 - k_lo2 + 1]
            
            # Quadratic terms: only at k=2*src, k=2*dst, k=src+dst
            quad = np.zeros(2 * n - 1)
            if 2 * src < 2 * n - 1:
                quad[2 * src] += 1.0
            if 2 * dst < 2 * n - 1:
                quad[2 * dst] += 1.0
            if src + dst < 2 * n - 1:
                quad[src + dst] -= 2.0
            
            # Ternary search for optimal delta in [0, max_delta]
            lo = 0.0
            hi = max_delta
            if triple_search:
                for _ in range(20):
                    t1 = lo + (hi - lo) / 3.0
                    t2 = hi - (hi - lo) / 3.0
                    m1 = np.max(b + t1 * effect + t1 * t1 * quad)
                    m2 = np.max(b + t2 * effect + t2 * t2 * quad)
                    if m1 < m2:
                        hi = t2
                    else:
                        lo = t1
                best_d = 0.5 * (lo + hi)
            else:
                best_d = 0.5 * max_delta
            
            # Evaluate at best_d and at 0
            m_best = np.max(b + best_d * effect + best_d * best_d * quad)
            if m_best < best_m_trial - 1e-14:
                best_m_trial = m_best
                best_delta = best_d
                best_direction = direction
        
        if best_delta > 1e-14 and best_m_trial < best_m - 1e-14:
            # Apply the move
            if best_direction == 1:
                a[i] -= best_delta
                a[j] += best_delta
            else:
                a[j] -= best_delta
                a[i] += best_delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s
            b = fast_conv(a)
            if b.max() < best_m - 1e-15:
                best_m = b.max()
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        iterations += 1
        if verbose and iterations % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairCD iter={iterations} ({elapsed:.1f}s): J={J:.8f}, stall={stall}")
    
    return best_a, best_m

def block_cd_search(a, time_limit, verbose=False, seed=456, block_size=4):
    """
    Block coordinate descent: move a small 'profile' from one region to another.
    Generalizes pair-CD to multi-coordinate moves.
    """
    n = len(a)
    a = norm_sum1(a).copy()
    b = fast_conv(a)
    best_a = a.copy()
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iterations = 0
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Pick source and target blocks
        ws = min(block_size, n // 4)
        i_start = int(rng.integers(0, n - ws + 1))
        j_start = int(rng.integers(0, n - ws + 1))
        if abs(i_start - j_start) < ws:
            continue  # blocks overlap
        
        src_idx = np.arange(i_start, i_start + ws)
        dst_idx = np.arange(j_start, j_start + ws)
        src_vals = a[src_idx].copy()
        
        if src_vals.max() < 1e-12:
            stall += 1
            continue
        
        # Effect: a[src_idx] -= alpha * src_vals, a[dst_idx] += alpha * src_vals
        # For the convolution, the effect on each lag k is:
        # delta_b[k] = -alpha * sum_{p in src} src_vals[p] * a[k-(i_start+p)] 
        #              + alpha * sum_{q in dst} src_vals[q-i_start] * a[k-(j_start+q)]
        #              + alpha^2 * cross_terms
        # This is O(n*ws) to compute, which is fine for small ws.
        
        # Simplified: just compute effect for linear term
        effect = np.zeros(2 * n - 1)
        for p in range(ws):
            sv = src_vals[p]
            if sv < 1e-14:
                continue
            src_pos = i_start + p
            dst_pos = j_start + p
            # Contribution from -alpha * sv at src_pos
            k_lo = src_pos
            k_hi = min(src_pos + n - 1, 2 * n - 2)
            if k_lo <= k_hi:
                effect[k_lo:k_hi + 1] -= alpha_sign * sv * a[0:k_hi - k_lo + 1]
            # Contribution from +alpha * sv at dst_pos
            k_lo = dst_pos
            k_hi = min(dst_pos + n - 1, 2 * n - 2)
            if k_lo <= k_hi:
                effect[k_lo:k_hi + 1] += alpha_sign * sv * a[0:k_hi - k_lo + 1]
        
        # This is getting complex with the quadratic terms for multi-block moves.
        # Let's simplify: just try a few alpha values directly.
        best_alpha = 0.0
        best_m_block = best_m
        for alpha_val in [0.01, 0.03, 0.05, 0.1, 0.2, -0.01, -0.03, -0.05, -0.1, -0.2]:
            a_trial = a.copy()
            a_trial[src_idx] -= alpha_val * src_vals
            a_trial[dst_idx] += alpha_val * src_vals
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m_block - 1e-14:
                best_m_block = m_trial
                best_alpha = alpha_val
        
        if best_alpha != 0.0 and best_m_block < best_m - 1e-14:
            a[src_idx] -= best_alpha * src_vals
            a[dst_idx] += best_alpha * src_vals
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            b = fast_conv(a)
            if b.max() < best_m:
                best_m = b.max()
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        iterations += 1
        if verbose and iterations % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    BlockCD iter={iterations} ({elapsed:.1f}s): J={J:.8f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    """Create initial sequences of various shapes."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bilinear_asym':
        # Asymmetric: high on left, low on right
        a = np.maximum(1.0 - x, 0.0) * x + 0.3 * (1.0 - x)
    elif kind == 'exp_asym':
        a = np.exp(-3.0 * x)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy: NMD for coarse optimization, then intensive pair-CD for fine-tuning.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2400
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: NMD from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 1: NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t1, top_k=30, c0=6.0, verbose=True)
        try_update(a_out, f"NMD_parent")
    
    # === Phase 2: Intensive Pair-CD (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 600.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: PairCD n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = pair_cd_search(norm_sum1(best_seq), t2, verbose=True, seed=123, greedy_fraction=0.5, triple_search=True)
        try_update(a_out, f"PairCD_main_n{n_cur}")
    
    # === Phase 3: NMD polish with small steps ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 120.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: NMD polish n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t3, top_k=10, c0=1.5, verbose=True)
        try_update(a_out, f"NMD_polish_n{n_cur}")
    
    # === Phase 4: More Pair-CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 350.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: PairCD round2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = pair_cd_search(norm_sum1(best_seq), t4, verbose=True, seed=789, greedy_fraction=0.6, triple_search=True)
        try_update(a_out, f"PairCD_r2_n{n_cur}")
    
    # === Phase 5: Try slightly larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        n_alt = n0 + 200
        t5_total = min(remaining * 0.25, 300.0)
        print(f"\n=== Phase 5: Alt length n={n_alt} (budget={t5_total:.0f}s) ===")
        if t5_total > 60:
            a_alt = interpolate_sequence(best_seq, n_alt)
            t5a = t5_total * 0.5
            a_out, _ = nesterov_mirror_descent(a_alt, t5a, top_k=20, c0=4.0, verbose=False)
            try_update(a_out, f"NMD_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40 and len(best_seq) == n_alt:
                a_out, _ = pair_cd_search(norm_sum1(best_seq), min(remaining * 0.4, 150.0), verbose=False, seed=321)
                try_update(a_out, f"PairCD_n{n_alt}")
    
    # === Phase 6: Final intensive Pair-CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.30, 300.0)
    print(f"\n=== Phase 6: Final PairCD n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = pair_cd_search(norm_sum1(best_seq), t6, verbose=True, seed=555, greedy_fraction=0.7, triple_search=True)
        try_update(a_out, f"PairCD_final_n{n_cur}")
    
    # === Phase 7: Block CD for structural moves ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 100.0)
    print(f"\n=== Phase 7: BlockCD n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = block_cd_search(norm_sum1(best_seq), t7, verbose=False, seed=99, block_size=3)
        try_update(a_out, f"BlockCD_n{n_cur}")
        
        # Follow up with pair-CD on the result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = pair_cd_search(norm_sum1(best_seq), min(remaining * 0.5, 80.0), verbose=False, seed=111)
            try_update(a_out, f"PairCD_after_block_n{n_cur}")
    
    # === Phase 8: Final NMD + PairCD squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final squeeze n={n_cur} (budget={remaining:.0f}s) ===")
        t8a = min(remaining * 0.3, 60.0)
        if t8a > 10:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t8a, top_k=5, c0=0.8, verbose=False)
            try_update(a_out, f"NMD_squeeze_n{n_cur}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = pair_cd_search(norm_sum1(best_seq), min(remaining - 5, 120.0), verbose=True, seed=2024, greedy_fraction=0.8, triple_search=True)
            try_update(a_out, f"PairCD_squeeze_n{n_cur}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
