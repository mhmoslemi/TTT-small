"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent with full FFT-based gradient (log-sum-exp surrogate)
2. Nesterov acceleration with adaptive step size
3. Multi-beta annealing and diverse initializations
4. Local refinement via coordinate descent and random walk
5. Multi-length search with interpolation
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, R, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*R[k]))) w.r.t. a.
    g[m] = 2 * sum_j p[m+j] * a[j] where p = softmax(beta*R).
    """
    n = len(a)
    # Softmax with stability
    R_max = R.max()
    exp_vals = np.exp(beta * (R - R_max))
    p = exp_vals / exp_vals.sum()
    
    # Reverse p for convolution
    p_rev = p[::-1].copy()
    
    # Linear conv
    conv_len = (2*n - 1) + n - 1  # 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(P * A, fft_size)[:conv_len]
    
    # g[m] = 2 * C[2n-2-m]
    g = 2.0 * C[n-1:2*n-1][::-1]
    return g

def mirror_descent_smooth(a0, time_limit, beta, c0=3.0, restart_every=5000, verbose=False):
    """
    Smoothed mirror descent with Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        mu = (t - 1.0) / (t + 1.5)
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute R and gradient
        R = fast_conv(y)
        g = compute_smooth_grad(y, R, beta)
        
        # Adaptive step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent update (multiplicative)
        # x_new[i] = y[i] * exp(-eta * g[i]) / Z
        log_x_new = np.log(y) - eta * g
        log_x_new -= log_x_new.max()  # stability
        x_new = np.exp(log_x_new)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        x_prev = x
        x = x_new
        
        # Track best
        M = fast_conv(x).max()
        if M < best_m - 1e-16:
            best_m = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stuck
        if stall > restart_every:
            stall = 0
            restart_count += 1
            # Perturb the best
            noise_scale = min(0.3, 0.05 * restart_count)
            pert = rng.normal(0, noise_scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SMD restart {restart_count} (t={t}, {elapsed:.0f}s): J={2*n*best_m:.6f}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta}")
    
    return best_a, best_m


def local_refine(a0, time_limit, verbose=False):
    """

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, R, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*R[k]))) w.r.t. a.
    Result: g[m] = 2 * sum_j p[m+j] * a[j] where p = softmax(beta*R).
