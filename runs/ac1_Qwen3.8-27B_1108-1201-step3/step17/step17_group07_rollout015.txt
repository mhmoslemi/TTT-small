def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_subgrad_full(y, ac, top_k):
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=25, c0=3.0, verbose=False):

            break

        near_max = np.sum(ac >= M * 0.998)

        g = compute_subgrad_full(y, ac, k_use)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        x_prev = x
        x = a_new
        M_new = fast_conv(x).max()

def remez_lp_enhanced(a0, time_limit, solver=None, verbose=False, max_active=250):

    a = norm_sum1(a0)
    best_a = a.copy()

    t0 = time.monotonic()
    r = 0.05
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_conv(a)
        max_b = b.max()
        # Adaptive active set: all lags within adaptive threshold of max
        # Start with tight threshold, expand if too few constraints
        threshold_frac = 0.99995 - min(0.005, iter_count * 0.00001)
        active = np.where(b >= max_b * threshold_frac)[0]
        if len(active) < 20:
            threshold_frac = 0.999
            active = np.where(b >= max_b * threshold_frac)[0]
        if len(active) > max_active:
            active = np.argsort(b)[::-1][:max_active]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= 0)

        # Trust region
        a_lower = np.maximum(a * (1.0 - r), 0.0)
        a_upper = a * (1.0 + r)
        constraints.append(a_var >= a_lower)
        constraints.append(a_var <= a_upper)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            r = max(0.001, r * 0.5)
            no_improve += 1
            if no_improve > 30:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            r = max(0.001, r * 0.5)
            no_improve += 1
            if no_improve > 30:
                break
            continue

        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            r = max(0.001, r * 0.5)
            no_improve += 1
            continue
        a_new = a_new / s

            best_trial_m = m_full

            for _ in range(20):
                mid = (lo + hi) / 2
                a_trial = (1 - mid) * a + mid * a_new
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    hi = mid
                    continue
                a_trial = a_trial / st
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-18:
                    lo = mid
                    best_frac = mid
                    best_trial_m = m_trial
                else:
                    hi = mid

        if best_frac > 0:
            a = (1 - best_frac) * a + best_frac * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = fast_conv(a).max()
            if best_m < 2.0 * n * best_m / (1.0 ** 2):  # always true, just tracking
                best_a = a.copy()
            no_improve = 0
            r = min(0.1, r * 1.05)
        else:
            no_improve += 1
            r = max(0.0005, r * 0.6)
            if no_improve > 40:

                rng = np.random.default_rng(iter_count)
                idx = rng.choice(n, size=min(n // 5, 500), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_conv(a).max()
                best_a = a.copy()

                r = 0.03

            J = 2.0 * n * best_m

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            best_local_frac = 0
            best_local_m = best_m
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_local_m - 1e-16:
                    best_local_m = m_trial
                    best_local_frac = frac
                    a_best_local = a_trial.copy()
            if best_local_frac > 0:
                a = a_best_local
                best_m = best_local_m
                best_a = a.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sin_bump':
        a = np.maximum(np.sin(np.pi * x) ** 0.7, 0.0)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        else:

            r = max(1e-4, r * 0.7)
            if no_improve > 60:
                break

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 10.0, 1790.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n0 = len(best_seq)
    print(f"Starting optimization with n={n0}, score={best_score:.10f}")

    # === Phase 1: NMD from parent (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t1, top_k=30, c0=4.0, verbose=True)
        try_update(a_out, f"NMD1_n{n0}")

    # === Phase 2: Enhanced Remez LP (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 2: Enhanced Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_lp_enhanced(norm_sum1(best_seq), t2, solver=solver, verbose=True, max_active=200)
        try_update(a_out, f"Remez2_n{len(best_seq)}")

    # === Phase 3: NMD polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: NMD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t3, top_k=15, c0=1.5, verbose=False)
        try_update(a_out, f"NMD3_n{len(best_seq)}")

    # === Phase 4: Final Remez with tiny trust region (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Final Remez n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = remez_lp_final(norm_sum1(best_seq), t4, solver=solver, max_active=300, r_init=0.01)
        try_update(a_out, f"Remez4_n{len(best_seq)}")

    # === Phase 5: Coordinate descent polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 5: CD polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t5, verbose=False, rng_seed=777)
        try_update(a_out, f"CD5_n{len(best_seq)}")

    # === Phase 6: Alternate length exploration (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_alt = n0 + 200
        t6 = min(remaining * 0.5, 120.0)
        if t6 > 20:
            print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
            a_alt = interpolate_sequence(best_seq, n_alt)
            a_out, _ = nesterov_mirror_descent(a_alt, t6, top_k=25, c0=3.0, verbose=False)
            try_update(a_out, f"NMD6_n{n_alt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = remez_lp_enhanced(norm_sum1(best_seq), remaining - 5, solver=solver, verbose=False, max_active=200)
                try_update(a_out, f"Remez6_n{len(best_seq)}")
        else:
            a_out, _ = remez_lp_enhanced(norm_sum1(best_seq), remaining - 5, solver=solver, verbose=False, max_active=200)
            try_update(a_out, f"Remez6_n{len(best_seq)}")

    # === Phase 7: Final aggressive push (remaining time, min 10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 7: Final push n={len(best_seq)} (budget={remaining:.0f}s) ===")
        t7a = remaining * 0.6
        if t7a > 10:
            a_out, _ = remez_lp_final(norm_sum1(best_seq), t7a, solver=solver, max_active=300, r_init=0.005)
            try_update(a_out, f"Remez7a_n{len(best_seq)}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 5:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), remaining - 2, top_k=8, c0=0.8, verbose=False)
            try_update(a_out, f"NMD7b_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
