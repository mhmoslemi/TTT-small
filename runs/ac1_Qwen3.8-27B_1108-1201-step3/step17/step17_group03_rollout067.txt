def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def make_parametric(n, family='sin2'):
    x = np.linspace(0.0, 1.0, n)

        f = np.sin(np.pi * x) ** 4

    elif family == 'beta':
        f = np.maximum(x * (1.0 - x), 1e-10)

def compute_subgradient(a, ac, top_k=50, sharpness=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** sharpness
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def lp_subproblem(f, M, active_idx, solver, n):
    K = len(active_idx)
    T_sel = np.zeros((K, n))
    for row in range(K):
        k_int = int(active_idx[row])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            T_sel[row, i] = f[k_int - i]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sel @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val is None or b_val.sum() < 1e-15:
        return None
    return b_val / b_val.sum()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def freq_perturb(a, strength=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

        for new_n in longer_lengths[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 150:
                break
            print(f"\n=== Phase 6: Extend n={new_n} ===")
            a_long = interpolate_sequence(best_seq, new_n)

            t6a = min(remaining * 0.30, 200.0)
            if t6a > 30:
                a_out, _ = mirror_descent_optimize(a_long, t6a, top_k=50, sharpness=4.0,
                                                   c0=3.0, restart_every=2000, verbose=False)
                best_seq, best_score = update_best(a_out, best_seq, best_score, t_global, f"md_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 100 and new_n <= 4500:
                t6b = min(remaining * 0.25, 150.0)
                if t6b > 25:
                    a_out, _ = lp_frank_wolfe(a_long, t6b, solver, top_K=200,
                                              margin_frac=2e-3, verbose=False)
                    best_seq, best_score = update_best(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 7: Frequency diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Freq diversify (n={n_best}) ===")
    rng = np.random.default_rng(200)
    n_trials = 3 if remaining > 300 else (2 if remaining > 180 else 1)
    for trial in range(n_trials):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)

        t7a = min(remaining * 0.25, 120.0)
        if t7a > 25:
            a_out, _ = mirror_descent_optimize(a_pert, t7a, top_k=50, sharpness=5.0,
                                               c0=4.0, restart_every=1500, verbose=False)
            best_seq, best_score = update_best(a_out, best_seq, best_score, t_global, f"md_f{trial}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) <= 4500:
            t7b = min(remaining * 0.25, 120.0)
            if t7b > 20:
                a_out, _ = lp_frank_wolfe(a_pert, t7b, solver, top_K=250,
                                          margin_frac=1e-3, verbose=False)
                best_seq, best_score = update_best(a_out, best_seq, best_score, t_global, f"lp_f{trial}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.30, 180.0)
    if t8 > 25 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t8, solver, top_K=500, margin_frac=1e-4,
                                  verbose=True, use_iqcd=True)
        best_seq, best_score = update_best(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final peak elim ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final peak elim ===")
    if remaining > 30:
        t9 = min(remaining * 0.5, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_elimination(a_in, t9, verbose=True, seed=777)
            best_seq, best_score = update_best(a_out, best_seq, best_score, t_global, "peak_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sin2')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
