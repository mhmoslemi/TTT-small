"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def compute_subgradient_exact(a, conv, tol=1e-10):
    """Compute exact subgradient of max(conv) - indicator of argmax set."""
    n = len(a)
    M = conv.max()
    # Find all lags within tol of the max
    tight = np.where(conv >= M - tol)[0]
    g = np.zeros(n)
    for k in tight:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    # Normalize by number of tight lags (average subgradient)
    if len(tight) > 0:
        g /= len(tight)
    return g

def compute_subgradient_soft(a, conv, top_k=50, beta=5.0):
    """Soft subgradient using top-K lags with exponential weighting."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g

def subgradient_descent(a0, time_limit, top_k=100, beta=8.0, c0=5.0,
                        restart_every=3000, verbose=False, seed=42):
    """Accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c0 = c0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Subgradient
        conv = fast_conv(y)
        g = compute_subgradient_soft(y, conv, top_k=cur_top_k, beta_w=cur_beta)
        
        # Polyak step size
        eta = cur_c0 / (np.sqrt(t) + 1.0)
        
        # Mirror step (KL projection)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive restart
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            # Adapt parameters
            cur_c0 = max(cur_c0 * 0.85, 0.5)
            cur_beta = max(cur_beta - 1.5, 2.0)
            cur_top_k = min(cur_top_k + 20, 300)
            if verbose:
                print(f"    SGrad restart t={t}: J={best_val:.6f}")
        
        if verbose and t % 5000 == 0:
            print(f"    SGrad t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")
    
    return best_a, best_val

def pressure_based_optimize(a0, time_limit, verbose=False, seed=42):
    """
    Pressure-based mass transfer optimization.
    Move mass from positions that contribute to peak lags
    to positions that don't.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        
        # Find top contributing lags
        n_top = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_top]
        
        # Compute pressure: for each position, how much it contributes to top lags
        pressure = np.zeros(n)
        for k in top_lags:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                # Contribution of position i to lag k is a[i] * a[k-i]
                contrib = a[idx] * a[k_i - idx]
                pressure[idx] += contrib
        
        # Normalize pressure by position value to get "responsibility"
        # High responsibility = this position is a big contributor to peaks
        resp = np.zeros(n)
        mask = a > 1e-15
        resp[mask] = pressure[mask] / a[mask]
        
        # Source: position with highest responsibility
        # But we want to move mass FROM high-resp TO low-resp
        src_candidates = np.where(resp > np.median(resp))[0]
        if len(src_candidates) == 0:
            stall += 1
            continue
        src = src_candidates[np.argmax(resp[src_candidates])]
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Destination: position with lowest responsibility that can accept mass
        dst_candidates = np.where((resp < np.median(resp)) & np.arange(n) != src)[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.arange(n) != src
            dst_candidates = np.where(dst_candidates)[0]
        dst = dst_candidates[np.argmin(resp[dst_candidates])]
        
        # Try multiple transfer amounts
        improved = False
        fracs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]
        for frac in fracs:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-15:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-15:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturb to escape
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
        
        if verbose and stall % 200 == 0:
            print(f"    Pressure ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J

def make_geometric_decay(n, ratio, seed=0):
    """Geometric decay: a_i ∝ ratio^i, normalized."""
    i = np.arange(n, dtype=np.float64)
    a = ratio ** i
    s = a.sum()
    return a / s

def make_power_law(n, power, seed=0):
    """Power law: a_i ∝ (n+1-i)^power (decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = (n + 1 - i) ** power
    s = a.sum()
    return a / s

def make_asymmetric_smooth(n, alpha, beta, seed=0):
    """Asymmetric smooth: a_i ∝ exp(-alpha*i)*(1-exp(-beta*(n-1-i)))^0.5"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) * np.sqrt(np.maximum(1 - np.exp(-beta * (n - 1 - i)), 1e-15))
    s = a.sum()
    return a / s

def make_truncated_geom(n, ratio, trunc_frac=0.3):
    """Geometric decay but with a floor to avoid too-small tail."""
    i = np.arange(n, dtype=np.float64)
    a = ratio ** i
    floor_val = a[int(n * trunc_frac)]
    a = np.maximum(a, floor_val)
    s = a.sum()
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

        scored.append((J, lbl, init))

    top_inits = scored[:min(8, len(scored))]

        t_pr = min(remaining * 0.4, 80.0)

            a_out, _ = pressure_based_optimize(best_a.copy(), t_pr, verbose=False,
                                                seed=int(time.monotonic()*1000) % 100000)

    if best_a is not None:

        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
