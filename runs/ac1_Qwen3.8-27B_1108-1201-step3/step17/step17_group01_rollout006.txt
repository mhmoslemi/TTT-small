import numpy as np

def fast_conv(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad(a, conv, top_k=30, temperature=0.003):
    """Compute (sub)gradient of max_k conv[k] using softmax weighting."""
    n = len(a)
    conv_max = conv.max()
    # Find peaks within threshold
    threshold = conv_max - temperature * 8
    active = np.where(conv >= threshold)[0]
    if len(active) > top_k:
        # Keep only top_k
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def amd_optimize(a0, time_limit, verbose=False, seed=42):
    """Nesterov-accelerated mirror descent with adaptive subgradient."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adaptive parameters
    top_k = 20
    temperature = 0.003
    c0 = 3.0
    stall_count = 0
    max_restart = 5000
    restart_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient

        g = compute_subgrad(y, conv_y, top_k=top_k, temperature=temperature)
        
        # Mirror descent step

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x = a_new

        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled

            stall_count = 0
            restart_count += 1
            # Perturb

            pert = rng.normal(0, strength, n) * best_x

            x = x_r

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(x) for x in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score

        n_base = 4000
        x = np.linspace(0, 1, n_base)

    remaining = time_budget - (time.monotonic() - t_global)

        inits.append(make_mixed_structure(n0, pos1=0.1, pos2=0.8, width=0.12))
        labels.append("mixed_2")
        inits.append(make_weighted_structure(n0, alpha=0.5, beta=0.3))
        labels.append("weighted")
        
        # Sinc-like
        x = np.linspace(0, 1, n0)
        for bw in [3.0, 6.0, 10.0]:
            a = np.sinc((x - 0.5) * bw) ** 2
            a = np.maximum(a, 0.0)
            inits.append(norm_sum1(a))
            labels.append(f"sinc{bw}")

        for mode in [1, 2, 3]:
            a = np.cos(np.pi * mode * (x - 0.5)) ** 2
            a = np.maximum(a, 0.0)
            inits.append(norm_sum1(a))
            labels.append(f"cos{mode}")

        x_c = np.abs(np.arange(n0) - n0 / 2) + 1.0
        for pw in [0.5, 1.0]:

        a_in = norm_sum1(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    candidate_ns = []
    for delta in [-1000, -500, 500, 1000, 2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:

    for new_n in candidate_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:

        print(f"  Trying n={new_n}...")
        # Interpolate
        old_x = np.linspace(0.0, 1.0, n_best)
        new_x = np.linspace(0.0, 1.0, new_n)
        a_interp = np.interp(new_x, old_x, best_seq / best_seq.sum())
        a_interp = np.maximum(a_interp, 0.0)
        a_interp = a_interp / a_interp.sum()
        try_update(a_interp, f"interp_n{new_n}")
        
        t4a = min(remaining * 0.35, 150.0)
        if t4a > 15:
            a_out, _ = amd_optimize(a_interp, t4a, verbose=False, seed=777)
            try_update(a_out, f"amd_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t4b = min(remaining * 0.25, 100.0)
            if t4b > 10:
                a_out, _ = coord_descent_polish(a_interp, t4b, verbose=False, seed=555)
                try_update(a_out, f"coord_n{new_n}")
    
    # Phase 5: Final AMD + coordinate descent on absolute best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final Polish (n={n_best}) ===")
    
    if remaining > 30:
        t5a = min(remaining * 0.5, 200.0)
        if t5a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t5a, verbose=True, seed=2024)
            try_update(a_out, "amd_final")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t5b = min(remaining * 0.6, 150.0)
        if t5b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_polish(a_in, t5b, verbose=True, seed=31415)
            try_update(a_out, "coord_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_sequence([float(v) for v in best_seq])
    
    return [float(x) for x in best_seq]
