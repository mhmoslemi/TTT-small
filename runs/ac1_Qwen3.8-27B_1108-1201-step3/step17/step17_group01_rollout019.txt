```python
import numpy as np
import time

def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, c, k):
    """Compute gradient of c_k with respect to a.
    grad[i] = 2 * a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # For each i in range(n), k-i must be in [0, n)
    # So i must be in [k-(n-1), k] intersect [0, n-1]
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def multi_peak_subgradient(a, c, top_k=20, temperature=10.0):
    """Compute weighted subgradient using top peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    
    # Get top k peaks weighted by softmax
    max_c = c.max()
    weights = np.exp((c - max_c) / (temperature * max_c + 1e-30))
    # Only consider top peaks
    top_indices = np.argsort(c)[::-1][:k_max]
    w = np.zeros(len(c))
    w[top_indices] = weights[top_indices]
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-10:
            grad += w[k] * compute_subgradient(a, c, k)
    return grad
```

        x = a_new
        a = a_new
        
        # Track best
        c_new = fast_conv(a)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        # Periodic spectral smoothing
        if t % 5000 == 0:
            a = spectral_smooth_step(a, axtent=0.3, verbose=False)
            c_check = fast_conv(a)
            if c_check.max() < best_m:
                best_m = c_check.max()
                best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, T={T:.6f}")
    
    return best_a, best_m


def spectral_smooth_step(a, axtent=0.3, verbose=False):
    """One step of spectral flattening: compress dynamic range of |A|."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    # Compress dynamic range
    ratio = mag / (mag_mean + 1e-30)
    mag_new = mag_mean * ratio ** (1.0 - axtent)
    mag_new = np.maximum(mag_new, 1e-30)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    
    # Blend with original
    a_out = (1.0 - axtent * 0.5) * a + axtent * 0.5 * a_new
    return norm_simplex(a_out)


def spectral_optimize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with adaptive step size."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    step = 0.3
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = spectral_smooth_step(a, axtent=step)
        c = fast_conv(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            step = min(0.5, step * 1.05)
        else:
            # Revert and reduce step
            a = norm_simplex(0.5 * a + 0.5 * best_a)
            step = max(0.05, step * 0.7)
            
            if step < 0.05:
                break
    
    return best_a, best_m


def differential_evolution(a0, time_limit, pop_size=12, verbose=False):
    """Differential evolution in the sequence space."""
    n = len(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Initialize population
    population = [norm_simplex(a0)]
    for i in range(pop_size - 1):
        pert = rng.normal(0, 0.1, n)
        candidate = a0 + pert
        population.append(norm_simplex(candidate))
    
    scores = [fast_conv(p).max() for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_m = scores[best_idx]
    
    F = 0.8  # differential weight
    CR = 0.9  # crossover rate
    
    gen = 0
    while time.monotonic() - t0 < time_limit:
        gen += 1
        # Sort population
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
        if scores[0] < best_m:
            best_m = scores[0]
            best_a = population[0].copy()
        
        # Create new candidate from worst member
        worst_idx = len(population) - 1
        r1 = rng.integers(0, worst_idx)
        r2 = rng.integers(0, worst_idx)
        if r1 == r2:
            r2 = (r2 + 1) % worst_idx
        
        # Differential mutation
        mutant = population[r1] + F * (population[r2] - population[worst_idx])
        
        # Crossover
        trial = np.zeros(n)
        j_rand = rng.integers(n)
        for j in range(n):
            if rng.random() < CR or j == j_rand:
                trial[j] = mutant[j]
            else:
                trial[j] = population[worst_idx][j]
        
        trial = norm_simplex(trial)
        trial_score = fast_conv(trial).max()
        
        if trial_score < scores[worst_idx]:
            population[worst_idx] = trial
            scores[worst_idx] = trial_score
        
        # Occasionally add random perturbation to best
        if gen % 50 == 0:
            pert = rng.normal(0, 0.02, n)
            candidate = norm_simplex(best_a + pert)
            sc = fast_conv(candidate).max()
            worst_idx = len(population) - 1
            if sc < scores[worst_idx]:
                population[worst_idx] = candidate
                scores[worst_idx] = sc
        
        if verbose and gen % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DE gen={gen} ({elapsed:.0f}s): best_J={2*n*best_m:.6f}")
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between positions."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Find top peaks
        n_top = min(10, len(c))
        top_peaks = np.argsort(c)[::-1][:n_top]
        
        improved = False
        
        # Try moving mass from positions that contribute to peaks
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            # Find the position with highest contribution to this peak
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Try moving small amount to a random other position
            for attempt in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def make_init_flat(n):
    return np.ones(n) / n


def make_init_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_simplex(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def make_init_bowl(n, sharpness=2.0):
    """High at edges, low in middle (bowl shape)."""
    x = np.linspace(0, 1, n)
    a = (4.0 * (x - 0.5) ** 2) ** sharpness + 0.1
    return norm_simplex(a)


def make_init_dual_peak(n, pos1=0.25, pos2=0.75, width=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((x - pos2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_simplex(a)


def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_simplex(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_simplex(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        best_seq = make_init_cosine(n_base, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: FTRL optimization with diverse inits at base length ===
    print(f"\n=== Phase 1: FTRL (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)
    
    if t_phase1 > 60:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_simplex(parent))
            labels.append("parent")
        
        inits.append(make_init_cosine(n0, mode=1))
        labels.append("cos1")
        
        inits.append(make_init_bowl(n0, sharpness=2.0))
        labels.append("bowl2")
        
        inits.append(make_init_bowl(n0, sharpness=3.0))
        labels.append("bowl3")
        
        inits.append(make_init_dual_peak(n0, 0.2, 0.8, 0.12))
        labels.append("dualpeak")
        
        inits.append(make_init_ramp(n0, power=0.5))
        labels.append("sqrt_ramp")
        
        inits.append(make_init_sinc(n0, bw=6.0))
        labels.append("sinc6")
        
        # Try a few random smooth inits
        rng_init = np.random.default_rng(77)
        x = np.linspace(0, 1, n0)
        for seed in [1, 2]:
            a_r = np.ones(n0) * 0.5
            for f in range(1, 12):
                coeff = rng_init.normal(0, 0.12 / np.sqrt(f))
                phase = rng_init.uniform(0, 2 * np.pi)
                a_r += coeff * np.sin(2 * np.pi * f * x + phase)
            a_r = np.maximum(a_r, 0.001)
            inits.append(norm_simplex(a_r))
            labels.append(f"rand{seed}")
        
        # Deduplicate
        seen_set = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 6))
            if sig not in seen_set:
                seen_set.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels
        
        print(f"  Running FTRL on {len(inits)} initializations...")
        t_per = t_phase1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            remaining_now = time_budget - 30 - elapsed_so_far
            if remaining_now < 10:
                break
            t_avail = min(t_per, remaining_now)
            if t_avail < 8:
                break
            print(f"    FTRL '{lbl}' (budget={t_avail:.0f}s)...")
            a_out, _ = ftrl_optimize(init, t_avail, verbose=False)
            try_update(a_out, f"ftrl_{lbl}")
            # Quick spectral polish
            remaining_now = time_budget - 30 - (time.monotonic() - t_global)
            if remaining_now > 15:
                a_sp, _ = spectral_optimize(a_out, min(15.0, remaining_now * 0.3), verbose=False)
                try_update(a_sp, f"spectral_{lbl}")
    
    # === Phase 2: Differential evolution on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Differential Evolution (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_simplex(best_seq)
        a_out, _ = differential_evolution(a_in, t2, pop_size=10, verbose=False)
        try_update(a_out, f"de_n{n_best}")
        # FTRL polish after DE
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.10, 100.0)
        if t2b > 15:
            a_out, _ = ftrl_optimize(a_out, t2b, verbose=False)
            try_update(a_out, f"ftrl_after_de_n{n_best}")
    
    # === Phase 3: More FTRL with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FTRL variant (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_simplex(best_seq)
        # Perturb slightly before FTRL
        rng3 = np.random.default_rng(99)
        perturbed = norm_simplex(a_in + rng3.normal(0, 0.01, n_best))
        a_out, _ = ftrl_optimize(perturbed, t3, verbose=False)
        try_update(a_out, f"ftrl_variant_n{n_best}")
    
    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = norm_simplex(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, verbose=False)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length variation ===")
    
    # Try longer
    for delta in [500, 1000]:
        new_n = n_best + delta
        if new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")
        
        t5 = min(remaining * 0.25, 150.0)
        if t5 > 20:
            a_out, _ = ftrl_optimize(a_long, t5, verbose=False)
            try_update(a_out, f"ftrl_n{new_n}")
            a_sp, _ = spectral_optimize(a_out, min(30.0, remaining * 0.1), verbose=False)
            try_update(a_sp, f"spectral_n{new_n}")
    
    # Try shorter
    for delta in [500, 1000]:
        new_n = n_best - delta
        if new_n < 2000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_short = interpolate_sequence(best_seq, new_n)
        try_update(a_short, f"interp_short_n{new_n}")
        
        t5s = min(remaining * 0.25, 120.0)
        if t5s > 20:
            a_out, _ = ftrl_optimize(a_short, t5s, verbose=False)
            try_update(a_out, f"ftrl_short_n{new_n}")
    
    # === Phase 6: Final intensive FTRL on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final FTRL polish (n={n_best}) ===")
    t6 = min(remaining * 0.40, 300.0)
    if t6 > 20:
        a_in = norm_simplex(best_seq)
        a_out, _ = ftrl_optimize(a_in, t6, verbose=True)
        try_update(a_out, f"ftrl_final_n{n_best}")
        
        # Spectral polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_sp, _ = spectral_optimize(a_out, min(30.0, remaining * 0.5), verbose=False)
            try_update(a_sp, f"spectral_final_n{n_best}")
    
    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 7: Final CD (n={n_best}) ===")
        t7 = min(remaining * 0.6, 80.0)
        if t7 > 5:
            a_in = norm_simplex(best_seq)
            a_out, _ = coordinate_descent(a_in, t7, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_init_cosine(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
