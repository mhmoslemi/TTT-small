Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fictitious Play / Dual Descent with FFT-accelerated CG inner solver
2. Temperature annealing in the dual (autocorrelation peak weights)
3. Adaptive multiplicative updates (AMD) for polishing
4. Multi-init diverse search

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def hankel_matvec(lam, x):
    """Compute Q @ x where Q_{ij} = lam_{i+j}, Q is n x n."""
    n = len(x)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    lam_rev = lam[::-1].copy()
    L = np.fft.rfft(lam_rev, fft_size)
    X = np.fft.rfft(x, fft_size)
    conv = np.fft.irfft(L * X, fft_size)[:size]
    # (Qx)_i = conv[2n-2-i] for i = 0,...,n-1
    return conv[2*n-2 : n-1 : -1]

def hankel_cg_solve(lam, rhs, max_it=100, tol=1e-10):
    """Solve Q x = rhs using CG, where Q_{ij} = lam_{i+j}."""
    n = len(rhs)
    x = np.ones(n) / n  # initial guess
    r = rhs - hankel_matvec(lam, x)
    p = r.copy()
    rs_old = r @ r
    for it in range(max_it):
        Ap = hankel_matvec(lam, p)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = r_new @ r_new
        if np.sqrt(rs_new) < tol * np.sqrt(rs_old) + 1e-30:
            break
        p = r_new + (rs_new / rs_old) * p
        r = r_new
        rs_old = rs_new
    return x

"""

2. Temperature annealing in the dual (autocorrelation peak weights)
3. Adaptive multiplicative mirror descent (AMD) for polishing

5. Diverse multi-init search with coordinate descent polish

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_matvec(lam, x):
    """Compute Q @ x where Q_{ij} = lam_{i+j}, Q is n x n Hankel matrix."""
    n = len(x)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    lam_rev = lam[::-1].copy()
    L = np.fft.rfft(lam_rev, fft_size)
    X = np.fft.rfft(x, fft_size)
    conv = np.fft.irfft(L * X, fft_size)[:size]
    return conv[2 * n - 2 : n - 1 : -1]


def hankel_cg_solve(lam, rhs, max_it=60, tol=1e-10):
    """Solve Q x = rhs using preconditioned CG, where Q_{ij} = lam_{i+j}."""
    n = len(rhs)
    # Jacobi preconditioner: diagonal of Q is lam[0], lam[2], lam[4], ...
    diag = lam[0::2][:n]
    diag = np.maximum(diag, 1e-30)
    z_scale = 1.0 / diag

    x = np.ones(n) / n
    r = rhs - hankel_matvec(lam, x)
    z = r * z_scale
    p = z.copy()
    rz_old = r @ z
    for it in range(max_it):
        Ap = hankel_matvec(lam, p)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rz_old / pAp
        x += alpha * p
        r -= alpha * Ap
        z = r * z_scale
        rz_new = r @ z
        if abs(rz_new) < tol * max(abs(rz_old), 1e-30):
            break
        beta_cg = rz_new / rz_old if abs(rz_old) > 1e-30 else 0.0
        p = z + beta_cg * p
        rz_old = rz_new
    return x


def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def fictitious_play(a0, time_limit, beta_start=5.0, beta_end=0.8,
                    alpha_start=0.6, alpha_end=0.1, max_cg_it=50,
                    verbose=True, seed=42):
    """
    Fictitious Play / Dual Descent optimizer.
    Alternates between:
    - Primal: update a via CG solve of Q(lambda) a = 1
    - Dual: update lambda from autocorrelation peaks (annealed)
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stagnation = 0
    it = 0
    total_iter_target = 20000

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, it / max(1, total_iter_target))
        # Anneal beta (peak concentration)
        beta = beta_start * (beta_end / beta_start) ** progress
        # Anneal alpha (blend rate)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        # Add small randomization to alpha to avoid cycles
        alpha *= (1.0 + 0.1 * rng.standard_normal())
        alpha = np.clip(alpha, 0.01, 1.0)

        # Compute autocorrelation
        ac = fast_autocorr(a)

        # Form dual weights (concentrate on peaks)
        ac_pos = np.maximum(ac, 1e-30)
        log_ac = np.log(ac_pos)
        # Softmax-style weights with temperature controlled by beta
        lam = np.exp(beta * (log_ac - log_ac.max()))
        lam_sum = lam.sum()
        if lam_sum < 1e-30 or not np.isfinite(lam_sum):
            lam = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            lam = lam / lam_sum

        # Ensure lam is strictly positive for PD guarantee
        lam = np.maximum(lam, 1e-15)
        lam = lam / lam.sum()

        # Solve Q(lambda) a_new = 1
        rhs = np.ones(n)
        try:
            a_new = hankel_cg_solve(lam, rhs, max_it=max_cg_it, tol=1e-12)
        except Exception:
            a_new = a.copy()

        # Check for numerical issues
        if not np.all(np.isfinite(a_new)):
            a_new = a.copy()

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Blend update
        a_blended = (1.0 - alpha) * a + alpha * a_new
        a_blended = np.maximum(a_blended, 0.0)
        s = a_blended.sum()
        if s < 1e-15:
            a_blended = a.copy()
        else:
            a_blended = a_blended / s

        # Evaluate
        M = fast_autocorr(a_blended).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_blended.copy()
            a = a_blended
            stagnation = 0
        else:
            # Try smaller step
            alpha_small = alpha * 0.3
            a_try = (1.0 - alpha_small) * a + alpha_small * a_new
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st > 1e-15:
                a_try = a_try / st
                M_try = fast_autocorr(a_try).max()
                if M_try < best_m - 1e-15:
                    best_m = M_try
                    best_a = a_try.copy()
                    a = a_try
                    stagnation = 0
                elif M_try < M:
                    a = a_try
            else:
                stagnation += 1
            if M >= best_m:
                stagnation += 1

        # Stagnation handling: perturbation + parameter reset
        if stagnation > 300:
            pert_scale = 0.03 + 0.02 * min(stagnation / 1000, 3.0)
            pert = rng.normal(0, pert_scale, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stagnation = 0
            # Occasionally randomize beta
            if rng.random() < 0.3:
                beta_start = rng.uniform(2.0, 8.0)
                total_iter_target = it + 10000  # reset annealing schedule

        it += 1
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FP it={it} ({elapsed:.0f}s): J={J:.8f}, beta={beta:.2f}, "
                  f"alpha={alpha:.3f}, stall={stagnation}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                 restart_every=5000, verbose=False, seed=42):
    """Nesterov-accelerated mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        # Multi-peak subgradient
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            strength = min(0.15, 0.02 + 0.02 * (stall_count // restart_every + 1))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.3, 1.5)
            gamma = c0 * (1.0 + 0.15 * min(stall_count // restart_every + 1, 5))
            t = 0  # reset Nesterov momentum

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.8f}, stall={stall_count}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, iters=2000, verbose=False):
    """Spectral flattening with proper DC constraint handling."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Progressive flattening
        progress = it / max(1, iters)
        eta = 0.3 + 0.5 * progress
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        # Preserve DC component
        P_target[0] = P[0]

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecFlatten it={it} ({elapsed:.0f}s): J={J:.8f}")

    return best_a, best_m

            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try moving to several candidate positions
                candidates = rng.choice(n, size=min(20, n), replace=False)
                for dst in candidates:
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def make_structured_init(n, kind, seed=0):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "linear":
        a = x
    elif kind == "sqrt":
        a = np.sqrt(x)
    elif kind == "cos1":
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif kind == "cos2":
        a = np.cos(2 * np.pi * (x - 0.5)) ** 2
    elif kind == "sinc":
        a = np.sinc((x - 0.5) * 6.0) ** 2
    elif kind == "hump3":
        a = sum(np.exp(-((x - c) ** 2) / (2 * 0.15 ** 2)) for c in [0.25, 0.5, 0.75])
    elif kind == "hump5":
        a = sum(np.exp(-((x - c) ** 2) / (2 * 0.1 ** 2)) for c in [0.2, 0.35, 0.5, 0.65, 0.8])
    elif kind == "asym_ramp":
        a = x ** 1.5
    elif kind == "asym_inv":
        a = (1 - x) ** 1.5
    elif kind == "two_hump_asym":
        a = np.exp(-((x - 0.3) ** 2) / (2 * 0.12 ** 2)) + 0.5 * np.exp(-((x - 0.7) ** 2) / (2 * 0.08 ** 2))
    else:
        rng = np.random.default_rng(seed)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.1 / np.sqrt(f)) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    return norm_sum1(np.maximum(a, 0.0))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with fictitious play as the core optimizer."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_structured_init(n_base, "cos1")
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.8f}")

    # === Phase 1: Quick AMD warmup to get a better starting point ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: AMD warmup (n={n0}, {t1:.0f}s) ===")
        a_out, _ = amd_optimize(best_seq, t1, top_k=40, beta=4.0, c0=3.0,
                                restart_every=3000, seed=42)
        try_update(a_out, "amd_warmup")

    # === Phase 2: Main Fictitious Play optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.55, 1100.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Fictitious Play (n={n0}, {t2:.0f}s) ===")

        # Run 1: standard annealing
        a_in = norm_sum1(best_seq)
        a_out, m_out = fictitious_play(a_in, t2 * 0.5,
                                       beta_start=5.0, beta_end=0.8,
                                       alpha_start=0.5, alpha_end=0.1,
                                       max_cg_it=50, verbose=True, seed=42)
        try_update(a_out, "fp_run1")

        # Run 2: from the result of run 1, with different parameters
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.25, 500.0)
        if t2b > 20:
            a_in2 = norm_sum1(best_seq)
            a_out2, _ = fictitious_play(a_in2, t2b,
                                        beta_start=3.0, beta_end=0.5,
                                        alpha_start=0.4, alpha_end=0.05,
                                        max_cg_it=40, verbose=True, seed=123)
            try_update(a_out2, "fp_run2")

    # === Phase 3: AMD polishing on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        print(f"\n=== Phase 3: AMD polish (n={n_best}, {t3:.0f}s) ===")
        a_out, _ = amd_optimize(best_seq, t3, top_k=60, beta=3.5, c0=3.0,
                                restart_every=4000, seed=77)
        try_update(a_out, "amd_polish")

    # === Phase 4: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        print(f"\n=== Phase 4: Spectral flattening (n={n_best}, {t4:.0f}s) ===")
        a_out, _ = spectral_flatten(best_seq, t4, iters=1000, verbose=False)
        try_update(a_out, "spectral")
        # AMD after spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t4b = min(remaining * 0.1, 80.0)
            if t4b > 10:
                a_out, _ = amd_optimize(best_seq, t4b, top_k=50, beta=3.0,
                                        c0=2.5, restart_every=3000, seed=99)
                try_update(a_out, "amd_after_spec")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Coordinate descent (n={n_best}, {t5:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t5, verbose=False, seed=42)
        try_update(a_out, "coord")

    # === Phase 6: Try another FP run with fresh init ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 20:
        print(f"\n=== Phase 6: FP with diverse init ===")
        # Try a few different structured inits at the best length
        inits_to_try = ["asym_ramp", "two_hump_asym", "hump5"]
        t_per = t6 / len(inits_to_try)
        for kind in inits_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            a_init = make_structured_init(n_best, kind)
            # Blend with current best for a hybrid init
            a_hybrid = 0.5 * norm_sum1(best_seq) + 0.5 * a_init
            a_hybrid = norm_sum1(a_hybrid)
            a_out, _ = fictitious_play(a_hybrid, t_per,
                                       beta_start=4.0, beta_end=0.6,
                                       alpha_start=0.5, alpha_end=0.1,
                                       max_cg_it=40, verbose=False,
                                       seed=np.random.default_rng().integers(10000))
            try_update(a_out, f"fp_{kind}")

    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Length exploration ===")
        for new_n in [n_best + 300, n_best - 300]:
            if new_n < 1000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_long = interpolate_sequence(best_seq, new_n)
            try_update(a_long, f"interp_n{new_n}")
            t7 = min(remaining * 0.4, 80.0)
            if t7 > 10:
                a_out, _ = amd_optimize(a_long, t7, top_k=50, beta=4.0,
                                        c0=3.5, restart_every=3000, seed=555)
                try_update(a_out, f"amd_n{new_n}")

    # === Phase 8: Final AMD with aggressive parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 8: Final AMD ===")
        t8 = min(remaining * 0.6, 60.0)
        if t8 > 10:
            a_out, _ = amd_optimize(best_seq, t8, top_k=80, beta=2.5,
                                    c0=2.0, restart_every=6000, seed=314)
            try_update(a_out, "amd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_structured_init(n_fb, "cos1")
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    return [float(x) for x in best_seq]
