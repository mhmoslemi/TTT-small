Cross-correlation can be computed with FFT:
(h ⊛ w)[k] = IFFT(FFT(h) * conj(FFT(w)))[k]

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed softmax mirror descent with top-K gradient approximation
2. LP-based Frank-Wolfe with line search for high-quality steps
3. Adaptive parameter tuning (mu annealing, constraint selection)
4. Multi-shape initialization with diverse starting points
5. Coordinate descent and peak-pair methods for final polish
"""

import numpy as np
import time
import cvxpy as cp

_solver = None

def get_lp_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    return _solver


def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_u_shape(n, c=2.0):
    """Symmetric U-shape: high at edges, low in middle."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + c * x ** 2
    return normalize_nonneg(a)


def make_parabola(n, power=1.5):
    """Parabolic shape."""
    x = np.linspace(-1, 1, n)
    a = 1.0 - power * x ** 2
    a = np.maximum(a, 0.0)
    return normalize_nonneg(a)


def make_cosine(n, power=1.0):
    """Cosine-shaped (bell)."""
    x = np.linspace(-1, 1, n)
    a = np.cos(np.pi * x / 2) ** (2 * power)
    return normalize_nonneg(a)


def make_w_shape(n, depth=0.5):
    """W-shape: high at edges and center, low at 1/4 and 3/4."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + depth * (4 * x ** 2 - 1) ** 2
    return normalize_nonneg(a)


def make_spectral(n, seed=0, n_modes=30):
    """Random spectral initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.1 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    return normalize_nonneg(a)


def softmax_topk(ac, mu, top_k=50):
    """Compute top-k softmax weights for autocorrelation peaks."""
    n_ac = len(ac)
    k = min(top_k, n_ac)
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    # Numerically stable softmax
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(shifted / mu)
    total = exp_vals.sum()
    if total < 1e-300:
        weights = np.ones(k) / k
    else:
        weights = exp_vals / total
    return top_idx, weights


def smoothed_grad_topk(a, ac, mu, top_k=50):
    """
    Compute approximate smoothed gradient using top-K peaks.
    grad[j] = (1/mu) * sum_{k in top} p[k] * a[k-j] for l=0..min(n-1,k)
    """
    n = len(a)
    top_idx, weights = softmax_topk(ac, mu, top_k)
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        k = int(idx)
        # Contribution: w/mu * [a[k], a[k-1], ..., a[max(0,k-n+1)], 0, ...]
        end = min(k + 1, n)  # number of non-zero entries
        if end == 0:
            continue
        # a[k], a[k-1], ..., a[k-end+1]
        seg = a[k::-1][:end]
        grad[:end] += (w / mu) * seg
    return grad


def smoothed_mirror_descent(a0, time_limit, mu_init=None, top_k=50, 
                             eta_init=1.0, mu_decay=0.9995, 
                             anneal_every=200, rng_seed=42, verbose=False):
    """
    Softmax-smoothed mirror descent with top-K gradient approximation.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    if mu_init is None:
        mu_init = max(0.01 * best_m, 1e-6)
    mu = mu_init
    rng = np.random.default_rng(rng_seed)
    
    t = 0
    stall = 0
    max_stall = anneal_every * 3
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Anneal: decrease mu, restart
            mu *= mu_decay ** max(1, stall // 10)
            # Perturb
            if rng is not None:
                pert = rng.normal(0, 0.01, n) * a
                a = normalize_nonneg(a + pert)
            stall = 0
        
        # Compute gradient
        eta = eta_init / (np.sqrt(t) + 1.0)
        grad = smoothed_grad_topk(a, ac, mu, top_k=top_k)
        
        # Mirror descent update: a_new = exp(log(a) - eta * grad) / sum
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * grad
        # Shift for numerical stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = normalize_nonneg(a_new)
        
        # Periodically decrease mu
        if t % anneal_every == 0:
            mu *= mu_decay
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.0f}s): J={J:.6f}, mu={mu:.6f}, stall={stall}")
    
    return best_a, best_m


def lp_frank_wolfe_line(a0, time_limit, solver, top_K_frac=0.5, min_K=50, 
                         max_K=None, verbose=True, rng_seed=0):
    """
    LP-based Frank-Wolfe with exact line search.
    Uses adaptive constraint selection based on proximity to max.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    if max_K is None:
        max_K = 2 * n - 1
    
    # Adaptive K: start with fewer constraints, increase over time
    K = min(min_K + len(ac) // 10, max_K)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Select binding constraints: those within (1 - eps) of max
        # eps decreases over time (tighter selection)
        eps = max(0.001, 0.1 * np.exp(-t / 500.0))
        threshold = M_curr * (1.0 - eps)
        binding_mask = ac >= threshold
        n_binding = binding_mask.sum()
        
        # Also include at least min_K constraints
        if n_binding < min_K:
            # Take top min_K
            top_idx = np.argsort(ac)[::-1][:min_K]
            binding_mask = np.zeros(len(ac), dtype=bool)
            binding_mask[top_idx] = True
            n_binding = min_K
        
        # Additional top-K selection if too many binding
        if n_binding > K:
            top_idx = np.argsort(ac)[::-1][:K]
            binding_mask = np.zeros(len(ac), dtype=bool)
            binding_mask[top_idx] = True
            n_binding = K
        
        # Build constraint matrix: T_sel @ b <= M_sel
        # T_sel[i, j] = a[j] if (i-j) is a valid lag, i.e., the constraint is:
        # sum_j a[j] * b[i-j] <= M_i (when i-j is in [0, n-1])
        # Actually, the constraint is: (a * b)[i] <= M_i for binding i
        # (a * b)[i] = sum_j a[j] * b[i-j] for valid j
        
        # For the LP, the constraint (a*b)[i] <= M_i becomes:
        # sum_{j: 0<=i-j<n} a[j] * b[i-j] <= M_i
        # = sum_{j: max(0,i-n+1) <= j <= min(n-1, i)} a[j] * b[i-j] <= M_i
        
        binding_idx = np.where(binding_mask)[0]
        n_b = len(binding_idx)
        
        # Build the constraint matrix sparsely
        # For each binding index i, the row is: a[j] for j in [max(0,i-n+1), min(n-1,i)]
        # placed at positions [i-j] for those j
        
        # This is a banded matrix. Let me build it as a dense matrix for cvxpy.
        # For n up to ~3000 and n_b up to ~3000, this is 9M entries which is fine.
        
        T_sel = np.zeros((n_b, n))
        M_sel = np.zeros(n_b)
        
        for row, i in enumerate(binding_idx):
            i_int = int(i)
            j_start = max(0, i_int - n + 1)
            j_end = min(n - 1, i_int)
            for j in range(j_start, j_end + 1):
                T_sel[row, i_int - j] = a[j]
            M_sel[row] = ac[i_int]
        
        # Solve LP: max sum(b) s.t. T_sel @ b <= M_sel, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Line search: find best t in [0, 1]
        # f(t) = max(conv((1-t)*a + t*direction, (1-t)*a + t*direction))
        # We want to minimize f(t)
        # Use golden section or bisection
        
        def obj_at_t(t_val):
            a_t = (1.0 - t_val) * a + t_val * direction
            a_t = np.maximum(a_t, 0.0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                return float('inf')
            a_t = a_t / s_t
            return fast_autocorr(a_t).max()
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate ~0.618
        lo, hi = 0.0, 1.0
        c = hi - gr * (hi - lo)
        d_ = lo + gr * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d_)
        
        for _ in range(25):  # 25 bisections gives precision ~ 1/2^25
            if time.monotonic() - t0 > time_limit:
                break
            if fc < fd:
                hi = d_
                d_ = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = obj_at_t(c)
            else:
                lo = c
                c = d_
                fc = fd
                d_ = lo + gr * (hi - lo)
                fd = obj_at_t(d_)
        
        t_best = (c + d_) / 2.0
        a_new = (1.0 - t_best) * a + t_best * direction
        a_new = normalize_nonneg(a_new)
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Increase K over time
        if t % 20 == 0:
            K = min(K + 20, max_K)
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, n_bind={n_b}")
    
    return best_a, best_m
