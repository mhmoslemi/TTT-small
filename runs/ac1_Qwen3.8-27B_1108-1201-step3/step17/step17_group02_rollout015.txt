import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_grad(a, w, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*w))) w.r.t. a."""
    n = len(a)
    bw = beta * w
    bw_max = bw.max()
    exp_bw = np.exp(bw - bw_max)
    p = exp_bw / exp_bw.sum()
    
    # grad[j] = 2 * sum_{i=0}^{n-1} p[j+i] * a[i]
    # Use FFT cross-correlation
    N = 1 << (3 * n - 2 - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fp = np.fft.rfft(p, N)
    C = np.fft.irfft(np.conj(Fa) * Fp, N)
    grad = 2.0 * C[:n]
    return grad

def lse_optimize(a0, time_limit, beta_start=1.0, beta_end=50.0, 
                 eta=0.1, n_restarts=5, verbose=True, seed=42):
    """LSE-based mirror descent optimization."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_w = fast_autocorr(a)
    best_M = best_w.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for restart in range(n_restarts):
        x = a.copy()
        x_prev = a.copy()
        
        for t in range(1, 1000000):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Adaptive beta
            frac = min(1.0, t / 500.0)
            beta = beta_start + (beta_end - beta_start) * frac
            
            # Nesterov momentum
            mu = 0.9 + 0.09 * min(1.0, t / 100.0)
            y = x + mu * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
            
            # Compute gradient
            w = fast_autocorr(y)
            g = lse_grad(y, w, beta)
            
            # Mirror descent step
            eta_t = eta / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta_t * g
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            
            x_prev = x
            x = a_new
            
            # Check improvement
            w_new = fast_autocorr(x)
            M_new = w_new.max()
            if M_new < best_M:
                best_M = M_new
                best_a = x.copy()
                a = x.copy()
        
        # Perturb for next restart
        if time.monotonic() - t0 > time_limit:
            break
        pert = rng.normal(0, 0.05, n) * best_a
        a = best_a + pert
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return best_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (LSE) gradient with FFT-based computation for smooth max approximation
2. Exponentiated gradient (mirror descent) on the probability simplex
3. Adaptive beta scheduling with Nesterov momentum
4. Multi-restart with diverse initializations (parametric families + parent)
5. Top-k peak gradient for fine polishing
6. Multi-length exploration
7. Coordinate descent for final peak reduction

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_gradient(a, w, beta):
    """Compute gradient of LSE(beta*w) w.r.t. a, where w = autocorr(a).
    grad[j] = 2 * sum_{i=0}^{n-1} p[j+i] * a[i]
    where p = softmax(beta * w)
    """
    n = len(a)
    bw = beta * w
    bw_max = bw.max()
    exp_bw = np.exp(bw - bw_max)
    p = exp_bw / exp_bw.sum()
    
    # Compute cross-correlation: c[j] = sum_{i=0}^{n-1} a[i] * p[j+i]
    # Using FFT: c = ifft(conj(fft(a)) * fft(p))
    N = 1 << (3 * n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fp = np.fft.rfft(p, N)
    C = np.fft.irfft(np.conj(Fa) * Fp, N)
    grad = 2.0 * C[:n]
    return grad


def topk_gradient(a, w, top_k=50, beta_w=5.0):
    """Compute gradient focusing on top-k autocorrelation peaks."""
    n = len(a)
    k_max = min(top_k, len(w))
    top_idx = np.argsort(w)[::-1][:k_max]
    top_vals = np.maximum(w[top_idx], 1e-30)
    weights = top_vals ** beta_w
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k_max) / k_max
    else:
        weights = weights / wsum
    
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        partner = k_int - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad

                       eta=0.15, max_iters=200000, verbose=True, seed=42,

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    w = fast_autocorr(a)
    best_M = w.max()
    J_best = 2.0 * n * best_M
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    momentum = 0.5
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, t / 2000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Nesterov momentum (increasing over time)
        mu = 0.5 + 0.4 * min(1.0, t / 500.0)
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        w = fast_autocorr(y)
        if frac > topk_switch_frac and use_topk_polish:
            # Blend LSE and top-k gradients
            g_lse = lse_gradient(y, w, beta)
            g_top = topk_gradient(y, w, top_k=80, beta_w=6.0)
            blend = (frac - topk_switch_frac) / (1.0 - topk_switch_frac)
            g = (1 - blend * 0.5) * g_lse + blend * 0.5 * g_top
        else:
            g = lse_gradient(y, w, beta)
        
        # Step size (decaying)
        eta_t = eta / (1.0 + 0.3 * np.log1p(t))
        
        # Mirror descent (exponentiated gradient)
        log_a = np.log(y) - eta_t * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Evaluate
        w_new = fast_autocorr(x)
        M_new = w_new.max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            J_best = 2.0 * n * best_M
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            momentum = 0.5
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      LSE-AMD restart t={t} ({elapsed:.0f}s): J={J_best:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-AMD t={t} ({elapsed:.0f}s): J={J_best:.6f}, beta={beta:.1f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      LSE-AMD done ({elapsed:.0f}s): J={J_best:.6f}")
    return best_a, 2.0 * n * best_M


def projected_gradient_descent(a0, time_limit, verbose=False, seed=42):
    """Projected gradient descent with backtracking on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    lr = 0.01
    momentum_buf = np.zeros(n)
    beta_m = 0.9
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        w = fast_autocorr(a)
        beta = 5.0 + 5.0 * min(1.0, t / 1000.0)
        g = lse_gradient(a, w, beta)
        
        momentum_buf = beta_m * momentum_buf + (1 - beta_m) * g
        a_new = a - lr * momentum_buf
        a_new = project_simplex(a_new)
        
        w_new = fast_autocorr(a_new)
        M_new = w_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.1)
        else:
            a = a_new  # still accept to keep moving
            lr = lr * 0.95
            if lr < 1e-5:
                lr = 1e-3
                pert = rng.normal(0, 0.01, n) * a
                a = np.maximum(a + pert, 0)
                a = a / a.sum()
    
    return best_a, 2.0 * n * best_M


def coordinate_peak_reduction(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent targeting the highest autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        w = fast_autocorr(a)
        top_k = np.argsort(w)[::-1][:10]
        improved = False
        
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            
            # Find best destination: minimize increase in autocorrelation
            # Use a simple heuristic: move to position with low "energy"
            energy = a ** 2
            candidates = np.argsort(energy)[:n // 2]
            candidates = [c for c in candidates if c != src]
            if not candidates:
                candidates = list(range(n))
            
            for frac in [0.01, 0.02, 0.005, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                for dst in candidates[:5]:
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-16:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if not improved and stall > 30:
                dst = int(rng.integers(n))
                if dst != src:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, 2.0 * n * best_M


def make_gaussian_init(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def make_sine_init(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def make_power_init(n, p=1.0):
    x = np.linspace(0.01, 0.99, n)
    a = x ** p + (1 - x) ** p
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def make_modulated_init(n, alpha=0.5):
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search: LSE mirror descent with multi-init, multi-length, and polishing."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p = p / p.sum()
            p_score = evaluate_safe(p)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        best_seq = make_sine_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: LSE optimization on parent length ===
    print(f"\n=== Phase 1: LSE-AMD on n={n0} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    
    if t1 > 60:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(parent / parent.sum())
            labels.append("parent")
        
        inits.append(make_sine_init(n0))
        labels.append("sine")
        
        inits.append(make_gaussian_init(n0, 0.5, 0.35))
        labels.append("gauss")
        
        inits.append(make_power_init(n0, 0.7))
        labels.append("power0.7")
        
        inits.append(make_modulated_init(n0, 0.3))
        labels.append("mod0.3")
        
        inits.append(make_gaussian_init(n0, 0.3, 0.2))
        labels.append("gauss_left")
        
        inits.append(make_gaussian_init(n0, 0.7, 0.2))
        labels.append("gauss_right")
        
        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        inits = unique_inits
        labels = unique_labels
        t_per = max(30.0, t1 / len(inits))
        
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 20:
                break
            print(f"  LSE '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = lse_mirror_descent(init, t_avail, beta_start=2.0, beta_end=60.0,
                                           eta=0.12, max_iters=100000, verbose=False,
                                           seed=100 + idx, use_topk_polish=True,
                                           topk_switch_frac=0.6)
            try_update(a_out, f"lse_{lbl}")
    
    # === Phase 2: Longer sequences ===
    print(f"\n=== Phase 2: Longer sequences ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    
    trial_lengths = [n0 + 500, n0 + 1000, n0 + 1500, 5000]
    trial_lengths = [l for l in trial_lengths if 1000 <= l <= 5500 and l != n0]
    trial_lengths = list(dict.fromkeys(trial_lengths))
    
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_init = interpolate_sequence(best_seq, new_n)
        
        t_avail = min(t2 / len(trial_lengths[:3]), remaining - 80)
        if t_avail < 30:
            break
        a_out, _ = lse_mirror_descent(a_init, t_avail, beta_start=2.0, beta_end=50.0,
                                       eta=0.1, max_iters=80000, verbose=False,
                                       seed=new_n % 997, use_topk_polish=True)
        try_update(a_out, f"lse_n{new_n}")
        
        # Also try a fresh init at this length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_init2 = make_sine_init(new_n)
            t_avail2 = min(60.0, remaining - 80)
            if t_avail2 > 20:
                a_out2, _ = lse_mirror_descent(a_init2, t_avail2, beta_start=3.0, beta_end=40.0,
                                                eta=0.1, max_iters=50000, verbose=False,
                                                seed=new_n + 500)
                try_update(a_out2, f"lse_sine_n{new_n}")
    
    # === Phase 3: Focused LSE on best (higher beta, more iterations) ===
    print(f"\n=== Phase 3: Focused LSE polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    
    if t3 > 30:
        a_in = best_seq / best_seq.sum()
        n_best = len(a_in)
        a_out, _ = lse_mirror_descent(a_in, t3, beta_start=20.0, beta_end=150.0,
                                       eta=0.08, max_iters=150000, verbose=True,
                                       seed=777, use_topk_polish=True, topk_switch_frac=0.3)
        try_update(a_out, "lse_polish")
    
    # === Phase 4: Coordinate peak reduction ===
    print(f"\n=== Phase 4: Coordinate peak reduction (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    
    if t4 > 20:
        a_in = best_seq / best_seq.sum()
        a_out, _ = coordinate_peak_reduction(a_in, t4, verbose=True, seed=456)
        try_update(a_out, "coord")
    
    # === Phase 5: Second LSE round on updated best ===
    print(f"\n=== Phase 5: LSE round 2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    
    if t5 > 20:
        a_in = best_seq / best_seq.sum()
        a_out, _ = lse_mirror_descent(a_in, t5, beta_start=30.0, beta_end=200.0,
                                       eta=0.05, max_iters=100000, verbose=False,
                                       seed=888, use_topk_polish=True, topk_switch_frac=0.5)
        try_update(a_out, "lse_r2")
    
    # === Phase 6: Final coordinate pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 6: Final coordinate pass ===")
        t6 = min(remaining * 0.4, 100.0)
        if t6 > 15:
            a_in = best_seq / best_seq.sum()
            a_out, _ = coordinate_peak_reduction(a_in, t6, verbose=False, seed=999)
            try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_sine_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
