def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_smooth_max_grad(a, t, top_k_weights=300):
    """Compute smooth-max objective gradient for the normalized sequence."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return 0.0, np.zeros_like(a)

    scaled = t * (ac - ac_max)
    scaled = np.maximum(scaled, -50.0)
    exp_scaled = np.exp(scaled)
    soft_max = ac_max + (1.0 / t) * np.log(exp_scaled.sum())
    obj = 2.0 * n * soft_max

    w = exp_scaled / exp_scaled.sum()

    n_top = min(top_k_weights, len(w))
    top_idx = np.argsort(w)[::-1][:n_top]

    grad = np.zeros(n)
    for k_val in top_idx:
        wk = w[k_val]
        if wk < 1e-15:
            continue
        k_int = int(k_val)
        if k_int == 0:
            grad += 2.0 * wk * a
        else:
            i_max = n - k_int
            if i_max > 0:
                grad[0:i_max] += wk * a[k_int:n]
            i_min = k_int
            if i_min < n:
                grad[i_min:n] += wk * a[0:n - k_int]

    grad *= 2.0 * n
    return obj, grad


def smooth_max_lbfgs(a0, time_limit, t_start=50.0, t_end=500.0,
                     n_temps=6, max_iter_total=200, verbose=True):
    """Optimize using smooth-max L-BFGS with decreasing temperature."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(t_start, t_end, n_temps)
    iters_per_temp = max(10, max_iter_total // n_temps)

    for ti, t in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        def objective(x):
            a_norm = np.maximum(x, 1e-15)
            s = a_norm.sum()
            if s < 1e-12:
                return 1e10
            a_norm = a_norm / s
            obj, _ = compute_smooth_max_grad(a_norm, t, top_k_weights=200)
            return obj

        def gradient(x):
            a_norm = np.maximum(x, 1e-15)
            s = a_norm.sum()
            if s < 1e-12:
                return np.zeros_like(x)
            a_norm = a_norm / s
            _, grad = compute_smooth_max_grad(a_norm, t, top_k_weights=200)
            # Chain rule for normalization
            # d(a_norm[i])/d(x[j]) = (delta[i,j]*s - a_norm[i]) / s^2
            # Simplified: just return grad (approximately correct for small steps)
            return grad

        x0 = a.copy()
        try:
            result = scipy_minimize(
                objective, x0, jac=gradient, method='L-BFGS-B',
                bounds=[(1e-10, 1000.0)] * n,
                options={'maxiter': iters_per_temp, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            if result.success or result.fun < objective(x0):
                a_new = norm_sum1(result.x)
                m_new = fast_autocorr(a_new).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            if verbose:
                print(f"    L-BFGS temp={t:.0f} error: {e}")
            continue

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothMax t={t:.0f} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3, dual_check=True):
    """LP Frank-Wolfe with adaptive margins and optional dual infeasibility check."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 5:
            active_idx = np.argsort(margins)[:top_K * 5]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:min(top_K, len(ac))]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        if K == 0:
            break

        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Dual infeasibility check: see if any non-active constraint is violated
        if dual_check and t > 100 and t % 500 == 0:
            ac_g = fast_autocorr(g_dir * b_sum)
            violated = np.where(ac_g > M_curr * (1 + 1e-6))[0]
            if len(violated) > 0:
                # Add violated constraints and re-solve
                new_active = np.union1d(active_idx, violated[:50])
                K2 = len(new_active)
                T_sel2 = np.zeros((K2, n))
                for row, k_int in enumerate(new_active):
                    k_int = int(k_int)
                    i_start = max(0, k_int - n + 1)
                    i_end = min(n, k_int + 1)
                    for i in range(i_start, i_end):
                        T_sel2[row, i] = y[k_int - i]
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T_sel2 @ b_var2 <= M_curr]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                    if prob2.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.maximum(
                            np.asarray(b_var2.value, dtype=np.float64), 0.0)
                        b_sum = b_val.sum()
                        if b_sum > 1e-15:
                            g_dir = b_val / b_sum
                except Exception:
                    pass

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten margin on improvement
            margin_frac = max(margin_frac * 0.85, 1e-5)

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, "
                  f"K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def smart_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-crushing coordinate descent using full autocorrelation pressure."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    top_peaks = 5

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peaks (excluding lag 0 which is sum(a^2))
        peak_threshold = 0.95 * M_curr
        peak_lags = np.where(ac >= peak_threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.argsort(ac)[::-1][:top_peaks]

        # Compute pressure: for each position i, how much does it contribute
        # to the top peaks?
        pressure = np.zeros(n)
        for k_int in peak_lags[:top_peaks]:
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                pressure[i] += a[j]

        # Find high-pressure source and low-pressure destination
        improved = False
        n_top_src = min(20, n)
        top_src = np.argsort(pressure)[::-1][:n_top_src]

        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(src)
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving a fraction to low-pressure positions
            best_new_m = M_curr
            best_move = None

            # Find low-pressure destinations
            low_pressure_mask = pressure < 0.3 * pressure[src]
            if low_pressure_mask.sum() < 10:
                low_pressure_mask = pressure < np.median(pressure)
            low_idx = np.where(low_pressure_mask)[0]

            n_trials = min(40, len(low_idx))
            if n_trials == 0:
                low_idx = np.arange(n)
                n_trials = 20

            trial_idx = rng.choice(low_idx, size=min(n_trials, len(low_idx)),
                                   replace=False)

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for dst in trial_idx:
                    dst = int(dst)
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

                if best_new_m < M_curr * (1 - 1e-6):
                    break

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDesc final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'block':
        a = np.zeros(n)
        m = n // 2
        start = n // 4
        a[start:start + m] = 1.0
        return norm_sum1(a)
    elif family == 'bimodal':
        x = np.linspace(0, 1, n)
        c1, c2 = 0.25, 0.75
        w = 0.15
        f = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
        return norm_sum1(f)
    elif family == 'triangular':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smooth-max L-BFGS optimization ===
    print(f"\n=== Phase 1: Smooth-max L-BFGS (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 270.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_max_lbfgs(a_in, t1, t_start=30.0, t_end=800.0,
                                     n_temps=5, max_iter_total=100, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                           t_global, "smoothmax")

    # === Phase 2: LP Frank-Wolfe (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 400.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   top_K=300, margin_frac=2e-3, dual_check=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                           t_global, "lp_main")

    # === Phase 3: Smart coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smart coord descent (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smart_coordinate_descent(a_in, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                           t_global, "coord1")

    # === Phase 4: Second LP with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   top_K=400, margin_frac=5e-4, dual_check=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                           t_global, "lp_tight")

    # === Phase 5: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent 2 (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = smart_coordinate_descent(a_in, t5, verbose=True, rng_seed=77)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                           t_global, "coord2")

    # === Phase 6: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 200, n_best + 500, n_best + 1000]))
    longer_lengths = [l for l in longer_lengths if l <= 5000 and remaining > 250]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"\n=== Phase 6: Extend to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.25, 200.0)
        if t6a > 30 and new_n <= 4500:
            a_out, _ = lp_frank_wolfe(a_long, t6a, solver, verbose=True,
                                       top_K=250, margin_frac=2e-3,
                                       dual_check=False)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                               t_global, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.30, 100.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = smart_coordinate_descent(a_in, t6b, verbose=True,
                                                    rng_seed=123)
                best_seq, best_score = try_update(a_out, best_seq, best_score,
                                                   t_global, f"coord_n{new_n}")

    # === Phase 7: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Freq-diversify (n={n_best}) ===")
    rng = np.random.default_rng(99)
    t7_per = min(remaining * 0.08, 120.0)
    if t7_per > 40:
        for trial in range(2):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t7_per + 30:
                break
            a_pert = freq_perturb(best_seq, strength=0.01, rng=rng)
            # Quick smooth-max warm start
            a_warm, _ = smooth_max_lbfgs(a_pert, min(t7_per * 0.3, 40.0),
                                          t_start=50.0, t_end=200.0,
                                          n_temps=2, max_iter_total=30,
                                          verbose=False)
            # Then LP
            t_lp = min(remaining * 0.06, 80.0)
            if t_lp > 20 and len(a_warm) <= 4500:
                a_out, _ = lp_frank_wolfe(a_warm, t_lp, solver, verbose=False,
                                           top_K=200, margin_frac=1e-3,
                                           dual_check=False)
                best_seq, best_score = try_update(a_out, best_seq, best_score,
                                                   t_global, f"freq{trial}")

    # === Phase 8: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True,
                                   top_K=500, margin_frac=1e-4,
                                   dual_check=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                           t_global, "lp_final")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.35, 150.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = smart_coordinate_descent(a_in, t9, verbose=True,
                                                rng_seed=555)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                               t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety: ensure we return before budget
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 20:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning now")

    return [float(x) for x in best_seq]
