    """Compute gradient of weighted sum of convolution values.
    
    grad_j = 2 * sum_i q_weights[i+j] * a[i] for j=0..n-1
    This is 2 * cross-correlation of q_weights and a.

    # We want: for each j in 0..n-1: sum_{i=0}^{n-1} q_weights[i+j] * a[i]
    # This is np.correlate(q_weights, a, mode='valid')
    grad = 2.0 * np.correlate(q_weights, a, mode='valid')

def logsumexp_descent(a0, time_limit, tau_start=50.0, tau_end=1.0, n_tau_steps=30, verbose=False):
    """Mirror descent with log-sum-exp smoothing and Nesterov acceleration."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()  # Nesterov auxiliary variable
    x_prev = a.copy()
    rng = np.random.default_rng(12345 + n)
    
    for tau_idx in range(n_tau_steps):
        # Interpolate tau from tau_start to tau_end (geometric)
        frac = tau_idx / max(1, n_tau_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        n_iters = max(100, 20000 // n_tau_steps)
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            # Nesterov extrapolation
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / (2.0 * (it + 1) + 1)
            y = beta * x + (1 - beta) * x_prev
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
            
            # Compute convolution
            q = fast_conv(y)
            
            # Compute softmax weights
            q_max = q.max()
            exp_q = np.exp(tau * (q - q_max))
            p = exp_q / exp_q.sum()
            
            # Compute gradient
            grad = compute_gradient(p, y)
            
            # Mirror descent step with adaptive learning rate
            obj_current = fast_conv(y).max()
            L = tau * obj_current + 1.0  # estimated Lipschitz constant
            eta = 1.0 / (np.sqrt(it + 1.0) * np.sqrt(L))
            
            # Ensure step is not too large
            max_grad = np.abs(grad).max()
            if max_grad > 0:
                eta = min(eta, 0.5 / max_grad)
            
            log_a = np.log(y) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            
            x_prev = x
            x = a_new
            
            # Update best

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            
            if verbose and it % 200 == 0:

                J = 2.0 * n * best_obj

    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Identify active constraints (near max)
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # Build LP: minimize M
    # s.t. for each active k: b[k] + grad_bk . (a' - a) <= M
    #      sum(a') = 1
    #      a' >= 0
    #      |a' - a| <= r (trust region, approximated as a*(1-r) <= a' <= a*(1+r))
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        # grad of b[k] w.r.t. a[j] = 2*a[k-j] if 0<=k-j<n else 0
        # Linearization: b[k](a') ≈ b[k](a) + 2*sum_j a[k-j]*a'[j] - 2*sum_j a[k-j]*a[j]
        # = b[k](a) + 2*(a_rev_k · a') - 2*(a_rev_k · a)
        # where a_rev_k[j] = a[k-j] if 0<=k-j<n else 0
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        idx = np.arange(j_start, j_end + 1)
        coeffs = 2.0 * a[k_int - idx]
        # b[k](a') ≈ b[k](a) + coeffs · a'[idx] - coeffs · a[idx]
        # = b[k](a) + coeffs · a'[idx] - b[k](a) + b[k](a) - coeffs · a[idx]
        # Actually: the linearization of the quadratic form q_k(a) = sum_i a[i]*a[k-i]
        # around point a is: q_k(a) + 2*sum_j a[k-j]*(a'[j] - a[j])
        # = q_k(a) + 2*sum_j a[k-j]*a'[j] - 2*sum_j a[k-j]*a[j]
        # = q_k(a) + 2*sum_j a[k-j]*a'[j] - 2*q_k(a)
        # = 2*sum_j a[k-j]*a'[j] - q_k(a)
        
        # So the constraint is: 2*sum_j a[k-j]*a'[j] - q_k(a) <= M
        # i.e., 2*sum_j a[k-j]*a'[j] <= M + q_k(a)
        linear_val = cp.sum(cp.multiply(coeffs, a_var[idx]))
        constraints.append(linear_val <= M_var + b[k_int])
    
    # Trust region
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_method(a0, time_limit, solver=None, r_init=0.12, verbose=False):
    """Iterative LP Remez method with line search."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_step(a, max_cons=40, r=r, solver=solver)
        
        # Line search
        accepted = False
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            obj_trial = fast_conv(a_trial).max()
            if obj_trial < best_obj - 1e-18:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.25, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.7)
            if no_improve > 30:
                # Perturb to escape
                rng = np.random.default_rng(t * 7 + 3)
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                a[idx] *= rng.uniform(0.5, 2.0, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = 0.1
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"  Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, rng_seed=42):
    """Final coordinate descent polishing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                obj_trial = fast_conv(a_trial).max()
                if obj_trial < best_obj - 1e-17:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_obj

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'skew':
        # Asymmetric: peaked on the left
        a = np.exp(-6.0 * (x - 0.35) ** 2 / 0.15)
    elif kind == 'bimodal':
        a = np.exp(-20.0 * (x - 0.3) ** 2) + np.exp(-20.0 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Wait, I realize there might be an issue with my LP Remez step. Let me double-check the linearization.

The quadratic form for lag k is:
q_k(a) = sum_{i: valid} a[i] * a[k-i]

The gradient at point a:
d/d a[j] q_k(a) = 2 * a[k-j] (if 0 <= k-j < n)

The first-order Taylor approximation:
q_k(a + d) ≈ q_k(a) + sum_j (d/d a[j] q_k(a)) * d[j]
= q_k(a) + 2 * sum_j a[k-j] * d[j]

If we set a' = a + d, then d[j] = a'[j] - a[j]:
q_k(a') ≈ q_k(a) + 2 * sum_j a[k-j] * (a'[j] - a[j])
= q_k(a) + 2 * sum_j a[k-j] * a'[j] - 2 * sum_j a[k-j] * a[j]
= q_k(a) + 2 * sum_j a[k-j] * a'[j] - 2 * q_k(a)
= 2 * sum_j a[k-j] * a'[j] - q_k(a)

So the constraint q_k(a') <= M becomes:
2 * sum_j a[k-j] * a'[j] - q_k(a) <= M
i.e.,
2 * sum_j a[k-j] * a'[j] <= M + q_k(a)
