- Objective: minimize sum_i |delta_i| (or minimize the reduction in C[k*])
- Constraints:

The last constraint is the linearization of C[k] around the current point:
C[k](a + delta) ≈ C[k](a) + 2 * (delta, a_shifted_k)

where a_shifted_k[i] = a[k-i] (valid).

- Constraints:
  - sum_i delta_i = 0
  - a_i + delta_i >= 0

def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(p, a):
    """Compute cross_corr(p,a)[j] = sum_l p[j+l]*a[l] for j=0..n-1 via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AF = np.fft.rfft(a[::-1], fft_size)
    cross_full = np.fft.irfft(P * AF, fft_size)[:size]
    return cross_full[n - 1:]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

                        j_max += 1
                    if a[j_max] > amt:
                        a_trial[j_max] -= amt
                        a_trial = a_trial / a_trial.sum()
                        J_t = compute_J(a_trial)
                        if J_t < best_J - 1e-12:
                            a = a_trial
                            best_J = J_t
                            best_a = a.copy()
                            improvement = True
                continue

            # Ternary search on a[i] in [0, ai + ai] = [0, 2*ai]
            lo = 0.0
            hi = 2.0 * ai

            # Evaluate a few points to narrow down
            candidates = [lo + (hi - lo) * f for f in [0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]]
            best_candidate = ai
            best_candidate_J = np.inf

            rest_sum = a.sum() - ai

            for new_ai in candidates:
                if new_ai < 0 or new_ai > 2 * ai + 1e-10:
                    continue
                if abs(new_ai - ai) < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_ai
                # Redistribute the difference among other coordinates proportionally
                diff = ai - new_ai
                if abs(diff) < 1e-15:
                    continue
                # Scale other coordinates proportionally
                others = a_trial.copy()
                others[i] = 0
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                # We need sum(a_trial) = 1
                # a_trial[i] = new_ai, sum of others = 1 - new_ai
                # Scale others: others * (1 - new_ai) / others_sum
                others = others * (1.0 - new_ai) / others_sum
                a_trial = np.concatenate([a_trial[:i], [new_ai], others[i:]]) if i < n - 1 else np.concatenate([a_trial[:i], [new_ai]])
                # Actually let me just do it properly
                a_trial2 = np.copy(a)
                a_trial2[i] = new_ai
                others_mask = np.ones(n, dtype=bool)
                others_mask[i] = False
                others_sum_curr = a_trial2[others_mask].sum()
                target_others = 1.0 - new_ai
                if others_sum_curr > 1e-15:
                    a_trial2[others_mask] *= target_others / others_sum_curr
                a_trial2 = np.maximum(a_trial2, 0.0)
                s2 = a_trial2.sum()
                if s2 < 1e-15:
                    continue
                a_trial2 = a_trial2 / s2
                J_t = compute_J(a_trial2)
                if J_t < best_candidate_J:
                    best_candidate_J = J_t
                    best_candidate = new_ai

            if best_candidate_J < best_J - 1e-12:
                # Apply the best candidate
                a[i] = best_candidate
                others_mask = np.ones(n, dtype=bool)
                others_mask[i] = False
                others_sum_curr = a[others_mask].sum()
                target_others = 1.0 - best_candidate
                if others_sum_curr > 1e-15:
                    a[others_mask] *= target_others / others_sum_curr
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                best_J = best_candidate_J
                best_a = a.copy()
                improvement = True

        if verbose and pass_count % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordLS pass={pass_count} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def two_point_search(a0, time_limit, verbose=False, seed=55):
    """
    Two-point mass transfer search: pick pair (i,j), find optimal transfer amount.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Pick random pairs with bias toward peak contributors
        C = fast_autocorr(a)
        top_lags = np.argsort(C)[::-1][:5]

        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                    break

            # Choose i (source) and j (destination)
            if rng.random() < 0.5 and len(top_lags) > 0:
                k = int(top_lags[0])
                i = int(rng.integers(max(1, k - n + 1), min(n, k + 1)))
            else:
                i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue

            # Try multiple transfer amounts
            best_amt = 0
            best_trial_J = best_J
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                J_t = compute_J(a_trial)
                if J_t < best_trial_J:
                    best_trial_J = J_t
                    best_amt = amt

            if best_amt > 0:
                a[i] -= best_amt
                a[j] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    2Pt iter ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_symmetric_init(n):
    """Cosine bump initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_asymmetric_init(n):
    """Asymmetric: high at left, low at right."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / 0.3) ** 2) + 0.005
    return norm_sum1(a)


def make_multi_bump_init(n, n_bumps=5):
    """Multiple Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_bumps):
        center = (i + 0.5) / n_bumps
        a += np.exp(-((x - center) / 0.05) ** 2)
    a += 0.001
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
