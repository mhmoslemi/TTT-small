    # This is the cross-correlation of w and a, evaluated at position i.
    # corr(w, a)[i] = sum_j w[i+j] * a[j]
    # = sum_j w_rev[conv_len-1-i-j] * a[j]  where w_rev = w[::-1]
    # Hmm, let me just use the FFT approach directly.
    
    # Simple approach: for the gradient, convolve w with a (with appropriate alignment)
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j in [0, n-1] and i+j in [0, 2n-2]
    # This is the same as: (full_convolution(w_padded, a_flipped))[2n-2-i]... 
    # Let me just do it via the shift-and-multiply approach for clarity and correctness.
    
    # Most efficient: use the fact that grad = 2 * IFFT(FFT(w_padded) * FFT(a_padded_conj))
    # but the alignment is tricky. Let me use a direct approach with FFT.
    
    # Actually, the simplest correct approach:
    # grad[i] = 2 * sum_{j=max(0, i-(2n-2))}^{min(n-1, 2n-2-i)} ... no this is getting complicated.
    
    # Let me use the most direct approach: 
    # The subgradient of max_k conv[k] (or soft approximation) w.r.t. a[i] is:
    # For hard max at k*: grad[i] = 2*a[k*-i] if 0<=k*-i<n else 0
    # For soft-max: grad[i] = 2 * sum_k w[k] * a[k-i] (with a[k-i]=0 if out of range)
    # 
    # This is 2 * (correlation of w with a), computed at each position i.
    # corr[i] = sum_k w[k] * a[k-i] = sum_j w[i+j] * a[j]
    # 
    # Using FFT: corr = ifft(fft(w, N) * conj(fft(a, N))) where N >= 2n-1
    # Wait, I need to be careful. The standard cross-correlation:
    # (w ⋆ a)[i] = sum_k w[k] * conj(a[k-i]) = sum_k w[k] * a[k-i] (for real a)
    # This equals ifft(fft(w) * conj(fft(a))) with zero-padding to length >= len(w)+len(a)-1
    # But our sum is over k from 0 to 2n-2, and a[k-i] is only non-zero for 0<=k-i<n.
    # So effectively: (w ⋆ a)[i] where a is zero-padded to length 2n-1.
    
    N = 1 << (2 * n - 2).bit_length()
    w_fft = np.fft.fft(w, N)
    a_padded = np.zeros(N)
    a_padded[:n] = a
    a_fft = np.fft.fft(a_padded, N)
    corr = np.fft.ifft(w_fft * np.conj(a_fft)).real
    grad = 2.0 * corr[:n]
    
    return grad


def compute_subgradient_topk(a, ac, top_k=80, beta=6.0):
    """Compute subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, verbose=False, rng_seed=42, use_softmax=True):
    """
    Nesterov-accelerated mirror descent with adaptive subgradient.
    Alternates between soft-max (all peaks) and top-k (focused) subgradients.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0
    cur_top_k = 60
    cur_beta = 6.0
    temperature = 0.3  # Start focused on top peak
    
    # Temperature schedule: start low (focused), gradually increase (broader view)
    # Then periodically reset to low for exploitation
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: oscillate between focused and broad
        # Use a periodic schedule with increasing mean
        phase = (t % 2000) / 2000.0
        base_temp = 0.2 + 0.1 * (t / 50000.0)  # slowly increasing base
        temperature = base_temp + 0.3 * np.sin(np.pi * phase) ** 2
        temperature = max(0.1, min(temperature, 5.0))
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        
        # Choose subgradient type based on stall
        if stall_count > 3000 and use_softmax:
            g = compute_soft_subgradient_all(y, ac, temperature)
        else:
            g = compute_subgradient_topk(y, ac, top_k=cur_top_k, beta=cur_beta)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameter updates
        if stall_count > 5000:
            cur_beta = max(cur_beta * 0.85, 2.0)
            cur_top_k = min(cur_top_k + 10, 200)
            
        if stall_count > 8000:
            stall_count = 0
            # Surgical restart: perturb the positions most responsible for the top peak
            ac_best = fast_autocorr(best_a)
            top_peak = int(np.argmax(ac_best))
            s = max(0, top_peak - n + 1)
            e = min(n, top_peak + 1)
            idx = np.arange(s, e)
            partner = top_peak - idx
            contrib = best_a[idx] * best_a[partner]
            # Perturb the highest-contributing positions
            top_contrib = np.argsort(contrib)[::-1][:5]
            for tc in top_contrib:
                pert_val = rng.uniform(0.005, 0.03) * best_a[idx[tc]]
                best_a[idx[tc]] -= pert_val
                # Move to a random low-contribution position
                j = int(rng.integers(n))
                best_a[j] += pert_val
            best_a = np.maximum(best_a, 1e-10)
            best_a = best_a / best_a.sum()
            best_m = fast_autocorr(best_a).max()
            x = best_a.copy()
            x_prev = best_a.copy()
            cur_top_k = min(cur_top_k + 5, 150)
            cur_beta = max(cur_beta - 1.0, 3.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, temp={temperature:.2f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, use_all_constraints=False, top_K=None):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if use_all_constraints:
        K_sel = 2 * n - 1
    elif top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(500, 2 * n - 1)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if use_all_constraints or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Select binding constraints: those closest to M_curr
            # conv[k] = sum_i y[i]*y[k-i]
            # For efficiency, compute only for a subset
            constraint_vals = ac  # We already have the full convolution
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix for selected indices
        # T_sel[i, j] = 1 if j+j' = binding_idx[i] for some valid pair... 
        # Actually, the constraint is: for each selected k, sum_i b[i]*b[k-i] <= M
        # This is QUADRATIC in b, not linear!
        # 
        # Wait - the LP formulation from the paper uses:
        # maximize sum(b) subject to (f * g)[k] <= M for all k, b >= 0
        # where f is the CURRENT sequence and g is the LP variable.
        # This IS linear in g because f is fixed!
        # (f * g)[k] = sum_i f[i] * g[k-i] = (T_k @ g) where T_k is a matrix.
        
        # So we need: for each selected k, (T_k @ b) <= M_curr
        # where (T_k @ b)[k] = sum_i f[i] * b[k-i]
        
        # T_k is a matrix of size (1 x n) where T_k[i] = f[k-i] if 0<=k-i<n else 0
        # Actually it's simpler: (f*b)[k] = sum_{i} f[i]*b[k-i]
        # As a function of b: it's a linear form with coefficients f[k-0], f[k-1], ..., f[k-(n-1)]
        # i.e., the coefficient of b[j] in (f*b)[k] is f[k-j] (if 0<=k-j<n)
        
        # So row k of the constraint matrix is: [f[k], f[k-1], ..., f[k-(n-1)]] (padded with 0s)
        
        f_curr = y  # Use current y as the fixed sequence in the LP
        
        # Build the constraint matrix efficiently
        # For constraint k: coeff[j] = f_curr[k-j] if 0 <= k-j < n else 0
        # This is a "skewed diagonal" of f_curr
        
        n_constraints = len(binding_idx)
        A_mat = np.zeros((n_constraints, n))
        for idx, k in enumerate(binding_idx):
            k_int = int(k)
            # coeff[j] = f_curr[k_int - j] for j in [max(0, k_int-n+1), min(n-1, k_int)]
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                A_mat[idx, j_start:j_end] = f_curr[k_int - (np.arange(j_start, j_end))][::-1]
                # Wait, let me be more careful.
                # coeff[j] = f_curr[k_int - j]
                # For j = j_start: coeff = f_curr[k_int - j_start]
                # For j = j_end-1: coeff = f_curr[k_int - (j_end-1)]
                # So the coefficients are f_curr[k_int - j_start], f_curr[k_int - (j_start+1)], ...
                # = f_curr[k_int - j_start : k_int - j_start + (j_end-j_start)][::-1]
                # = f_curr[k_int - (j_end-1) : k_int - j_start + 1][::-1]
                # Hmm, let me just do it element by element for correctness.
                pass
            # Redo:
            for j in range(j_start, j_end):
                A_mat[idx, j] = f_curr[k_int - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A_mat @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

        # Adaptively increase K
        if not use_all_constraints and t % 30 == 0:
            K_sel = min(K_sel + 50, 2 * n - 1)

    return best_a, best_m


def pair_exchange_search(a0, time_limit, verbose=True, rng_seed=100):
    """
    Novel local search using pair exchanges and targeted mass redistribution.
    Key innovation: try swapping values between positions, not just moving mass.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    n_top = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            if contrib.max() < 1e-20:
                continue

            # Try reducing the top contributing pairs
            top_pairs = np.argsort(contrib)[::-1][:5]
            
            for tp in top_pairs:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[tp]
                src_p = partner[tp]
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Strategy 1: Move mass from src to a random position
                for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    # Try several destinations
                    for _try in range(3):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break

                # Strategy 2: Swap portions of a[src] and a[src_p]
                if not improved and src != src_p:
                    swap_frac = rng.uniform(0.1, 0.5)
                    amt_swap = min(a[src], a[src_p]) * swap_frac
                    a_trial = a.copy()
                    a_trial[src] -= amt_swap
                    a_trial[src_p] += amt_swap
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 10 == 0:
                n_pert = min(10, n)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for pi in pert_idx:
                    frac = rng.uniform(0.005, 0.05)
                    j = int(rng.integers(n))
                    if j != pi and a[pi] > 1e-12:
                        amt = a[pi] * frac
                        a[pi] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairExchange ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PairExchange final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_finetune(a0, time_limit, verbose=True, rng_seed=200):
    """
    Fine-grained coordinate descent: for each peak, try optimal single-coordinate move.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:5]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Try moving small amounts to several candidate destinations
            # Choose destinations with low "peak participation"
            sens = np.zeros(n)
            for kk in top_k:
                kki = int(kk)
                ss = max(0, kki - n + 1)
                ee = min(n, kki + 1)
                ii = np.arange(ss, ee)
                sens[ii] += a[kki - ii]
            
            # Sort by sensitivity (low first)
            low_sens = np.argsort(sens)[:20]
            
            for dst in low_sens:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Small random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    """Generate smooth non-negative sequence via Fourier synthesis."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_weighted_shape(n, alpha=1.5, beta=1.5):
    """Beta-distribution shaped sequence (generalized U-shape/dome)."""
    x = np.linspace(0.001, 0.999, n)
    a = x ** alpha * (1 - x) ** beta
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_ramp(n, power=1.0):
    """Ramp-shaped sequence (asymmetric)."""
    x = np.linspace(0, 1, n)
    a = x ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_two_mode(n, pos1=0.25, pos2=0.75, width=0.15):
    """Two Gaussian modes."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2) + 0.001
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    Combines soft-max AMD, LP-FW, pair-exchange search, and coordinate finetuning.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start AMD with soft-max ===
    print(f"\n=== Phase 1: Multi-start AMD (n={n0}) ===")
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_weighted_shape(n0, alpha=1.2, beta=1.2))
        labels.append("beta_1.2")

        inits.append(make_weighted_shape(n0, alpha=2.0, beta=2.0))
        labels.append("beta_2.0")

        inits.append(make_weighted_shape(n0, alpha=0.8, beta=0.8))
        labels.append("beta_0.8")

        inits.append(make_two_mode(n0, pos1=0.2, pos2=0.8, width=0.2))
        labels.append("2mode")

        inits.append(spectral_init(n0, n_modes=25, seed=42))
        labels.append("spectral")

        inits.append(spectral_init(n0, n_modes=8, seed=7))
        labels.append("spectral_low")

        # Asymmetric initializations (novel)
        inits.append(make_ramp(n0, power=0.5))
        labels.append("ramp05")

        inits.append(make_ramp(n0, power=2.0))
        labels.append("ramp2")

        # Blend parent with structured
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_weighted_shape(n0, 1.5, 1.5)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(25.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - elapsed)
            if t_avail < 20:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, verbose=False,
                                     rng_seed=42 + idx * 13, use_softmax=True)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW on best ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K = min(2000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    use_all_constraints=use_all, top_K=top_K)
        try_update(a_out, "lp_main")

    # === Phase 3: Pair-exchange search ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 3: Pair-exchange search (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_exchange_search(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "pair_exch")

    # === Phase 4: AMD polish with higher top_k ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 4: AMD polish (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 25:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, verbose=False,
                                 rng_seed=200, use_softmax=True)
        try_update(a_out, "amd_polish")

    # === Phase 5: LP-FW round 2 ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 5: LP-FW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    use_all_constraints=use_all, top_K=top_K)
        try_update(a_out, "lp_r2")

    # === Phase 6: Coordinate finetune ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 6: Coordinate finetune (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_finetune(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (current n={n_best}) ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600]:
        new_n = n_best + delta
        if 2000 <= new_n <= 5000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 180.0)
        if t7a > 30:
            a_out, _ = amd_optimize(a_long, t7a, verbose=False,
                                     rng_seed=new_n % 1000, use_softmax=True)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 3500:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 20:
                a_in = norm_sum1(a_out)
                use_all = (new_n <= 2000)
                top_K = min(1500, 2 * new_n - 1)
                a_out2, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                             use_all_constraints=use_all, top_K=top_K)
                try_update(a_out2, f"lp_n{new_n}")

    # === Phase 8: Final LP ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 5000:
        t8 = min(remaining * 0.4, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 2500)
            top_K = min(4000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True,
                                        use_all_constraints=use_all, top_K=top_K)
            try_update(a_out, "lp_final")

    # === Phase 9: Final pair-exchange + coordinate ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = pair_exchange_search(a_in, t9 * 0.5, verbose=False, rng_seed=999)
            try_update(a_out, "pair_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9b = min(remaining * 0.6, 50.0)
            if t9b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_finetune(a_in, t9b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
