    n = len(a)
    b = fast_conv(a)
    # Softmax smooth
    b_shifted = b / tau
    b_max = b_shifted.max()
    exp_b = np.exp(b_shifted - b_max)
    weights = exp_b / exp_b.sum()
    smooth_max = tau * (np.log(exp_b.sum()) + b_max)

            idx = np.arange(j_start, j_end + 1)
            g[idx] += weights[k] * 2.0 * a[k_int - idx]

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_max_and_grad(a, tau):
    """Compute softmax-smoothed max of conv(a,a) and its gradient w.r.t. a.
    
    Returns (smooth_max_value, gradient_array).
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Softmax
    b_scaled = b / tau
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    Z = exp_b.sum()
    w = exp_b / Z  # softmax weights
    smooth_max = tau * (np.log(Z) + b_max)
    
    # Gradient: g[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # = 2 * (a_rev * w)[n-1 : 2n-1] where a_rev = a[::-1]
    a_rev = a[::-1]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_full = np.fft.irfft(A_rev * W, fft_size)[:n + len(w) - 1]
    g = 2.0 * conv_full[n-1 : 2*n-1]
    
    return smooth_max, g

        return np.ones(n) / n

    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)

def softmax_grad_fft(a, tau):
    """Compute softmax-smoothed max of conv(a,a) and full gradient using FFT.
    
    Returns (smooth_max, gradient).
    """
    n = len(a)
    b = fast_conv(a)

    b_max_val = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max_val)
    Z = exp_b.sum()
    w = exp_b / Z

    a_rev = a[::-1]

    g = 2.0 * conv_full[n - 1:2 * n - 1]
    
    return smooth_max, g

        constraints = [cp.sum(a_var) == 1]
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
        constraints.append(a_var <= a * (1.0 + r))
        constraints.append(a_var >= a * (1.0 - r))
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        a_new_raw = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_val = a_new_raw.sum()
        if s_val < 1e-15:
            continue
        a_new_raw = a_new_raw / s_val
        accepted = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1.0 - frac) * a + frac * a_new_raw
            a_trial = np.maximum(a_trial, 0.0)
            s2 = a_trial.sum()
            if s2 < 1e-15:
                continue
            a_trial = a_trial / s2
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                r = min(0.25, r * 1.1)
                no_improve = 0
                break
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.65)
            if no_improve > 12:
                idx = rng.choice(n, size=min(n // 4, 400), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = norm_sum1(a)
                no_improve = 0
                r = 0.06
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    return best_a, best_m

def cd_polish(a0, time_limit, seed=0):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed + n)
    stall = 0
    max_stall = max(30, n // 80)
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(80, n // 4)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.005, 0.01, 0.03, 0.06, 0.12, 0.2]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_a, best_m

def spectral_reshape(a0, n_peaks=5, strength=0.3, seed=0):
    n = len(a0)
    a = norm_sum1(a0).copy()
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    B_mag = np.abs(A) ** 2
    b_lin = np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(a, fft_size), fft_size)[:2 * n - 1]
    peak_locs = np.argpartition(b_lin, -n_peaks)[-n_peaks:]
    freq_contrib = np.zeros(len(A))
    for k in peak_locs:
        k_int = int(k)
        freq_contrib += np.abs(np.fft.rfft(np.zeros(fft_size)))**2  # placeholder
    target = B_mag.copy()
    top_freqs = np.argsort(B_mag)[::-1][:min(20, len(B_mag))]
    for f_idx in top_freqs:
        if f_idx == 0:
            continue
        target[f_idx] *= (1.0 - strength * 0.1)
    new_mag = np.sqrt(target)
    A_new = new_mag * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bimodal':
        a = np.exp(-50.0 * (x - 0.25) ** 2) + np.exp(-50.0 * (x - 0.75) ** 2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to(a0, new_n):
    old_n = len(a0)
    if old_n == new_n:
        return a0.copy()
    s = a0.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = a0 / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = 2400
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Softmax PG from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Softmax PG n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = softmax_pg(norm_sum1(best_seq), t1, tau0=None, verbose=True, seed=1)
        update_best(a_out, f"SPG_parent_n{n0}")

    # === Phase 2: Remez polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_lp(norm_sum1(best_seq), t2, solver=solver, verbose=True, seed=2)
        update_best(a_out, f"Remez_n{len(best_seq)}")

    # === Phase 3: Softmax PG with different params (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: SPG variant n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        b_init = fast_conv(norm_sum1(best_seq))
        tau3 = b_init.max() * 0.15
        a_out, _ = softmax_pg(norm_sum1(best_seq), t3, tau0=tau3, verbose=True, seed=100)
        update_best(a_out, f"SPG_var_n{len(best_seq)}")

    # === Phase 4: Multi-init exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4_total = min(remaining * 0.10, 170.0)
    if t4_total > 60:
        print(f"\n=== Phase 4: Multi-init (budget={t4_total:.0f}s) ===")
        inits = ['raised_cos', 'gaussian', 'bimodal']
        t_per = t4_total / len(inits)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = make_init(n0, kind)
            a_out, _ = softmax_pg(a_init, min(t_per, remaining - 5), tau0=None, verbose=False, seed=50)
            update_best(a_out, f"SPG_{kind}_n{n0}")

    # === Phase 5: Alternate lengths (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5_total = min(remaining * 0.15, 250.0)
    if t5_total > 60:
        print(f"\n=== Phase 5: Alt lengths (budget={t5_total:.0f}s) ===")
        alt_lengths = [n0 - 200, n0 + 200, n0 + 400]
        alt_lengths = [x for x in alt_lengths if 1000 <= x <= 4000]
        t_per_len = t5_total / max(1, len(alt_lengths))
        for n_alt in alt_lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_alt = interpolate_to(best_seq, n_alt)
            t_here = min(t_per_len, remaining - 5)
            if t_here > 15:
                a_out, _ = softmax_pg(a_alt, t_here, tau0=None, verbose=False, seed=n_alt)
                update_best(a_out, f"SPG_n{n_alt}")

    # === Phase 6: Remez + CD final polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6a = min(remaining * 0.06, 100.0)
    if t6a > 15:
        print(f"\n=== Phase 6a: Remez final (budget={t6a:.0f}s) ===")
        a_out, _ = remez_lp(norm_sum1(best_seq), t6a, solver=solver, verbose=False, seed=7)
        update_best(a_out, f"Remez_final_n{len(best_seq)}")

    remaining = time_budget - (time.monotonic() - t_global)
    t6b = min(remaining * 0.04, 60.0)
    if t6b > 10:
        print(f"\n=== Phase 6b: CD polish (budget={t6b:.0f}s) ===")
        a_out, _ = cd_polish(norm_sum1(best_seq), t6b, seed=999)
        update_best(a_out, f"CD_n{len(best_seq)}")

    # === Phase 7: Final push with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final push (budget={remaining:.0f}s) ===")
        t7a = min(remaining * 0.6, remaining - 10)
        if t7a > 10:
            b_best = fast_conv(norm_sum1(best_seq))
            tau7 = b_best.max() * 0.05
            a_out, _ = softmax_pg(norm_sum1(best_seq), t7a, tau0=tau7, verbose=True, seed=200)
            update_best(a_out, f"SPG_final_n{len(best_seq)}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = remez_lp(norm_sum1(best_seq), remaining - 5, solver=solver, verbose=False, seed=300)
            update_best(a_out, f"Remez_last_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
