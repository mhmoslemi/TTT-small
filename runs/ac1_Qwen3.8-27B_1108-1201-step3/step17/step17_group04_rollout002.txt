So each iteration is about $O(n \log n) \approx 4000 \times 12 \approx 50000$ operations. In Python with numpy, this might take about 1-5 ms per iteration. So we can do maybe 200K-1M iterations in 1800 seconds.

For LP-FW, each iteration involves solving an LP with $n$ variables and up to $2n-1$ constraints. For $n = 4000$, this might take 10-100 ms per LP solve (depending on the solver). So we can do maybe 10K-100K LP iterations.

Let me now write the code:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max (log-sum-exp) subgradient descent with Nesterov acceleration
2. FFT-based gradient computation for O(n log n) iterations
3. LP Frank-Wolfe with full constraint set
4. Adaptive temperature annealing
5. Local peak-reduction coordinate moves
6. Multi-length exploration with spectral initialization
"""

import numpy as np
import time
import cvxpy as cp

# --- Global solver ---
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_gradient(a, ac, tau):
    """
    Compute gradient of log-sum-exp approximation of max(ac).
    Returns the subgradient of (1/tau)*log(sum exp(tau*ac)).
    
    The gradient w.r.t. a is: 2 * sum_k w_k * (T_k a)
    where w_k = softmax(tau * ac) and (T_k a)_i = a_{k-i} (if valid).
    
    This equals 2 * c_i where c_i = sum_j w_{i+j} * a_j
    which is a cross-correlation computable via FFT.
    """
    n = len(a)
    # Compute softmax weights
    ac_shifted = ac - ac.max()  # for numerical stability
    exp_ac = np.exp(tau * ac_shifted)
    w = exp_ac / exp_ac.sum()
    
    # Compute c_i = sum_j w_{i+j} * a_j for i = 0, ..., n-1
    # This is cross-correlation: c = corr(w, a) at specific lags
    # c_i = sum_{j=0}^{n-1} w_{i+j} * a_j
    # Let's use the convolution approach:
    # Reverse a: a_rev[j] = a[n-1-j]
    # Then c_i = sum_j w_{i+j} * a_rev[n-1-j] = (w * a_rev)_{i+n-1}
    a_rev = a[::-1]
    # Convolve w (length 2n-1) with a_rev (length n)
    # Result has length (2n-1) + n - 1 = 3n - 2
    # We want positions i+n-1 for i = 0, ..., n-1, i.e., positions n-1, ..., 2n-2
    conv = np.convolve(w, a_rev, mode='full')
    c = conv[n-1:2*n-1]  # Length n
    
    return 2.0 * c


def softmax_gradient_fft(a, ac, tau):
    """FFT-based version of softmax_gradient for speed."""
    n = len(a)
    ac_shifted = ac - ac.max()
    exp_ac = np.exp(tau * ac_shifted)
    w = exp_ac / exp_ac.sum()
    
    # We need c_i = sum_j w_{i+j} * a_j
    # Use FFT: c = ifft(conj(fft(w)) * fft(a)) at appropriate indices
    # Actually, this is cross-correlation. Let me use the direct formula:
    # c_i = sum_{j=0}^{n-1} w_{i+j} * a_j
    # = (w convolved with reverse(a)) at position i+n-1
    
    a_rev = a[::-1]
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A_rev = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A_rev, fft_size)
    c = conv[n-1:2*n-1]
    
    return 2.0 * c


def nesterov_softmax_sgd(a0, time_limit, verbose=False, 
                          tau0=100.0, tau_min=5.0, tau_decay=0.999,
                          c_step=3.0, seed=42):
    """
    Nesterov-accelerated mirror descent with log-sum-exp soft max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    tau = tau0
    stall = 0
    lower_bound = 1.0 / (2 * n - 1)
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        M = ac.max()
        g = softmax_gradient_fft(y, ac, tau)
        
        # Step size
        gap = M - lower_bound
        if gap < 1e-12:
            break
        eta = c_step * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)
        
        # Mirror descent step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Decay temperature
        tau = max(tau_decay * tau, tau_min)
        
        # Track best
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 2000:
            stall = 0
            sigma = 0.02
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            tau = max(tau * 0.5, tau_min)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMaxSGD t={t} ({elapsed:.0f}s): J={best_J:.6f}, tau={tau:.1f}")
    
    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=None):
    """LP Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if top_K is None:
        top_K = min(2 * n - 1, 2000)
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix (sparse would be better but dense for now)
        # T[k, j] = y[j] if 0 <= k-j < n, else 0
        # For efficiency, build only the selected constraints
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select binding constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        # Build the submatrix for selected constraints
        # T_sel[k, j] = y[j] if 0 <= binding_idx[k] - j < n
        T_sel = np.zeros((K_sel, n))
        for idx_i, k_val in enumerate(binding_idx):
            k_int = int(k_val)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            T_sel[idx_i, j_start:j_end] = y[j_start:j_end]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel), cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_moves(a0, time_limit, verbose=False, seed=77):
    """
    Local coordinate moves: transfer mass from peak-contributing positions
    to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]
        
        # Compute per-position sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii in range(len(idx)):
                cc = a[idx[ii]] * a[j_idx[ii]]
                sensitivity[idx[ii]] += cc
                sensitivity[j_idx[ii]] += cc
        
        improved = False
        n_tries = 50
        
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick source (high sensitivity) and destination (low sensitivity)
            sens_thresh = sensitivity.mean() + 0.5 * sensitivity.std()
            high_sens = np.where(sensitivity > sens_thresh)[0]
            low_sens = np.where(sensitivity < sensitivity.mean())[0]
            
            if len(high_sens) == 0 or len(low_sens) == 0:
                break
            
            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst or a[src] < 1e-10:
                continue
            
            # Try different transfer amounts
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Larger perturbation
            if stall % 20 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordMoves ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J


def make_symmetric_init(n):
    """Cosine-squared initialization (high at edges)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_asymmetric_init(n, skew=0.3):
    """Asymmetric initialization biased to one side."""
    x = np.linspace(0, 1, n)
    # Skewed distribution: more mass on the left
    a = np.exp(-((x - skew) / 0.3) ** 2) + np.exp(-((x - (1-skew)) / 0.5) ** 2) + 0.005
    return norm_sum1(a)


def make_linear_ramp_init(n):
    """Linear ramp from one end."""
    x = np.linspace(0, 1, n)
    a = (1 - x) ** 0.5 + 0.01
    return norm_sum1(a)


def make_bimodal_init(n, center1=0.2, center2=0.7, sigma=0.1):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center1) / sigma) ** 2) + np.exp(-((x - center2) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
