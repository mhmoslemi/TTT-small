    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:

def compute_subgrad(a, beta=10.0, top_k=60):
    """Compute soft subgradient of max(conv(a,a)) using top-k lags."""
    n = len(a)
    ac = fast_autocorr(a)
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k) / k
    else:
        w_top = w_top / w_sum
    w = np.zeros(len(ac))

    w_pad[:len(w)] = w

    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()

            x = x_r
            x_prev = x_r.copy()

    return best_a, best_m


def build_constraint_matrix(a, all_constraints=True, top_K=800):
    """Build the T matrix for LP: T @ b <= M."""
    n = len(a)
    if all_constraints:
        n_c = 2 * n - 1
        T = np.zeros((n_c, n))
        for j in range(n):
            T[j:j + n, j] = a
        return T
    else:
        ac = fast_autocorr(a)
        K = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K]
        T = np.zeros((K, n))
        for col_idx, m in enumerate(top_idx):
            mi = int(m)
            js = max(0, mi - n + 1)
            je = min(n - 1, mi)
            js_arr = np.arange(js, je + 1)
            T[col_idx, js_arr] = a[mi - js_arr]
        return T


def lp_frank_wolfe(a0, time_limit, solver, all_constraints=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        T = build_constraint_matrix(y, all_constraints=all_constraints, top_K=800)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(T.shape[0]), cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx
        if t == 1:
            x_new = g_dir.copy()

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, seed=123):
    """Coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:20]
        improved = False

        for k in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    sens[d] += a[partner]

            mask = np.abs(np.arange(n) - src) > n // 10
            candidates = np.where(mask & (sens < sens.mean()))[0]
            if len(candidates) == 0:
                candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            best_dst = candidates[np.argmin(sens[candidates])]
            if best_dst == src:
                continue

            for frac in [0.02, 0.01, 0.05, 0.005, 0.03, 0.008]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.04)
                    a[i] -= a[i] * frac
                    a[j] += a[i] * frac + a[i] * frac
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, seed=0):
    """Create smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in [1, 2, 3, 4, 5]:
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase optimization combining
    mirror descent, LP-FW, and coordinate descent with adaptive tuning.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-FW on parent (most effective technique) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW Full (n={n0}) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60 and n0 <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, all_constraints=True)
        try_update(a_out, f"lp1_n{n0}")

    # === Phase 2: Mirror descent with multiple inits ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent ===")
    t2 = min(remaining * 0.20, 340.0)
    if t2 > 40:
        inits = [norm_sum1(best_seq)]
        inits.append(make_init(n0, seed=11))
        inits.append(make_init(n0, seed=23))
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_init(n0, seed=5)
            inits.append(norm_sum1(blend))
        t_per = max(20.0, t2 / len(inits))
        for idx, init in enumerate(inits):
            rem = time_budget - 60 - (time.monotonic() - t_global)
            if rem < 15:
                break
            ta = min(t_per, rem)
            a_out, _ = mirror_descent(init, ta, beta_init=10.0, c0=4.0, seed=100 + idx)
            try_update(a_out, f"md_{idx}_n{n0}")

    # === Phase 3: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW Round 2 (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 40 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, all_constraints=True)
        try_update(a_out, f"lp2_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord Descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, seed=77)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration ===")
    for new_n in [n_best + 200, n_best - 200]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        if new_n < 500 or new_n > 5000:
            continue
        print(f"  Trying n={new_n}")
        a_long = interpolate_seq(best_seq, new_n)
        t5a = min(remaining * 0.3, 100.0)
        if t5a > 20:
            a_out, _ = mirror_descent(a_long, t5a, beta_init=8.0, c0=3.0, seed=555)
            try_update(a_out, f"md_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 3500:
            t5b = min(remaining * 0.4, 100.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe(a_long, t5b, solver, all_constraints=False)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 4000:
        t6 = min(remaining * 0.7, 200.0)
        if t6 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, all_constraints=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Coord (n={n_best}) ===")
    if remaining > 15:
        t7 = min(remaining * 0.6, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent(a_in, t7, seed=333)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_init(3200, seed=1)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
