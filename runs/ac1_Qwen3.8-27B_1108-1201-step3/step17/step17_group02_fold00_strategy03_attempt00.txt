                    cand = [max(0,x) for x in cand]
                    if sum(cand) < 0.01:

                    else:
                        cand = cand / sum(cand)
                    
                    # Skip if value exceeds stability threshold
                    error_val = evaluate_sequence(cand)
                    if not best_found or error_val < best_sequence_value:
                        best_sequence_value = error_val
                        best_seq = cand
                        best_found = True
