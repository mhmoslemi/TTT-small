def fast_autocorr(a):

    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_constraint_matrix(y, active_lags):
    """Build constraint matrix T[m,j] = y[m-j] if valid for lag m."""
    n = len(y)
    T = np.zeros((len(active_lags), n))

        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        T[i, js] = y[m - js]
    return T

def make_smooth_init(n, seed=0):
    """Create smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in [1, 2, 3, 4, 5]:
        coeff = rng.normal(0, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    f_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)

import numpy as np

import cvxpy as cp


def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def build_constraint_matrix(y, active_lags):
    """Build constraint matrix T where T[i, j] = y[m-i - j] for lag m."""
    n = len(y)
    T = np.zeros((len(active_lags), n))
    for i in range(len(active_lags)):
        m = int(active_lags[i])
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        T[i, js] = y[m - js]
    return T

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iterations=5000, K_init=800, K_max=2000):

    LP-based Frank-Wolfe with adaptive constraint working set and Nesterov acceleration.
    The working set starts small and grows to include newly violated constraints.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_acc = 0.0

    ac = fast_autocorr(y)
    K_curr = min(K_init, 2 * n - 1)
    active_lags = np.argsort(ac)[::-1][:K_curr].copy()

    for t in range(1, max_iterations + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Evaluate current y
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Update working set: add constraints that are near the peak
        # Use a margin to include "near-active" constraints

        near_active = np.where(ac >= M_curr * (1.0 - margin))[0]
        if len(near_active) > K_max:
            active_lags = near_active[np.argsort(ac[near_active])[::-1][:K_max]].copy()
            K_curr = K_max
        else:
            combined = np.unique(np.concatenate([active_lags, near_active]))
            if len(combined) > K_max:

            K_curr = len(combined)

        n_cons = T.shape[0]
        if n_cons == 0:

        constraints = [T @ b_var <= M_curr * np.ones(n_cons),
                       cp.sum(b_var) == 1]

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):

        b_sum = b_val.sum()
        if b_sum < 1e-15:

        s_direction = b_val / b_sum

        # Nesterov step with proper momentum

        y_fw = np.maximum(y_fw, 0.0)

        if sy < 1e-15:

            y_fw = y_fw / sy

            y_new = y_fw + beta_t * (y_fw - y)
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y_fw.copy()
            else:
                y_new = y_new / sy

        y = y_new

        # Track best (check both y and the FW point)

        if my < best_m:

def compute_subgrad(y, beta):
    """Compute soft-max subgradient of max(conv(y,y)) using FFT."""
    n = len(y)
    ac = fast_autocorr(y)
    M_curr = ac.max()
    shifted = ac - ac.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / w_sum
    # Mask tiny weights
    w[w < 1e-14] = 0.0
    if w.sum() < 1e-15:
        w = np.ones_like(ac) / len(ac)

    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = y[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, M_curr


def mirrored_descent_accelerated(a0, time_limit, verbose=True, beta_init=15.0, beta_final=4.0,
                                  c0=4.0, seed=42, restart_every=4000):
    """
    Nesterov-accelerated mirror descent on the simplex with adaptive soft-max subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1

        if stall > restart_every:
            # Diversity restart
            if rng.random() < 0.5:
                # Gaussian perturbation around best
                scale = max(0.02, 0.08 * (1.0 - iterations / (2.0 * restart_every * 3)))
                pert = rng.normal(0, scale, n) * best_a
                x = best_a + pert
            else:
                # Structured perturbation: shift mass between halves
                half = n // 2
                x = best_a.copy()
                amt = 0.05 * x[half:].sum()
                x[half:] -= amt / (n - half)
                x[:half] += amt / half
                x = np.maximum(x, 1e-10)
                x = x / x.sum()
            x_prev = x.copy()
            stall = 0
            continue

        # Nesterov momentum
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Adaptive beta for subgradient
        progress = min(1.0, iterations / 8000.0)
        cur_beta = beta_init + (beta_final - beta_init) * progress

        # Compute subgradient
        g, M_curr = compute_subgrad(y, cur_beta)

        # Mirror descent step (entropic regularization on simplex)
        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x.copy()
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if verbose and iterations % 2000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD iter={iterations} ({time.monotonic()-t0:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, stall={stall}")

    return best_a, best_m


def smart_coordinate_descent(a0, time_limit, verbose=True, seed=123, max_stall=800):
    """
    Coordinate descent with sensitivity-guided moves and line search.
    Focuses on coordinates contributing most to peak lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    moves = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peak lags
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improvement = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)

            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue

            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute sensitivity for each destination
            sens = np.zeros(n)
            for k2 in top_lags[:8]:
                k2 = int(k2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                for d in range(s2, e2 + 1):
                    partner = k2 - d
                    sens[d] += a[partner]

            # Score: prefer low sensitivity and far distance
            dist = np.abs(np.arange(n) - src).astype(float)
            scores = sens / (1.0 + 10.0 * dist / n)
            scores[src] = np.inf
            best_dst = int(np.argmin(scores))

            if best_dst == src:
                continue

            # Line search: try multiple fractions
            best_trial_m = best_m
            best_frac = None
            fractions = [0.02, 0.01, 0.03, 0.005, 0.05, 0.015, 0.04, 0.008, 0.025, 0.06]
            for frac in fractions:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_m:
                    best_trial_m = M_trial
                    best_frac = frac

            if best_frac is not None and best_trial_m < best_m - 1e-16:
                amt = src_val * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improvement = True
                stall = 0
                moves += 1

        if not improvement:
            stall += 1
            # Escape local minimum with random moves
            if stall % 15 == 0:
                for _ in range(8):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.06)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improvement = True
                            break

        if verbose and stall % 100 == 0:
            J = 2.0 * n * best_m
            print(f"    Coord ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}, moves={moves}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    Coord final ({time.monotonic()-t0:.0f}s): J={J:.6f}, moves={moves}")

    return best_a, best_m


def make_smooth_init(n, seed=0):
    """Create smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in [1, 2, 3, 4, 5, 6]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    f_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gradient_polish(a0, time_limit, verbose=True, seed=0, max_iters=50000):
    """
    Fast gradient-based polish using projected gradient descent with
    backtracking line search. Works well as a final refinement step.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.02  # initial step size
    stall = 0

    for iteration in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            # Reduce step size and perturb
            eta *= 0.5
            pert = rng.normal(0, 0.01, n) * a
            a = a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            continue

        # Compute subgradient with high beta (focus on peak)
        beta = 20.0
        g, M_curr = compute_subgrad(a, beta)

        # Projected gradient step
        a_trial = a - eta * g
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            eta *= 0.5
            continue
        a_trial = a_trial / s_t

        M_trial = fast_autocorr(a_trial).max()

        if M_trial < best_m - 1e-16:
            a = a_trial
            best_m = M_trial
            best_a = a.copy()
            stall = 0
            # Increase step size slightly
            eta = min(eta * 1.05, 0.1)
        else:
            stall += 1
            eta *= 0.95

        # Occasionally try a different beta
        if iteration % 1000 == 0:
            beta2 = 8.0
            g2, _ = compute_subgrad(a, beta2)
            a_trial2 = a - eta * g2
            a_trial2 = np.maximum(a_trial2, 0.0)
            s2 = a_trial2.sum()
            if s2 > 1e-15:
                a_trial2 = a_trial2 / s2
                M2 = fast_autocorr(a_trial2).max()
                if M2 < best_m - 1e-16:
                    a = a_trial2
                    best_m = M2
                    best_a = a.copy()
                    stall = 0

        if verbose and iteration % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    GradPolish iter={iteration} ({time.monotonic()-t0:.0f}s): J={J:.6f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization combining LP-FW, mirror descent, coordinate descent,
    and gradient polish. Uses adaptive constraint sets and Nesterov acceleration.
    """
    t0 = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_smooth_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP-FW with adaptive constraints (35% time) ===
    print(f"\n=== Phase 1: LP-FW Adaptive ===")
    remaining = time_budget - (time.monotonic() - t0)
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True,
                                    max_iterations=5000, K_init=600, K_max=1500)
        try_update(a_out, "lp_fw_p1")

    # === Phase 2: Accelerated Mirror Descent (20% time) ===
    print(f"\n=== Phase 2: MD Accelerated ===")
    remaining = time_budget - (time.monotonic() - t0)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirrored_descent_accelerated(a_in, t2, verbose=True,
                                                  beta_init=15.0, beta_final=4.0,
                                                  c0=4.0, seed=42, restart_every=3000)
        try_update(a_out, "md_p2")

    # === Phase 3: Gradient Polish (15% time) ===
    print(f"\n=== Phase 3: Gradient Polish ===")
    remaining = time_budget - (time.monotonic() - t0)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_polish(a_in, t3, verbose=True, seed=7)
        try_update(a_out, "gradpolish_p3")

    # === Phase 4: Smart Coordinate Descent (15% time) ===
    print(f"\n=== Phase 4: Coord Descent ===")
    remaining = time_budget - (time.monotonic() - t0)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = smart_coordinate_descent(a_in, t4, verbose=True, seed=42, max_stall=1000)
        try_update(a_out, "coord_p4")

    # === Phase 5: Second LP-FW round (10% time) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW Round 2 ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    max_iterations=4000, K_init=800, K_max=1800)
        try_update(a_out, "lp_fw_p5")

    # === Phase 6: Final gradient polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 6: Final Polish ===")
    if remaining > 15:
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = gradient_polish(a_in, t6, verbose=True, seed=999)
            try_update(a_out, "gradpolish_final")

    # === Phase 7: Final coord if time permits ===
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 20:
        print(f"\n=== Phase 7: Final Coord ===")
        t7 = min(remaining * 0.6, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = smart_coordinate_descent(a_in, t7, verbose=True, seed=555, max_stall=500)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t0:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_smooth_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
