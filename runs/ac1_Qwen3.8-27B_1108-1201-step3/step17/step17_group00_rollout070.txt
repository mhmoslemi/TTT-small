- solve_lp_oracle(y, solver, max_K=None):
    C=build_conv_csr(y, max_K)
    if C is None: return None
    n=len(y); g=cp.Variable(n, nonneg=True); M=cp.Variable()
    constraints=[cp.sum(g)==1, C@g <= M]
    # Maybe add M >= C.min? no.
    prob=cp.Problem(cp.Minimize(M), constraints)
    try: prob.solve(solver=solver, verbose=False, max_time=20)
    except Exception: return None
    if prob.status not in ('optimal','optimal_inaccurate'): return None
    if g.value is None: return None
    gv=np.asarray(g.value,dtype=np.float64).ravel()
    gv=np.nan_to_num(gv, nan=0.0, posinf=0.0, neginf=0.0)
    gv=np.maximum(gv,0)
    gs=gv.sum()
    if gs < 1e-15: return None

```
   try:
```

```
      return None
   if prob.status not in ('optimal','optimal_inaccurate'): return None
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))**2.

Strategy:
- Start from the provided parent sequence when available.
- Use a fast surrogate objective based on FFT autocorrelation for structured searches.
- Main solver: Frank-Wolfe where the linearizing direction is obtained from a sparse LP oracle:
  min M subject to sum(g)=1, g>=0, (y*g)_k <= M for a large subset of lag constraints.
- Use Nesterov acceleration in the Frank-Wolfe iterate.
- Polish with coordinate descent and smooth-perturbation search.
- Explore slightly longer sequences by interpolation and re-optimization.
"""

import numpy as np
import time
from scipy import sparse

try:
    import cvxpy as cp
    HAVE_CP = True
except Exception:
    cvxpy = None
    HAVE_CP = False

_conv_cache = {}
_solver = None
_best_seq = None
_best_score = float("inf")
_t_global = 0.0


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    if not HAVE_CP:
        return None

    names = ["CLARABEL", "MOSEK", "PDLP", "SCS", "ECOS"]
    for name in names:
        s = getattr(cp, name, None)
        if s is None:
            continue
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _solver = s
                print(f"  LP solver selected: {name}")
                return s
        except Exception:
            pass

    s = cp.ECOS
    _solver = s
    print("  LP solver selected: ECOS fallback")
    return s


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def local_score(a):
    a = norm_sum1(a)
    n = a.size
    if n == 0:
        return float("inf")
    m = fast_autocorr(a).max()
    return float(2.0 * n * m)


def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        seq = [float(x) for x in arr]
        val = float(evaluate_sequence(seq))
        return val
    except Exception:
        return float("inf")


def check_update(a, label):

    if arr.size == 0:

def build_conv_csr(y, max_K=None):
    n = len(y)
    size = 2 * n - 1
    if max_K is None or max_K >= size:
        if n <= 2000 and n in _conv_cache:
            rows, cols = _conv_cache[n]
            vals = np.asarray(y[rows - cols], dtype=np.float64)
            return sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))

        rows_parts = []
        cols_parts = []
        for k in range(size):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                rows_parts.append(np.full(i_end - i_start, k, dtype=np.int32))
                cols_parts.append(np.arange(i_start, i_end, dtype=np.int32))

        if not rows_parts:
            return None

        rows = np.concatenate(rows_parts)
        cols = np.concatenate(cols_parts)

        if n <= 2000 and len(_conv_cache) < 3:
            _conv_cache[n] = (rows, cols)

        vals = np.asarray(y[rows - cols], dtype=np.float64)
        return sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))
    else:
        ac = fast_autocorr(y)
        if ac.size <= max_K:
            k_list = np.arange(ac.size)
        else:
            k_list = np.argsort(ac)[::-1][:max_K]

        rows_parts = []
        cols_parts = []
        for kk in k_list:
            k = int(kk)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                rows_parts.append(np.full(i_end - i_start, k, dtype=np.int32))
                cols_parts.append(np.arange(i_start, i_end, dtype=np.int32))

        if not rows_parts:
            return None

        rows = np.concatenate(rows_parts)
        cols = np.concatenate(cols_parts)
        vals = np.asarray(y[rows - cols], dtype=np.float64)
        return sparse.csr_matrix((vals, (rows, cols)), shape=(len(k_list), n))


def solve_lp_oracle(y, solver, max_K=None):
    if not HAVE_CP or solver is None:
        return None

    C = build_conv_csr(y, max_K)
    if C is None:
        return None
    n = len(y)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    constraints = [
        cp.sum(g) == 1,
        C @ g <= M,
        M >= 0
    ]

    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    if g.value is None:
        return None

    gv = np.asarray(g.value, dtype=np.float64).ravel()
    gv = np.nan_to_num(gv, nan=0.0, posinf=0.0, neginf=0.0)
    gv = np.maximum(gv, 0.0)
    gs = gv.sum()
    if gs <= 1e-15:
        return None
    return gv / gs


def random_smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    spec = np.fft.rfft(rng.standard_normal(n))
    freqs = np.fft.rfftfreq(n)
    spec = spec * np.exp(-6.0 * freqs)
    a = np.fft.irfft(spec, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def add_smooth_perturb(a, amount, seed):
    a = norm_sum1(a)
    n = a.size
    rng = np.random.default_rng(seed)
    spec = np.fft.rfft(rng.standard_normal(n))
    freqs = np.fft.rfftfreq(n)
    spec = spec * np.exp(-6.0 * freqs)
    p = np.fft.irfft(spec, n)
    a = a + amount * p
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    f = norm_sum1(f)
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return norm_sum1(g)


def parametric_init(n, time_limit, start_a, start_score):
    t0 = time.monotonic()
    best_a = norm_sum1(start_a)
    best_score = min(float(start_score), local_score(best_a))

    i = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    ones = np.ones(n, dtype=np.float64) / n
    base = best_a / best_a.sum()

    # Power and decreasing power shapes.
    for p in [0.2, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = (i + 1.0) ** p
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

        if time.monotonic() - t0 > time_limit:
            break
        a = (n - i) ** p
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

    # Beta-like shapes.
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta in [0.5, 0.8, 1.0, 1.5, 2.0]:
            if time.monotonic() - t0 > time_limit:
                break
            a = np.maximum(i + 1.0, 0.5) ** alpha * np.maximum(n - i, 0.5) ** beta
            sc = local_score(a)
            if sc < best_score:
                best_score = sc
                best_a = norm_sum1(a)

    # Gaussian bumps.
    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for sigma in [0.08, 0.12, 0.18, 0.25, 0.35]:
            if time.monotonic() - t0 > time_limit:
                break
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            sc = local_score(a)
            if sc < best_score:
                best_score = sc
                best_a = norm_sum1(a)

    # Two-bump shapes.
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sigma in [0.05, 0.08, 0.12, 0.18]:
                if time.monotonic() - t0 > time_limit:
                    break
                a = (np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) +
                     np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma)))
                sc = local_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = norm_sum1(a)

    # Sigmoid shapes.
    for c in [0.3, 0.5, 0.7]:
        for s in [3, 5, 10, 20, 40]:
            if time.monotonic() - t0 > time_limit:
                break
            a = 1.0 / (1.0 + np.exp(-s * (x - c)))
            sc = local_score(a)
            if sc < best_score:
                best_score = sc
                best_a = norm_sum1(a)

    # Sine / periodic-ish bumps.
    for f in [1, 2, 3]:
        for p in [1, 2, 3]:
            if time.monotonic() - t0 > time_limit:
                break
            a = np.maximum(np.sin(np.pi * f * x) ** p, 0.01)
            sc = local_score(a)
            if sc < best_score:
                best_score = sc
                best_a = norm_sum1(a)

    # Exponential shapes.
    for r in [1, 2, 4, 8, 15]:
        if time.monotonic() - t0 > time_limit:
            break
        a = np.exp(-r * x)
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

        if time.monotonic() - t0 > time_limit:
            break
        a = np.exp(r * (x - 1.0))
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

    # Sinc-like.
    for f in [0.5, 1.0, 1.5, 2.0]:
        if time.monotonic() - t0 > time_limit:
            break
        a = np.maximum(np.sinc(f * x - 0.5) ** 2, 0.001)
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

    # Blend with current parent / uniform / power.
    power = norm_sum1((i + 1.0) ** 1.0)
    for w in [0.2, 0.35, 0.5, 0.65, 0.8]:
        if time.monotonic() - t0 > time_limit:
            break
        a = w * base + (1.0 - w) * ones
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

        if time.monotonic() - t0 > time_limit:
            break
        a = w * base + (1.0 - w) * power
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

    # Smooth random initializations.
    for seed in [1, 2, 3, 7, 11, 17, 23, 42, 64, 99, 128, 256]:
        if time.monotonic() - t0 > time_limit:
            break
        a = random_smooth_init(n, seed)
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

    # Smooth perturbations of start.
    for amount, seed in [(0.02, 11), (0.05, 23), (0.10, 37), (0.20, 53), (0.35, 71)]:
        if time.monotonic() - t0 > time_limit:
            break
        a = add_smooth_perturb(start_a, amount, seed)
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_a = norm_sum1(a)

    return best_a, best_score


def fw_optimize(a0, time_limit, solver, max_K=None, verbose=True, seed=101):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = float(fast_autocorr(y).max())
    x_acc = y.copy()

    if max_K is None:
        max_K = choose_max_K(n)

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    bad_oracle = 0
    iter_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t == 1:
            y_look = y.copy()
        else:
            beta = (float(t) - 1.0) / (float(t) + 2.0)
            y_look = x_acc + beta * (x_acc - y)
            y_look = np.maximum(y_look, 0.0)
            sy = y_look.sum()
            if sy > 1e-15:
                y_look = y_look / sy
            else:
                y_look = y.copy()

        g = solve_lp_oracle(y_look, solver, max_K)

        if g is None:
            bad_oracle += 1
            if bad_oracle > 10:
                break

            # Fallback: projected negative soft-gradient direction.
            ac = fast_autocorr(y)
            top_k = np.argsort(ac)[::-1][:5]
            grad = np.zeros(n, dtype=np.float64)
            for kk in top_k:
                k = int(kk)
                s = max(0, k - n + 1)
                e = min(n, k + 1)
                if s >= e:
                    continue
                idx = np.arange(s, e)
                grad[idx] += 2.0 * y[k - idx]

            eta = 0.35 / np.sqrt(float(t) + 1.0)
            y_new = y - eta * grad
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new > 1e-15:
                y_new = y_new / s_new
            else:
                y_new = y.copy()

            x_new = y_new.copy()
            M_new = float(fast_autocorr(y_new).max())
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = y_new.copy()
                stall = 0
            else:
                stall += 1

            y = y_new
            x_acc = x_new
            iter_count += 1
            continue

        bad_oracle = 0
        alpha = 2.0 / (float(t) + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = y.copy()

        if t == 1:
            x_new = y_new.copy()
        else:
            beta = (float(t) - 1.0) / (float(t) + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
            else:
                x_new = y_new.copy()

        improved = False
        m1 = float(fast_autocorr(y_new).max())
        if m1 < best_m - 1e-15:
            best_m = m1
            best_a = y_new.copy()
            improved = True

        m2 = float(fast_autocorr(x_new).max())
        if m2 < best_m - 1e-15:
            best_m = m2
            best_a = x_new.copy()
            improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        if stall > 8000:
            perturb = rng.normal(0.0, 0.03, n) * best_a
            y = best_a + perturb
            y = np.maximum(y, 1e-12)
            ss = y.sum()
            if ss > 1e-15:
                y = y / ss
            else:
                y = best_a.copy()
            x_acc = y.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    FW restart at t={t} ({elapsed:.1f}s), local J={2.0*n*best_m:.9f}")
            continue

        y = y_new
        x_acc = x_new
        iter_count += 1

        if verbose and t % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={t} ({elapsed:.1f}s), local J={2.0*n*best_m:.9f}, K={max_K}")

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    FW done ({elapsed:.1f}s), local J={2.0*n*best_m:.9f}, iters={iter_count}")
    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=5, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    round_no = 0

    while time.monotonic() - t0 < time_limit and stall < 600:
        round_no += 1
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:25]
        improved = False

        for pos in range(min(12, len(top_k))):
            if time.monotonic() - t0 > time_limit:
                break

            k = int(top_k[pos])
            s = max(0, k - n + 1)
            e = min(n - 1, k)
            if s > e:
                continue

            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.size == 0 or contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = float(a[src])
            if src_val < 1e-12:
                continue

            scores = np.zeros(n, dtype=np.float64)
            in_peak = np.zeros(n, dtype=bool)
            for kj in top_k[:8]:
                kij = int(kj)
                sj = max(0, kij - n + 1)
                ej = min(n - 1, kij)
                if sj > ej:
                    continue
                idxj = np.arange(sj, ej + 1)
                partner = kij - idxj
                scores[idxj] += a[partner]
                scores[partner] += a[idxj]
                in_peak[sj:ej + 1] = True

            candidates = np.where(~in_peak)[0]
            if candidates.size == 0:
                candidates = np.arange(n)
            else:
                cand_scores = scores[candidates] + 0.25 * a[candidates]
                dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                dst = int(rng.integers(n))

            found = False
            for frac in [0.01, 0.02, 0.04, 0.07, 0.12, 0.20]:
                amt = src_val * frac
                if amt <= 0.0:
                    continue
                trial = a.copy()
                trial[src] -= amt
                trial[dst] += amt
                trial = np.maximum(trial, 0.0)
                st = trial.sum()
                if st <= 1e-15:
                    continue
                trial = trial / st
                mt = float(fast_autocorr(trial).max())
                if mt < best_m - 1e-16:
                    a = trial
                    best_m = mt
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 120:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = float(rng.uniform(0.004, 0.02))
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    sa = a.sum()
                    if sa > 1e-15:
                        a = a / sa
                        mn = float(fast_autocorr(a).max())
                        if mn < best_m:
                            best_m = mn
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.003, 0.025))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                    mn = float(fast_autocorr(a).max())
                    if mn < best_m:
                        best_m = mn
                        best_a = a.copy()
                        stall = 0

        if verbose and round_no % 75 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD round={round_no} ({elapsed:.1f}s), local J={2.0*n*best_m:.9f}, stall={stall}")

    return best_a, best_m


def perturb_search(a0, time_limit, seed=99, verbose=True):
    n = a0.size
    best = norm_sum1(a0)
    best_score = evaluate_safe(best)
    current = best.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        seed2 = int(rng.integers(0, 2**31 - 1))
        p = random_smooth_init(n, seed2)
        w = float(rng.uniform(0.05, 0.65))
        trial = (1.0 - w) * current + w * p
        trial = norm_sum1(trial)
        sc = evaluate_safe(trial)

        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best = trial.copy()
            current = trial.copy()
            stall = 0
        else:
            stall += 1

        if stall > 120:
            seed3 = int(rng.integers(0, 2**31 - 1))
            p = random_smooth_init(n, seed3)
            w = float(rng.uniform(0.2, 0.85))
            current = norm_sum1((1.0 - w) * best + w * p)
            stall = 0

        if verbose and it % 40 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Perturb it={it} ({elapsed:.1f}s), best={best_score:.9f}, stall={stall}")

    return best, best_score


def propose_candidate(budget_s=1800.0):
    global _best_seq, _best_score, _t_global

    t0 = time.monotonic()
    _t_global = t0
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    if time_budget < 120.0:
        time_budget = max(60.0, float(budget_s) - 20.0)

    print(f"Total time budget: {time_budget:.1f}s")
    solver = get_solver()

    # Initialize from parent when available.
    best_seq = None
    best_score = float("inf")

    try:
        parent = globals().get("height_sequence_1", None)
        if parent is not None:
            p = np.asarray(parent, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and p.sum() > 0.01:
                best_seq = norm_sum1(p)
                best_score = evaluate_safe(best_seq)
                print(f"Parent loaded: n={best_seq.size}, score={best_score:.9f}")
    except Exception:
        pass

    if best_seq is None or not np.isfinite(best_score):
        n_fb = 5200
        x = np.linspace(0.0, 1.0, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback init: n={best_seq.size}, score={best_score:.9f}")

    _best_seq = best_seq.copy()
    _best_score = float(best_score)

    local0 = local_score(_best_seq)
    print(f"Initial: local J={local0:.9f}, evaluator score={_best_score:.9f}")

    # Phase 1: parametric initialization search using fast surrogate.
    remaining = time_budget - (time.monotonic() - t0)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10.0:
        print(f"\n=== Phase 1: parametric init, n={len(_best_seq)} ===")
        a_out, _ = parametric_init(len(_best_seq), t1, _best_seq, local0)
        check_update(a_out, "param")

    # Phase 2: main LP-Frank-Wolfe on current best length.
    remaining = time_budget - (time.monotonic() - t0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 2: FW-LP, n={n_cur} ===")
    t2 = min(remaining * 0.45, 720.0)
    if t2 > 30.0:
        a_in = norm_sum1(_best_seq)
        a_out, _ = fw_optimize(a_in, t2, solver, max_K=choose_max_K(n_cur), verbose=True, seed=101)
        check_update(a_out, f"fw_main_{n_cur}")

    # Phase 3: coordinate polish.
    remaining = time_budget - (time.monotonic() - t0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 3: coordinate descent, n={n_cur} ===")
    t3 = min(remaining * 0.12, 160.0)
    if t3 > 20.0:
        a_in = norm_sum1(_best_seq)
        a_out, _ = coordinate_descent(a_in, t3, rng_seed=17, verbose=True)
        check_update(a_out, f"cd_{n_cur}")

    # Phase 4: try longer lengths.
    base_n = len(_best_seq)
    ext_candidates = []
    for dn in [600, 1200, 2000]:
        nn = base_n + dn
        if nn > base_n and nn <= 7500:
            ext_candidates.append(nn)
    ext_candidates = ext_candidates[:2]

    for new_n in ext_candidates:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 220.0:
            break

        print(f"\n=== Phase 4: extend to n={new_n} ===")
        a_long = interpolate_sequence(_best_seq, new_n)
        check_update(a_long, f"interp_{new_n}")

        t4a = min(remaining * 0.45, 240.0)
        if t4a > 25.0:
            a_in = norm_sum1(_best_seq)
            # FW direction at old length then interpolate? Simpler: run FW on interpolated sequence.
            a_out, _ = fw_optimize(a_long, t4a, solver, max_K=choose_max_K(new_n), verbose=False, seed=new_n % 10000)
            check_update(a_out, f"fw_ext_{new_n}")

        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 90.0:
            t4b = min(remaining * 0.25, 90.0)
            if t4b > 15.0:
                a_out, _ = coordinate_descent(a_long, t4b, rng_seed=new_n % 100000, verbose=False)
                check_update(a_out, f"cd_ext_{new_n}")

    # Phase 5: final FW-LP on best sequence.
    remaining = time_budget - (time.monotonic() - t0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 5: final FW-LP, n={n_cur} ===")
    t5 = min(remaining * 0.45, 330.0)
    if t5 > 30.0:
        a_in = norm_sum1(_best_seq)
        a_out, _ = fw_optimize(a_in, t5, solver, max_K=choose_max_K(n_cur), verbose=True, seed=777)
        check_update(a_out, f"fw_final_{n_cur}")

    # Phase 6: smooth perturbation search using the real evaluator.
    remaining = time_budget - (time.monotonic() - t0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 6: evaluator-guided perturbations, n={n_cur} ===")
    t6 = min(remaining * 0.40, 160.0)
    if t6 > 20.0:
        a_out, _ = perturb_search(_best_seq, t6, seed=314, verbose=True)
        check_update(a_out, f"perturb_{n_cur}")

    # Phase 7: final coordinate polish.
    remaining = time_budget - (time.monotonic() - t0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 7: final coordinate polish, n={n_cur} ===")
    t7 = min(remaining * 0.65, 140.0)
    if t7 > 15.0:
        a_in = norm_sum1(_best_seq)
        a_out, _ = coordinate_descent(a_in, t7, rng_seed=2718, verbose=True)
        check_update(a_out, f"cd_final_{n_cur}")

    # Final safety normalization.
    if _best_seq is None:
        n_fb = 5200
        x = np.linspace(0.0, 1.0, n_fb)
        _best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_score = evaluate_safe(_best_seq)

    out = np.asarray(_best_seq, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s < 0.01:
        out = np.ones(1000, dtype=np.float64) / 1000.0
    else:
        out = out / s

    check_update(out, "final_normalized")

    elapsed = time.monotonic() - t0
    final_score = evaluate_safe(_best_seq)
    print(f"\nFINAL: score={final_score:.9f}, n={len(_best_seq)}, time={elapsed:.1f}s")

    return [float(v) for v in out]
```
