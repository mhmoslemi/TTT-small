def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(v):
    """Project v onto probability simplex {x>=0, sum x = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng_arr = np.arange(1, n + 1)
    cond = u * rng_arr > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def grad_lag(a, k):
    """Compute df_k/da for given lag k. Vectorized O(n)."""
    n = len(a)
    g = np.zeros(n)
    k = int(k)
    # Term 1: a[k-j] for j where 0 <= k-j < n => max(0,k-n+1) <= j <= min(n-1,k)
    j1_s = max(0, k - n + 1)
    j1_e = min(n, k + 1)
    if j1_s < j1_e:
        idx = np.arange(j1_s, j1_e)
        g[idx] += a[k - idx]
    # Term 2: a[j-k] for j where 0 <= j-k < n => max(0,k) <= j <= min(n-1,k+n-1)
    j2_s = max(0, k)
    j2_e = min(n, k + n)
    if j2_s < j2_e:
        idx = np.arange(j2_s, j2_e)
        g[idx] += a[idx - k]
    return g

def compute_J_subgrad(a, active_lags, M):
    """Compute subgradient of J = 2n*M/S^2 at point a (sum a = 1)."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        g += grad_lag(a, k)
    g /= len(active_lags)
    return 2.0 * n * (g - 2.0 * M * a)

            if i == j or a[i] < 1e-14:

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.12)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(3.0 * (x - 0.5))
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
