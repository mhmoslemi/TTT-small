```python
from scipy.optimize import minimize

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    x = np.asarray(x, dtype=np.float64)
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def get_grad_a(a, beta):
    """Gradient of smooth max of autocorrelation w.r.t. a."""
    c = fast_autocorr(a)
    bc = beta * c
    bc_shifted = bc - bc.max()
    p = np.exp(bc_shifted)
    p = p / p.sum()
    n = len(a)
```

- Multiple L-BFGS runs with different β schedules.

Let me also think about the LP polish more carefully. The FW LP is:
max sum(b) s.t. T @ b <= M, b >= 0, sum(b) = 1

where M = max(conv(a,a)) at the current point, and T is the matrix with T[k,j] = a[k-j] (valid positions).

The solution b* gives a FW direction. The FW update is:
a_new = (1-α)a + α*b*

The optimal α can be found by 1D line search: minimize max(conv(a_new, a_new)) over α ∈ [0,1].

This is a 1D optimization problem that can be solved efficiently (the function is piecewise quadratic in α).

For the line search, I'll just use scipy's minimize_scalar.

A good schedule might be: [3, 5, 10, 20, 40, 80, 120, 200]

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B optimization in log-space (softmax parameterization)
2. Smooth max (log-sum-exp) objective with beta annealing
3. LP Frank-Wolfe polishing with exact line search
4. Multiple restarts and length sweeping
"""

import numpy as np
import time

# Global best tracking

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def smooth_max_grad_a(a, beta):
    """
    Compute smooth max of autocorrelation and its gradient w.r.t. a.
    Returns (smooth_value, grad_a, true_max).
    """
    c = fast_autocorr(a)
    bc = beta * c
    m = bc.max()
    exp_bc = np.exp(bc - m)
    p = exp_bc / exp_bc.sum()
    smooth_val = (m + np.log(exp_bc.sum())) / beta
    
    n = len(a)
    # grad_a[j] = 2 * sum_k p[k] * a[k-j] (valid a indices)
    # = 2 * cross_correlation(a, p)[j]
    p_rev = p[::-1]
    L = n + len(p) - 1
    fsize = 1 << (L - 1).bit_length()
    cc = np.fft.irfft(np.fft.rfft(a, fsize) * np.fft.rfft(p_rev, fsize), fsize)
    grad_a = 2.0 * cc[len(p) - 1:len(p) - 1 + n]
    
    return smooth_val, grad_a, c.max()

def lbfgs_objective(theta, beta):
    a = softmax(theta)
    val, _, _ = smooth_max_grad_a(a, beta)
    return val

def lbfgs_jacobian(theta, beta):
    a = softmax(theta)
    _, grad_a, _ = smooth_max_grad_a(a, beta)
    mean_g = np.dot(a, grad_a)
    return a * (grad_a - mean_g)

def lbfgs_optimize(a_init, time_limit, beta_schedule, verbose=True, seed=42):
    """
    Run L-BFGS-B with annealed beta schedule.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    theta = np.log(a)
    
    best_score = 2.0 * n * fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for beta in beta_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        try:
            result = minimize(
                fun=lbfgs_objective,
                x0=theta,
                jac=lbfgs_jacobian,
                args=(beta,),
                method='L-BFGS-B',
                options={'maxiter': 300, 'gtol': 1e-10, 'maxfun': 500, 'iprint': -1}
            )
            if result.x is not None:
                a_new = softmax(result.x)
                c_max = fast_autocorr(a_new).max()
                score = 2.0 * n * c_max
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                theta = result.x
                if verbose:
                    print(f"    L-BFGS β={beta:.0f}: J={score:.6f} (time={time.monotonic()-t0:.0f}s)")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS β={beta:.0f}: error {e}")
    
    return best_a, best_score

def lp_fw_step(a, M, top_K=200, solver=None):
    """
    One Frank-Wolfe step: solve LP to find best direction.
    Returns (direction, success).
    """
    n = len(a)
    c = fast_autocorr(a)
    
    if top_K and top_K < 2 * n - 1:
        top_idx = np.argsort(c)[::-1][:top_K]
    else:
        top_idx = np.arange(2 * n - 1)
    
    K = len(top_idx)
    T = np.zeros((K, n))
    for i, m in enumerate(top_idx):
        m_int = int(m)
        j_start = max(0, m_int - n + 1)
        j_end = min(n - 1, m_int)
        js = np.arange(j_start, j_end + 1)
        T[i, js] = a[m_int - js]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(K), cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        if solver is None:
            solver = cp.CLARABEL
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for s in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=s, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, False
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, False
    
    direction = b_val / b_sum
    return direction, True

def fw_line_search(a, direction, M_target):
    """Find optimal alpha for FW step via line search."""
    n = len(a)
    
    def f_alpha(alpha):
        a_try = (1 - alpha) * a + alpha * direction
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            return float('inf')
        a_try = a_try / s
        return fast_autocorr(a_try).max()
    
    result = minimize_scalar(f_alpha, bounds=(0, 1), method='bounded',
                             options={'xatol': 1e-10})
    return result.x, result.fun

def fw_optimize(a_init, time_limit, top_K=200, max_iter=50, verbose=True, solver=None):
    """
    Frank-Wolfe optimization with exact line search.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(a).max()
        
        direction, success = lp_fw_step(a, M_curr, top_K=top_K, solver=solver)
        if not success:
            break
        
        alpha, f_alpha = fw_line_search(a, direction, M_curr)
        a_new = (1 - alpha) * a + alpha * direction
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and (it % 5 == 0 or it == max_iter - 1):
            print(f"    FW iter={it}: alpha={alpha:.4f}, J={2*n*M_new:.6f} (time={time.monotonic()-t0:.0f}s)")
    
    return best_a, 2.0 * n * best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```
