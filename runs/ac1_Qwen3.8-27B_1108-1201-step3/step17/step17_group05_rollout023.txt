```python
"""
Minimizes J = 2*N*max(conv(f,f))/(sum(f))^2 using:
1. Log-sum-exp smoothed mirror descent with adaptive temperature
2. FFT-based O(N log N) gradient computation
3. Multiple diverse initializations and restarts
4. Local coordinate descent refinement
5. Length optimization
"""

import numpy as np
import time

def fast_conv(a, b=None):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_gradient_smooth(f, beta):
    """
    Compute gradient of h_beta(f) = (1/beta)*log(sum exp(beta*conv(f,f)))
    Returns (grad, h_val, max_conv)
    """
    N = len(f)
    conv_f = fast_conv(f, f)  # length 2N-1
    M = conv_f.max()
    
    shifted = conv_f - M
    exp_vals = np.exp(beta * shifted)
    w_sum = exp_vals.sum()
    if w_sum < 1e-30:
        w = np.ones(2*N-1) / (2*N-1)
    else:
        w = exp_vals / w_sum
    
    h_val = M + np.log(w_sum) / beta
    
    # grad[j] = 2 * sum_m f[m] * w[m+j]
    # = 2 * conv(f_rev, w)[j + N - 1]
    f_rev = f[::-1]
    conv_len = N + (2*N-1) - 1  # 3N-2
    fft_size = 1 << (conv_len - 1).bit_length()
    F_rev = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(F_rev * W, fft_size)[:conv_len]
    
    grad = 2.0 * conv_result[N-1 : 2*N-1]
    
    return grad, h_val, M

def mirror_descent(f0, time_limit, beta_init=20.0, beta_end=5.0, 
                   c=3.0, seed=42, verbose=False):
    """
    Mirror descent (exponentiated gradient) on the probability simplex.
    Minimizes the log-sum-exp smoothed objective.
    """
    N = len(f0)
    f = normalize(f0)
    best_f = f.copy()
    best_M = fast_conv(f, f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iteration = 0
    stall_count = 0
    best_stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if iteration % 100000 == 0:
            break
        
        # Adaptive beta: start high, decrease over time
        progress = min(1.0, iteration / 50000.0)
        beta = beta_init - (beta_init - beta_end) * progress
        
        # Compute smoothed gradient
        grad, h_val, M_curr = compute_gradient_smooth(f, beta)
        
        # Mirror descent step: f_new[j] ∝ f[j] * exp(-eta * grad[j])
        eta = c / (np.sqrt(iteration) + 1.0)
        
        log_f = np.log(np.maximum(f, 1e-300))
        log_f_new = log_f - eta * grad
        log_f_new -= log_f_new.max()  # for numerical stability
        f_new = np.exp(log_f_new)
        f_new = np.maximum(f_new, 1e-300)
        f_new = f_new / f_new.sum()
        
        # Evaluate
        M_new = fast_conv(f_new, f_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f_new.copy()
            stall_count = 0
            best_stall = 0
        else:
            stall_count += 1
            best_stall += 1
        
        f = f_new
        
        # Restart if stalled
        if stall_count > 3000:
            stall_count = 0
            # Perturb around best
            scale = rng.uniform(0.01, 0.1)
            pert = rng.normal(0, scale, N) * best_f
            f_r = best_f + pert
            f_r = np.maximum(f_r, 1e-10)
            f_r = f_r / f_r.sum()
            f = f_r
            
            # Occasionally do a more drastic restart
            if rng.random() < 0.2:
                f = normalize(f + rng.normal(0, 0.5, N))
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * N * best_M
                print(f"      MD restart it={iteration} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * N * best_M
            print(f"      MD it={iteration} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_f, best_M
```

```python
def coordinate_descent(f0, time_limit, seed=42, verbose=False):
    """
    Coordinate descent on the simplex, focusing on reducing the peak convolution value.
    """
    N = len(f0)
    f = normalize(f0)
    best_f = f.copy()
    best_M = fast_conv(f, f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        conv_f = fast_conv(f, f)
        M_curr = conv_f.max()
        peak_lag = int(np.argmax(conv_f))
        
        # Find positions contributing to the peak
        js = max(0, peak_lag - N + 1)
        je = min(N-1, peak_lag)
        if js > je:
            stall += 1
            continue
        
        idx = np.arange(js, je+1)
        contrib = f[idx] * f[peak_lag - idx]
        
        # Find the pair with maximum contribution
        max_ci = np.argmax(contrib)
        src = int(idx[max_ci])
        dst = int(peak_lag - src)  # the paired position
        
        # We want to reduce f[src] * f[dst]
        # Option 1: Move mass from src to a different position
        # Option 2: Move mass from dst to a different position
        
        # Compute sensitivity: for each position p, how much would moving mass there
        # affect the peak lags?
        top_lags = np.argsort(conv_f)[::-1][:min(20, 2*N-1)]
        
        # For each candidate destination, compute the impact
        best_improvement = 0
        best_move = None
        
        # Try moving a fraction of mass from src (or dst) to various destinations
        for source in [src, dst]:
            if f[source] < 1e-10:
                continue
            for frac in [0.01, 0.005, 0.02, 0.001, 0.05]:
                amt = f[source] * frac
                if amt < 1e-15:
                    continue
                
                # Try several destinations
                # Prefer positions far from both src and dst, and with low local density
                candidates = []
                for _ in range(30):

                    if d != source:
                        candidates.append(d)

                            best_move = (source, d, amt, f_trial)
        
        if best_move is not None:

            f = f_trial
            M_new = fast_conv(f, f).max()
            if M_new < best_M:
                best_M = M_new
                best_f = f.copy()
            stall = 0

            # Random perturbation

                i = int(rng.integers(N))
                j = int(rng.integers(N))
                if i != j and f[i] > 1e-10:
                    amt = f[i] * rng.uniform(0.005, 0.05)
                    f[i] -= amt
                    f[j] += amt

                    M_new = fast_conv(f, f).max()

            J = 2.0 * N * best_M
            print(f"      Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_f, best_M
```
