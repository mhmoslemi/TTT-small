```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def lp_step(f, M, time_limit=60):
    """Solve LP to find best g such that f*g <= M, g >= 0, maximizing sum(g)."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
```

def lp_best_response(f, verbose=False):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response iterations (solving max sum(g) s.t. f*g <= M, g>=0)
2. Nesterov-accelerated mirror descent with FFT-based subgradients
3. Multi-strategy initialization (power ramps, bimodal, log-ramp, cosine-modulated)
4. Coordinate descent with pressure-based mass transfer
5. Adaptive length sweeping with interpolation
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J_norm(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a, conv, top_k=80, beta_w=10.0):
    """Compute soft subgradient of max(conv) at a using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent(a0, time_limit, top_k=80, beta_w=10.0, c0=4.0,
                   restart_every=6000, verbose=False, seed=12345):
    """Nesterov-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_val = eval_J_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        conv = fast_conv(y)
        g = compute_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        val = eval_J_norm(a_new)

            stall = 0

            stall += 1

def lp_best_response(f, time_limit_lp=30.0):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector (may be scaled by caller).
    """
    n = len(f)
    conv_f = fast_conv(f)
    M = conv_f.max()
    if M < 1e-30:
        return f.copy()

    # Build constraint matrix A (2n-1 x n): A[k,j] = f[k-j] if 0<=k-j<n
    j = np.arange(n)
    offsets = np.arange(n)
    rows = (j[:, None] + offsets[None, :]).ravel()
    cols = np.repeat(j, n)
    vals = np.tile(f, n)
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': time_limit_lp, 'presolve': True})

    if result.success:
        g = np.maximum(result.x, 0.0)
        if g.sum() > 1e-15:
            return g
    return f.copy()

def make_power_ramp(n, p=1.0):

    for freq in range(1, 10):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric_ramp(n, p=1.0, shift=0.3):
    x = np.linspace(0, 1, n)
    a = ((x + shift / (1 - 2 * shift)) ** p) * ((1 - x) ** 0.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_log_ramp(n, base=1.03):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = base ** x
    return a / a.sum()


def make_cosine_ramp(n, p=1.0):
    x = np.linspace(0, 1, n)
    a = (x ** p) * (0.5 + 0.5 * np.cos(np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_inverse_ramp(n, p=1.0):
    """Mass concentrated at the beginning."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-p)
    return a / a.sum()


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """
    Main optimization driver combining LP best-response, mirror descent,
    coordinate descent, and length sweeping.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J_norm(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.8f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J_norm(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is not None:
        best_a = parent.copy()
        best_score = float(eval_J_norm(parent))
        n0 = len(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J_norm(best_a)
        print(f"Initial: n={n0}, J={best_score:.8f}")

    # === Phase 1: Quick MD on diverse inits ===
    print(f"\n=== Phase 1: MD Multi-Init (n={n0}) ===")
    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.7))
    labels.append("ramp0.7")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp1.0")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp1.5")
    inits.append(make_power_ramp(n0, 2.5))
    labels.append("ramp2.5")
    inits.append(make_inverse_ramp(n0, 1.0))
    labels.append("inv_ramp")
    inits.append(make_bimodal(n0))
    labels.append("bimodal")
    inits.append(make_log_ramp(n0, 1.04))
    labels.append("log1.04")
    inits.append(make_cosine_ramp(n0, 1.0))
    labels.append("cos_ramp")
    inits.append(make_asymmetric_ramp(n0, p=1.5, shift=0.15))
    labels.append("asym1.5")

    # Deduplicate
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 50)], 4))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    t_per = max(8.0, time_budget * 0.12 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        remaining = time_budget - 60 - elapsed
        if remaining < 10:
            break
        t_avail = min(t_per, remaining)
        if t_avail < 5:
            break
        print(f"  MD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = mirror_descent(init, t_avail, top_k=80, beta_w=10.0, c0=4.0,
                                  restart_every=5000, seed=100 + idx)
        try_update(a_out, f"md_{lbl}")

    # === Phase 2: LP-based refinement ===
    print(f"\n=== Phase 2: LP Best-Response (n={len(best_a)}) ===")
    remaining = time_budget - 60 - (time.monotonic() - t_global)
    n_cur = len(best_a)
    lp_algebra = n_cur <= 3500  # LP feasible for moderate n

    if lp_algebra and remaining > 60:
        a_work = best_a.copy()
        lp_best_val = eval_J_norm(a_work)
        t0_lp = time.monotonic()
        lp_time_limit = min(remaining * 0.38, 550.0)
        lp_it = 0
        lp_stall = 0

        while time.monotonic() - t0_lp < lp_time_limit:
            time_left = lp_time_limit - (time.monotonic() - t0_lp)
            if time_left < 15:
                break

            lp_it += 1
            # Solve LP
            g = lp_best_response(a_work, time_limit_lp=min(45.0, time_left * 0.5))
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g_scaled = g * (a_work.sum() / g_sum)

            # Line search
            improved = False
            for t in [0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]:
                a_try = (1 - t) * a_work + t * g_scaled
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try /= s_try
                J_try = eval_J_norm(a_try)
                if J_try < lp_best_val - 1e-14:
                    a_work = a_try
                    lp_best_val = J_try
                    improved = True
                    lp_stall = 0
                    break

            if improved:
                if lp_best_val < best_score:
                    best_score = lp_best_val
                    best_a = a_work.copy()
                if lp_it % 5 == 1:
                    print(f"  LP {lp_it}: J={lp_best_val:.8f} (t={time.monotonic()-t_global:.0f}s)")
            else:
                lp_stall += 1
                if lp_stall > 3:
                    # Perturb and try again
                    rng = np.random.default_rng(lp_it * 7919)
                    pert = rng.normal(0, 0.008, n_cur) * a_work
                    a_work_pert = a_work + pert
                    a_work_pert = np.maximum(a_work_pert, 1e-10)
                    a_work_pert /= a_work_pert.sum()
                    J_pert = eval_J_norm(a_work_pert)
                    if J_pert < lp_best_val + 0.001:
                        a_work = a_work_pert
                    else:
                        a_work = a_work_pert
                    lp_stall = 0
                    # Also try MD for a bit
                    time_left2 = lp_time_limit - (time.monotonic() - t0_lp)
                    if time_left2 > 25:
                        a_md, _ = mirror_descent(a_work.copy(), min(12.0, time_left2 * 0.25),
                                                top_k=100, beta_w=8.0, c0=3.5, seed=lp_it)
                        if eval_J_norm(a_md) < lp_best_val:
                            a_work = a_md
                            lp_best_val = eval_J_norm(a_work)
                            if lp_best_val < best_score:
                                best_score = lp_best_val
                                best_a = a_work.copy()
                elif lp_stall > 0:
                    # Between LPs, do a few MD steps
                    time_left2 = lp_time_limit - (time.monotonic() - t0_lp)
                    if time_left2 > 20:
                        a_md, _ = mirror_descent(a_work.copy(), min(10.0, time_left2 * 0.2),
                                                top_k=100, beta_w=8.0, c0=3.5, seed=1000 + lp_it)
                        if eval_J_norm(a_md) < lp_best_val:
                            a_work = a_md
                            lp_best_val = eval_J_norm(a_work)
                            if lp_best_val < best_score:
                                best_score = lp_best_val
                                best_a = a_work.copy()

    try_update(a_work if lp_algebra and 'a_work' in dir() else best_a, "lp_phase")

    # === Phase 3: Extended MD ===
    remaining = time_budget - 60 - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Extended MD (n={len(best_a)}) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 20:
        a_out, _ = mirror_descent(best_a.copy(), t3, top_k=100, beta_w=8.0,
                                  c0=3.5, restart_every=8000, verbose=False, seed=777)
        try_update(a_out, "md_ext")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - 60 - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent (n={len(best_a)}) ===")
    t4 = min(remaining * 0.13, 160.0)
    if t4 > 10:
        a_out, _ = coordinate_descent(best_a.copy(), t4, verbose=False, seed=314)
        try_update(a_out, "coord")

    # === Phase 5: MD after CD ===
    remaining = time_budget - 60 - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MD Post-CD ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_out, _ = mirror_descent(best_a.copy(), t5, top_k=120, beta_w=6.0, c0=3.0,
                                  restart_every=10000, seed=555)
        try_update(a_out, "md_post_cd")

    # === Phase 6: Length sweep ===
    remaining = time_budget - 60 - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-400, -200, 200, 400, 600]:
        nl = n_best + d
        if 2000 <= nl <= 4500:
            lengths_to_try.append(nl)
    for nl in [2500, 3000, 3500, 4000]:
        if nl not in lengths_to_try and abs(nl - n_best) > 100:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - 60 - (time.monotonic() - t_global)
        if remaining < 35:
            break
        print(f"  Length: n={nl} (remaining {remaining:.0f}s)")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.35, 70.0)
        if t_md > 8:
            a_out, _ = mirror_descent(a_int, t_md, top_k=80, beta_w=10.0, c0=3.5, seed=nl)
            try_update(a_out, f"len{nl}")

    # === Phase 7: Final polish ===
    remaining = time_budget - 60 - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish (n={len(best_a)}) ===")
    t7 = min(remaining * 0.45, 140.0)
    if t7 > 10:
        a_out, _ = mirror_descent(best_a.copy(), t7, top_k=150, beta_w=5.0,
                                  c0=2.5, restart_every=12000, seed=999, verbose=False)
        try_update(a_out, "final_md")

    remaining = time_budget - 60 - (time.monotonic() - t_global)
    if remaining > 10:
        a_out, _ = coordinate_descent(best_a.copy(), remaining * 0.5, seed=271)
        try_update(a_out, "final_cd")

    # Final output
    if best_a is not None:
        # Ensure valid output
        final_a = np.maximum(best_a, 0.0)
        final_a = np.minimum(final_a, 1000.0)
        s_final = final_a.sum()
        if s_final < 0.01:
            final_a = final_a * (1.0 / s_final) if s_final > 0 else np.ones(len(final_a))
        final_J = eval_J_norm(final_a / final_a.sum())
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.8f}, n={len(final_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in final_a]
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J_norm(fb):.8f}")
        return [float(x) for x in fb]
