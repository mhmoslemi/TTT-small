Actually, it's easier to work with the logarithm: let a_i = exp(x_i) / Z where Z = sum(exp(x_j)). Then the constraint sum(a) = 1 is automatically satisfied, and a >= 0 is automatic.

The gradient with respect to x:
df/dx_i = (df/da) * (da_i/dx_i)
da_i/dx_i = a_i * (1 - a_i)  (for the softmax parameterization)
da_i/dx_j = -a_i * a_j  (for i != j)

So df/dx = (df/da) * a - a * sum_j (df/da_j * a_j)
= a * (grad_f - sum_j grad_f_j * a_j)

where grad_f = gradient of f with respect to a.

This is a standard softmax gradient formula.

Actually, for the L-BFGS approach, I think it's simpler to just work with the original variables a (with a >= 0, sum(a) = 1) and use the L-BFGS-B method with bounds and a constraint. But scipy's L-BFGS-B doesn't handle equality constraints directly.

Alternative: work with x (unconstrained) and use the softmax parameterization. Then there are no constraints, and we can use plain L-BFGS.

```python
def softmax_obj(x, beta):
    """Compute smoothed objective and gradient."""
    n = len(x)
    a = np.exp(x - x.max())
    a /= a.sum()
    b = fast_conv(a)
    beta_b = beta * b
    beta_b_shift = beta_b - beta_b.max()
    logsumexp_val = beta_b.max() + np.log(np.exp(beta_b_shift).sum())
    f = logsumexp_val / beta
    
    # Gradient with respect to a
    weights = np.exp(beta_b_shift) / np.exp(beta_b_shift).sum()
    grad_a = np.zeros(n)
    # Only iterate over significant weights
    sig_mask = weights > 1e-12
    for k in np.where(sig_mask)[0]:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad_a[j_start:j_end+1] += (2.0 / beta) * weights[k] * a[k - np.arange(j_start, j_end+1)]
    
    # Convert to gradient with respect to x
    # df/dx_i = sum_j (df/da_j) * da_j/dx_i
    # = a_i * (grad_a_i - sum_j grad_a_j * a_j)
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return f, grad_x
```

This should work. The main cost per evaluation is the convolution (O(n log n)) plus the weighted sum over lags (O(n * n) worst case, but typically much less with the significance threshold).

For n=2600, the convolution takes about 50-100 microseconds. The weighted sum over lags: if we have 1000 significant lags, it's 1000 * 2600 = 2.6M multiply-adds, which is maybe 1-2 ms. L-BFGS typically needs 100-1000 iterations to converge, so total time would be 1-20 seconds. Very feasible!

```python
def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_obj_and_grad(x, beta):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    x is the unbounded parameter; a = softmax(x).
    """
    n = len(x)
    # softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    
    b = fast_conv(a)
    beta_b = beta * b
    max_bb = beta_b.max()
    exp_bb = np.exp(beta_b - max_bb)
    sum_exp = exp_bb.sum()
    
    f = (max_bb + np.log(sum_exp)) / beta
    
    # Weights for the active lags
    weights = exp_bb / sum_exp
    
    # Gradient with respect to a
    grad_a = np.zeros(n)
    # Only process lags with significant weight
    sig_idx = np.where(weights > 1e-10 * sum_exp)[0]
    for k in sig_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            grad_a[idx] += (2.0 / beta) * weights[k_int] * a[k_int - idx]
    
    # Convert to gradient w.r.t. x (softmax chain rule)
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return f, grad_x

def lbfgs_smoothed(a0, time_limit, beta_start=50.0, beta_end=2000.0, verbose=False):
    """
    Optimize using L-BFGS on the smoothed objective with adaptive beta.
    """
    n = len(a0)
    a = norm_sum1(a0)
    # Initialize x from a
    x = np.log(a + 1e-30)
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    
    # Adaptive beta schedule
    n_phases = 5
    betas = np.logspace(np.log10(beta_start), np.log10(beta_end), n_phases)
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        # Define objective and gradient for scipy
        def objective(x_flat):
            val, grad = smooth_obj_and_grad(x_flat, beta)
            return val, grad
        
        # Run L-BFGS
        result = minimize(objective, x, method='L-BFGS-B', jac=True,
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxfun': 5000, 'iprint': -1})
        
        # Update
        x_opt = result.x
        a_new = np.exp(x_opt - x_opt.max())
        a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            best_score = 2.0 * n * m_new
        
        x = x_opt
        a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LFGS beta={beta:.0f} ({elapsed:.1f}s): M={m_new:.10f}, score={best_score:.8f}")
    
    return best_a, best_m

def nmd_large_active(a0, time_limit, verbose=False):

                g[idx] += 2.0 * y[k_int - idx]

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)

            stall += 1

            stall = 0

            strategy = int(rng.integers(4))
            if strategy == 0:

            x = x_r

        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) > max_cons:
            active = active[np.argsort(b[active])[::-1][:max_cons]]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        constraints = [cp.sum(a_var) == 1]

            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]

        constraints.append(a_var <= a * (1.0 + r))
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)

            else:
                prob.solve(verbose=False)

            return best_a, best_m
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):

                a = a / a.sum()

        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:

            a_trial = np.maximum(a_trial, 0.0)

                best_a = a.copy()
                accepted = True
                no_improve = 0

        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.75)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

The "ideal" solution would have b[k] = M for all k (perfectly flat autocorrelation). This is achieved by a "top-hat" in the frequency domain: |A(f)|^2 = constant. For non-negative sequences, the best we can do is approximate a flat periodogram while maintaining non-negativity in the time domain. This is related to the "minimum variance" or "minimum peak" problem.

The optimal solution is likely to be smooth (no sharp features) and approximately symmetric, with a shape like a raised cosine or a polynomial.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_obj_and_grad(x, beta):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. unbounded param x."""
    n = len(x)
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()

    b = fast_conv(a)
    beta_b = beta * b
    max_bb = beta_b.max()
    exp_bb = np.exp(beta_b - max_bb)
    sum_exp = exp_bb.sum()

    f = (max_bb + np.log(sum_exp)) / beta

    weights = exp_bb / sum_exp

    grad_a = np.zeros(n)
    sig_thresh = 1e-12
    sig_idx = np.where(weights > sig_thresh)[0]
    for k in sig_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            grad_a[idx] += (2.0 / beta) * weights[k_int] * a[k_int - idx]

    grad_x = a * (grad_a - np.dot(grad_a, a))

    return f, grad_x

def lbfgs_smoothed(a0, time_limit, beta_start=50.0, beta_end=5000.0, verbose=True):

    a = norm_sum1(a0)

    best_m = fast_conv(a).max()

        if time.monotonic() - t0 > time_limit:

        if remaining < 15:
            break

        def objective_func(x_flat):
            return smooth_obj_and_grad(x_flat, beta)

        max_iter = 300 if beta < 500 else 150

        result = minimize(objective_func, x, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxfun': 8000, 'iprint': -1, 'maxls': 50})

        x_opt = result.x
        a_new = np.exp(x_opt - x_opt.max())
        a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0.0)

        m_new = fast_conv(a_new).max()

            best_m = m_new
            best_a = a_new.copy()

        x = x_opt.copy()
        a = a_new.copy()

                  f"M={m_new:.12f}, obj={result.fun:.10f}, iter={result.nit}")

                g[idx] += 2.0 * y[k_int - idx]

        max_log = log_a.max()

        # Gradually tighten the active set

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) > max_cons:
            order = np.argsort(b[active])[::-1]
            active = active[order[:max_cons]]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1]

        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

        constraints.append(a_var <= a * (1.0 + r))

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            return best_a, best_m

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            r = max(0.005, r * 0.8)
            if no_improve > 12:
                rng = np.random.default_rng(t + 7)
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                a[idx] *= rng.uniform(0.75, 1.25, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
            continue

        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_trial = (1.0 - t_frac) * a + t_frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.15, r * 1.03)
                break

        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.7)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Remez t={t} ({elapsed:.1f}s): M={best_m:.12f}, r={r:.4f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
