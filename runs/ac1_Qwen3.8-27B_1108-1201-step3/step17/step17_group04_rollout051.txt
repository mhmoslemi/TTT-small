"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP Frank-Wolfe with
symmetry exploitation, backtracking line search, and parametric initialization.

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
import sys

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

        return float('inf')

    n = len(a)
