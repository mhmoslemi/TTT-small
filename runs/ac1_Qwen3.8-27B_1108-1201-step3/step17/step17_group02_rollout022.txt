def logsumexp_optimize(a0, time_limit, beta_start=10, beta_end=1000, ...):
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = max(conv(a,a))
    
    for step in range(max_steps):
        if time elapsed > time_limit: break
        
        # Adaptive beta
        beta = beta_start * (beta_end/beta_start) ** (step / max_steps)
        
        # Compute weights
        c = conv(a, a)  # length 2n-1
        w = exp(beta * c - max(beta * c))  # numerically stable
        w = w / sum(w)
        
        # Compute gradient
        g = (2/beta) * np.correlate(w, a, mode='valid')
        
        # Mirror descent step (multiplicative update)
        # a_new[j] = a[j] * exp(-eta * g[j]) / sum(...)
        eta = learning_rate / sqrt(step + 1)
        a_new = a * exp(-eta * g)
        a_new = a_new / sum(a_new)
        
        # Track best
        if max(conv(a_new, a_new)) < best_m:
            best_m = max(conv(a_new, a_new))
            best_a = a_new.copy()
        
        a = a_new

    a = rng.uniform(0, 1, n)
    a = a / a.sum()

        # Get magnitude

        # But we can't just set it to constant because the zero frequency should be sum(a) = 1
        # Let's use: new_mag = (mag^p + constant)^(1/(2p)) for some p
        # Or simply: new_mag = (mag^2 + eps) / (1 + eps) ...

def spectral_flat_init(n, n_iter=200, seed=42, floor=0.001):
    """Generate sequence with flat power spectrum via iterative sieve."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = rng.uniform(0.5, 1.0, n)
    a = a / a.sum()

    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Reduce peaks in power spectrum
        mean_p = power.mean()
        if mean_p < 1e-30:
            break
        excess = np.maximum(0, power - mean_p)
        if excess.max() < 1e-15:
            break
        alpha = 0.3 + 0.3 * (it / n_iter)
        new_power = power - alpha * excess
        new_power = np.maximum(new_power, mean_p * floor)
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = np.sqrt(new_power) * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

    return a

def lse_amd_optimize(a0, time_limit, beta_start=20.0, beta_end=5000.0,
                     c0=4.0, nesterov=True, restart_every=8000,
                     verbose=False, rng_seed=42):
    """Log-sum-exp smooth max with Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    fft_size = 1 << (2 * n).bit_length()

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta: exponential ramp
        frac = min(1.0, t / 100000.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Nesterov step (geometric)
        if nesterov and t > 2:
            mom = (t - 1.0) / (t + 1.5)
            ratio = x / np.maximum(x_prev, 1e-30)
            y = x * np.power(np.maximum(ratio, 1e-30), mom)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        # Compute autocorrelation and weights
        c = fast_autocorr(y)
        bc = beta * c
        bc_max = bc.max()
        w = np.exp(bc - bc_max)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(c)) / len(c)
        else:
            w = w / w_sum

        # Gradient: df/da[j] = (2/beta) * sum_m w[j+m] * a[m]
        # This is the valid cross-correlation of w (len 2n-1) and a (len n)
        g = (2.0 / beta) * np.correlate(w, a, mode='valid')

        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart if stalled
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LSE-AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-AMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_a, best_m


def lse_amd_multiscale(a0, time_limit, verbose=False, rng_seed=42):
    """Multi-scale LSE-AMD: first optimize a downsampled version, then refine."""
    n = len(a0)
    t0 = time.monotonic()

    # Scale 1: downsample by factor 4
    n_small = max(200, n // 4)
    a_small = np.interp(np.linspace(0, 1, n_small), np.linspace(0, 1, n), a0)
    a_small = norm_sum1(a_small)

    t1 = time_limit * 0.2
    a_small_out, _ = lse_amd_optimize(a_small, t1, beta_start=10, beta_end=2000,
                                       c0=3.5, verbose=False, rng_seed=rng_seed)

    # Upscale and use as starting point
    a_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_small), a_small_out)
    a_up = norm_sum1(a_up)

    # Blend with original
    blend = 0.3 * a_up + 0.7 * norm_sum1(a0)
    a_init = norm_sum1(blend)

    # Full-scale optimization
    t2 = time_limit - (time.monotonic() - t0) - 5
    a_out, m_out = lse_amd_optimize(a_init, t2, beta_start=20, beta_end=5000,
                                     c0=4.0, verbose=verbose, rng_seed=rng_seed + 1)
    return a_out, m_out


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=None):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(400, 2 * n - 1)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix (sparse)
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            T_full = np.zeros((2 * n - 1, n))
            for j in range(n):
                T_full[j:j + n, j] = y
            constraint_vals = T_full @ y
            margins = M_curr - constraint_vals
            binding_idx = np.argsort(margins)[:K_sel]
            del T_full

        # Build selected constraints
        T_sel = np.zeros((len(binding_idx), n))
        for row_i, k in enumerate(binding_idx):
            ki = int(k)
            for j in range(n):
                if 0 <= ki - j < n:
                    T_sel[row_i, j] = y[j] + y[ki - j] if ki - j != j else 2 * y[j]
                # Actually: dc[k]/da[j] = 2*a[k-j] if 0<=k-j<n else 0
                # But for the constraint T@b <= M, we need (f*g)[k] = sum_i f[i]*g[k-i]
                # For Frank-Wolfe: maximize sum(b) s.t. sum_i y[i]*b[k-i] <= M
                # So T[k, j] = y[k-j] if 0<=k-j<n else 0
            T_sel[row_i] = 0.0
            for j in range(n):
                m = ki - j
                if 0 <= m < n:
                    T_sel[row_i, j] = y[m]

        M_sel = M_curr * np.ones(len(binding_idx))

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

        if t % 30 == 0 and K_sel < 2 * n - 1:
            K_sel = min(K_sel + 50, 2 * n - 1)

    return best_a, best_m


def peak_pair_reduction(a0, time_limit, verbose=False, rng_seed=77):
    """Targeted peak-pair reduction: redistribute mass from peak-contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    n_peaks = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_k[:6]:
            ki = int(k)
            for j in range(n):
                m = ki - j
                if 0 <= m < n:
                    sens[j] += a[m]

        improved = False
        for k in top_k[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            contrib = np.zeros(n)
            for j in range(n):
                m = ki - j
                if 0 <= m < n:
                    contrib[j] = a[j] * a[m]
            if contrib.max() < 1e-20:
                continue

            src = np.argmax(contrib)
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination (low sensitivity)
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]

            found = False
            for frac in [0.02, 0.01, 0.005, 0.002, 0.05, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 50:
                j = int(rng.integers(n))
                frac = rng.uniform(0.005, 0.03)
                amt = src_val * frac
                a[src] = max(0, a[src] - amt)
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if not improved:
            stall += 1
            # Multi-point perturbation
            n_pert = min(10, n)
            idx = rng.choice(n, size=n_pert, replace=False)
            for i in idx:
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakPair ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with multi-destination trial."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_top = 10
    n_dst = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]

        # Sensitivity
        sens = np.zeros(n)
        for k in top_k[:5]:
            ki = int(k)
            for j in range(n):
                m = ki - j
                if 0 <= m < n:
                    sens[j] += a[m]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            contrib = np.zeros(n)
            for j in range(n):
                m = ki - j
                if 0 <= m < n:
                    contrib[j] = a[j] * a[m]
            if contrib.max() < 1e-20:
                continue

            src = np.argmax(contrib)
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try multiple destinations
            exclude = {src}
            candidates = [d for d in range(n) if d not in exclude]
            candidates.sort(key=lambda d: sens[d])
            best_dst = candidates[:n_dst]

            found = False
            for dst in best_dst:
                if found:
                    break
                for frac in [0.01, 0.005, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_sine_sq(n):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.sin(np.pi * x) ** 2)


def make_triangular(n):
    x = np.linspace(0, 1, n)
    return norm_sum1(1.0 - np.abs(2 * x - 1))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
