def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):

def get_subgradient(a, ac, top_k=20, beta=3.0, exact_frac=0.5, rng=None):
    n = len(a)
    grad = np.zeros(n)
    use_exact = (rng is None) or (rng.random() < exact_frac)
    if use_exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def compute_danger(a, ac, n_peaks=15):
    n = len(a)
    top_lags = np.argsort(ac)[::-1][:n_peaks]
    danger = np.zeros(n)
    for k in top_lags:
        k_int = int(k)
        for i in range(n):
            v1 = k_int - i
            v2 = i - k_int
            if 0 <= v1 < n:
                danger[i] += a[v1]
            if 0 <= v2 < n:
                danger[i] += a[v2]
    return danger, top_lags

def anneal_polish(a0, time_limit, seed=999):

def parametric_shapes(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    xc = x - 0.5
    shapes = []
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 10.0]:
        a = 1.0 + alpha * xc**2
        shapes.append(norm_sum1(np.maximum(a, 0.01)))
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = x**p + (1 - x)**p + 0.01
        shapes.append(norm_sum1(a))
    for d in [0.2, 0.25, 0.3, 0.35]:
        for sig in [0.06, 0.1, 0.15]:
            a = np.exp(-(x - (0.5 - d))**2 / (2 * sig**2)) + np.exp(-(x - (0.5 + d))**2 / (2 * sig**2))
            shapes.append(norm_sum1(a))
    for sig in [0.1, 0.15, 0.2, 0.3]:
        a = np.exp(-xc**2 / (2 * sig**2))
        shapes.append(norm_sum1(a))
    for _ in range(20):
        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(4, 10)
        amp2 = rng.uniform(0.1, 0.5)
        a = 1.0 + amp2 * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 6.28))
        a += 0.05 * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 6.28))
        shapes.append(norm_sum1(np.maximum(a, 0.01)))
    return shapes


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
