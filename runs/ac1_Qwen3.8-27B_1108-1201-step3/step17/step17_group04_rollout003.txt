        j_idx = k_int - idx
        grad[idx] += 2.0 * probs[k_int] * a[j_idx]
    return grad


# === Initialization strategies ===

def make_uniform_init(n):
    return np.ones(n) / n


def make_symmetric_init(n):
    """Cosine-shaped: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law decay from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_bimodal_init(n):
    """Two bumps at 1/4 and 3/4 positions."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.25) / 0.1) ** 2) + np.exp(-((x - 0.75) / 0.1) ** 2) + 0.005
    return norm_sum1(a)


def make_periodic_init(n, freq=3):
    """Periodic structure."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x) + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_random_init(n, seed=0):
    """Random positive initialization."""
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / ss
