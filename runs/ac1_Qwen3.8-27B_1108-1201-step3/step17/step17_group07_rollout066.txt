def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgrad_full(y, ac, top_k):
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - np.arange(j_start, j_end + 1)]
    g /= k_max
    return g

def lp_remez_single(a, max_cons=50, r=0.1, solver=None):
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    active = np.argsort(b)[::-1][:max_cons]

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - np.arange(j_start, j_end + 1)]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    constraints.append(a_var >= a * (1.0 - r))
    constraints.append(a_var <= a * (1.0 + r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.05)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bell':
        a = np.exp(-10.0 * (x - 0.5) ** 2)
    elif kind == 'U_shape':
        a = (2.0 * x - 1.0) ** 2 + 0.1
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

                a_out = np.maximum(a_out, 0.0)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [Remez final n{len(best_seq)}]: {sc:.8f}")

    # Phase 6: Final NMD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        t6a = min(remaining * 0.4, 100.0)
        if t6a > 10:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t6a, top_k=10, c0=1.5, verbose=False)
            a_out = np.maximum(a_out, 0.0)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [NMD final n{n_best}]: {sc:.8f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.4, 80.0)
            if t6b > 10:
                a_out, _ = remez_lp_method(norm_sum1(best_seq), t6b, solver=solver, r_init=0.02, verbose=False)
                a_out = np.maximum(a_out, 0.0)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [Remez last n{n_best}]: {sc:.8f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t6c = min(remaining * 0.4, 50.0)
            if t6c > 8:
                a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t6c, verbose=False, rng_seed=999)
                a_out = np.maximum(a_out, 0.0)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  *** NEW BEST [CD last n{n_best}]: {sc:.8f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
