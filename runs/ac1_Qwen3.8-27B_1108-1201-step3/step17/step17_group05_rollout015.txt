def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_asymmetric_init(n, seed=0, type='ramp'):

def fw_optimize(a0, time_limit, verbose=True, K_start=100, K_max=500, seed=42):

    Nesterov-accelerated Frank-Wolfe with adaptive constraint LP.
    """

    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov extrapolation

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        M_curr = ac.max()

        top_idx = np.argsort(ac)[::-1][:K_use]
        
        # Build LP matrix T: K_use x n
        # T[k_row, j] = y[k_j - j] if 0 <= k_j - j < n, else 0
        T = np.zeros((K_use, n))
        for row, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                T[row, js] = y[m_int - js]
        
        # Solve LP: minimize t s.t. T@b <= t, sum(b) = 1, b >= 0
        # Variables: (b_0, ..., b_{n-1}, t)
        # c = [0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Inequality: T@b - t <= 0 => [T, -1] @ [b; t] <= 0
        A_ub = np.hstack([T, -np.ones((K_use, 1))])
        b_ub = np.zeros(K_use)
        
        # Equality: sum(b) = 1 => [1, ..., 1, 0] @ [b; t] = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            result = so.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                bounds=bounds, method='highs',
                                options={'presolve': True, 'time_limit': 5})
        except Exception:

        if result is None or not result.success:

            js = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))

            b_dir = np.zeros(n)

            else:
                b_dir = b_dir / b_sum
        
        # FW step
        alpha = 2.0 / (iters + 2.0)
        x_new = (1 - alpha) * x + alpha * b_dir
        x_new = np.maximum(x_new, 0)
        sx_new = x_new.sum()
        if sx_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx_new
        
        x_prev = x
        x = x_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()

            stall = 0
        else:

            x = norm_sum1(x)
            x_prev = x.copy()
            stall = 0
            K = K_start
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={iters} ({elapsed:.0f}s): J={J:.6f}, K={K}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_autocorr(a):
    """Compute np.convolve(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def make_asymmetric_init(n, seed=0, typ='exp'):
    """Create asymmetric non-negative initializations (sum=1)."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if typ == 'exp':
        alpha = rng.uniform(1.5, 4.0)
        a = np.exp(-alpha * x)
    elif typ == 'ramp':
        slope = rng.uniform(0.5, 2.0)
        a = 1.0 + slope * x
    elif typ == 'half_gauss':
        mu = rng.uniform(0.15, 0.45)
        sigma = rng.uniform(0.15, 0.35)
        a = np.exp(-((x - mu) / sigma) ** 2)
    elif typ == 'sine_asym':
        f = int(rng.integers(1, 5))
        phase = rng.uniform(0, np.pi)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * x + phase)

            phase = rng.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * f * x + phase)
    else:
        a = 1.0 + x  # default ramp
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)

def build_t_matrix(y, top_idx, n):
    """Build the constraint matrix T where T[row, j] = y[m-j] if valid."""
    K = len(top_idx)
    T = np.zeros((K, n), dtype=np.float64)
    for row in range(K):
        m = int(top_idx[row])
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            T[row, js] = y[m - js]
    return T


def solve_fw_lp(T, M, n, fallback_dir=None):
    """
    Solve LP: maximize sum(b) subject to T@b <= M, b >= 0.
    Returns normalized direction b/sum(b).
    """
    K = T.shape[0]
    # minimize -sum(b) subject to T@b <= M*ones, b >= 0
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = so.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            bs = b.sum()
            if bs > 1e-15:
                return b / bs
    except Exception:
        pass
    if fallback_dir is not None:
        return fallback_dir
    return np.ones(n) / n


def solve_fw_lp_minmax(T, M, n, fallback_dir=None):
    """
    Solve LP: minimize t subject to T@b <= t, sum(b)=1, b>=0.
    This finds the best direction in the simplex that minimizes peak conv.
    """
    K = T.shape[0]
    # Variables: (b_0,...,b_{n-1}, t)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    # Inequality: T@b - t <= 0
    A_ub = np.hstack([T, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds: b_i >= 0, t >= 0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = so.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            bs = b.sum()
            if bs > 1e-15:
                return b / bs
    except Exception:
        pass
    if fallback_dir is not None:
        return fallback_dir
    return np.ones(n) / n


def fw_optimize(a0, time_limit, verbose=True, K_start=150, K_max=500,

    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()

    stall = 0
    K = K_start
    iters = 0
    lp_fail_count = 0

        iters += 1

        # Nesterov extrapolation

            lag_probs = ac[top_lags] - ac[top_lags].min() + 1e-10
            lag_probs = lag_probs / lag_probs.sum()

            # Find most contributing pair for this lag
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)

            # Pick source from top contributors

            src = int(js[src_local])
