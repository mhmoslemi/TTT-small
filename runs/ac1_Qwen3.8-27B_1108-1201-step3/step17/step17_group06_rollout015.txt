def compute_subgradient(a, conv, top_k=80, beta_w=10.0):
    """Compute weighted subgradient of max(conv) using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]

        return False

    # We need a wrapper that doesn't use closures
    # Instead, use a global-like approach via the callback
    result = minimize(
        fun=smooth_obj_grad_wrapper,
        x0=z0,
        args=(p,),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'maxfun': 50000, 'ftol': 1e-12,
                 'gtol': 1e-10, 'iprint': -1}
    )

    # This approach has issues with time checking in callback.
    # Let me use a manual L-BFGS loop instead.
    return None, None


def smooth_obj_grad_wrapper(z, p):
    """Wrapper for scipy minimize."""
    return smooth_obj_grad(z, p)


def lbf_smooth_manual(a0, time_limit, p=8.0, max_iter=3000):
    """Manual L-BFGS-B style optimization with time checking."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15).astype(np.float64)
    a0 = a0 / a0.sum()
    z = np.log(a0) - np.log(a0).mean()

    t0 = time.monotonic()
    best_z = z.copy()
    best_val = 2.0 * n * eval_J(a0)

    # L-BFGS parameters
    m = 20  # memory
    W = []  # list of (s, y) pairs
    rho = []
    lr = 1.0
    c = 0.0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        f, g = smooth_obj_grad(z, p)

        if f < best_val:
            best_val = f
            best_z = z.copy()

        # L-BFGS two-loop recursion
        q = g.copy()
        alphas = np.zeros(len(W))
        for i in range(len(W) - 1, -1, -1):
            s_i, y_i = W[i]
            alphas[i] = rho[i] * np.dot(s_i, q)
            q -= alphas[i] * y_i

        # Initial direction
        if len(W) == 0:
            gamma = 1.0
        else:
            s_last, y_last = W[-1]
            sy = np.dot(s_last, y_last)
            if sy > 1e-15:
                gamma = sy / np.dot(y_last, y_last)
            else:
                gamma = 1.0

        r = gamma * q

        for i in range(len(W)):
            s_i, y_i = W[i]
            beta_i = rho[i] * np.dot(y_i, r)
            r += (alphas[i] - beta_i) * s_i

        # Backtracking line search
        alpha = lr
        f_new = None
        for _ in range(30):
            z_new = z + alpha * r
            f_new, _ = smooth_obj_grad(z_new, p)
            if f_new < f - 1e-4 * abs(g) * abs(alpha * r):
                break
            alpha *= 0.5
            if alpha < 1e-15:
                break

        if f_new is None or f_new >= f:
            # Try small random perturbation
            break

        s_k = z_new - z
        y_k = smooth_obj_grad(z_new, p)[1] - g
        sy = np.dot(s_k, y_k)

        z = z_new

        if sy > 1e-15:
            W.append((s_k, y_k))
            rho.append(1.0 / sy)
            if len(W) > m:
                W.pop(0)
                rho.pop(0)

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_val


def adam_mirror_descent(a0, time_limit, top_k=80, beta_w=10.0,
                        lr_init=0.1, beta1=0.9, beta2=0.999,
                        restart_every=5000, verbose=False):
    """Adam-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w
    lr = lr_init

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        t += 1

        # Compute subgradient
        conv = fast_conv(a)
        g = compute_subgradient(a, conv, top_k=cur_top_k, beta_w=cur_beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Mirror descent step with Adam
        eta = lr / (np.sqrt(t) + 1.0)
        log_r = np.log(a + 1e-30) - eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        a = a_new

        # Track best
        val = eval_J(a)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Learning rate decay
        lr = lr_init / (1 + 0.01 * t)

        # Restart logic
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            a = x_r.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            # Adapt parameters
            cur_top_k = min(cur_top_k + 20, 300)
            cur_beta = max(cur_beta - 0.5, 2.0)
            # Reset adaptive lr
            lr = lr_init * 0.5
            if verbose:
                print(f"    AMD restart: J={best_val:.6f}, k={cur_top_k}, beta={cur_beta:.1f}")

        if verbose and it % 5000 == 0:
            print(f"    AMD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def nesterov_mirror_descent(a0, time_limit, top_k=80, beta_w=10.0, c0=4.0,
                            restart_every=6000, verbose=False):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        conv = fast_conv(y)
        g = compute_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        eta = c0 / (np.sqrt(t) + 1.0)

        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.04)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 300)
            cur_beta = max(cur_beta - 0.5, 2.0)
            c0 = max(c0 * 0.92, 1.0)

        if verbose and t % 10000 == 0:
            print(f"    NMD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with multi-step mass transfer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        n_lags = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        # Compute pressure at each position
        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        # Find source
        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 10
            continue
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 10
            continue

        # Find destination
        dest_scores = np.zeros(n)
        for k in top_lags[:6]:
            k_i = int(k)
            for i in range(n):
                j = k_i - i
                if 0 <= j < n:
                    dest_scores[i] += a[j]
        dest_scores[src] = np.inf
        dest_scores += 0.05 * pressure
        dst = int(np.argmin(dest_scores))

        # Try multiple transfer fractions
        improved = False
        for frac in [0.003, 0.005, 0.008, 0.01, 0.02, 0.03, 0.05, 0.08, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-15:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 200 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine_taper(n, alpha=0.3):
    """Cosine-tapered distribution."""
    x = np.linspace(0, 1, n)
    a = (1.0 + np.cos(np.pi * (x - alpha))) / (1.0 + np.cos(np.pi * alpha))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p / p.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is not None:
        try_update(parent, "parent")
    else:
        for n_try in [2000, 3000, 4000]:
            try_update(make_power_ramp(n_try, 1.0), f"ramp_n{n_try}")

    if best_a is None:
        best_a = make_power_ramp(3000, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    # === Phase 1: L-BFGS on smooth objective ===
    print(f"\n=== Phase 1: L-BFGS Smooth (budget remaining={time_budget-(time.monotonic()-t_global):.0f}s) ===")
    n_cur = len(best_a)
    remaining = time_budget - (time.monotonic() - t_global)
    p_vals = [6.0, 10.0, 20.0]
    t_per_lbf = max(20.0, remaining * 0.25 / len(p_vals))
    for p in p_vals:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_avail = min(t_per_lbf, remaining - 60)
        if t_avail < 15:
            break
        print(f"  L-BFGS p={p} ({t_avail:.0f}s), n={n_cur}...")
        a_out, _ = lbf_smooth_manual(best_a.copy(), t_avail, p=p, max_iter=2000)
        try_update(a_out, f"lbf_p{p:.0f}")

    # === Phase 2: Adam Mirror Descent (multiple configs) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam Mirror Descent ===")
    configs = [
        (80, 12.0, 0.15),
        (120, 8.0, 0.10),
        (200, 5.0, 0.08),
    ]
    t_per_amd = max(25.0, remaining * 0.35 / len(configs))
    for idx, (tk, bw, lr) in enumerate(configs):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t_avail = min(t_per_amd, remaining - 60)
        if t_avail < 20:
            break
        print(f"  AMD k={tk}, beta={bw}, lr={lr} ({t_avail:.0f}s)...")
        a_out, _ = adam_mirror_descent(best_a.copy(), t_avail, top_k=tk,
                                       beta_w=bw, lr_init=lr)
        try_update(a_out, f"amd_{idx}")

    # === Phase 3: Nesterov Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Nesterov MD ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_out, _ = nesterov_mirror_descent(best_a.copy(), t3, top_k=100,
                                           beta_w=8.0, c0=3.5,
                                           restart_every=8000, verbose=True)
        try_update(a_out, "nmd")

    # === Phase 4: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent ===")

    # === Phase 5: MD after CD ===

            break
        if abs(nl - n_best) <= 30:

                                               beta_w=10.0, c0=3.0)
            try_update(a_out, f"len{nl}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and len(best_a) != nl:
            t_cd = min(remaining * 0.10, 30.0)
            if t_cd > 8:
                a_cur = interpolate_seq(best_a, nl)
                a_out, _ = coordinate_descent(a_cur, t_cd, seed=nl)
                try_update(a_out, f"len{nl}_cd")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 7: Final Polish (n={n_cur}) ===")
    t7 = min(remaining * 0.35, 250.0)
    if t7 > 20:
        # Alternate between AMD and CD
        a_cur = best_a.copy()

        for round_i in range(3):

            if remaining < 20:
                break
            t_sub = min(sub_budget, remaining - 10)
            if round_i % 2 == 0:
                a_out, _ = adam_mirror_descent(a_cur, t_sub, top_k=200,
                                               beta_w=4.0, lr_init=0.03)
            else:
                a_out, _ = coordinate_descent(a_cur, t_sub, seed=100 + round_i)
            if a_out is not None and eval_J(a_out) <= eval_J(a_cur) + 1e-10:
                a_cur = a_out
            else:
                a_cur = best_a.copy()
        try_update(a_cur, "final_polish")

    # === Phase 8: One more L-BFGS round ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 8: Final L-BFGS ===")
        t8 = remaining - 15
        a_out, _ = lbf_smooth_manual(best_a.copy(), t8, p=12.0, max_iter=3000)
        try_update(a_out, "final_lbf")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]

        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
