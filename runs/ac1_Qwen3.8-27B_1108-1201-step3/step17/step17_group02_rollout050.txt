    # Use FFT for cross-correlation
    # Need L >= (2n-1) + n - 1 = 3n - 2
    L = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, L)
    Ap = np.fft.rfft(a, L)
    cross = np.fft.irfft(P * np.conj(Ap), L)[:n]
    
    grad = 2.0 * cross
    return obj, grad, c


def project_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def nesterov_gradient_descent(a0, time_limit, beta_init=10.0, beta_final=50.0,
                                eta_init=0.5, verbose=False, rng_seed=42):
    """
    Nesterov-accelerated projected gradient descent on the smooth max objective.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta = eta_init
    beta = beta_init
    stall = 0
    max_stall = 3000
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        t = iter_count + 1
        
        # Adaptive beta: increase over time for sharper max approximation
        progress = min(1.0, t / 5000.0)
        beta = beta_init + (beta_final - beta_init) * progress
        
        # Nesterov extrapolation
        momentum = (t - 1.0) / (t + 2.0)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        obj, grad, c = compute_smooth_gradient(y, beta)
        
        # Gradient step
        eta_t = eta_init / (1.0 + 0.1 * np.sqrt(t))  # slowly decreasing
        z = y - eta_t * grad
        x_new = project_simplex(z)
        
        x_prev = x
        x = x_new
        iter_count += 1
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_val - 1e-15:
            best_val = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    NGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")
    
    return best_a, best_val


def lp_frank_wolfe_oa(a0, time_limit, solver, top_K_init=100, verbose=True):
    """
    LP-based Frank-Wolfe with outer approximation.
    The constraint set is adaptively grown to include newly violated constraints.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    # Active constraint set (indices of lags k)
    ac = fast_autocorr(y)
    M_curr = ac.max()
    active_set = set(np.argsort(ac)[::-1][:top_K_init].tolist())
    
    total_iters = 0
    oa_count = 0
    
    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        t = total_iters
        
        # Recompute autocorrelation and update active set if needed
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Build constraint matrix for active set
        active_list = sorted(active_set)
        K = len(active_list)
        
        if K > 4 * n:  # cap on constraints
            active_set = set(np.argsort(ac)[::-1][:4 * n].tolist())
            active_list = sorted(active_set)
            K = len(active_list)
        
        # Build T matrix: T[k,j] = y[j] if 0 <= k-j < n else 0
        # i.e., T[k] is the "partner" vector for lag k
        T_sel = np.zeros((K, n))
        for idx, k in enumerate(active_list):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s < e:
                T_sel[idx, s:e] = y[ki - (np.arange(s, e))]
        
        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Check if direction violates any non-active constraints (outer approximation)
        ac_dir = fast_autocorr(direction)
        if K < 2 * n - 1:
            # Check for violations
            non_active = np.array([k for k in range(2 * n - 1) if k not in active_set])
            if len(non_active) > 0:
                violations = ac_dir[non_active] - M_curr
                if violations.max() > M_curr * 0.01:  # 1% tolerance
                    # Add violated constraints
                    n_add = min(50, len(non_active))
                    worst = non_active[np.argsort(violations)[::-1][:n_add]]
                    active_set.update(worst.tolist())
                    oa_count += 1
                    continue  # Re-solve with larger constraint set
        
        # Frank-Wolfe update with Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Momentum
        if t == 1:
            x_new = direction.copy()
        else:
            beta_mom = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_mom * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x_mom = x_new
        
        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()
        
        # Periodically refresh active set to top peaks of current y
        if t % 50 == 0:
            ac_y = fast_autocorr(y)
            top = np.argsort(ac_y)[::-1][:max(top_K_init, len(active_set))]
            active_set = set(top.tolist())
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-OA iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(active_set)}, OA={oa_count}")
    
    return best_a, best_m


def peak_shaving_simultaneous(a0, time_limit, verbose=True, rng_seed=77):
    """
    Simultaneous multi-peak reduction: identify top peaks, compute sensitivities,
    and perform coordinated moves to reduce all peaks at once.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    n_peaks = 8
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]
        
        if top_k[0] + 1 <= 0:  # shouldn't happen
            break
        
        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            partner = ki - idx
            sens[idx] += a[partner]
        
        # For each top peak, find the most contributing pair
        moves = []  # (src, dst, amount, expected_reduction)
        occupied_src = set()
        
        for k in top_k[:5]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            # Find top contributing pairs not already used
            pair_order = np.argsort(contrib)[::-1]
            for po in pair_order[:10]:
                src = int(idx[po])
                if src in occupied_src or a[src] < 1e-12:
                    continue
                # Find best destination (lowest sensitivity)
                exclude = {src, int(partner[po])}
                candidates = np.array([d for d in range(n) if d not in exclude])
                if len(candidates) == 0:
                    break
                dst = candidates[np.argmin(sens[candidates])]
                
                if dst == src:
                    continue
                
                moves.append((src, dst, k))
                occupied_src.add(src)
                break
        
        if not moves:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
            continue
        
        # Try coordinated moves
        improved = False
        for base_frac in [0.01, 0.005, 0.02, 0.03, 0.05]:
            a_trial = a.copy()
            for src, dst, k in moves:
                amt = a_trial[src] * base_frac
                if amt < 1e-15:
                    continue
                a_trial[src] -= amt
                a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            stall += 1
            # Try individual moves
            for src, dst, k in moves:
                if time.monotonic() - t0 > time_limit:
                    break
                for frac in [0.005, 0.01, 0.02, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakShave final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return project_simplex(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return project_simplex(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return project_simplex(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Uses Nesterov-accelerated projected gradient, LP-FW with outer approximation,
    and simultaneous peak shaving to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = project_simplex(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4200
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Nesterov gradient descent with multiple inits ===
    print(f"\n=== Phase 1: Nesterov GD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 300.0)
    
    if t_phase1 > 60:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(project_simplex(parent))
            labels.append("parent")
        
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal")
        inits.append(spectral_init(n0, 30, 42))
        labels.append("spectral")
        
        if parent is not None:
            blend = 0.5 * project_simplex(parent) + 0.5 * make_u_shape(n0, 2.0)
            inits.append(project_simplex(blend))
            labels.append("blend")
        
        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 15:
                break
            print(f"  NGD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = nesterov_gradient_descent(init, t_avail,
                                                   beta_init=10.0, beta_final=80.0,
                                                   eta_init=1.0, verbose=False,
                                                   rng_seed=42 + idx)
            try_update(a_out, f"ngd_{lbl}")
    
    # === Phase 2: LP-FW with outer approximation (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW OA (n={n_best}) ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 60:
        a_in = project_simplex(best_seq)
        a_out, _ = lp_frank_wolfe_oa(a_in, t2, solver, top_K_init=200, verbose=True)
        try_update(a_out, "lp_oa_main")
    
    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak shaving (n={n_best}) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 30:
        a_in = project_simplex(best_seq)
        a_out, _ = peak_shaving_simultaneous(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "peakshave")
    
    # === Phase 4: NGD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: NGD polish (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 30:
        a_in = project_simplex(best_seq)
        a_out, _ = nesterov_gradient_descent(a_in, t4,
                                               beta_init=50.0, beta_final=200.0,
                                               eta_init=0.5, verbose=False,
                                               rng_seed=200)
        try_update(a_out, "ngd_polish")
    
    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW OA round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 40:
        a_in = project_simplex(best_seq)
        a_out, _ = lp_frank_wolfe_a
