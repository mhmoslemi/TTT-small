```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Power-law family exploration for strong initializations
2. Adam optimizer on log-parameterization with active-set subgradients
3. Smooth max via log-sum-exp for all-active-subset optimization
4. Beale-Zangvier quadratic Remez for polishing
5. Local search with FFT-diff for micro-optimizations
Key insight: non-symmetric (increasing) sequences shift the convolution peak
to positions where fewer terms contribute, reducing the maximum asymptotically
to 4/3 (vs 2 for uniform).
"""
import numpy as np
import time

def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgrad_all(y, ac, n_active=15, use_smooth=False, beta=50.0):
    """Compute active-set subgradient. O(n*k_active)."""
    n = len(y)
    M = ac.max()
    if M < 1e-15:
        return np.zeros(n)
    if use_smooth and beta > 0:
        shifted = ac - M
        # Numerically stable log-sum-exp
        weights = np.exp(shifted * beta)
        total_w = weights.sum()
        if total_w < 1e-15:
            weights = np.zeros_like(ac)
            weights[np.argmax(ac)] = 1.0
            total_w = 1.0
        else:
            weights /= total_w
    else:
        if n_active >= len(ac):
            top_idx = np.arange(len(ac))
        else:
            top_idx = np.argpartition(ac, -n_active)[-n_active:]
        weights = np.zeros_like(ac)
        top_vals = ac[top_idx]
        w_sum = top_vals.sum()
        if w_sum < 1e-15:
            weights[top_idx[np.argmax(top_vals)]] = 1.0
            w_sum = 1.0
        weights[top_idx] = top_vals / w_sum
    nz = np.nonzero(weights > 1e-12)[0]
    if len(nz) == 0:
        nz = np.array([int(np.argmax(ac))])
    g = np.zeros(n)
    for k in nz:
        k_int = int(k)
        w_k = weights[k_int]
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += w_k * 2.0 * y[k_int - idx]
    return g

def adam_logspace(a0, time_budget, n_active=15, use_smooth=True, smooth_beta=80.0, lr=0.005, verbose=False):
    """Adam optimizer on log-parameterization with active-set subgradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    theta = np.log(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    stall = 0
    best_stall_thresh = 3000
    rng = np.random.default_rng(7 + n)
    t0 = time.monotonic()
    verbose_count = 0
    while time.monotonic() - t0 < time_budget:
        t += 1
        ac = fast_conv(a)
        M = ac.max()
        g_a = compute_subgrad_all(a, ac, n_active, use_smooth, smooth_beta)
        grad_theta = a * (g_a - a @ g_a)
        m = b1 * m + (1 - b1) * grad_theta
        v = b2 * v + (1 - b2) * grad_theta**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.exp(theta - theta.max())
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new /= s
        a = a_new
        obj = 2.0 * n * fast_conv(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > best_stall_thresh:
            strategy = int(rng.integers(4))
            if strategy == 0:
                a_p = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            elif strategy == 1:
                a_p = best_a.copy()
                idx = rng.choice(n, size=max(1, n // 5), replace=False)
                a_p[idx] *= rng.uniform(0.6, 1.4, len(idx))
            elif strategy == 2:
                a_p = best_a.copy()
                shift = int(rng.integers(1, max(2, n // 10)))
                a_p = np.concatenate([a_p[shift:], a_p[:shift]])
            else:
                a_p = best_a * (1.0 + 0.02 * rng.standard_normal(n))
            a_p = np.maximum(a_p, 1e-12)
            a_p /= a_p.sum()
            theta = np.log(a_p)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0
            best_stall_thresh = max(500, best_stall_thresh * 0.7)
            if use_smooth and smooth_beta < 200:
                smooth_beta = min(200.0, smooth_beta * 1.3)
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            verbose_count += 1
            print(f"      Adam t={t} ({elapsed:.0f}s): obj={best_obj:.8f}, stall={stall}, beta={smooth_beta:.0f}")
    return best_a, best_obj

def remez_quadratic(a0, time_budget, max_active=30, verbose=False):
    """Beale-Zangvier style quadratic Remez: max representing 2a*(a' - a) under constraints."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0
    r = 0.05
    base = a.copy()
    while time.monotonic() - t0 < time_budget:
        b = fast_conv(base)
        M = b.max()
        thresh = M * (1.0 - 0.002)
        active = np.where(b >= thresh)[0]
        if len(active) > max_active:
            active = np.argsort(b)[::-1][:max_active]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        conds = []
        valid_count = 0
        for k in active:
            k_i = int(k)
            j_start = max(0, k_i - n + 1)
            j_end = min(n - 1, k_i)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                ci = 2.0 * base[k_i - idx]
                rhs = b[k_i]
                ci_ext = np.zeros(n)
                ci_ext[idx] = ci
                conds.append((ci_ext, rhs))
                valid_count += 1
        if valid_count == 0:
            a_trial = base.copy()
        else:
            A = np.array([c[0] for c in conds])
            rhs = np.array([c[1] for c in conds])
            rhs = np.maximum(rhs, 0.0)
            a_trial = np.zeros(n)
            a_trial[np.argmax(base)] = rhs.min()
            lam = np.ones(valid_count) / valid_count
            for _ in range(200):
                viol = A @ a_trial - rhs
                if (viol < -1e-12).sum() == 0 and a_trial.min() > -1e-12:
                    break
                active_c = viol > -1e-10
                if active_c.sum() == 0:
                    break
                avg_rhs = rhs[active_c].mean()
                a_trial = np.maximum(avg_rhs * np.ones(n), A[active_c].mean(axis=0))
                lam_new = np.zeros(valid_count)
                lam_new[active_c] = 1.0 / active_c.sum()
                if np.linalg.norm(lam_new - lam) < 1e-8:
                    lam = lam_new
                    break
                lam = lam_new
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                a_trial = base.copy()
            else:
                a_trial /= s
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_new = (1.0 - alpha) * base + alpha * a_trial
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            m_new = fast_conv(a_new).max()
            if m_new < best_m - 1e-14:
                base = a_new
                best_m = m_new
                best_a = a_new.copy()
                accepted = True
                r = min(0.15, r * 1.05)
                break
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.8)
            if no_improve > 25:
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 1e-10)
                base = a_pert / a_pert.sum()
                no_improve = 0
        if verbose and (no_improve % 10 == 0 or accepted):
            elapsed = time.monotonic() - t0
            print(f"      Remez ({elapsed:.0f}s): max={best_m:.8f}, r={r:.4f}, no_impr={no_improve}")
    return best_a, best_m

def local_search_polish(a0, time_budget, verbose=False):
    """Local search using FFT-diff for fast candidate evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_conv(a0).max()
    a = a / s
    best_a = a.copy()
    best_m = fast_conv(a).max()
    nbase = 2 * n - 1
    fft_size = 1 << (nbase - 1).bit_length()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = max(30, n // 50)
    # Precompute FFT of current a
    A_fft = np.fft.rfft(a, fft_size)
    B_fft = A_fft * A_fft
    conv_base = np.fft.irfft(B_fft, fft_size)[:nbase]
    M_base = conv_base.max()
    while time.monotonic() - t0 < time_budget and stall < max_stall:
        improved = False
        n_tries = min(80, n // 4)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_budget:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                if amt < 1e-15:
                    break
                delta = np.zeros(fft_size)
                delta[i] = -amt
                delta[j] = amt
                D = np.fft.rfft(delta, fft_size)
                conv_new = conv_base + 2.0 * np.fft.irfft(A_fft * D, fft_size)[:nbase] + np.fft.irfft(D * D, fft_size)[:nbase]
                M_new = conv_new.max()
                if M_new < M_base - 1e-14:
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s2 = a.sum()
                    a = a / s2
                    A_fft = np.fft.rfft(a, fft_size)
                    B_fft = A_fft * A_fft
                    conv_base = np.fft.irfft(B_fft, fft_size)[:nbase]
                    M_base = conv_base.max()
                    best_m = M_base
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            stall += 1
        else:
            stall = 0
        if verbose and stall % max(1, max_stall // 10) == 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalSearch ({elapsed:.0f}s): max={best_m:.8f}, stall={stall}")
    return best_a, best_m

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def make_power_law(n, p):
    """Create a[i] = (i+1)^p normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_exp_seq(n, alpha):
    """Create a[i] = exp(alpha*i/n) normalized."""
    x = np.linspace(0, 1, n)
    a = np.exp(alpha * x)
    s = a.sum()
    return a / s

def make_gaussian_offset(n, mu, sigma):
    """Gaussian centered at mu with width sigma, normalized."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    s = a.sum()
    return a / s

def interpolate_to_length(a, new_n):
    """Interpolate sequence to a different length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return result / rs

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")
    best_seq = None
    best_score = float('inf')
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    if best_seq is None:
        best_seq = make_power_law(n_base, 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init (power law p=1): n={n_base}, score={best_score:.8f}")
    n0 = len(best_seq)
    # Phase 1: Power-law family exploration (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.10
    print(f"\n=== Phase 1: Family exploration (budget={t1:.0f}s) ===")
    if t1 > 10:
        t_per = t1 / 16.0
        candidates = []
        for p in [0.3, 0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 2.5]:
            a_cand = make_power_law(n0, p)
            candidates.append(a_cand)
            sc = evaluate_safe(a_cand)
            if np.isfinite(sc):
                print(f"    power p={p}: score={sc:.8f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = a_cand.copy()
                    print(f"    -> NEW BEST: {sc:.8f}")
        for alpha in [0.5, 1.0, 2.0, 3.0]:
            a_cand = make_exp_seq(n0, alpha)
            sc = evaluate_safe(a_cand)
            if np.isfinite(sc):
                print(f"    exp alpha={alpha}: score={sc:.8f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = a_cand.copy()
                    print(f"    -> NEW BEST: {sc:.8f}")
        for mu in [0.3, 0.5, 0.7]:
            a_cand = make_gaussian_offset(n0, mu, 0.2)
            sc = evaluate_safe(a_cand)
            if np.isfinite(sc):
                print(f"    gauss mu={mu}: score={sc:.8f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = a_cand.copy()
                    print(f"    -> NEW BEST: {sc:.8f}")
        if parent_loaded:
            sc_p = evaluate_safe(best_seq)
            if sc_p < best_score:
                best_score = sc_p
    # Phase 2: Adam from best power-law (25% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Adam from best init (budget={t2:.0f}s, n={len(best_seq)}) ===")
    if t2 > 30:
        a_out, _ = adam_logspace(norm_sum1(best_seq), t2, n_active=15, use_smooth=True,
                                 smooth_beta=80.0, lr=0.005, verbose=True)
        try_update(a_out, f"Adam_phase2_n{len(best_seq)}")
    # Phase 3: Remez quadratic polish (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Remez polish (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = remez_quadratic(norm_sum1(best_seq), t3, max_active=30, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
    # Phase 4: Adam again with refined smooth (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: Adam refined (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = adam_logspace(norm_sum1(best_seq), t4, n_active=20, use_smooth=True,
                                 smooth_beta=150.0, lr=0.002, verbose=True)
        try_update(a_out, f"Adam_phase4_n{len(best_seq)}")
    # Phase 5: Adam from different inits (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.15
    print(f"\n=== Phase 5: Adam multi-init (budget={t5:.0f}s) ===")
    if t5 > 30:
        inits = [make_power_law(n0, 0.7), make_power_law(n0, 1.5), make_exp_seq(n0, 1.5)]
        t_per = t5 / len(inits)
        for i, a_init in enumerate(inits):
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 < 15:
                break
            a_out, _ = adam_logspace(a_init, min(t_per, remaining2 - 5), n_active=12,
                                     use_smooth=True, smooth_beta=50.0, lr=0.004, verbose=False)
            try_update(a_out, f"Adam_multi_{i}_n{n0}")
    # Phase 6: Local search polish (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: Local search (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = local_search_polish(norm_sum1(best_seq), t6, verbose=True)
        try_update(a_out, f"LocalSearch_n{len(best_seq)}")
    # Phase 7: Try longer length (5% time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n0 + 400
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_to_length(best_seq, n_alt)
        t7 = remaining * 0.7
        if t7 > 20:
            a_out, _ = adam_logspace(a_alt, t7, n_active=15, use_smooth=True,
                                     smooth_beta=80.0, lr=0.004, verbose=True)
            try_update(a_out, f"Adam_alt_n{n_alt}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = remez_quadratic(norm_sum1(best_seq), remaining - 5, max_active=30, verbose=False)
            try_update(a_out, f"Remez_final_{len(best_seq)}")
    # Final: one more Adam push on global best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Final push (budget={remaining:.0f}s) ===")
        a_out, _ = adam_logspace(norm_sum1(best_seq), remaining - 5, n_active=10,
                                 use_smooth=True, smooth_beta=200.0, lr=0.001, verbose=False)
        try_update(a_out, f"Final_n{len(best_seq)}")
    # Final validation
    if best_seq is None:
        best_seq = make_power_law(2400, 1.0)
        best_score = evaluate_safe(best_seq)
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = make_power_law(2400, 1.0)
        best_score = evaluate_safe(best_seq)
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    return [float(x) for x in best_seq]

def norm_sum1(a):
    """Normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
```
