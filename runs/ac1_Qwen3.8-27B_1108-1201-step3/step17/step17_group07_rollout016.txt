    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative gradient descent (mirror descent) with adaptive top-K subgradient
2. Aggressive LP Remez with large trust regions for polishing
3. Multi-scale approach: optimize at small n first, then upscale
4. Coordinate descent + random pair moves for final polish
5. Multi-length and multi-init exploration

Key insight: The subgradient at lag k is 2*a[k-j] (a shifted copy of a),
making each gradient step O(n*K). The soft gradient (log-sum-exp weighted)
can be computed via FFT for full O(n log n) steps.

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient_topk(a, b, top_k):
    """Compute subgradient from top-K lags of b."""
    n = len(a)
    size = len(b)
    k_max = min(top_k, size)
    if k_max == size:
        top_idx = np.arange(size)
    else:
        top_idx = np.argpartition(b, -k_max)[-k_max:]
    
    g = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[ki - idx]
    g /= k_max
    return g

def compute_soft_gradient_fft(a, b, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*b)) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Softmax weights
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()
    
    # g[j] = (2/beta) * sum_k w[k] * a[k-j] for valid k-j
    # = (2/beta) * (cross_correlation of w and a)[j]
    # Cross-correlation = conv(flip(w), a)
    w_flip = np.zeros(fft_size)
    w_flip[:size] = w[::-1]
    
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    W = np.fft.rfft(w_flip, fft_size)
    
    g = np.fft.irfft(A * W, fft_size)[:n] * (2.0 / beta)
    return g

def multiplicative_grad_descent(a0, time_limit, verbose=False, rng_seed=0):
    """Multiplicative gradient descent with adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Adaptive parameters
    eta = 0.5  # initial step size
    beta = 50.0  # log-sum-exp temperature
    stall = 0
    max_stall = 500
    use_soft = True  # use FFT-based soft gradient
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if (t - 1) % 500 == 0 and time.monotonic() - t0 > time_limit:
            break
            
        b = fast_conv(a)
        M = b.max()
        
        # Check for improvement
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive step size
        eta = 0.8 / (1.0 + 0.01 * t)
        beta = min(200.0, 20.0 * (1.0 + 0.005 * t))
        
        # Compute gradient
        if use_soft:
            k_adapt = max(10, int(1.5 * np.sum(b >= M * 0.99)))
            if k_adapt > 50 and t > 100:
                try:
                    g = compute_soft_gradient_fft(a, b, beta)
                except Exception:
                    g = compute_subgradient_topk(a, b, min(k_adapt, 80))
            else:
                g = compute_subgradient_topk(a, b, min(max(k_adapt, 10), 80))
        else:
            g = compute_subgradient_topk(a, b, 20)
        
        # Multiplicative update (mirror descent step)
        # a_new[j] = a[j] * exp(-eta * g[j]) / normalization
        log_a = np.log(np.maximum(a, 1e-300)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Line search: try damped versions
        best_trial_m = fast_conv(a_new).max()
        if best_trial_m >= best_m + 1e-10:
            # Damp
            for alpha in [0.5, 0.3, 0.1, 0.05]:
                a_trial = (1 - alpha) * a + alpha * a_new
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial /= st
                    mt = fast_conv(a_trial).max()
                    if mt < best_m:
                        a_new = a_trial
                        best_trial_m = mt
                        break
            else:
                a_new = a  # no update
        
        a = a_new
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = max(200, int(max_stall * 0.7))
            strategy = int(rng.integers(4))
            if strategy == 0:
                a = best_a * (1 + rng.normal(0, 0.03, n))
            elif strategy == 1:
                a = best_a.copy()
                idx = rng.choice(n, size=min(n//3, 800), replace=False)
                a[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                # Frequency domain perturbation
                A_fft = np.fft.rfft(a)
                A_fft[np.random.choice(len(A_fft), size=n//10, replace=False)] *= rng.uniform(0.8, 1.2, n//10)
                a = np.fft.irfft(A_fft, n)
            else:
                a = best_a * (1 + rng.normal(0, 0.01, n))
            
            a = np.maximum(a, 1e-10)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall}, eta={eta:.4f}")
    
    return best_a, best_m

def lp_remez_aggressive(a, max_cons, r, solver=None):
    """Aggressive LP Remez step with large trust region."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active lags - use all lags above 99.5% of max
    threshold = max_b * 0.995
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        # Take the most important ones
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    # Linearized autocorrelation constraints
    for k in active:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[ki - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[ki])
    
    # Trust region (allow large changes)
    constraints.append(a_var <= a * (1.0 + r) + 1e-6)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(solver=solver, verbose=False)
    except Exception:
        return a.copy(), b.max()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), b.max()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), b.max()
    a_new = a_new / s
    m_new = fast_conv(a_new).max()
    return a_new, m_new

def remez_loop(a0, time_limit, solver=None, r_init=0.3, max_cons=50, verbose=False):
    """LP Remez with aggressive trust region and adaptive shrinking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(777)
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand, m_cand = lp_remez_aggressive(a, max_cons, r, solver)
        
        # Line search with multiple step sizes
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1 - alpha) * a + alpha * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                r = min(0.8, r * 1.05)  # slowly increase trust region
                no_improve = 0
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.05, r * 0.8)
            if no_improve > 15:
                # Perturb to escape
                idx = rng.choice(n, size=min(n//4, 600), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                r = max(r, 0.2)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.3f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, rng_seed=42):

    n = len(a0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-5.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'sine_bell':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'asym':
        a = np.exp(-3.0 * (x - 0.4) ** 2 / 0.06)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            if t6b > 20:
                a_out, _ = remez_loop(norm_sum1(best_seq), t6b, solver=solver, r_init=0.15, max_cons=70, verbose=False)
                try_update(a_out, f"Remez_alt_{n_best}")
    
    # === Phase 7: CD polish (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.07, 80.0)
    print(f"\n=== Phase 7: CD polish n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t7, rng_seed=314)
        try_update(a_out, f"CD_{len(best_seq)}")
    
    # === Phase 8: Final Remez + MGD push ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final push n={n_best} (budget={remaining:.0f}s) ===")
        t8a = min(remaining * 0.5, 100.0)
        if t8a > 15:
            a_out, _ = multiplicative_grad_descent(norm_sum1(best_seq), t8a, verbose=False, rng_seed=2718)
            try_update(a_out, f"MGD_final_{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = remez_loop(norm_sum1(best_seq), remaining - 5, solver=solver, r_init=0.05, max_cons=100, verbose=False)
            try_update(a_out, f"Remez_last_{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
