```python
def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(p, a):
    """Compute cross-correlation (p ⋆ a)[i] = sum_k p[k]*a[k-i] using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(np.asarray(p, dtype=np.float64), fft_size)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    # Cross-correlation = ifft(conj(P) * A)
    result = np.fft.irfft(np.conj(P) * A, fft_size)[:size]
    return result
```

```python
    a = a0.copy()

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break

        y = a + momentum * v

        b_y = fast_conv(y)
        M = b_y.max()
        top_idx = np.argpartition(b_y, -top_k)[-top_k:]

        for k in top_idx:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:

                g[idx] += 2.0 * y[k_int - idx]
```

```python
def project_onto_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def fast_grad_conv(a, p):
    """Compute grad_f[i] = 2 * sum_k p[k] * a[k-i] using FFT.
    a has length n, p has length 2n-1.
    Returns array of length n."""
    n = len(a)
    q = p[::-1]  # flip(p)
    conv_len = len(q) + n - 1  # = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    Q = np.fft.rfft(q, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(Q * A, fft_size)
    # We need elements at positions 2n-2 through 3n-3
    start = 2 * n - 2
    return 2.0 * conv[start:start+n]
```

```python
def fast_grad_conv(a, p):
    """Compute grad_f[i] = 2 * sum_k p[k] * a[k-i] using FFT.
    a has length n, p has length 2n-1.
    Returns array of length n."""
    n = len(a)
    p_rev = p[::-1]
    conv_len = len(p_rev) + n - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(P * A, fft_size)
    # Extract: conv[n-1 : 2*n-1] reversed
    result = conv[n-1 : 2*n-1][::-1]
    return 2.0 * result
```
