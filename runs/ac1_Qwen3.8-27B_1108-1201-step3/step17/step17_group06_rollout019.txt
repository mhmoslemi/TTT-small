"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed L-BFGS optimization with adaptive soft-max parameter
2. Multiple initialization strategies
3. Length variation search
4. Coordinate descent polishing
5. Riemannian optimization on the probability simplex
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J_normalized(a):
    """Compute J = 2*n*max(conv(a,a)) for a normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def softmax(x):
    """Numerically stable softmax."""
    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()

def smooth_objective(x, beta):
    """
    Compute (1/beta)*log(sum_k exp(beta*c_k)) / (sum(x_softmax))^2 ... 
    Actually, since we parameterize a = softmax(x), sum(a) = 1 always.
    So objective = 2*n*(1/beta)*log(sum_k exp(beta*c_k))
    where c = conv(a,a).
    """
    a = softmax(x)
    n = len(a)
    conv = fast_conv(a)
    # Soft-max of conv
    c_max = conv.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max
    exp_bc = np.exp(beta * (conv - c_max))
    log_sum_exp = c_max + np.log(exp_bc.sum())
    soft_max_val = log_sum_exp / beta
    return 2.0 * n * soft_max_val

def smooth_objective_grad(x, beta):
    """
    Gradient of the smoothed objective w.r.t. x.
    
    F(x) = 2n * (1/beta) * log(sum_k exp(beta * c_k))
    where c_k = conv(a,a)_k, a = softmax(x)
    
    dF/dx_i = 2n * sum_k p_k * dc_k/da_i * da_i/dx_i
    
    where p_k = exp(beta*c_k) / sum_j exp(beta*c_j)
    
    dc_k/da_i = a_{k-i} + a_{k-i}*[2i=k] (for valid indices)
    
    da_i/dx_i = a_i * (1 - a_i)
    da_j/dx_i = -a_i * a_j for j != i
    
    So: dF/dx_i = 2n * a_i * (g_i - sum_j g_j * a_j)
    where g_i = sum_k p_k * dc_k/da_i
    """
    a = softmax(x)
    n = len(a)
    conv = fast_conv(a)
    
    c_max = conv.max()
    exp_bc = np.exp(beta * (conv - c_max))
    p = exp_bc / exp_bc.sum()
    
    # Compute g = sum_k p_k * (dc/da)
    g = np.zeros(n)
    # Only iterate over significant p values
    threshold = 1e-12
    active = p > threshold
    active_idx = np.where(active)[0]
    
    for k in active_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        # dc_k/da_i = a_{k-i} + a_{k-i} if 2*i == k
        grad_k = a[j_idx].copy()
        double = (2 * idx == k_int)
        if double.any():
            grad_k[double] += a[j_idx[double]]
        g[idx] += p[k] * grad_k
    
    # Transform to gradient w.r.t. x
    mean_g = np.dot(g, a)
    grad_x = 2.0 * n * a * (g - mean_g)
    
    return grad_x

def optimize_smoothed(a_start, time_limit, beta_start=5.0, beta_end=50.0, 
                       n_beta_steps=5, verbose=False):
    """
    Optimize using smoothed L-BFGS with adaptive beta.
    """
    n = len(a_start)
    best_a = a_start.copy()
    best_val = eval_J_normalized(best_a)
    t0 = time.monotonic()
    
    # Convert to log domain
    a_log = np.maximum(a_start, 1e-30)
    x0 = np.log(a_log / a_log.sum())
    
    for step in range(n_beta_steps):
        frac = step / max(n_beta_steps - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        t_step = (time.monotonic() - t0)
        t_remaining = time_limit - t_step
        if t_remaining < 10:
            break
        
        t_this_step = t_remaining / (n_beta_steps - step) * 0.9
        
        if verbose:
            print(f"    Beta step {step}: beta={beta:.1f}, t_avail={t_this_step:.0f}s")
        
        def fun(x):
            return smooth_objective(x, beta)
        
        def grad(x):
            return smooth_objective_grad(x, beta)
        
        # L-BFGS with time limit
        # scipy's L-BFGS-B doesn't have a built-in time limit, so we use maxiter
        # Estimate iterations: each gradient eval takes ~T seconds
        t_test_start = time.monotonic()
        _ = grad(x0)
        t_grad = time.monotonic() - t_test_start
        
        max_iter = max(10, int(t_this_step / (t_grad * 3)))
        
        result = minimize(fun, x0, jac=grad, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'gtol': 1e-8, 'maxfun': max_iter * 2})
        
        # Convert back
        a_new = softmax(result.x)
        val = eval_J_normalized(a_new)
        
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            x0 = result.x.copy()
            if verbose:
                print(f"    Improved to J={best_val:.6f} at beta={beta:.1f}")
        else:
            # Keep x0 as is if no improvement
            pass
    
    return best_a, best_val

def coordinate_descent_smooth(a0, time_limit, beta=20.0, verbose=False):

    a = a / s

        conv = fast_conv(a)
        c_max = conv.max()
        exp_bc = np.exp(beta * (conv - c_max))
        p = exp_bc / exp_bc.sum()

        g_a = np.zeros(n)

                grad_k[double] += a[j_idx[double]]
            g_a[idx] += p[k] * grad_k

        g_projected = g_a - np.mean(g_a)

        # We want to decrease the objective, so move in -gradient direction
        # But we need to maintain simplicity: pick the most "negative" projected gradient
        # with significant mass

            stall += 1
            if stall > 100:

            continue

        # Actually, negative gradient means decreasing a_i would increase objective
        # So we want to increase a_i where gradient is most negative
        src = candidates[np.argmin(g_projected[candidates])]
        
        # Find destination: where gradient is most positive (decreasing a_j helps most)
        pos_candidates = np.where(g_projected > 1e-10)[0]
        if len(pos_candidates) == 0:
            stall += 1
            continue
        dst = pos_candidates[np.argmax(g_projected[pos_candidates])]
        
        # Try transfer
        improved = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-15:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            val = eval_J_normalized(a_t)
            if val < best_val - 1e-14:
                a = a_t
                best_val = val
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    val = eval_J_normalized(a)
                    if val < best_val:
                        best_val = val
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_val

def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_bimodal(n, width=80.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.003
    return a / a.sum()

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 8):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

def fast_gradient_G(a, p, n):
    """Compute G_i = sum_m a_m * p_{m+i} + p_{2i} * a_i for i=0,...,n-1"""
    a_rev = a[::-1]
    # Full convolution of a_rev (length n) and p (length 2n-1)
    size = n + (2*n - 1) - 1  # = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_ap = np.fft.irfft(A * P, fft_size)[:size]
    
    G = conv_ap[n-1 : 2*n-1].copy()  # indices n-1 to 2n-2
    # Add correction: p[2i] * a[i]
    G += p[0 : 2*n-1 : 2] * a  # p[0], p[2], p[4], ..., p[2n-2]
    
    return G

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed L-BFGS with O(n log n) FFT-based gradient via cross-correlation
2. Adaptive soft-max beta schedule for smooth-to-sharp transition
3. Multiple initialization and length exploration
4. Subgradient coordinate descent for final polishing
5. IFTA-inspired spectral smoothing as auxiliary method

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J_normalized(a):
    """Compute J = 2*n*max(conv(a,a)) for a normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def fast_gradient_G(a, p, n):
    """
    Compute G_i = sum_m a_m * p_{m+i} + p_{2i} * a_i for i=0,...,n-1
    using FFT-based convolution. O(n log n).
    """
    a_rev = a[::-1]
    p_len = len(p)  # 2n-1
    size = n + p_len - 1  # 3n-2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_ap = np.fft.irfft(A * P, fft_size)[:size]
    # G[i] = conv_ap[i + n - 1] for i=0,...,n-1
    G = conv_ap[n - 1: 2 * n - 1].copy()
    # Add correction: p[2i] * a[i]
    G += p[0:2 * n - 1: 2] * a
    return G


def smooth_objective(x, beta):
    """Compute smoothed objective value."""
    a = softmax(x)
    n = len(a)
    conv = fast_conv(a)
    c_max = conv.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max
    exp_bc = np.exp(beta * (conv - c_max))
    log_sum = c_max + np.log(exp_bc.sum())
    return 2.0 * n * log_sum / beta


def smooth_objective_grad(x, beta):
    """Compute gradient of smoothed objective using O(n log n) FFT method."""
    a = softmax(x)
    n = len(a)
    conv = fast_conv(a)
    c_max = conv.max()
    exp_bc = np.exp(beta * (conv - c_max))
    p = exp_bc / exp_bc.sum()

    G = fast_gradient_G(a, p, n)
    H = np.dot(a, G)
    grad_x = 2.0 * n * a * (G - H)
    return grad_x


def optimize_lbfgs(a_start, time_limit, beta_start=3.0, beta_end=30.0,
                   n_steps=5, verbose=True):
    """
    Optimize using L-BFGS with adaptive beta schedule.
    """
    n = len(a_start)
    best_a = a_start.copy()
    best_val = eval_J_normalized(best_a)
    t0 = time.monotonic()

    # Convert to log domain
    a_safe = np.maximum(a_start, 1e-30)
    x0 = np.log(a_safe / a_safe.sum())

    for step in range(n_steps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break
        frac = step / max(n_steps - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        t_remaining = time_limit - elapsed
        t_this = t_remaining / (n_steps - step) * 0.85

        if verbose:
            print(f"      [L-BFGS] beta={beta:.1f}, t_avail={t_this:.0f}s")

        # Time a single gradient to estimate max iterations
        t_g0 = time.monotonic()
        _ = smooth_objective_grad(x0, beta)
        t_grad = max(time.monotonic() - t_g0, 0.001)

        max_iter = max(5, int(t_this / (t_grad * 2.5)))

        def fun(x):
            return smooth_objective(x, beta)

        def grad(x):
            return smooth_objective_grad(x, beta)

        try:
            result = minimize(fun, x0, jac=grad, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'gtol': 1e-10,
                                      'maxfun': max_iter * 3, 'ftol': 1e-12})
            a_new = softmax(result.x)
            val = eval_J_normalized(a_new)
            if val < best_val - 1e-15:
                best_val = val
                best_a = a_new.copy()
                x0 = result.x.copy()
                if verbose:
                    print(f"      [L-BFGS] Improved: J={best_val:.7f}")
            else:
                if verbose:
                    print(f"      [L-BFGS] No improvement (J={val:.7f} vs {best_val:.7f})")
        except Exception as e:
            if verbose:
                print(f"      [L-BFGS] Error: {e}")

    return best_a, best_val


def subgradient_coord_descent(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent using exact subgradient (single peak).
    Transfers mass from high-contribution positions to low-contribution ones.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = eval_J_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        k_max = int(np.argmax(conv))

        # Compute subgradient at the peak
        # dc_kmax/da_i = a[kmax-i] + a[kmax-i]*[2i=kmax]
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_max - idx
        grad_peak = a[j_idx].copy()
        double = (2 * idx == k_max)
        if double.any():
            grad_peak[double] += a[j_idx[double]]

        # Project to tangent space of simplex
        g = np.zeros(n)
        g[idx] = grad_peak
        g_proj = g - np.mean(g)

        # Source: position where we want to increase (most negative projected grad)
        # Dest: position where we want to decrease (most positive projected grad)
        neg_mask = g_proj < -1e-14
        pos_mask = g_proj > 1e-14

        if not neg_mask.any() or not pos_mask.any():
            # Try top-2 peaks
            if np.sum(conv > M * 0.95) > 1:
                order = np.argsort(conv)[::-1]
                k2 = int(order[1])
                i2s = max(0, k2 - n + 1)
                i2e = min(n - 1, k2)
                idx2 = np.arange(i2s, i2e + 1)
                j2 = k2 - idx2
                g2 = np.zeros(n)
                g2[idx2] = a[j2]
                d2 = (2 * idx2 == k2)
                if d2.any():
                    g2[d2] += a[j2[d2]]
                g_total = g + g2
                g_total -= np.mean(g_total)
                neg_mask = g_total < -1e-14
                pos_mask = g_total > 1e-14
                if not neg_mask.any() or not pos_mask.any():
                    stall += 1
                    continue
            else:
                stall += 1
                if stall > 20:
                    break
                continue

        src = int(np.argmin(np.where(neg_mask, g_proj, 0)))
        dst = int(np.argmax(np.where(pos_mask, g_proj, 0)))

        if src == dst or a[src] < 1e-14:
            stall += 1
            continue

        # Try multiple transfer fractions
        improved = False
        for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.1, 0.2]:
            amt = a[src] * frac
            if amt < 1e-16:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-15:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            val = eval_J_normalized(a_t)
            if val < best_val - 1e-14:
                a = a_t
                best_val = val
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.005)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        val = eval_J_normalized(a)
                        if val < best_val:
                            best_val = val
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 200 == 0 and stall > 0:
            print(f"      [CD] stall={stall}, J={best_val:.7f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_val


def ifta_smooth(a0, n_iters=200, clip_frac=0.01):
    """
    IFTA-inspired spectral smoothing: iteratively flatten the autocorrelation peaks.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    a /= a.sum()
    a_curr = a.copy()

    for it in range(n_iters):
        conv = fast_conv(a_curr)
        M = conv.max()
        # Soft clip: reduce peaks
        threshold = M * (1.0 - clip_frac)
        excess = np.maximum(conv - threshold, 0)
        conv_new = conv - 0.5 * excess

        # Get spectrum: FFT(conv) = |FFT(a)|^2
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        C_fft = np.fft.rfft(conv_new, fft_size)

        # Check if real part is non-negative (valid autocorrelation spectrum)
        if np.min(C_fft.real) < -1e-10:
            # Fallback: use original
            C_fft_orig = np.fft.rfft(conv, fft_size)
            max_neg = -np.min(C_fft.real)
            C_fft = C_fft + max_neg * np.ones_like(C_fft)
            # This shifts the spectrum, which adds a delta at lag 0... approximate
            # Better: just reduce the clipping
            conv_new = conv - 0.1 * excess
            C_fft = np.fft.rfft(conv_new, fft_size)
            if np.min(C_fft.real) < -1e-10:
                C_fft = np.maximum(C_fft.real, 0) + 1j * C_fft.imag

        # A_fft = sqrt(C_fft) (element-wise, taking principal square root)
        # Since C_fft should be real and >= 0 for valid autocorrelation
        C_mag = np.sqrt(np.maximum(C_fft.real, 0))
        A_fft = C_mag  # Real, non-negative spectrum

        # Inverse FFT to get time domain
        a_new = np.fft.irfft(A_fft, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a_curr = a_new

    return a_curr


def make_power_ramp(n, p=1.0):

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

    print(f"\n=== Phase 1: L-BFGS beta 3->40 (n={n0}, budget={remaining:.0f}s) ===")
    t1 = remaining * 0.30
    a_out, _ = optimize_lbfgs(best_a.copy(), t1,
                               beta_start=3.0, beta_end=40.0,
                               n_steps=6, verbose=True)
    try_update(a_out, "lbgfs_p1")

    # === Phase 2: High-beta L-BFGS refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: L-BFGS beta 20->100 (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.20
    a_out, _ = optimize_lbfgs(best_a.copy(), t2,
                               beta_start=20.0, beta_end=100.0,
                               n_steps=5, verbose=True)
    try_update(a_out, "lbgfs_p2")

    # === Phase 3: Subgradient coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Subgradient CD (budget={remaining:.0f}s) ===")
    t3 = remaining * 0.12
    a_out, _ = subgradient_coord_descent(best_a.copy(), t3, verbose=True, seed=7)
    try_update(a_out, "cd_p3")

    # === Phase 4: L-BFGS after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BFGS after CD (budget={remaining:.0f}s) ===")
    t4 = remaining * 0.15
    a_out, _ = optimize_lbfgs(best_a.copy(), t4,
                               beta_start=30.0, beta_end=80.0,
                               n_steps=4, verbose=True)
    try_update(a_out, "lbgfs_p4")

    # === Phase 5: IFTA smoothing + L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: IFTA + L-BFGS (budget={remaining:.0f}s) ===")
    if remaining > 40:
        a_ifta = ifta_smooth(best_a.copy(), n_iters=300, clip_frac=0.005)
        try_update(a_ifta, "ifta")
        t5 = min(remaining * 0.25, 120.0)
        if t5 > 15:
            a_out, _ = optimize_lbfgs(a_ifta, t5,
                                       beta_start=5.0, beta_end=50.0,
                                       n_steps=4, verbose=True)
            try_update(a_out, "ifta_lbfgs")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Exploration (budget={remaining:.0f}s) ===")
    n_cur = len(best_a)
    # Try nearby lengths
    length_candidates = []
    for d in [-200, -100, 100, 200, 400, -400]:
        nl = n_cur + d
        if 1500 <= nl <= 6000:
            length_candidates.append(nl)

    for nl in sorted(length_candidates):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 35:
            break
        if abs(nl - n_cur) <= 30:
            continue
        a_int = interpolate_seq(best_a, nl)
        v_int = eval_J_normalized(a_int)
        if v_int < best_score:
            best_score = v_int
            best_a = a_int.copy()
            print(f"  Interpolated to n={nl}: J={v_int:.7f}")
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 12:
            a_out, _ = optimize_lbfgs(a_int, t_md,
                                       beta_start=5.0, beta_end=40.0,
                                       n_steps=3, verbose=False)
            try_update(a_out, f"len_{nl}")

    # === Phase 7: Final high-beta L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final L-BFGS beta 50->300 (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.30
    if t7 > 15:
        a_out, _ = optimize_lbfgs(best_a.copy(), t7,
                                   beta_start=50.0, beta_end=300.0,
                                   n_steps=4, verbose=True)
        try_update(a_out, "final_lbfgs")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final CD (budget={remaining:.0f}s) ===")
    t8 = remaining * 0.35
    if t8 > 10:
        a_out, _ = subgradient_coord_descent(best_a.copy(), t8, verbose=True, seed=314)
        try_update(a_out, "final_cd")

    # === Phase 9: One more L-BFGS if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        print(f"\n=== Phase 9: Last L-BFGS (budget={remaining:.0f}s) ===")
        a_out, _ = optimize_lbfgs(best_a.copy(), remaining - 5,
                                   beta_start=100.0, beta_end=500.0,
                                   n_steps=3, verbose=False)
        try_update(a_out, "last_lbfgs")

    # Final validation
    if best_a is not None:
        final_J = eval_J_normalized(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.7f}, n={len(best_a)}, time={elapsed:.0f}s")
        return [float(x) for x in best_a]
    else:
        n_fb = 3200
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J_normalized(fb):.7f}")
        return [float(x) for x in fb]
