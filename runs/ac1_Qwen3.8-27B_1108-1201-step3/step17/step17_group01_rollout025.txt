import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    return ac

def softmax_gradient(a, beta):
    """Compute the gradient of the log-sum-exp approximation to max(ac).
    
    Returns g where g_i = 2 * sum_k w_k * a_{|k-i|}
    with w = softmax(beta * ac).
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Compute softmax weights (with numerical stability)
    ac_centered = ac - ac.max()
    w = np.exp(beta * ac_centered)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / w_sum
    
    # Compute g_i = 2 * sum_k w_k * a_{|k-i|} via FFT
    # S_a[i] = C[i] + (a*b)[i] - a[0]*b[i]
    # where C = cross_corr(a, w), (a*b) = conv(a, w)
    
    N = 1 << (3 * n - 2).bit_length()  # FFT size >= 3n-2
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    
    # Cross-correlation: C[i] = sum_j a[j] * w[i+j]
    # FFT(C) = conj(FFT(a)) * FFT(w)  (for real signals)
    C = np.fft.irfft(np.conj(A) * W, N)[:n]
    
    # Convolution: (a*w)[i] = sum_j a[j] * w[i-j]
    # FFT(a*w) = FFT(a) * FFT(w)
    conv_aw = np.fft.irfft(A * W, N)[:n]
    
    S_a = C + conv_aw - a[0] * w[:n]
    g = 2.0 * S_a
    
    return g, ac, w

def normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def mirror_descent_step(a, g, eta):
    """Multiplicative weights update: a_new_i ∝ a_i * exp(-eta * g_i)"""
    log_factor = -eta * g
    # Numerical stability
    log_factor -= log_factor.max()
    factor = np.exp(log_factor)
    a_new = a * factor
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def spectral_shaping_step(a, n_iters=50, eta=0.5):
    """Gerchberg-Saxton-like spectral flattening step."""
    n = len(a)
    a = normalize_nonneg(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Compress dynamic range
        P_new = P_mean * (P / P_mean) ** (1.0 - eta)
        P_new = np.maximum(P_new, 1e-30)
        
        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = normalize_nonneg(a)
    
    return best_a

                    if m_trial < best_m - 1e-14:

                        best_m = m_trial

            i = int(rng.integers(n))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated softmax mirror descent with Nesterov acceleration
2. Adaptive temperature schedule for the soft-max approximation to the max
3. Spectral shaping (Gerchberg-Saxton) as periodic perturbation

5. Coordinate descent polish on top autocorrelation peaks

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    return ac


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_gradient(a, beta):
    """Compute gradient of log-sum-exp approx to max(autocorr(a)).
    Returns g where g_i = 2 * sum_k softmax(beta*ac)_k * a_{|k-i|}.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    ac = fast_autocorr(a)

    # Numerically stable softmax
    ac_shifted = ac - ac.max()
    w = np.exp(beta * ac_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / w_sum

    # Compute g_i = 2 * sum_k w_k * a_{|k-i|}
    # S_a[i] = cross_corr(a,w)[i] + conv(a,w)[i] - a[0]*w[i]
    N = 1 << (3 * n - 2).bit_length()

    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)

    # Cross-correlation: cross[i] = sum_j a[j] * w[j+i]
    cross = np.fft.irfft(np.conj(A) * W, N)[:n]

    # Convolution: conv[i] = sum_j a[j] * w[i-j]
    conv = np.fft.irfft(A * W, N)[:n]

    S_a = cross + conv - a[0] * w[:n]
    g = 2.0 * S_a

    return g


def mirror_descent_step(a, g, eta):
    """Multiplicative weights: a_new_i proportional to a_i * exp(-eta * g_i)."""
    log_factor = -eta * g
    log_factor -= log_factor.max()
    factor = np.exp(log_factor)
    a_new = a * factor
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def spectral_shaping_step(a, n_iters=200, eta=0.5):

    n = len(a)
    a = normalize_nonneg(a).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        P_new = P_mean * np.power(P / P_mean, 1.0 - eta)
        P_new = np.maximum(P_new, 1e-30)

        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-30:

            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = normalize_nonneg(a)

                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new

def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return a_interp / ss

def make_multihump(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return normalize_nonneg(a)


def make_random_smooth(n, seed=0, n_freq=8):

    x = np.linspace(0, 1, n)

        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return normalize_nonneg(a)

                                  c0=4.0, verbose=False)
        _try_update(a_out, "md_highbeta")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_in = normalize_nonneg(best_seq)
        a_out = coordinate_descent_polish(a_in, time_limit=t4, rng_seed=42)
        _try_update(a_out, "coord")

    # === Phase 5: Multi-n search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-n (base={n_best}) ===")
        candidates_n = [n_best + 500, n_best + 1500, n_best + 3000,
                        max(1000, n_best - 1000)]
        candidates_n = sorted(set(x for x in candidates_n if 500 <= x <= 10000))

        for new_n in candidates_n[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={new_n}...")
            a_interp = interpolate_seq(best_seq, new_n)
            _try_update(a_interp, f"interp_n{new_n}")

            t5 = min(remaining * 0.25, 150.0)
            if t5 > 20:
                a_out, _ = main_optimize(a_interp, t5, beta_start=3.0, beta_end=50.0,
                                          c0=5.0, verbose=False)
                _try_update(a_out, f"md_n{new_n}")

    # === Phase 6: Different initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 6: Alt inits (n={len(best_seq)}) ===")
        n_cur = len(best_seq)
        inits = [
            (make_sinc_like(n_cur, 4.0), "sinc4"),
            (make_sinc_like(n_cur, 8.0), "sinc8"),
            (make_multihump(n_cur, 5, 0.12), "5hump"),
            (make_random_smooth(n_cur, seed=77, n_freq=6), "rnd77"),
        ]
        t_per = max(15.0, remaining / (len(inits) + 1))
        for init, lbl in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 15:
                break
            a_out, _ = main_optimize(init, t_avail, beta_start=3.0, beta_end=50.0,
                                      c0=5.0, verbose=False)
            _try_update(a_out, f"alt_{lbl}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish (n={len(best_seq)}) ===")
        t7 = min(remaining * 0.35, 100.0)
        if t7 > 10:
            a_in = normalize_nonneg(best_seq)
            a_out, _ = main_optimize(a_in, t7, beta_start=20.0, beta_end=100.0,
                                      c0=3.5, verbose=False)
            _try_update(a_out, "md_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.3, 60.0)
            if t7b > 10:
                a_in = normalize_nonneg(best_seq)
                a_out = coordinate_descent_polish(a_in, time_limit=t7b, rng_seed=999)
                _try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found, using fallback")
        n_fb = 3000
        best_seq = make_sinc_like(n_fb, 5.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
