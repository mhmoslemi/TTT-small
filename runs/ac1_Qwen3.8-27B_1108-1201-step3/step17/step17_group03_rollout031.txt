The gradient of this smooth function is easy to compute, and we can use accelerated proximal gradient (FISTA) on the simplex.

The gradient of L(a) = (1/β) * log(sum_k exp(β*c[k])) is:
∇L(a)_i = (1/β) * sum_k exp(β*c[k]) * β * ∂c[k]/∂a[i] / sum_k exp(β*c[k])
= sum_k p_k * 2*a[k-i] / sum_k p_k * exp(0)  ... 

where p_k = exp(β*c[k]) / sum_j exp(β*c[j]) (softmax probabilities).

= 2 * sum_{k: i in S_k} p_k * a[k-i]

This is a weighted sum of the shifted a's, where the weights are the softmax probabilities of the autocorrelation peaks.

def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lp_frank_wolfe_minimax(a0, time_limit, solver, top_k=None, verbose=True,
                           use_all_lags=True, nesterov=True, line_search=True):
    """
    LP-Frank-Wolfe with minimax LP subproblem.
    
    At each iteration:
    1. Solve: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0
    2. Nesterov step: x <- y + beta*(y - x_prev)
    3. Update: y <- (1-alpha)*y + alpha*b
    """
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()  # Nesterov momentum variable
    x_prev = y.copy()
    
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    num_lags = 2 * n - 1
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build the constraint matrix: A[k, i] = a[k-i] for valid (k,i)
        # (a*b)[k] = sum_i A[k,i] * b[i] <= t
        # For efficiency, only include active lags if top_k is specified
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if use_all_lags or top_k is None:
            active_idx = np.arange(num_lags)
        else:
            # Include lags that are within a certain fraction of the max
            # Plus always include the top-k
            threshold = M_curr * 0.95
            active_mask = ac >= threshold
            top_idx = np.argsort(ac)[::-1][:top_k]
            active_idx = np.unique(np.concatenate([np.where(active_mask)[0], top_idx]))
        
        K = len(active_idx)
        
        # Build constraint matrix for active lags
        # A[k, i] = y[k-i] if 0 <= k-i < n, else 0
        A_mat = np.zeros((K, n))
        for row, k in enumerate(active_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                A_mat[row, i] = y[k_int - i]
        
        # Solve: min t s.t. A_mat @ b <= t, sum(b) = 1, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [A_mat @ b_var <= t_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-12:
            continue
        b_val = b_val / b_sum
        
        # Frank-Wolfe step with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        
        y_new = (1.0 - alpha) * y + alpha * b_val
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        else:
            x_new = y_new.copy()
        
        # Line search: try a few step sizes and pick the best
        if line_search:
            best_y = y_new
            best_y_m = fast_autocorr(y_new).max()
            for frac in [0.5, 1.0, 1.5, 2.0]:
                a_trial = (1.0 - alpha * frac) * y + alpha * frac * b_val
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_y_m:
                    best_y_m = m_trial
                    best_y = a_trial.copy()
            y_new = best_y
            
            if nesterov and t > 1:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                s_x = x_new.sum()
                if s_x < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / s_x
        
        y = y_new
        x = x_new
        
        # Update best
        m_y = fast_autocorr(y).max()
        if m_y < best_m:
            best_m = m_y
            best_a = y.copy()
        
        m_x = fast_autocorr(x).max()
        if m_x < best_m:
            best_m = m_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, t_val={prob.value:.6f}")
    
    return best_a, best_m


def coordinate_descent_peak(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent: move mass from peak contributors to flat areas."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        
        if total_contrib < 1e-20:
            stall += 50
            break
        
        # Try moving mass from top contributors to random destinations
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        
        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            best_new_m = M_curr
            best_move = None
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(50):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)
                
                if best_move is not None:
                    break
            
            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 40 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD final ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    """Perturb in frequency domain to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    """Generate parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_constraint_matrix(y, active_lags):
    """Build A where A[row, i] = y[k - i] for the given active lags."""
    n = len(y)
    k_arr = np.asarray(active_lags, dtype=np.int64)

def lp_frank_wolfe_minimax(a0, time_limit, solver, top_k=None, verbose=True,
                           nesterov=True, line_search=True):
    """
    LP-Frank-Wolfe with minimax LP subproblem.
    Solves: min t s.t. (a*b)[k] <= t for selected k, sum(b)=1, b>=0
    """
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()
    x_prev = y.copy()

    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    num_lags = 2 * n - 1

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if top_k is None:
            active_idx = np.arange(num_lags)
        else:
            threshold = M_curr * 0.90
            active_mask = ac >= threshold
            top_idx = np.argsort(ac)[::-1][:top_k]
            active_idx = np.unique(np.concatenate([np.where(active_mask)[0], top_idx]))

        K = len(active_idx)
        A_mat = build_constraint_matrix(y, active_idx)

        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [A_mat @ b_var <= t_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-12:
            continue
        b_val = b_val / b_sum

        alpha = 2.0 / (t + 2.0)

        if line_search:
            best_y = None
            best_y_m = float('inf')
            for frac in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                a_trial = (1.0 - alpha * frac) * y + alpha * frac * b_val
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_y_m:
                    best_y_m = m_trial
                    best_y = a_trial.copy()
            if best_y is None:
                best_y = y.copy()
            y_new = best_y
        else:
            y_new = (1.0 - alpha) * y + alpha * b_val
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        if nesterov and t > 1:

            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        else:
            x_new = y_new.copy()

        y = y_new
        x = x_new

        m_y = fast_autocorr(y).max()
        if m_y < best_m:
            best_m = m_y
            best_a = y.copy()

        m_x = fast_autocorr(x).max()
        if m_x < best_m:

def coordinate_descent_peak(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent: move mass from peak contributors to flat areas."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 50
            continue

        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(40):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

                if best_move is not None:
                    break

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    """Perturb phase in frequency domain to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    """Generate parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        return a_norm.copy(), sc

    """
    Main search function. Uses LP-Frank-Wolfe with minimax subproblem as
    the primary optimizer, combined with coordinate descent polish and
    multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick LP-FW warmup with top-K ===
    print(f"\n=== Phase 1: LP-FW warmup (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.07, 120.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_minimax(a_in, t1, solver, top_k=300,
                                           verbose=True, nesterov=True,
                                           line_search=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "warmup")

    # === Phase 2: Main LP-FW with ALL lags ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW all-lags main (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_minimax(a_in, t2, solver, top_k=None,
                                           verbose=True, nesterov=True,
                                           line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_all")

    # === Phase 3: LP-FW top-K fine-tune ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW top-K fine (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_minimax(a_in, t3, solver, top_k=150,
                                           verbose=True, nesterov=True,
                                           line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_fine")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: CD polish (n={n_best}) ===")
    t4 = min(remaining * 0.08, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_peak(a_in, t4, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd1")

    # === Phase 5: Frequency diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Freq diversify (n={n_best}) ===")
    rng = np.random.default_rng(99)
    t5_per = min(remaining * 0.04, 75.0)
    if t5_per > 25:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t5_per + 60:
                break
            a_pert = freq_perturb(best_seq, strength=0.012, rng=rng)
            a_out, _ = lp_frank_wolfe_minimax(a_pert, t5_per, solver, top_k=200,
                                               verbose=False, nesterov=True,
                                               line_search=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [300, 600, -300, 900]:
        new_n = n_best + delta
        if 2500 <= new_n <= 5200 and new_n != n_best:
            candidate_lengths.append(new_n)
    candidate_lengths = sorted(set(candidate_lengths))

    if candidate_lengths and remaining > 200:
        print(f"\n=== Phase 6: Length exploration ===")
        for new_n in candidate_lengths[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={new_n}")
            a_long = interpolate_sequence(best_seq, new_n)
            t6 = min(remaining * 0.12, 180.0)
            if t6 > 20:
                a_out, _ = lp_frank_wolfe_minimax(a_long, t6, solver, top_k=250,
                                                   verbose=False, nesterov=True,
                                                   line_search=True)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"n{new_n}")

    # === Phase 7: Final LP-FW all-lags ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW all-lags (n={n_best}) ===")
    t7 = min(remaining * 0.20, 300.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_minimax(a_in, t7, solver, top_k=None,
                                           verbose=True, nesterov=True,
                                           line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "final_lp")

    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD (n={n_best}) ===")
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_peak(a_in, t8, verbose=True, rng_seed=777)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "final_cd")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_parametric(3600, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
