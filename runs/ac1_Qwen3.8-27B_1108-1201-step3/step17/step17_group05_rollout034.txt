def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_subgrad_toppk(a, top_k=10, beta=10.0):
    """
    Compute soft subgradient focusing on top-K peak lags.
    Uses FFT for efficient computation.
    Returns (gradient, current_max).
    """
    n = len(a)
    ac = fast_autocorr(a)
    K = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:K]
    top_vals = ac[top_idx]
    
    # Soft-max weighting over top K lags
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(K) / K
    else:
        w_top = w_top / w_sum
    
    # Build full weight vector
    w = np.zeros(len(ac))
    for wi, idx in zip(w_top, top_idx):
        w[int(idx)] = wi
    
    # Compute gradient via FFT: grad[i] = 2 * sum_m w[m] * a[m-i] for valid m-i
    # = 2 * (a_rev * w)[n-1 : 2n-1] where a_rev = a[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev_padded = np.zeros(fft_size)
    a_rev_padded[:n] = a[::-1]
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()


def subgradient_optimize(a0, time_limit, verbose=True, seed=42):
    """
    Nesterov-accelerated mirror descent with adaptive top-K subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    grad_ema = np.zeros(n)
    ema_alpha = 0.3
    c0 = 4.0
    stall_count = 0
    restart_every = 3000
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        # Adaptive beta: start sharp, maintain sharpness
        progress = min(1.0, iter_count / 20000.0)
        beta_t = 25.0 - 15.0 * progress  # 25 -> 10
        top_k = 5 + int(5 * (1 - progress))  # 10 -> 5

        # Nesterov extrapolation
        beta_nest = (iter_count - 1.0) / (iter_count + 2.0)
        y = x + beta_nest * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient with momentum
        g, M_curr = compute_subgrad_toppk(y, top_k=top_k, beta=beta_t)
        grad_ema = ema_alpha * g + (1 - ema_alpha) * grad_ema
        g_use = grad_ema

        # Mirror descent step
        eta = c0 / (np.sqrt(iter_count) + 1.0)
        log_y = np.log(y)
        log_a_new = log_y - eta * g_use
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x.copy()
        x = a_new

        # Evaluate

            best_a = x.copy()

        else:

        # Restart if stalled
        if stall_count > restart_every:

                pert = rng.normal(0, 0.02, n) * best_a

def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True, top_K=300):
    """
    LP-based Frank-Wolfe with top-K constraints and Nesterov acceleration.
    """
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        if iter_count % 50 == 0 and time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K binding constraints
        K = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix T: T[m, j] = y[m-j] if valid
        T = np.zeros((K, n))
        for col_idx, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            T[col_idx, js] = y[m_int - js]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K),
                        cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (iter_count + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        prev_y = y
        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m - 1e-14:
            best_m = my
            best_a = y.copy()

        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_flat_spectrum_init(n, seed=0):
    """Create initialization with nearly flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Random phase, flat magnitude
    freqs = np.fft.rfftfreq(n)
    magnitudes = np.ones(n // 2 + 1)
    magnitudes[0] = 0  # DC component
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    spectrum = magnitudes * np.exp(1j * phases)
    a = np.fft.irfft(spectrum * n, n)
    # Clip and normalize
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_envelope_init(n, seed=0, asymmetry=0.3):

    # === Phase 1: Subgradient descent from parent (primary optimization) ===
    print(f"\n=== Phase 1: Subgradient Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 650.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t1, verbose=True, seed=42)
        try_update(a_out, f"sg_main_n{n0}")

    # === Phase 2: Subgradient with different seed (diversity) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        # Slight perturbation before second run
        rng = np.random.default_rng(99)
        a_in = a_in + rng.normal(0, 0.005, n0) * a_in
        a_in = norm_sum1(a_in)
        a_out, _ = subgradient_optimize(a_in, t2, verbose=False, seed=123)
        try_update(a_out, "sg_diverse")

    # === Phase 3: LP Frank-Wolfe refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_optimize(a_in, t3, solver, verbose=True, top_K=400)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_optimize(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Second subgradient round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: SG Round 2 ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t5, verbose=False, seed=777)
        try_update(a_out, f"sg_r2_n{n_best}")

    # === Phase 6: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best + 200, n_best - 200]
    candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000]

    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 6: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 200.0)
        if t6a > 30:
            a_out, _ = subgradient_optimize(a_long, t6a, verbose=False, seed=555)
            try_update(a_out, f"sg_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 3800:
            t6b = min(remaining * 0.40, 200.0)
            if t6b > 30:
                a_out, _ = lp_frank_wolfe_optimize(a_long, t6b, solver, verbose=False, top_K=300)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Coord (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_optimize(a_in, t7, verbose=True, seed=888)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 4000:
        print(f"\n=== Phase 8: Final LP ===")
        t8 = min(remaining * 0.7, 250.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_optimize(a_in, t8, solver, verbose=True, top_K=500)
            try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_envelope_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
