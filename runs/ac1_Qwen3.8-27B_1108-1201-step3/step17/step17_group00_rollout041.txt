def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_gradient_smoothed(a, b, T):
    """Compute gradient of log(sum(exp(b/T))) w.r.t. a using FFT-based cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()

    # g[m] = sum_j a[j] * w[m+j] for m=0,...,n-1
    # = conv(a, flip(w))[2n-2-m]
    v = w[::-1]
    fft_len = 1 << (n + len(v) - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_len)
    V_fft = np.fft.rfft(v, fft_len)
    conv_av = np.fft.irfft(A_fft * V_fft, fft_len)
    g = conv_av[n - 1:2 * n - 1][::-1]
    return (2.0 / T) * g


def fista_optimize(a0, time_limit, verbose=True):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    T_min = 0.001
    T_decay = 0.9995
    eta = 0.5
    momentum = 1.0
    stall_count = 0
    max_stall = 5000
    iter_count = 0

    x = a.copy()
    y = a.copy()
    t_nesterov = 1.0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        g = compute_gradient_smoothed(y, b_y, T)

        x_new = y - eta * g
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov)) / 2.0
        beta = (t_nesterov - 1.0) / t_new
        y_new = x_new + beta * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new / sy

        t_nesterov = t_new
        x = x_new
        y = y_new
        iter_count += 1

        b_x = fast_autocorr(x)
        M_x = b_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        T = max(T * T_decay, T_min)

        if stall_count > max_stall:
            stall_count = 0
            T = max(T * 2.0, 0.01)
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x_r.copy()
            t_nesterov = 1.0
            if verbose:
                elapsed = time.monotonic() - t0

def spectral_perturb(a, strength=0.1, rng=None):
    """Add random spectral components to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = rng.uniform(0, 2 * np.pi, len(A))
    noise = strength * np.exp(1j * phase) * np.ones_like(A)
    A_new = A + noise
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)

                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 60:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_search(n, time_limit, best_init, best_score_init):
    """Search over parametric families for a good initial sequence."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = best_init.copy()
    best_val = best_score_init
    i_arr = np.arange(n, dtype=np.float64)

    def try_update_local(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = (i_arr + 1.0) ** p
        try_update_local(a, f"pow_{p}")
        a = (n - i_arr) ** p
        try_update_local(a, f"ipow_{p}")

    for alpha in [0.3, 0.5, 1.0, 2.0]:
        for beta in [0.3, 0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            try_update_local(a, f"beta_{alpha}_{beta}")

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_update_local(a, f"gc{c:0.1f}s{sigma:0.1f}")

    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sigma in [0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                try_update_local(a, f"bi{c1:0.1f}_{c2:0.1f}_{sigma:0.2f}")

    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50, 100]:
            a = 1.0 / (1.0 + np.exp(-s * (x - c)))
            try_update_local(a, f"sg{c:0.1f}_{s}")

    for freq in [1, 2, 3, 4]:
        for ph in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01)
            try_update_local(a, f"sn{freq}_{ph:0.2f}")

    for rate in [1, 2, 5, 10, 20]:
        a = np.exp(-rate * x)
        try_update_local(a, f"er{rate}")
        a = np.exp(rate * x)
        try_update_local(a, f"enr{rate}")

    # Raised cosine and other window shapes
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    try_update_local(a, "raised_cos")
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    try_update_local(a, "inv_raised_cos")

    # Hamming-like
    n_real = n
    i_h = np.arange(n_real)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i_h / (n_real - 1))
    try_update_local(a, "hamming")

    # Blend with parent
    if best_score_init < 2.5:
        for w in [0.2, 0.3, 0.5, 0.7]:
            a_pow = (i_arr + 1.0) ** 0.7
            a_blend = w * best_init / best_init.sum() + (1.0 - w) * a_pow / a_pow.sum()
            try_update_local(a_blend, f"bl{w:0.1f}")
            a_exp = np.exp(-2.0 * x)
            a_blend2 = w * best_init / best_init.sum() + (1.0 - w) * a_exp / a_exp.sum()
            try_update_local(a_blend2, f"ble{w:0.1f}")

    print(f"  Parametric done: best J={best_val:.6f}")
    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        val_init = 2.0 * n0 * fast_autocorr(best_seq).max()
        a_param, val_param = parametric_search(n0, t1, best_seq, val_init)
        check_and_update(a_param, "parametric")

    # === Phase 2: FISTA with spectral perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FISTA (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = fista_optimize(a_in, t2 * 0.7, verbose=True)
        check_and_update(a_out, f"fista1_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        # FISTA with spectral perturbation restarts
        a_in = norm_sum1(best_seq)
        rng_sp = np.random.default_rng(99)
        a_curr = a_in.copy()
        t2b = min(remaining * 0.15, 200.0)
        n_perturb = 5
        t_per_perturb = t2b / n_perturb
        for pi in range(n_perturb):
            remaining_sub = time_budget - (time.monotonic() - t_global) - 40
            if remaining_sub < 30:
                break
            a_pert = spectral_perturb(a_curr, strength=0.05, rng=rng_sp)
            a_out, _ = fista_optimize(a_pert, min(t_per_perturb, remaining_sub), verbose=False)
            check_and_update(a_out, f"fsp{pi}_n{n_best}")
            if a_out is not None and fast_autocorr(a_out).max() < fast_autocorr(a_curr).max():
                a_curr = a_out
