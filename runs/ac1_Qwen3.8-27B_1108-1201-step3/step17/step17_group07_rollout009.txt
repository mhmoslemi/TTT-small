def compute_subgradient_exact(y, ac, threshold_frac=0.99):
    """Compute subgradient averaging over all lags above threshold_frac * max.
    O(n * K) where K is number of active lags."""
    n = len(y)
    M = ac.max()
    threshold = threshold_frac * M
    active_lags = np.where(ac >= threshold)[0]
    
    if len(active_lags) == 0:
        active_lags = np.array([int(np.argmax(ac))])
    
    g = np.zeros(n)
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= len(active_lags)
    return g

                # Flip symmetry
                x_r = best_a[::-1].copy()
                pert = rng.normal(0, 0.005, n)
                x_r *= (1 + pert)
            else:
                x_r = best_a * (1 + rng.normal(0, 0.003, n))
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.7)
            
            # Adjust threshold
            threshold_frac = min(0.999, threshold_frac + 0.002)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD2 t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, thresh={thresh:.4f}")
    
    return best_a, best_m

def targeted_smo(a0, time_limit, verbose=False, rng_seed=777):
    """Targeted SMO: move mass from positions that contribute most to the peak lag."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        max_lag = int(np.argmax(ac))
        M = ac[max_lag]
        
        # Find positions that contribute most to b[max_lag]
        j_start = max(0, max_lag - n + 1)
        j_end = min(n - 1, max_lag)
        if j_start > j_end:
            stall += 1
            continue
        
        idx_range = np.arange(j_start, j_end + 1)
        contributions = a[idx_range] * a[max_lag - idx_range]
        
        # Find the position with highest contribution
        sorted_idx = np.argsort(contributions)[::-1]
        
        improved = False
        n_tries = min(20, len(idx_range))
        
        for si in range(n_tries):
            j = int(idx_range[sorted_idx[si]])
            if a[j] < 1e-12:
                continue
            
            # Find best target: position j' where a[max_lag - j'] is small
            # and moving mass there doesn't increase other lags too much
            # Simple heuristic: move to a position where the "partner" value is low
            candidates = []
            n_cand = min(50, n // 5)
            cand_positions = rng.choice(n, size=n_cand, replace=False)
            
            best_delta = 0
            best_target = -1
            best_frac = 0
            
            for j_prime in cand_positions:
                if j_prime == j:
                    continue
                # Change in b[max_lag] from moving delta from j to j_prime:
                # delta_b = delta * (a[max_lag - j_prime] - a[max_lag - j])
                # We want this negative, so a[max_lag-j_prime] < a[max_lag-j]
                j_partner = max_lag - j
                jp_partner = max_lag - j_prime
                if 0 <= j_partner < n:
                    val_j = a[j_partner]
                else:
                    val_j = 0
                if 0 <= jp_partner < n:
                    val_jp = a[jp_partner]
                else:
                    val_jp = 0
                
                if val_jp < val_j:
                    # Good candidate: compute how much we can move
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = a[j] * frac
                        a_trial = a.copy()
                        a_trial[j] -= amt
                        a_trial[j_prime] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        m_trial = fast_conv(a_trial).max()
                        if m_trial < best_m - 1e-16:
                            if (best_m - m_trial) > best_delta:
                                best_delta = best_m - m_trial
                                best_target = int(j_prime)
                                best_frac = frac
                            break
            
            if best_target >= 0:
                amt = a[j] * best_frac
                a[j] -= amt
                a[best_target] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_conv(a).max()
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def lp_remez_single(a, max_cons=50, r=0.08, solver=None):
    """Single LP Remez step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * 0.998
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * (1.0 - r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver=None, r_init=0.06, verbose=False):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_single(a, max_cons=min(60, n // 50), r=r, solver=solver)
        
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.25, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.75)
            if no_improve > 15:
                idx = rng.choice(n, size=min(n // 6, 500), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = 0.03
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with targeted moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(80, n // 4)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.03, 0.08, 0.15]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = (1.0 - (2.0 * x - 1.0) ** 4)
    elif kind == 'bimodal':
        a = np.exp(-50.0 * (x - 0.3) ** 2) + np.exp(-50.0 * (x - 0.7) ** 2)
    elif kind == 'asym_tri':
        a = np.maximum(1.0 - np.abs((x - 0.4) * 2.5), 0.0)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: NMD from parent with exact subgradient (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: NMD v2 n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent_v2(norm_sum1(best_seq), t1, threshold_frac=0.99, c0=3.5, verbose=True)
        try_update(a_out, f"NMD2_parent_n{n0}")
    
    # === Phase 2: Targeted SMO (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Targeted SMO n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = targeted_smo(norm_sum1(best_seq), t2, verbose=False)
        try_update(a_out, f"SMO_n{len(best_seq)}")
    
    # === Phase 3: LP Remez (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: LP Remez n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t3, solver=solver, r_init=0.08, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
    
    # === Phase 4: Multi-length exploration (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4_total = min(remaining * 0.20, 300.0)
    if t4_total > 60:
        print(f"\n=== Phase 4: Multi-length exploration (budget={t4_total:.0f}s) ===")
        n_candidates = [n0 - 100, n0 + 100, n0 + 200, n0 - 200, n0 + 300]
        n_candidates = [n for n in n_candidates if n >= 500]
        t_per = t4_total / len(n_candidates)
        
        for n_cand in n_candidates:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            print(f"  Exploring n={n_cand} (t_per={min(t_per, remaining-5):.0f}s)")
            a_alt = interpolate_sequence(best_seq, n_cand)
            a_out, _ = nesterov_mirror_descent_v2(a_alt, min(t_per, remaining - 10), 
                                                   threshold_frac=0.99, c0=3.0, verbose=False)
            try_update(a_out, f"NMD2_n{n_cand}")
            
            # Quick SMO polish on the best at this length
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = targeted_smo(norm_sum1(a_out), min(remaining * 0.3, 40), verbose=False, rng_seed=n_cand)
                try_update(a_out2, f"SMO_n{n_cand}")
    
    # === Phase 5: NMD with different thresholds (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: NMD varied threshold n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        # Try a more aggressive threshold (include more near-max lags)
        a_out, _ = nesterov_mirror_descent_v2(norm_sum1(best_seq), t5, 
                                               threshold_frac=0.95, c0=2.0, verbose=True)
        try_update(a_out, f"NMD2_wide_n{len(best_seq)}")
    
    # === Phase 6: CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 6: CD polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t6, verbose=False, rng_seed=999)
        try_update(a_out, f"CD_{len(best_seq)}")
    
    # === Phase 7: Final LP Remez + SMO (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t7, solver=solver, r_init=0.03, verbose=False)
        try_update(a_out, f"Remez_final_{len(best_seq)}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, _ = targeted_smo(norm_sum1(best_seq), remaining - 5, verbose=False, rng_seed=42)
            try_update(a_out2, f"SMO_final_{len(best_seq)}")
    
    # === Phase 8: Last-ditch diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 8: Diversification (budget={remaining:.0f}s) ===")
        rng = np.random.default_rng(2024)
        
        # Try reverse of best
        a_rev = best_seq[::-1].copy()
        sc = evaluate_safe(a_rev)
        print(f"  Reverse score: {sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_rev.copy()
        
        # Try perturbed reverse
        a_pert = a_rev * (1 + rng.normal(0, 0.01, len(a_rev)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        try_update(a_pert, "PerturbedReverse")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = nesterov_mirror_descent_v2(norm_sum1(best_seq), remaining - 5,
                                                   threshold_frac=0.99, c0=1.5, verbose=False)
            try_update(a_out, "NMD2_last")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
