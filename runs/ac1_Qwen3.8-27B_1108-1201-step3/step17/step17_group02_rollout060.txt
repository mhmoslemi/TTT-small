                m = fast_autocorr(a_trial).max()

                    best_a = a_trial.copy()
    
    return best_a

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, ac, top_k=10, sharpness=20.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** sharpness
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def adam_mirror_descent(a0, time_limit, top_k=15, sharpness=20.0, lr=0.15,
                        beta1=0.9, beta2=0.999, restart_every=3000,
                        rng_seed=42, n_peaks_schedule=None):
    """Adam-accelerated mirror descent in log-space with sharp peak targeting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    theta = np.log(a) - np.log(a).mean()  # log-space parameter
    m_theta = np.zeros(n)
    v_theta = np.zeros(n)
    x = a.copy()
    x_prev = a.copy()
    nesterov_mu = 0.0
    stall_count = 0
    cur_top_k = top_k
    cur_sharp = sharpness

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        nesterov_mu = (t - 1.0) / (t + 2.0)
        y = x + nesterov_mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        ac = fast_autocorr(y)
        g_a = compute_subgradient(y, ac, top_k=cur_top_k, sharpness=cur_sharp)

        # Convert to theta-space gradient (centered)
        g_theta = y * (g_a - np.dot(y, g_a))

        # Adam update
        m_theta = beta1 * m_theta + (1 - beta1) * g_theta
        v_theta = beta2 * v_theta + (1 - beta2) * g_theta ** 2
        m_hat = m_theta / (1 - beta1 ** t)
        v_hat = v_theta / (1 - beta2 ** t)

        eta = lr / (np.sqrt(t) + 1.0) ** 0.3  # slower decay than 1/sqrt(t)
        theta_new = theta - eta * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Convert back to probability
        theta_new = theta_new - theta_new.max()
        a_new = np.exp(theta_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        theta = theta_new - theta_new.max()

        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameter tuning
        if stall_count > restart_every // 2:
            cur_sharp = min(cur_sharp * 0.95, 5.0)  # soften if stuck
            cur_top_k = min(cur_top_k + 3, 50)

        # Restart with perturbation
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            theta = np.log(x) - np.log(x).max()
            m_theta = np.zeros(n)
            v_theta = np.zeros(n)
            cur_top_k = min(cur_top_k + 5, 80)
            cur_sharp = max(cur_sharp - 2.0, 5.0)

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, init_K=100, max_K=None):
    """LP-based Frank-Wolfe with dual interval structure and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_nest = a.copy()
    K_sel = init_K

    if max_K is None:
        max_K = min(2 * n - 1, 2000)

    # Precompute sparse structure
    rows_list = []
    cols_list = []
    col_starts = []
    for j in range(n):
        r_start = j
        r_end = min(2 * n - 1, j + n)
        col_starts.append(len(rows_list))
        rows_list.extend(range(r_start, r_end))
        cols_list.extend([j] * (r_end - r_start))
    rows_arr = np.array(rows_list, dtype=np.int32)
    cols_arr = np.array(cols_list, dtype=np.int32)
    n_nz = len(rows_list)
    col_ends = col_starts[1:] + [n_nz] if n > 1 else [n_nz]

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select constraints (top K by autocorrelation value)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build value array for selected constraints
        vals = np.zeros(n_nz)
        for j in range(n):
            cs = col_starts[j]
            ce = col_ends[j]
            vals[cs:ce] = y[j]

        # Filter to selected lags
        mask = np.zeros(2 * n - 1, dtype=bool)
        mask[binding_idx] = True
        row_mask = mask[rows_arr]
        sel_rows = rows_arr[row_mask]
        sel_cols = cols_arr[row_mask]
        sel_vals = vals[row_mask]

        T_sel = coo_matrix((sel_vals, (sel_rows, sel_cols)), shape=(len(binding_idx), n))

        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64) if b_var.value is not None else None
        if b_val is None:
            continue
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x_nest = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x_nest).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nest.copy()

        # Adaptively increase K
        if t % 10 == 0:
            K_sel = min(K_sel + 20, max_K)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def block_swap_lns(a0, time_limit, verbose=True, rng_seed=55):
    """Large neighborhood search via block permutations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        stall += 1
        improved = False

        # Try different block sizes
        block_size = int(rng.choice([3, 5, 8, 12, 20, 30, 50]))
        if block_size >= n:
            block_size = n // 2

        # Pick two random blocks to swap
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue

        a_trial = a.copy()
        a_trial[i:i+block_size] = a[j:j+block_size].copy()
        a_trial[j:j+block_size] = a[i:i+block_size].copy()
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        M_trial = fast_autocorr(a_trial).max()
        if M_trial < best_m - 1e-16:
            a = a_trial
            best_m = M_trial
            best_a = a.copy()
            stall = 0
            improved = True
        elif M_trial < best_m * 1.01 and rng.random() < 0.1:
            # Accept slightly worse (simulated annealing style)
            a = a_trial

        # Also try reversal of a block
        if not improved and stall > 50:
            i = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            a_trial[i:i+block_size] = a[i:i+block_size][::-1].copy()
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0

        # Random perturbation when very stuck
        if stall > 100:
            n_pert = int(rng.integers(3, 10))
            idx = rng.choice(n, n_pert, replace=False)
            a_trial = a.copy()
            a_trial[idx] *= rng.uniform(0.8, 1.2, n_pert)
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
            else:
                # Try averaging with best
                a_trial2 = 0.5 * best_a + 0.5 * a_trial2 if False else 0.5 * best_a + 0.5 * a_trial
                a_trial2 = np.maximum(a_trial2, 1e-10)
                a_trial2 = a_trial2 / a_trial2.sum()
                M2 = fast_autocorr(a_trial2).max()
                if M2 < best_m:
                    a = a_trial2
                    best_m = M2
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LNS ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def spectral_shaping(a0, time_limit, verbose=True, rng_seed=100):
    """Add optimized sinusoidal perturbations to shape the autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = np.arange(n)

    # Try various frequencies and amplitudes
    freqs = list(range(1, 30)) + [n // k for k in range(2, 10) if n // k > 0]
    freqs = [f for f in freqs if 0 < f < n]
    amplitudes = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]

    rng.shuffle(freqs)
    rng.shuffle(amplitudes)

    for amp in amplitudes:
        if time.monotonic() - t0 > time_limit:
            break
        for freq in freqs[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            for phase in [0, np.pi / 4, np.pi / 2, 3 * np.pi / 4]:
                jitter = amp * np.sin(2 * np.pi * freq * x / n + phase)
                a_trial = best_a * (1.0 + jitter)
                a_trial = np.maximum(a_trial, 1e-10)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    best_m = M_trial
                    best_a = a_trial.copy()
                    if verbose:
                        J = 2.0 * n * best_m
                        print(f"    Spectral: J={J:.6f} (freq={freq}, amp={amp}, phase={phase:.2f})")

    # Also try to dampen specific peak regions
    ac = fast_autocorr(best_a)
    top_peak = int(np.argmax(ac))
    peak_region = int(n // 10)
    center = top_peak if top_peak < n else n - 1 - top_peak

    for damp_frac in [0.8, 0.85, 0.9]:
        a_trial = best_a.copy()
        a_trial[max(0, center - peak_region):center + peak_region + 1] *= damp_frac
        a_trial = np.maximum(a_trial, 1e-10)
        a_trial = a_trial / a_trial.sum()
        M_trial = fast_autocorr(a_trial).max()
        if M_trial < best_m:
            best_m = M_trial
            best_a = a_trial.copy()

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.12) / width) ** 2) + np.exp(-((x - 0.88) / width) ** 2) + 0.003
    return norm_sum1(a)


def make_powershape(n, p=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + p * np.abs(x) ** 3
    return norm_sum1(a)


def spectral_init(n, n_modes=25, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_u_shape(n_base, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init Adam MD ===
    print(f"\n=== Phase 1: Multi-init Adam MD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_powershape(n0, 2.5))
        labels.append("pow2.5")
        inits.append(make_bimodal(n0, 0.18))
        labels.append("bimodal")
        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spectral")

        # Blend parent with U-shape
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        seen_sigs = []
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = np.round(init[::max(1, n0 // 40)], 4)
            is_dup = False
            for prev_sig in seen_sigs:
                if np.allclose(sig, prev_sig, atol=0.05):
                    is_dup = True
                    break
            if not is_dup:
                seen_sigs.append(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(25.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  Adam MD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = adam_mirror_descent(init, t_avail, top_k=15, sharpness=25.0,
                                            lr=0.2, restart_every=3000, rng_seed=100 + idx)
            try_update(a_out, f"adam_{lbl}")

    # === Phase 2: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 380.0)
    if t2 > 50 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, init_K=80, max_K=min(800, 2*n_best))
        try_update(a_out, "lp_main")

    # === Phase 3: Block swap LNS ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Block swap LNS (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = block_swap_lns(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "lns")

    # === Phase 4: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral shaping (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_shaping(a_in, t4, verbose=True, rng_seed=200)
        try_update(a_out, "spectral")

    # === Phase 5: Adam MD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Adam polish (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 25:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_mirror_descent(a_in, t5, top_k=10, sharpness=30.0,
                                        lr=0.12, restart_every=5000, rng_seed=300)
        try_update(a_out, "adam_polish")

    # === Phase 6: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP-FW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.18, 300.0)
    if t6 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, init_K=150, max_K=min(1500, 2*n_best))
        try_update(a_out, "lp_r2")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish (n={len(best_seq)}) ===")
    n_best = len(best_seq)

    if remaining > 40:
        t7a = min(remaining * 0.35, 120.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = adam_mirror_descent(a_in, t7a, top_k=8, sharpness=35.0,
                                            lr=0.1, restart_every=4000, rng_seed=500)
            try_update(a_out, "adam_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7b = min(remaining * 0.3, 80.0)
        if t7b > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_shaping(a_in, t7b, verbose=False, rng_seed=600)
            try_update(a_out, "spectral_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and n_best <= 5000:
        t7c = min(remaining * 0.4, 100.0)
        if t7c > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7c, solver, verbose=False, init_K=200, max_K=min(2000, 2*n_best))
            try_update(a_out, "lp_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
