`a_new[j] = a[j] * exp(-η * grad[j]) / Z`

where `Z = sum_j a[j] * exp(-η * grad[j])`.

This automatically maintains `a >= 0` and `sum(a) = 1`.

Let me think about the step size. The gradient of `F_λ` has magnitude roughly `O(1)` (since `a` sums to 1 and `p` sums to 1, the gradient entries are at most `2*max(a)`). So `η = 1/sqrt(t)` should work.

Actually, for mirror descent on the simplex with a strongly convex function, the rate is `O(1/sqrt(t))` for the subgradient method. For a smooth function, we can get `O(1/t)` with proper step sizes.

The smooth max `F_λ` has Lipschitz constant (with respect to the ℓ1 norm, which is dual to the ℓ∞ norm used in mirror descent) of roughly `2 * max(a) * ||p||_1 = 2 * max(a)`. Since `sum(a) = 1` and `a >= 0`, `max(a) <= 1`, so the Lipschitz constant is at most 2.

For mirror descent with entropy, the step size should be `η_t = c / sqrt(t)` for subgradient, or `η_t = c * L / t` for smooth (with proper line search).

Let me just use a backtracking line search to be safe.

**LP Remez improvement:**

The key improvement I want to make to the LP Remez is:
1. Use more constraints (up to 200-500 active lags)
2. Use a two-sided trust region: `a * (1-r) <= a' <= a * (1+r)`
3. Solve with high precision
4. Do multiple Remez iterations in a row (the "Remez cycle")

Let me now write the complete code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) mirror descent with adaptive temperature
2. LP-based Remez with many active constraints
3. Nesterov acceleration
4. Multi-length exploration and final polish
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max_grad(a, lam):
    """Compute smooth max of conv(a,a) and its gradient.
    Returns (value, gradient) where gradient is w.r.t. a on the simplex.
    """
    n = len(a)
    b = fast_conv(a)
    # Numerically stable softmax
    lb = lam * (b - b.max())
    exp_lb = np.exp(lb)
    p = exp_lb / exp_lb.sum()
    val = b.max() + np.log(exp_lb.sum()) / lam
    
    # Gradient: grad[m] = 2 * sum_{l=0}^{n-1} p[m+l] * a[l]
    # = 2 * correlate(p, a, 'full')[m] for m = 0,...,n-1
    c = np.correlate(p, a, 'full')[:n]
    grad = 2.0 * c
    return val, grad

def smooth_max_descent(a0, time_limit, lam_init=5.0, lam_max=500.0, 
                         lam_growth=1.05, period=50, verbose=False):
    """Mirror descent on the smooth max objective with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    lam = lam_init
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 1000
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        y = y / sy
        
        val, grad = smooth_max_grad(y, lam)
        
        # Mirror descent step with backtracking
        # Step size: start with 1/sqrt(t), backtrack if needed
        eta0 = 2.0 / (np.sqrt(t) + 1.0)
        
        accepted = False
        for _bt in range(30):
            eta = eta0 * (0.5 ** _bt)
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            
            m_new = fast_conv(a_new).max()
            if m_new < best_m + 1e-15:  # allow tiny increase
                x_prev = x
                x = a_new
                if m_new < best_m - 1e-18:
                    best_m = m_new
                    best_a = x.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
        
        if not accepted:
            stall += 1
            # If stuck, just do a small step
            eta = eta0 * (0.5 ** 30)
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            x_prev = x
            x = a_new
        
        # Update lambda
        if t % period == 0:
            lam = min(lam * lam_growth, lam_max)
        
        # Restart if stuck too long
        if stall > max_stall:
            stall = 0
            max_stall = max(200, int(max_stall * 0.5))
            rng = np.random.default_rng(42 + t)
            pert = rng.normal(0, 0.01, n)
            x = best_a * (1 + pert)
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            x_prev = x.copy()
            lam = max(lam_init, lam * 0.5)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  SMD t={t} ({elapsed:.1f}s): J={J:.8f}, lam={lam:.1f}, stall={stall}")
    
    return best_a, best_m

def lp_remez_step(a, max_cons=100, r=0.1, solver=None):
    """Single LP Remez step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints: all lags with b[k] >= threshold
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        # Take the top max_cons
        active = active[np.argsort(b[active])[::-1][:max_cons]]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    n_active = len(active)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for i in range(n_active):
        k_int = int(active[i])
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Two-sided trust region
    low = np.maximum(a * (1.0 - r), 0.0)
    high = a * (1.0 + r)
    constraints.append(a_var >= low)
    constraints.append(a_var <= high)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), None
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), None
    return a_new / s, prob.status

def remez_method(a0, time_limit, solver=None, r_init=0.15, max_cons=100, verbose=False):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand, status = lp_remez_step(a, max_cons=max_cons, r=r, solver=solver)
        
        if a_cand is None or np.array_equal(a_cand, a):
            no_improve += 1
        else:
            # Line search
            accepted = False
            for t_frac in [1.0, 0.8, 0.5, 0.3, 0.1]:
                a_trial = (1 - t_frac) * a + t_frac * a_cand
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    accepted = True
                    no_improve = 0
                    r = min(0.4, r * 1.05)
                    break
            
            if not accepted:
                no_improve += 1
                r = max(0.01, r * 0.7)
        
        if no_improve > 15:
            # Perturb to escape
            rng = np.random.default_rng(t)
            idx = rng.choice(n, size=min(n//4, 500), replace=False)
            a[idx] *= rng.uniform(0.7, 1.3, len(idx))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            r = r_init
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, no_improve={no_improve}")
    
    return best_a, best_m

def nesterov_subgrad_descent(a0, time_limit, top_k=30, c0=3.0, verbose=False):
    """Nesterov mirror descent with top-k subgradient (from previous code, improved)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_conv(y)
        M = ac.max()
        
        # Adaptive top_k
        near_max = np.sum(ac >= M * 0.99)
        k_use = min(max(top_k, int(near_max) + 3), min(150, len(ac)))
        
        top_idx = np.argpartition(ac, -k_use)[-k_use:]
        
        g = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                g[idx] += 2.0 * y[k_int - idx]
        g /= k_use
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.6))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//5, 400), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            else:
                # Smaller perturbation
                x_r = best_a * (1 + rng.normal(0, 0.003, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.7)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, c={c:.3f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, rng_seed=123):
    """Coordinate descent with random pair moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_rounds = min(50, n // 5)
        
        for _ in range(n_rounds):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-17:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multi_scale_lp(a0, time_limit, solver=None, verbose=False):
    """Multi-scale LP Remez: solve at coarser scale, then refine."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Start with small trust region and increase
    r_values = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4]
    max_cons_values = [50, 100, 200, 300, 400, 500, 600]
    
    for r, mc in zip(r_values, max_cons_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        a_out, _ = remez_method(a, remaining * 0.8, solver=solver, 
                                  r_init=r, max_cons=mc, verbose=verbose)
        if a_out is not None:
            a = a_out
            m = fast_conv(a).max()
            if m < best_m - 1e-18:
                best_m = m
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  MultiScale r={r:.3f}, mc={mc}: J={J:.8f} ({elapsed:.1f}s)")
    
    return best_a, best_m
```

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def smooth_max_grad(a, lam):
    """Compute smooth max of conv(a,a) and its gradient via FFT-based correlation."""
    n = len(a)
    b = fast_conv(a)
    lb = lam * (b - b.max())
    exp_lb = np.exp(lb)
    p = exp_lb / exp_lb.sum()
    val = b.max() + np.log(exp_lb.sum()) / lam
    # grad[m] = 2 * sum_{l=0}^{n-1} p[m+l] * a[l]
    # = 2 * correlate(p, a, 'full')[m] for m = 0,...,n-1
    c = np.correlate(p, a, 'full')[:n]
    grad = 2.0 * c
    return val, grad

def lp_solve_with_active(a, active, r, solver):
    """Solve LP Remez with given active set and trust region. Returns a_cand or None."""
    n = len(a)
    b = fast_conv(a)

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    low = np.maximum(a * (1.0 - r), 0.0)
    high = a * (1.0 + r) + 1e-15
    constraints.append(a_var >= low)
    constraints.append(a_var <= high)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s

            m_trial = fast_conv(a_trial).max()
            if m_trial < best_trial_m - 1e-17:
                best_trial = a_trial
                best_trial_m = m_trial

        if best_trial is not None:
            a = best_trial
            if best_trial_m < best_m - 1e-18:
                best_m = best_trial_m
                best_a = a.copy()
            no_improve = 0

            b_new = fast_conv(a)
            M_new = b_new.max()
            threshold_new = M_new * 0.997
            new_active = set(int(k) for k in np.where(b_new >= threshold_new)[0])
            old_active = set(active)
            combined = old_active | new_active
            if len(combined) > max_active:
                b_vals = sorted(combined, key=lambda k: -b_new[k])
                active = b_vals[:max_active]
            else:
                active = list(combined)
            r = min(0.35, r * 1.02)
        else:
            no_improve += 1
            r = max(0.005, r * 0.6)
            if no_improve > 20:
                rng = np.random.default_rng(t * 7 + 3)
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                a[idx] *= rng.uniform(0.75, 1.25, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                r = r_init * 0.5
                no_improve = 0

        if verbose and t % 8 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            avg_lp = np.mean(lp_times[-10:]) if lp_times else 0
            print(f"    RemezCyc t={t} ({elapsed:.1f}s): J={J:.8f}, |A|={len(active)}, r={r:.4f}, lp={avg_lp*1000:.0f}ms")

    return best_a, best_m

def nesterov_subgrad(a0, time_limit, top_k=30, c0=3.0, verbose=False):
    """Nesterov mirror descent with top-k subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        ac = fast_conv(y)
        M = ac.max()
        near_max = np.sum(ac >= M * 0.99)
        k_use = min(max(top_k, int(near_max) + 3), min(150, len(ac)))
        top_idx = np.argpartition(ac, -k_use)[-k_use:]
        g = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                g[idx] += 2.0 * y[k_int - idx]
        g /= k_use
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        x_prev = x
        x = a_new
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            max_stall = max(300, int(max_stall * 0.6))
            strat = int(rng.integers(4))
            if strat == 0:
                x = best_a * (1 + rng.normal(0, 0.012, n))
            elif strat == 1:
                x = best_a.copy()
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                x[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strat == 2:
                x = best_a * (1 + rng.normal(0, 0.004, n))
            else:
                x = best_a * (1 + rng.normal(0, 0.002, n))
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c = max(0.3, c * 0.7)
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall}, c={c:.3f}")
    return best_a, best_m
