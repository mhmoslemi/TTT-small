def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def exact_subgrad(a, k_star):
    """
    Compute exact subgradient of max(conv(a,a)) at lag k_star.
    g[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g


def soft_subgrad(a, beta=5.0, top_k=None):
    """
    Compute soft subgradient using exp-weighted combination of top-k lags.
    """
    n = len(a)
    ac = fast_autocorr(a)
    if top_k is not None and top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = np.zeros(len(ac))
        weights = ac[top_idx]
        weights = np.exp(beta * (weights - weights.max()))
        weights = weights / weights.sum()
        w[top_idx] = weights
    else:
        shifted = ac - ac.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / w_sum
        w[w < 1e-12] = 0
        if w.sum() < 1e-15:
            w = np.ones_like(ac) / len(ac)
    # grad[i] = 2 * sum_m w[m] * a[m-i] for valid m-i
    # = 2 * (a_rev * w)[i] where a_rev[i] = a[n-1-i]
    # Using FFT: grad = 2 * IFFT(FFT(a_rev) * FFT(w))
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:total_len]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()


def md_optimize(a0, time_limit, beta_init=6.0, beta_min=3.0, c0=4.0,

    Mirror descent with Nesterov momentum and mixed exact/soft subgradients.

    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta_init
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        beta_t = (it - 1.0) / (it + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Mixed subgradient: exact for top lag + soft for exploration
        if it % 3 < int(use_exact_frac * 3):
            # Use exact subgradient at the top lag
            ac_y = fast_autocorr(y)
            k_star = int(np.argmax(ac_y))
            g_exact = exact_subgrad(y, k_star)
            # Blend with a small amount of soft subgrad for exploration
            g_soft, _ = soft_subgrad(y, beta=cur_beta, top_k=30)
            g = 0.7 * g_exact + 0.3 * g_soft
        else:
            g, M_curr = soft_subgrad(y, beta=cur_beta, top_k=50 if it > top_k_switch else None)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(it) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if rng.random() < 0.2:
                x = rng.dirichlet(np.ones(n) * 0.7)
                x_prev = x.copy()
    
    return best_a, best_m


def lp_fw_optimize(a0, time_limit, solver, top_K=None, verbose=True):
    """
    LP Frank-Wolfe with adaptive constraint selection and Nesterov acceleration.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_nesterov = a.copy()
    prev_y = a.copy()
    
    if top_K is None:
        top_K = min(300, 2 * n - 1)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select constraints: top-K most binding
        K = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix (sparse-ish)
        T = np.zeros((K, n))
        for col, m in enumerate(top_idx):
            mi = int(m)
            js = max(0, mi - n + 1)
            je = min(n - 1, mi)
            idx = np.arange(js, je + 1)
            T[col, idx] = y[mi - idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constr = [T @ b_var <= M_curr * np.ones(K), cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constr)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        g_dir = b_val / bs
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        prev_y = y
        y = y_new
        x_nesterov = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_nesterov).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nesterov.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def multi_peak_coord(a0, time_limit, seed=42, n_peaks=10):
    """
    Multi-peak coordinate descent with line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    moves = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 150:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        top_lags = np.argsort(ac)[::-1][:n_peaks]
        
        improvement = False
        
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            
            # Find source: position contributing most to this lag
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Compute destination sensitivity (how much moving to d affects all top peaks)
            sens = np.zeros(n)
            for k2 in top_lags[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]
            
            # Find best destination: low sensitivity, far from source
            dist = np.abs(np.arange(n) - src).astype(float)
            score = sens + 0.01 * (1.0 / (1.0 + dist / n))
            # Avoid the source position
            score[src] = np.inf
            dst = int(np.argmin(score))
            if dst == src:
                continue
            
            # Line search: try to find optimal transfer amount
            best_frac = None
            best_m_trial = best_m
            for frac in [0.05, 0.03, 0.02, 0.01, 0.08, 0.04, 0.015, 0.025, 0.005]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                M_t = fast_autocorr(a_t).max()
                if M_t < best_m_trial:
                    best_m_trial = M_t
                    best_frac = frac
            
            if best_frac is not None:
                amt = src_val * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_m_trial
                best_a = a.copy()
                improvement = True
                moves += 1
                stall = 0
        
        if not improvement:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_ramp_init(n, seed=0, bias=0.3):
    """Create a ramp-like initialization with asymmetry."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Asymmetric profile: higher on one side
    a = (1.0 + bias * (2.0 * x - 1.0))
    a += 0.05 * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)
