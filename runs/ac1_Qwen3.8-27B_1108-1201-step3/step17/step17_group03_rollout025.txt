import time
from scipy.optimize import approx_fprime
import cvxpy as cp

_global_best = None

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, nonnegative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_lse_objective(a, beta):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(a)
    ac = fast_autocorr(a)
    # Log-sum-exp of autocorrelation
    max_ac = ac.max()
    if max_ac < 1e-30:
        return 2.0 * n * max_ac, np.zeros(n)
    
    # p_k = softmax(beta * ac)
    shifted = beta * (ac - max_ac)  # for numerical stability
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    lse_val = max_ac + np.log(exp_shifted.sum()) / beta
    
    # Gradient: 2 * conv(p, reverse(a))
    # (p * reverse(a))[i + n - 1] for i = 0,...,n-1
    a_rev = a[::-1]

    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * AR, fft_size)
    grad = 2.0 * conv_pa[n-1:2*n-1]  # indices n-1 to 2n-2
    
    obj = 2.0 * n * lse_val
    return obj, grad

def softmax(z):
    """Numerically stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def softmax_gradient_chain(grad_a, a):
    """Chain rule through softmax: grad_z = a * (grad_a - sum(a * grad_a))"""
    offset = np.dot(a, grad_a)
    return a * (grad_a - offset)

              beta_ramp=200, max_iter=5000, verbose=True, n_restart=3000):

    n = len(a0)

    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()

    m = 10  # memory
    s_list = []  # position differences
    y_list = []  # gradient differences
    rho_list = []
    
    prev_grad_z = None
    prev_z = z.copy()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        beta = min(beta_init + (beta_max - beta_init) * it / max(beta_ramp, 1), beta_max)
        
        # Current a from z
        a_curr = softmax(z)
        
        # Compute objective and gradient in a-space
        obj_a, grad_a = compute_lse_objective(a_curr, beta)
        
        # Chain through softmax to get gradient in z-space
        grad_z = softmax_gradient_chain(grad_a, a_curr)
        
        # L-BFGS two-loop recursion for direction
        if prev_grad_z is None:
            # First iteration: steepest descent
            direction = -grad_z
            alpha = 1.0
        else:
            # Build L-BFGS history
            s = z - prev_z
            y = grad_z - prev_grad_z
            sy = np.dot(s, y)
            if sy > 1e-15:
                s_list.append(s.copy())
                y_list.append(y.copy())
                rho_list.append(1.0 / sy)
                if len(s_list) > m:
                    s_list.pop(0)
                    y_list.pop(0)
                    rho_list.pop(0)
            
            # Two-loop recursion
            r = -grad_z.copy()
            q = r.copy()
            alpha_vals = [0.0] * len(s_list)
            
            for i in range(len(s_list) - 1, -1, -1):
                alpha_vals[i] = rho_list[i] * np.dot(s_list[i], q)
                q = q - alpha_vals[i] * y_list[i]

            if len(s_list) > 0:
                gamma = np.dot(s_list[-1], y_list[-1]) / np.dot(y_list[-1], y_list[-1])
                if gamma < 1e-10:
                    gamma = 1.0
            else:
                gamma = 1.0
            r = gamma * q
            
            for i in range(len(s_list)):
                beta_val = rho_list[i] * np.dot(y_list[i], r)
                r = r + (alpha_vals[i] - beta_val) * s_list[i]
            
            direction = r

        alpha = 1.0
        c1 = 1e-4

                accepted = True
                break

            z_trial = z + alpha * direction
            a_trial = softmax(z_trial)
            obj_trial, _ = compute_lse_objective(a_trial, beta)
            if obj_trial < obj_curr:
                accepted = True
            else:

                prev_z = z.copy()
                prev_grad_z = grad_z.copy()
                continue
        
        z = z_trial
        prev_z = z.copy()
        prev_grad_z = grad_z.copy()
        
        # Update best
        a_curr = softmax(z)
        ac = fast_autocorr(a_curr)
        current_val = 2.0 * n * ac.max()
        if current_val < best_val:

        # Convergence check
        if grad_norm < 1e-10:
            break
    
    return best_a, best_val

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to nonnegative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_lse_gradient(a, beta, top_k=200):
    """
    Compute gradient of log-sum-exp smoothed objective.
    Returns (lse_value, gradient) where lse_value approximates max(ac).
    """
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    if max_ac < 1e-30:
        return 0.0, np.zeros(n)

    # Use top-k for efficiency when beta is large
    if beta > 10 and top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        ac_top = ac[top_idx]
        max_top = ac_top.max()
        shifted = beta * (ac_top - max_top)
        exp_shifted = np.exp(shifted)
        p_top = exp_shifted / exp_shifted.sum()
        lse_val = max_top + np.log(exp_shifted.sum()) / beta
        # Build full p vector (zero outside top_k)
        p = np.zeros_like(ac)
        p[top_idx] = p_top
    else:
        shifted = beta * (ac - max_ac)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()
        lse_val = max_ac + np.log(exp_shifted.sum()) / beta

    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] for valid (k-i) in [0,n-1]
    # = 2 * (p * a_rev)[i + n - 1]
    a_rev = a[::-1]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * AR, fft_size)
    grad = 2.0 * conv_pa[n - 1:2 * n - 1]

    return lse_val, grad

                a = a_new

def lp_frank_wolfe_short(a0, solver, max_iters=50, top_K=200, margin_frac=1e-3,
                         verbose=False):
    """Short LP Frank-Wolfe for interlaced polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    for t in range(1, max_iters + 1):
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active constraints: top K peaks within margin
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K:
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < 5:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                T_sel[row, idx] = a[k_int - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

    if verbose:
        J = 2.0 * n * best_m
        print(f"    LP-short: J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe_full(a0, solver, time_limit, top_K_init=100, top_K_max=500,
                        margin_init=5e-3, verbose=True):
    """Full LP Frank-Wolfe with adaptive K and margin."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    cur_K = top_K_init
    cur_margin = margin_init
    stall = 0
    max_stall = 50

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive K
        cur_K = min(top_K_init + t * 2, top_K_max)
        # Adaptive margin (decrease over time)
        cur_margin = max(margin_init * (0.95 ** (t // 10)), 1e-5)

        margins = M_curr - ac
        threshold = cur_margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > cur_K:
            active_idx = np.argsort(margins)[:cur_K]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:cur_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                T_sel[row, idx] = a[k_int - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            cur_K = max(cur_K // 2, 50)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        g_dir = b_val / b_sum

        # FW+ step size: min of 2/(t+2) and line search
        alpha_fw = 2.0 / (t + 2.0)

        # Line search
        best_alpha = alpha_fw
        best_obj = M_curr
        for trial_alpha in [alpha_fw, alpha_fw * 0.5, alpha_fw * 2.0, 1.0]:
            if trial_alpha > 1.0 or trial_alpha < 0.01:
                continue
            a_trial = (1.0 - trial_alpha) * a + trial_alpha * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_obj:
                best_obj = m_trial
                best_alpha = trial_alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
            # Tighten margin when improving
            cur_margin = max(cur_margin * 0.9, 1e-5)
        else:
            stall += 1
            if stall > max_stall:
                # Perturb to escape
                rng = np.random.default_rng(t)
                pert = rng.normal(0, 0.01, n) * best_a
                a_new = best_a + pert
                a_new = np.maximum(a_new, 1e-10)
                a_new = a_new / a_new.sum()
                a = a_new
                stall = 0
                cur_margin = margin_init

        a = a_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={cur_margin:.2e}")

    return best_a, best_m


def peak_crush_cd(a0, time_limit, verbose=True, rng_seed=42):
    """Spectrum-aware peak-crushing coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top 3 peaks
        peak_indices = np.argsort(ac)[::-1][:3]

        improved = False
        for peak_k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break

            peak_k = int(peak_k)
            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[peak_k - idx]
            total_contrib = contrib.sum()

            if total_contrib < 1e-20:
                continue

            # Find top contributing pairs
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]

            found_move = False
            for li in top_local:
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                best_new_m = M_curr
                best_move = None

                # Try various fractions and destinations
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue

                    # Spectrum-aware destination selection
                    # Move mass to where it causes least interference
                    n_trials = 40
                    for _ in range(n_trials):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)

                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        found_move = True
                        break

            if found_move:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush done ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.02, rng=None):
    """Perturb in frequency domain."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    """Generate parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'raised_cos':
        x = np.linspace(0, 1, n)
        return norm_sum1(0.5 * (1 + np.cos(2 * np.pi * x)))
    elif family == 'gauss':
        x = np.linspace(0, 1, n)
        mu, sig = 0.5, 0.2
        return norm_sum1(np.exp(-0.5 * ((x - mu) / sig) ** 2))
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 1.5 * (1 - x) ** 1.5)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, J={best_score:.6f}")

    # === Phase 1: Mirror descent with LSE (400s) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    print(f"\n=== Phase 1: MD-LSE (n={n0}) ===")
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_lse(a_in, t1, beta_init=2.0, beta_max=20.0,
                                      beta_ramp_iters=8000, c0=2.5,
                                      verbose=True, lp_every=800, solver=solver)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "md_lse")

    # === Phase 2: LP-FW main (500s) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.28, 500.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_full(a_in, solver, t2, top_K_init=100,
                                        top_K_max=400, margin_init=3e-3, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Multi-length exploration (350s) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Multi-length (base n={n_best}) ===")
    candidate_lengths = []
    # Try nearby lengths
    for delta in [-400, -200, 200, 400, 600]:
        new_n = n_best + delta
        if 2000 <= new_n <= 5500:
            candidate_lengths.append(new_n)
    candidate_lengths = sorted(set(candidate_lengths))

    t3_per = min(remaining * 0.08, 80.0)
    for new_n in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
        if remaining < t3_per + 30:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick MD-LSE
        t_quick = min(t3_per * 0.4, remaining * 0.3)
        if t_quick > 10:
            a_out, _ = mirror_descent_lse(a_interp, t_quick, beta_init=3.0,
                                         beta_max=15.0, beta_ramp_iters=3000,
                                         c0=2.0, verbose=False,
                                         lp_every=1000, solver=None)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                             t_global, f"ml_n{new_n}_md")

        # Quick LP-FW
        remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
        t_lp = min(t3_per * 0.5, remaining * 0.4)
        if t_lp > 15 and new_n <= 4500:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_full(a_in, solver, t_lp, top_K_init=80,
                                           top_K_max=250, margin_init=3e-3, verbose=False)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                             t_global, f"ml_n{new_n}_lp")

    # === Phase 4: LP-FW tight (400s) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.22, 400.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_full(a_in, solver, t4, top_K_init=200,
                                        top_K_max=600, margin_init=1e-3, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 5: Frequency perturbation + re-optimize (200s) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Freq perturb (n={n_best}) ===")
    rng = np.random.default_rng(77)
    t5_per = min(remaining * 0.10, 100.0)
    if t5_per > 25:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
            if remaining < t5_per + 20:
                break
            a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
            # Short MD-LSE
            t_short = t5_per * 0.5
            a_out, _ = mirror_descent_lse(a_pert, t_short, beta_init=5.0,
                                         beta_max=20.0, beta_ramp_iters=2000,
                                         c0=2.0, verbose=False,
                                         lp_every=500, solver=solver)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                             t_global, f"freq{trial}")

            # Short LP
            remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
            if remaining > 30:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe_full(a_in, solver, t5_per * 0.4,
                                               top_K_init=150, top_K_max=350,
                                               margin_init=1e-3, verbose=False)
                best_seq, best_score = try_update(a_out, best_seq, best_score,
                                                 t_global, f"freq{trial}_lp")

    # === Phase 6: Peak-crushing CD (150s) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Peak-crush (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush_cd(a_in, t6, verbose=True, rng_seed=123)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc1")

    # === Phase 7: Final LP-FW (remaining) ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.50, 300.0)
    if t7 > 20 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_full(a_in, solver, t7, top_K_init=300,
                                        top_K_max=800, margin_init=5e-4, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final peak-crush ===
    remaining = time_budget - (time.monotonic() - t0_global_wrapper(t_global))
    print(f"\n=== Phase 8: Final peak-crush ===")
    t8 = min(remaining * 0.40, 120.0)
    if t8 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush_cd(a_in, t8, verbose=True, rng_seed=777)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety check
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 15:
        print(f"WARNING: Only {remaining_time:.0f}s remaining!")

    return [float(x) for x in best_seq]


def t0_global_wrapper(t_global):
    """Return the global start time (helper for computing remaining budget)."""
    return t_global
