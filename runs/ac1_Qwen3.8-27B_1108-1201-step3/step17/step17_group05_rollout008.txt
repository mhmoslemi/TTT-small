import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute conv(a,a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv(a, b):
    """Compute conv(a,b) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]

def compute_subgrad_active(a, threshold=0.99):
    """
    Compute subgradient using active lags (where conv >= threshold * max).
    Returns (gradient, max_value, active_lags).
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), M, np.array([n-1])
    
    # Active lags: where ac[k] >= threshold * M
    active = np.where(ac >= threshold * M)[0]
    
    # Weights: proportional to ac values (or uniform over active)
    w = ac[active]
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    
    # grad[j] = 2 * sum_{k in active} w[k] * a[k-j] for valid k-j
    # This is 2 * conv(a_rev, w_full)[n-1 : 2n-1]
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(active):
        w_full[int(k)] = w[i]
    
    total_len = n + len(w_full) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_full)] = w_full
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, M, active

    """
    Line search for FW step: minimize max_k conv(a_alpha, a_alpha)_k
    along a_alpha = (1-alpha)*a + alpha*b, alpha in [0, 1].
    
    conv(a_alpha, a_alpha)[k] = (1-alpha)^2 * ac_a[k] + 2*alpha*(1-alpha)*conv_ab[k] + alpha^2 * ac_b[k]
    
    Returns best alpha.
    """

    def f_of_alpha(alpha):
        if alpha <= 0:
            return ac_a.max()
        if alpha >= 1:
            return ac_b.max()
        val = (1-alpha)**2 * ac_a + 2*alpha*(1-alpha)*conv_ab + alpha**2 * ac_b
        return val.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
