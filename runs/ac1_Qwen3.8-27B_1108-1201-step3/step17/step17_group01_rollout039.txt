def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_peak_gradient(a, k):
    """Compute gradient of (a*a)_k w.r.t. a.
    grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    return grad


def compute_multi_peak_gradient(a, ac, top_k=20, temperature=3.0):
    """Compute weighted multi-peak gradient with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    # Use softmax over top peaks for smooth weighting
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Softmax-like weighting
    max_val = top_vals[0]
    if max_val < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        exp_vals = np.exp(-temperature * (max_val - top_vals) / max(max_val, 1e-30))
        w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += wi * 2.0 * a[j_idx]
    return grad


def peak_shaving_optimize(a0, time_limit, top_k_start=5, top_k_end=80,

        cur_temp = temp_start + (temp_end - temp_start) * progress

        # Nesterov momentum
        beta = (iteration) / (iteration + 2.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and multi-peak gradient

        # Mirror descent step (exponentiated gradient)

        max_log = log_a.max()

def freq_flatten_optimize(a0, time_limit, iters=5000, verbose=False):
    """Frequency-domain flattening with adaptive exponent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Adaptive flattening exponent: start strong, get gentler
        alpha = 0.5 * (1.0 + 0.5 * it / max(1, iters))
        # Target: compress dynamic range
        P_ratio = P / (P_mean + 1e-30)
        P_target = P_mean * P_ratio ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqFlatten it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Direct coordinate descent targeting the worst peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_worst = int(np.argmax(ac))

        # Get gradient for the worst peak
        g = compute_peak_gradient(a, k_worst)

        # Try to move mass from high-gradient to low-gradient positions
        # while reducing the peak
        n_positions = max(1, n // 20)
        high_idx = np.argsort(g)[::-1][:n_positions]
        low_idx = np.argsort(g)[:n_positions]

        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_trial = a.copy()
            src_total = a[high_idx].sum()
            if src_total < 1e-15:
                break
            # Move proportional mass from high to low
            move_amt = src_total * frac
            # Distribute from high positions
            redist = (a[high_idx] / (src_total + 1e-30)) * move_amt
            a_trial[high_idx] -= redist
            # Distribute to low positions proportionally to their current value + epsilon
            low_weights = a[low_idx] + 1e-10
            a_trial[low_idx] += (low_weights / low_weights.sum()) * move_amt

            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()

            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def multi_resolution_optimize(a0, time_limit, verbose=False):
    """Coarse-to-fine optimization: optimize block sums, then refine."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Coarse level: group into blocks
    for block_size in [4, 2]:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        n_blocks = n // block_size
        if n_blocks < 10:
            continue

        # Extract block sums
        blocks = a[:n_blocks * block_size].reshape(n_blocks, block_size).sum(axis=1)
        t_coarse = min(remaining * 0.4, 60.0)

        # Optimize block structure using peak shaving
        if t_coarse > 5:
            blocks_opt, _ = peak_shaving_optimize(blocks, t_coarse,
                                                   top_k_start=3, top_k_end=20,
                                                   temp_start=1.0, temp_end=3.0,
                                                   c0=2.0, verbose=False)
            # Reconstruct: distribute block sums uniformly within blocks
            a_new = np.zeros(n)
            for i in range(n_blocks):
                a_new[i*block_size:(i+1)*block_size] = blocks_opt[i] / block_size
            # Keep tail
            if n_blocks * block_size < n:
                a_new[n_blocks*block_size:] = a[n_blocks*block_size:]
            a_new = norm_sum1(a_new)
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
                a = a_new

    # Fine level: full resolution optimization
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 20:
        a_out, m_out = peak_shaving_optimize(a, remaining,
                                             top_k_start=10, top_k_end=60,
                                             temp_start=3.0, temp_end=8.0,
                                             c0=3.0, verbose=False)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()

    return best_a, best_m


def make_comb(n, period=3, duty=1):
    """Create a comb pattern: [1,0,0,1,0,0,...]"""
    a = np.zeros(n)
    for i in range(0, n, period):
        for j in range(min(duty, period)):
            a[i+j] = 1.0
    if a.sum() < 1e-15:
        a = np.ones(n)
    return norm_sum1(a)


def make_block(n, block_on=3, block_off=3):
    """Create a block pattern: [1,1,1,0,0,0,1,1,1,...]"""
    a = np.zeros(n)
    period = block_on + block_off
    for i in range(0, n, period):
        a[i:i+block_on] = 1.0
    if a.sum() < 1e-15:
        a = np.ones(n)
    return norm_sum1(a)


def make_tapered(n, alpha=2.0):
    """Tapered edges: high in middle, low at edges (raised cosine-like)."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** alpha
    # Actually let's do: high at center, taper at edges
    t = np.abs(x - 0.5) * 2  # 0 at center, 1 at edges
    a = (1 - t) ** alpha
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def make_edge_concentrated(n, alpha=2.0):
    """Mass concentrated at edges."""
    x = np.linspace(0, 1, n)
    t = np.abs(x - 0.5) * 2  # 1 at edges, 0 at center
    a = t ** alpha + 0.1
    return norm_sum1(a)


def make_bimodal(n, sep=0.5, width=0.15):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    c1 = (1 - sep) / 2
    c2 = (1 + sep) / 2
    a = np.exp(-((x - c1) ** 2) / (2 * width ** 2)) + np.exp(-((x - c2) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_sum1(a)


def make_multipeak(n, n_peaks=5, width=0.08):
    """Multiple evenly-spaced Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_peaks):
        center = (i + 1) / (n_peaks + 1)
        a += np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a += 0.05
    return norm_sum1(a)


def make_power_law(n, power=1.0):

def make_cosine_window(n, mode=1):

def make_random_smooth(n, seed=0, n_freq=8):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    """Compute gradient of (a*a)_k w.r.t. a.
