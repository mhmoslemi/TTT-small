        else:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane accelerated Frank-Wolfe with adaptive constraint subset and line search
2. Symmetry exploitation (only first n of 2n-1 autocorrelation lags needed)
3. Soft-max mirror descent with adaptive temperature for polishing
4. Multi-length exploration (n=3200, 4000, 4800)
5. Nesterov momentum and multiple diverse restarts
Key innovation: LP without sum=1 constraint (maximizes sum(b) for well-defined direction),
symmetry halving, and aggressive line search to accelerate convergence.

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute conv(a, a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def normalize_nonneg(a):
    """Clip negatives and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_constrained_LHS(a, top_indices):
    """
    Build the constraint matrix rows for the LP.
    For each lag m in top_indices, the row is: [a[m], a[m-1], ..., 0, 0, ..., 0]
    (padded to length n), representing the coefficients of b in conv(a,b)[m].
    Only uses the first n lags by symmetry.
    """
    n = len(a)
    K = len(top_indices)
    T = np.zeros((K, n))
    for col_idx, m in enumerate(top_indices):
        mi = int(m)
        if mi < 0 or mi >= 2 * n - 1:
            continue
        j_start = max(0, mi - n + 1)
        j_end = min(n - 1, mi)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        T[col_idx, js] = a[mi - js]
    return T


def lp_fw_optimize(a0, time_limit, solver, K_init=50, K_growth=1.3,
                   K_max_frac=1.0, verbose=True, seed=42):
    """
    Cutting-plane Frank-Wolfe with line search.
    At each iteration:
      1. Compute autocorrelation
      2. Select top-K lags (growing over time)
      3. Solve LP: max sum(b) s.t. b>=0, conv(a,b)[top-K] <= M
      4. Line search along direction b/sum(b)
      5. Nesterov extrapolation
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_M = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    K = K_init
    K_max = int(K_max_frac * n)  # use first n lags by symmetry
    y = a.copy()  # Nesterov point
    x_mom = a.copy()
    prev_M = best_M
    stall = 0
    max_stall = 2000

    iter_count = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iter_count += 1
        elapsed_check = time.monotonic() - t0

        # Compute autocorrelation at Nesterov point
        ac_y = fast_autocorr(y)
        M_y = ac_y.max()

        # Select top-K lags
        K_curr = min(K, n)  # at most n (symmetry)
        if K_curr >= len(ac_y):
            top_idx = np.arange(len(ac_y))
        else:
            top_idx = np.argsort(ac_y)[::-1][:K_curr]
        top_idx = np.sort(top_idx)  # sort for determinism

        # Build constraint matrix
        T = build_constrained_LHS(y, top_idx)
        # Remove rows that are all zero
        row_sums = T.sum(axis=1)
        valid = row_sums > 1e-20
        T = T[valid]
        rhs = M_y * np.ones(len(T))
        K_sel = len(T)

        if K_sel == 0:
            direction = np.ones(n) / n
        else:
            # Solve LP: max sum(b) s.t. b >= 0, T @ b <= rhs, sum(b) <= 2
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= rhs, cp.sum(b_var) <= 2.0]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                direction = np.ones(n) / n
            else:
                direction = b_val / b_sum

        # Line search: try multiple alpha values
        candidates = [2.0 / (iter_count + 2.0), 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 0.95]
        candidates = [min(max(c, 0.01), 0.99) for c in candidates]
        # Remove duplicates
        seen = set()
        unique_candidates = []
        for c in candidates:
            cr = round(c, 4)
            if cr not in seen:
                seen.add(cr)
                unique_candidates.append(c)

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = float('inf')
        a_current = normalize_nonneg(a)

        for alpha in unique_candidates:
            a_trial = (1.0 - alpha) * a_current + alpha * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        # Update
        a_new = (1.0 - best_alpha) * a_current + best_alpha * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()

        # Nesterov update
        beta_t = (iter_count - 1.0) / (iter_count + 2.0)
        y_new = a_new + beta_t * (a_new - x_mom)
        y_new = np.maximum(y_new, 1e-15)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        x_mom = a_new
        a = a_new
        # Update y at the Nesterov point for next iteration
        # Re-evaluate at y for next iteration's constraint selection
        y = y_new

        # Track best
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Also check y
        M_y_check = fast_autocorr(y).max()
        if M_y_check < best_M - 1e-14:
            best_M = M_y_check
            best_a = y.copy()
            stall = 0

        # Grow K
        if iter_count % 30 == 0:
            K = min(int(K * K_growth), K_max)

        # Periodic restart if stalled
        if stall > 500 and stall % 500 == 0:
            # Perturb around best
            pert_scale = 0.02 * (0.5 + 0.5 * rng.random())
            pert = rng.normal(0, pert_scale, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            x_mom = a_r.copy()
            y = a_r.copy()
            stall = 0
            if verbose:
                print(f"    LP-FW restart ({time.monotonic()-t0:.0f}s): J={2.0*n*best_M:.6f}")

        if verbose and iter_count % 200 == 0:
            J = 2.0 * n * best_M
            print(f"    LP-FW iter={iter_count} K={K_curr} ({elapsed_check:.0f}s): J={J:.6f}")

    if verbose:
        J = 2.0 * n * best_M
        print(f"    LP-FW final ({time.monotonic()-t0:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_M


def mirror_descent_optimize(a0, time_limit, beta_init=10.0, beta_min=3.0,
                            c0=4.0, restart_every=4000, seed=42, verbose=True):
    """
    Nesterov-accelerated mirror descent with adaptive soft-subgradient.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta_init
    stall_count = 0

    for iterations in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft subgradient
        ac_y = fast_autocorr(y)
        shifted = ac_y - ac_y.max()
        w = np.exp(cur_beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac_y) / len(ac_y)
        else:
            w = w / w_sum

        # Gradient via FFT
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        a_rev_padded = np.zeros(fft_size)
        a_rev_padded[:n] = y[::-1]
        w_padded = np.zeros(fft_size)
        w_padded[:len(w)] = w
        conv = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
        g = 2.0 * conv[n - 1:2 * n - 1]

        # Mirror descent step
        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x.copy()
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive beta
        cur_beta = max(beta_min, beta_init - 0.002 * iterations)

        # Restart
        if stall_count > restart_every:
            stall_count = 0
            pert_scale = 0.03
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if rng.random() < 0.2:
                x_r = rng.dirichlet(np.ones(n) * 0.8)
                x = x_r
                x_prev = x_r.copy()
            if verbose:
                J = 2.0 * n * best_m
                print(f"    MD restart iter={iterations} ({time.monotonic()-t0:.0f}s): J={J:.6f}")

        if verbose and iterations % 2000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD iter={iterations} ({time.monotonic()-t0:.0f}s): J={J:.6f}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    MD final ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, seed=42, verbose=True):
    """
    Fast coordinate descent focusing on peak-reducing moves.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    moves = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peak lags
        n_top = min(20, n)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destinations
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    sens[d] += a[partner]

            # Prefer far, low-sensitivity destinations
            dist = np.abs(np.arange(n) - src).astype(float)
            dist[dist == 0] = 1.0
            scores = sens / (1.0 + dist / (n * 0.1))
            candidates = np.argsort(scores)[:min(n // 4, 500)]

            for dst in candidates[:20]:
                if dst == src:
                    continue
                for frac in [0.03, 0.05, 0.01, 0.08, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        moves += 1
                        break
                if improved:
                    break
            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and (moves % 200 == 0 or stall % 100 == 0):
            J = 2.0 * n * best_M
            print(f"    Coord ({time.monotonic()-t0:.0f}s): J={J:.6f}, moves={moves}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_M
        print(f"    Coord final ({time.monotonic()-t0:.0f}s): J={J:.6f}, moves={moves}")
    return best_a, best_M


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, seed=0, style="smooth"):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if style == "smooth":
        a = np.ones(n) * 0.5
        for f in [1, 2, 3, 4, 5]:
            coeff = rng.normal(0, 0.1)
            phase = rng.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * f * x + phase)
        a = np.maximum(a, 0.0)
    elif style == "bimodal":
        a = 0.3 + 0.4 * np.exp(-((x - 0.25) ** 2) / 0.02) + 0.4 * np.exp(-((x - 0.75) ** 2) / 0.02)
        a += rng.normal(0, 0.02, n)
        a = np.maximum(a, 0.0)
    elif style == "cosine":
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x * 3 + rng.uniform(0, 2 * np.pi))
        a += rng.normal(0, 0.01, n)
        a = np.maximum(a, 0.0)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return normalize_nonneg(a)

        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-FW on parent length (main phase) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW (n={n0}) ===")
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t1, solver, K_init=50, K_growth=1.4,
                                  K_max_frac=1.0, verbose=True, seed=42)
        try_update(a_out, f"lpfw1_n{n0}")

    # === Phase 2: Mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: MD Polish (n={n_cur}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t2, beta_init=12.0, beta_min=4.0,
                                           c0=3.0, restart_every=3000, seed=77, verbose=True)
        try_update(a_out, f"md_n{n_cur}")

    # === Phase 3: Second LP-FW round on improved seq ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP-FW Round 2 (n={n_cur}) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t3, solver, K_init=100, K_growth=1.5,
                                  K_max_frac=1.0, verbose=True, seed=123)
        try_update(a_out, f"lpfw2_n{n_cur}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord Polish (n={n_cur}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, seed=555, verbose=True)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Try larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    target_lengths = [n_cur + 500, n_cur + 1000]
    target_lengths = [l for l in target_lengths if 3000 <= l <= 6000]

    for new_n in target_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: Larger n={new_n} ===")
        a_interp = interpolate_to_length(best_seq, new_n)

        t5a = min(remaining * 0.25, 200.0)
        if t5a > 30:
            a_out, _ = lp_fw_optimize(a_interp, t5a, solver, K_init=50, K_growth=1.4,
                                      K_max_frac=1.0, verbose=True, seed=999)
            try_update(a_out, f"lpfw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.20, 120.0)
            if t5b > 20:
                a_out, _ = mirror_descent_optimize(a_interp, t5b, beta_init=8.0, beta_min=3.0,
                                                   c0=3.0, restart_every=2000, seed=888, verbose=False)
                try_update(a_out, f"md_n{new_n}")

    # === Phase 6: Final LP polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_cur}) ===")
    t6 = min(remaining * 0.40, 200.0)
    if t6 > 30:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t6, solver, K_init=200, K_growth=1.5,
                                  K_max_frac=1.0, verbose=True, seed=314)
        try_update(a_out, f"lpfw_final_n{n_cur}")

    # === Phase 7: Final coordinate + MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final Polish (n={n_cur}) ===")
        t7a = min(remaining * 0.4, 100.0)
        if t7a > 15:
            a_in = normalize_nonneg(best_seq)
            a_out, _ = coordinate_polish(a_in, t7a, seed=271, verbose=False)
            try_update(a_out, f"coord_final_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.5, 80.0)
            if t7b > 10:
                a_in = normalize_nonneg(best_seq)
                a_out, _ = mirror_descent_optimize(a_in, t7b, beta_init=15.0, beta_min=6.0,
                                                   c0=2.0, restart_every=5000, seed=161, verbose=False)
                try_update(a_out, f"md_ult_final_n{n_cur}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 3200
        best_seq = make_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
