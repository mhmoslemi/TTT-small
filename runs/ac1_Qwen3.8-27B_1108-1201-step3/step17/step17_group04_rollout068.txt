def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, np.maximum(f_norm, 0.0))
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. FFT-accelerated Nesterov exponentiated gradient descent with log-sum-exp smoothing
2. Multi-scale temperature annealing for global-to-local transition
3. B-spline, Gaussian, and power-law initializations
4. Peak-shaving coordinate refinement
5. Multi-start with diverse initializations and length exploration
Key innovation: mirror descent (exponentiated gradient) naturally handles simplex
constraint, combined with Nesterov acceleration and adaptive log-sum-exp temperature
for smooth-yet-accurate max-convolution optimization.
"""

import numpy as np
import time


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_obj(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def soft_max_vals(ac, temp):
    m = ac.max()
    scaled = (ac - m) / temp
    scaled = np.clip(scaled, -700.0, 0.0)
    w = np.exp(scaled)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(ac)
        w[np.argmax(ac)] = 1.0
    else:
        w = w / sw
    return w


def nesterov_exp_gradient(a0, time_limit, temp0=None, lr0=0.5,
                           verbose=False, seed=42, max_iter=5000000,
                           cool_rate=0.99995, restart_stall=4000):
    a0 = np.asarray(a0, dtype=np.float64)
    n = len(a0)
    if temp0 is None:
        temp0 = 3.0 / n
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    N_fft = 1 << (3 * n + 16).bit_length()
    temp = temp0
    lr = lr0
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    best_stall = 0
    min_temp = temp0 * 0.01

    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 2:
            mu = (t - 2.0) / (t + 2.0)
            y = x + mu * (x - x_prev)
        else:
            y = x.copy()

        y = np.maximum(y, 1e-15)
        ys = y.sum()
        if ys < 1e-15:
            y = x.copy()
        else:
            y = y / ys

        ac = fast_autocorr(y)
        w = soft_max_vals(ac, temp)

        W = np.fft.rfft(w, N_fft)
        Y = np.fft.rfft(y, N_fft)
        corr = np.fft.irfft(W * np.conj(Y), N_fft)
        grad = 2.0 * corr[:n]

        g_scale = np.abs(grad).max()
        if g_scale < 1e-15:
            x_prev = x
            x = a.copy()
            continue

        lr_eff = min(lr / g_scale, 3.0)
        step_factors = np.exp(-lr_eff * grad / g_scale)
        step_factors = np.clip(step_factors, 1e-10, 1e10)

        x_new = y * step_factors
        x_new = np.maximum(x_new, 1e-30)
        xs = x_new.sum()
        if xs < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / xs

        x_prev = x
        x = x_new

        J_new = compute_obj(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1

        if stall > 800:
            lr = max(lr * 0.85, 0.05)
            stall = 0

        temp = max(temp * cool_rate, min_temp)

        if best_stall > restart_stall:
            sigma = 0.015 + 0.005 * min(best_stall // restart_stall, 10)
            pert = rng.normal(0.0, sigma, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            x_prev = x.copy()
            temp = min(temp * 2.0, temp0 * 0.3)
            lr = max(lr, 0.3)
            best_stall = 0

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ExpGrad t={t} ({elapsed:.0f}s): J={best_J:.8f}, temp={temp:.6f}, lr={lr:.3f}")

    return best_a, best_J


def peak_shave(a0, time_limit, verbose=False, seed=77):
    a0 = np.asarray(a0, dtype=np.float64)
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_order = np.argsort(ac)[::-1][:8]

        contrib_vals = np.zeros(n)
        for rank, k in enumerate(top_order):
            k_int = int(k)
            wt = 1.0 / (rank + 1)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    contrib_vals[i] += wt * a[i] * a[j]
                    contrib_vals[j] += wt * a[i] * a[j]

        high_idx = np.where((contrib_vals > contrib_vals.mean() + 0.3 * contrib_vals.std()) & (a > 1e-7))[0]
        low_idx = np.where((contrib_vals < contrib_vals.mean() - 0.2 * contrib_vals.std()) | (contrib_vals < 1e-12))[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            stall += 1
            if stall % 50 == 0:
                p = int(rng.integers(n))
                q = int(rng.integers(n))
                if p != q and a[p] > 1e-8:
                    amt = a[p] * rng.uniform(0.005, 0.05)
                    a[p] -= amt
                    a[q] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    J_new = compute_obj(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
            continue

        improved = False
        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break

            src = int(high_idx[rng.integers(len(high_idx))])
            dst = int(low_idx[rng.integers(len(low_idx))])
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-9:
                continue

            fracs = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.5]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                J_trial = compute_obj(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found and stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    J_new = compute_obj(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                npert = max(3, n // 100)
                perts = rng.choice(n, size=npert, replace=False)
                for p in perts:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.1)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst2 = int(rng.integers(n))
                        a[dst2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_obj(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): J={best_J:.8f}, stall={stall}")

    return best_a, best_J


def gaussian_init(n, sigma=0.25):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return norm_sum1(a)


def power_law_init(n, power=0.7):
    x = np.linspace(0.0, 1.0, n)
    a = np.maximum(1.0 - np.abs(x - 0.5) * 2.0, 0.0) ** power + 0.001
    return norm_sum1(a)


def bspline_init(n, order=3):
    x = np.linspace(-0.5, 0.5, n)
    a = np.maximum(1.0 - np.abs(x) * 2.0, 0.0)
    for _ in range(order - 1):
        ac = fast_autocorr(a)
        a = ac[:n]
        s = a.sum()
        if s > 1e-15:
            a = a / s
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return np.asarray(f, dtype=np.float64).copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, np.maximum(f_norm, 0.0))
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
