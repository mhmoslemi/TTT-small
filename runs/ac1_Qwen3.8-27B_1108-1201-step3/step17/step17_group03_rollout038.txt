"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative mirror descent with Nesterov momentum and soft-peak weighting
2. LP Frank-Wolfe for directed refinement (adaptive margins)
3. Targeted peak-crushing coordinate descent
4. Multi-length exploration with interpolation
5. Frequency-domain diversification for escaping local minima
Key insight: subgradient of c_k w.r.t. a[p] = 2*a[k-p] * 1[0<=k-p<n]
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full convolution using FFT - O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgradient(a, ac, top_k=80, temperature=5.0):
    """
    Compute weighted subgradient from top peaks.
    dc_k/da[p] = 2*a[k-p] * 1[0 <= k-p < n]
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = (top_vals / top_vals.max()) ** temperature
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        p_start = max(0, k_int - n + 1)
        p_end = min(n, k_int + 1)
        if p_start >= p_end:
            continue
        idx = np.arange(p_start, p_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def mirror_descent(a0, time_limit, top_k=80, temperature0=12.0, temperature_min=2.5,
                   c0=4.0, momentum=0.92, verbose=True, rng_seed=42,
                   restart_count=2000):
    """
    Fast multiplicative mirror descent with Nesterov momentum.
    Uses correct subgradient: dc_k/da[p] = 2*a[k-p]*1[0<=k-p<n]
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for it in range(1, 50_000_000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum (ramp up over first 200 iters)
        beta = momentum * min(1.0, it / 200.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Adaptive temperature: start broad, narrow focus
        frac = min(1.0, it / (time_limit * 50.0)) if time_limit > 0 else 1.0
        cur_temp = temperature0 + (temperature_min - temperature0) * frac

        # Adaptive step size
        eta = c0 / (np.sqrt(it + 1) + 1.0)

        # Compute subgradient
        g = compute_subgradient(y, ac, top_k=top_k, temperature=cur_temp)

        # Multiplicative (exponential) update
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation when stuck
        if stall > restart_count:
            stall = 0
            # Frequency-domain perturbation
            fft_size = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(best_a, fft_size)
            phase = np.angle(A)
            phase_pert = phase + rng.normal(0, 0.02, len(phase))
            A_new = np.abs(A) * np.exp(1j * phase_pert)
            a_pert = np.fft.irfft(A_new, fft_size)[:n]
            a_pert = np.maximum(a_pert, 1e-15)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
            x = a_pert
            x_prev = a_pert.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD it={it} ({elapsed:.0f}s): J={J:.6f}, temp={cur_temp:.1f}, stall={stall}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, top_K=200, margin_frac=2e-3, verbose=True):
    """
    LP-based Frank-Wolfe with adaptive constraint selection.
    Solves: max sum(b) s.t. T@b <= M, b >= 0
    where T is built from active constraints (top peaks).
    """
    try:
        import cvxpy as cp
    except ImportError:
        if verbose:
            print("    cvxpy not available, skipping LP phase")
        return a0, fast_autocorr(a0).max()

    # Find a working solver
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for it in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active constraints: peaks within margin_frac of max
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix T: T[row, i] = y[k-row - i] for valid indices
        T = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            p_start = max(0, int(k_int) - n + 1)
            p_end = min(n, int(k_int) + 1)
            if p_start >= p_end:
                continue
            idx = np.arange(p_start, p_end)
            j = int(k_int) - idx
            T[row, idx] = y[j]

        # Solve LP: max sum(b) s.t. T@b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # FW update with Nesterov
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov step
        if it == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (it - 1.0) / (it + 2.0)
            x_new = y_new + 0.9 * beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new

        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten margin as we improve
            margin_frac = max(margin_frac * 0.85, 1e-5)

        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()

        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP it={it} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m

def peak_crush(a0, time_limit, verbose=True, rng_seed=123):
    """
    Targeted coordinate descent: move mass from peak-contributing elements
    to random non-contributing elements to reduce the maximum autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Find contributing pairs for peak_k
        p_start = max(0, peak_k - n + 1)
        p_end = min(n, peak_k + 1)
        if p_start >= p_end:
            stall += 1
            continue

        idx = np.arange(p_start, p_end)
        j = peak_k - idx
        contrib = a[idx] * a[j]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        # Try moving mass from top contributing elements
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try different fractions and destinations
            best_move_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(20):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_move_m:
                        best_move_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if i != j2 and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def freq_perturb(a, strength=0.015, rng=None):
    """Perturb in frequency domain to create diverse candidates."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = np.abs(A) * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def make_parametric(n, family='sine'):
    """Create parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'raised_cos':
        x = np.linspace(0, 1, n)
        return norm_sum1((1 + np.cos(2 * np.pi * x)) / 2 + 0.01)
    elif family == 'beta_sym':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 1.5 * (1 - x) ** 1.5)
    elif family == 'gauss':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.exp(-4 * (x - 0.5) ** 2))
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

def try_update(a, best_seq, best_score, t_global, label=""):

    print(f"\n=== Phase 5: Multi-length (n={n_best}) ===")
    longer_lengths = sorted(set([n_best + 400, n_best + 800, n_best + 1200]))
    longer_lengths = [l for l in longer_lengths if l <= 6000 and remaining > 100]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.15, 120.0)
        if t5a > 20:
            a_out, _ = mirror_descent(a_long, t5a, top_k=80, temperature0=10.0,
                                       temperature_min=3.0, c0=4.0, momentum=0.92,
                                       verbose=False, restart_count=4000)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 4500:
            t5b = min(remaining * 0.08, 80.0)
            if t5b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_crush(a_in, t5b, verbose=False, rng_seed=77)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"pc_n{new_n}")

    # === Phase 6: Frequency diversification + re-optimize (8% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Freq-diversify (n={n_best}) ===")
    rng = np.random.default_rng(99)
    t6_per = min(remaining * 0.06, 80.0)
    if t6_per > 20:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 20:
                break
            a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
            a_out, _ = mirror_descent(a_pert, t6_per, top_k=70, temperature0=10.0,
                                       temperature_min=3.0, c0=4.5, momentum=0.93,
                                       verbose=False, restart_count=2000)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 7: Final LP refinement (5% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.05, 100.0)
    if t7 > 20 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, top_K=350, margin_frac=5e-4, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final peak-crush (5% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final peak-crush (n={n_best}) ===")
    t8 = min(remaining * 0.05, 80.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush(a_in, t8, verbose=True, rng_seed=555)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety check
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
