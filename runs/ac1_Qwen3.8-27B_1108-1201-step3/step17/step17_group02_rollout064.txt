"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP-based Frank-Wolfe with Nesterov acceleration (exact FW step when n small)
2. Top-K adaptive-constraint LP-FW for larger n
3. Peak-aware accelerated mirror descent with correct subgradients
4. Sensitivity-based coordinate descent polishing
5. Multi-length exploration via interpolation + stochastic restarts
The objective is scale-invariant so we always minimize max_k (a*a)[k] with sum(a)=1.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")

        except Exception:
            continue

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def max_ac(a):
    return float(fast_autocorr(a).max())


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_triangular(n):
    x = np.arange(n)
    a = np.minimum(x + 1, n - x)
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0.0, np.pi, n)
    a = np.sin(x) ** 2
    return norm_sum1(a)


def make_gaussian(n, sigma_frac=0.3):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1.0, 1.0, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=24, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0.0, 0.18 / np.sqrt(freq))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += coeff * np.sin(2.0 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def lp_solve_fw_direction(a, M, solver, top_K=None):
    """Solve max sum(b) s.t. (a*b)[k] <= M for selected k, b>=0; return b/sum(b) or None."""
    n = len(a)
    num_constraints = 2 * n - 1
    if top_K is not None and top_K < num_constraints:
        ac = fast_autocorr(a)
        margins = M - ac
        binding_idx = np.argsort(margins)[:top_K]
    else:
        binding_idx = np.arange(num_constraints)
    n_bc = len(binding_idx)
    T = np.zeros((n_bc, n))
    for ii in range(n_bc):
        k = int(binding_idx[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            T[ii, j_start:j_end] = a[k - j_end + 1 : k - j_start + 1][::-1]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def fw_optimize(a0, time_limit, solver, top_K=None, verbose=True, rng_seed=42):

        M_curr = max_ac(y_new)
        g = lp_solve_fw_direction(y_new, M_curr, solver, top_K=top_K)

        if g is None:

            x = y_new
            iters += 1
            stall += 1

def compute_subgradient(a, ac, top_k=120, temp_scale=0.2):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    denom = max(float(ac[top_idx].max()) * temp_scale, 1e-12)
    w = np.exp((top_vals - top_vals.max()) / denom)
    w_sum = w.sum()
    w = w / w_sum if w_sum > 1e-30 else np.ones(k_max) / k_max
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

        y = y / sy if sy > 1e-15 else x.copy()
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_k)
        eta = gamma / (np.sqrt(iters) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        a_new = a_new / sn if sn > 1e-15 else y.copy()
        x_prev = x
        x = a_new
        mnew = max_ac(x)
        if mnew < best_m - 1e-16:

            stall = 0
        else:

            x = x_r
            x_prev = x_r.copy()
        if verbose and iters % 20000 == 0:
            print(f"    MD it={iters} ({time.monotonic()-t0:.0f}s): J={2.0*n*best_m:.6f} stall={stall}")
    if verbose:

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
