import numpy as np

import cvxpy as cp

_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDL9P, cp.SCS]:

        except Exception:

def fast_conv(a, b=None):
    n1 = len(a)
    n2 = len(b) if b is not None else n1
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    if b is None:
        B = A
    else:
        B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def peak_subgradient(a, beta=20.0, top_k=200):
    """
    Compute subgradient of max(conv(a,a)) using soft-max over top-k lags.
    Returns (gradient, max_value).
    """
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k_max) / k_max
    else:
        w_top = w_top / w_sum
    
    # Build sparse weight vector
    w = np.zeros(len(ac))
    for wi, idx in zip(w_top, top_idx):
        w[int(idx)] = wi
    
    # grad[j] = 2 * sum_m w[m] * a[m-j] (valid range)
    # = 2 * conv(a_rev, w)[n-1 : 2n-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev_padded = np.zeros(fft_size)
    a_rev_padded[:n] = a[::-1]
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv_result = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad, ac.max()

def multiplicative_gd(a0, time_limit, verbose=True, beta=15.0, c0=3.0,
                      restart_every=3000, seed=42):
    """
    Peak-directed multiplicative gradient descent with Nesterov momentum.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        g, M_curr = peak_subgradient(y, beta=beta, top_k=200)
        
        # Adaptive step size
        eta = gamma / (np.sqrt(iterations) + 1.0)
        
        # Multiplicative (exponential) update
        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with spectral perturbation
        if stall > restart_every:
            stall = 0
            # Spectral perturbation: add a random sinusoid
            f = rng.integers(1, n // 2)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            perturbation = amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
            x_new = best_a + perturbation * best_a
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new.copy()
            
            # Sometimes try a different base
            if rng.random() < 0.2:
                x_new = best_a * (1 + amp * rng.normal(0, 1, n))
                x_new = np.maximum(x_new, 1e-10)
                x_new = x_new / x_new.sum()
                x = x_new
                x_prev = x_new.copy()
        
        if verbose and iterations % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MDGD iter={iterations} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, full_constraints=True):
    """
    Correct LP Frank-Wolfe: linear subproblem minimizes gradient @ b.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient
        g, M_curr = peak_subgradient(a, beta=10.0, top_k=300)
        
        if M_curr < 1e-16:
            break
        
        # Build constraint matrix T (2n-1 x n Toeplitz)
        if full_constraints:
            K = 2 * n - 1
            T = np.zeros((K, n))
            for j in range(n):
                T[j:j+n, j] = a
        else:
            # Use only top constraints
            ac = fast_autocorr(a)
            K = min(500, 2 * n - 1)
            top_idx = np.argsort(ac)[::-1][:K]
            T = np.zeros((K, n))
            for col_idx, m in enumerate(top_idx):
                mi = int(m)
                js = max(0, mi - n + 1)
                je = min(n - 1, mi)
                idx = np.arange(js, je + 1)
                T[col_idx, idx] = a[mi - idx]
        
        # LP: minimize g @ b s.t. T @ b <= M, b >= 0, sum(b) = 1
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(g @ b_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        s_star = b_val / b_sum
        
        # FW step
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * s_star
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        m_new = fast_autocorr(a).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def parametric_init(n, f1, c1, f2=0, c2=0, p=1.0, seed=0):
    """Generate a[i] = (1 + c1*cos(2πf1*i/n + φ1) + c2*cos(2πf2*i/n + φ2))^p"""
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    base = np.ones(n)
    if c1 != 0:
        phi1 = rng.uniform(0, 2 * np.pi)
        base += c1 * np.cos(2 * np.pi * f1 * i / n + phi1)
    if c2 != 0:
        phi2 = rng.uniform(0, 2 * np.pi)
        base += c2 * np.cos(2 * np.pi * f2 * i / n + phi2)
    base = np.maximum(base, 0.01)
    a = base ** p
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
