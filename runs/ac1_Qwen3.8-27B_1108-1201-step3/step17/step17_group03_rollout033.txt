def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_softmax_subgradient(a, ac, top_k=50, beta=5.0):
    """Compute soft subgradient using softmax-weighted top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Softmax weights
    max_val = top_vals.max()
    exp_vals = np.exp(beta * (top_vals - max_val))
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_hard_subgradient(a, ac, top_k=50):
    """Compute hard subgradient at top-k peaks (uniform weights)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * a[j_idx] / k_max
    return grad

        for ls_iter in range(10):
            # Mirror descent step (log space)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-20:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        
        if not accepted:
            # Accept anyway with smallest step
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart with structured perturbation
        if stall_count > restart_every:
            stall_count = 0
            pert_type = rng.integers(3)
            if pert_type == 0:
                # Gaussian noise
                pert = rng.normal(0, 0.02, n) * best_a
                x_r = best_a + pert
            elif pert_type == 1:
                # Frequency domain shift
                fft_size = 1 << (2*n - 1).bit_length()
                A = np.fft.rfft(best_a, fft_size)
                shift = rng.normal(0, 0.02, len(A))
                A_pert = A * np.exp(1j * shift)
                x_r = np.fft.irfft(A_pert, fft_size)[:n]
            else:
                # Local value swap
                x_r = best_a.copy()
                for _ in range(n // 100):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j:
                        amt = x_r[i] * rng.uniform(0.01, 0.05)
                        x_r[i] -= amt
                        x_r[j] += amt
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta_t:.1f}, top_k={top_k_t}, stall={stall_count}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_K=1000):
    """
    LP Frank-Wolfe with adaptive active set and Nesterov acceleration.
    """
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    margin_frac = 2e-3
    stall_count = 0
    
    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < 1e-16:
            break
        
        # Adaptive active set
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        
        # Scale active set size with progress
        progress = t / 500.0
        target_K = min(int(100 + progress * 50), max_K)
        
        if n_active > target_K:
            active_idx = np.argsort(margins)[:target_K]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:target_K]
        else:
            active_idx = np.where(active_mask)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix
        # For each active lag k, the constraint is sum_i y[i]*y[k-i] <= M_curr
        # Linearized: sum_i [y[k-i]*y[i] + y[i]*y[k-i]] / 2 * delta_i ... 
        # Actually: d/d(a_i) [sum_j a[j]*a[k-j]] = a[k-i] + a[i-k] (with bounds)
        # For the LP, we linearize the constraint around y:
        # sum_i (a[k-i] + a[i-k]) * b_i <= ... hmm, this is the gradient, not the constraint.
        
        # The constraint is: sum_i b[i] * y[k-i] <= M_curr (using y as the "partner")
        # This is a linear constraint in b!
        # Because conv(y,b)[k] = sum_i y[i]*b[k-i], and we want this <= M_curr.
        # Wait, the original constraint is conv(a,a)[k] <= M.
        # Linearizing around the current point y: 
        # conv(y + delta, y + delta)[k] ≈ conv(y,y)[k] + conv(y,delta)[k] + conv(delta,y)[k]
        # We want conv(y,delta)[k] + conv(delta,y)[k] <= 0 (to not increase the peak)
        # conv(y,delta)[k] = sum_i y[i]*delta[k-i]
        # conv(delta,y)[k] = sum_i delta[i]*y[k-i]
        # Together: sum_i (y[k-i] + y[i-k]) * delta[i] ... 
        # Hmm, but delta = b - y where b is the new direction.
        
        # Actually, in Frank-Wolfe, we find b that maximizes sum(b) subject to
        # the linearized constraints. The linearized constraint at lag k is:
        # grad_k . b <= M_curr - conv(y,y)[k] + ... 
        # Actually, the standard FW approach here is:
        # Find b >= 0 that maximizes sum(b) subject to conv(y, b)[k] <= M_curr for active k.
        # This is because we're looking for a direction b such that moving from y
        # toward b doesn't increase the autocorrelation peaks.
        
        # conv(y, b)[k] = sum_i y[i] * b[k-i] (with appropriate bounds)
        # This is linear in b!
        
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                # conv(y,b)[k] = sum_i y[i] * b[k-i]
                # For a given k, b index is k-i
                b_idx = k_int - i
                if 0 <= b_idx < n:
                    T_sel[row, i] = y[b_idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # FW step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov step
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Track best
        my = fast_autocorr(y).max()
        mx = fast_autocorr(x).max()
        
        improved = False
        if my < best_m:
            best_m = my
            best_a = y.copy()
            improved = True
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            improved = True
        
        if improved:
            stall_count = 0
            # Tighten margin when improving
            margin_frac = max(margin_frac * 0.95, 1e-5)
        else:
            stall_count += 1
            if stall_count > 100:
                # Loosen margin and restart
                margin_frac = min(margin_frac * 2.0, 5e-3)
                stall_count = 0
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")
    
    return best_a, best_m


def polish_active_set(a0, time_limit, solver, verbose=True):
    """
    Polish by solving a sequence of LPs with increasingly tight active sets.
    Goal: equalize the top peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Multiple passes with different K values
    for K in [50, 100, 200, 400, 600, 800]:
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top K peaks
        top_idx = np.argsort(ac)[::-1][:K]
        
        # Solve LP: maximize sum(b) s.t. conv(a, b)[k] <= M_curr for k in top_idx, b >= 0
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(top_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                b_idx = k_int - i
                if 0 <= b_idx < n:
                    T_sel[row, i] = a[b_idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_dir = b_val / b_sum
        
        # Move toward b_dir
        # Try different step sizes
        for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1 - alpha) * a + alpha * b_dir
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                break
        
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        if verbose:
            print(f"    Polish K={K} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=42):
    """
    Targeted peak-crushing: identify the maximum peak and redistribute mass
    from contributing elements to random targets.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find top contributors to this peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total = contrib.sum()
        
        if total < 1e-20:
            stall += 1
            continue
        
        # Try to reduce the peak by moving mass from high contributors
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        
        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            best_new_m = M_curr
            best_move = None
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(50):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)
            
            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush done ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def make_parametric(n, family='sine'):
    """Generate parametric initial sequences."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

def norm_sum1(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_lp_matrix(a, active_idx, n):
    """Build constraint matrix for LP: T @ b <= M where row k is conv(a,b)[k]."""
    K = len(active_idx)
    T = np.zeros((K, n))
    for row, k_int in enumerate(active_idx):
        k_int = int(k_int)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            b_idx = k_int - i
            if 0 <= b_idx < n:
                T[row, i] = a[b_idx]
    return T


def compute_softmax_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    max_val = top_vals.max()
    exp_vals = np.exp(beta * (top_vals - max_val))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def peak_crush(a0, time_limit, verbose=True, rng_seed=42):

            if src_val < 1e-12:
                continue

                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:

                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)

                M_new = fast_autocorr(a).max()

def make_parametric(n, family='sine'):

        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search: combines AMD, LP-FW, multi-K polishing, peak crushing,
    and multi-length exploration to minimize the evaluation function.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
        best_n = len(best_seq)
    else:
        best_n = 3600
        best_seq = make_parametric(best_n, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={best_n}, score={best_score:.6f}")

    # === Phase 1: AMD with adaptive temperature ===
    print(f"\n=== Phase 1: AMD adaptive (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.13, 250.0)
    if t1 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t1, verbose=True, use_adaptive_temp=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_adapt")

    # === Phase 2: LP Frank-Wolfe main ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.22, 420.0)
    if t2 > 50 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, max_K=800, init_margin=2e-3)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Multi-K polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Multi-K polish (n={n_best}) ===")
    t3 = min(remaining * 0.12, 210.0)
    if t3 > 30 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = polish_multi_k(a_in, t3, solver, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "polish1")

    # === Phase 4: Peak crushing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak crush (n={n_best}) ===")
    t4 = min(remaining * 0.10, 170.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush(a_in, t4, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "peakcrush1")

    # === Phase 5: Second LP-FW with tighter params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW tight (n={n_best}) ===")
    t5 = min(remaining * 0.16, 280.0)
    if t5 > 40 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, max_K=1000, init_margin=1e-3)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = sorted(set([
        max(1500, n_best - 600),
        max(1500, n_best - 300),
        n_best + 300,
        n_best + 600,
        n_best + 1000,
    ]))
    candidate_lengths = [l for l in candidate_lengths if l <= 6500 and remaining > 150]

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Explore n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.12, 120.0)
        if t6a > 30:
            a_out, _ = amd_optimize(a_long, t6a, verbose=False, use_adaptive_temp=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5000:
            t6b = min(remaining * 0.10, 100.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False, max_K=500, init_margin=2e-3)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 7: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    t7 = min(remaining * 0.18, 220.0)
    if t7 > 30 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = polish_multi_k(a_in, t7, solver, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "polish_final")

    # === Phase 8: Final peak crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final peak crush (n={n_best}) ===")
        t8 = min(remaining * 0.35, 200.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_crush(a_in, t8, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining!")

    return [float(x) for x in best_seq]
