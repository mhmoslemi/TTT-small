def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_power_law(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_sinc_like(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_flat_top(n):
    """Flat-top window (Blackman-Harris derivative)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0
    a -= 0.426559 / n * np.ones(n)
    # Use standard flat-top formula
    w = 0.374846 - 0.336813 * np.cos(2 * np.pi * x) + 0.168919 * np.cos(4 * np.pi * x) \
        - 0.067499 * np.cos(6 * np.pi * x) + 0.012133 * np.cos(8 * np.pi * x)
    return norm_sum1(w)


def make_nuttall(n):
    """Nuttall window."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 0.3635819 - 0.4891775 * np.cos(2 * np.pi * x) + 0.1365995 * np.cos(4 * np.pi * x) \
        - 0.0106411 * np.cos(6 * np.pi * x)

            (make_flat_top(n_best), "flattop"),
            (make_nuttall(n_best), "nuttall"),
            (make_sinc_like(n_best, bandwidth=6.0), "sinc6"),
            (make_sinc_like(n_best, bandwidth=10.0), "sinc10"),
            (make_cosine(n_best, mode=2), "cos2"),
        ]
        t_per = min(60.0, remaining / len(inits))
        for init, lbl in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            ta = min(t_per, remaining - 10)
            if ta < 15:
                break
            print(f"  MD '{lbl}' (t={ta:.0f}s)...")
            a_out, _ = amd_optimize(init, ta, top_k=50, beta=5.0, c0=4.0,
                                     restart_every=3000, verbose=False, seed=hash(lbl) % 10000)
            try_update(a_out, f"md_{lbl}_n{n_best}")

    # === Phase 8: Final aggressive AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final AMD (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.5, 150.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t8, top_k=100, beta=3.0, c0=2.5,
                                     restart_every=8000, verbose=True, seed=5555)
            try_update(a_out, f"amd_final_n{n_best}")

    # === Phase 9: Final SA ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final SA (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = simulated_annealing(a_in, t9, seed=31415, verbose=False)
            try_update(a_out, f"sa_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
