def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K=300,

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

            break

        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive active set: include all constraints within margin_frac * M
        margins = M_curr - ac
        threshold = max(current_margin * M_curr, min_margin)
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        # Cap active set size
        if n_active > top_K:
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:max(top_K, 50)]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix: (F g)_k = sum_i a[k-i] * g[i]
        # F[k, i] = a[k - i] if 0 <= k-i < n else 0
        # For efficiency, build only the rows we need
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)

        if sy < 1e-15:

def pair_coordinate_descent(a0, time_limit, verbose=True, seed=777):
    """
    Coordinate descent that targets peak lags by redistributing mass
    from "linked pairs" (a_i, a_{i+k*}) that contribute most to the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top few peaks
        n_peaks = min(5, len(ac))
        peak_lags = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for peak_k in peak_lags:
            k_int = int(peak_k)
            if k_int == 0:
                continue  # Skip lag 0 (sum of squares, harder to reduce)

            # Find contributing pairs
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            total_c = contrib.sum()
            if total_c < 1e-20:
                continue

            # Try reducing the top contributors
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]

            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                partner = k_int - src
                if partner < 0 or partner >= n:
                    continue
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Try moving a fraction from src to a random target
                best_trial_m = M_curr
                best_move = None

                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    n_trials = 20
                    for _ in range(n_trials):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_trial_m - 1e-18:
                            best_trial_m = M_trial
                            best_move = (src, dst, amt)

                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-18:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:

        J = 2.0 * n * best_m

    return best_a, best_m

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m_trial:
                    best_m_trial = M_trial
                    best_frac = amt
            if best_frac is not None:
                a[src] = max(a[src] - best_frac, 0.0)
                a[dst] += best_frac
                a = a / a.sum()
                if fast_autocorr(a).max() < best_m:
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            continue

        # For non-zero lag, find the pair with max product
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]

        if contrib.sum() < 1e-20:
            stall += 1
            continue

        # Try to reduce the top contributor
        top_li = int(np.argmax(contrib))
        src = int(idx[top_li])
        src_val = a[src]
        if src_val < 1e-12:
            stall += 1
            continue

        # Find best destination by checking gradient effect
        best_dst = -1
        best_delta = 0.0
        for trial_dst in range(0, n, max(1, n // 200)):
            if trial_dst == src:
                continue
            # Gradient change at peak: 2*(a[peak_k-trial_dst] - a[peak_k-src])
            g_src = a[peak_k - src] if 0 <= peak_k - src < n else 0.0
            g_dst = a[peak_k - trial_dst] if 0 <= peak_k - trial_dst < n else 0.0
            delta = 2.0 * (g_dst - g_src)
            if delta < best_delta:
                best_delta = delta
                best_dst = trial_dst

        if best_dst < 0:
            stall += 1
            continue

        # Try different amounts
        best_amt = None
        best_m_trial = M_curr
        for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.1]:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m_trial:
                best_m_trial = M_trial
                best_amt = amt

        if best_amt is not None:
            a[src] = max(a[src] - best_amt, 0.0)
            a[best_dst] += best_amt
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m - 1e-18:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Squeeze ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Squeeze final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function to find the best sequence."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick AMD warm-start ===
    print(f"\n=== Phase 1: AMD warm-start (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 80.0)
    if t1 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = autoreg_accelerated(a_in, t1, c0=4.0, verbose=False, seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd1")

    # === Phase 2: LP-FW main optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 380.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, verbose=True, top_K=350,
                                      margin_frac=2e-3, min_margin=5e-6)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Squeeze descent (targeted peak reduction) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Squeeze (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.10, 160.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = squeeze_descent(a_in, t3, verbose=True, seed=222)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "squeeze1")

    # === Phase 4: LP-FW tight ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True, top_K=450,
                                      margin_frac=8e-4, min_margin=2e-6)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 5: Pair coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: PairCD (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_coordinate_descent(a_in, t5, verbose=True, seed=777)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "paircd")

    # === Phase 6: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Freq-diversify (n={n_best}) ===")
    rng = np.random.default_rng(99)
    t6_per = min(remaining * 0.07, 100.0)
    if remaining > t6_per * 2 + 60:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 40:
                break
            a_pert = freq_perturb(best_seq, strength=0.012, rng=rng)
            a_out, _ = autoreg_accelerated(a_pert, t6_per, c0=4.5, verbose=False, seed=100+trial)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 7: LP-FW final on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP-FW final (n={n_best}) ===")
    t7 = min(remaining * 0.15, 220.0)
    if t7 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t7, solver, verbose=True, top_K=500,
                                      margin_frac=4e-4, min_margin=1e-6)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final squeeze (n={len(best_seq)}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.25, 180.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = squeeze_descent(a_in, t8, verbose=True, seed=555)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "squeeze_final")

    # === Phase 9: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    alt_lengths = [n_best - 200, n_best + 200, n_best + 400, n_best - 400]
    alt_lengths = [l for l in alt_lengths if 2000 <= l <= 5500]
    alt_lengths = sorted(set(alt_lengths))

    for new_n in alt_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 9: Try n={new_n} (remaining {remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t9a = min(remaining * 0.25, 180.0)
        if t9a > 30:
            a_out, _ = autoreg_accelerated(a_long, t9a, c0=3.5, verbose=False, seed=new_n % 1000)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 4500:
            t9b = min(remaining * 0.25, 150.0)
            if t9b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe_v2(a_in, t9b, solver, verbose=False, top_K=300,
                                              margin_frac=2e-3, min_margin=5e-6)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety: ensure we return before budget
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
