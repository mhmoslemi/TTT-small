            for mult in [0.3, 0.7, 1.0, 2.0, 5.0]:
                eta = eta_base * mult
                log_r = np.log(y) - eta * g
                log_r -= log_r.max()
                cand = np.exp(log_r)
                cand = np.maximum(cand, 1e-30)
                cand /= cand.sum()
                val = eval_J(cand)
                if val < best_new_val:
                    best_new_val = val
                    a_new = cand
        else:
            eta = eta_base
            log_r = np.log(y) - eta * g
            log_r -= log_r.max()
            a_new = np.exp(log_r)
            a_new = np.maximum(a_new, 1e-30)
            a_new /= a_new.sum()
            best_new_val = eval_J(a_new)

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        if best_new_val < best_val - 1e-14:
            best_val = best_new_val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart logic with diverse strategies
        if stall > restart_every:
            stall = 0
            strategy = rng.integers(5)
            if strategy == 0:
                # Geometric perturbation
                scale = rng.uniform(0.01, 0.06)
                pert = rng.normal(0, scale, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Multiplicative perturbation
                scale = rng.uniform(0.02, 0.15)
                mult = np.exp(rng.normal(0, scale, n))
                x_r = best_a * mult
            elif strategy == 2:
                # Sparse perturbation
                n_pert = max(1, n // 8)
                idx = rng.choice(n, size=n_pert, replace=False)
                x_r = best_a.copy()
                x_r[idx] *= np.exp(rng.normal(0, 0.15, n_pert))
            elif strategy == 3:
                # Blend with random structure
                blend = make_smooth_random(n, seed=int(rng.integers(100000)))
                mix = rng.uniform(0.1, 0.5)
                x_r = (1 - mix) * best_a + mix * blend
            else:
                # Peak-targeted: increase mass at low-pressure positions
                conv_bd = fast_conv(best_a)
                pressure = np.zeros(n)
                top_lags = np.argsort(conv_bd)[::-1][:10]
                for k_l in top_lags:
                    k_int = int(k_l)
                    i_s = max(0, k_int - n + 1)
                    i_e = min(n - 1, k_int)
                    if i_s <= i_e:
                        idx = np.arange(i_s, i_e + 1)
                        pressure[idx] += best_a[k_int - idx]
                low_pressure = np.where(pressure < np.median(pressure))[0]
                if len(low_pressure) > 0:
                    x_r = best_a.copy()
                    chosen = rng.choice(low_pressure, size=min(50, len(low_pressure)), replace=False)
                    for ci in chosen:
                        donor = int(np.argmax(pressure * best_a))
                        amt = best_a[donor] * rng.uniform(0.01, 0.05)
                        x_r[donor] -= amt
                        x_r[ci] += amt
                else:
                    x_r = best_a + rng.normal(0, 0.01, n) * best_a

            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()

            # Adapt parameters
            cur_top_k = min(cur_top_k + 15, 250)
            cur_beta = max(cur_beta - 0.5, 2.0)
            cur_c0 = max(cur_c0 * 0.88, 0.5)
            if verbose:
                print(f"    MD restart t={t}: J={best_val:.6f}, k={cur_top_k}, b={cur_beta:.1f}, c0={cur_c0:.2f}")

        if verbose and t % 20000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, stall={stall}")

    return best_a, best_val


def smooth_objective_and_grad(z, a_fixed_sum=True):
    """
    Smooth objective using log-sum-exp approximation to max(conv).
    We parameterize a = softmax(z) to automatically satisfy a>=0, sum(a)=1.
    Returns (objective, gradient_wrt_z).
    """
    # Convert z to a via softmax
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)

    # Compute convolution
    conv = fast_conv(a)

    # Log-sum-exp with temperature
    T = 0.01
    conv_shifted = conv / T
    conv_shifted -= conv_shifted.max()
    lse = T * (conv_shifted.max() + np.log(np.sum(np.exp(conv_shifted))))
    obj = 2.0 * n * lse

    # Gradient: d(obj)/d(z) = d(obj)/d(a) * da/dz
    # d(obj)/d(a[i]) from the softmax weights on conv
    weights = np.exp(conv_shifted)
    weights /= weights.sum()

    g_a = np.zeros(n)
    for wi, k in zip(weights, range(len(conv))):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g_a[idx] += 2.0 * wi * a[j_idx]
    g_a = 2.0 * n * g_a

    # da/dz for softmax: a_i * (delta_ij - a_j)
    g_z = g_a * a - a * np.dot(g_a, a)

    return obj, g_z


def scipy_polish(a0, time_limit, n_inner_max=500):
    """Use scipy L-BFGS-B with smooth log-sum-exp objective for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s

    # Parameterize via log(a) + offset to stay positive
    # z = log(a) gives a = exp(z - logsumexp(z)) via softmax
    z0 = np.log(np.maximum(a, 1e-30))

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    n_iter = 0

    def fun(z):
        nonlocal n_iter
        obj, _ = smooth_objective_and_grad(z)
        n_iter += 1
        return obj

    def fun_and_grad(z):
        nonlocal n_iter
        obj, grad = smooth_objective_and_grad(z)
        n_iter += 1
        return obj, grad

    # Run L-BFGS-B with time check via callback
    state = {'stop': False}

    def callback(zk):
        if time.monotonic() - t0 > time_limit:
            state['stop'] = True
        return state['stop']

    try:
        result = scipy.optimize.minimize(
            fun_and_grad, z0, method='L-BFGS-B',
            options={'maxiter': n_inner_max, 'maxfun': n_inner_max * 2,
                     'callback': callback, 'gtol': 1e-8}
        )
        if result.x is not None:
            z_opt = result.x
            z_sh = z_opt - z_opt.max()
            a_opt = np.exp(z_sh)
            a_opt = np.maximum(a_opt, 0.0)
            a_opt /= a_opt.sum()
            val_opt = eval_J(a_opt)
            if val_opt < best_val:
                best_a = a_opt
                best_val = val_opt
    except Exception as e:
        pass

    return best_a, best_val


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=50.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.trim_zeros(np.exp(-width * (x - 0.85) ** 2)) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.8
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_chirp(n, cycles=1.0, amplitude=0.5, offset=1.0):
    """Linear chirp: a_i = max(0, offset + amplitude*cos(2*pi*cycles*i^2/n))."""
    i = np.arange(n, dtype=np.float64)
    phase = 2 * np.pi * cycles * i * i / n
    a = offset + amplitude * np.cos(phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine(n, freq=1.0, amplitude=0.4, offset=1.0):
    """Single cosine: a_i = max(0, offset + amplitude*cos(2*pi*freq*i/n))."""
    i = np.arange(n, dtype=np.float64)
    a = offset + amplitude * np.cos(2 * np.pi * freq * i / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_multi_cosine(n, freqs=None, amplitudes=None):
    """Multi-frequency cosine superposition."""
    if freqs is None:
        freqs = [1.0, 2.5, 4.0]
    if amplitudes is None:
        amplitudes = [0.3, 0.2, 0.1]
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, amp in zip(freqs, amplitudes):
        a += amp * np.cos(2 * np.pi * f * i / n + 0.5 * f)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric_ramp(n, p=1.0, shift=0.3):
    """Asymmetric ramp with peak shifted."""
    x = np.linspace(0, 1, n)
    a = (x + shift / max(0.1, 1 - 2 * shift)) ** p * ((1 - x) ** 0.3)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_sparse_burst(n, n_bursts=5, width_frac=0.1, seed=0):
    """Sparse bursts at random positions."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) * 0.05
    for _ in range(n_bursts):
        center = int(rng.uniform(0, n))
        width = max(1, int(n * width_frac * rng.uniform(0.5, 2.0)))
        for i in range(max(0, center - width), min(n, center + width + 1)):
            dist = abs(i - center) / max(1, width)
            a[i] += np.exp(-2 * dist * dist)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_square_wave(n, periods=3, duty=0.5):
    """Square wave approximation."""
    x = np.linspace(0, periods, n, endpoint=False)
    frac = x % 1.0
    a = np.where(frac < duty, 1.0, 0.1)
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def generate_diverse_candidates(n, parent=None):
    """Generate a diverse set of candidate initializations."""
    candidates = []
    labels = []

    if parent is not None:
        candidates.append(parent.copy())
        labels.append("parent")

    # Power ramps
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(make_power_ramp(n, p))
        labels.append(f"ramp_{p}")

    # Bimodal
    for w in [30, 50, 80]:
        candidates.append(make_bimodal(n, width=w))
        labels.append(f"bimodal_{w}")

    # Smooth random
    for seed in [1, 7, 42, 99, 123, 777]:
        candidates.append(make_smooth_random(n, seed=seed))
        labels.append(f"smooth_{seed}")

    # Chirps
    for cycles in [0.5, 1.0, 2.0, 3.0]:
        for amp in [0.3, 0.5, 0.7]:
            candidates.append(make_chirp(n, cycles=cycles, amplitude=amp))
            labels.append(f"chirp_c{cycles}_a{amp}")

    # Cosines
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.2, 0.4, 0.6]:
            candidates.append(make_cosine(n, freq=freq, amplitude=amp))
            labels.append(f"cos_f{freq}_a{amp}")

    # Multi-cosine
    candidates.append(make_multi_cosine(n, freqs=[1, 3, 7], amplitudes=[0.3, 0.2, 0.15]))
    labels.append("multicos_1")
    candidates.append(make_multi_cosine(n, freqs=[2, 5, 11], amplitudes=[0.25, 0.2, 0.1]))
    labels.append("multicos_2")
    candidates.append(make_multi_cosine(n, freqs=[1, 2, 4, 8], amplitudes=[0.2, 0.15, 0.1, 0.05]))
    labels.append("multicos_3")

    # Asymmetric ramps
    for shift in [0.2, 0.4, 0.6]:
        candidates.append(make_asymmetric_ramp(n, p=1.0, shift=shift))
        labels.append(f"asym_{shift}")

    # Sparse bursts
    for nb in [3, 5, 8]:
        candidates.append(make_sparse_burst(n, n_bursts=nb, seed=nb))
        labels.append(f"burst_{nb}")

    # Square wave
    for per in [2, 3, 5]:
        candidates.append(make_square_wave(n, periods=per))
        labels.append(f"sq_{per}")

    # Blends with parent
    if parent is not None:
        for frac in [0.3, 0.5, 0.7]:
            blend = frac * parent + (1 - frac) * make_power_ramp(n, 1.0)
            candidates.append(blend / blend.sum())
            labels.append(f"blend_p{frac}")
        for frac in [0.5, 0.7]:
            blend = frac * parent + (1 - frac) * make_chirp(n, cycles=1.0, amplitude=0.5)
            candidates.append(blend / blend.sum())
            labels.append(f"blend_chirp{frac}")

    return candidates, labels


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with pressure-based mass transfer (improved)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Compute pressure from top lags
        n_lags = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[k_i - idx]

        # Find source: high pressure * high mass
        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-14:
            stall += 1
            continue

        # Find destination: low pressure, low impact on peaks
        dest_scores = np.zeros(n)
        for k in top_lags[:8]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_scores[idx] += a[k_i - idx]

        dest_scores[src] = np.inf
        dest_scores += 0.2 * pressure
        dst = int(np.argmin(dest_scores))

        # Try multiple transfer fractions
        improved = False
        for frac in [0.003, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.005, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

    return best_a, best_J


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    # Load parent if available
    parent = None
    parent_n = 3200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            parent_n = len(parent)
            pJ = eval_J(parent)
            print(f"Parent loaded: n={parent_n}, J={pJ:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_power_ramp(parent_n, 1.0)
        print(f"No parent found. Using power ramp init: n={parent_n}, J={eval_J(parent):.6f}")

    best_a = parent.copy()
    best_score = eval_J(parent)
    print(f"Initial best: J={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score - 1e-14:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = parent_n

    # === Phase 1: Quick diverse initialization (40s) ===
    print(f"\n=== Phase 1: Diverse Init (n={n0}) ===")
    t1_end = t_global + 40.0
    candidates, labels = generate_diverse_candidates(n0, parent)

    # Score all candidates quickly
    scored = []
    for c, lbl in zip(candidates, labels):
        if time.monotonic() > t1_end:
            break
        J_c = eval_J(c)
        scored.append((J_c, lbl, c))

    scored.sort(key=lambda x: x[0])
    print(f"  Evaluated {len(scored)} candidates. Top 5:")
    for J_c, lbl, _ in scored[:5]:
        print(f"    J={J_c:.6f} [{lbl}]")

    # Update best from scoring
    for J_c, lbl, c in scored[:3]:
        try_update(c, f"init_{lbl}")

    # === Phase 2: Deep MD on top candidates (1000s) ===
    print(f"\n=== Phase 2: Deep MD ===")
    t2_start = time.monotonic()
    t2_end = t2_start + min(time_budget * 0.50, 900.0)

    # Pick top diverse candidates (not all similar)
    top_cands = []
    top_labels = []
    seen_sigs = set()
    for J_c, lbl, c in scored:
        if time.monotonic() > t2_end:
            break
        sig = tuple(np.round(c[::max(1, n0 // 50)], 2))
        if sig not in seen_sigs:
            seen_sigs.add(sig)
            top_cands.append(c)
            top_labels.append(lbl)
        if len(top_cands) >= 6:
            break

    # Also ensure parent is included
    if parent not in top_cands:
        top_cands.insert(0, parent.copy())
        top_labels.insert(0, "parent")

    for idx, (cand, lbl) in enumerate(zip(top_cands, top_labels)):
        remaining = t2_end - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(remaining * 0.6, 250.0)
        if t_alloc < 15:
            break
        print(f"  MD [{lbl}] alloc={t_alloc:.0f}s...")
        a_out, _ = mirror_descent(cand, t_alloc, top_k=60, beta_w=10.0,
                                  c0=4.0, restart_every=3000, seed=100 + idx)
        try_update(a_out, f"md_{lbl}")

    # === Phase 3: Extended MD on current best (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Extended MD on best (J={best_score:.6f}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30:
        a_out, _ = mirror_descent(best_a.copy(), t3, top_k=100, beta_w=8.0,
                                  c0=3.0, restart_every=5000, seed=777)
        try_update(a_out, "md_ext")

    # === Phase 4: Coordinate descent (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_out, _ = coordinate_descent(best_a.copy(), t4, verbose=True, seed=314)
        try_update(a_out, "coord1")

    # === Phase 5: MD after CD (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MD after CD ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 15:
        a_out, _ = mirror_descent(best_a.copy(), t5, top_k=120, beta_w=6.0,
                                  c0=2.5, restart_every=4000, seed=555)
