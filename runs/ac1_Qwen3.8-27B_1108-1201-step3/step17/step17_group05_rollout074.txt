```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

The algorithm uses:
  - FFT-based autocorrelation and objective evaluations.
  - Polynomial smoothed-max subgradients (annealed q), with adaptive
    active-lag sharpness and line searches using multiplicative and
    additive simplex-preserving steps.
  - LP-based directional jumps: for the current a, solve min tau s.t.
    conv(a,b) over selected lags <= tau, b >= 0, sum(b)=1, and mix with b.
  - Coordinate-polish moves from high-risk to low-risk positions.
  - Length interpolation / longer-sequence trials to avoid local minima.

All helper functions are top-level. The required entry point is
propose_candidate().
"""

import time
import numpy as np

try:
    import cvxpy as cp
except Exception:
    cp = None

_LP_SOLVER = None


def fft_autocorr(a):
    """Full linear autocorrelation conv(a, a) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, m)
    return np.fft.irfft(A * A, m)[:size]


def normalize_nonneg(a):
    """Clip to nonnegative, replace nonfinite by zero, normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def internal_score(a):
    """Fast score assuming a is already normalized to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    ac = fft_autocorr(a)
    m = float(np.max(ac)) if ac.size else 0.0
    if not np.isfinite(m):
        return float("inf")
    if m < 0.0 and m > -1e-12:
        m = 0.0
    return 2.0 * float(n) * m


def validate_score(a):
    """Validate using the provided evaluate_sequence if available."""
    try:
        seq = [float(x) for x in a]
        val = float(evaluate_sequence(seq))
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float("inf")


def make_smooth_init(n, seed=0):
    """Smooth positive initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n, dtype=np.float64) * 0.65
    for f in range(1, 9):
        amp = rng.uniform(0.03, 0.12)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.sin(2.0 * np.pi * f * x + phase)
    a = np.maximum(a, 0.05)
    return normalize_nonneg(a)


def random_dirichlet_init(n, seed_or_rng):
    """Random Dirichlet initialization."""
    if isinstance(seed_or_rng, (int, np.integer)):
        rng = np.random.default_rng(int(seed_or_rng))
    else:
        rng = seed_or_rng
    alpha = float(rng.choice([0.2, 0.5, 1.0, 2.0, 4.0]))
    return normalize_nonneg(rng.dirichlet(np.ones(n, dtype=np.float64) * alpha))


def perturb_vector(a, rng, scale=0.02):
    """Multiplicative Gaussian perturbation, renormalized."""
    noise = rng.normal(0.0, scale, size=a.shape) * (0.5 + a / max(np.max(a), 1e-30))
    x = a + noise
    x = np.maximum(x, 1e-14)
    return normalize_nonneg(x)


def resample_sequence(a, new_n):
    """Linearly interpolate a normalized sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    old_n = a.size
    new_n = int(new_n)
    if new_n <= 0 or old_n == new_n:
        return normalize_nonneg(a)
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    return normalize_nonneg(b)


def topk_weights(ac, k=10):
    """Uniform weights on the top-k lags."""
    ac = np.asarray(ac, dtype=np.float64)
    if ac.size == 0:
        return np.array([], dtype=np.float64)
    k = int(k)
    if k <= 0:
        k = 1
    if k >= ac.size:
        return np.ones(ac.size, dtype=np.float64) / float(ac.size)
    idx = np.argsort(ac)[::-1][:k]
    w = np.zeros(ac.size, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def poly_weights(ac, q):
    """
    Weights proportional to ac^q, normalized.
    q=1 gives ac itself; large q approximates max-subgradient.
    """
    ac = np.asarray(ac, dtype=np.float64)
    if ac.size == 0:
        return np.array([], dtype=np.float64), 0.0
    M = float(np.max(ac))
    if not np.isfinite(M) or M <= 1e-300:
        w = np.ones(ac.size, dtype=np.float64) / float(ac.size)
        return w, max(M, 0.0)
    q = float(q)
    if q < 0.1:
        q = 0.1
    if q > 1000.0:
        q = 1000.0

    ratio = np.maximum(ac, 0.0) / M
    pos = ratio > 1e-300
    logv = np.zeros(ac.size, dtype=np.float64)
    if np.any(pos):
        logv[pos] = np.log(ratio[pos])
    logv[~pos] = -50.0
    ex = np.clip(q * logv, -745.0, 0.0)

    w = np.zeros(ac.size, dtype=np.float64)
    if np.any(pos):
        w[pos] = np.exp(ex[pos])
    s = float(w.sum())
    if s <= 1e-300:
        w = np.zeros(ac.size, dtype=np.float64)
        w[int(np.argmax(ac))] = 1.0
        return w, M

    w /= s
    w[w < 1e-25] = 0.0
    s = float(w.sum())
    if s > 1e-300:
        w /= s
    else:
        w = topk_weights(ac, k=8)
    return w, M


def grad_from_weights(a, w):
    """
    Gradient direction:
    g[j] = 2 * sum_k w[k] * a[k-j] (valid indices only).
    Uses direct summation for sparse weights and FFT for dense weights.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if w.size == 0:
        return np.zeros(n, dtype=np.float64)

    nz = np.flatnonzero(w > 1e-25)
    if nz.size == 0:
        return np.zeros(n, dtype=np.float64)

    if nz.size <= 900:
        g = np.zeros(n, dtype=np.float64)
        for kll in nz:
            k = int(kll)
            wk = float(w[k])
            if wk == 0.0:
                continue
            js = k - n + 1
            if js < 0:
                js = 0
            je = k
            if je > n - 1:
                je = n - 1
            if js > je:
                continue
            start = k - js
            stop = k - je - 1
            if stop < 0:
                seg = a[start::-1]
            else:
                seg = a[start:stop:-1]
            g[js:je + 1] += wk * seg
        return 2.0 * g

    L = w.size
    total = n + L - 1
    m = 1 << (total - 1).bit_length()
    arev = np.zeros(m, dtype=np.float64)
    arev[:n] = a[::-1]
    wpad = np.zeros(m, dtype=np.float64)
    wpad[:L] = w
    conv = np.fft.irfft(np.fft.rfft(arev) * np.fft.rfft(wpad), m)[:total]
    return 2.0 * conv[n - 1:2 * n - 1]


def mirror_step(a, g_unit, eta):
    """Multiplicative simplex-preserving step: a <- normalize(a*exp(-eta*g))."""
    if eta <= 0.0:
        return None
    loga = np.log(np.maximum(a, 1e-300))
    out = np.exp(loga - float(eta) * g_unit)
    out = np.where(np.isfinite(out), out, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return None
    return out / s


def additive_step(a, g_unit, alpha):
    """Additive simplex-preserving step: a <- normalize(clip(a - alpha*g, 0, inf))."""
    if alpha <= 0.0:
        return None
    out = a - float(alpha) * g_unit
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return None
    return out / s


def line_search_gradient(a, g, current_score):
    """
    Try several multiplicative and additive steps and keep the best.
    g is projected direction with nonnegative values.
    """
    n = a.size
    gm = float(np.max(g)) if g.size else 0.0
    if gm <= 1e-300:
        return a, current_score

    g_unit = g / gm
    best_a = a
    best_s = float(current_score)
    amax = float(np.max(a)) if a.size else 0.0

    if n <= 4200:
        mirror_steps = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3)
        additive_gammas = (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2)
    else:
        mirror_steps = (3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1)
        additive_gammas = (3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2)

    for eta in mirror_steps:
        c = mirror_step(a, g_unit, eta)
        if c is None:
            continue
        sc = internal_score(c)
        if sc < best_s - 1e-14:
            best_s = sc
            best_a = c

    if amax > 1e-300:
        for gamma in additive_gammas:
            alpha = float(gamma) * amax
            c = additive_step(a, g_unit, alpha)
            if c is None:
                continue
            sc = internal_score(c)
            if sc < best_s - 1e-14:
                best_s = sc
                best_a = c

    return best_a, best_s


def poly_optimize(a0, time_limit, seed=0, q_init=2.0, q_max=100.0,
                  verbose=False, stop_stall=3000):
    """
    Annealed polynomial smoothed-max subgradient optimization.
    """
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a, float("inf")

    if time_limit <= 0:
        ac = fft_autocorr(a)
        m = float(np.max(ac)) if ac.size else 0.0
        return a, 2.0 * n * max(m, 0.0)

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    ac = fft_autocorr(a)
    M = float(np.max(ac)) if ac.size else 0.0
    current_score = 2.0 * n * max(M, 0.0)
    best_a = a.copy()
    best_score = current_score

    q = float(q_init)
    progress_stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        ac = fft_autocorr(a)
        M = float(np.max(ac)) if ac.size else 0.0
        old_score = 2.0 * n * max(M, 0.0)

        if old_score < best_score - 1e-13:
            best_score = old_score
            best_a = a.copy()

        # Schedule q to increase with elapsed time, but keep it controllable.
        if time_limit > 0:
            rem_frac = max(0.0, 1.0 - elapsed / max(time_limit, 1e-12))
            q_target = max(float(q_init), float(q_max) * (1.0 - rem_frac))
        else:
            q_target = float(q_max)

        w, _ = poly_weights(ac, q)
        s2 = float(np.dot(w, w))
        eff_k = (1.0 / s2) if s2 > 1e-300 else float(ac.size)

        # Adaptive active-lag sharpness.
        if eff_k < 5.0:
            if q > 0.5:
                q *= 0.8
            else:
                w = topk_weights(ac, max(5, min(80, int(np.sqrt(float(ac.size))))))
        elif eff_k > max(600, n // 4):
            if q < float(q_max):
                q *= 1.15

        q = 0.85 * q + 0.15 * q_target
        if q < 0.5:
            q = 0.5
        if q > float(q_max):
            q = float(q_max)

        w, _ = poly_weights(ac, q)
        g = grad_from_weights(a, w)
        gm = float(np.max(g)) if g.size else 0.0

        # If direction is almost flat, try a smoother weight once.
        if gm > 1e-300 and (gm - float(np.min(g))) < 1e-12 * gm:
            w_alt, _ = poly_weights(ac, max(0.5, q * 0.5))
            g_alt = grad_from_weights(a, w_alt)
            if float(np.max(g_alt)) > gm:
                g = g_alt

        new_a, new_score = line_search_gradient(a, g, old_score)

        if new_score < old_score - 1e-14:
            a = new_a
            old_score = new_score
            progress_stall = 0
        else:
            progress_stall += 1

        if old_score < best_score - 1e-13:
            best_score = old_score
            best_a = a.copy()
            if verbose and (it % 500 == 0 or best_score < 1.52):
                print(f"    poly-opt it={it} t={elapsed:.0f}s J={best_score:.6f} q={q:.1f}")

        # Escape from stagnation.
        if progress_stall > 800:
            if rng.random() < 0.55:
                a = perturb_vector(best_a, rng, rng.uniform(0.01, 0.05))
            else:
                a = random_dirichlet_init(n, rng)
            q = max(0.7, min(float(q_max) * 0.35, 4.0))
            progress_stall = 0

        if progress_stall > stop_stall:
            break

    return best_a, best_score


def get_lp_solver():
    """Choose a cvxpy LP solver if possible."""
    global _LP_SOLVER
    if cp is None:
        return None
    if _LP_SOLVER is not None:
        return _LP_SOLVER

    names = []
    for name in ["CLARABEL", "ECOS", "PDLP", "SCS"]:
        attr = getattr(cp, name, None)
        if attr is not None:
            names.append(name)

    for name in names:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _LP_SOLVER = name
                print(f"    LP solver: {name}")
                return name
        except Exception:
            continue

    _LP_SOLVER = None
    print("    LP solver: none")
    return None


def lp_direction_optimize(a0, time_limit, solver, verbose=False):
    """
    LP-based directional jumps:
    minimize tau subject to conv(a,b)[k] <= tau for selected active lags,
    b >= 0, sum(b)=1, then mix a with b.
    """
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0 or solver is None or time_limit <= 0:
        ac = fft_autocorr(a)
        M = float(np.max(ac)) if ac.size else 0.0
        return a, 2.0 * n * max(M, 0.0)

    t0 = time.monotonic()
    ac = fft_autocorr(a)
    M = float(np.max(ac)) if ac.size else 0.0
    best_a = a.copy()
    best_score = 2.0 * n * max(M, 0.0)
    current_score = best_score

    fails = 0
    solves = 0
    max_solves = 25

    k_schedule = [60, 120, 240, 480]
    k_schedule = [k for k in k_schedule if k <= max(20, 2 * n - 1)]
    if len(k_schedule) == 0:
        k_schedule = [max(20, min(120, n))]

    cycle = 0
    while time.monotonic() - t0 < time_limit and fails < 12 and solves < max_solves:
        ac = fft_autocorr(a)
        M = float(np.max(ac)) if ac.size else 0.0
        current_score = 2.0 * n * max(M, 0.0)
        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()

        k = int(k_schedule[cycle % len(k_schedule)])
        cycle += 1
        solves += 1

        top_idx = np.argsort(ac)[::-1][:k]
        T = np.zeros((len(top_idx), n), dtype=np.float64)
        for row, kll in enumerate(top_idx):
            kk = int(kll)
            js = kk - n + 1
            if js < 0:
                js = 0
            je = kk
            if je > n - 1:
                je = n - 1
            if js > je:
                continue
            start = kk - js
            stop = kk - je - 1
            if stop < 0:
                seg = a[start::-1]
            else:
                seg = a[start:stop:-1]
            T[row, js:je + 1] = seg

        bvar = cp.Variable(n, nonneg=True)
        tau = cp.Variable()
        cons = [
            cp.sum(bvar) == 1,
            T @ bvar <= tau,
            tau >= 0
        ]
        prob = cp.Problem(cp.Minimize(tau), cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fails += 1
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            fails += 1
            continue

        bval = np.asarray(bvar.value, dtype=np.float64).flatten()
        bval = np.where(np.isfinite(bval), bval, 0.0)
        bval = np.maximum(bval, 0.0)
        bs = float(bval.sum())
        if bs <= 1e-14:
            fails += 1
            continue
        bval = bval / bs

        base_score = current_score
        best_cand = a
        best_cand_score = base_score
        alphas = (0.002, 0.005, 0.01, 0.03, 0.1, 0.25, 0.5, 1.0)
        for alp in alphas:
            c = (1.0 - float(alp)) * a + float(alp) * bval
            c = normalize_nonneg(c)
            sc = internal_score(c)
            if sc < best_cand_score - 1e-14:
                best_cand_score = sc
                best_cand = c

        if best_cand_score < base_score - 1e-14:
            a = best_cand
            current_score = best_cand_score
            fails = 0
            if current_score < best_score - 1e-13:
                best_score = current_score
                best_a = a.copy()
                if verbose:
                    print(f"    LP-JUMP solve={solves} K={k} t={time.monotonic()-t0:.0f}s J={best_score:.6f}")
        else:
            fails += 1

    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=0, verbose=False):
    """
    Single-move coordinate polish from high gradient-risk positions to low
    gradient-risk positions.
    """
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0 or time_limit <= 0:
        ac = fft_autocorr(a)
        M = float(np.max(ac)) if ac.size else 0.0
        return a, 2.0 * n * max(M, 0.0)

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    ac = fft_autocorr(a)
    M = float(np.max(ac)) if ac.size else 0.0
    current_score = 2.0 * n * max(M, 0.0)
    best_a = a.copy()
    best_score = current_score

    stall = 0
    scan = 0

    count_tops = 8 if n > 5000 else 12
    gammas = (0.02, 0.05, 0.10, 0.20)

    while time.monotonic() - t0 < time_limit:
        scan += 1
        ac = fft_autocorr(a)
        M = float(np.max(ac)) if ac.size else 0.0
        current_score = 2.0 * n * max(M, 0.0)
        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()

        w, _ = poly_weights(ac, q=18.0)
        g = grad_from_weights(a, w)
        order = np.argsort(g)
        dsts = order[:count_tops]
        srcs = order[::-1][:count_tops]

        found = False
        best_move = None
        best_move_score = current_score

        for src in srcs:
            src = int(src)
            if a[src] <= 1e-13:
                continue
            for dst in dsts:
                dst = int(dst)
                if dst == src:
                    continue
                for gamma in gammas:
                    amt = float(a[src]) * float(gamma)
                    if amt <= 0.0:
                        continue
                    c = a.copy()
                    c[src] -= amt
                    c[dst] += amt
                    c = np.maximum(c, 0.0)
                    s = float(c.sum())
                    if s <= 1e-15:
                        continue
                    c /= s
                    sc = internal_score(c)
                    if sc < best_move_score - 1e-14:
                        best_move_score = sc
                        best_move = (src, dst, float(gamma))
                        found = True

        if found:
            src, dst, gamma = best_move
            amt = float(a[src]) * gamma
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0.0)
            a = a / max(float(a.sum()), 1e-15)
            current_score = best_move_score
            stall = 0
            if current_score < best_score - 1e-13:
                best_score = current_score
                best_a = a.copy()
                if verbose and (scan % 50 == 0 or best_score < 1.51):
                    print(f"    coord-polish scan={scan} t={time.monotonic()-t0:.0f}s J={best_score:.6f}")
        else:
            # Try some random moves.
            improved = False
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] <= 1e-13:
                    continue
                gamma = float(rng.uniform(0.005, 0.18))
                amt = float(a[i]) * gamma
                c = a.copy()
                c[i] -= amt
                c[j] += amt
                c = np.maximum(c, 0.0)
                s = float(c.sum())
                if s <= 1e-15:
                    continue
                c /= s
                sc = internal_score(c)
                if sc < current_score - 1e-14:
                    a = c
                    current_score = sc
                    improved = True
                    break

            if improved:
                stall = 0
                if current_score < best_score - 1e-13:
                    best_score = current_score
                    best_a = a.copy()
            else:
                stall += 1
                if stall > 20:
                    break

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """
    Entry point. Searches for a nonnegative sequence minimizing the objective.
    Returns a list of floats.
    """
    t_global = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 45.0, 1755.0))
    solver = get_lp_solver()

    best_seq = None
    best_score = float("inf")

    # Try parent if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-10:
                best_seq = normalize_nonneg(p)
                best_score = internal_score(best_seq)
                print(f"Parent: n={best_seq.size}, internal_score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 3200
        # Choose a few initializations and keep best.
        for seed in range(8):
            a = make_smooth_init(n0, seed=seed * 17 + 3)
            sc = internal_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a
        print(f"Default init: n={best_seq.size}, internal_score={best_score:.6f}")

    def _elapsed():
        return time.monotonic() - t_global

    # Main polynomial subgradient run at current length.
    rem = total_budget - _elapsed()
    t1 = min(rem * 0.42, 620.0)
    if t1 > 30:
        a_out, sc_out = poly_optimize(best_seq, t1, seed=1,
                                      q_init=1.5, q_max=90.0,
                                      verbose=True, stop_stall=3500)
        if sc_out < best_score:
            best_seq = a_out
            best_score = sc_out
            print(f"*** Improved to {best_score:.6f} (n={best_seq.size}) after poly main.")

    # Short second run from a perturbed point.
    rem = total_budget - _elapsed()
    t1b = min(rem * 0.12, 150.0)
    if t1b > 20:
        rng = np.random.default_rng(123)
        a_in = perturb_vector(best_seq, rng, 0.02)
        a_out, sc_out = poly_optimize(a_in, t1b, seed=42,
                                      q_init=1.0, q_max=70.0,
                                      verbose=False, stop_stall=2500)
        if sc_out < best_score:
            best_seq = a_out
            best_score = sc_out
            print(f"*** Improved to {best_score:.6f} (n={best_seq.size}) after poly perturbed.")

    # LP directional jumps.
    rem = total_budget - _elapsed()
    t_lp = min(rem * 0.18, 230.0)
    if t_lp > 30 and solver is not None:
        a_out, sc_out = lp_direction_optimize(best_seq, t_lp, solver, verbose=True)
        if sc_out < best_score:
            best_seq = a_out
            best_score = sc_out
            print(f"*** Improved to {best_score:.6f} (n={best_seq.size}) after LP jumps.")

        # Short polish after LP.
        rem = total_budget - _elapsed()
        t_pol = min(rem * 0.08, 70.0)
        if t_pol > 15:
            a_out, sc_out = poly_optimize(best_seq, t_pol, seed=99,
                                          q_init=5.0, q_max=120.0,
                                          verbose=False, stop_stall=2000)
            if sc_out < best_score:
                best_seq = a_out
                best_score = sc_out
                print(f"*** Improved to {best_score:.6f} (n={best_seq.size}) after LP polish.")

    # Coordinate polish at current best length.
    rem = total_budget - _elapsed()
    t_coord = min(rem * 0.15, 200.0)
    if t_coord > 20:
        a_out, sc_out = coordinate_polish(best_seq, t_coord, seed=5, verbose=True)
        if sc_out < best_score:
            best_seq = a_out
            best_score = sc_out
            print(f"*** Improved to {best_score:.6f} (n={best_seq.size}) after coordinate polish.")

    # Try nearby/longer lengths.
    base_n = int(best_seq.size)
    cand_lengths = []
    for delta in (256, 512, 1024, 2048):
        ln = base_n + delta
        if 800 <= ln <= 8000:
            cand_lengths.append(ln)
    if base_n > 1500:
        cand_lengths.append(base_n - 512)

    seen = set()
    ordered_lengths = []
    for ln in sorted(cand_lengths):
        if ln not in seen:
            seen.add(ln)
            ordered_lengths.append(ln)

    for new_n in ordered_lengths:
        rem = total_budget - _elapsed()
        if rem < 120:
            break
        a_res = resample_sequence(best_seq, new_n)

        t_md = min(rem * 0.35, 160.0)
        if t_md > 25:
            a_out, sc_out = poly_optimize(a_res, t_md, seed=1000 + (new_n % 1000),
                                          q_init=1.5, q_max=80.0,
                                          verbose=False, stop_stall=2500)
            if sc_out < best_score:
                best_seq = a_out
                best_score = sc_out
                print(f"*** Improved to {best_score:.6f} at n={new_n} after resample poly.")

                # If longer length helped, do a short coordinate polish there.
                rem = total_budget - _elapsed()
                if rem > 50:
                    tc = min(rem * 0.25, 80.0)
                    if tc > 20:
                        a_out2, sc2 = coordinate_polish(best_seq, tc, seed=2000 + (new_n % 500),
                                                        verbose=False)
                        if sc2 < best_score:
                            best_seq = a_out2
                            best_score = sc2
                            print(f"*** Improved to {best_score:.6f} at n={new_n} after resample coord.")
        else:
            # At least validate the resample.
            sc_res = internal_score(a_res)
            if sc_res < best_score:
                best_seq = a_res
                best_score = sc_res
                print(f"Resample improved to {best_score:.6f} at n={new_n}.")

    # Final polish at the best length.
    rem = total_budget - _elapsed()
    if rem > 70:
        t_final = min(rem * 0.5, 240.0)
        a_out, sc_out = poly_optimize(best_seq, t_final, seed=777,
                                      q_init=5.0, q_max=160.0,
                                      verbose=True, stop_stall=5000)
        if sc_out < best_score:
            best_seq = a_out
            best_score = sc_out
            print(f"*** Improved to {best_score:.6f} after final poly.")

    rem = total_budget - _elapsed()
    if rem > 40:
        t_cf = min(rem * 0.6, 120.0)
        a_out, sc_out = coordinate_polish(best_seq, t_cf, seed=888, verbose=True)
        if sc_out < best_score:
            best_seq = a_out
            best_score = sc_out
            print(f"*** Improved to {best_score:.6f} after final coordinate.")

    # Final validation with the true evaluator if available.
    final_val = validate_score(best_seq)
    if np.isfinite(final_val) and final_val < best_score:
        best_score = final_val

    if not np.isfinite(best_score) or best_seq is None:
        best_seq = make_smooth_init(2048, seed=1)
        best_score = validate_score(best_seq)

    print(f"FINAL: score={best_score:.6f}, n={best_seq.size}, time={time.monotonic()-t_global:.0f}s")
    return [float(x) for x in best_seq]
```
