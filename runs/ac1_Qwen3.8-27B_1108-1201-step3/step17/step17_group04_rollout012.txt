def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def peak_subgradient(a, k):
    """Exact subgradient of conv[a,a][k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad


def peak_cycling_descent(a0, time_limit, verbose=False, seed=42,
                          top_k_soft=8, soft_beta=4.0):
    """
    Peak-cycling mirror descent: always target the current peak with exact subgradient,
    occasionally use soft top-K for smoothing. Adaptive step via line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    eta_base = 0.5

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        # Peak cycling: target the argmax peak with exact subgradient
        k_peak = int(np.argmax(ac))
        g = peak_subgradient(y, k_peak)

        # Occasionally mix in soft top-K for broader smoothing
        if t % 5 == 0 and top_k_soft > 0:
            top_idx = np.argsort(ac)[::-1][:top_k_soft]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** soft_beta
            w_sum = w.sum()
            if w_sum > 1e-30:
                w = w / w_sum
            g_soft = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                ki_int = int(ki)
                g_soft += wi * peak_subgradient(y, ki_int)
            g = 0.5 * g + 0.5 * g_soft

        # Adaptive step size
        gap = M - (1.0 / (2 * n - 1))
        if gap < 0:
            gap = 0
        norm_g = np.dot(g, g)
        eta = eta_base * gap / (norm_g + 1e-20)
        eta = min(eta, 1.0 / (np.sqrt(t + 1.0)))
        eta = max(eta, 1e-10)

        # Mirror descent step (exponential weights)
        log_a = np.log(y + 1e-30) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > 3000:
            stall = 0
            sigma = rng.uniform(0.005, 0.02)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta_base = rng.uniform(0.3, 1.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    PC restart ({elapsed:.0f}s): J={best_J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PC t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def pairwise_moves(a0, time_limit, verbose=False, seed=99):
    """
    Targeted pairwise moves: identify peak contributors and find optimal
    transfer destinations using sensitivity analysis.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300

    # Precompute for each lag k, which positions contribute
    # conv[k] = sum_i a[i]*a[k-i], positions i in [max(0,k-n+1), min(n-1,k)]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        k_peak = int(np.argmax(ac))

        # Get contributors to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        contributors = np.arange(i_start, i_end)
        partners = k_peak - contributors

        # Sensitivity: for each contributor i, reducing a[i] reduces
        # conv[k_peak] by 2*a[i]*a[k_peak-i] but affects other lags
        sensitivity = np.zeros(n)
        for i in contributors:
            sensitivity[i] = a[i] * a[partners[list(contributors).index(i)]]
            sensitivity[partners[list(contributors).index(i)]] += sensitivity[i]

        # Find high-sensitivity source and low-sensitivity destination
        high_sens_threshold = np.percentile(sensitivity[sensitivity > 0], 70) if (sensitivity > 0).any() else 0
        low_sens_threshold = np.percentile(sensitivity[sensitivity < high_sens_threshold], 50) if (sensitivity < high_sens_threshold).any() else 0

        high_pos = np.where((sensitivity >= high_sens_threshold) & (a > 1e-8))[0]
        low_pos = np.where((sensitivity <= low_sens_threshold) | (sensitivity == 0))[0]
        low_pos = low_pos[np.isin(low_pos, contributors) == False]  # Don't move within peak contributors

        if len(high_pos) == 0 or len(low_pos) == 0:
            # Fallback: random moves
            high_pos = np.where(a > np.median(a))[0]
            low_pos = np.where(a < np.median(a))[0]

        improved = False

        for attempt in range(50):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_pos[rng.integers(len(high_pos))]
            dst = low_pos[rng.integers(len(low_pos))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Binary search for optimal transfer amount
            lo, hi = 0.0, src_val * 0.8
            best_amt = 0
            best_trial_J = best_J

            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_amt = mid
                    lo = mid
                else:
                    hi = mid

            if best_amt > 0 and best_trial_J < best_J - 1e-14:
                a[src] -= best_amt
                a[dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall > 50:
                # Larger perturbation
                n_pert = max(10, n // 50)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    if a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.01, 0.08)
                        a[i] -= amt
                        j = int(rng.integers(n))
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pairwise ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP Frank-Wolfe with Nesterov acceleration for the min-max formulation."""
    n = len(a0)
    a = norm_sum1(a0)

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K binding constraints
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix for selected lags
        # conv[k] = sum_i a[i]*a[k-i] <= M for i in valid range
        # Linearize: for fixed y, conv_y[k] = sum_i y[i]*y[k-i]
        # Direction: we want x such that sum_i x[i]*y[k-i] + y[i]*x[k-i] <= M
        # This gives: sum_i (y[k-i] + y[i-x... ]) -- let's use the standard FW approach
        # Minimize <g, x> where g is subgradient at the peak

        # Use the peak subgradient as the linear objective
        k_peak = int(np.argmax(ac))
        g = peak_subgradient(y, k_peak)

        # LP: minimize g^T x subject to x >= 0, sum(x) = 1
        # This is solved by putting all mass on argmin(g)
        min_idx = int(np.argmin(g))
        g_dir = np.zeros(n)
        g_dir[min_idx] = 1.0

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K=400):
    """LP FW with full constraint set using cvxpy."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        K_sel = min(top_K, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix: for each lag k, conv[k] = sum_i a[i]*a[k-i]
        # Linearize around y: direction d should satisfy sum_i (d[i]*y[k-i] + y[i]*d[k-i])/2 <= M
        # i.e., sum_i d[i]*(y[k-i] + y[k-i])/2 ... 
        # Actually: conv_y+d[k] ≈ conv_y[k] + sum_i (d[i]*y[k-i] + y[i]*d[k-i])
        # = conv_y[k] + 2 * sum_i d[i]*y[k-i] (by symmetry of the sum)
        # Wait: sum_i d[i]*y[k-i] + sum_i y[i]*d[k-i] = sum_i d[i]*y[k-i] + sum_j d[j]*y[k-j] = 2*sum_i d[i]*y[k-i]
        # So we need: 2 * sum_i d[i]*y[k-i] <= M - conv_y[k] = M - ac[k] (for binding constraints)
        # For the max constraint: 2 * sum_i d[i]*y[k_peak-i] <= 0 (to not increase the peak)

        # Build the linear constraint matrix
        T = np.zeros((K_sel, n))
        for row, k in enumerate(top_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T[row, idx] = 2.0 * y[j_idx]

        rhs = M_curr - ac[top_lags]  # slack for each constraint

        d_var = cp.Variable(n)
        constraints = [
            d_var >= -y,  # y + d >= 0
            cp.sum(d_var) == 0,  # preserve sum
            T @ d_var <= rhs
        ]

        # Objective: minimize the peak increase (use negative dot with gradient)
        k_peak = int(top_lags[0])
        g_peak = peak_subgradient(y, k_peak)
        objective = cp.Minimize(cp.sum(cp.multiply(d_var, np.maximum(g_peak, 0))))

        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        d_val = np.asarray(d_var.value, dtype=np.float64)
        y_new = y + d_val
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        alpha = 2.0 / (t + 2.0)
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_inits(n):
    """Generate diverse initializations."""
    inits = []
    labels = []

    # Uniform
    inits.append(np.ones(n) / n)
    labels.append("uniform")

    # Cosine window (edge-concentrated)
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.05
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    labels.append("cos_edge")

    # Linear ramp (asymmetric)
    a = np.arange(1, n + 1, dtype=np.float64)
    inits.append(a / a.sum())
    labels.append("ramp")

    # Quadratic bowl (centered low, edges high)
    x = np.linspace(-1, 1, n)
    a = x ** 2 + 0.1
    inits.append(a / a.sum())
    labels.append("bowl")

    # Two-bump asymmetric
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / 0.08) ** 2) + 0.7 * np.exp(-((x - 0.75) / 0.12) ** 2) + 0.01
    inits.append(a / a.sum())
    labels.append("two_bump_asym")

    # S-curve (smooth asymmetric)
    a = 1.0 / (1.0 + np.exp(-8 * (x - 0.4))) + 0.01
    inits.append(a / a.sum())
    labels.append("sigmoid")

    # Smooth random
    rng = np.random.default_rng(123)
    a = np.abs(rng.normal(1.0, 0.3, n))
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    inits.append(a / a.sum())
    labels.append("smooth_rand")

    return inits, labels


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with peak-cycling optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Peak cycling from parent ===
    print(f"\n=== Phase 1: Peak Cycling (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = peak_cycling_descent(norm_sum1(best_seq), t1, verbose=True,
                                         top_k_soft=8, soft_beta=4.0, seed=42)
        try_update(a_out, "peakcycle_parent")

    # === Phase 2: Diverse initializations ===
    print(f"\n=== Phase 2: Diverse Inits (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = min(remaining * 0.20, 300.0)
    if t2_total > 30:
        inits, labels = make_inits(n0)
        # Add parent as an init
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        n_inits = len(inits)
        t_per = max(15.0, t2_total / n_inits)

        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  PC init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = peak_cycling_descent(init, t_avail, verbose=False,
                                             top_k_soft=6, soft_beta=5.0, seed=100 + idx)
            try_update(a_out, f"pc_{lbl}")

    # === Phase 3: Pairwise moves on best ===
    print(f"\n=== Phase 3: Pairwise Moves (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_moves(a_in, t3, verbose=True, seed=77)
        try_update(a_out, f"pairwise_n{len(a_in)}")

    # === Phase 4: LP Frank-Wolfe v2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW2 (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t4, solver, verbose=True, top_K=300)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 5: Peak cycling with higher soft params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: PC polish (n={n_best}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_cycling_descent(a_in, t5, verbose=False,
                                         top_k_soft=12, soft_beta=3.0, seed=555)
        try_update(a_out, f"pc_polish_n{n_best}")

    # === Phase 6: Pairwise moves round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Pairwise round 2 (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_moves(a_in, t6, verbose=True, seed=222)
        try_update(a_out, f"pairwise2_n{n_best}")

    # === Phase 7: Longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Longer sequences ===")
    for new_n in [n_best + 400, n_best + 800]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60 or new_n > 5000:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(norm_sum1(best_seq), new_n)
        t7a = min(remaining * 0.15, 120.0)
        if t7a > 20:
            a_out, _ = peak_cycling_descent(a_long, t7a, verbose=False,
                                             top_k_soft=10, soft_beta=3.5, seed=666)
            try_update(a_out, f"pc_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_in = norm_sum1(best_seq)
            a_long2 = interpolate_sequence(a_in, new_n)
            a_out, _ = pairwise_moves(a_long2, min(remaining * 0.1, 60.0), verbose=False, seed=333)
            try_update(a_out, f"pairwise_n{new_n}")

    # === Phase 8: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 3500:
        t8 = min(remaining * 0.3, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t8, solver, verbose=True, top_K=400)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 9: Final peak cycling ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final PC (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.5, 120.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_cycling_descent(a_in, t9, verbose=False,
                                             top_k_soft=15, soft_beta=2.5, seed=4321)
            try_update(a_out, f"pc_final_n{n_best}")

    # === Phase 10: Final pairwise ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 10: Final Pairwise (n={n_best}) ===")
    if remaining > 15:
        t10 = min(remaining * 0.6, 80.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_moves(a_in, t10, verbose=True, seed=7777)
            try_update(a_out, f"pairwise_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
