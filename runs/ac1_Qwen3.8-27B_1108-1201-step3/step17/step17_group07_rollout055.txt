def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)

    tau = (cssv[rho] - 1) / (rho + 1.0)

            # k >= n: b[k] = sum_{j=k-n+1}^{n-1} a[j]*a[k-j]
            # grad = a[j-k] + a[j+k] but j-k < 0 always (since j < n <= k)
            # and j+k >= k >= n, so a[j+k] = 0
            # So grad = a[j-k] for valid j, but j-k < 0... 
            # Actually for k >= n: b[k] = sum_j a[j]*a[k-j] where k-j is in [0,n-1]
            # i.e., j is in [k-n+1, n-1]
            # db[k]/da[j] = a[k-j] for j in [k-n+1, n-1] (only one term since j+k >= n)
            j_lo = k_int - n + 1
            j_hi = n - 1
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += a[k_int - j_hi:k_int - j_lo]
    
    g /= len(active)
    return g, active, M

def nesterov_subgrad(a0, time_limit, verbose=False, epsilon=0.005, eta0=0.01):
    """Nesterov-accelerated subgradient with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = eta0
    stall = 0
    max_stall = 5000
    rng = np.random.default_rng(12345 + n)
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        b = fast_conv(y)
        M = b.max()
        g, active, M = compute_active_subgrad(y, b, epsilon)
        
        # Project gradient onto tangent space of simplex (remove mean component)
        g_proj = g - g.mean()
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-15:
            break
        
        # Backtracking line search
        accepted = False
        for _ in range(15):
            a_new = project_simplex(y - eta * g)
            b_new = fast_conv(a_new)
            M_new = b_new.max()
            if M_new < M - 1e-16:
                accepted = True
                break
            eta *= 0.7
        
        if not accepted:
            # Accept the step anyway (it's subgradient, so we might not decrease at every step)
            a_new = project_simplex(y - eta * g)
            b_new = fast_conv(a_new)
            M_new = b_new.max()
        
        x_prev = x
        x = a_new
        
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            eta = min(eta * 1.05, eta0 * 2)
        else:
            stall += 1
            if stall % 100 == 0:
                eta *= 0.95
        
        if stall > max_stall:
            stall = 0
            max_stall = max(1000, int(max_stall * 0.6))
            eta = eta0 * rng.uniform(0.3, 2.0)
            # Targeted perturbation
            strategy = int(rng.integers(4))
            if strategy == 0:
                # Perturb near the peak lags
                peak_lags = np.where(fast_conv(best_a) >= best_m * 0.999)[0]
                pert = np.zeros(n)
                for k in peak_lags[:20]:
                    k_int = int(k)
                    if k_int < n:
                        pert[k_int:] += rng.normal(0, 0.01, n - k_int)
                        pert[:n - k_int] += rng.normal(0, 0.01, n - k_int)
                x = best_a * (1.0 + pert)
            elif strategy == 1:
                x = best_a * (1.0 + rng.normal(0, 0.01, n))
            elif strategy == 2:
                # Random subset perturbation
                idx = rng.choice(n, size=min(n//3, 500), replace=False)
                x = best_a.copy()
                x[idx] *= rng.uniform(0.7, 1.3, len(idx))
            else:
                x = best_a + rng.normal(0, 0.005, n)
            
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            x_prev = x.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NS t={t} ({elapsed:.1f}s): J={J:.8f}, |A|={len(active)}, eta={eta:.5f}, stall={stall}")
    
    return best_a, best_m

def lp_remez_step(a, max_active=50, r=0.05, solver=None):
    """Single LP Remez step with dual variable extraction."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Select active constraints: all within 0.1% of max, up to max_active
    threshold = M * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        # Keep the top max_active
        active = active[np.argsort(b[active])[::-1][:max_active]]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    dual_vars = []
    
    for idx, k in enumerate(active):
        k_int = int(k)
        # b[k] quadratic linearized around a: b[k](a') ≈ 2*sum_j a[j]*a'[j+k] - b[k] (first order)
        # Actually: b[k](a') = sum_j a'[j]*a'[k-j], linearized: sum_j (a[j]*a'[k-j] + a[j+k]*a'[j])
        # = sum_j a[j]*a'[k-j] + sum_j a[j+k]*a'[j]
        # Let me use: b[k](a') ≈ 2 * (a shifted) dot (a' shifted) - b[k](a)
        # More precisely: b[k](a') ≈ b[k](a) + sum_j (a[k-j]*a'[j] + a[j]*a'[k-j])
        # = b[k](a) + 2*sum_j a[k-j]*a'[j] (by reindexing, these are the same sum)
        # Wait: sum_j a[k-j]*a'[j] and sum_j a[j]*a'[k-j]. Let l=k-j in the second: sum_l a[k-l]*a'[l]. Same!
        # So b[k](a') ≈ b[k](a) + 2*sum_j a[k-j]*a'[j]
        
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n - 1, k_int)
        if j_lo <= j_hi:
            idx_arr = np.arange(j_lo, j_hi + 1)
            coeffs = a[k_int - idx_arr]
            # Linearized: b[k](a') ≈ b[k](a) + 2*sum coeffs[j]*a'[j]
            lhs = 2.0 * cp.sum(cp.multiply(coeffs, a_var[idx_arr]))
            constraints.append(lhs - M_var <= b[k_int])
            dual_vars.append(constraints[-1])
    
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), None
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), None
    
    # Extract dual variables for active constraints
    lam = None
    try:
        lam_vals = []
        for c in dual_vars:
            d = c.dual_value
            if d is not None and np.isfinite(d):
                lam_vals.append(max(0.0, float(d)))
            else:
                lam_vals.append(0.0)
        lam = np.array(lam_vals)
        if lam.sum() > 1e-15:
            lam = lam / lam.sum()
        else:
            lam = None
    except Exception:
        lam = None
    
    return a_new / s, (active, lam)

def remez_method(a0, time_limit, solver=None, r_init=0.08, verbose=False):
    """LP Remez with adaptive trust region and dual-guided line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    max_no_improve = 30
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand, dual_info = lp_remez_step(a, max_active=40, r=r, solver=solver)
        
        # Line search
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1.0 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.25, r * 1.15)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.7)
            if no_improve > max_no_improve:
                rng = np.random.default_rng(t * 7 + 3)
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = r_init * 0.5
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, no_imp={no_improve}")
    
    return best_a, best_m

def greedy_pair_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Greedy pair-exchange coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_attempts = min(200, n // 2)
        
        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick random i, try to move mass from i to j
            i = int(rng.integers(n))
            if a[i] < 1e-12:
                continue
            
            for frac in [0.001, 0.005, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                if amt < 1e-15:
                    break
                
                # Try several random j positions
                for _ in range(5):
                    j = int(rng.integers(n))
                    if j == i:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    
                    b_trial = fast_conv(a_trial)
                    m_trial = b_trial.max()
                    
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            
            # Occasionally do a more structured move
            if stall % 3 == 0:
                # Move mass from high values to low values at "good" positions
                b = fast_conv(a)
                peak_lags = np.where(b >= b.max() * 0.999)[0]
                # Find positions that contribute most to peak lags
                contrib = np.zeros(n)
                for k in peak_lags[:10]:
                    k_int = int(k)
                    if k_int < n:
                        contrib[:n-k_int] += a[k_int:]  # a[j+k] for j from 0
                        contrib[k_int:] += a[:n-k_int]  # a[j-k] for j from k
                # Move mass from high-contrib high-value positions to low-contrib positions
                high_idx = np.where(contrib > np.median(contrib) + 0.1 * contrib.max())[0]
                low_idx = np.where(contrib < np.median(contrib) - 0.1 * contrib.max())[0]
                if len(high_idx) > 0 and len(low_idx) > 0:
                    hi = high_idx[int(rng.integers(len(high_idx)))]
                    lo = low_idx[int(rng.integers(len(low_idx)))]
                    for frac in [0.01, 0.03, 0.05]:
                        amt = a[hi] * frac
                        a_trial = a.copy()
                        a_trial[hi] -= amt
                        a_trial[lo] += amt
                        if a_trial[hi] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        m_trial = fast_conv(a_trial).max()
                        if m_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    """Create various initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bump':
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    elif kind == 'two_bump':
        a = np.exp(-100.0 * (x - 0.3) ** 2) + np.exp(-100.0 * (x - 0.7) ** 2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def dual_guided_subgrad(a, b, active_lags, lam, timestep):
    """Compute dual-guided subgradient direction."""
    n = len(a)
    if lam is None or len(lam) == 0:
        g, active, M = compute_active_subgrad(a, b, 0.005)
        return g
    g = np.zeros(n)
    for idx, k in enumerate(active_lags):
        k_int = int(k)
        w = lam[idx] if idx < len(lam) else 0.0
        if k_int == 0:
            g += 2.0 * w * a
        elif k_int < n:
            g[k_int:] += w * a[:n - k_int]
            g[:n - k_int] += w * a[k_int:]
        else:
            j_lo = k_int - n + 1
            j_hi = n - 1
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += w * a[k_int - j_hi:k_int - j_lo]
    total = lam.sum() if lam.sum() > 1e-15 else 1.0
    g /= total
    return g

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2600
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Nesterov subgradient from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Nesterov Subgradient n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_subgrad(norm_sum1(best_seq), t1, verbose=True, epsilon=0.004, eta0=0.008)
        try_update(a_out, f"NS_parent_n{n0}")
    
    # === Phase 2: Remez LP (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: Remez LP n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_method(norm_sum1(best_seq), t2, solver=solver, r_init=0.06, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
    
    # === Phase 3: Nesterov subgradient refinement (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: NS refinement n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = nesterov_subgrad(norm_sum1(best_seq), t3, verbose=True, epsilon=0.002, eta0=0.003)
        try_update(a_out, f"NS_refine_n{len(best_seq)}")
    
    # === Phase 4: Greedy pair descent (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Greedy CD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = greedy_pair_descent(norm_sum1(best_seq), t4, verbose=False, rng_seed=777)
        try_update(a_out, f"CD_n{len(best_seq)}")
    
    # === Phase 5: Remez polish with tight trust region (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Remez polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = remez_method(norm_sum1(best_seq), t5, solver=solver, r_init=0.02, verbose=True)
        try_update(a_out, f"Remez_polish_n{len(best_seq)}")
    
    # === Phase 6: Multi-init exploration (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6_total = min(remaining * 0.05, 100.0)
    if t6_total > 40:
        print(f"\n=== Phase 6: Multi-init n={n0} (budget={t6_total:.0f}s) ===")
        inits = ['raised_cos', 'gaussian', 'two_bump']

            if remaining < 15:

        remaining = time_budget - (time.monotonic() - t_global)

            try_update(a_out, f"NS_final_{n_best}")
    
    # === Phase 8: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n0 + 100
        print(f"\n=== Phase 8: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
