            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_subgradient(a, ac, exact=True, top_k=20, beta=2.0):
    """Compute subgradient of max(conv(a,a)) or smoothed version."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_mirror_descent(a0, time_limit, top_k_init=20, beta_init=2.0,
                            c_init=3.0, exact_frac=0.5, restart_every=3000,
                            seed=42, verbose=False):
    """Nesterov-accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k_init
    cur_beta = beta_init
    cur_c = c_init
    cur_exact_frac = exact_frac
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Adaptive subgradient
        if rng.random() < cur_exact_frac:
            g = get_subgradient(y, ac, exact=True)
        else:
            g = get_subgradient(y, ac, exact=False, top_k=cur_top_k, beta=cur_beta)
        
        # Polyak step size
        lower_bound = 1.0 / n
        gap = max(M - lower_bound, 1e-10)
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)
        
        # Mirror descent step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive restart
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall / restart_every, 3.0)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.0)
            cur_c = min(cur_c + 0.5, 10.0)
            cur_exact_frac = min(cur_exact_frac + 0.05, 0.8)
            restart_every = min(restart_every + 1000, 10000)
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MGD restart ({elapsed:.0f}s): J={best_J:.6f}")
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MGD t={t} ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J


def coordinate_descent(a0, time_limit, seed=123, verbose=False):
    """
    Coordinate descent: try changing one element at a time.
    For each element, try increasing and decreasing it, adjusting others to maintain sum=1.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Identify the peak lag
        peak_k = int(np.argmax(ac))
        
        # Elements contributing to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        peak_elements = np.arange(i_start, i_end)
        
        # Try modifying peak elements
        for i in peak_elements:
            if a[i] < 1e-10:
                continue
            
            # Try decreasing a[i] and distributing to non-peak elements
            non_peak = np.setdiff1d(np.arange(n), peak_elements)
            if len(non_peak) == 0:
                continue
            
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                # Distribute to random non-peak element
                j = non_peak[rng.integers(len(non_peak))]
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    stall = 0
                    break
        
        if stall > 0:
            stall += 1
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=600,
                   update_freq=5):
    """LP Frank-Wolfe with dynamic constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < 1e-16:
            break
        
        # Select top-K constraints
        K_sel = min(top_K, 2 * n - 1)
        margins = M_curr - ac
        binding_idx = np.argsort(margins)[:K_sel]
        
        # Build constraint matrix (sparse representation)
        # For constraint k: sum_{i} a[i] * y[k-i] <= M_curr
        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T_sel[row, idx] = y[j_idx]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Step size (line search or fixed)
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
