def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_softmax_gradient(a, T, top_k=200):
    """Compute gradient of soft-max approximation to max(autocorrelation)."""
    n = len(a)
    ac = fast_autocorr(a)
    # Use top-k lags for efficiency
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    vals = ac[top_idx]
    # Soft weights
    max_val = vals[0]
    exp_vals = np.exp((vals - max_val) / T)
    w = exp_vals / exp_vals.sum()
    # Gradient: sum_k w_k * dc[k]/da[j]
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        # dc[k]/da[i] = a[k-i] + a[i-k] (with bounds)
        left = k_int - idx
        right = idx - k_int
        contrib = np.zeros(n)
        valid_l = (left >= 0) & (left < n)
        contrib[idx[valid_l]] += a[left[valid_l]]
        valid_r = (right >= 0) & (right < n)
        contrib[idx[valid_r]] += a[right[valid_r]]
        # Fix double counting at i = k-i (i.e., k=2i)
        if k_int % 2 == 0:
            mid = k_int // 2
            if 0 <= mid < n:
                i_start_m = max(0, k_int - n + 1)
                i_end_m = min(n, k_int + 1)
                if i_start_m <= mid < i_end_m:
                    contrib[mid] -= a[mid]  # subtract one copy
        grad += wi * contrib
    return grad, ac


def softmax_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent on soft-max approximation with annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    # Temperature schedule: start high, anneal down
    T_init = best_m * 0.5
    T_final = max(best_m * 0.001, 1e-10)
    n_iter = 50000

    x = a.copy()
    x_prev = a.copy()
    gamma = 1.0

    for t in range(1, n_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        frac = (t - 1) / max(n_iter - 1, 1)
        T = T_init * (T_final / T_init) ** frac

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient
        grad, ac_y = compute_softmax_gradient(y, T, top_k=100)

        # Mirror descent step (exponential family)
        eta = gamma / (np.sqrt(t) + 1.0) * T
        log_new = np.log(y) - eta * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        # Update best
        M = fast_autocorr(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = x.copy()

        # Periodic restart to escape local minima
        if t % 5000 == 0:
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            M_r = fast_autocorr(x_r).max()
            if M_r < best_m:
                best_m = M_r
                best_a = x_r.copy()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, T={T:.2e}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K=200,
                       margin_frac=2e-3, line_search=True):
    """LP Frank-Wolfe with adaptive active set and optional quadratic line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    margin = margin_frac

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Adaptive active set
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        if line_search:
            # Quadratic line search along FW direction
            best_alpha = 2.0 / (t + 2.0)
            best_M = M_curr
            for alpha in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5,
                          2.0 / (t + 2.0)]:
                a_trial = (1.0 - alpha) * y + alpha * g_dir
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_alpha = alpha

            a_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / sa
        else:
            alpha = 2.0 / (t + 2.0)
            a_new = (1.0 - alpha) * y + alpha * g_dir
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / sa

        # Nesterov acceleration
        if t == 1:
            x_new = a_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = a_new + beta_t * (a_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = a_new.copy()
            else:
                x_new = x_new / sx

        y = a_new
        x = x_new

        # Update best
        My = fast_autocorr(y).max()
        if My < best_m:
            best_m = My
            best_a = y.copy()
            # Tighten margin when improving
            margin = max(margin * 0.9, 1e-5)

        Mx = fast_autocorr(x).max()
        if Mx < best_m:
            best_m = Mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPv2 iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, K={K}, m={margin:.1e}")

    return best_a, best_m


def danger_weighted_peak_crush(a0, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent that moves mass from 'dangerous' to 'safe' positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peaks
        n_peaks = min(20, len(ac))
        peak_idx = np.argsort(ac)[::-1][:n_peaks]
        peak_vals = ac[peak_idx]
        # Weights proportional to peak values
        w = peak_vals ** 3
        w = w / w.sum()

        # Compute danger for each position
        danger = np.zeros(n)
        for wi, k in zip(w, peak_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            # Contribution of a[i] to c[k] is a[i] * a[k-i]
            left = k_int - idx
            right = idx - k_int
            contrib = np.zeros(n)
            valid_l = (left >= 0) & (left < n)
            contrib[idx[valid_l]] += a[idx[valid_l]] * a[left[valid_l]]
            valid_r = (right >= 0) & (right < n)
            contrib[idx[valid_r]] += a[idx[valid_r]] * a[right[valid_r]]
            danger += wi * contrib

        # Normalize danger
        d_max = danger.max()
        if d_max < 1e-20:
            stall += 1
            break
        danger = danger / d_max

        # Safe positions: low danger, high value
        # We want to move FROM high danger TO low danger
        # Pick top dangerous source and random safe destination
        n_top_src = min(30, n)
        src_candidates = np.argsort(danger)[::-1][:n_top_src]
        src_candidates = src_candidates[a[src_candidates] > 1e-8]

        if len(src_candidates) == 0:
            stall += 1
            continue

        improved = False
        for src in src_candidates[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-10:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-14:
                    continue
                # Try random destinations, prefer low danger
                n_trials = 40
                for _ in range(n_trials):
                    # Sample destination weighted by (1 - danger)
                    safe_weights = np.maximum(1.0 - danger, 0.01)
                    safe_weights[src] = 0.0
                    tw = safe_weights / safe_weights.sum()
                    dst = int(rng.choice(n, p=tw))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m - 1e-15:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)
                if best_move is not None:
                    break

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 40 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    DWPC ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_flattening(a0, time_limit, verbose=True):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Find frequency bins with highest power
        max_power = power.max()
        if max_power < 1e-15:
            break

        # Target: reduce high-power bins
        threshold = 0.7 * max_power
        high_bins = power > threshold

        # Compute correction in frequency domain
        # Reduce amplitude at high bins
        A_corr = A.copy()
        A_corr[high_bins] *= 0.95

        # Transform back
        a_new = np.fft.irfft(A_corr, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Also try blending
        alpha = 0.1 + 0.05 * np.sin(t * 0.01)
        a_blend = (1 - alpha) * a + alpha * a_new
        a_blend = np.maximum(a_blend, 0.0)
        sb = a_blend.sum()
        if sb < 1e-15:
            a_blend = a_new
        else:
            a_blend = a_blend / sb

        M = fast_autocorr(a_blend).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a_blend.copy()
        a = a_blend

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpecFlatten t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            best_seq = a_norm.copy()
            best_score = sc

    # === Phase 1: Soft-max mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Soft-max MD (n={n0}) ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 30:
        a_out, _ = softmax_mirror_descent(best_seq, t1, verbose=True)
        try_update(a_out, "smd")

    # === Phase 2: LP Frank-Wolfe v2 with line search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-FW v2 (n={n_cur}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40 and n_cur <= 5000:
        a_out, _ = lp_frank_wolfe_v2(best_seq, t2, solver, verbose=True,
                                      top_K=250, margin_frac=2e-3, line_search=True)
        try_update(a_out, "lpfw2")

    # === Phase 3: Danger-weighted peak crushing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: DWPC (n={n_cur}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = danger_weighted_peak_crush(best_seq, t3, verbose=True, rng_seed=42)
        try_update(a_out, "dwpc")

    # === Phase 4: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Spectral flatten (n={n_cur}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = spectral_flattening(best_seq, t4, verbose=True)
        try_update(a_out, "specflat")

    # === Phase 5: LP-FW with tighter margin ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP-FW tight (n={n_cur}) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30 and n_cur <= 5000:
        a_out, _ = lp_frank_wolfe_v2(best_seq, t5, solver, verbose=True,
                                      top_K=350, margin_frac=5e-4, line_search=True)
        try_update(a_out, "lpfw_tight")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    lengths_to_try = sorted(set([
        n_cur - 400, n_cur - 200, n_cur + 200, n_cur + 400, n_cur + 800
    ]))
    lengths_to_try = [l for l in lengths_to_try if 2000 <= l <= 5500]

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick SMD
        t_sub = min(remaining * 0.20, 120.0)
        if t_sub > 20:
            a_out, _ = softmax_mirror_descent(a_long, t_sub, verbose=False)
            try_update(a_out, f"smd_n{new_n}")

        # Quick LP
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4500:
            t_sub2 = min(remaining * 0.15, 100.0)
            if t_sub2 > 20:
                a_in = norm_sum1(best_seq) if len(best_seq) == new_n else a_long
                a_out, _ = lp_frank_wolfe_v2(a_in, t_sub2, solver, verbose=False,
                                              top_K=200, margin_frac=2e-3, line_search=True)
                try_update(a_out, f"lpfw_n{new_n}")

    # === Phase 7: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Freq diversify (n={n_cur}) ===")
    rng = np.random.default_rng(123)
    t7_per = min(remaining * 0.08, 100.0)
    if t7_per > 30:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t7_per + 20:
                break
            a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
            a_out, _ = lp_frank_wolfe_v2(a_pert, t7_per, solver, verbose=False,
                                          top_K=200, margin_frac=1e-3, line_search=True)
            try_update(a_out, f"freq{trial}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_cur}) ===")
    t8 = min(remaining * 0.20, 250.0)
    if t8 > 30 and n_cur <= 5000:
        a_out, _ = lp_frank_wolfe_v2(best_seq, t8, solver, verbose=True,
                                      top_K=400, margin_frac=1e-4, line_search=True)
        try_update(a_out, "lp_final")

    # === Phase 9: Final DWPC polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final DWPC (n={n_cur}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.30, 200.0)
        if t9 > 15:
            a_out, _ = danger_weighted_peak_crush(best_seq, t9, verbose=True, rng_seed=777)
            try_update(a_out, "dwpc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
