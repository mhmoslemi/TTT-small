    weights = np.exp(shifted)
    weights /= weights.sum()

    a_flip = a[::-1]
    total_len = conv_len + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a_flip, fft_size)
    corr = np.fft.irfft(W * A, fft_size)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return g


def topk_subgradient(a, conv, top_k=80, beta_w=10.0):
    """Compute subgradient using top-K lags with exponential weighting."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent_smooth(a0, time_limit, T_start=8.0, T_end=0.3, c0=3.0,
                          n_iters_max=30000, verbose=False):
    """Nesterov-accelerated mirror descent with annealed smoothing."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    stall = 0
    T_current = T_start

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, n_iters_max + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Annealing: exponential decay
        progress = min(1.0, t / (n_iters_max * 0.85))
        T_current = T_start * (T_end / T_start) ** progress

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Smoothed subgradient
        conv = fast_conv(y)
        g = smoothed_subgradient(y, conv, T_current)

        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            g_norm = 1e-15
        eta = c0 / (np.sqrt(t) + 1.0) / max(1.0, g_norm / (2.0 * (t + 1) ** 0.5 / c0))
        eta = min(eta, 5.0)

        # Mirror step (KL projection onto simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stuck
        if stall > 3000:
            stall = 0
            rng = np.random.default_rng(t)
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-12)
            x /= x.sum()
            x_prev = x.copy()

        if verbose and t % 5000 == 0:
            print(f"    MD_smooth t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, T={T_current:.3f}")

    return best_a, best_val


def mirror_descent_topk(a0, time_limit, top_k=100, beta_w=8.0, c0=3.5,
                        restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent with top-K subgradient."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)
    cur_top_k = top_k
    cur_beta = beta_w

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_conv(y)
        g = topk_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        eta = c0 / (np.sqrt(t) + 1.0)

        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.04)
            pert = rng.normal(0, scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-12)
            x /= x.sum()
            x_prev = x.copy()
            cur_top_k = min(cur_top_k + 20, 250)
            cur_beta = max(cur_beta - 0.5, 2.0)
            c0 = max(c0 * 0.95, 1.0)
            if verbose:
                print(f"    MD_topk restart t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

        if verbose and t % 5000 == 0:
            print(f"    MD_topk t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def coord_descent_peakafter(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with structured peak-aware mass transfer."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        n_peaks = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_peaks]

        # Compute pressure: which positions contribute most to peaks
        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        # Source: highest pressure * mass
        src_scores = pressure * a
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        # Destination: minimize increase in peak values
        dest_penalty = np.zeros(n)
        for k in top_lags[:10]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_penalty[idx] += a[k_i - idx]

        dest_penalty[src] = np.inf
        dest_penalty += 0.05 * pressure
        dst = int(np.argmin(dest_penalty))

        # Try multiple transfer fractions
        improved = False
        for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.08, 0.15, 0.25, 0.4]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.005, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 100 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=60.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.003
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier components."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric(n, p=1.0, shift=0.25):
    """Asymmetric distribution."""
    x = np.linspace(0, 1, n)
    a = ((x + shift / max(1e-6, 1 - 2 * shift)) ** p) * ((1 - x) ** 0.4)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_triangular(n, peak=0.3):
    """Triangular distribution with peak at specified location."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - np.abs(x - peak) / (0.5 + 0.3 * (1 - peak)))
    a += 0.002
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def block_perturbation(a, n_blocks=16, scale=0.02, seed=0):
    """Perturb blocks of the sequence together."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a_new = a.copy()
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        pert = rng.normal(0, scale, end - start)
        a_new[start:end] += pert * a[start:end]
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-strategy approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"Starting optimization with n={n0}, initial J={best_score:.6f}")

    # === Phase 1: Multi-init with annealed smoothing (exploration) ===
    print(f"\n=== Phase 1: Annealed MD Multi-Init ===")
    remaining = time_budget - (time.monotonic() - t_global)

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.7))
    labels.append("ramp07")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp10")
    inits.append(make_power_ramp(n0, 1.4))
    labels.append("ramp14")
    inits.append(make_bimodal(n0))
    labels.append("bimodal")
    inits.append(make_smooth_random(n0, seed=7))
    labels.append("smooth7")
    inits.append(make_asymmetric(n0, p=1.2, shift=0.2))
    labels.append("asym")
    inits.append(make_triangular(n0, peak=0.25))
    labels.append("tri25")

    if parent is not None:
        blend1 = 0.6 * parent + 0.4 * make_power_ramp(n0, 1.5)
        inits.append(blend1 / blend1.sum())
        labels.append("blend")

    # Deduplicate
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 80)], 3))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    t_per = max(10.0, remaining * 0.25 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 120:
            break
        t_avail = min(t_per, time_budget - 120 - elapsed)
        if t_avail < 8:
            break
        print(f"  AnnealMD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = mirror_descent_smooth(init, t_avail, T_start=6.0, T_end=0.5, c0=3.0)
        try_update(a_out, f"anneal_{lbl}")

    # === Phase 2: Extended annealed MD on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended Annealed MD ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = mirror_descent_smooth(best_a.copy(), t2, T_start=4.0, T_end=0.2,
                                         c0=2.5, verbose=True)
        try_update(a_out, "anneal_ext")

    # === Phase 3: Top-K mirror descent (exploitation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Top-K MD Exploitation ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        a_out, _ = mirror_descent_topk(best_a.copy(), t3, top_k=100, beta_w=7.0,
                                       c0=3.0, restart_every=4000, verbose=True)
        try_update(a_out, "topk_exploit")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak-aware Coord Descent ===")
    t4 = min(remaining * 0.10, 160.0)
    if t4 > 15:

    # === Phase 5: MD after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MD after CD ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_out, _ = mirror_descent_smooth(best_a.copy(), t5, T_start=3.0, T_end=0.15,
                                         c0=2.0)
        try_update(a_out, "md_after_cd")

    # === Phase 6: Block perturbation search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Block Perturbation Search ===")
    t6_budget = min(remaining * 0.08, 120.0)

        rng6 = np.random.default_rng(555)
        n_trials = int(t6_budget / 2.0)
        for trial in range(n_trials):
            if time.monotonic() - t_global > time_budget - 60:
                break
            scale = rng6.uniform(0.005, 0.08)
            n_blocks = int(rng6.integers(4, 32))
            a_pert = block_perturbation(best_a, n_blocks=n_blocks, scale=scale,
                                        seed=trial * 1000 + 7)
            # Quick MD refinement
            t_mini = min(3.0, (time_budget - 60 - (time.monotonic() - t_global)) / 2)
            if t_mini > 1.0:
                a_out, _ = mirror_descent_topk(a_pert, t_mini, top_k=80, beta_w=8.0, c0=2.5)
                try_update(a_out, f"block{trial}")

    # === Phase 7: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-300, -150, 150, 300, 500, -500]:
        nl = n_best + d
        if 2000 <= nl <= 5000:
            lengths_to_try.append(nl)
    for nl in [2500, 3000, 3500, 4000]:
        if nl not in lengths_to_try and 2000 <= nl <= 5000:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length sweep: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.30, 80.0)
        if t_md > 10:
            a_out, _ = mirror_descent_smooth(a_int, t_md, T_start=5.0, T_end=0.3, c0=3.0)
            try_update(a_out, f"len{nl}_smooth")

            if remaining > 15:
                t_topk = min(remaining * 0.15, 40.0)

                    try_update(a_out2, f"len{nl}_topk")

    # === Phase 8: Final polish - alternating smooth + topk ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 8: Final Polish (n={n_cur}) ===")
    t8 = min(remaining * 0.20, 200.0)

                                         c0=2.0, verbose=True)

        if remaining > 15:

                                           top_k=150, beta_w=5.0, c0=2.0,

    remaining = time_budget - (time.monotonic() - t_global)

        a_out, _ = mirror_descent_smooth(best_a.copy(), remaining - 5,
                                         T_start=1.5, T_end=0.05, c0=1.5)

    # Final evaluation
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
