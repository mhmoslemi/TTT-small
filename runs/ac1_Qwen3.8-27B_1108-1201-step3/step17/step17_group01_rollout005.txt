def build_constraint_matrix(y, binding_idx, n):
    """Build the constraint matrix for the LP.
    For the bilinear linearization: L_b[k] = sum_j (y[j]*b[k-j] + y[k-j]*b[j])
    So T[k,j] = y[j] + y[k-j] (if both indices valid)
    But we handle the case k-j == j (i.e., k=2j) carefully.

    Actually simpler: T[k,j] = y[j] + y[k-j] for valid k-j, and if k==2j, it's 2*y[j].
    """
    K = len(binding_idx)
    T = np.zeros((K, n))
    for i, k in enumerate(binding_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        # T[i, j] = y[j] + y[k-j] (with y[k-j] = 0 if k-j out of range, but k-j is in [j_start, j_end] range)
        # k-j for j in [j_start, j_end]: k-j in [k-j_end, k-j_start] = [k-(k)=0 ... k-j_start]

        # So k-j is always in [0, n-1] for j in [j_start, j_end]. Good.
        T[i, js] = y[js] + y[k_int - js]

def nesterov_subgradient(a0, time_limit, top_k=80, beta_pow=4.0, c0=5.0, verbose=False):

        else:

            if j_start > j_end:

            js = np.arange(j_start, j_end + 1)
            grad[js] += wi * (y[js] + y[k_int - js])

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-15)

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-FW on parent with improved method ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW improved (n={n0}) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t1, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n0}")

    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coord descent (n={n_best}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t2, verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 3: Nesterov subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov subgrad (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t3, top_k=100, beta_pow=3.5, c0=4.0, verbose=False)
        try_update(a_out, f"nesterov_n{n_best}")

    # === Phase 4: LP-FW with different params (more constraints) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW more constraints (n={n_best}) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t4, solver, verbose=True)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 5: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length search ===")

    candidate_lengths = []
    # Try nearby lengths
    for delta in [-400, -200, 200, 400, 600]:
        ln = n_best + delta
        if 1000 <= ln <= 8000:
            candidate_lengths.append(ln)

    # Deduplicate and sort by proximity to current best
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        if new_n == n_best:
            continue

        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t5a = min(remaining * 0.35, 150.0)
        if t5a > 30:
            a_out, _ = lp_frank_wolfe_improved(a_interp, t5a, solver, verbose=False)
            try_update(a_out, f"lpfw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.25, 80.0)
            if t5b > 20:
                a_out, _ = coordinate_descent_polish(a_interp, t5b, verbose=False)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Diverse restart + LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 6: Diverse restarts (n={n_best}) ===")

        restarts = [
            ("cos_taper", None),
            ("power_law", None),
            ("perturbed", None),
        ]

        for label, _ in restarts:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break

            if label == "cos_taper":
                x = np.linspace(0, 1, n_best)
                a_init = (np.cos(np.pi * (x - 0.5)) ** 2) * 0.3 + 0.7 * best_seq / best_seq.sum()
                a_init = norm_sum1(a_init)
            elif label == "power_law":
                idx = np.arange(n_best, dtype=np.float64)
                center = (n_best - 1) / 2.0
                a_init = 1.0 / (1.0 + 0.5 * np.abs(idx - center) / n_best)
                a_init = norm_sum1(a_init)
                # Blend with best
                a_init = 0.5 * a_init + 0.5 * norm_sum1(best_seq)
                a_init = norm_sum1(a_init)
            else:  # perturbed
                n_best_local = n_best
                pert = rng.normal(0, 0.05, n_best_local) * norm_sum1(best_seq)
                a_init = norm_sum1(norm_sum1(best_seq) + pert)

            try_update(a_init, f"init_{label}")

            t6 = min(remaining * 0.3, 100.0)
            if t6 > 20:
                a_out, _ = lp_frank_wolfe_improved(a_init, t6, solver, verbose=False)
                try_update(a_out, f"lpfw_{label}")

    # === Phase 7: Final coordinate polish on overall best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_polish(a_in, t7, verbose=True)
            try_update(a_out, f"final_coord_n{n_best}")

    # === Phase 8: One last LP-FW if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_improved(a_in, t8, solver, verbose=True)
            try_update(a_out, f"final_lpfw_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
