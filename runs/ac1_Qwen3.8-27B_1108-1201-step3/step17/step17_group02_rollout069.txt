def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.array([0.0])
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(out, 0.0)
```
The direct convolution has (a*a)_k = sum_i a_i a_{k-i}. FFT of a times FFT of a (not conjugate) gives circular convolution of a with a, which matches full convolution for zero-padded? ```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))**2 for non-negative sequences.

Strategy:
- Start from the provided parent construction when available.
- Use a fast FFT-based autocorrelation/convolution surrogate.
- Run a hybrid local search:
  * Linearized transfer optimization: moves mass from peak-contributing
    coordinates to low-sensitivity coordinates using a safe linear step.
  * Multiplicative mirror descent with smooth peak weighting.
- Perform low-dimensional parametric shape searches to escape local minima.
- Try nearby length variations and finish with a final polish phase.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

_PARAM_N = 1
_PARAM_FAMILY = "poly"


def sigmoid_scalar(z):
    z = float(z)
    if z > 50.0:
        return 1.0
    if z < -50.0:
        return 0.0
    return 1.0 / (1.0 + np.exp(-z))


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(out, 0.0)


def score_from_normalized(a):
    a = norm_sum1(a)
    if a.size == 0:
        return float("inf")
    ac = fast_autocorr(a)
    return float(2.0 * a.size * ac.max())


def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        if not np.all(np.isfinite(arr)):
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if float(arr.sum()) < 0.01:
            return float("inf")
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def interpolate_sequence(f, new_n):
    f = norm_sum1(f)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    if f.size == new_n:
        return f.copy()
    old_n = f.size
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-30:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return out / s


def accumulate_grad(grad, a, k, coeff=1.0):
    n = len(a)
    k = int(k)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    if i_start >= i_end:
        return
    d_start = k - (i_end - 1)
    d_end = k - i_start + 1
    grad[d_start:d_end] += float(coeff) * a[i_start:i_end][::-1]


def make_uniform(n):
    return np.ones(int(n), dtype=np.float64) / float(n)


def make_rational(n, p, eps=1e-12):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    p = max(0.0, float(p))
    x = np.linspace(-1.0, 1.0, n)
    a = 1.0 / (1.0 + p * x * x) + eps
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_poly(n, c1, c2, c3, eps=1e-10):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    x = np.linspace(-1.0, 1.0, n)
    x2 = x * x
    x4 = x2 * x2
    x6 = x4 * x2
    a = 1.0 + float(c1) * x2 + float(c2) * x4 + float(c3) * x6 + eps
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_inv_beta(n, p, eps=1e-12):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    p = max(0.0, float(p))
    x = np.linspace(0.5 / n, 1.0 - 0.5 / n, n)
    a = np.power(x, -p) * np.power(1.0 - x, -p) + eps
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_bimodal(n, d, w, base):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    d = float(np.clip(d, 0.03, 0.47))
    w = float(max(0.01, w))
    base = float(max(1e-6, base))
    x = np.linspace(0.0, 1.0, n)
    a = base + np.exp(-((x - d) / w) * ((x - d) / w))
    a = a + np.exp(-((x - (1.0 - d)) / w) * ((x - (1.0 - d)) / w))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def param_dim(family):
    if family == "poly":
        return 3
    if family == "bimodal":
        return 3
    return 1


def param_shape_from_raw(x):
    x = np.asarray(x, dtype=np.float64)
    n = int(_PARAM_N)
    fam = _PARAM_FAMILY

    if fam == "inv_beta":
        p = 2.0 * sigmoid_scalar(x[0])
        return make_inv_beta(n, p)

    if fam == "rational":
        p = 25.0 * sigmoid_scalar(x[0])
        return make_rational(n, p)

    if fam == "bimodal":
        d = 0.05 + 0.40 * sigmoid_scalar(x[0])
        w = 0.02 + 0.35 * sigmoid_scalar(x[1])
        base = 0.01 + 0.60 * sigmoid_scalar(x[2])
        return make_bimodal(n, d, w, base)

    # default poly
    c1 = -3.0 + 6.0 * sigmoid_scalar(x[0])
    c2 = -2.0 + 4.0 * sigmoid_scalar(x[1])
    c3 = -1.0 + 2.0 * sigmoid_scalar(x[2])
    return make_poly(n, c1, c2, c3)


def param_objective(x):
    a = param_shape_from_raw(x)
    ac = fast_autocorr(a)
    return float(2.0 * int(_PARAM_N) * ac.max())


def parametric_search(n, time_limit, verbose=False, seed=11):
    global _PARAM_N, _PARAM_FAMILY

    n = int(n)
    t0 = time.monotonic()
    time_limit = float(max(10.0, time_limit))
    rng = np.random.default_rng(seed)

    best_a = make_rational(n, 2.0)
    best_j = score_from_normalized(best_a)

    families = ["poly", "rational", "inv_beta", "bimodal"]
    fam_budget = time_limit / max(1, len(families))

    for fam in families:
        if time.monotonic() - t0 >= time_limit:
            break

        _PARAM_FAMILY = fam
        _PARAM_N = n
        dim = param_dim(fam)

        fam_best_x = None
        fam_best_f = float("inf")

        def eval_raw(x):
            nonlocal fam_best_x, fam_best_f
            x = np.asarray(x, dtype=np.float64)
            f = param_objective(x)
            if np.isfinite(f) and f < fam_best_f:
                fam_best_f = f
                fam_best_x = x.copy()
            return f

        # Random warm-start / wide sampling.
        max_random = max(8, int(fam_budget / 0.35))
        for r in range(max_random):
            if time.monotonic() - t0 >= time_limit:
                break
            x = rng.normal(0.0, 0.9, size=dim)
            eval_raw(x)

        # Bounded optimizer if available.
        if _scipy_minimize is not None and fam_best_x is not None:
            for attempt in range(2):
                if time.monotonic() - t0 >= time_limit:
                    break
                x0 = fam_best_x.copy()
                if attempt > 0:
                    x0 = x0 + rng.normal(0.0, 0.6, size=dim)
                try:
                    opts = {
                        "maxiter": 35,
                        "xatol": 1e-4,
                        "fatol": 1e-9,
                        "disp": False,
                    }
                    res = _scipy_minimize(param_objective, x0, method="Powell", options=opts)
                    if res is not None:
                        eval_raw(res.x)
                except Exception:
                    pass

        if fam_best_x is not None and np.isfinite(fam_best_f):
            a = param_shape_from_raw(fam_best_x)
            j = 2.0 * n * fast_autocorr(a).max()
            if np.isfinite(j) and j < best_j:
                best_j = float(j)
                best_a = a.copy()
        elif time.monotonic() - t0 < time_limit:
            # Fallback: a few deterministic points.
            for xvec in (np.zeros(dim), np.ones(dim), -np.ones(dim)):
                a = param_shape_from_raw(xvec)
                j = 2.0 * n * fast_autocorr(a).max()
                if np.isfinite(j) and j < best_j:
                    best_j = float(j)
                    best_a = a.copy()

        if verbose:
            print(f"      param fam={fam}: j={best_j:.9f}")

    return best_a, float(best_j)

            best = x.copy()
            stall = 0

        if M <= 1e-30:
            break

        if top_k >= ac.size:
            top_idx = np.argsort(ac)[::-1]
        else:
            top_idx = np.argpartition(ac, ac.size - top_k)[-top_k:]
            top_idx = top_idx[np.argsort(ac[top_idx])[::-1]]
        top_vals = ac[top_idx]
        Mtop = float(top_vals[0])
        if Mtop <= 1e-30:
            break

        w = np.exp(beta * (top_vals - Mtop))
        ws = float(w.sum())
        if ws > 1e-30:
            w = w / ws
        else:
            w = np.ones_like(top_vals) / top_vals.size

        grad = np.zeros(n, dtype=np.float64)
        for wi, idx in zip(w, top_idx):
            accumulate_grad(grad, x, int(idx), 2.0 * float(wi))

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        gmax = float(momentum.max())
        if gmax <= 1e-30:
            break

        eta = float(c0) / (gmax * (np.sqrt(float(t)) + 1.0))
        improved = False
        for scale in (1.0, 2.5, 5.0, 0.4):
            z = -eta * scale * momentum
            z = np.clip(z, -50.0, 0.0)
            x2 = x * np.exp(z)
            x2 = np.maximum(x2, 1e-30)
            s2 = float(x2.sum())
            if s2 <= 1e-30:
                continue
            x2 /= s2
            ac2 = fast_autocorr(x2)
            M2 = float(ac2.max())
            if M2 < best_m - 1e-15:
                x = x2
                best_m = M2
                best = x2.copy()
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            if stall >= restart:
                noise = rng.normal(0.0, 0.045, n) * best
                x2 = best + noise
                x2 = np.maximum(x2, 1e-15)
                x2 /= x2.sum()
                x = x2.copy()
                momentum = np.zeros(n, dtype=np.float64)
                stall = 0
                restart = int(min(5000, restart * 1.2))

        if verbose and t % 10000 == 0:
            print(f"        md t={t} J=2n*M={2.0 * n * best_m:.9f} stall={stall}")

    return best, float(best_m)


def hybrid_local(a0, time_limit, verbose=False, seed=1):
    best = norm_sum1(a0)
    n = int(best.size)
    best_m = float(fast_autocorr(best).max())
    working = best.copy()
    working_m = best_m
    t0 = time.monotonic()
    time_limit = float(max(10.0, time_limit))
    rng = np.random.default_rng(seed)
    state = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 8.0:
            break
        cur = min(25.0, rem * 0.97)
        if cur < 5.0:
            cur = rem

        start = working.copy()
        if state % 4 == 0:
            out, out_m = linear_transfer_optimize(
                start, cur, verbose=False, seed=seed + state,
                active_init=4, sens_count=90, pool_size=80
            )
        elif state % 4 == 1:
            out, out_m = linear_transfer_optimize(
                start, cur, verbose=False, seed=seed + 1000 + state,
                active_init=2, sens_count=50, pool_size=60
            )
        elif state % 4 == 2:
            out, out_m = md_optimize(
                start, cur, verbose=False, seed=seed + 2000 + state,
                top_k=45, beta=5.5, c0=0.9
            )
        else:
            out, out_m = md_optimize(
                start, cur, verbose=False, seed=seed + 3000 + state,
                top_k=80, beta=2.5, c0=0.5
            )

        out_m = float(out_m)
        if out_m < best_m - 1e-14:
            best_m = out_m
            best = out.copy()
            working = out.copy()
            working_m = out_m
            no_improve = 0
        else:
            no_improve += 1
            if np.isfinite(out_m) and out_m < working_m - 1e-12:
                working = out.copy()
                working_m = out_m
            elif no_improve >= 4:
                scale = 0.02 + 0.005 * min(10.0, float(no_improve))
                noise = rng.normal(0.0, scale, n) * best
                working = best + noise
                working = np.maximum(working, 1e-15)
                working /= working.sum()
                working_m = float(fast_autocorr(working).max())

        state += 1
        if verbose and state % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    hybrid t={elapsed:.0f}s J=2n*M={2.0 * n * best_m:.9f}")

    return best, float(best_m)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = float(budget_s)
    if not np.isfinite(time_budget) or time_budget <= 60.0:
        time_budget = 1800.0
    time_budget = min(time_budget - 25.0, 1775.0)
    if time_budget < 120.0:
        time_budget = 120.0
    deadline = t_global + time_budget

    best = None
    best_score = float("inf")

    # Load parent construction if present.
    try:
        parent = height_sequence_1
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 1e-12:
            p = norm_sum1(p)
            sc = evaluate_safe(p)
            if np.isfinite(sc):
                best = p.copy()
                best_score = float(sc)
                print(f"Parent loaded: n={len(p)}, official_score={best_score:.9f}")
    except Exception:
        pass

    if best is None:
        n_fb = 4200
        best = make_poly(n_fb, -1.6, 0.7, -0.25)
        best_score = evaluate_safe(best)
        if not np.isfinite(best_score):
            best_score = score_from_normalized(best)
        print(f"Fallback initialization: n={len(best)}, score={best_score:.9f}")

    # Quick hybrid polish on the initial best.
    rem = deadline - time.monotonic()
    quick = min(130.0, rem * 0.14)
    if quick > 25.0:
        print(f"\n=== Quick hybrid polish (budget {quick:.0f}s) ===")
        a, m = hybrid_local(best, quick, verbose=False, seed=1)
        a = norm_sum1(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"  Quick improved: {best_score:.9f} (n={len(a)})")
        rem = deadline - time.monotonic()

    # Parametric global search over a few lengths.
    rem = deadline - time.monotonic()
    param_total = min(360.0, rem * 0.22)
    if param_total > 80.0:
        print(f"\n=== Parametric search (budget {param_total:.0f}s) ===")
        n_cur = int(len(best))
        length_plan = [(n_cur, 0.40)]
        for cand in (n_cur + 500, n_cur - 500, 5200, 3600):
            cand = int(cand)
            if 2200 <= cand <= 7000 and not any(abs(x - cand) <= 50 for x, _ in length_plan):
                length_plan.append((cand, 0.20))

        for n_try, frac in length_plan:
            rem = deadline - time.monotonic()
            tw = min(param_total * frac, rem * 0.9)
            if tw < 25.0:
                continue
            print(f"  Parametric n={n_try}, time={tw:.0f}s")
            a, j = parametric_search(n_try, tw, verbose=True, seed=17 + n_try)
            a = norm_sum1(a)
            sc = evaluate_safe(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"    NEW BEST parametric: {best_score:.9f} (n={n_try})")
        rem = deadline - time.monotonic()

    # Main hybrid local search.
    rem = deadline - time.monotonic()
    main = min(780.0, rem * 0.42)
    if main > 60.0:
        print(f"\n=== Main hybrid local search (budget {main:.0f}s, n={len(best)}) ===")
        a, m = hybrid_local(best, main, verbose=True, seed=101)
        a = norm_sum1(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"  Main improved: {best_score:.9f}")
        rem = deadline - time.monotonic()

    # Length exploration from the best sequence.
    rem = deadline - time.monotonic()
    len_total = min(260.0, rem * 0.18)
    if len_total > 70.0:
        print(f"\n=== Length exploration (budget {len_total:.0f}s) ===")
        n_cur = int(len(best))
        cand_lengths = []
        for delta in (700, -700, 1300, 500, -350):
            nn = int(n_cur + delta)
            if 2500 <= nn <= 7000 and not any(abs(x - nn) <= 50 for x in cand_lengths):
                cand_lengths.append(nn)
        # Avoid too many heavy lengths.
        cand_lengths = cand_lengths[:2]
        for nn in cand_lengths:
            rem = deadline - time.monotonic()
            tw = min(len_total / max(1, len(cand_lengths)), rem * 0.9)
            if tw < 35.0:
                continue
            print(f"  Interpolate and polish to n={nn}, time={tw:.0f}s")
            a0 = interpolate_sequence(best, nn)
            a, m = hybrid_local(a0, tw, verbose=False, seed=555 + nn)
            a = norm_sum1(a)
            sc = evaluate_safe(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"    NEW BEST length={nn}: {best_score:.9f}")

    # Final polish.
    rem = deadline - time.monotonic()
    final = min(300.0, rem * 0.55)
    if final > 25.0:
        print(f"\n=== Final polish (budget {final:.0f}s, n={len(best)}) ===")
        a, m = hybrid_local(best, final, verbose=True, seed=909)
        a = norm_sum1(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"  Final improved: {best_score:.9f}")

    # Clean tiny values, renormalize, and validate officially.
    if best is None:
        best = make_poly(4200, -1.6, 0.7, -0.25)

    best = np.asarray(best, dtype=np.float64).copy()
    best = np.maximum(best, 0.0)
    best[best < 1e-18] = 0.0
    s = float(best.sum())
    if s <= 1e-15:
        best = make_uniform(4200)
        s = 1.0
    else:
        best = best / s

    official = evaluate_safe(best)
    if np.isfinite(official):
        best_score = float(official)

    print(f"\nFINAL: official_score={best_score:.9f}, n={len(best)}, runtime={time.monotonic()-t_global:.1f}s")

    return [float(x) for x in best]
