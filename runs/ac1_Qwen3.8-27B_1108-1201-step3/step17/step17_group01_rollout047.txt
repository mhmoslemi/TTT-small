"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgradient(a, ac, top_k=50, beta=5.0):
    """Compute weighted multi-peak subgradient vector g[i]."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # For each i in range, grad[i] += wi * (a[k-i] + a[i-k])
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        # Since i_start >= k-n+1 and i_end <= k+1, both are valid for all i in range
        # But we need to be careful: a[k-i] and a[i-k] might be the same element
        # when i = k-i, i.e., 2i = k
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def mirror_descent(a0, time_limit, top_k=80, beta=3.0, c0=5.0, 
                   restart_every=5000, spectral_smooth=False, smooth_alpha=0.1,
                   verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with multi-peak subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0
    
    while True:
        if time.monotonic() - t0 > time_limit:
            break
        
        t = restart_count * restart_every + int((time.monotonic() - t0) * 100)  # approximate iteration count
        # Actually, let me use a simple counter
        break
    
    # Use a proper iteration counter
    t_iter = 0
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_nest = (t_iter - 1.0) / (t_iter + 1.5) if t_iter > 1 else 0.0
        y = x + beta_nest * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)
        
        eta = gamma / (np.sqrt(t_iter) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Optional: spectral smoothing
        if spectral_smooth and t_iter % 10 == 0:
            fft_size = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a_new, fft_size)
            P = np.abs(A) ** 2
            # Smooth the power spectrum
            P_smooth = np.convolve(P, np.ones(5) / 5, mode='same')
            P_mix = (1 - smooth_alpha) * P + smooth_alpha * P_smooth
            P_mix = np.maximum(P_mix, 1e-30)
            mag_new = np.sqrt(P_mix)
            phase = np.angle(A)
            A_mix = mag_new * np.exp(1j * phase)
            a_smooth = np.real(np.fft.irfft(A_mix, fft_size)[:n])
            a_smooth = np.maximum(a_smooth, 0.0)
            ss = a_smooth.sum()
            if ss > 1e-15:
                a_smooth = a_smooth / ss
                a_new = 0.7 * a_new + 0.3 * a_smooth
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M = fast_autocorr(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 15, 200)
            cur_beta = max(cur_beta - 0.5, 1.5)
            gamma = c0 * (1.0 + 0.15 * min(restart_count, 5))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart #{restart_count} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")
        
        if verbose and t_iter % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={t_iter} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    
    return best_a, best_m

def spectral_flatten(a0, time_limit, iters=5000, eta0=0.4, verbose=False):
    """Iterative spectrum flattening to reduce convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        P_min = P[P > 1e-30].min() if P[P > 1e-30].size > 0 else 1e-30
        P_mean = P.mean()
        
        if P_mean < 1e-30:
            break
        
        # Soft compression of dynamic range
        eta = eta0 * (1.0 + 0.5 * it / max(1, iters))
        # Compress: P_new = P_mean * (P/P_mean)^(1-eta)
        ratio = P / max(P_mean, 1e-30)
        P_target = P_mean * np.power(np.maximum(ratio, 1e-30), 1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)
        # Preserve DC component
        P_target[0] = P[0]
        
        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.real(np.fft.irfft(A_new, fft_size)[:n])
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.6 * a + 0.4 * a_new
            a = norm_sum1(a)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecFlatten it={it} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=False, rng_seed=42):
    """Targeted coordinate descent to shave the highest peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Try to reduce the top few peaks
        top_k = np.argsort(ac)[::-1][:min(10, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find the source position that most contributes to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving some mass from src to a random position
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                
                # Try several random destinations
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, top_K=300, verbose=False):

                    solver = s
                    break

        return best_a, best_m
    
    y = a.copy()
