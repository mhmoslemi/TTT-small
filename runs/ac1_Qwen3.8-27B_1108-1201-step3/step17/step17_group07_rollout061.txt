    # cross[j] = sum_m w[m] * a[m-j] (with a padded to 0)
    # = sum_{m=j}^{j+n-1} w[m] * a[m-j] (since a[m-j] = 0 outside [0,n-1])
    # = sum_{k=0}^{n-1} w[k+j] * a[k]
    # This is exactly what we want!
    
    grad = 4.0 * n * cross
    return grad, b, w

def nesterov_fft_mirror_descent(a0, time_limit, tau_start=10.0, tau_end=300.0, 
                                  anneal_rate=0.9999, verbose=False):
    """
    Nesterov-accelerated mirror descent on the simplex using full FFT-based gradient.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    tau = tau_start
    stall_count = 0
    max_stall = 5000
    rng = np.random.default_rng(42 + n)
    c = 2.0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute full gradient
        grad, b, w = compute_full_gradient(y, tau)
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Evaluate
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Anneal temperature
        tau = min(tau * anneal_rate, tau_end)
        
        # Restart if stalled
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(1000, int(max_stall * 0.7))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.03, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//5, 500), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            else:
                # Asymmetric perturbation (important since optimum is asymmetric)
                shift = rng.uniform(-0.05, 0.05)
                x_r = np.roll(best_a, int(round(shift * n)))
                x_r = np.maximum(x_r, 0)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.7)
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD-FFT t={t} ({elapsed:.1f}s): J={J:.8f}, tau={tau:.1f}, stall={stall_count}")
    
    return best_a, best_m

def remez_lp_enhanced(a, max_cons=200, r=0.05, solver=None):
    """
    Enhanced LP Remez step with larger active set.
    Linearizes b_k(a') ≈ b_k(a) + 2*sum_j a'_{j} * a_{k-j} for active k.
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints (all lags within 1% of max)
    threshold = max_b * 0.995
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        # Keep the largest ones
        active_sorted = active[np.argsort(b[active])[::-1]]
        active = active_sorted[:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # Build LP
    d = cp.Variable(n)  # perturbations
    M = cp.Variable()
    
    constraints = [cp.sum(d) == 0]  # maintain sum = 1
    constraints.append(d >= -a)     # non-negativity: a + d >= 0
    constraints.append(d <= r)      # upper trust region
    
    # Linearized active constraints
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            # b_k(a+d) ≈ b_k(a) + 2 * sum_{j in idx} d_j * a[k-j]
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, d[idx])) + b[k_int] <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    d_val = np.asarray(d.value, dtype=np.float64)
    a_new = a + d_val
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def pairwise_cd(a0, time_limit, verbose=False, rng_seed=123):
    """Pairwise coordinate descent with optimal step along each pair."""

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.4) ** 2 / 0.15)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'skewed':
        a = np.exp(-3.0 * x)  # skewed to left
    elif kind == 'bimodal':
        a = np.exp(-20*(x-0.3)**2) + 0.5*np.exp(-20*(x-0.7)**2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Compute linear convolution a*a via FFT in O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def compute_full_gradient(a, tau):
    """
    Compute full gradient of log-sum-exp smoothed objective via FFT.
    Returns (gradient, b, weights) where:
      gradient_j = 4n * sum_k w_k * a_{k-j} for valid indices
      b = a * a (convolution)
      w = softmax(tau * b)
    """
    n = len(a)
    b = fast_conv(a)
    
    # Softmax weights with numerical stability
    tau_b = tau * b
    tau_b -= tau_b.max()
    w = np.exp(tau_b)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(b) / len(b)
    else:
        w = w / w_sum
    
    # Cross-correlation via FFT: cross[j] = sum_m w[m] * a[m-j]
    # Need fft_size >= (len(w) + len(a) - 1) = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    
    grad = 4.0 * n * cross
    return grad, b, w

    try:
        prob.solve(solver=solver, verbose=False)

        return a.copy()

    a_new = a + d_val
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    """Compute linear convolution a*a via FFT in O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Clamp to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_full_gradient(a, tau):
    """
    Compute full gradient of log-sum-exp smoothed objective via FFT.
    Returns (gradient, b, weights) where:
      gradient_j = 4n * sum_k w_k * a_{k-j} for valid indices
      b = a * a (convolution)
      w = softmax(tau * b)
    """
    n = len(a)
    b = fast_conv(a)
    
    # Softmax weights with numerical stability
    tau_b = tau * b
    tau_b -= tau_b.max()
    w = np.exp(tau_b)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(b) / len(b)
    else:
        w = w / w_sum
    
    # Cross-correlation via FFT: cross[j] = sum_m w[m] * a[m-j]
    # Need fft_size >= (len(w) + len(a) - 1) = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    
    grad = 4.0 * n * cross
    return grad, b, w
