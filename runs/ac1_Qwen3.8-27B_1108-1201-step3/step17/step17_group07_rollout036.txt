for better convergence, and a more aggressive cutting-plane Remez that verifies
all quadratic constraints after each LP solve.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smoothed_gradient(conv_vals, a, tau):
    """Compute gradient of (2n/tau)*logsumexp(conv/tau) w.r.t. a (sum a = 1)."""
    n = len(a)
    cmax = conv_vals.max()
    log_w = (conv_vals - cmax) / tau
    max_log_w = log_w.max()
    exp_log_w = np.exp(log_w - max_log_w)
    w = exp_log_w / exp_log_w.sum()
    
    active = np.where(w > 1e-10)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        g[idx] += w[k_int] * 2.0 * a[k_int - idx]
    return g

def smoothed_nmd(a0, time_limit, tau_init=None, verbose=False):
    """Smoothed mirror descent with log-sum-exp and adaptive tau."""
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_conv(a)
    M0 = conv.max()
    if tau_init is None:
        tau = 0.05 * M0
    else:
        tau = tau_init
    best_a = a.copy()
    best_m = M0
    t0 = time.monotonic()
    stall = 0
    max_stall = 5000
    rng = np.random.default_rng(777 + n)
    
    tau_min = 0.01 * M0
    tau_decay = 0.9997
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        if M < best_m - 1e-18:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        g = smoothed_gradient(conv, a, tau)
        g_max = np.abs(g).max()
        if g_max < 1e-30:
            break
        g = g / g_max
        
        # Try multiple step sizes (log-spaced)
        best_trial_m = M
        best_trial = a
        for eta in [0.05, 0.1, 0.2, 0.4, 0.8, 1.5, 3.0]:
            a_trial = a * np.exp(-eta * g)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial / a_trial.sum()
            conv_t = fast_conv(a_trial)
            m_t = conv_t.max()
            if m_t < best_trial_m:
                best_trial_m = m_t
                best_trial = a_trial
            if m_t < M - 1e-6:
                break
        
        a = best_trial
        
        # Anneal tau
        tau = max(tau_min, tau * tau_decay)
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = max(1000, int(max_stall * 0.6))
            strategy = int(rng.integers(5))
            if strategy == 0:
                a_r = best_a * (1 + rng.normal(0, 0.03, n))
            elif strategy == 1:
                a_r = best_a.copy()
                idx = rng.choice(n, size=n // 5, replace=False)
                a_r[idx] *= rng.uniform(0.3, 2.0, len(idx))
            elif strategy == 2:
                # Smooth restart: mix with a flat sequence
                mix = 0.1 + 0.2 * rng.uniform()
                a_r = (1 - mix) * best_a + mix * np.ones(n) / n
            elif strategy == 3:
                # Perturb active region
                conv_best = fast_conv(best_a)
                M_best = conv_best.max()
                active_idx = np.where(conv_best >= 0.95 * M_best)[0]
                # Map active lags to positions in a
                a_r = best_a * (1 + rng.normal(0, 0.05, n))
            else:
                tau = max(tau_min, tau * 5)  # Increase smoothing temporarily
                a_r = best_a.copy()
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.1f}s): J={J:.8f}, tau={tau:.3e}, stall={stall}")
    
    return best_a, best_m

def lp_remez_enhanced(a0, time_limit, solver=None, verbose=False):
    """Enhanced LP Remez with full constraint verification."""
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_conv(a)
    best_m = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    r = 0.08
    no_improve = 0
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Active set: all lags within 99.5% of max
        threshold = M * 0.995
        active = np.where(conv >= threshold)[0]
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]
        
        # Build LP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, a_var <= a * (1.0 + r), a_var >= a * max(0, 1.0 - r)]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            # Linearized: 2*sum a_curr[k-j]*a[j] <= M + conv_curr[k]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var + conv[k_int])
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            r = max(0.01, r * 0.5)
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(t)
                a_pert = a * (1 + rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                a = a_pert / a_pert.sum()
                no_improve = 0
                r = 0.08
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            r = max(0.01, r * 0.5)
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(t)
                a_pert = a * (1 + rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                a = a_pert / a_pert.sum()
                no_improve = 0
                r = 0.08
            continue
        
        a_lp = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_lp = a_lp.sum()
        if s_lp < 1e-15:
            r = max(0.01, r * 0.5)
            no_improve += 1
            continue
        a_lp = a_lp / s_lp
        
        # Line search
        accepted = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1 - frac) * a + frac * a_lp
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            conv_t = fast_conv(a_trial)
            m_t = conv_t.max()
            if m_t < best_m - 1e-18:
                a = a_trial.copy()
                best_m = m_t
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.3, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.7)
            if no_improve > 25:
                rng = np.random.default_rng(t + 1)
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = 0.1
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, active={len(active)}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent with random pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    conv = fast_conv(a)
    best_m = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                conv_t = fast_conv(a_trial)
                m_t = conv_t.max()
                if m_t < best_m - 1e-16:
                    a = a_trial
                    best_m = m_t
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if found:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        t = 2.0 * x - 1.0
        a = np.maximum(1.0 - t ** 4, 0.0)
    elif kind == 'barnes':
        # Two-hump
        a = np.exp(-20.0 * (x - 0.3) ** 2) + np.exp(-20.0 * (x - 0.7) ** 2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multiple phases."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'raised_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Smoothed NMD from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Smoothed NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, _ = smoothed_nmd(norm_sum1(best_seq), t1, verbose=True)
        try_update(a_out, f"SMD_parent_n{n0}")
    
    # === Phase 2: LP Remez on best (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 2: LP Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = lp_remez_enhanced(norm_sum1(best_seq), t2, solver=solver, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
    
    # === Phase 3: Smoothed NMD polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: SMD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        conv_best = fast_conv(best_seq)
        tau_small = 0.02 * conv_best.max()
        a_out, _ = smoothed_nmd(norm_sum1(best_seq), t3, tau_init=tau_small, verbose=True)
        try_update(a_out, f"SMD_polish_n{len(best_seq)}")
    
    # === Phase 4: LP Remez polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Remez polish (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_remez_enhanced(norm_sum1(best_seq), t4, solver=solver, verbose=True)
        try_update(a_out, f"Remez_polish_{len(best_seq)}")
    
    # === Phase 5: CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 5: CD polish (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t5, rng_seed=555)
        try_update(a_out, f"CD_{len(best_seq)}")
    
    # === Phase 6: Multi-init exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6_total = min(remaining * 0.10, 150.0)
    if t6_total > 60:
        print(f"\n=== Phase 6: Multi-init (budget={t6_total:.0f}s) ===")
        inits = ['barnes', 'gaussian', 'quartic']
        t_per = t6_total / len(inits)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = make_init(n0, kind)
            # Mix with current best for warm start
            a_mixed = 0.5 * a_init + 0.5 * norm_sum1(best_seq)
            a_mixed = np.maximum(a_mixed, 0)
            a_mixed = a_mixed / a_mixed.sum()
            a_out, _ = smoothed_nmd(a_mixed, min(t_per, remaining - 5), verbose=False)
            try_update(a_out, f"SMD_{kind}_n{n0}")
    
    # === Phase 7: Try different length (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        # Try a longer sequence
        n_alt = n0 + 400
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        
        t7a = min(remaining * 0.4, 200.0)
        if t7a > 30
