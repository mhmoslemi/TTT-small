def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_smoothed_gradient(a, beta):
    n = len(a)
    c = fast_autocorr(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    a_rev = a[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A_rev * P, fft_size)[:conv_len]
    grad = (2.0 / beta) * conv_result[n - 1:n - 1 + n]
    return grad, c, p


def multiplicative_weight_optimize(a0, time_limit, beta_schedule, lr, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    x_prev = a.copy()
    total_iters = 0
    stall_count = 0
    max_stall = 5000

    n_betas = len(beta_schedule)
    iters_per_beta = max(200, int(50000.0 * time_limit / (n_betas * n)))

    for bi in range(n_betas):
        beta = beta_schedule[bi]
        beta_iters = min(iters_per_beta, max(100, int(time_limit * 300 / (n_betas * n))))
        for t in range(1, beta_iters + 1):
            if time.monotonic() - t0 > time_limit:
                break
            beta_t = (t - 1.0) / (t + 1.5)
            y = a + beta_t * (a - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            y = y / sy

            grad, c, p = compute_smoothed_gradient(y, beta)
            grad_inf = np.abs(grad).max()
            if grad_inf < 1e-15:
                break
            eta = lr / (grad_inf + 1e-15)
            log_update = -eta * grad
            log_update_max = log_update.max()
            x_new = y * np.exp(log_update - log_update_max)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new / sx

            x_prev = a
            a = x_new
            total_iters += 1

            M_new = fast_autocorr(a).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

            if stall_count > max_stall:
                stall_count = 0
                pert = rng.normal(0, 0.05, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                a = a_r
                x_prev = a_r.copy()

            if verbose and t % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MW iter={total_iters} beta={beta:.1f} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_condensed_fw(a0, time_limit, solver, initial_K=200, max_K=2000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0
    rng = np.random.default_rng(7)

    ac_y = fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:25]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 15 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed, verbose=True):

                if s_t < 1e-15:

                a_trial = a_trial / s_t

                    best_m = M_trial

                    stall = 0
                    break

            if not found and stall > 60:

                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt

                    M_new = fast_autocorr(a).max()

def parametric_init(n, best_init=None):

        candidates.append((i_arr + 1.0) ** p)
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        candidates.append((n - i_arr + 1.0) ** p)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            candidates.append(np.maximum(i_arr + 1.0, 0.5) ** alpha *
                              np.maximum(n - i_arr, 0.5) ** beta)
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
            candidates.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            candidates.append(np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01))
    for rate in [0.5, 1, 2, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    if best_init is not None:
        for w in [0.2, 0.5, 0.8]:
            a_pow = (i_arr + 1.0) ** 1.0
            a_blend = w * best_init / best_init.sum() + (1.0 - w) * a_pow / a_pow.sum()
            candidates.append(a_blend)

    best_c = None
    best_val = float('inf')
    for c in candidates:
        c = norm_sum1(c)
        m = fast_autocorr(c).max()
        if m < best_val:
            best_val = m
            best_c = c.copy()
    return best_c, 2.0 * n * best_val


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 3000
        best_seq, best_score_est = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Parametric search for better init
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_param, val_param = parametric_init(n0, best_seq)
        check_and_update(a_param, "parametric")

    n_best = len(best_seq)

    # Phase 2: Smoothed multiplicative weight (main workhorse)
    print(f"\n=== Phase 2: Smoothed MW optimization (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 700.0)
    if t2 > 30:
        beta_schedule = [10.0, 30.0, 60.0, 100.0, 200.0, 400.0, 800.0, 1500.0]
        a_in = norm_sum1(best_seq)
        a_out, m_out = multiplicative_weight_optimize(a_in, t2, beta_schedule, lr=1.0,
                                                      verbose=True)
        check_and_update(a_out, f"mw_n{n_best}")

    # Phase 3: Second MW pass with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MW pass 2 (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        beta_schedule2 = [50.0, 150.0, 300.0, 600.0, 1200.0, 2500.0]
        a_in = norm_sum1(best_seq)
        a_out, m_out = multiplicative_weight_optimize(a_in, t3, beta_schedule2, lr=0.7,
                                                      verbose=True)
        check_and_update(a_out, f"mw2_n{n_best}")

    # Phase 4: LP-Frank-Wolfe polishing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-CW polishing (n={n_best}) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_condensed_fw(a_in, t4, solver, initial_K=250, max_K=2500,
                                   verbose=True)
        check_and_update(a_out, f"lp_n{n_best}")

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, rng_seed=42, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # Phase 6: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    lengths_to_try = [n_best - 200, n_best + 200, n_best + 500, n_best + 1000]
    lengths_to_try = [l for l in lengths_to_try if 1000 <= l <= 7000]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Length {new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        sc = evaluate_safe(a_long)
        a_in = norm_sum1(a_long)

        t6m = min(remaining * 0.25, 150.0)
        if t6m > 20:
            beta_schedule6 = [50.0, 150.0, 400.0, 1000.0]
            a_out, _ = multiplicative_weight_optimize(a_in, t6m, beta_schedule6, lr=0.8,
                                                      verbose=False)
            check_and_update(a_out, f"mw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5500:
            t6lp = min(remaining * 0.30, 120.0)
            if t6lp > 20:
                a_out, _ = lp_condensed_fw(a_in, t6lp, solver, initial_K=200,
                                           max_K=1500, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")

    # Phase 7: Final LP polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 6000:
        t7 = min(remaining * 0.50, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_condensed_fw(a_in, t7, solver, initial_K=350, max_K=3000,
                                       verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.60, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, rng_seed=999, verbose=True)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        a_fb, _ = parametric_init(n_fb)
        best_seq = a_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
