def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_initial(n, family='sine'):
    x = np.linspace(0, 1, n)
    if family == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif family == 'cosine':
        a = np.cos(np.pi * x) ** 2 + 0.1
    elif family == 'uniform':

    elif family == 'beta':

def lp_frank_wolfe(a0, time_limit, solver, top_K=200, margin_frac=1e-3,
                    verbose=False, use_backtrack=True, line_search_max=8):
    """LP Frank-Wolfe with adaptive margins and backtracking."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    # Precompute some useful things

        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < 1e-16:
            break
        
        # Select active constraints: lags where ac[k] is close to M_curr
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        
        if n_active > top_K:
            # Keep only the top_K most violated
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < 10:
            # If too few active, take top 10
            active_idx = np.argsort(margins)[:max(10, top_K // 4)]
        else:
            active_idx = np.where(active_mask)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix T: T[k, i] = y[k - i] for valid i
        # This is the cross-correlation kernel
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                i_idx = np.arange(i_start, i_end)
                T_sel[row, i_idx] = y[k_int - i_idx]
        
        # Solve LP: maximize sum(g) subject to T_sel @ g <= M_curr, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ g_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Frank-Wolfe update with backtracking
        if use_backtrack:
            best_alpha = 2.0 / (t + 2.0)
            best_m_trial = M_curr
            for alpha in [2.0/(t+2), 2.0*(2.0/(t+2)), 0.5*2.0/(t+2)]:
                y_trial = (1 - alpha) * y + alpha * g_dir
                y_trial = np.maximum(y_trial, 0.0)
                s = y_trial.sum()
                if s < 1e-15:
                    continue
                y_trial = y_trial / s
                m_trial = fast_autocorr(y_trial).max()
                if m_trial < best_m_trial:
                    best_m_trial = m_trial
                    best_alpha = alpha
            
            y_new = (1 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s = y_new.sum()
            if s < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s = y_new.sum()
            if s < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s
        
        y = y_new
        m_new = M_curr  # approximate, will be exact next iteration
        
        if m_new < best_m:
            best_m = m_new
            best_a = y.copy()
            # Tighten margin on improvement
            margin_frac = max(margin_frac * 0.85, 1e-5)
        
        if verbose and t % 10 == 0:

def coordinate_refine(a0, time_limit, verbose=False):
    """Coordinate descent: move mass from peaks to valleys."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find contributing pairs
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total = contrib.sum()
        
        if total < 1e-20:
            stall += 1
            continue
        
        # Try moving mass from high-contributing positions
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        
        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            best_reduction = 0
            best_dst = -1
            best_frac = 0
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try multiple destinations
                for trial in range(20):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    reduction = best_m - M_trial
                    if reduction > best_reduction:
                        best_reduction = reduction
                        best_dst = dst
                        best_frac = frac
            
            if best_dst >= 0 and best_reduction > 0:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max

            w = w / w_sum

            if i_start >= i_end:

            i_idx = np.arange(i_start, i_end)
            j_idx = k_int - i_idx
            grad[i_idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        base_m = fast_autocorr(y).max()

            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:

            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        
        if not accepted:
            x_prev = x
            x = y.copy()
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
    
    return best_a, best_m

def interpolate_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
