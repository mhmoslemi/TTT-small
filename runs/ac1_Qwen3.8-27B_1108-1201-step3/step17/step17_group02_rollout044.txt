The gradient w.r.t. z:
d(a_i)/d(z_j) = a_i * (delta(i,j) - a_j)

So d(objective)/d(z_j) = sum_i d(objective)/d(a_i) * a_i * (delta(i,j) - a_j)
= a_j * (d(objective)/d(a_j) - sum_i d(objective)/d(a_i) * a_i)

Let g = gradient of objective w.r.t. a (computed as above)
Then d(objective)/d(z_j) = a_j * (g_j - sum_i g_i * a_i)

This is the gradient of the objective w.r.t. the softmax parameter z.

Let me implement this with L-BFGS-B. For n=4200, each gradient evaluation takes O(n log n) for the FFT + O(n) for the rest. L-BFGS-B might need 200-1000 iterations. Total time: maybe 10-60 seconds. That's feasible.

def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax(z):
    z_max = z.max()
    ez = np.exp(z - z_max)
    return ez / ez.sum()

def lse_objective_and_grad(z, T):
    """Compute log-sum-exp objective and gradient w.r.t. z."""
    a = softmax(z)
    n = len(a)
    ac = fast_autocorr(a)
    # Log-sum-exp of T * ac
    T_ac = T * ac
    max_val = T_ac.max()
    log_sum = max_val + np.log(np.sum(np.exp(T_ac - max_val)))
    obj = log_sum / T  # This is (1/T)*log(sum exp(T*ac)) ≈ max(ac)
    
    # Weights w_k = exp(T*ac[k]) / sum_l exp(T*ac[l])
    w = np.exp(T_ac - max_val)
    w = w / w.sum()
    
    # Gradient of obj w.r.t. a: grad_a[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
    # This is 2 * cross-correlation of w and a
    # Using FFT:
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # cross-correlation: corr[j] = sum_k w[k] * a[k-j]
    # = ifft(conj(fft(w)) * fft(a))[:n] ... need to verify
    # Actually: ifft(conj(W) * A)[j] = sum_k w[k] * conj(a)[(k-j) mod N]
    # For real a with zero padding, this gives sum_k w[k] * a[k-j] (with wrap-around)
    # But with enough zero padding, wrap-around doesn't affect the first n elements
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    grad_a = 2.0 * corr[:n]
    
    # Gradient w.r.t. z: grad_z[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    ga_dot_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - ga_dot_a)
    
    return obj, grad_z

def scipy_smooth_optimize(a0, time_limit, T_start=10.0, T_end=500.0, verbose=True):
    """Optimize using scipy L-BFGS-B with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_sum1(a0)
    # Initial z: log(a) (since softmax(log(a)) ∝ a, normalized)
    z0 = np.log(np.maximum(a, 1e-30))
    
    best_score = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Try multiple temperatures
    T_values = [T_start, T_start*3, T_start*9, T_end]
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        t_rem = time_limit - (time.monotonic() - t0)
        if t_rem < 10:
            break
        
        def obj_grad(z):
            obj, grad = lse_objective_and_grad(z, T)
            return obj, grad
        
        try:
            result = scipy_minimize(obj_grad, z0, jac=True, method='L-BFGS-B',
                                    options={'maxiter': 500, 'maxfun': 2000,
                                             'ftol': 1e-12, 'gtol': 1e-8,
                                             'maxls': 50})
            if result.success or result.fun < best_score:
                a_new = softmax(result.x)
                ac = fast_autocorr(a_new)
                J = 2.0 * n * ac.max()
                if J < best_score:
                    best_score = J
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    LSE T={T}: J={J:.6f} (iters={result.nit})")
        except Exception as e:
            if verbose:
                print(f"    LSE T={T} failed: {e}")
        
        # Update z0 for next temperature
        z0 = result.x.copy()
    
    return best_a, best_score / (2.0 * n)  # Return normalized max autocorr


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def mirror_descent_smooth(a0, time_limit, verbose=True, rng_seed=42):
    """Mirror descent with smooth log-sum-exp subgradient and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 5.0
    stall = 0
    
    T = 20.0  # Temperature
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute smooth subgradient
        ac = fast_autocorr(y)
        T_ac = T * ac
        max_val = T_ac.max()
        w = np.exp(T_ac - max_val)
        w = w / w.sum()
        
        # grad_a[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
        fft_size = 1 << (2 * n - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(y, fft_size)
        corr = np.fft.irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * corr[:n]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive temperature
        if stall > 3000:
            T = min(T * 1.5, 200.0)
        
        # Restart
        if stall > 8000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            T = max(T / 2, 10.0)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, T={T:.1f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent: try moving mass from high-peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    n_top = 20
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Compute sensitivity
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]
            
            if dst == src:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(1, min(5, n))
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
    
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
