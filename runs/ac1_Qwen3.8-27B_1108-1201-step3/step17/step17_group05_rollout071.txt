Minimizes  2*n*max(conv(a,a)) / (sum(a))^2  for a non-negative step function a.

All internal optimization is done on the simplex sum(a)=1 (clipping keeps the
evaluate_sequence objective identical, since it equals 2*n*max(conv(a,a)) there).

Multi-phase strategy:
  1. Fast accelerated projected gradient (FISTA) with the EXACT subgradient of
     max_m conv(a,a)[m]  (g_k = a_{m*-k} + a_{k-m*}), O(n log n) per step.
  2. CuttING-plane LP Frank-Wolfe: each outer step solves a SMALL LP (only the

  4. Symmetry averaging (a + a[::-1])/2 as a free convex improvement, length

Key improvement over the previous state of the art: the exact-direction

_GLOBAL_SOLVER = None


def get_solver():
    """Pick the first working LP solver (fast conic preferred)."""
    global _GLOBAL_SOLVER
    if _GLOBAL_SOLVER is not None:
        return _GLOBAL_SOLVER
    order = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.PDLP, cp.SCS, cp.ECOS_PP]
    for s in order:
        try:
            x = cp.Variable(12, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _GLOBAL_SOLVER = s
                print("  LP solver: %s" % s)
                return _GLOBAL_SOLVER
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _GLOBAL_SOLVER


def fast_autocorr(a):
    """conv(a, a) (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_from_ac(ac, n):
    """Return 2*n*max(ac) for a sum-1 sequence (matches evaluate_sequence)."""
    return 2.0 * n * float(np.max(ac))


def compute_subgrad(a, m_star):
    """Exact subgradient of max_m conv(a,a)[m] at the maximizer m_star."""
    n = len(a)
    g = np.zeros(n)
    k_lo = max(0, m_star - n + 1)
    k_hi = min(n - 1, m_star)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        g[ks] += a[m_star - ks]
    k_lo2 = max(0, m_star)
    k_hi2 = min(n - 1, m_star + n - 1)
    if k_lo2 <= k_hi2:
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        g[ks2] += a[ks2 - m_star]
    return g


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum x = 1} (order-statistic method)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1).astype(np.float64)
    cond = u - (cssv - 1.0) / rho > 0.0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho_hat = np.max(rho[cond])
    theta = (cssv[int(rho_hat) - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """(a + a[::-1])/2 projected back onto the simplex (free convex improvement)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.full_like(a, 1.0 / len(a))
    return a_sym / s


def make_smooth_init(n, seed=0, amp=0.06, maxfreq=10):
    """Smooth positive initialisation via a low-frequency Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.full(n, 0.5)
    for f in range(1, maxfreq + 1):
        a += rng.normal(0.0, amp) * np.sin(2.0 * np.pi * f * x + rng.uniform(0.0, 2.0 * np.pi))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return a / s


def interpolate_sequence(f, new_n):
    """Linear interpolation of a sum-1 sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return np.full(new_n, 1.0 / new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, fn)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.full(new_n, 1.0 / new_n)
    return out / ss


def fista_optimize(a0, time_limit, L=4.0, verbose=False, restart_interval=4000,
                   seed=42, sym=True):
    """Accelerated projected gradient with the exact subgradient."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_J = objective_from_ac(fast_autocorr(a), n)
    t0 = time.monotonic()

    y = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    cur_L = L
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        ac_y = fast_autocorr(y)
        m_star = int(np.argmax(ac_y))
        g = compute_subgrad(y, m_star)

        eta = 1.0 / cur_L
        z = y - eta * g
        x_new = project_simplex(z)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new
        y = x_new + beta * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = x_new.copy()
        t_k = t_new
        x_prev = x_new

        cand = x_new
        J = objective_from_ac(fast_autocorr(cand), n)
        if sym and (it % 3 == 0):
            cs = symmetrize(cand)
            Js = objective_from_ac(fast_autocorr(cs), n)
            if Js < J:
                J = Js
                cand = cs
        if J < best_J - 1e-15:
            best_J = J
            best_a = cand.copy()
            stall = 0
        else:
            stall += 1
            if stall % 400 == 0:
                cur_L *= 1.1  # conservative response to stalling

        if stall >= restart_interval:
            stall = 0
            cur_L = L
            t_k = 1.0
            pert = rng.normal(0.0, 0.03, n) * best_a
            xr = best_a + pert
            xr = project_simplex(np.maximum(xr, 0.0))
            if sym and rng.random() < 0.5:
                xr = symmetrize(xr)
            y = xr.copy()
            x_prev = xr.copy()
            if verbose:
                print("    FISTA restart it=%d (%.0fs): J=%.6f" % (it, time.monotonic() - t0, best_J))

        if verbose and it % 20000 == 0:
            print("    FISTA it=%d (%.0fs): J=%.6f stall=%d" % (it, time.monotonic() - t0, best_J, stall))

    return best_a, best_J

        it += 1
        ac_a = fast_autocorr(a)
        m_star = int(np.argmax(ac_a))

        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a.copy()
        a = a_new

        cand = a
        J = objective_from_ac(fast_autocorr(cand), n)
        if sym and (it % 3 == 0):
            cs = symmetrize(cand)
            Js = objective_from_ac(fast_autocorr(cs), n)
            if Js < J:
                J = Js
                cand = cs
        if J < best_J - 1e-15:
            best_J = J
            best_a = cand.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            stall = 0
            eta *= 0.85
            if rng.random() < 0.3:
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = project_simplex(np.maximum(best_a + pert, 0.0))

        if verbose and it % 30000 == 0:
            print("    FGM it=%d (%.0fs): J=%.6f eta=%.4f" % (it, time.monotonic() - t0, best_J, eta))

    return best_a, best_J


def build_toeplitz_subset(a, lags):
    """T[i, j] = a[lags[i]-j] when 0<=lags[i]-j<n else 0 (vectorised gather)."""
    n = len(a)
    lags_arr = np.asarray(lags, dtype=np.int64)
    j_range = np.arange(n)
    idx = lags_arr[:, None] - j_range[None, :]
    valid = (idx >= 0) & (idx < n)
    idxc = np.clip(idx, 0, n - 1)
    return np.where(valid, a[idxc], 0.0)


def fw_vertex(a, M, lags, solver):
    """Solve the linearised LP over the lag-set `lags`; returns vertex b or None."""
    n = len(a)
    K = len(lags)
    T = build_toeplitz_subset(a, lags)
    b_var = cp.Variable(n, nonneg=True)
    cons = [T @ b_var <= M * np.ones(K), cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    bs = b.sum()
    if bs < 1e-15:
        return None
    return b / bs


def lp_cutting_plane(a0, time_limit, solver, verbose=True, init_K=200,
                     max_K=2600, line_search=True):
    """Frank-Wolfe with a cutting-plane linearised LP (exact direction) + line search."""
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_J = objective_from_ac(fast_autocorr(a), n)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        ac = fast_autocorr(a)
        M = float(np.max(ac))
        if M < 1e-15:
            break

        # start from the most binding lags
        K0 = min(init_K, 2 * n - 1)
        lags = np.argsort(ac)[::-1][:K0].astype(np.int64)

        # saturate: keep adding the worst violated lag until the LP vertex
        # satisfies every lag  ->  vertex equals the exact FW direction.
        b = None
        for _inner in range(8):
            if time.monotonic() - t0 > time_limit:
                break
            b = fw_vertex(a, M, lags, solver)
            if b is None:
                break
            acb = fast_autocorr(b)
            viol = float(np.max(acb)) - M
            if viol <= 1e-9:
                break  # feasible for all lags -> exact direction
            # add worst violators
            room = max(0, max_K - len(lags))
            if room == 0:
                break
            add = np.argsort(acb)[::-1][:max(1, room // 2)].astype(np.int64)
            lags = np.unique(np.concatenate([lags, add]))
            lags = lags[:max_K]

        if b is None:
            continue

        # Frank-Wolfe step with line search (objective evals are cheap FFTs)
        if line_search:
            alphas = [2.0 / (it + 2.0), 0.5, 0.33, 0.2, 0.7, 0.9]
            alphas = [min(1.0, max(0.0, x)) for x in alphas]
            best_alpha = alphas[0]
            best_step_J = objective_from_ac(fast_autocorr(a), n)
            for al in alphas:
                cand = (1.0 - al) * a + al * b
                cand = project_simplex(cand)
                Jc = objective_from_ac(fast_autocorr(cand), n)
                if Jc < best_step_J:
                    best_step_J = Jc
                    best_alpha = al
            a = (1.0 - best_alpha) * a + best_alpha * b
            a = project_simplex(a)
        else:
            al = 2.0 / (it + 2.0)
            a = project_simplex((1.0 - al) * a + al * b)

        J = objective_from_ac(fast_autocorr(a), n)
        cand = a
        cs = symmetrize(a)
        Js = objective_from_ac(fast_autocorr(cs), n)
        if Js < J:
            J = Js
            cand = cs
        if J < best_J:
            best_J = J
            best_a = cand.copy()

        gap = M - float(np.max(fast_autocorr(b)))
        if verbose and it % 5 == 0:
            print("    LP-FW it=%d (%.0fs): J=%.6f K=%d gap=%.2e"
                  % (it, time.monotonic() - t0, best_J, len(lags), max(0.0, gap)))
        if gap <= 1e-11 * max(1.0, M) and it > 3:
            break

    return best_a, best_J


def safe_score(a):
    """Call the external evaluate_sequence defensively."""
    try:
        vals = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        s = evaluate_sequence(vals)
        return float(s)
    except Exception:
        return float("inf")

    seen = set()
    for lbl, ini in inits:
        sig = tuple(np.round(ini[::max(1, n0 // 40)], 2))
        if sig in seen:
            continue
        seen.add(sig)
        consider(ini, "init_" + lbl)

    # ---- Phase 1: FISTA on each init (fast local optimisation) ----
    rem = time_budget - (time.monotonic() - t_global)
    t_p1 = min(rem * 0.28, 480.0)
    print("\n=== Phase 1: FISTA (exact subgradient) ===")
    if t_p1 > 60 and best_seq is not None:
        per = max(15.0, t_p1 / len(seen))
        k = 0
        for lbl, ini in inits:
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 12:
                break
            k += 1
            aout, _ = fista_optimize(best_seq if k == 1 else ini, t_avail,
                                     L=4.0, verbose=False, seed=100 + k)
            consider(aout, "fista_" + lbl)

    # ---- Phase 2: cutting-plane LP Frank-Wolfe (exact directions) ----
    rem = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    t_p2 = min(rem * 0.30, 520.0)
    print("\n=== Phase 2: LP Frank-Wolfe (cutting plane) n=%d ===" % n_best)
    if t_p2 > 40 and n_best <= 5000:
        aout, _ = lp_cutting_plane(best_seq, t_p2, solver, verbose=True,
                                   init_K=200, max_K=2600, line_search=True)
        consider(aout, "lpfw_1")

    # ---- Phase 3: FGM mirror-descent filler ----
    rem = time_budget - (time.monotonic() - t_global)
    t_p3 = min(rem * 0.10, 150.0)
    print("\n=== Phase 3: FGM mirror descent ===")
    if t_p3 > 20:
        aout, _ = fgm_optimize(best_seq, t_p3, eta_init=0.30, verbose=False, seed=7)
        consider(aout, "fgm")

    # ---- Phase 4: length exploration ----
    base_n = len(best_seq) if best_seq is not None else n0
    cand_lengths = [base_n + 300, base_n - 300, base_n + 700]
    cand_lengths = [l for l in cand_lengths if 1200 <= l <= 5000 and l != base_n]
    print("\n=== Phase 4: length exploration ===")
    for new_n in cand_lengths[:2]:
        rem = time_budget - (time.monotonic() - t_global)
        if rem < 90:
            break
        a_long = interpolate_sequence(best_seq, new_n)
        tA = min(rem * 0.45, 150.0)
        if tA > 15:
            aout, _ = fista_optimize(a_long, tA, L=4.0, verbose=False, seed=211)
            consider(aout, "fista_n%d" % new_n)
            rem = time_budget - (time.monotonic() - t_global)
            if rem > 60 and new_n <= 4500:
                tB = min(rem * 0.5, 160.0)
                aout, _ = lp_cutting_plane(best_seq, tB, solver, verbose=False,
                                           init_K=150, max_K=2000, line_search=True)
                aout2, _ = lp_cutting_plane(a_long, tB, solver, verbose=False,
                                            init_K=150, max_K=2000, line_search=True)
                consider(aout, "lpfw_n%d" % new_n)
                consider(aout2, "lpfw_interp_n%d" % new_n)

    # ---- Phase 5: final LP Frank-Wolfe + FISTA polish on global best ----
    rem = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else base_n
    print("\n=== Phase 5: final polish n=%d ===" % n_best)
    if rem > 30 and n_best <= 5000:
        tB = min(rem * 0.55, 200.0)
        aout, _ = lp_cutting_plane(best_seq, tB, solver, verbose=True,
                                   init_K=300, max_K=2800, line_search=True)
        consider(aout, "lpfw_final")
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 20:
            aout, _ = fista_optimize(best_seq, min(rem * 0.6, 120.0), L=3.5,
                                     verbose=False, seed=999)
            consider(aout, "fista_final")

    # ---- final evaluation & return ----
    if best_seq is None:
        fb = make_smooth_init(3200, seed=1)
        best_seq = fb
        best_score = safe_score(fb)
        print("Fallback score=%.6f" % best_score)
    else:
        final_sc = safe_score(best_seq)
        if np.isfinite(final_sc):
            best_score = final_sc
        # one last symmetrisation cannot hurt
        a_sym = symmetrize(best_seq)
        sc_sym = safe_score(a_sym)
        if np.isfinite(sc_sym) and sc_sym < best_score:
            best_seq = a_sym
            best_score = sc_sym
    print("\nFINAL: score=%.6f n=%d time=%.0fs"
          % (best_score, len(best_seq), time.monotonic() - t_global))
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64).ravel()]
