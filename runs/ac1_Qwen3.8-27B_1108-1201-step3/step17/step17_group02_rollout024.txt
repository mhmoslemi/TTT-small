def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_constraint_rows(a, binding_idx):
    n = len(a)
    K = len(binding_idx)
    T = np.zeros((K, n))
    for idx, k in enumerate(binding_idx):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        if s < e:
            T[idx, s:e] = a[ki + 1 - e: ki + 1 - s][::-1]
    return T


def logsumexp_subgradient(a, ac, top_k=80, beta=8.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        if s >= e:
            continue
        idx = np.arange(s, e)
        partner = ki - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad


def logsumexp_subgradient_v2(a, ac, top_k=80, beta=8.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    shifted = top_vals - top_vals[0]
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        if s >= e:
            continue
        idx = np.arange(s, e)
        partner = ki - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad


def amd_optimize(a0, time_limit, top_k=60, beta_start=10.0, beta_end=4.0,

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta_start

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = t / max(restart_every, 1)
        cur_beta = max(beta_start - (beta_start - beta_end) * min(frac, 1.0), beta_end)

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = logsumexp_subgradient_v2(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_y = np.log(y)
        log_a_new = log_y - eta * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            sigma = 0.02 + 0.01 * np.random.default_rng(t).uniform()
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, K_init=100, K_max=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if K_max is None:
        K_max = min(2 * n - 1, 3000)

    y = a.copy()
    x = a.copy()
    K_sel = K_init
    iter_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if K_sel >= 2 * n - 1:

        else:
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]

        T_sel = build_constraint_rows(y, binding_idx)
        M_vec = M_curr * np.ones(len(binding_idx))

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t <= 3:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

def pair_local_search(a0, time_limit, verbose=True, rng_seed=99):

    a = norm_sum1(a0)
    best_a = a.copy()

    n_src = 30

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            sens[idx] += a[partner]

        src_candidates = np.argsort(sens)[::-1][:n_src]
        low_sens = np.argsort(sens)[:n_src]

        improved = False

        for src_i, src in enumerate(src_candidates):
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

                if dst == src:
                    continue

                found = False
                for frac in [0.02, 0.01, 0.005, 0.03, 0.05, 0.015, 0.008]:

                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

                if found:
                    break

            if improved and time.monotonic() - t0 > time_limit:
                break

        if not improved:
            stall += 1
            if stall > 50:
                n_pert = min(10, n)

def spectral_init(n, n_modes=25, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Multi-init AMD
    print(f"\n=== Phase 1: Multi-init AMD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.28, 450.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_bell(n0, 0.35))
        labels.append("bell")
        inits.append(make_bimodal(n0, 0.18))
        labels.append("bimodal")
        inits.append(spectral_init(n0, 30, seed=42))
        labels.append("spec1")
        inits.append(spectral_init(n0, 15, seed=7))
        labels.append("spec2")
        inits.append(make_asymmetric(n0, 0.5))
        labels.append("asym")

        if parent is not None:
            blend1 = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend1))
            labels.append("blend_u")
            blend2 = 0.6 * norm_sum1(parent) + 0.4 * make_bell(n0, 0.4)
            inits.append(norm_sum1(blend2))
            labels.append("blend_bell")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(25.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta_start=12.0, beta_end=4.0,
                                     c0=5.0, restart_every=4000, verbose=False, rng_seed=100 + idx)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        K_init = min(80, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 2500)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, K_init=K_init, K_max=K_max)
        try_update(a_out, "lp_main")

    # Phase 3: Pair local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Pair local search (n={n_best}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_local_search(a_in, t3, verbose=True, rng_seed=55)
        try_update(a_out, "pairls")

    # Phase 4: Gradient pair search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Gradient pair search (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_pair_search(a_in, t4, verbose=True, rng_seed=77)
        try_update(a_out, "gradpair")

    # Phase 5: AMD polish with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: AMD polish (n={n_best}) ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t5, top_k=100, beta_start=6.0, beta_end=3.0,
                                 c0=4.0, restart_every=6000, verbose=False, rng_seed=200)
        try_update(a_out, "amd_polish")

    # Phase 6: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW r2 (n={n_best}) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        K_init = min(200, 2 * n_best - 1)
        K_max = min(2 * n_best - 1, 4000)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, K_init=K_init, K_max=K_max)
        try_update(a_out, "lp_r2")

    # Phase 7: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500, -1500]:
        new_n = n_best + delta
        if 800 <= new_n <= 5500:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.4, 200.0)
        if t7a > 30:
            a_out, _ = amd_optimize(a_interp, t7a, top_k=60, beta_start=10.0, beta_end=4.0,
                                     c0=4.5, restart_every=5000, verbose=False, rng_seed=new_n % 500)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 4500:
            t7b = min(remaining * 0.35, 120.0)
            if t7b > 25:
                a_in = interpolate_sequence(best_seq, new_n)
                K_init = min(100, 2 * new_n - 1)
                K_max = min(2 * new_n - 1, 2000)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=True, K_init=K_init, K_max=K_max)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8a = min(remaining * 0.4, 100.0)
        if t8a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = pair_local_search(a_in, t8a, verbose=False, rng_seed=333)
            try_update(a_out, "pairls_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.4, 80.0)
            if t8b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = gradient_pair_search(a_in, t8b, verbose=False, rng_seed=444)
                try_update(a_out, "gradpair_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25 and n_best <= 5000:
            t8c = min(remaining * 0.5, 80.0)
            if t8c > 20:
                a_in = norm_sum1(best_seq)
                K_init = min(300, 2 * n_best - 1)
                K_max = min(2 * n_best - 1, 4000)
                a_out, _ = lp_frank_wolfe(a_in, t8c, solver, verbose=True, K_init=K_init, K_max=K_max)
                try_update(a_out, "lp_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
