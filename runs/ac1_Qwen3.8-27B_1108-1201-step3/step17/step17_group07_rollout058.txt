"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed L-BFGS optimization with softmax parameterization
2. Adaptive temperature (β) schedule for sharpening the max approximation
3. Multi-start with diverse initializations (parent, sinusoidal, asymmetric shapes)
4. Peak-shaving coordinate descent for final polish
5. Multi-length exploration
Key difference from previous approach: uses smooth L-BFGS on the reparameterized
problem instead of subgradient mirror descent, enabling faster convergence to
sharper local minima.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_cross_correlation_fft(a, p):
    """Compute c[j] = sum_m a[m] * p[m+j] for j=0..n-1.
    a has length n, p has length 2n-1.
    Uses FFT: c = IFFT(FFT(a,L) * FFT(flip(p),L))[j + len(p) - 1]
    """
    n = len(a)
    lp = len(p)  # 2n-1
    L = 1 << (n + lp - 1).bit_length()
    A = np.fft.rfft(a, L)
    P_flip = np.fft.rfft(np.flip(p), L)
    conv = np.fft.irfft(A * P_flip, L)
    # conv[k] = sum_m a[m] * flip(p)[k-m] = sum_m a[m] * p[lp-1-(k-m)]
    # We want sum_m a[m] * p[m+j], so set lp-1-(k-m) = m+j → k = lp-1-j
    c = conv[lp - 1 - np.arange(n)]
    return c

def logsumexp_objective_and_grad(z, beta):
    """Objective and gradient for z (log-parameters), where a = softmax(z).
    Returns (f, g) where f = 2*n*logsumexp(beta*conv(a,a)) and g = grad w.r.t. z.
    """
    n = len(z)
    # Compute a = softmax(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute b = conv(a, a)
    b = fast_conv(a)
    
    # Log-sum-exp
    M = b.max()
    exp_b = np.exp(beta * (b - M))
    S = exp_b.sum()
    p = exp_b / S
    lse = M + np.log(S) / beta  # (1/beta) * log(sum(exp(beta*b)))
    
    f = 2.0 * n * lse
    
    # Gradient of lse w.r.t. a_j:
    # d(dse)/da_j = beta * sum_k p[k] * db_k/da_j = beta * sum_k p[k] * 2*a[k-j]*[0<=k-j<n]
    # = 2*beta * sum_k p[k] * a[k-j] (where valid)
    # = 2*beta * c[j] where c[j] = sum_m a[m]*p[j+m] = cross-correlation
    c = compute_cross_correlation_fft(a, p)
    grad_a = 2.0 * beta * c
    
    # Chain rule through softmax: da_j/dz_i = a_j*(delta_ji - a_i)
    # df/dz_i = sum_j (df/da_j) * a_j * (delta_ji - a_i)
    #         = a_i * (grad_a[i] - sum_j a_j * grad_a[j])
    weighted_sum = np.dot(a, grad_a)
    grad_z = a * (grad_a - weighted_sum)
    
    return f, grad_z

    n = len(a0)

    betas = np.geomspace(beta_init, beta_end, n_betas)
    
    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Run L-BFGS
        options = {

        }
        
        try:
            result = minimize(
                logsumexp_objective_and_grad,

                options=options
            )
            z = result.x
        except Exception as e:

                print(f"    L-BFGS error at beta={beta}: {e}")
            break
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Evaluate
        b = fast_conv(a)
        f = 2.0 * n * b.max()
        if f < best_f:
            best_f = f
            best_a = a.copy()
        
        if verbose:

            print(f"    L-BFGS beta={beta:.1f} ({elapsed:.1f}s): f={f:.8f}, best={best_f:.8f}")
    
    return best_a, best_f
