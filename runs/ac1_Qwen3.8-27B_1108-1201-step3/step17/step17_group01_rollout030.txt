"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth-max gradient (log-sum-exp of convolution peaks)
2. Nesterov-accelerated mirror descent on the simplex with adaptive temperature
3. Gerchberg-Saxton frequency-domain initialization with edge enhancement
4. Structured restarts via spectral smoothing and partial reversal
5. Targeted local search exploiting convolution gradient structure
6. Multi-length exploration to find optimal sequence length

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_max_gradient(a, beta):
    """
    Compute gradient of log(sum(exp(beta*C))) where C = conv(a,a).
    This gives a smooth approximation to the subgradient of max(C).
    Uses FFT cross-correlation for O(n log n) complexity.
    """
    n = len(a)
    C = fast_autocorr(a)
    
    # Softmax weights over convolution values
    beta_C = beta * C
    max_bc = beta_C.max()
    exp_bc = np.exp(beta_C - max_bc)
    w = exp_bc / exp_bc.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (a[j]=0 outside [0,n))
    # This is 2 * cross-correlation of w and a
    N = 1 << (4 * n - 2).bit_length()
    x = np.zeros(N)
    x[:n] = a
    y = np.zeros(N)
    y[:2 * n - 1] = w
    
    X = np.fft.fft(x)
    Y = np.fft.fft(y)
    
    g = np.fft.ifft(Y * np.conj(X)).real[:n] * 2.0
    return g


def topk_subgradient(a, C, top_k=100, beta=3.0):
    """Compute subgradient using top-k peaks with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(C))
    top_idx = np.argsort(C)[::-1][:k_max]
    top_vals = np.maximum(C[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def gs_initialize(n, n_iter=200, edge_boost=1.5, seed=42):
    """
    Gerchberg-Saxton initialization with edge enhancement preference.
    Alternates between flat spectrum constraint and time-domain non-negativity + edge boost.
    """
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Start with random phase
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    target_mag = np.ones(fft_size // 2 + 1) / np.sqrt(fft_size)
    
    a = np.zeros(n)
    for it in range(n_iter):
        # Frequency domain: set magnitudes to target, keep phase
        A = target_mag * np.exp(1j * phase)
        a_complex = np.fft.irfft(A, fft_size)[:n]
        a_real = np.real(a_complex)
        
        # Time domain: project to non-negative with edge boost preference
        a_proj = np.maximum(a_real, 0.0)
        
        # Apply edge boost: multiply edges by a factor to encourage U-shape
        edge_width = max(1, n // 20)
        edge_factor = np.ones(n)
        for i in range(edge_width):
            f = edge_boost * (1.0 - i / edge_width) + 1.0
            edge_factor[i] = f
            edge_factor[n - 1 - i] = f
        a_proj = a_proj * edge_factor
        
        # Also try a mild "smile" shape preference
        x = np.linspace(0, 1, n)
        smile = 1.0 + 0.3 * (2 * x - 1) ** 2
        a_proj = a_proj * smile
        
        s = a_proj.sum()
        if s < 1e-15:
            a_proj = np.ones(n) / n
        else:
            a_proj = a_proj / s
        
        # Update phase from time-domain
        A_new = np.fft.rfft(a_proj, fft_size)
        phase = np.angle(A_new)
        a = a_proj.copy()
    
    return normalize_sum1(a)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                                     beta_start=20.0, beta_end=100.0,
                                     c0=2.5, restart_every=3000)
            try_update(a_out, "amd_after_spectral")
    
    # === Phase 3: Local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Local search (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = local_search(best_seq, t3, rng_seed=42)
        try_update(a_out, "local")
    
    # === Phase 4: AMD with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: AMD variant (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        # Different beta schedule
        a_out, _ = amd_optimize(best_seq, t4,
                                 beta_start=15.0, beta_end=200.0,
                                 c0=2.0, restart_every=5000)
        try_update(a_out, "amd_high_beta")
        
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.08, 100.0)
        if t4b > 15:
            a_out, _ = amd_optimize(best_seq, t4b,
                                     beta_start=5.0, beta_end=30.0,
                                     c0=4.0, restart_every=2000)
            try_update(a_out, "amd_low_beta")
    
    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length search ===")
    n_best = len(best_seq)
    candidate_ns = [n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]
    candidate_ns = [n for n in candidate_ns if 1500 <= n <= 6000]
    # Remove duplicates and sort by proximity to current best
    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_best))
    
    for new_n in candidate_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")
        
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 15:
            a_out, _ = amd_optimize(a_interp, t5a,
                                     beta_start=10.0, beta_end=60.0,
                                     c0=3.0, restart_every=3000)
            try_update(a_out, f"amd_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.15, 60.0)
            if t5b > 10:
                a_out, _ = spectral_refine(a_out, t5b)
                try_update(a_out, f"spectral_n{new_n}")
    
    # === Phase 6: Final intense AMD on overall best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final AMD (n={len(best_seq)}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.35, 250.0)
        a_out, _ = amd_optimize(best_seq, t6,
                                 beta_start=30.0, beta_end=300.0,
                                 c0=2.0, restart_every=8000)
        try_update(a_out, "amd_final_intense")
    
    # === Phase 7: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final local search (n={len(best_seq)}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.30, 100.0)
        a_out, _ = local_search(best_seq, t7, rng_seed=999)
        try_update(a_out, "local_final")
    
    # === Phase 8: One more AMD with very high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 8: Final polish ===")
        t8 = min(remaining * 0.5, 80.0)
        a_out, _ = amd_optimize(best_seq, t8,
                                 beta_start=50.0, beta_end=500.0,
                                 c0=1.5, restart_every=10000)
        try_update(a_out, "polish")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize_sum1((2 * x - 1) ** 2 + 0.1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
