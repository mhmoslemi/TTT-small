Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov mirror descent

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * np.real(A), fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))

def safe_eval(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')

def peak_grad(a, k):
    """Subgradient of autocorr[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    if k == 0:
        g = 2.0 * a
    else:
        # g[j] = a[j+k] (if j+k<n) + a[j-k] (if j-k>=0)
        j1 = np.arange(0, n - k)
        g[j1] += a[j1 + k]
        j2 = np.arange(k, n)
        g[j2] += a[j2 - k]
    return g

def nesterov_mirror(a0, time_limit, seed=42):
    """Nesterov-accelerated mirror descent on max(autocorr)."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_val = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation and find peak
        ac = fast_autocorr(y)
        k_peak = int(np.argmax(ac))
        
        # Subgradient
        g = peak_grad(y, k_peak)
        
        # Polyak step size
        lower_bound = 1.0 / n  # minimum possible max(autocorr) when sum=1
        gap = ac[k_peak] - lower_bound
        eta = 3.0 * max(gap, 0) / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)
        
        # Mirror descent step (exponential)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        J_new = compute_J(x)
        if J_new < best_val - 1e-12:
            best_val = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart

            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
    
    return best_a, best_val

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        mean_s = sens.mean()
        std_s = sens.std() + 1e-15

        high_idx = np.where((sens > mean_s + 0.3 * std_s) & (a > 1e-8))[0]
        low_idx = np.where(sens < mean_s - 0.3 * std_s)[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            stall += 3
            continue

        improved = False
        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            src = high_idx[rng.integers(len(high_idx))]
            dst = low_idx[rng.integers(len(low_idx))]
            if src == dst:
                continue
            frac = rng.choice([0.002, 0.005, 0.01, 0.02, 0.05, 0.08, 0.12])
            amt = a[src] * frac
            if amt < 1e-14:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            J_trial = compute_J(a_trial)
            if J_trial < best_val - 1e-14:
                a = a_trial
                best_val = J_trial
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                n_pert = max(5, n // 30)
                idx = rng.choice(n, size=n_pert, replace=False)
                for i in idx:
                    if a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.01, 0.06)
                        a[i] -= amt
                        j = int(rng.integers(n))
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_val:
                    best_val = J_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_val


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def init_cosine(n):
    x = np.linspace(0, np.pi, n)
    a = np.cos(x) ** 2 + 0.01
    return norm_sum1(a)


def init_gaussian_edges(n, sigma=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)
