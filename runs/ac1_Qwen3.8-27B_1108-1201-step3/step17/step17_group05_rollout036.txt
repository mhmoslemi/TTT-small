def _fast_autocorr(a):
    """Compute conv(a, a) using FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def _smooth_max_value(ac, beta=100.0):
    """Compute smooth max of ac using log-sum-exp."""
    m = ac.max()
    if m < 1e-30:
        return m
    shifted = ac - m
    log_sum = m + np.log(np.exp(beta * shifted).sum()) / beta
    return log_sum

def _smooth_gradient(a, ac, beta=100.0, top_k=200):
    """
    Compute approximate gradient of smooth_max(conv(a,a)) w.r.t. a.
    g[j] = 2 * sum_m w[m] * a[m-j] where w is softmax over ac with temperature beta.
    Uses top-k lags for efficiency.
    """
    n = len(a)
    ac_max = ac.max()
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    shifted = top_vals - ac_max
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k) / k
    else:
        w_top = w_top / w_sum
    # Compute g[j] = 2 * sum_{m in top} w[m] * a[m-j] for valid m-j
    g = np.zeros(n)
    for wi, mi in zip(w_top, top_idx):
        mi = int(mi)
        # a[m-j] is valid when 0 <= m-j < n, i.e., m-n+1 <= j <= m
        j_start = max(0, mi - n + 1)
        j_end = min(n - 1, mi)
        if j_start <= j_end:
            g[j_start:j_end + 1] += 2.0 * wi * a[mi - np.arange(j_start, j_end + 1)]
    return g

def _objective(a):
    """Compute the objective value for a normalized sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    ac = _fast_autocorr(a)
    max_b = ac.max()
    return float(2.0 * n * max_b / (s * s))

def _fista_optimize(a0, time_limit=300, beta=80.0, top_k=150, verbose=False, seed=42):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) with smooth max objective.
    Optimize on simplex: min f(a) s.t. a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_obj = _objective(a)
    
    t0 = time.monotonic()
    x = a.copy()
    z = a.copy()
    t_mom = 1.0
    
    lgamma = np.log(a + 1e-30)
    # Lipschitz estimate: for the smooth max, L ~ 2*beta*max(a) approximately
    L = 2.0 * beta * max(a.max(), 0.1) + 1.0
    step = 1.0 / L
    
    stall = 0
    max_stall = 500
    rng = np.random.default_rng(seed)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = z + (t_mom - 1.0) / t_new * (z - x)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = z.copy()
        else:
            y = y / sy
        
        # Compute gradient
        ac = _fast_autocorr(y)
        g = _smooth_gradient(y, ac, beta=beta, top_k=top_k)
        
        # Gradient step
        x_new = y - step * g
        x_new = _project_simplex(x_new, 1.0)
        
        # Check objective
        obj = _objective(x_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive: if stalled, increase beta (sharper max approximation)
        if stall % 200 == 0 and stall > 0:
            beta *= 1.2
            step = 1.0 / (2.0 * beta * max(x_new.max(), 0.05) + 1.0)
            if verbose:
                print(f"      beta={beta:.1f}, step={step:.6f}")
        
        # Diversification: random perturbation every 300 stalls
        if stall % 300 == 0:
            pert = rng.normal(0, 0.02, n) * best_a
            x_pert = best_a + pert
            x_pert = _project_simplex(x_pert, 1.0)
            if _objective(x_pert) < best_obj:
                best_obj = _objective(x_pert)
                best_a = x_pert.copy()
                x = x_pert.copy()
                z = x_pert.copy()
                stall = 0
            else:
                # Reset momentum
                x = best_a + rng.normal(0, 0.01, n)
                x = _project_simplex(x, 1.0)
                z = x.copy()
        
        x = x_new
        z = x_new + (t_mom - 1.0) / t_new * (x - x)  # z = x_new for next iter
        t_mom = t_new
        
        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FISTA iter={iteration} ({elapsed:.0f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def _peak_directed_search(a0, time_limit=120, verbose=False, seed=42):
    """
    Peak-directed local search: identify peak lags and perform mass transfers
    to reduce the peak autocorrelation.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_obj = _objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    transfers = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        M = ac.max()
        
        # Find top peak lags
        n_top = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            
            # Find contributing pairs for this lag
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            js = np.arange(j_start, j_end + 1)
            partners = ki - js
            contrib = a[js] * a[partners]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the most contributing source
            src_local = np.argmax(contrib)
            src = int(js[src_local])
            partner = int(partners[src_local])
            
            # Compute sensitivity of each position to ALL top peaks
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    p = k2i - d
                    if 0 <= p < n:
                        sens[d] += a[p]
            
            # Find good destinations: low sensitivity, far from source
            dist = np.abs(np.arange(n) - src).astype(float)
            score = sens + 0.1 / (dist + 1.0)  # prefer far, low sensitivity
            # Exclude positions that would create new peaks
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            mask[partner] = False
            
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue
            best_dst = candidates[np.argmin(score[candidates])]
            
            # Try multiple transfer fractions
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            found = False
            for frac in [0.05, 0.02, 0.01, 0.08, 0.005, 0.03, 0.1, 0.002]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                ob = _objective(a_trial)
                if ob < best_obj - 1e-14:
                    a = a_trial
                    best_obj = ob
                    best_a = a.copy()
                    improved = True
                    transfers += 1
                    found = True
                    stall = 0
                    break
            
            if found:
                break  # restart peak analysis after a move
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 20 == 0:
                n_pert = rng.integers(1, min(5, n))
                indices = rng.choice(n, n_pert, replace=False)
                for idx in indices:
                    if a[idx] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[idx] * frac
                        dst = int(rng.integers(n))
                        a[idx] -= amt
                        a[dst] += amt
                a = _project_simplex(a, 1.0)
                ob = _objective(a)
                if ob < best_obj:
                    best_obj = ob
                    best_a = a.copy()
                    stall = 0
        
        if verbose and (transfers % 100 == 0 or stall % 100 == 0):
            elapsed = time.monotonic() - t0
            print(f"      PeakSearch ({elapsed:.0f}s): obj={best_obj:.6f}, trans={transfers}, stall={stall}")
    
    return best_a, best_obj

def _make_power_law_init(n, alpha=1.0, seed=0):
    """Power-law initialization biased towards one end."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return _norm_sum1(a)

def _make_two_hump_init(n, p=0.3, width=0.1, seed=0):
    """Two-hump initialization with peaks at positions p and 1-p."""
    x = np.linspace(0, 1, n)
    sigma = width
    a = np.exp(-((x - p) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - (1 - p)) ** 2) / (2 * sigma ** 2))
    return _norm_sum1(a)

def _make_exp_init(n, rate=2.0):
    """Exponential initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return _norm_sum1(a)

def _make_sine_init(n, freq=3, amp=0.3, seed=0):
    """Sine-modulated initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return _norm_sum1(a)

def _make_smooth_random(n, seed=0):
    """Smooth random initialization via low-frequency Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.01)
    return _norm_sum1(a)

def _interpolate_seq(a, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase optimization with diverse initializations,
    FISTA, peak-directed search, and LP polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = _norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2000
        best_seq = _make_smooth_random(n_base, seed=1)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Try diverse lengths and initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.35
    
    lengths_to_try = sorted(set([n0, 1500, 2000, 2500, 3200, 4000]))
    lengths_to_try = [l for l in lengths_to_try if 500 <= l <= 5000]
    
    # For each length, generate a few diverse inits
    init_configs = []
    
    for n_try in lengths_to_try[:4]:
        inits = []
        labels = []
        
        # Power law
        inits.append(_make_power_law_init(n_try, alpha=0.8))
        labels.append(f"power_n{n_try}")
        
        # Two hump
        inits.append(_make_two_hump_init(n_try, p=0.25, width=0.08))
        labels.append(f"2hump_n{n_try}")
        
        # Exponential
        inits.append(_make_exp_init(n_try, rate=1.5))
        labels.append(f"exp_n{n_try}")
        
        # Sine
        inits.append(_make_sine_init(n_try, freq=2, amp=0.25, seed=n_try))
        labels.append(f"sine_n{n_try}")
        
        # Smooth random
        inits.append(_make_smooth_random(n_try, seed=n_try + 100))
        labels.append(f"smooth_n{n_try}")
        
        # Parent interpolated (if available)
        if parent is not None:
            inits.append(_interpolate_seq(parent, n_try))
            labels.append(f"parent_n{n_try}")
        
        init_configs.extend(zip(inits, labels))
    
    # Remove duplicates and limit count
    seen_sigs = set()
    unique_configs = []
    for init, lbl in init_configs:
        sig = tuple(np.round(init[::max(1, len(init) // 20)], 2))
        if sig not in seen_sigs:
            seen_sigs.add(sig)
            unique_configs.append((init, lbl))
    
    # Limit to top 12 configs
    unique_configs = unique_configs[:12]
    
    t_per_init = max(10.0, t_phase1 / len(unique_configs))
    
    for idx, (init, lbl) in enumerate(unique_configs):
        elapsed_so_far = time.monotonic() - t_global
        t_avail = min(t_per_init, time_budget - 30 - elapsed_so_far)
        if t_avail < 10:
            break
        
        print(f"  FISTA [{lbl}]: n={len(init)}, t_avail={t_avail:.0f}s")
        
        # First evaluate the init
        init_score = _objective(init)
        sc = _evaluate_safe(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = init.copy()
            print(f"    Init already good: {sc:.6f}")
        
        # Run FISTA
        a_out, _ = _fista_optimize(init, time_limit=min(t_avail * 0.6, 200),
                                    beta=60.0, top_k=100, verbose=False, seed=42 + idx)
        try_update(a_out, f"fista_{lbl}")
        
        # If FISTA improved, do peak-directed search
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_in = _norm_sum1(best_seq)
            if len(a_in) == len(init):
                t_peak = min(remaining * 0.2, 60)
                if t_peak > 15:
                    a_out2, _ = _peak_directed_search(a_in, time_limit=t_peak,
                                                       verbose=False, seed=99 + idx)
                    try_update(a_out2, f"peak_{lbl}")
    
    # === Phase 2: Focused optimization on best found ===
    print(f"\n=== Phase 2: Focused Optimization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"  Best so far: score={best_score:.6f}, n={len(best_seq)}")
    
    n_best = len(best_seq)
    t_phase2 = remaining * 0.25
    
    if t_phase2 > 30:
        # FISTA with higher beta (sharper)
        t_fista = min(t_phase2 * 0.5, 200)
        if t_fista > 20:
            print(f"  FISTA sharp (n={n_best}, t={t_fista:.0f}s)")
            a_out, _ = _fista_optimize(best_seq, time_limit=t_fista,
                                        beta=120.0, top_k=200, verbose=True, seed=777)
            try_update(a_out, "fista_sharp")
        
        # Peak-directed search
        remaining = time_budget - (time.monotonic() - t_global)
        t_peak = min(remaining * 0.3, 120)
        if t_peak > 15:
            print(f"  Peak search (n={n_best}, t={t_peak:.0f}s)")
            a_out, _ = _peak_directed_search(best_seq, time_limit=t_peak,
                                              verbose=True, seed=555)
            try_update(a_out, f"peak_focus_n{n_best}")
    
    # === Phase 3: LP Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Polish ===")
    t_lp = min(remaining * 0.2, 180)
    if t_lp > 30 and n_best <= 4000:
        print(f"  LP-FW (n={n_best}, t={t_lp:.0f}s)")
        a_out, _ = _lp_polish(best_seq, time_limit=t_lp, verbose=True)
        try_update(a_out, f"lp_n{n_best}")
    
    # === Phase 4: Second round of diverse search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Second Round ===")
    
    if remaining > 120:
        # Try a few more exotic initializations
        exotic_inits = []
        n_cur = len(best_seq)
        
        # Asymmetric power law (stronger bias)
        exotic_inits.append((_make_power_law_init(n_cur, alpha=2.0), "power2"))
        
        # Very narrow two-hump
        exotic_inits.append((_make_two_hump_init(n_cur, p=0.2, width=0.03), "narrow2h"))
        
        # Multi-frequency sine
        rng_ex = np.random.default_rng(1234)
        x = np.linspace(0, 1, n_cur)
        a_ex = np.ones(n_cur) * 0.5
        for f in [1, 2, 5, 8]:
            a_ex += 0.05 * rng_ex.normal() * np.sin(2 * np.pi * f * x + rng_ex.uniform(0, 6.28))
        a_ex = np.maximum(a_ex, 0.01)
        exotic_inits.append((_norm_sum1(a_ex), "multifreq"))
        
        # Blend of best and power law
        blend = 0.5 * best_seq + 0.5 * _make_power_law_init(n_cur, alpha=1.0)
        exotic_inits.append((_norm_sum1(blend), "blend_power"))
        
        # Inverted: high in middle, low at edges
        x = np.linspace(0, 1, n_cur)
        a_mid = 1.0 + 0.5 * np.cos(np.pi * x)  # peak in middle
        a_mid = np.maximum(a_mid, 0.01)
        exotic_inits.append((_norm_sum1(a_mid), "mid_peak"))
        
        t_per = max(15.0, min(remaining * 0.4 / len(exotic_inits), 80))
        for init, lbl in exotic_inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  Exotic [{lbl}] (t={t_avail:.0f}s)")
            
            a_out, _ = _fista_optimize(init, time_limit=min(t_avail * 0.5, 100),
                                        beta=80.0, top_k=100, verbose=False, seed=200)
            try_update(a_out, f"exotic_{lbl}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = _peak_directed_search(_norm_sum1(best_seq), 
                                                   time_limit=min(remaining * 0.15, 40),
                                                   verbose=False, seed=300)
                try_update(a_out2, f"peak_after_{lbl}")
    
    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final Polish ===")
    
    if remaining > 30:
        # Short FISTA with very high beta
        t_fin = min(remaining * 0.3, 100)
        if t_fin > 10:
            a_out, _ = _fista_optimize(best_seq, time_limit=t_fin,
                                        beta=200.0, top_k=300, verbose=False, seed=9999)
            try_update(a_out, "final_fista")
        
        # Peak search
        remaining = time_budget - (time.monotonic() - t_global)
        t_fs = min(remaining * 0.3, 80)
        if t_fs > 10:
            a_out, _ = _peak_directed_search(best_seq, time_limit=t_fs,
                                              verbose=False, seed=8888)
            try_update(a_out, "final_peak")
        
        # LP if time permits
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and len(best_seq) <= 3500:
            t_flp = min(remaining * 0.5, 60)
            if t_flp > 15:
                a_out, _ = _lp_polish(best_seq, time_limit=t_flp, verbose=False)
                try_update(a_out, "final_lp")
    
    # === Try adjacent lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Adjacent Lengths ===")
        for delta in [-200, 200]:
            new_n = n_best + delta
            if new_n < 500 or new_n > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"  Trying n={new_n}")
            a_interp = _interpolate_seq(best_seq, new_n)
            t_short = min(remaining * 0.3, 60)
            if t_short > 15:
                a_out, _ = _fista_optimize(a_interp, time_limit=t_short,
                                            beta=80.0, top_k=100, verbose=False, seed=500+new_n)
                try_update(a_out, f"adj_n{new_n}")
                
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 20:
                    a_out2, _ = _peak_directed_search(a_interp, time_limit=min(remaining*0.3, 30),
                                                       verbose=False, seed=600+new_n)
                    try_update(a_out2, f"adj_peak_n{new_n}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 1500
        best_seq = _make_smooth_random(n_fb, seed=1)
        best_score = _evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
