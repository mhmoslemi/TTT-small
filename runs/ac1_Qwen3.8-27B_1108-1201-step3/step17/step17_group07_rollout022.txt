def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def fast_conv(a):
    """Compute linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_objective_and_grad(a, beta):
    """
    Compute the log-sum-exp smooth approximation to max(b) and its gradient.
    f(a) = log(sum_k exp(beta * b[k]))
    grad[i] = 2 * beta * sum_k p[k] * a[k-i] (valid k-i)
    where p = softmax(beta * b)
    
    Returns (f_value, grad) where grad includes the beta factor.
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    xb = beta * b
    xb_max = xb.max()
    exp_xb = np.exp(xb - xb_max)
    p = exp_xb / exp_xb.sum()
    f_val = xb_max + np.log(exp_xb.sum())
    
    # grad[i] = 2*beta * sum_k p[k] * a[k-i]
    # = 2*beta * (a_rev * p)[i+n-1]
    a_rev = a[::-1].copy()
    N = _next_pow2(n + len(p) - 1)
    A_rev = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(A_rev * P, N)
    grad = 2.0 * beta * conv[n - 1:2 * n - 1]
    
    return f_val, grad

def nesterov_smooth_descent(a0, time_limit, beta_start=1.0, beta_end=200.0, 
                             c0=2.0, verbose=False):
    """
    Nesterov-accelerated mirror descent on the log-sum-exp smoothed objective.
    Works in log-space (entropy regularization on the simplex).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)  # log-weights
    v_prev = np.zeros(n)
    a_prev = a.copy()
    
    n_iters_planned = 50000
    stall = 0
    max_stall = 500
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        frac = min(t / n_iters_planned, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Nesterov extrapolation in log-space
        beta_t = (t - 1.0) / (t + 1.5)
        v_y = v + beta_t * (v - v_prev)
        y = np.exp(v_y - v_y.max())
        y = y / y.sum()
        
        # Compute gradient at y
        _, grad = smooth_objective_and_grad(y, beta)
        
        # Step size: eta = c / sqrt(t), scaled by beta since grad ~ beta
        eta = c0 / (np.sqrt(t) * beta + 1e-8)
        
        v_new = v - eta * grad
        v_new = v_new - v_new.max()  # numerical stability
        a_new = np.exp(v_new)
        a_new = a_new / a_new.sum()
        
        v_prev = v
        v = v_new
        a_prev = a
        a = a_new
        
        # Track best
        M = fast_conv(a).max()
        if M < best_M - 1e-18:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Restart: perturb best solution
            rng = np.random.default_rng(t * 7 + n)
            strategy = int(rng.integers(3))
            if strategy == 0:
                a_r = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                a_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 500), replace=False)
                a_r[idx] *= rng.uniform(0.7, 1.3, len(idx))
            else:
                a_r = best_a * (1 + rng.normal(0, 0.005, n))
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            a_prev = a_r
            v = np.log(a + 1e-15)
            v_prev = v.copy()
            stall = 0
            max_stall = max(200, int(max_stall * 0.7))
            c0 = max(0.3, c0 * 0.7)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SmoothND t={t} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, stall={stall}")
    
    return best_a, best_M

def lp_remez_step(a, max_active=50, trust_r=0.1, solver=None):
    """Single LP Remez step with trust region."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints (top max_active lags by b value)
    # Weight by how close they are to the max
    threshold = max_b * 0.99
    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        active = np.argsort(b)[::-1][:max_active]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            # Linearized: b[k](a') ≈ b[k](a) + sum_i coeffs[i] * (a'[i] - a[i])
            # We want: b[k](a') <= M
            # i.e., b[k](a) + sum_i coeffs[i] * a'[i] - sum_i coeffs[i] * a[i] <= M
            # Note: sum_i coeffs[i] * a[i] = sum_i 2*a[k-i]*a[i] = 2*b[k]... 
            # Wait, that's not right. Let me recompute.
            # b[k] = sum_{i=j_start}^{j_end} a[i] * a[k-i]
            # Linearization: b[k](a') ≈ b[k](a) + sum_i [a[k-i]*(a'[i]-a[i]) + a[i]*(a'[k-i]-a[k-i])]
            # = b[k](a) + sum_i a[k-i]*a'[i] - sum_i a[k-i]*a[i] + sum_i a[i]*a'[k-i] - sum_i a[i]*a[k-i]
            # = b[k](a) + sum_i a[k-i]*a'[i] - b[k](a) + sum_i a[i]*a'[k-i] - b[k](a)
            # = sum_i a[k-i]*a'[i] + sum_i a[i]*a'[k-i] - b[k](a)
            # Hmm, that doesn't simplify nicely. Let me use the gradient directly.
            # ∂b[k]/∂a[i] = 2*a[k-i] (if valid)
            # So: b[k](a') ≈ b[k](a) + sum_i 2*a[k-i] * (a'[i] - a[i])
            # = b[k](a) + 2*sum_i a[k-i]*a'[i] - 2*b[k](a)
            # = 2*sum_i a[k-i]*a'[i] - b[k](a)
            # We want this <= M:
            # 2*sum_i a[k-i]*a'[i] - b[k](a) <= M
            # 2*sum_i a[k-i]*a'[i] <= M + b[k](a)
            
            grad_k = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(grad_k, a_var[idx])) <= M_var + b[k_int])
    
    # Trust region: keep close to current a
    constraints.append(a_var <= a * (1.0 + trust_r))
    constraints.append(a_var >= a * (1.0 - trust_r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s

def remez_polish(a0, time_limit, solver=None, r_init=0.15, verbose=False):
    """LP Remez polishing loop with adaptive trust region."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_step(a, max_active=40, trust_r=r, solver=solver)
        
        if a_cand is None:
            r *= 0.7
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        accepted = False
        for frac in [1.0, 0.8, 0.5, 0.3, 0.1]:
            a_trial = (1 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-16:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.4, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.75)
            if no_improve > 40:
                # Perturb
                rng = np.random.default_rng(t * 13 + 7)
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, no_improve={no_improve}")
    
    return best_a, best_M

def coord_descent_polish(a0, time_limit, rng_seed=42):
    """Coordinate descent with random pair transfers."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_trials = min(200, n // 5)
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.001, 0.005, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                if amt > a[j] * 0.1:  # don't make j too dominant
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_M

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bell':
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    return f_interp / f_interp.sum()
