"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with full constraints and Nesterov acceleration
2. Nesterov-accelerated multiplicative weight updates (AMD) with exact subgradients  
3. Peak-pair coordinate descent for final polish
4. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=5):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    k_sel = np.argsort(ac)[::-1][:top_k]
    vals = np.maximum(ac[k_sel], 1e-30)
    w = vals / vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, k_sel):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        partner = ki - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad

def build_sparse_A(n, a, rows_idx, cols_idx):
    """Build the sparse constraint matrix for the LP."""
    vals = a[cols_idx]
    return sparse.csr_matrix((vals, (rows_idx, cols_idx)), shape=(2*n-1, n))

def precompute_lp_indices(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def lp_solve(a, M, solver_method='highs'):
    """Solve the LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_sp = build_sparse_A(n, a, *precompute_lp_indices(n))
    b_ub = M * np.ones(2*n-1)
    res = linprog(c, A_ub=A_sp, b_ub=b_ub, bounds=[(0, None)]*n, method=solver_method)
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_frank_wolfe(a0, time_limit, verbose=True, top_k_subgrad=5):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov extrapolated point
    x = a.copy()  # Main iterate
    rows_idx, cols_idx = precompute_lp_indices(n)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - y) if t > 1 else x.copy()
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Solve LP at y
        ac = fast_autocorr(y)
        M = ac.max()
        
        c = -np.ones(n)
        vals = y[cols_idx]
        A_sp = sparse.csr_matrix((vals, (rows_idx, cols_idx)), shape=(2*n-1, n))
        b_ub = M * np.ones(2*n-1)
        
        try:
            res = linprog(c, A_ub=A_sp, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        b_star = np.maximum(res.x, 0)
        b_sum = b_star.sum()
        if b_sum < 1e-15:
            break
        
        direction = b_star / b_sum
        
        # Update
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * x + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        
        x = x_new
        
        # Check improvement
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def amd_optimize(a0, time_limit, top_k=5, verbose=False, rng_seed=42):
    """Nesterov-accelerated multiplicative weight descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    y = a.copy()
    eta0 = 2.0
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - y) if t > 1 else x.copy()
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Subgradient
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=top_k)
        
        # Multiplicative weight update
        eta = eta0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        sa = a_new.sum()
        a_new = a_new / sa
        
        x = a_new
        mx = fast_autocorr(x).max()
        
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            # Restart with perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            y = x.copy()
            stall = 0
    
    return best_a, best_m

def coordinate_descent_peaks(a0, time_limit, verbose=False, rng_seed=42):
    """Peak-pair coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:10]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_partner = partner[np.argmax(contrib)]
            
            # Compute sensitivity
            sens = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                sens[idx2] += a[k2i - idx2]
            
            # Choose which to reduce
            if sens[src] >= sens[src_partner]:
                reduce_i = src
            else:
                reduce_i = src_partner
            
            if a[reduce_i] < 1e-12:
                continue
            
            # Find low-sensitivity destination
            exclude = {src, src_partner}
            mask = ~np.zeros(n, dtype=bool)
            for e in exclude:
                mask[e] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dst = candidates[np.argmin(sens[candidates])]
            if dst == reduce_i:
                continue
            
            # Try moving mass
            for frac in [0.01, 0.005, 0.02, 0.001, 0.05]:
                amt = a[reduce_i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_i] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                Mt = fast_autocorr(a_trial).max()
                if Mt < best_m - 1e-16:
                    a = a_trial
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                Mnew = fast_autocorr(a).max()
                if Mnew < best_m:
                    best_m = Mnew
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m
