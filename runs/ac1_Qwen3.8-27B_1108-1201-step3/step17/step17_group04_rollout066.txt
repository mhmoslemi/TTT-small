    n = len(a)
    size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def compute_gradient_fft(a, w):
    """grad[j] = 2 * sum_m a[m]*w[j+m]  (FFT cross-correlation)."""
    n = len(a)
    size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(A) * W, size)
    return 2.0 * corr[:n]

def _coord_eval(a, C, j, t, S, n):

    tot = S + t
    if tot <= 1e-300:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)"
                  % (label, sc, len(a), time.monotonic() - t_global))

    # parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            print("Parent: n=%d score=%.6f" % (len(parent), evaluate_safe(parent)))
    except Exception:
        pass

    if parent is None:
        parent = make_symmetric_init(3200)
        print("Default init n=%d" % len(parent))
    try_update(parent, "parent")
    n0 = len(parent)

    # ---- Phase 1: smooth descent from a few starts ----
    rem = time_budget - (time.monotonic() - t_global)
    t1 = min(rem * 0.40, 600.0)
    if t1 > 60:
        inits = [parent, make_symmetric_init(n0),
                 make_power_law_init(n0, 0.7), make_gaussian_edges(n0, 0.12)]
        tper = max(20.0, t1 / len(inits))
        for i, init in enumerate(inits):
            rem = time_budget - (time.monotonic() - t_global)
            if rem < 40:
                break
            t_av = min(tper, rem - 30)
            if t_av < 18:
                break
            print("  smooth init %d (%.0fs)" % (i, t_av))
            ao, _ = smooth_descent(init, t_av, verbose=False, seed=10 + i)
            try_update(ao, "smooth_%d" % i)
        # long smooth on best
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 60:
            ao, _ = smooth_descent(best_seq, min(rem * 0.5, 250.0), verbose=True, seed=900)
            try_update(ao, "smooth_long")

    # ---- Phase 2: coordinate descent (main improvement) ----
    rem = time_budget - (time.monotonic() - t_global)
    t2 = min(rem * 0.40, 550.0)
    if t2 > 40:
        print("  coord descent on n=%d (%.0fs)" % (len(best_seq), t2))
        ao, _ = coordinate_descent(best_seq, t2, verbose=True, seed=200)
        try_update(ao, "coord_n%d" % len(best_seq))

    # ---- Phase 3: LP Frank-Wolfe ----
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 40 and len(best_seq) <= 4000:
        t3 = min(rem * 0.25, 250.0)
        if t3 > 25:
            print("  lpfw on n=%d (%.0fs)" % (len(best_seq), t3))
            ao, _ = lp_frank_wolfe(best_seq, t3, solver, verbose=True, top_K=300)
            try_update(ao, "lpfw_n%d" % len(best_seq))
            # coord again after LP structural change
            rem = time_budget - (time.monotonic() - t_global)
            if rem > 30:
                ao, _ = coordinate_descent(best_seq, min(rem * 0.4, 150.0), verbose=False, seed=300)
                try_update(ao, "coord_after_lpfw")

    # ---- Phase 4: multi-length exploration ----
    rem = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidates_len = []
    for dn in [200, 400, -300]:
        L = n_best + dn
        if 1500 <= L <= 5000:
            candidates_len.append(L)
    for L in candidates_len[:2]:
        rem = time_budget - (time.monotonic() - t_global)
        if rem < 70:
            break
        print("  trying length %d" % L)
        aL = interpolate_sequence(best_seq, L)
        try_update(aL, "interp_n%d" % L)
        t_a = min(rem * 0.3, 120.0)
        if t_a > 25:
            ao, _ = smooth_descent(aL, t_a, verbose=False, seed=500 + L)
            try_update(ao, "smooth_n%d" % L)
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 40:
            ao, _ = coordinate_descent(best_seq, min(rem * 0.4, 120.0), verbose=False, seed=600 + L)
            try_update(ao, "coord_n%d" % L)

    # ---- final polish if time ----
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 25:
        ao, _ = coordinate_descent(best_seq, min(rem * 0.7, 120.0), verbose=True, seed=999)
        try_update(ao, "coord_final")

    # safety: ensure we return well before the hard limit
    if best_seq is None:
        best_seq = make_symmetric_init(3200)
        best_score = evaluate_safe(best_seq)
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("FINAL: score=%.6f n=%d t=%.0fs" % (best_score, len(best_seq), time.monotonic() - t_global))
    return [float(x) for x in best_seq]
