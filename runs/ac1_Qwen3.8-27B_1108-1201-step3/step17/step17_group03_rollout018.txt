def fast_autocorr(a):
    """Compute linear autocorrelation (convolution of a with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""

        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_parametric(n, family='sin2'):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    if family == 'sin2':
        f = np.sin(np.pi * x) ** 2
    elif family == 'sin4':
        f = np.sin(np.pi * x) ** 4
    elif family == 'raised_cosine':
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif family == 'beta':
        f = x * (1 - x)
    elif family == 'beta2':
        f = (x * (1 - x)) ** 2
    elif family == 'plateau':
        f = np.ones(n)
        quarter = n // 4
        f[:quarter] = 0.5
        f[-quarter:] = 0.5
    elif family == 'bimodal':

    elif family == 'asym':
        f = x * np.sin(np.pi * x) ** 2
    else:
        f = np.sin(np.pi * x) ** 2
    return norm_sum1(f)

    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** sharpness
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_optimize(a0, time_limit, top_k=50, sharpness=5.0, c0=4.0,
                            restart_every=3000, verbose=True, seed=42):
    """Mirror descent (multiplicative weight) with Nesterov acceleration and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_autocorr(y)
        M_y = ac_y.max()
        g = compute_subgradient(y, ac_y, top_k=top_k, sharpness=sharpness)

        # Backtracking line search
        eta = gamma / (np.sqrt(t) + 1.0)
        accepted = False
        for _ in range(8):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < M_y - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            eta *= 0.5

        if not accepted:
            # Try a smaller step
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Reduce sharpness to explore more broadly
            sharpness = max(sharpness - 0.5, 3.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_subproblem(f, M, active_idx, solver):
    """Solve the LP subproblem: max sum(b) s.t. b>=0, (f*b)[k]<=M for k in active_idx."""
    n = len(f)
    K = len(active_idx)
    
    # Build constraint matrix T where T[k, i] = f[active_idx[k] - i] if valid
    T_sel = np.zeros((K, n))
    for row, k_int in enumerate(active_idx):
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            T_sel[row, i] = f[k_int - i]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sel @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val / b_val.sum()


def lp_frank_wolfe(a0, time_limit, solver, top_K=200, margin_frac=2e-3,
                   verbose=True, use_iqcd=False, max_iterations=50000):
    """
    LP-based Frank-Wolfe with:
    - Adaptive constraint margin
    - Backtracking line search
    - Optional IQCD acceleration
    - Restart on stall
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    y = a.copy()
    x = a.copy()
    stall = 0
    margin = margin_frac
    cur_K = top_K

    for t in range(1, max_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active constraints
        margins_vec = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins_vec <= threshold
        n_active = active_mask.sum()

        if n_active > cur_K * 3:
            active_idx = np.argsort(margins_vec)[:cur_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins_vec)[:min(cur_K, len(ac))]
        else:
            active_idx = np.where(active_mask)[0]
        
        # Cap the number of constraints
        if len(active_idx) > cur_K:
            active_idx = active_idx[:cur_K]

        # Solve LP
        b_dir = lp_subproblem(y, M_curr, active_idx, solver)
        if b_dir is None:
            # Fallback: use subgradient direction
            g = compute_subgradient(y, ac, top_k=20, sharpness=4.0)
            log_a = np.log(np.maximum(y, 1e-15)) - 0.1 * g / max(g.max(), 1e-15)
            b_dir = np.exp(log_a - log_a.max())
            b_dir = np.maximum(b_dir, 1e-30)
            b_dir = b_dir / b_dir.sum()

        # Line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = alpha
        best_obj = M_curr
        best_step = (1.0 - alpha) * y + alpha * b_dir
        
        # Try multiple alphas
        for a_try in [alpha, alpha * 1.5, alpha * 0.7, min(alpha * 2.0, 1.0)]:
            a_try = max(0.01, min(a_try, 1.0))
            candidate = (1.0 - a_try) * y + a_try * b_dir
            candidate = np.maximum(candidate, 0.0)
            sc = candidate.sum()
            if sc < 1e-15:
                continue
            candidate = candidate / sc
            m_try = fast_autocorr(candidate).max()
            if m_try < best_obj:
                best_obj = m_try
                best_alpha = a_try
                best_step = candidate.copy()

        y_new = best_step
        
        # IQCD-style: try full step to b_dir
        if use_iqcd and t > 5:
            m_full = fast_autocorr(b_dir).max()
            if m_full < best_obj:
                y_new = b_dir.copy()
                best_obj = m_full

        # Extrapolation (Nesterov)
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        else:
            x_new = y_new.copy()

        y = y_new
        x = x_new

        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
            # Tighten margin
            margin = max(margin * 0.85, 1e-5)
        else:
            stall += 1

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            stall = 0

        # Restart
        if stall > 200:
            stall = 0
            margin = 5e-3
            cur_K = min(top_K + 50, 500)
            # Perturb
            pert = rng.normal(0, 0.01, n) * best_a
            y_r = best_a + pert
            y_r = np.maximum(y_r, 1e-10)
            y_r = y_r / y_r.sum()
            y = y_r
            x = y_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LPFW restart t={t} ({elapsed:.0f}s): J={J:.6f}, K={cur_K}")

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LPFW t={t} ({elapsed:.0f}s): J={J:.6f}, K={len(active_idx)}, margin={margin:.1e}")

    return best_a, best_m


def peak_elimination(a0, time_limit, verbose=True, seed=42):
    """
    Targeted peak elimination: identify the dominant peak in the autocorrelation
    and redistribute mass to reduce it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        top_peaks = np.argsort(ac)[::-1][:10]
        
        improved = False
        for peak_k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(peak_k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue

            # Find the largest contributors
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]
            
            for li in top_local:
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                best_new_m = M_curr
                best_move = None

                # Try moving fraction to random destinations
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    for _ in range(40):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)

                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakElim ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakElim final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def freq_perturb(a, strength=0.01, rng=None):
    """Perturb in frequency domain for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Combines:
    1. Diverse parametric initializations
    2. Fast mirror descent for initial optimization
    3. LP Frank-Wolfe for precise optimization
    4. Peak elimination for final polishing
    5. Multi-length exploration
    6. Frequency-domain diversification
    """
    t_global = time.monotonic()
    safety_margin = 60.0  # seconds to reserve for final evaluation
    time_budget = min(float(budget_s) - safety_margin, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            best_seq = a_norm.copy()
            best_score = sc

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sin2')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Try diverse initializations ===
    print(f"\n=== Phase 1: Diverse init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    families = ['sin2', 'sin4', 'raised_cosine', 'beta', 'beta2', 'bimodal', 'asym']
    for fam in families:
        if time.monotonic() - t_global > time_budget - 100:
            break
        candidate = make_parametric(n0, fam)
        try_update(candidate, f"init_{fam}")

    # === Phase 2: Fast mirror descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t2, top_k=60, sharpness=4.0,
                                            c0=3.5, restart_every=3000, verbose=True)
        try_update(a_out, "md_main")

    # === Phase 3: LP Frank-Wolfe (primary optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.28, 480.0)
    if t3 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, top_K=300, margin_frac=2e-3,
                                   verbose=True, use_iqcd=True)
        try_update(a_out, "lp_main")

    # === Phase 4: Peak elimination ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak elimination (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_elimination(a_in, t4, verbose=True, seed=42)
        try_update(a_out, "peak1")

    # === Phase 5: Second LP with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW tight (n={n_best}) ===")
    t5 = min(remaining * 0.18, 300.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, top_K=400, margin_frac=5e-4,
                                   verbose=True, use_iqcd=True)
        try_update(a_out, "lp_tight")

    # === Phase 6: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 200:
        longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
        longer_lengths = [l for l in longer_lengths if l <= 6000]
        
        for new_n in longer_lengths[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 150:
                break
            print(f"\n=== Phase 6: Extend to n={new_n} ===")
            a_long = interpolate_sequence(best_seq, new_n)
            
            t6a = min(remaining * 0.25, 200.0)
            if t6a > 30:
                a_out, _ = mirror_descent_optimize(a_long, t6a, top_k=50, sharpness=4.0,
                                                    c0=3.0, restart_every=2000, verbose=False)
                try_update(a_out, f"md_n{new_n}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 100 and new_n <= 4500:
                t6b = min(remaining * 0.20, 150.0)
                if t6b > 20:
                    a_out, _ = lp_frank_wolfe(a_long, t6b, solver, top_K=200,
                                               margin_frac=2e-3, verbose=False)
                    try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Freq diversify (n={n_best}) ===")
    rng = np.random.default_rng(123)
    n_trials = min(3, int(remaining / 80)) if remaining > 80 else 0
    for trial in range(n_trials):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
        t7 = min(remaining * 0.3, 150.0)
        if t7 > 30:
            a_out, _ = mirror_descent_optimize(a_pert, t7, top_k=50, sharpness=5.0,
                                                c0=4.0, restart_every=1500, verbose=False)
            try_update(a_out, f"freq{trial}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) <= 4000:
            t7b = min(remaining * 0.3, 120.0)
            if t7b > 20:
                a_out, _ = lp_frank_wolfe(a_pert, t7b, solver, top_K=250,
                                           margin_frac=1e-3, verbose=False)
                try_update(a_out, f"lp_freq{trial}")

    # === Phase 8: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t8, solver, top_K=500, margin_frac=2e-4,
                                   verbose=True, use_iqcd=True)
        try_update(a_out, "lp_final")

    # === Phase 9: Final peak elimination ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final peak elim (n={len(best_seq)}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.4, 150.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_elimination(a_in, t9, verbose=True, seed=999)
            try_update(a_out, "peak_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
