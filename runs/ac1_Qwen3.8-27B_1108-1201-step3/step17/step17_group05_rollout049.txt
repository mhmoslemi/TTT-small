    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective_value(a):
    """Compute 2*n*max(conv(a,a))*(sum(a))^2 ... wait, the objective is 2*n*max/(sum^2).
    If sum=1, it's just 2*n*max(conv(a,a))."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def soft_subgrad(a, beta=30.0):
    """Compute soft subgradient of max(conv(a,a))."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    shifted = ac - M
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / ws
    # g[j] = 2 * sum_m w[m] * a[m-j]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad, M

def exact_subgrad_topk(a, k=10):
    """Compute subgradient using only top-k lags (exact subgradient of max)."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    top_idx = np.argsort(ac)[::-1][:k]
    w = np.zeros(len(ac))
    w[top_idx] = 1.0 / k
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad, M, top_idx

def subgrad_optimize(a0, time_limit, beta_start=20.0, beta_end=80.0,
                     c0=5.0, restart_every=3000, seed=42, verbose=True):
    """Nesterov-accelerated subgradient descent with adaptive beta."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Adaptive beta: increase over time for sharper focus
        progress = min(1.0, iters / (restart_every * 2))
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Nesterov extrapolation
        alpha_n = (iters - 1.0) / (iters + 2.0)
        y = x + alpha_n * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        g, M_curr = soft_subgrad(y, beta=beta)
        
        # Step size
        eta = c0 / (np.sqrt(iters) + 1.0)
        
        # Mirror descent update (exponential family / simplex)
        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x.copy()
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            # Perturb around best
            noise_scale = max(0.005, 0.02 * (stall / restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_new = best_a + pert
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    SG restart ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, iter={iters}")
        
        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SG iter={iters} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

def coord_descent_optimize(a0, time_limit, seed=123, verbose=True):
    """Targeted coordinate descent with analytical delta optimization."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top contributing coordinates to peak lags
        n_top_lags = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top_lags]
        
        # Compute "cost" of each coordinate: how much it contributes to peaks
        cost = np.zeros(n)
        for m in top_lags[:10]:
            mi = int(m)
            js = max(0, mi - n + 1)
            je = min(n - 1, mi)
            idx = np.arange(js, je + 1)
            cost[idx] += a[idx] * a[mi - idx]
        
        # Top "expensive" coordinates
        n_expensive = min(50, n // 10)
        expensive_idx = np.argsort(cost)[::-1][:n_expensive]
        
        # Compute "sensitivity" of each coordinate: how much mass there
        # would affect peaks
        sens = np.zeros(n)
        for m in top_lags[:10]:
            mi = int(m)
            js = max(0, mi - n + 1)
            je = min(n - 1, mi)
            for d in range(js, je + 1):
                partner = mi - d
                sens[d] += a[partner]
        
        # Cheap coordinates (low sensitivity)
        cheap_idx = np.argsort(sens)[:n_expensive]
        
        # Try moves from expensive to cheap
        improvement_found = False
        
        for i in expensive_idx[:20]:
            if time.monotonic() - t0 > time_limit:
                    break
            if a[i] < 1e-12:
                continue
            
            for j in cheap_idx[:20]:
                if j == i:
                    continue
                
                # Compute optimal delta analytically
                # Change in ac[m] = delta * (a[m-j] - a[m-i])
                # We want to find delta that minimizes max(ac + delta * delta_vec)
                # where delta_vec[m] = a[m-j] - a[m-i]
                
                # Build delta_vec
                delta_vec = np.zeros(2 * n - 1)
                # Contribution from removing mass at i: -delta * a[m-i] for valid m-i
                for mi in range(n):
                    m = mi + i  # m - i = mi => m = mi + i
                    if 0 <= m < 2 * n - 1:
                        delta_vec[m] -= a[mi]
                # Contribution from adding mass at j: +delta * a[m-j]
                for mi in range(n):
                    m = mi + j
                    if 0 <= m < 2 * n - 1:
                        delta_vec[m] += a[mi]
                
                # New ac = ac + delta * delta_vec (approximately, for small delta)
                # We want: max(ac + delta * delta_vec) minimized
                # This is: find delta in [0, a[i]] that minimizes the max
                
                # The max increases where delta_vec > 0 and decreases where delta_vec < 0
                # Optimal delta is where the max of (ac + delta * delta_vec) is minimized
                
                # For simplicity, try a few delta values
                ai = a[i]
                for frac in [0.01, 0.005, 0.02, 0.002, 0.05]:
                    delta = ai * frac
                    if delta < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improvement_found = True
                        break
            
            if improvement_found:
                stall = 0
        
        if not improvement_found:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    CD final ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_M

def simulated_annealing_optimize(a0, time_limit, seed=99, verbose=True):
    """Simulated annealing with pair moves."""
    n = len(a0)

    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_end = 0.00001
    patience = 5000
    stall = 0

        # Annealing schedule
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Random pair move
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:

        frac = rng.uniform(0.001, 0.02)
        amt = a[i] * frac

        a_trial[j] += amt

        a_trial = a_trial / a_trial.sum()

                best_M = M_trial

                stall += 1

            if rng.random() < np.exp(-delta_M / T):
                a = a_trial
                stall += 1
            else:
                stall += 1

Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive subgradient descent, analytical coordinate descent,
simulated annealing, and multi-scale length exploration.
"""

import numpy as np
import time

# === Core utilities ===

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

# === Subgradient methods ===

def soft_subgrad(a, beta):
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    w = np.exp(beta * (ac - M))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w /= ws
    # g[j] = 2 * sum_m w[m] * a[m-j]
    fft_size = 1 << (n + len(w) - 2).bit_length()
    ar = np.zeros(fft_size); ar[:n] = a[::-1]
    wp = np.zeros(fft_size); wp[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(ar) * np.fft.rfft(wp), fft_size)
    return 2.0 * conv[n-1:2*n-1], M

def subgrad_descent(a0, tlimit, beta0=15, beta1=100, c0=4.0,

        g, _ = soft_subgrad(y, beta)
        eta = c0 / (np.sqrt(it) + 1.0)
        ln = np.log(y) - eta * g
        ln -= ln.max()
        x_new = np.exp(ln)
        x_new = np.maximum(x_new, 1e-30)
        x_new /= x_new.sum()
        x_prev = x
        x = x_new

            stall = 0
        else:

        if stall > restart_every:

            x = np.maximum(best_a + p, 1e-10)

def coord_descent(a0, tlimit, seed=0, verbose=False):

    a = norm_sum1(a0.copy())
    best_a, best_M = a.copy(), fast_autocorr(a).max()

            if 0 <= m < 2*n-1:

                e2 = min(n-1, m)
                for d in range(s2, e2+1):
                    sens[d] += a[m - d]

        top_cheap = np.argsort(sens)[:30]
        found = False
        for i in top_exp[:15]:

                if j == i: continue
                # Try several fractions
                for frac in [0.01, 0.005, 0.02, 0.002, 0.03]:
                    d = a[i] * frac
                    if d < 1e-15: continue
                    at = a.copy()
                    at[i] -= d
                    at[j] += d
                    if at[i] < 0: continue
                    at /= at.sum()
                    Mt = fast_autocorr(at).max()
                    if Mt < best_M - 1e-14:
                        a, best_M, best_a = at, Mt, at.copy()
                        found = True
                        break
                if found: break
            if found:
                stall = 0
        if not found:
            stall += 1
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    d = a[i] * rng.uniform(0.002, 0.01)
                    a[i] -= d; a[j] += d
                    a = np.maximum(a, 0); a /= a.sum()
                    M = fast_autocorr(a).max()
                    if M < best_M:
                        best_M, best_a = M, a.copy()
                        stall = 0
        if verbose and stall % 200 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={2*n*best_M:.6f}, stall={stall}")
    return best_a, best_M
