def _fft_size(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = _fft_size(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

def compute_objective_norm(a):
    """Compute 2*n*max(ac) assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def compute_smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*ac_k)) w.r.t. a.
    Returns (gradient, max_ac, logsumexp_val).
    Uses full softmax over all 2n-1 lags.

    ac = fast_autocorr(a)
    M = ac.max()

    shifted = ac - M

    # Inverse FFT to get real sequence
    full_spec = np.zeros(N // 2 + 1)
    full_spec[:min(n // 2 + 1, N // 2 + 1)] = spectrum[:min(n // 2 + 1, N // 2 + 1)]
    a = np.fft.irfft(full_spec, N)[:n]
    return normalize_nonneg(a)

# ---- Main smooth gradient descent ----

def smooth_gd(a0, time_limit, seed=42, beta_start=2.0, beta_end=80.0,
              eta_start=1.0, momentum=0.9, restart_every=2000,
              verbose=False, max_iter=5000000):
    """
    Nesterov-accelerated mirror descent on log-sum-exp smooth objective.
    Anneals temperature from beta_start to beta_end.
    """
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_J = compute_objective_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    mu = momentum
    stall = 0
    total_iters = 0

    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        progress = min(1.0, t / (max_iter * 0.8))
        beta = beta_start + (beta_end - beta_start) * progress

        # Nesterov extrapolation
        if t > 1:
            beta_t = (mu + (1 - mu) * (t - 1) / t)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()

        # Ensure y is valid (non-negative, sum=1)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient
        grad, M_curr, lse = compute_smooth_grad(y, beta)

        # Adaptive step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            eta = eta_start
        else:
            # Polyak-like: use gap between current and lower bound
            # Lower bound on smooth obj is roughly 2*n*(1/n) = 2
            # But we work on sum=1 simplex, objective is 2*n*max_ac
            # Use a simpler heuristic
            obj_curr = 2.0 * n * M_curr
            lower = 2.0  # theoretical minimum
            gap = max(obj_curr - lower, 0.01)
            eta = 0.5 * gap / (grad_norm_sq + 1e-10)
            eta = min(eta, eta_start * 2.0 / (1.0 + 0.01 * t))
            eta = max(eta, 1e-12)

        # Mirror descent step (exponential)
        log_y = np.log(y)
        log_a_new = log_y - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        total_iters += 1

        # Track best
        J_curr = compute_objective_norm(x)
        if J_curr < best_J - 1e-14:
            best_J = J_curr
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with structured perturbation
        if stall > restart_every:
            stall = 0
            # Structured restart: multiply by random smooth positive function
            n_knots = rng.integers(5, 15)
            knots = np.sort(rng.uniform(0, 1, n_knots))
            vals = rng.uniform(0.3, 2.0, n_knots)
            mult = np.interp(np.linspace(0, 1, n), knots, vals)
            x_r = best_a * mult
            # Add small random component
            x_r += rng.uniform(0, 0.05, n) * best_a
            x_r = normalize_nonneg(x_r)
            x = x_r
            x_prev = x_r.copy()

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      SGD restart ({elapsed:.0f}s): J={best_J:.7f}, beta={beta:.1f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD t={t} ({elapsed:.0f}s): J={best_J:.7f}, beta={beta:.1f}, eta={eta:.4f}")

    return best_a, best_J

# ---- Coordinate moves targeting peak lags ----

def peak_coord_moves(a0, time_limit, seed=77, verbose=False):
    """
    Targeted coordinate moves: for each of the top peak lags,
    identify high-contribution coordinates and move mass to low-contribution ones.
    Uses quadratic approximation for line search.
    """
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_J = compute_objective_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        J_curr = 2.0 * n * M_curr

        # Get top peak lags
        n_peaks = min(8, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity for each coordinate
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            # Contribution of a[idx[i]] to ac[k] is a[j_idx[i]]
            contrib = a[idx] * a[j_idx]
            for ii in range(len(idx)):
                sensitivity[idx[ii]] += contrib[ii]
                sensitivity[j_idx[ii]] += contrib[ii]

        # Find high and low sensitivity positions with meaningful mass
        thresh_high = sensitivity.mean() + 0.5 * sensitivity.std()
        thresh_low = sensitivity.mean() - sensitivity.std()
        high_pos = np.where((sensitivity > thresh_high) & (a > 1e-6))[0]
        low_pos = np.where((sensitivity < thresh_low) | (sensitivity == 0))[0]

        improved = False

        if len(high_pos) > 0 and len(low_pos) > 0:
            # Try multiple candidate moves
            n_tries = min(50, len(high_pos) * len(low_pos) // 10 + 5)
            for _ in range(n_tries):
                if time.monotonic() - t0 > time_limit:
                    break

                src = high_pos[rng.integers(len(high_pos))]
                dst = low_pos[rng.integers(len(low_pos))]
                if src == dst or a[src] < 1e-8:
                    continue

                # Try a range of transfer amounts via golden section
                max_amt = a[src] * 0.5
                best_frac = 0.1
                best_trial_J = compute_objective_norm(a)

                # Coarse search
                fracs = [0.01, 0.03, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6]
                for frac in fracs:
                    amt = a[src] * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t /= st
                    J_t = compute_objective_norm(a_t)
                    if J_t < best_trial_J:
                        best_trial_J = J_t
                        best_frac = frac

                # Apply best move
                if best_trial_J < best_J - 1e-14 and best_frac > 0:
                    amt = a[src] * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    best_J = best_trial_J
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            # Escalating random escape
            if stall % 100 == 0:
                n_pert = max(3, n // 200)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for p in pert_idx:
                    if a[p] > 1e-8:
                        amt = a[p] * rng.uniform(0.05, 0.3)
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                J_new = compute_objective_norm(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakCoord ({elapsed:.0f}s): J={best_J:.7f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      PeakCoord done ({elapsed:.0f}s): J={best_J:.7f}")
    return best_a, best_J
