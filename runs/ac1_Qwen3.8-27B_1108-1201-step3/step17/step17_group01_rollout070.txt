```
import numpy as np
import time

def _next_pow2(m):
    if m <= 1: return 1
    return 1 << (int(m)-1).bit_length()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones_like(a)/len(a)
    return a/s

def _autocorr(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2*n-1
    if fft_size is None or fft_size < need:
        fft_size = _next_pow2(need)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A*A, fft_size)[:need]
    return c, fft_size

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n=a.size
    c,_ = _autocorr(a)
    s=a.sum()
    if s <= 1e-300: return np.inf
    return float(2.0*n*max(0.0, float(c.max()))/(s*s))
```

```
    if a.ndim != 1 or a.size == 0:
        return np.ones(1)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64)/a.size
    a = a/s
    return a
```

```
def _autocorr(a, fft_size=None):
```

```
def _init_edge_bumps(n, alpha=2.0, sigma_frac=0.05):
    if n <= 1: return _init_uniform(n)
    i = np.arange(n, dtype=np.float64)
    sigma = max(1.0, sigma_frac * n)
    w = 1.0 + alpha * np.exp(-(i + 1.0) / sigma) + alpha * np.exp(-(n - i) / sigma)
    return _normalize(w)
```

```
def _init_random_smooth(n, seed=0, modes=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0,1,n)
    w = np.ones(n, dtype=np.float64)
    for f in range(1, modes+1):
        amp = rng.uniform(0.05, 0.25) / (f ** 0.5)
        ph = rng.uniform(0, 2*np.pi)
        w += amp * np.sin(2*np.pi*f*x + ph)
    w = np.maximum(w, 0.01)
    return _normalize(w)
```

```
def _init_random_spectral(n, seed=0, iters=30, alpha_start=0.7):
    rng = np.random.default_rng(seed)
    a = np.ones(n) + rng.normal(0.0, 0.25, n)
    a = _normalize(a)
    F = _next_pow2(2*n - 1)
    alpha = alpha_start
    for _ in range(iters):
        A = np.fft.rfft(a, F)
        mag = np.abs(A)
        mean = mag.mean()
        if mean <= 0: break
        mag = mean * np.power(mag / max(mean,1e-300), 1.0 - alpha)
        A = mag * np.exp(1j * np.angle(A))
        a = np.fft.irfft(A, F)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 1e-300: break
        a = a / s
        alpha *= 0.98
        if alpha < 0.1: alpha = 0.1
    return _normalize(a)
```

```
def _time_mw(a0, time_limit, seed=1, beta_start=80.0, beta_end=20.0, c_start=2.5, c_end=0.8,
             momentum=0.3, restart_iters=10000, min_time=1.0):
    a = _normalize(a0)
    n = a.size
    if n <= 1 or time_limit <= 1e-3:
        return a, _score(a)
    F = _next_pow2(3*n - 2)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(seed)
    v = np.zeros(n) if momentum > 0 else None
    it = 0; last_improve = 0
    noise_level = 0.03
    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-9))
        beta = beta_start + (beta_end - beta_start) * frac
        step = c_start + (c_end - c_start) * frac
        A = np.fft.rfft(a, F)
        c = np.fft.irfft(A * A, F)[:2*n - 1]
        M = float(c.max())
        if M <= 1e-300:
            break
        score = 2.0 * n * M  # normalized
        if score < best_score - 1e-13:
            best_score = score; best_a = a.copy(); last_improve = it
        if it - last_improve > restart_iters:
            a = best_a + rng.normal(0.0, noise_level, n) * np.maximum(best_a, 1e-6)
            a = _normalize(a)
            if v is not None: v.fill(0.0)
            last_improve = it
            noise_level = min(0.2, noise_level * 1.2)
            continue
        z = beta * (c - M) / M
        # clip lower to -60
        z = np.where(z < -60.0, -60.0, z)
        z -= z.max()
        w = np.exp(z)
        sw = w.sum()
        if sw <= 1e-300:
            w = np.ones(2*n - 1, dtype=np.float64) / (2*n - 1)
        else:
            w = w / sw
        W = np.fft.rfft(w, F)
        g = 2.0 * np.fft.irfft(W * np.conjugate(A), F)[:n]
        g = np.maximum(g, 0.0)
        if momentum > 0:
            v = momentum * v + (1.0 - momentum) * g
            use = v
        else:
            use = g
        use_mean = float(use.mean())
        use = use - use_mean
        pos_max = float(use.max()); neg_min = float(use.min())
        span = max(pos_max, -neg_min, 1e-300)
        eta = step / span
        if eta > 10.0: eta = 10.0
        loga = np.log(np.maximum(a, 1e-300)) - eta * use
        loga -= loga.max()
        loga = np.clip(loga, -700.0, 0.0)
        a_new = np.exp(loga)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new <= 1e-300:
            break
        a = a_new / s_new
    return best_a, best_score
```

```python
"""
Search for a non-negative step-function sequence minimizing

    2 * n * max(full_convolve(a, a)) / (sum(a))^2

by combining:
  * fast FFT-based exact subgradients of the maximum autocorrelation,
  * multiplicative-weights mirror steps in the time domain,
  * multiplicative-weights mirror steps in the frequency/power-spectrum domain,
  * symmetry averaging,
  * multiple diverse initializations and adaptive length extensions,
  * final polishing with alternating time/spectral updates.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _autocorr(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    if fft_size is None or fft_size < need:
        fft_size = _next_pow2(need)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:need]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf")
    c = _autocorr(a)
    m = float(c.max())
    if m <= 0.0:
        return 0.0
    return float(2.0 * a.size * m / (s * s))


def _true_score(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        return float(val)
    except Exception:
        return float("inf")


def _try_update_best(candidate, candidate_score, best_seq, best_score):
    if candidate is None:
        return best_seq, best_score
    if not np.isfinite(candidate_score):
        return best_seq, best_score
    if np.isfinite(best_score) and candidate_score > best_score + 1e-3:
        return best_seq, best_score
    tv = _true_score(candidate)
    if np.isfinite(tv) and tv < best_score:
        return np.asarray(candidate, dtype=np.float64).copy(), float(tv)
    return best_seq, best_score


def _improve_with_sym(a, sc):
    a = np.asarray(a, dtype=np.float64)
    if a.size <= 2:
        return a, sc
    asy = _normalize(0.5 * (a + a[::-1]))
    s2 = _score(asy)
    if s2 < sc - 1e-12:
        return asy, s2
    return a, sc


def _interpolate_to_length(seq, new_n):
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return np.ones(new_n, dtype=np.float64) / float(max(1, new_n))
    new_n = int(max(1, new_n))
    if seq.size == new_n:
        return _normalize(seq)
    old_x = np.linspace(0.0, 1.0, seq.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, seq)
    return _normalize(y)


def _init_uniform(n):
    n = int(max(1, n))
    return np.ones(n, dtype=np.float64) / float(n)


def _init_edge_bumps(n, alpha=2.0, sigma_frac=0.05):
    n = int(max(1, n))
    if n <= 1:
        return _init_uniform(n)
    i = np.arange(n, dtype=np.float64)
    sigma = max(1.0, float(sigma_frac) * n)
    w = (
        1.0
        + float(alpha) * np.exp(-(i + 1.0) / sigma)
        + float(alpha) * np.exp(-(n - i) / sigma)
    )
    return _normalize(w)


def _init_cos_edge(n, alpha=1.5, mode=1):
    n = int(max(1, n))
    if n <= 1:
        return _init_uniform(n)
    x = np.linspace(0.0, 1.0, n)
    w = 1.0 + float(alpha) * (0.5 + 0.5 * np.cos(np.pi * int(mode) * x)) ** 2
    return _normalize(np.maximum(w, 1e-6))


def _init_random_smooth(n, seed=0, modes=8):
    n = int(max(1, n))
    if n <= 1:
        return _init_uniform(n)
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    w = np.ones(n, dtype=np.float64)
    for f in range(1, int(modes) + 1):
        amp = rng.uniform(0.05, 0.25) / (f ** 0.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        w += amp * np.sin(2.0 * np.pi * f * x + ph)
    w = np.maximum(w, 0.01)
    return _normalize(w)


def _init_random_spectral(n, seed=0, iters=30, alpha_start=0.7):
    n = int(max(1, n))
    if n <= 1:
        return _init_uniform(n)
    rng = np.random.default_rng(seed)
    a = np.ones(n, dtype=np.float64) + rng.normal(0.0, 0.25, n)
    a = _normalize(a)
    F = _next_pow2(2 * n - 1)
    alpha = float(alpha_start)
    for _ in range(int(iters)):
        A = np.fft.rfft(a, F)
        mag = np.abs(A)
        mean = float(mag.mean())
        if mean <= 1e-300:
            break
        mag = mean * np.power(mag / mean, 1.0 - alpha)
        A = mag * np.exp(1j * np.angle(A))
        a = np.fft.irfft(A, F)[:n]
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s <= 1e-300:
            break
        a = a / s
        alpha *= 0.98
        if alpha < 0.1:
            alpha = 0.1
    return _normalize(a)


def _make_starts(n, parent=None, max_count=7):
    n = int(max(1, n))
    starts = []

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.size > 0:
            p = _normalize(p)
            starts.append(_interpolate_to_length(p, n))

    starts.append(_init_edge_bumps(n, 2.5, 0.04))
    starts.append(_init_edge_bumps(n, 1.2, 0.10))
    starts.append(_init_cos_edge(n, 2.0, 1))
    starts.append(_init_cos_edge(n, 1.0, 2))
    starts.append(_init_random_smooth(n, 11, 8))
    starts.append(_init_random_smooth(n, 37, 12))
    starts.append(_init_random_spectral(n, 57, 30, 0.7))
    starts.append(_init_random_spectral(n, 913, 35, 0.55))

    seen = set()
    unique = []
    for s in starts:
        s = _normalize(s)
        step = max(1, len(s) // 40)
        sig = tuple(np.round(s[::step], 6))
        if sig not in seen:
            seen.add(sig)
            unique.append(s)
    return unique[: int(max_count)]


def _time_mw(
    a0,
    time_limit,
    seed=1,
    beta_start=80.0,
    beta_end=20.0,
    c_start=2.5,
    c_end=0.8,
    momentum=0.3,
    restart_iters=10000,
):
    a = _normalize(a0)
    n = a.size
    if n <= 1:
        return a, _score(a)
    time_limit = float(max(0.0, time_limit))
    if time_limit <= 1e-3:
        return a, _score(a)

    F = _next_pow2(3 * n - 2)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(seed)
    v = np.zeros(n, dtype=np.float64) if float(momentum) > 0.0 else None
    it = 0
    last_improve = 0
    noise_level = 0.03

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-9))
        beta = float(beta_start) + (float(beta_end) - float(beta_start)) * frac
        step = float(c_start) + (float(c_end) - float(c_start)) * frac

        A = np.fft.rfft(a, F)
        c = np.fft.irfft(A * A, F)[: 2 * n - 1]
        M = float(c.max())
        if M <= 1e-300:
            break
        s = float(a.sum())
        score = (2.0 * n * M) / (s * s) if s > 0 else float("inf")

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            last_improve = it

        if it - last_improve > int(restart_iters):
            floor = max(1e-6, 1e-4 / n)
            a = best_a + rng.normal(0.0, noise_level, n) * np.maximum(best_a, floor)
            a = _normalize(a)
            if v is not None:
                v.fill(0.0)
            last_improve = it
            noise_level = min(0.25, noise_level * 1.25)
            continue

        z = beta * (c - M) / M
        z = np.clip(z, -60.0, 0.0)
        z -= z.max()
        w = np.exp(z)
        sw = float(w.sum())
        if sw <= 1e-300:
            w = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
        else:
            w = w / sw

        W = np.fft.rfft(w, F)
        g = 2.0 * np.fft.irfft(W * np.conjugate(A), F)[:n]
        g = np.maximum(g, 0.0)

        if momentum > 0.0:
            v = float(momentum) * v + (1.0 - float(momentum)) * g
            use = v
        else:
            use = g

        use = use - float(use.mean())
        span = max(float(use.max()), -float(use.min()), 1e-300)
        eta = step / span
        if eta > 6.0:
            eta = 6.0

        loga = np.log(np.maximum(a, 1e-300)) - eta * use
        loga -= loga.max()
        loga = np.clip(loga, -700.0, 0.0)
        a_new = np.exp(loga)
        a_new = np.maximum(a_new, 1e-300)
        s_new = float(a_new.sum())
        if s_new <= 1e-300:
            break
        a = a_new / s_new

    return best_a, best_score

            break
        s = float(a.sum())
        score = (2.0 * n * M) / (s * s) if s > 0 else float("inf")

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            last_improve = it

        if it - last_improve > int(restart_iters):
            floor = max(1e-6, 2e-4 / n)
            a = best_a + rng.normal(0.0, noise_level, n) * np.maximum(best_a, floor)
            a = _normalize(a)
            last_improve = it
            noise_level = min(0.30, noise_level * 1.25)
            continue

        z = beta * (c - M) / M
        z = np.clip(z, -60.0, 0.0)
        z -= z.max()
        w = np.exp(z)
        sw = float(w.sum())
        if sw <= 1e-300:
            w = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
        else:
            w = w / sw

        Wfull = np.fft.rfft(w, F)
        pgrad = np.real(Wfull).astype(np.float64, copy=True)
        if pgrad.size > 2:
            pgrad[1:-1] *= 2.0
        pgrad -= float(pgrad.mean())
        span = float(np.abs(pgrad).max())
        if span <= 1e-300:
            continue

        eta = step / span
        if eta > 4.0:
            eta = 4.0

        Pnew = P * np.exp(np.clip(-eta * pgrad, -700.0, 700.0))
        Pnew = np.maximum(Pnew, 0.0)
        if float(Pnew.sum()) <= 1e-300:
            break

        Anew = np.sqrt(Pnew) * np.exp(1j * np.angle(A))
        cand = np.fft.irfft(Anew, F)[:n]
        cand = np.maximum(cand, 0.0)
        sc = float(cand.sum())
        if sc <= 1e-300:
            break
        cand = cand / sc

        if it % int(max(1, score_every)) == 0:
            scand = _score(cand)
            if scand < score - 1e-12:
                a = cand
            else:
                blended = (1.0 - mix) * a + mix * cand
                blended = _normalize(blended)
                scblend = _score(blended)
                if scblend < score:
                    a = blended
                elif rng.random() < 0.015:
                    a = cand
        else:
            a = (1.0 - mix) * a + mix * cand
            a = _normalize(a)

    return best_a, best_score


def _optimize_block(a0, time_limit, seed=1):
    a = _normalize(a0)
    sc = _score(a)
    time_limit = float(max(0.0, time_limit))
    if time_limit <= 2.0:
        return a, sc

    t1 = time_limit * 0.38
    if t1 > 4.0:
        a1, s1 = _time_mw(
            a,
            t1,
            seed=seed,
            beta_start=90.0,
            beta_end=25.0,
            c_start=2.8,
            c_end=0.9,
            momentum=0.0,
            restart_iters=12000,
        )
        if s1 < sc:
            a, sc = a1, s1
        a, sc = _improve_with_sym(a, sc)

    t2 = time_limit * 0.28
    if t2 > 4.0:
        a1, s1 = _spectral_mw(
            a,
            t2,
            seed=seed + 1,
            beta_start=85.0,
            beta_end=22.0,
            step_start=1.2,
            step_end=0.35,
            mix_start=0.75,
            mix_end=0.35,
            restart_iters=6000,
            score_every=5,
        )
        if s1 < sc:
            a, sc = a1, s1
        a, sc = _improve_with_sym(a, sc)

    t3 = time_limit * 0.24
    if t3 > 4.0:
        a1, s1 = _time_mw(
            a,
            t3,
            seed=seed + 2,
            beta_start=65.0,
            beta_end=18.0,
            c_start=2.2,
            c_end=0.7,
            momentum=0.35,
            restart_iters=9000,
        )
        if s1 < sc:
            a, sc = a1, s1
        a, sc = _improve_with_sym(a, sc)

    t4 = time_limit * 0.10
    if t4 > 3.0:
        a1, s1 = _spectral_mw(
            a,
            t4,
            seed=seed + 3,
            beta_start=60.0,
            beta_end=16.0,
            step_start=0.9,
            step_end=0.28,
            mix_start=0.6,
            mix_end=0.3,
            restart_iters=4000,
            score_every=4,
        )
        if s1 < sc:
            a, sc = a1, s1
        a, sc = _improve_with_sym(a, sc)

    return a, sc


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = min(float(budget_s) - 45.0, 1755.0)
    if total_budget <= 30.0:
        total_budget = max(10.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")
    search_seq = None
    search_score = float("inf")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = p.reshape(-1)
        if (
            p.ndim == 1
            and p.size > 0
            and np.isfinite(p).all()
            and (np.maximum(p, 0.0).sum() > 1e-10)
        ):
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 12000:
                p = _interpolate_to_length(p, 8000)
            parent = p.copy()
    except Exception:
        parent = None

    if parent is not None:
        n_base = int(parent.size)
        pnorm = _normalize(parent)
        search_seq = pnorm
        search_score = _score(pnorm)
        best_score = _true_score(pnorm)
        best_seq = pnorm.copy()
        if not np.isfinite(best_score):
            best_score = search_score
    else:
        n_base = 4000
        search_seq = _init_edge_bumps(n_base, 2.0, 0.06)
        search_score = _score(search_seq)
        best_score = _true_score(search_seq)
        best_seq = search_seq.copy()
        if not np.isfinite(best_score):
            best_score = search_score

    print(
        f"initial: n={len(search_seq)}, true={best_score:.6f}, internal={search_score:.6f}"
    )

    # Phase 1: strong refinement of the base length.
    phase1_budget = min(total_budget * 0.42, 720.0)

    t = min(phase1_budget * 0.32, 240.0)
    if t > 20.0:
        a, sc = _optimize_block(search_seq, t, seed=1)
        if sc < search_score:
            search_seq, search_score = a, sc
        best_seq, best_score = _try_update_best(
            search_seq, search_score, best_seq, best_score
        )
        print(
            f"phase1 base block: true={best_score:.6f}, internal={search_score:.6f}"
        )

    starts = _make_starts(n_base, parent, max_count=8)
    for idx, st in enumerate(starts):
        if idx == 0:
            continue
        if idx > 4:
            break
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining < 70.0:
            break
        tblock = min(110.0, remaining * 0.12)
        if tblock < 25.0:
            break
        a, sc = _optimize_block(st, tblock, seed=20 + idx)
        if sc < search_score:
            search_seq, search_score = a, sc
        best_seq, best_score = _try_update_best(a, sc, best_seq, best_score)
        print(
            f"phase1 start {idx}: candidate internal={sc:.6f}, best true={best_score:.6f}"
        )

    # Phase 2: try somewhat longer sequences.
    longer_targets = []
    for target in [
        max(int(n_base) + 700, 4800),
        max(int(n_base) + 1500, 5800),
        max(int(n_base) + 2400, 6800),
    ]:
        target = int(min(int(target), 8500))
        if target > int(n_base) and target not in longer_targets:
            longer_targets.append(target)

    for target in longer_targets:
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining < 130.0:
            break
        if target <= len(search_seq):
            continue
        print(f"trying longer length n={target}")
        init = _interpolate_to_length(search_seq, target)
        t1 = min(150.0, remaining * 0.18)
        a, sc = _optimize_block(init, t1, seed=100 + (target % 100))
        if sc < search_score:
            search_seq, search_score = a, sc
        best_seq, best_score = _try_update_best(a, sc, best_seq, best_score)
        print(
            f"longer n={target}: internal={sc:.6f}, best true={best_score:.6f}"
        )

        remaining = total_budget - (time.monotonic() - t_global)
        if remaining > 90.0 and sc < search_score + 1e-6:
            t2 = min(80.0, remaining * 0.12)
            a2, sc2 = _optimize_block(a, t2, seed=300 + (target % 100))
            if sc2 < search_score:
                search_seq, search_score = a2, sc2
            best_seq, best_score = _try_update_best(a2, sc2, best_seq, best_score)
            print(
                f"longer2 n={target}: internal={sc2:.6f}, best true={best_score:.6f}"
            )

    # Phase 3: final alternating polish at the best currently known length.
    seed = 1000
    alt = 0
    while True:
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining < 35.0:
            break

        if alt % 2 == 0:
            tchunk = min(remaining * 0.35, 120.0)
            a, sc = _time_mw(
                search_seq,
                tchunk,
                seed=seed,
                beta_start=50.0,
                beta_end=12.0,
                c_start=1.6,
                c_end=0.5,
                momentum=0.3,
                restart_iters=8000,
            )
        else:
            tchunk = min(remaining * 0.30, 100.0)
            a, sc = _spectral_mw(
                search_seq,
                tchunk,
                seed=seed + 1,
                beta_start=60.0,
                beta_end=18.0,
                step_start=0.9,
                step_end=0.28,
                mix_start=0.65,
                mix_end=0.32,
                restart_iters=5000,
                score_every=4,
            )

        if sc < search_score:
            search_seq, search_score = a, sc
        search_seq, search_score = _improve_with_sym(search_seq, search_score)
        best_seq, best_score = _try_update_best(
            search_seq, search_score, best_seq, best_score
        )

        if alt % 2 == 1:
            print(
                f"final polish alt={alt}: best true={best_score:.6f}, "
                f"internal={search_score:.6f}, n={len(search_seq)}"
            )

        seed += 17
        alt += 1

    # Final validation and fallback.
    if search_seq is not None:
        ss = _score(search_seq)
        best_seq, best_score = _try_update_best(search_seq, ss, best_seq, best_score)

    if best_seq is None:
        best_seq = search_seq if search_seq is not None else _init_uniform(2000)

    arr = np.asarray(best_seq, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 0.01:
        arr = _normalize(_init_uniform(2000))
        s = float(arr.sum())
    else:
        arr = arr / s
        arr = np.maximum(arr, 0.0)

    final_true = _true_score(arr)
    if not np.isfinite(final_true):
        arr = _init_uniform(2000)
        final_true = 2.0

    print(
        f"FINAL: score={final_true:.6f}, n={len(arr)}, "
        f"elapsed={time.monotonic() - t_global:.1f}s"
    )
    return [float(x) for x in arr]
```
