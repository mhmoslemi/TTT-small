import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_peak_gradient(a, ac, top_k=100, beta=4.0, temperature=1.0):
    """
    Compute a soft gradient that emphasizes the top peaks.
    Uses a softmax-like weighting over peaks with temperature control.
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weighting with temperature
    scaled = top_vals / max(temperature, 1e-10)
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def adam_optimize_log(a0, time_limit, top_k=80, lr=0.01, beta1=0.9, beta2=0.999,
                      eps=1e-8, temperature=1.0, adaptive_temp=True, verbose=True, rng_seed=42):
    """
    Adam optimizer in log-domain. Parametrize a = softmax(w) * sum_target.
    This naturally handles non-negativity and the simplex constraint.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Initialize log-domain: w = log(a) (shifted to avoid -inf)
    a_safe = np.maximum(a, 1e-15)
    w = np.log(a_safe)
    
    # Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    cur_temp = temperature
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute softmax to get a from w
        w_shift = w - w.max()
        a_new = np.exp(w_shift)
        a_new = a_new / a_new.sum()
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(a_new)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a_new.copy()
        
        # Adaptive temperature
        if adaptive_temp and t % 500 == 0:
            cur_temp = max(0.1, temperature * 0.95 ** (t // 500))
        
        # Compute gradient in a-space
        g_a = compute_peak_gradient(a_new, ac, top_k=top_k, beta=4.0, temperature=cur_temp)
        
        # Transform to w-space: dw = g_a / a_new (Jacobian of log)
        g_w = g_a / np.maximum(a_new, 1e-15)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_w
        v = beta2 * v + (1 - beta2) * g_w**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AdamLog t={t} ({elapsed:.0f}s): J={J:.6f}, temp={cur_temp:.3f}")
    
    return best_a, best_m


def spectral_shaping(a0, time_limit, verbose=True, rng_seed=13):
    """
    Directly optimize in the spectral domain.
    The power spectrum S(f) = |A(f)|^2 determines the autocorrelation.
    We want to minimize max autocorrelation = max of inverse DFT of S(f).
    
    Strategy: iteratively adjust the magnitude of A(f) to flatten the 
    resulting autocorrelation, while maintaining non-negativity of a.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Get initial spectrum
    A = np.fft.rfft(a, fft_size)
    
    lr_spec = 0.1
    gamma = 0.01  # non-negativity enforcement strength
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current autocorrelation
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        
        # Peaks in autocorrelation domain
        top_k = np.argsort(ac)[::-1][:200]
        
        # For each peak, we want to reduce it.
        # In spectral domain, reducing peak k means reducing the 
        # contribution at frequency k. The phase of exp(2*pi*i*k*f)
        # tells us which spectral bins contribute.
        
        # Alternative approach: use the gradient in spectral domain
        # The autocorrelation at lag k is: r(k) = sum_f S(f) * exp(2*pi*i*k*f)
        # To reduce r(k), we want to reduce S(f) where exp(2*pi*i*k*f) is positive (aligned)
        
        # Simpler: compute gradient of max(r) w.r.t. |A(f)|^2
        # Then update |A| in that direction
        
        # For each top peak, compute the "phase alignment" with each frequency bin
        # freq bin m has frequency m/fft_size
        freqs = np.arange(fft_size // 2 + 1) / fft_size
        
        # For peak at lag k, the phase is exp(2*pi*i*k*m/fft_size)
        # The real part tells us the sign of contribution
        
        # Build gradient for |A|^2
        grad_S = np.zeros(len(A))
        for k in top_k[:50]:
            ki = int(k)
            phase = np.cos(2 * np.pi * ki * freqs)
            # Positive phase means this frequency contributes positively to this peak
            # We want to reduce S where phase is positive
            grad_S += phase / len(top_k[:50])
        
        # Update |A|: reduce where grad_S > 0 (aligned with peaks)
        mag = np.abs(A)
        phase_A = np.angle(A)
        
        # Multiplicative update on magnitude
        delta = lr_spec * grad_S
        mag_new = mag * np.exp(-delta)
        mag_new = np.maximum(mag_new, 1e-10)
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase_A)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Enforce non-negativity (soft)
        a_new = np.maximum(a_new, 0.0)
        
        # Also add a small amount of the "target" shape to prevent collapse
        # Target: something with low autocorrelation peak
        a_new = a_new + gamma * best_a
        a_new = a_new / a_new.sum()
        
        # Check if this improved
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        
        if M_new < M_curr:
            a = a_new
            A = np.fft.rfft(a, fft_size)
        else:
            # Adaptive: reduce step size
            lr_spec *= 0.95
            # Try with smaller step
            if lr_spec < 1e-4:
                # Perturb
                a_pert = a + rng.normal(0, 0.005, n) * a
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
                A = np.fft.rfft(a, fft_size)
                lr_spec = 0.1
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral t={t} ({elapsed:.0f}s): J={J:.6f}, lr={lr_spec:.4f}")
    
    return best_a, best_m


def population_search(a0, time_limit, pop_size=5, verbose=True, rng_seed=55):
    """
    Maintain a small population of candidate sequences.
    Mutate using various operators, select based on fitness.
    """
    n = len(a0)
    population = [norm_sum1(a0)]
    scores = [fast_autocorr(population[0]).max()]
    
    while len(population) < pop_size:
        # Create diverse initial population
        pert = rng.normal(0, 0.05, n) * norm_sum1(a0)
        new_member = norm_sum1(a0 + pert)
        population.append(new_member)
        scores.append(fast_autocorr(new_member).max())
    
    best_a = population[0].copy()
    best_m = scores[0]
    
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        op = iter_count % 4
        
        if op == 0:
            # Gaussian perturbation on best
            parent = population[np.argmin(scores)]
            sigma = 0.02 + 0.01 * (1 - (time.monotonic() - t0) / time_limit)
            child = parent + rng.normal(0, sigma, n) * parent
            child = np.maximum(child, 0)
            child = child / child.sum()
        elif op == 1:
            # Crossover between two best
            i, j = np.argsort(scores)[:2]
            if i == j:
                i = np.random.randint(pop_size)
            alpha = rng.uniform(0.2, 0.8)
            child = alpha * population[i] + (1 - alpha) * population[j]
            child = np.maximum(child, 0)
            child = child / child.sum()
        elif op == 2:
            # Targeted peak reduction
            parent = population[np.argmin(scores)]
            ac = fast_autocorr(parent)
            top_peak = int(np.argmax(ac))
            s = max(0, top_peak - n + 1)
            e = min(n - 1, top_peak)
            idx = np.arange(s, e + 1)
            contrib = parent[idx] * parent[top_peak - idx]
            src = idx[np.argmax(contrib)]
            dst = int(rng.integers(n))
            if dst == src:
                dst = (dst + 1) % n
            frac = rng.uniform(0.01, 0.08)
            child = parent.copy()
            amt = child[src] * frac
            child[src] -= amt
            child[dst] += amt
            child = np.maximum(child, 0)
            child = child / child.sum()
        else:
            # Spectral perturbation
            parent = population[np.argmin(scores)]
            fft_size = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(parent, fft_size)
            mag = np.abs(A) * (1 + rng.normal(0, 0.05, len(A)))
            mag = np.maximum(mag, 0)
            A_new = mag * np.exp(1j * np.angle(A))
            child = np.fft.irfft(A_new, fft_size)[:n]
            child = np.maximum(child, 0)
            child = child / child.sum()
        
        child_m = fast_autocorr(child).max()
        
        if child_m < best_m - 1e-15:
            best_m = child_m
            best_a = child.copy()
            print(f"    Pop t={iter_count}: NEW BEST J={2.0*n*child_m:.6f}")
        
        # Tournament selection: replace worst with child if better
        worst_idx = np.argmax(scores)
        if child_m < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_m
        
        # Also replace best if child is better
        if child_m < min(scores):
            population[np.argmin(scores)] = child
            scores = [fast_autocorr(p).max() for p in population]
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pop t={iter_count} ({elapsed:.0f}s): best J={2.0*n*best_m:.6f}")
    
    return best_a, best_m


def active_set_peak_reduction(a0, time_limit, verbose=True, rng_seed=99):
    """
    Active set method: identify the active (binding) constraints (peaks at max),
    then solve an LP to find the best direction to reduce them.
    More aggressive than the previous peak-pair method.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    solver = get_solver()
    max_iters = 200
    
    for outer in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        
        # Find all peaks at (or near) the maximum
        tol = 1e-6
        active_idx = np.where(ac >= M_curr - tol)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(ac))])
        
        # Build the LP: maximize sum(b) subject to:
        # sum(b) = 1, b >= 0, T_sel @ b <= M_curr
        # where T_sel is the constraint matrix for active peaks
        
        # For peak at lag k: sum_i b[i] * a[k-i] <= M_curr (for valid i)
        # Actually, the convolution (a*b)[k] = sum_i a[i]*b[k-i]
        # But we want T such that (T @ b)[k] = (a*b)[k]
        # (a*b)[k] = sum_i a[i] * b[k-i] = sum_j a[k-j] * b[j]
        # So T[k, j] = a[k-j] for valid j
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]
        
        for k in active_idx:
            ki = int(k)
            # (a*b)[ki] = sum_j a[ki-j] * b[j] for j where 0 <= ki-j < n
            # i.e., ki-n+1 <= j <= ki and 0 <= j < n
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            # T_row[j] = a[ki - j] for j in [j_start, j_end]
            T_row = cp.zeros(n)
            for j in range(j_start, j_end + 1):
                T_row[j] = a[ki - j]
            constraints.append(T_row @ b_var <= M_curr)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        # Try multiple step sizes
        found = False
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.4, 0.6]:
            a_trial = (1 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M_curr - 1e-12:
                a = a_trial
                found = True
                break
        
        if not found:
            # Smaller step
            a_trial = 0.995 * a + 0.005 * g_dir
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M_curr:
                a = a_trial
            else:
                # Stuck, perturb
                a_pert = a + rng.normal(0, 0.01, n) * a
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
        
        if verbose and outer % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ActiveSet outer={outer} ({elapsed:.0f}s): J={J:.6f}, n_active={len(active_idx)}")
    
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def suggest_direction(a, ac, n_top=10):
    """
    Suggest gradient direction for reducing top peaks.
    Uses a more sophisticated approach: for each peak, compute the 
    "antigradient" that most efficiently reduces it.
    """
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:n_top]
    top_vals = ac[top_idx]
    
    # Weight proportional to peak value (emphasis on larger peaks)
    w = top_vals ** 3
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad


def burn_in_optimize(a0, time_limit, verbose=True, rng_seed=1):
    """
    Optimizer that starts aggressive and gets more conservative over time.
    Uses a combination of Adam-like updates and targeted moves.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Use log-parameterization with Adam
    a_safe = np.maximum(a, 1e-15)
    w = np.log(a_safe)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr0 = 0.05
    decay = 0.999
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        lr = lr0 * (decay ** (t // 100))
        
        # Get a from w
        w_shift = w - w.max()
        a_curr = np.exp(w_shift)
        a_curr = a_curr / a_curr.sum()
        
        ac = fast_autocorr(a_curr)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a_curr.copy()
        
        # Gradient
        g_a = suggest_direction(a_curr, ac, n_top=50)
        g_w = g_a / np.maximum(a_curr, 1e-15)
        
        # Adam
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * g_w
        v = beta2 * v + (1 - beta2) * g_w**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic restart with perturbation
        if t % 5000 == 0:
            w_pert = w + rng.normal(0, 0.02, n)
            w_shift = w_pert - w_pert.max()
            a_pert = np.exp(w_shift)
            a_pert = a_pert / a_pert.sum()
            M_pert = fast_autocorr(a_pert).max()
            if M_pert < best_m:
                w = w_pert
                best_m = M_pert
                best_a = a_pert.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    BurnIn t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_peak_gradient(a, ac, top_k=100, temperature=1.0):
    """Compute soft gradient emphasizing top peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    scaled = top_vals / max(temperature, 1e-10)
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            best_a = a.copy()

        tol = M_curr * 1e-6 + 1e-12
        active_idx = np.where(ac >= M_curr - tol)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(ac))])
        # Limit active constraints for speed
        if len(active_idx) > 20:
            active_idx = active_idx[:20]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]

        for k in active_idx:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            T_row = np.zeros(n)
            for j in range(j_start, j_end + 1):
                T_row[j] = a[ki - j]
            constraints.append(T_row @ b_var <= M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        found = False
        for alpha in [0.4, 0.2, 0.3, 0.1, 0.15, 0.05, 0.5]:
            a_trial = (1 - alpha) * a + alpha * g_dir
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M_curr - 1e-12:
                a = a_trial
                found = True
                break

        if not found:
            a_trial = 0.995 * a + 0.005 * g_dir
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M_curr:
                a = a_trial
            else:
                a_pert = a + rng.normal(0, 0.01, n) * a
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert

        if verbose and outer % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ActiveSet outer={outer} ({elapsed:.0f}s): J={J:.6f}, n_act={len(active_idx)}")

    return best_a, best_m


def burn_in_optimize(a0, time_limit, verbose=True, rng_seed=1):
    """Adam in log-domain with decaying LR and periodic perturbations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    a_safe = np.maximum(a, 1e-15)
    w = np.log(a_safe)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr0 = 0.04
    decay = 0.9995
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    top_k = 60

    while time.monotonic() - t0 < time_limit:
        t += 1
        lr = lr0 * (decay ** (t // 50))
        w_shift = w - w.max()
        a_curr = np.exp(w_shift)
        a_curr = a_curr / a_curr.sum()

        ac = fast_autocorr(a_curr)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a_curr.copy()

        temp = max(0.1, 1.0 * 0.98 ** (t // 1000))
        g_a = compute_peak_gradient(a_curr, ac, top_k=top_k, temperature=temp)
        g_w = g_a / np.maximum(a_curr, 1e-15)

        m = beta1 * m + (1 - beta1) * g_w
        v = beta2 * v + (1 - beta2) * g_w ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            w_pert = w + rng.normal(0, 0.015, n)
            w_shift2 = w_pert - w_pert.max()
            a_pert = np.exp(w_shift2)
            a_pert = a_pert / a_pert.sum()
            M_pert = fast_autocorr(a_pert).max()
            if M_pert < best_m:
                w = w_pert
                best_m = M_pert
                best_a = a_pert.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    BurnIn t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def try_update_seq(a, best_state, label, t_global):
    """Update best_state dict with new sequence if improved."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy search combining Adam log-domain, spectral shaping,
    population search, active-set LP, and burn-in optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_state = {'seq': None, 'score': float('inf')}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = norm_sum1(parent)
            best_state['score'] = p_score
    except Exception:
        pass

    if best_state['seq'] is None:
        n_base = 4200
        x = np.linspace(0, 1, n_base)
        best_state['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        best_state['score'] = evaluate_safe(best_state['seq'])
        print(f"Default init: n={len(best_state['seq'])}, score={best_state['score']:.6f}")

    # === Phase 1: Adam log-domain on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam log-domain (n={len(best_state['seq'])}) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 60:
        a_in = norm_sum1(best_state['seq'])
        a_out, _ = adam_optimize_log(a_in, t1, top_k=80, lr=0.008,
                                     temperature=1.0, adaptive_temp=True,
                                     verbose=True, rng_seed=42)
        try_update_seq(a_out, best_state, "adam_log", t_global)

    # === Phase 2: Burn-in optimizer ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Burn-in (n={len(best_state['seq'])}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 50:
        a_in = norm_sum1(best_state['seq'])
        a_out, _ = burn_in_optimize(a_in, t2, verbose=True, rng_seed=1)
        try_update_seq(a_out, best_state, "burnin", t_global)

    # === Phase 3: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral shaping (n={len(best_state['seq'])}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 40:
        a_in = norm_sum1(best_state['seq'])
        a_out, _ = spectral_shaping(a_in, t3, verbose=True, rng_seed=13)
        try_update_seq(a_out, best_state, "spectral", t_global)

    # === Phase 4: Active-set LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Active-set LP (n={len(best_state['seq'])}) ===")
    n_best = len(best_state['seq'])
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40 and n_best <= 4500:
        a_in = norm_sum1(best_state['seq'])
        a_out, _ = active_set_peak_reduction(a_in, t4, verbose=True, rng_seed=99)
        try_update_seq(a_out, best_state, "active_set", t_global)

    # === Phase 5: Population search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Population search (n={len(best_state['seq'])}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40:
        a_in = norm_sum1(best_state['seq'])
        a_out, _ = population_search(a_in, t5, pop_size=5, verbose=True, rng_seed=55)
        try_update_seq(a_out, best_state, "population", t_global)

    # === Phase 6: Adam log round 2 (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Adam log round 2 (n={len(best_state['seq'])}) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 30:
        a_in = norm_sum1(best_state['seq'])
        a_out, _ = adam_optimize_log(a_in, t6, top_k=120, lr=0.005,
                                     temperature=0.5, adaptive_temp=True,
                                     verbose=True, rng_seed=100)
        try_update_seq(a_out, best_state, "adam_log2", t_global)

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 300, -300]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100][:2]

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_state['seq'], new_n)

        t7a = min(remaining * 0.4, 150.0)
        if t7a > 30:
            a_out, _ = adam_optimize_log(a_long, t7a, top_k=80, lr=0.008,
                                         temperature=1.0, adaptive_temp=True,
                                         verbose=True, rng_seed=new_n % 1000)
            try_update_seq(a_out, best_state, f"adam_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 20:
                a_in2 = interpolate_sequence(best_state['seq'], new_n)
                a_out, _ = burn_in_optimize(a_in2, t7b, verbose=False, rng_seed=new_n % 500)
                try_update_seq(a_out, best_state, f"burnin_n{new_n}", t_global)

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_state['seq'])
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.5, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_state['seq'])
            a_out, _ = active_set_peak_reduction(a_in, t8, verbose=True, rng_seed=200)
            try_update_seq(a_out, best_state, "final_active", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.6, 80.0)
            if t8b > 15:
                a_in = norm_sum1(best_state['seq'])
                a_out, _ = adam_optimize_log(a_in, t8b, top_k=150, lr=0.003,
                                             temperature=0.3, adaptive_temp=False,
                                             verbose=False, rng_seed=300)
                try_update_seq(a_out, best_state, "final_adam", t_global)

    # Final validation
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_state['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        best_state['score'] = evaluate_safe(best_state['seq'])
        print(f"Fallback: score={best_state['score']:.6f}")

    return [float(x) for x in best_state['seq']]
