def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def lp_direction(a, solver, full=True, top_k=300):
    """Find the LP descent direction."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    # Build constraint: conv(a, b)[m] <= t for all m
    # conv(a,b)[m] = sum_j a[|m-j|] * b[j] (with appropriate bounds)
    # = sum_j a[m-j] * b[j] for 0<=m-j<n, 0<=j<n

        T = np.zeros((2*n-1, n))
        for j in range(n):
            T[j:j+n, j] = a
    else:

        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:top_k]
        T = np.zeros((top_k, n))
        for i, m in enumerate(top_idx):

            j_start = max(0, m_i - n + 1)
            j_end = min(n-1, m_i)
            js = np.arange(j_start, j_end+1)
            T[i, js] = a[m_i - js]

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    return np.maximum(b_var.value, 0), t_var.value

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with Nesterov acceleration (min t s.t. T@b <= t, sum(b)=1)
2. Smart coordinate descent with peak-focused sensitivity

4. Length exploration
The key improvement over the previous approach is the LP direction-finding step,
which directly minimizes the peak cross-correlation, giving a more principled
descent direction.

# Global solver

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_crosscorr(a, b):
    """Compute conv(a, b) using FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_T_matrix(a, top_k=None):
    """Build the constraint matrix T where T[m,j] = a[m-j] if valid."""
    n = len(a)
    if top_k is None:
        K = 2 * n - 1
        T = np.zeros((K, n))
        for j in range(n):
            T[j:j+n, j] = a
        return T
    else:
        # Use top-k most binding lags (based on self-conv)
        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:top_k]
        T = np.zeros((top_k, n))
        for i, m in enumerate(top_idx):
            mi = int(m)
            j_start = max(0, mi - n + 1)
            j_end = min(n - 1, mi)
            js = np.arange(j_start, j_end + 1)
            T[i, js] = a[mi - js]
        return T


def lp_descent_direction(a, solver, top_k=None):
    """
    Find the LP descent direction:
    min t s.t. conv(a,b)[m] <= t for all m, sum(b)=1, b>=0
    Returns (b*, t*) or (None, None) on failure.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    T = build_T_matrix(a, top_k=top_k)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None
    b_val = b_val / b_sum
    return b_val, float(t_var.value)


def nesterov_fw_lp(a0, time_limit, solver, verbose=True, use_full_first=10, top_k=400):
    """
    Nesterov-accelerated Frank-Wolfe with LP direction finding.
    """
    n = len(a0)
    a = norm_sum1(a0)
    x = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:

        # Extrapolated point
        y = np.maximum(x, 1e-15)

        if sy < 1e-15:
            y = a.copy()
        else:

        # LP direction
        if t <= use_full_first and n <= 4000:

        if s is None:
            # Fallback: use negative subgradient
            ac = fast_autocorr(y)
            peak_idx = np.argmax(ac)
            s = y.copy()  # default: stay
            # Actually let's try a simple stochastic direction
            rng = np.random.default_rng(t)
            s = rng.dirichlet(np.ones(n) * 0.1)
        
        # FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * s
        y_new = np.maximum(y_new, 0.0)
        sy_new = y_new.sum()
        if sy_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy_new
        
        # Nesterov
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx_new = x_new.sum()
            if sx_new < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx_new
        
        y = y_new
        x = x_new
        
        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, t_lp={t_val:.6f}")

def coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent focused on peak reduction.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    peels = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))
        
        # Find the most contributing pair for the peak lag
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        if js > je:
            break
        
        idx = np.arange(js, je + 1)
        partners = peak_lag - idx
        contrib = a[idx] * a[partners]
        
        if contrib.max() < 1e-20:
            break
        
        # Source: position with max contribution
        src_local = np.argmax(contrib)
        src = int(idx[src_local])
        partner = int(partners[src_local])
        
        # Destination: find position with low contribution to top lags
        top_lags = np.argsort(ac)[::-1][:20]
        sens = np.zeros(n)
        for m in top_lags:
            mi = int(m)
            s2 = max(0, mi - n + 1)
            e2 = min(n - 1, mi)
            for d in range(s2, e2 + 1):
                p = mi - d
                if 0 <= p < n:
                    sens[d] += a[p]
        
        # Find best destination
        sens[src] = np.inf  # don't move to self
        candidates = np.where(sens < sens.mean())[0]
        if len(candidates) == 0:
            candidates = np.argsort(sens)[:max(1, n//10)]
        
        # Score by inverse distance to src and partner
        dist = np.minimum(np.abs(candidates - src), np.abs(candidates - partner)).astype(float)
        dist = np.maximum(dist, 1)
        scores = sens[candidates] * (1 + dist / n)
        best_dst = int(candidates[np.argmin(scores)])
        
        # Try move
        improved = False
        for frac in [0.01, 0.005, 0.02, 0.003, 0.015, 0.008, 0.03]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m * (1 - 1e-8):
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                peels += 1
                stall = 0
                break

            stall += 1
            # Random perturbation to escape

                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:

                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m * (1 - 1e-8):
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

                        break

            elapsed = time.monotonic() - t0

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""

2. Peak-focused coordinate descent with adaptive moves

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(20, nonneg=True)
            t = cp.Variable()
            prob = cp.Problem(cp.Minimize(t), [x.sum() == 1, cp.abs(x) <= t])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_T_matrix(a, top_k=None):
    n = len(a)
    if top_k is None:
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a
        return T
    else:
        ac = fast_autocorr(a)
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:k]
        T = np.zeros((k, n))
        for i, m in enumerate(top_idx):
            mi = int(m)
            j_start = max(0, mi - n + 1)
            j_end = min(n - 1, mi)
            js = np.arange(j_start, j_end + 1)
            T[i, js] = a[mi - js]
        return T


def lp_descent_direction(a, solver, top_k=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_T_matrix(a, top_k=top_k)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None
    b_val = b_val / b_sum
    return b_val, float(t_var.value)


def nesterov_fw_lp(a0, time_limit, solver, verbose=True, full_iters=5, top_k=500):

    a = norm_sum1(a0)
    x = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    slow_count = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        iter_start = time.monotonic()

        y = np.maximum(x, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        use_full = (t <= full_iters) and (n <= 3500) and (slow_count < 3)
        s, t_val = lp_descent_direction(y, solver, top_k=None if use_full else top_k)
        iter_elapsed = time.monotonic() - iter_start
        if iter_elapsed > 20:
            slow_count += 1
        else:
            slow_count = max(0, slow_count - 1)

        if s is None:
            rng = np.random.default_rng(t * 7919)
            s = rng.dirichlet(np.ones(n) * 0.1)

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * s
        y_new = np.maximum(y_new, 0.0)
        sy_new = y_new.sum()
        if sy_new < 1e-15:

        else:

def coordinate_descent(a0, time_limit, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    peels = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        peak_lag = int(np.argmax(ac))

        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        if js > je:
            break

        idx = np.arange(js, je + 1)
        partners = peak_lag - idx
        contrib = a[idx] * a[partners]

        if contrib.max() < 1e-20:
            break

        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])
        partner = int(partners[src_local])

        top_lags = np.argsort(ac)[::-1][:25]
        sens = np.zeros(n)
        for m in top_lags:
            mi = int(m)
            s2 = max(0, mi - n + 1)
            e2 = min(n - 1, mi)
            d_range = np.arange(s2, e2 + 1)
            p_range = mi - d_range
            valid = (p_range >= 0) & (p_range < n)
            if valid.any():
                sens[d_range[valid]] += a[p_range[valid]]

        sens[src] = np.inf
        mean_sens = sens.mean()
        candidates = np.where(sens < mean_sens)[0]
        if len(candidates) == 0:
            candidates = np.argsort(sens)[:max(1, n // 10)]

        dist_src = np.abs(candidates - src).astype(float)
        dist_partner = np.abs(candidates - partner).astype(float)
        dist = np.maximum(np.minimum(dist_src, dist_partner), 1.0)
        scores = sens[candidates] * (1.0 + dist / n)
        best_dst = int(candidates[np.argmin(scores)])

        improved = False
        fractions = [0.01, 0.005, 0.02, 0.003, 0.015, 0.008, 0.03, 0.002]
        for frac in fractions:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-10:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                peels += 1
                stall = 0
                break

        if not improved:
            # Try multiple random sources
            improved2 = False
            for _ in range(5):
                j = int(rng.integers(n))
                if a[j] < 1e-12:
                    continue
                d = int(rng.integers(n))
                if d == j:
                    continue
                frac = rng.uniform(0.005, 0.02)
                amt = a[j] * frac
                a_trial = a.copy()
                a_trial[j] -= amt
                a_trial[d] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-10:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    improved2 = True
                    peels += 1
                    stall = 0
                    break

            if not improved:
                stall += 1
                if stall > 300:
                    # Larger perturbation
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.05, 0.15)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            stall = 0  # reset to keep exploring

        if verbose and peels % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, peels={peels}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_smooth(a, sigma=8.0):
    n = len(a)
    A = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    window = np.exp(-2.0 * (np.pi * sigma * freqs) ** 2)
    A_smooth = A * window
    a_smooth = np.fft.irfft(A_smooth, n)
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s


def try_update(a, label, best_state):
    if a is None:
        return best_state
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_state
    a_norm = a / s
    seq_list = [float(x) for x in a_norm]
    try:
        sc = evaluate_sequence(seq_list)
    except Exception:
        return best_state
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a_norm.copy()
        best_state['n'] = len(a_norm)
        elapsed = time.monotonic() - best_state['t0']
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    return best_state


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    best_state = {'score': float('inf'), 'seq': None, 'n': 0, 't0': t0}

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_state = try_update(parent, "parent", best_state)
            print(f"Parent: n={len(parent)}, score={best_state['score']:.6f}")
    except Exception as e:
        print(f"No parent: {e}")

    if best_state['seq'] is None:
        n_init = 3200
        x = np.linspace(0, 1, n_init)
        a_init = 0.5 + 0.1 * np.sin(2 * np.pi * x) + 0.05 * np.sin(4 * np.pi * x)
        a_init = np.maximum(a_init, 0.0)
        a_init = a_init / a_init.sum()
        best_state = try_update(a_init, "default", best_state)

    n0 = best_state['n']

    # Phase 1: LP-FW from parent
    remaining = time_budget - (time.monotonic() - t0)
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP-FW parent (n={n0}, t={t1:.0f}s) ===")
        a_out, _ = nesterov_fw_lp(best_state['seq'], t1, solver, verbose=True)
        best_state = try_update(a_out, "lpfw_p1", best_state)

    # Phase 2: LP-FW from alternative inits
    remaining = time_budget - (time.monotonic() - t0)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-FW alt starts (t={t2:.0f}s) ===")
        x = np.linspace(0, 1, n0)
        inits = []
        for sv in [3, 11, 21]:
            rng = np.random.default_rng(sv)
            a_alt = np.ones(n0) * 0.5
            for f in [1, 2, 3, 4, 6, 8]:
                a_alt += rng.normal(0, 0.06) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
            a_alt = np.maximum(a_alt, 0.0)
            a_alt = 0.5 * (a_alt + a_alt[::-1])
            a_alt = a_alt / a_alt.sum()
            inits.append(a_alt)
        # Blend with parent
        blend = 0.5 * best_state['seq'] + 0.5 * inits[0]
        inits.append(blend / blend.sum())

        t_per = t2 / len(inits)
        for i, init in enumerate(inits):
            if time.monotonic() - t0 > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t0))
            if t_avail < 20:
                break
            print(f"  Alt {i} (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_fw_lp(init, t_avail, solver, verbose=False)
            best_state = try_update(a_out, f"lpfw_alt{i}", best_state)

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t0)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Coord descent (n={best_state['n']}, t={t3:.0f}s) ===")
        a_out, _ = coordinate_descent(best_state['seq'], t3, verbose=True, seed=42)
        best_state = try_update(a_out, "coord_p3", best_state)

    # Phase 4: Averaging and smoothing
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 50:
        print(f"\n=== Phase 4: Averaging ===")
        smooth = spectral_smooth(best_state['seq'], sigma=6.0)
        for w in [0.2, 0.4, 0.6, 0.8]:
            avg = w * best_state['seq'] + (1 - w) * smooth
            best_state = try_update(avg, f"avg_w{w}", best_state)
        # Polish best average
        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 20:
            tc = min(remaining * 0.3, 80.0)
            a_out, _ = coordinate_descent(best_state['seq'], tc, verbose=False, seed=77)
            best_state = try_update(a_out, "avg_polish", best_state)

    # Phase 5: Length exploration
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 80:
        print(f"\n=== Phase 5: Length exploration ===")
        n_best = best_state['n']
        for delta in [200, -200, 400, -400]:
            remaining = time_budget - (time.monotonic() - t0)
            if remaining < 50:
                break
            nl = n_best + delta
            if not (1500 <= nl <= 5000):
                continue
            print(f"  n={nl}...")
            a_interp = interpolate_sequence(best_state['seq'], nl)
            tl = min(remaining * 0.2, 120.0)
            a_out, _ = nesterov_fw_lp(a_interp, tl, solver, verbose=False)
            best_state = try_update(a_out, f"lpfw_n{nl}", best_state)
            remaining = time_budget - (time.monotonic() - t0)
            if remaining > 20:
                tc = min(remaining * 0.15, 40.0)
                a_out2, _ = coordinate_descent(best_state['seq'], tc, verbose=False, seed=13)
                best_state = try_update(a_out2, f"coord_n{nl}", best_state)

    # Phase 6: Final LP polish
    remaining = time_budget - (time.monotonic() - t0)
    n_final = best_state['n']
    if remaining > 30 and n_final <= 4000:
        print(f"\n=== Phase 6: Final LP (n={n_final}) ===")
        t6 = min(remaining * 0.4, 150.0)
        a_out, _ = nesterov_fw_lp(best_state['seq'], t6, solver, verbose=True)
        best_state = try_update(a_out, "lp_final", best_state)

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 15:
        t7 = min(remaining * 0.6, 80.0)
        a_out, _ = coordinate_descent(best_state['seq'], t7, verbose=True, seed=200)
        best_state = try_update(a_out, "coord_final", best_state)

    # Final
    if best_state['seq'] is not None:
        final_seq = [float(x) for x in best_state['seq']]
        fs = evaluate_sequence(final_seq)
        print(f"\nFINAL: score={fs:.6f}, n={len(final_seq)}, time={time.monotonic()-t0:.0f}s")
        return final_seq
    else:
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        a_fb = 0.5 + 0.1 * np.sin(2 * np.pi * x)
        a_fb = np.maximum(a_fb, 0.0)
        a_fb = a_fb / a_fb.sum()
        print(f"Fallback: n={n_fb}")
        return [float(v) for v in a_fb]
