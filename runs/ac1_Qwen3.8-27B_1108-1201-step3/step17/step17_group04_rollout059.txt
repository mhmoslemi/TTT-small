- There exist multipliers λ_k >= 0 for each constraint c[k] <= t
- The gradient of the Lagrangian is zero: sum_k λ_k * H[k] * a + μ * 1 = 0 (where μ is the multiplier for sum(a)=1)

sum(a) = 1

This is a system of n+K+1 equations in n+K+1 unknowns (a, λ, μ, t). It could be solved with Newton's method!

1. Start with a guess for the active set

3. Check if the solution satisfies all constraints
4. If not, update the active set and repeat

def build_kkt_matrix(n, active_set, lambdas):
    """Build the KKT matrix M where M[j,i] = sum_{k in active} lambda_k * ([i+j=k] + [|i-j|=k])"""
    M = np.zeros((n, n))
    for k, lam in zip(active_set, lambdas):
        k = int(k)
        # Toeplitz part: |i-j| = k
        if k < n:
            # Super-diagonal
            idx = np.arange(n - k)
            M[idx, idx + k] += lam
            M[idx + k, idx] += lam
        # Hankel part: i+j = k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                M[i, j] += lam
    return M

2. Solve linear system for a
3. Compute c = conv(a,a)
4. If max(c[A]) == min(c[A]) (within tolerance): we found the KKT point
5. Else: update λ proportional to (c[k] - mean(c[A]))^+
6. Check if any k not in A has c[k] > t: add to A
7. Check if any k in A has c[k] < t - margin: remove from A
8. Repeat

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. KKT active-set method exploiting Hankel matrix structure of optimality conditions
2. Spectral shaping (Gerchberg-Saxton) for diverse initializations
3. Nesterov subgradient with backtracking line search
4. Multi-length exploration and adaptive refinement
Key innovation: KKT method that solves the linear system P@a = const where P is
a Hankel matrix built from the active constraint set, giving precise stationary points.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_subgradient(a, ac, exact=True, top_k=10, beta=3.0):
    """Compute subgradient of max(conv(a,a))."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] = 2.0 * a[j_idx[valid]]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            grad[idx[valid]] += 2.0 * wi * a[j_idx[valid]]
    return grad


def kkt_solve(a0, active_set, lambdas, max_iter=50):
    """
    Solve the KKT linear system for given active set and multipliers.
    P @ a = (-mu/2) * 1, sum(a) = 1, a >= 0
    where P[j,i] = sum_k lambda_k * delta_{i+j, k}
    Returns (a, success).
    """
    n = len(a0)
    # Build P matrix (Hankel structure)
    P = np.zeros((n, n), dtype=np.float64)
    for k, lam in zip(active_set, lambdas):
        k_int = int(k)
        if k_int < 0 or k_int >= 2 * n - 1:
            continue
        # P[j,i] = lambda when i+j = k
        i_start = max(0, k_int - (n - 1))
        i_end = min(n - 1, k_int)
        for i in range(i_start, i_end + 1):
            j = k_int - i
            if 0 <= j < n:
                P[i, j] += lam

    # Augmented system: [2P | -1; 1^T | 0] [a; mu] = [0; 1]
    A_mat = np.zeros((n + 1, n + 1), dtype=np.float64)
    b_vec = np.zeros(n + 1, dtype=np.float64)
    A_mat[:n, :n] = 2.0 * P
    A_mat[:n, n] = -1.0
    A_mat[n, :n] = 1.0
    A_mat[n, n] = 0.0
    b_vec[n] = 1.0

    # Solve with fallback
    try:
        sol = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A_mat, b_vec, rcond=None)[0]
        except Exception:
            return a0, False

    a_sol = np.maximum(sol[:n], 0.0)
    s = a_sol.sum()
    if s < 1e-15:
        return a0, False
    a_sol = a_sol / s
    return a_sol, True


def kkt_active_set(a0, time_limit, verbose=False, initial_K=8, max_K=30):
    """
    KKT active-set refinement algorithm.
    Iteratively identifies the active set, solves the KKT system,
    and updates multipliers until convergence.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    max_lag = 2 * n - 2
    for _ in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()
        t_val = M

        # Identify active set: lags where c[k] is close to max
        threshold = t_val * (1.0 - 0.01)
        active_candidates = np.where(ac >= threshold)[0]
        # Also include top K regardless
        top_idx = np.argsort(ac)[::-1][:initial_K]
        active_set = sorted(set(list(active_candidates[:max_K]) + list(top_idx[:initial_K])))
        active_set = [k for k in active_set if 0 <= k <= max_lag]

        if len(active_set) < 2:
            active_set = sorted(top_idx[:initial_K].tolist())

        K = len(active_set)
        lambdas = np.ones(K, dtype=np.float64) / K

        # Iterate on multipliers
        converged = False
        for lam_iter in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            a_new, success = kkt_solve(a, active_set, lambdas)
            if not success:
                break
            a_new = norm_sum1(a_new)

            ac_new = fast_autocorr(a_new)
            vals = ac_new[active_set]
            if len(vals) > 0 and np.all(np.isfinite(vals)):
                # Check if all active constraints are approximately tight
                spread = vals.max() - vals.min()
                if spread < 1e-6 * vals.max():
                    consecutive = 0
                else:
                    consecutive = lam_iter

                # Update multipliers: proportional to how much each exceeds mean
                mean_val = vals.mean()
                if mean_val > 1e-15:
                    ratios = np.maximum(vals / mean_val, 0.1)
                    ratios = ratios / ratios.sum() * K
                    lambdas = 0.5 * lambdas + 0.5 * ratios

                # Check improvement
                J_new = 2.0 * n * ac_new.max()
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = a_new.copy()

                a = a_new

                if spread < 1e-5 * vals.max() + 1e-15:
                    converged = True
                    break
            else:
                break

        # After multiplier iteration, check active set validity
        ac = fast_autocorr(a)
        t_curr = ac.max()
        # Add newly violated constraints
        violated = np.where(ac > t_curr * 1.001)[0]
        if len(violated) > 0:
            new_set = sorted(set(list(active_set) + [int(v) for v in violated[:5]]))
            new_set = [k for k in new_set if 0 <= k <= max_lag]
            if len(new_set) != len(active_set):
                active_set = new_set[:max_K]
                if verbose:
                    print(f"    KKT: expanded active set to {len(active_set)}")

        # Remove constraints that are no longer tight
        tight_mask = np.array([ac[k] > t_curr * 0.99 for k in active_set])
        if tight_mask.sum() < len(active_set) and tight_mask.sum() >= 2:
            active_set = [active_set[i] for i in range(len(active_set)) if tight_mask[i]]

        if converged:
            if verbose:
                print(f"    KKT: converged in {lam_iter+1} mult iterations, J={best_J:.8f}")
            break

    return best_a, best_J


def spectral_shaping(a0, time_limit, verbose=False, n_iters=200):
    """
    Gerchberg-Saxton style spectral shaping to flatten the convolution.
    Alternates between frequency-domain magnitude enforcement and time-domain non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    # Choose FFT size
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Frequency domain: compute FFT
        a_padded = np.zeros(N, dtype=np.float64)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)

        # Target: flatten magnitude while keeping DC fixed
        mag = np.abs(A)
        mag[0] = 1.0  # DC component fixed (sum(a) = 1)

        # Adaptive target for non-DC components
        # Want |A(f)|^2 to be as flat as possible
        # Target: C = sqrt((N * sum(a^2) - 1) / (N-1))
        sum_a2 = np.sum(a ** 2)
        C = np.sqrt(max((N * sum_a2 - 1.0) / max(N - 1, 1), 1e-10))

        # Smooth transition: blend between current and target
        alpha = 0.3 + 0.4 * (it / max(n_iters, 1))  # increase over iterations
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]
        target_mag[1:] = C

        # Blend
        new_mag = (1 - alpha) * mag + alpha * target_mag
        new_mag[0] = mag[0]
        new_mag[1:] = np.maximum(new_mag[1:], C * 0.5)  # floor

        # Reconstruct with new magnitude
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = A[0]  # keep DC exactly

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def nesterov_subgradient(a0, time_limit, verbose=False, top_k=15, beta=3.0, c=3.0,
                         exact_frac=0.4, restart_every=3000, seed=42):
    """Nesterov-accelerated mirror descent with adaptive subgradient mixing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        ac = fast_autocorr(y)
        M_val = ac.max()

        use_exact = rng.random() < exact_frac
        if use_exact:
            g = get_subgradient(y, ac, exact=True)
        else:
            g = get_subgradient(y, ac, exact=False, top_k=cur_top_k, beta=cur_beta)

        lower_bound = 1.0 / n
        gap = M_val - lower_bound
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.01 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 8, 60)
            cur_beta = max(cur_beta - 0.3, 1.5)
            cur_c = min(cur_c + 0.3, 6.0)
            exact_frac = min(exact_frac + 0.05, 0.65)

    return best_a, best_J
