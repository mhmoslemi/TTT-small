def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_equalization(a0, time_limit, gamma=0.7, verbose=False, seed=0):
    """
    Iterative spectral shaping: flatten the power spectrum via constant-modulus
    style iteration with non-negativity projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    fft_size = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    cur_gamma = gamma
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        target_mag = np.median(mag[mag > 1e-10])
        if target_mag < 1e-15:
            break

        new_mag = (mag + 1e-15) ** cur_gamma
        new_mag = new_mag * (target_mag / (new_mag.mean() + 1e-30))

        A_new = phase * new_mag
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            stall += 1
            cur_gamma = min(1.0, cur_gamma + 0.05)
            continue
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                cur_gamma = max(0.3, cur_gamma - 0.03)
                # Add small noise to escape
                a = a + rng.normal(0, 0.01, n) * a
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and (time.monotonic() - t0) % 10 < 0.01:
            elapsed = time.monotonic() - t0
            print(f"    SE ({elapsed:.0f}s): J={best_J:.6f}, gamma={cur_gamma:.2f}, stall={stall}")

    if verbose:
        print(f"    SE final: J={best_J:.6f}")
    return best_a, best_J


def water_filling_spectrum(a0, time_limit, verbose=False, seed=0):
    """
    Water-filling in the frequency domain: iteratively cap the power spectrum
    at a threshold and reconstruct.
    """
    n = len(a0)
    a = norm_sum1(a0)
    fft_size = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initial threshold: slightly above the mean
    A = np.fft.fft(a, fft_size)
    mag2 = np.abs(A) ** 2
    threshold = np.mean(mag2) * 1.5

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Water-filling: cap magnitude
        mag_capped = np.minimum(mag, threshold)
        # Add floor to avoid zeros
        mag_capped = np.maximum(mag_capped, 1e-8)

        A_new = phase * mag_capped
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            threshold *= 0.9
            continue
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Adjust threshold
            ac = fast_autocorr(a)
            peak = ac.max()
            mean_ac = ac.mean()
            if peak > 1.2 * mean_ac:
                threshold *= 0.95
            else:
                threshold *= 1.05

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    WF ({elapsed:.0f}s): J={best_J:.6f}, thr={threshold:.4f}")

    return best_a, best_J


class AdamState:
    def __init__(self, n):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0


def adam_update(state, grad, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    state.t += 1
    state.m = beta1 * state.m + (1 - beta1) * grad
    state.v = beta2 * state.v + (1 - beta2) * (grad ** 2)
    m_hat = state.m / (1 - beta1 ** state.t)
    v_hat = state.v / (1 - beta2 ** state.t)
    return lr * m_hat / (np.sqrt(v_hat) + eps)


def adam_softmax_descent(a0, time_limit, verbose=False, lr=0.5, alpha_smooth=20.0, top_k=20, seed=0):
    """
    Adam-optimized gradient descent on softmax parameterization.
    Uses logsumexp-like smoothing of the max autocorrelation.
    """
    n = len(a0)
    a_init = norm_sum1(a0)
    # Initialize theta from a
    theta = np.log(a_init + 1e-15)
    theta -= theta.mean()  # center for numerical stability

    rng = np.random.default_rng(seed)
    state = AdamState(n)
    best_a = a_init.copy()
    best_J = compute_J(a_init)
    t0 = time.monotonic()

    cur_lr = lr
    cur_top_k = top_k
    stall = 0
    restart_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute a from theta (with clipping for stability)
        theta_clipped = np.clip(theta, -30, 30)
        theta_shifted = theta_clipped - theta_clipped.max()
        a = np.exp(theta_shifted)
        s = a.sum()
        a = a / s

        # Compute autocorrelation
        ac = fast_autocorr(a)
        M = ac.max()

        # Smooth max: use top-k weighted sum
        k = min(cur_top_k, len(ac))
        top_idx = np.argpartition(ac, -k)[-k:]
        top_vals = ac[top_idx]
        # Soft weights
        w = top_vals ** 3.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / w_sum

        # Gradient of weighted sum of ac[k] w.r.t. a
        grad_a = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            grad_a[idx] += 2.0 * wi * a[j_idx]

        # Chain rule: d a_i / d theta_i (softmax Jacobian)
        # grad_theta_i = grad_a_i * a_i * (1 - a_i) - a_i * sum_j(grad_a_j * a_j)
        ga_dot_a = np.dot(grad_a, a)
        grad_theta = grad_a * a - a * ga_dot_a

        # Adam update
        update = adam_update(state, grad_theta, lr=cur_lr)
        theta = theta - update
        theta -= theta.mean()  # recenter

        # Evaluate
        theta_clipped = np.clip(theta, -30, 30)
        theta_shifted = theta_clipped - theta_clipped.max()
        a_new = np.exp(theta_shifted)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1

        # Adaptive learning rate
        if t % 5000 == 0:
            if stall > 2000:
                cur_lr *= 0.8
                cur_top_k = min(cur_top_k + 5, 60)
                restart_count += 1
                # Perturb theta
                theta = theta + rng.normal(0, 0.1 * cur_lr, n)
                theta -= theta.mean()
            elif stall > 500:
                cur_lr = max(0.01, cur_lr * 0.95)

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={best_J:.6f}, lr={cur_lr:.4f}")

    return best_a, best_J


def exact_subgradient_refine(a0, time_limit, verbose=False, seed=0):
    """
    Exact subgradient descent tracking only the top-1 peak,
    with Polyak step size and mirror descent (exponential) updates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_c = 3.0
    nesterov_beta = 0.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        if t > 1:
            nesterov_beta = (t - 1.0) / (t + 1.0)
            y = x + nesterov_beta * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        # Exact subgradient at argmax
        k = int(np.argmax(ac))
        g = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        g[idx] = 2.0 * y[j_idx]

        # Also add a small regularizing component to spread mass
        g = g + 0.01 * (M - ac.mean())

        # Polyak step
        lower_bound = 1.0 / (2 * n - 1)  # theoretical min of max autocorr
        gap = M - lower_bound
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step
        log_a = np.log(y + 1e-15) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            cur_c = min(cur_c + 0.5, 10.0)
            pert = rng.normal(0, 0.005, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ESD t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J

    cur_c = 3.0

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300, max_iters=5000):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def combined_refinement(a0, time_limit, solver, verbose=False):
    """
    Alternate between spectral equalization, top-K subgradient, and exact subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    methods = [
        ("SE", spectral_equalization, {"gamma": 0.7, "seed": 1}),
        ("TopK", topk_subgradient, {"top_k": 10, "beta_exponent": 3.0, "seed": 2}),
        ("ESD", exact_subgradient_refine, {"seed": 3}),
        ("SE2", spectral_equalization, {"gamma": 0.5, "seed": 4}),
        ("Adam", adam_softmax_descent, {"lr": 0.3, "top_k": 15, "seed": 5}),
    ]

    idx = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        t_alloc = min(remaining * 0.25, 120.0)
        if t_alloc < 10:
            t_alloc = remaining * 0.4

        name, func, kwargs = methods[idx % len(methods)]
        print(f"  Combined [{name}] (t={t_alloc:.0f}s), current J={best_J:.6f}")
        a_out, _ = func(norm_sum1(best_a), t_alloc, verbose=False, **kwargs)
        J_out = compute_J(a_out)
        if J_out < best_J - 1e-14:
            best_J = J_out
            best_a = a_out.copy()
            print(f"    Improved to J={best_J:.6f}")
        idx += 1

    return best_a, best_J


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_cosine_edge(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Spectral Equalization from multiple inits ===
    print(f"\n=== Phase 1: Spectral Equalization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 300.0)

    if t_phase1 > 30:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_cosine_edge(n0, power=1.2))
        labels.append("cos1.2")
        inits.append(make_edge_init(n0, alpha=1.5))
        labels.append("edge1.5")
        inits.append(make_tapered(n0, inner_power=0.6))
        labels.append("taper0.6")

        # Remove duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.05:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(15.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 10:
                break
            print(f"  SE init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = spectral_equalization(init, t_avail, gamma=0.7 + 0.1 * idx, verbose=False, seed=10 + idx)
            try_update(a_out, f"se_{lbl}")

    # === Phase 2: Adam-based descent ===
    print(f"\n=== Phase 2: Adam Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        print(f"  Adam on best (n={len(a_in)}, score={best_score:.6f})...")
        a_out, _ = adam_softmax_descent(a_in, t2, verbose=True, lr=0.4, top_k=15, seed=100)
        try_update(a_out, "adam_best")

    # === Phase 3: Top-K subgradient ===
    print(f"\n=== Phase 3: Top-K Subgradient ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        print(f"  TopK on best (n={len(a_in)})...")
        a_out, _ = topk_subgradient(a_in, t3, verbose=True, top_k=10, beta_exponent=3.0, seed=200)
        try_update(a_out, "topk_best")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP Frank-Wolfe (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=300)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 5: Combined refinement ===
    print(f"\n=== Phase 5: Combined Refinement ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = combined_refinement(a_in, t5, solver, verbose=True)
        try_update(a_out, "combined")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    lengths_to_try = []
    for delta in [300, 600, -300, -600, 1000]:
        new_n = n_best + delta
        if 800 <= new_n <= 6000 and new_n not in lengths_to_try and new_n != n_best:
            lengths_to_try.append(new_n)

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_seq(best_seq, new_n)

        # Quick SE on interpolated
        t_se = min(remaining * 0.2, 100.0)
        if t_se > 15:
            a_out, _ = spectral_equalization(a_interp, t_se, gamma=0.7, verbose=False, seed=500)
            try_update(a_out, f"se_n{new_n}")

        # Quick topK
        remaining = time_budget - (time.monotonic() - t_global)
        t_lk = min(remaining * 0.15, 80.0)
        if t_lk > 15:
            a_in2 = norm_sum1(best_seq) if len(best_seq) == n_best else norm_sum1(a_out)
            # Use the current best for this length
            best_at_len = interpolate_seq(best_seq, new_n)
            # Check if we improved at this length
            sc_check = evaluate_safe(best_at_len)
            if sc_check < best_score:
                best_at_len = best_at_len
            a_out2, _ = topk_subgradient(best_at_len, t_lk, verbose=False, top_k=10, seed=600)
            try_update(a_out2, f"topk_n{new_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"  Final: n={n_best}, score={best_score:.6f}")

    if remaining > 30:
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = exact_subgradient_refine(a_in, t7, verbose=True, seed=999)
            try_update(a_out, "esd_final")

    # === Phase 8: One more combined round if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t8 = min(remaining * 0.6, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = combined_refinement(a_in, t8, solver, verbose=False)
            try_update(a_out, "combined_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_cosine_edge(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
