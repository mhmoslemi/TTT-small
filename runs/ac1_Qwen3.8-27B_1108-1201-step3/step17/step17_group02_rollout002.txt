            lse = ac_max + tau * np.log(np.sum(np.exp((ac - ac_max) / tau)))
            return lse
        
        def obj_grad(x):
            a_vec = softmax_fn(x)
            ac = fast_autocorr(a_vec)
            ac_max = ac.max()
            lse = ac_max + tau * np.log(np.sum(np.exp((ac - ac_max) / tau)))
            
            p = np.exp((ac - ac_max) / tau)
            p = p / p.sum()
            
            # Gradient w.r.t. a
            # Use FFT for efficiency
            # grad_a[j] = 2 * sum_k p[k] * a[k-j] for 0 <= k-j < n
            # This equals 2 * (cross_correlation of p_reversed and a) [j]
            # Or: 2 * (convolution of p and a_reversed) [n-1+j]
            
            # Let me use the direct FFT approach:
            # Define p_ext of length 2n-1 (already is)
            # We want: grad_a[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
            # This is the cross-correlation of p (length 2n-1) with a (length n) at lag j
            # cross_corr[j] = sum_m p[j+m] * a[m]
            # In FFT: cross_corr = IFFT(conj(FFT(p, N)) * FFT(a, N)) where N >= 2n-1+n-1 = 3n-2
            # Actually, let me think about this differently.
            # sum_m p[j+m]*a[m] = (p_shifted * a)[j] where p_shifted is a "reversed" p
            # Let's define q[m] = p[n-1-m] (reversed p, length 2n-1)
            # Then conv(q, a)[k] = sum_i q[i]*a[k-i] = sum_i p[n-1-i]*a[k-i]
            # At k = n-1+j: sum_i p[n-1-i]*a[n-1+j-i]
            # Let m = n-1-i: sum_m p[m]*a[j+m]
            # That's exactly what we want!
            # So grad_a[j] = 2 * conv(q, a)[n-1+j] where q = p[n-1::-1] ... 
            # Wait, p has length 2n-1, indices 0 to 2n-2.
            # q[i] = p[n-1-i] for i = 0, ..., 2n-2
            # So q = p[n-1::-1] in numpy (reversed)
            # But p has length 2n-1, so p[n-1::-1] gives indices n-1, n-2, ..., 0, which is length n.
            # That's not right. I need to reverse the full p.
            # q = p[::-1] has length 2n-1, q[i] = p[2n-2-i]
            # conv(q, a, 'full')[k] = sum_i q[i]*a[k-i] = sum_i p[2n-2-i]*a[k-i]
            # At k = n-1+j: sum_i p[2n-2-i]*a[n-1+j-i]
            # Let m = 2n-2-i: i = 2n-2-m
            # = sum_m p[m]*a[n-1+j-(2n-2-m)] = sum_m p[m]*a[j+m+1]
            # Hmm, off by 1. Let me try a different index.
            # At k = n+j: sum_i p[2n-2-i]*a[n+j-i]
            # Let m = 2n-2-i: i = 2n-2-m
            # = sum_m p[m]*a[n+j-(2n-2-m)] = sum_m p[m]*a[j+m+2]
            # Still not quite right.
            
            # OK let me just use the straightforward approach:
            # grad_a[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
            # For j = 0: sum_m p[m] * a[m] (m from 0 to n-1)
            # For j = 1: sum_m p[1+m] * a[m] (m from 0 to n-1)
            # ...
            # For j = n-1: sum_m p[n-1+m] * a[m] (m from 0 to n-1)
            
            # This is like a "sliding dot product" of p with a.
            # I can compute this using:
            # grad_a = 2 * np.array([np.dot(p[j:j+n], a) for j in range(n)])
            # But this is O(n^2). For n=4200, that's ~17M operations, which is fine.
            
            # Or I can use FFT:
            # Define p_pad of length 2n-1 (already is)
            # grad_a[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
            # This is the (n-1-j)th element of the "anti-diagonal" convolution...
            # Actually, this is: (p * a_flipped)[offset]
            # Let a_flip = a[::-1] (reversed a)
            # conv(p, a_flip, 'full')[k] = sum_i p[i] * a_flip[k-i] = sum_i p[i] * a[n-1-(k-i)]
            # = sum_i p[i] * a[n-1-k+i]
            # At k = n-1-j: sum_i p[i] * a[n-1-(n-1-j)+i] = sum_i p[i] * a[j+i]
            # That's sum_i p[i]*a[j+i] which is what we want (with i ranging over valid indices)!
            # So grad_a[j] = conv(p, a_flip, 'full')[n-1-j]
            
            a_flip = a_vec[::-1]
            N = len(p) + n - 1  # 2n-1 + n - 1 = 3n-2
            fft_N = 1 << (N - 1).bit_length()
            P = np.fft.rfft(p, fft_N)
            AF = np.fft.rfft(a_flip, fft_N)
            conv_result = np.fft.irfft(P * AF, fft_N)
            # grad_a[j] = 2 * conv_result[n-1-j] for j=0,...,n-1
            grad_a = 2.0 * conv_result[n-1::-1]  # indices n-1, n-2, ..., 0
            
            # Verify: grad_a[j] = 2 * conv_result[n-1-j]
            # For j=0: 2 * conv_result[n-1]
            # For j=n-1: 2 * conv_result[0]
            # So grad_a = 2 * conv_result[0:n][::-1] = 2 * conv_result[:n][::-1]
            # Hmm, that gives grad_a[0] = 2*conv_result[n-1], grad_a[1] = 2*conv_result[n-2], etc.
            # Which is what we want.
            
            # Gradient w.r.t. x
            mean_g = np.dot(a_vec, grad_a)
            grad_x = a_vec * (grad_a - mean_g)
            
            return lse, grad_x
        
        # Run L-BFGS
        try:
            res = minimize(obj_grad, x, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter, 'gtol': 1e-12, 'iprint': -1})
            if res.x is not None:
                x = res.x
        except Exception:
            pass
        
        a_new = softmax_fn(x)
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        if outer % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LBFGS outer={outer} ({elapsed:.0f}s): tau={tau:.2e}, J={J:.6f}")
    
    return best_a, best_m
