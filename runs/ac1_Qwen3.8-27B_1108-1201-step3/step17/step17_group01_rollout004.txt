def lp_step(a, M, K, solver):
    """Solve the LP subproblem for Frank-Wolfe.
    
    max sum(b) s.t. (T_k a)^T b <= M for k in top_k, b >= 0
    """
    n = len(a)
    ac = fast_autocorr(a)
    top_k = np.argsort(ac)[::-1][:K]
    
    # Build sparse constraint matrix
    # For each selected k, the constraint is sum_i a[k-i] * b_i <= M
    # (where 0 <= k-i < n and 0 <= i < n)
    
    rows = []
    for k in top_k:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        vals = a[k - idx]
        rows.append((idx, vals))
    
    # Solve LP
    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) == 1]  # Normalize
    for idx, vals in rows:
        constraints.append(vals @ b_var[idx] <= M / a.sum())
    
    prob = cp.Problem(cp.Maximize(0), constraints)  # Feasibility
    # Actually we want max sum(b) s.t. constraints, b >= 0
    # But if we normalize sum(b) = 1, we just need feasibility
    
    # Better: max sum(b) s.t. (T_k a)^T b <= M * S, b >= 0
    # where S = sum(a)
    S = a.sum()
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for idx, vals in rows:
        constraints.append(vals @ b_var[idx] <= M * S)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = b_var.value
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s
