```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov mirror descent with FFT-based subgradients and adaptive top-K
2. L-BFGS refinement with log-sum-exp smoothing and temperature continuation
3. Coordinate descent with random destination sampling
4. Multiple L-BFGS restarts from diverse perturbations
5. Multi-strategy initialization and length sweeping
"""

import numpy as np
import time
from scipy.optimize import minimize as _scipy_min


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming a is normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def softmax_fn(t):
    """Stable softmax."""
    t = t - t.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(t)) / len(t)
    return e / s


def compute_wa_cross(w, a, n):
    """
    Compute wa_cross[j] = sum_{m=0}^{n-1} w[j+m]*a[m] for j=0,...,n-1.
    w has length 2n-1, a has length n.
    """
    a_rev = a[::-1].copy()
    total_len = (2 * n - 1) + n - 1  # 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * Ar, fft_size)
    return conv[n - 1:2 * n - 1]


def compute_sliding_sum_arr(wc, n):
    """
    Compute slide[j] = sum_{m=0}^{n-1} wc[j+m] for j=0,...,n-1.
    wc has length 2n-1.
    """
    padded = np.zeros(len(wc) + 1)
    padded[1:] = wc
    cumsum = np.cumsum(padded)
    return cumsum[n:2 * n] - cumsum[:n]


def smooth_obj_and_grad(t, temp, top_k):
    """
    Compute the smooth objective and gradient for softmax parameterization.
    Objective: 2n * logsumexp(conv(softmax(t)) / temp), restricted to top-K lags.
    Returns (objective_value, gradient_vector).
    """
    a = softmax_fn(t)
    n = len(a)
    conv = fast_conv(a)
    conv_len = 2 * n - 1

    if top_k is not None and top_k < conv_len:
        top_idx = np.argsort(conv)[::-1][:top_k]
        conv_work = conv[top_idx]
    else:
        conv_work = conv
        top_idx = np.arange(conv_len)

    scaled = conv_work / temp
    scaled_max = scaled.max()
    exp_scaled = np.exp(scaled - scaled_max)
    lse = scaled_max + np.log(exp_scaled.sum())
    obj = 2.0 * n * lse

    w_top = exp_scaled / exp_scaled.sum()

    w = np.zeros(conv_len)
    w[top_idx] = w_top

    wa_cross = compute_wa_cross(w, a, n)
    wc = w * conv
    wconv_slide = compute_sliding_sum_arr(wc, n)
    grad = 4.0 * n * a * (wa_cross - wconv_slide)

    # Project out the invariance direction
    grad = grad - np.mean(grad)

    return obj, grad


def _smooth_fun(t, temp, top_k):
    """Wrapper for scipy minimize - objective only."""
    return smooth_obj_and_grad(t, temp, top_k)[0]


def _smooth_grad(t, temp, top_k):
    """Wrapper for scipy minimize - gradient only."""
    return smooth_obj_and_grad(t, temp, top_k)[1]


def compute_subgradient(a, conv, top_k, beta_w):
    """Compute soft subgradient of max(conv) at a using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent(a0, time_limit, top_k, beta_w, c0, restart_every, verbose):

        if time.monotonic() - t0 > time_limit:

            y = x + beta_t * (x - x_prev)

        y = np.maximum(y, 1e-15)

            y = x.copy()
        else:
            y = y / sy

        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)

        x_prev = x

            stall = 0

            cur_top_k = min(cur_top_k + 20, 300)

            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    best_a = a.copy()

        remaining = time_limit - elapsed
        if remaining < 5:
            break
        time_per = remaining / (n_temps - idx)

        max_iter = max(30, min(500, int(time_per * 80)))
        max_fun = max(100, min(3000, int(time_per * 400)))

        try:
            result = _scipy_min(
                _smooth_fun, t_log,
                args=(temp, 200),
                jac=_smooth_grad,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-12,
                    'gtol': 1e-7,
                    'maxfun': max_fun,
                    'iprint': 0
                }
            )
            t_log = result.x
            a_new = softmax_fn(t_log)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            val = eval_J(a_new)
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
            if verbose:
                print(f"    L-BF temp={temp:.2e}: J={best_val:.6f} "
                      f"(nit={result.nit}, ok={result.success})")
        except Exception:
            if verbose:
                print(f"    L-BF temp={temp:.2e}: failed")
            break

    return best_a, best_val


def lbfgs_single(a0, time_limit, top_k, verbose):
    """Single L-BFGS run at a fixed temperature derived from the current solution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    conv = fast_conv(a)
    max_conv = conv.max()
    temp = max_conv * 0.5

    t_log = np.log(a)
    t_log = t_log - t_log.mean()

    t0 = time.monotonic()
    best_a = a.copy()
    best_val = eval_J(a)

    max_iter = max(50, int(time_limit * 60))
    max_fun = max(200, int(time_limit * 300))

    try:
        result = _scipy_min(
            _smooth_fun, t_log,
            args=(temp, top_k),
            jac=_smooth_grad,
            method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-12,
                'gtol': 1e-7,
                'maxfun': max_fun,
                'iprint': 0
            }
        )
        t_log = result.x
        a_new = softmax_fn(t_log)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        val = eval_J(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        if verbose:
            print(f"    L-BF single: J={best_val:.6f} (nit={result.nit})")
    except Exception:
        pass

    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose, seed):
    """Coordinate descent with random destination sampling."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)

        n_lags = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        # Random destination sampling
        n_samples = min(25, n // 20)
        candidates = set()
        for _ in range(n_samples):
            candidates.add(int(rng.integers(n)))
        candidates.discard(src)

        # Also add the position with minimum pressure
        candidates.add(int(np.argmin(pressure + 0.01 * a)))

        best_improvement = None
        best_J_try = best_J

        for dst in candidates:
            if dst == src:
                continue
            for frac in (0.01, 0.03, 0.05, 0.1, 0.2):
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t /= st
                J_t = eval_J(a_t)
                if J_t < best_J_try:
                    best_J_try = J_t
                    best_improvement = (dst, frac)

        if best_improvement is not None:
            dst, frac = best_improvement
            amt = a[src] * frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0.0)
            a /= a.sum()
            best_J = best_J_try
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 100 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 10):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric_ramp(n, p, shift):
    """Asymmetric ramp with peak shifted."""
    x = np.linspace(0, 1, n)
    denom = 1 - 2 * shift
    if abs(denom) < 1e-6:
        denom = 1e-3
    a = ((x + shift / denom) ** p) * ((1 - x) ** 0.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_chessboard(n, alpha):
    """Alternating pattern with decay."""
    x = np.arange(n, dtype=np.float64)
    a = np.where(x % 2 == 0, 1.0, alpha) * np.exp(-x / (n * 0.5))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver: MD + L-BFGS continuation + CD + restarts."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3200
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"\n=== Phase 1: MD Multi-Init (n={n0}) ===")
    elapsed = time.monotonic() - t_global

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.5))
    labels.append("ramp05")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp10")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp15")
    inits.append(make_power_ramp(n0, 2.0))
    labels.append("ramp20")
    inits.append(make_bimodal(n0, 80.0))
    labels.append("bimodal")
    inits.append(make_smooth_random(n0, 3))
    labels.append("smooth3")
    inits.append(make_smooth_random(n0, 99))
    labels.append("smooth99")
    inits.append(make_asymmetric_ramp(n0, 1.0, 0.2))
    labels.append("asym")
    inits.append(make_chessboard(n0, 0.4))
    labels.append("chess")

    if parent is not None:
        b1 = 0.5 * parent + 0.5 * make_power_ramp(n0, 1.5)
        inits.append(b1 / b1.sum())
        labels.append("blend1")
        b2 = 0.7 * parent + 0.3 * make_power_ramp(n0, 2.0)
        inits.append(b2 / b2.sum())
        labels.append("blend2")

    # Deduplicate by signature
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        step = max(1, n0 // 100)
        sig = tuple(np.round(init[::step], 3))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    remaining = time_budget - (time.monotonic() - t_global)
    t_per = max(10.0, remaining * 0.25 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 120:
            break
        t_avail = min(t_per, time_budget - 120 - elapsed)
        if t_avail < 8:
            break
        print(f"  MD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = mirror_descent(init, t_avail, 100, 5.0, 5.0, 8000, False)
        try_update(a_out, "md_" + lbl)

    # Phase 2: L-BFGS continuation on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: L-BF Continuation (n={len(best_a)}) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        a_out, _ = lbfgs_refine(best_a.copy(), t2, True)
        try_update(a_out, "lbfgs_cont")

    # Phase 3: CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: CD Polish (n={len(best_a)}) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, _ = coordinate_descent(best_a.copy(), t3, True, 77)
        try_update(a_out, "cd1")

    # Phase 4: L-BFGS restarts from diverse perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BF Restarts (n={len(best_a)}) ===")
    t4_total = min(remaining * 0.20, 250.0)
    if t4_total > 30:
        n_restart = 4
        t_per_r = t4_total / n_restart
        rng4 = np.random.default_rng(2024)
        n_cur = len(best_a)
        for ri in range(n_restart):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 100:
                break
            # Create diverse perturbation
            if ri == 0:
                # Small Gaussian
                pert = rng4.normal(0, 0.01, n_cur) * best_a
                a_r = best_a + pert
            elif ri == 1:
                # Power-law reweight
                x = np.linspace(1.0, 0.1, n_cur)
                a_r = best_a * x
            elif ri == 2:
                # Mix with bimodal
                a_r = 0.6 * best_a + 0.4 * make_bimodal(n_cur, 60.0)
            else:
                # Low-freq Fourier perturbation
                x = np.linspace(0, 1, n_cur)
                pert = 0.05 * np.sin(2 * np.pi * 3 * x + 1.0) + 0.03 * np.cos(2 * np.pi * 7 * x)
                a_r = best_a * (1.0 + pert)

            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            a_out, _ = lbfgs_single(a_r, t_per_r, 150, False)
            try_update(a_out, f"lbfgs_r{ri}")

    # Phase 5: Final L-BFGS continuation
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 5: Final L-BF (n={n_cur}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_out, _ = lbfgs_refine(best_a.copy(), t5, True)
        try_update(a_out, "final_lbfgs")

    # Phase 6: Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final CD (n={len(best_a)}) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 15:
        a_out, _ = coordinate_descent(best_a.copy(), t6, True, 314)
        try_update(a_out, "final_cd")

    # Phase 7: Length sweep
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in (-300, -150, 150, 300, 500):
        nl = n_best + d
        if 2000 <= nl <= 5000:
            lengths_to_try.append(nl)
    for nl in (2500, 3000, 3500, 4000):
        if nl not in lengths_to_try and 2000 <= nl <= 5000:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Sweep: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.3, 60.0)
        if t_md > 10:
            a_out, _ = mirror_descent(a_int, t_md, 100, 5.0, 4.0, 6000, False)
            try_update(a_out, f"len{nl}_md")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_lf = min(remaining * 0.2, 40.0)
            if t_lf > 8:
                a_out, _ = lbfgs_single(interpolate_seq(best_a, nl), t_lf, 150, False)
                try_update(a_out, f"len{nl}_lbfgs")

    # Phase 8: Ultimate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Ultimate Polish (n={len(best_a)}) ===")
    t8 = max(0, remaining - 15)
    if t8 > 20:
        a_out, _ = lbfgs_refine(best_a.copy(), t8, True)
        try_update(a_out, "ultimate_lbfgs")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = coordinate_descent(best_a.copy(), remaining - 5, False, 999)
            try_update(a_out, "ultimate_cd")

    # Final
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3200
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
```
