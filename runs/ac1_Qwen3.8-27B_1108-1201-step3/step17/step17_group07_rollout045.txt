def lbfgs_smoothed(a0, time_limit, verbose=False):
    """L-BFGS on the log-sum-exp smoothed objective in z-space."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    z0 = np.log(a + 1e-30) - np.log(a + 1e-30).mean()
    
    best_z = z0.copy()
    best_F = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Adaptive temperature schedule
    tau_start = 5.0
    tau_end = 200.0
    n_phases = 5
    phase_times = time_limit / n_phases
    
    for phase in range(n_phases):
        tau = tau_start * (tau_end / tau_start) ** (phase / max(n_phases - 1, 1))
        tau = max(tau, 3.0)
        
        def func_and_grad(z):
            F, grad, a_cur = softmax_obj(z, tau)
            return F, grad
        
        phase_start = time.monotonic()
        time_left = time_limit - (phase_start - t0)
        if time_left < 5:
            break
        
        result = sco.minimize(
            func_and_grad, z0, jac=True, method='L-BFGS-B',
            options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        z0 = result.x
        a_cur = np.exp(z0 - z0.max())
        a_cur = a_cur / a_cur.sum()
        m_cur = fast_conv(a_cur).max()
        
        if m_cur < best_F:
            best_F = m_cur
            best_z = z0.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_F
            print(f"    L-BFGS phase={phase} tau={tau:.1f} ({elapsed:.1f}s): J={J:.8f}")
    
    # Convert back
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, fast_conv(a_best).max()

def enhanced_remez(a0, time_limit, solver=None, verbose=False):
    """Enhanced LP Remez with better active set and trust region."""
    import cvxpy as cp
    
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = 0.05
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Better active set: use all elements above a threshold
        # Threshold based on the "flatness" of the top
        sorted_b = np.sort(b)[::-1]
        # Find where the "plateau" ends
        threshold = max_b * (1 - 0.005)
        active = np.where(b >= threshold)[0]
        
        if len(active) > 50:
            # Use top 50
            active = np.argsort(b)[::-1][:50]
        
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
        
        constraints.append(a_var <= a * (1.0 + r))
        constraints.append(a_var >= a * max(0.0, 1.0 - r))
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Line search
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_trial = (1 - t_frac) * a + t_frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                r = min(0.2, r * 1.05)
                break
        
        if not accepted:
            r = max(0.002, r * 0.6)
            if r < 0.003:
                break
    
    return best_a, best_m

def coord_descent_v2(a0, time_limit, rng_seed=42):
    """Improved coordinate descent with bisection on the optimal fraction."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            # Bisection on the optimal fraction to move
            lo, hi = 0.0, 0.9
            best_frac = 0.0
            best_trial_m = best_m
            
            for _ in range(8):
                mid = (lo + hi) / 2
                amt = a[i] * mid
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    hi = mid
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 0 and best_trial_m < best_m - 1e-16:
                amt = a[i] * best_frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_conv(a).max()
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""

import numpy as np
import scipy.optimize as sco
import cvxpy as cp
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size
