1. Smooth subgradient (log-sum-exp) mirror descent with adaptive beta
2. Frequency-domain spectrum flattening (Gerchberg-Saxton variant)
3. Coordinate descent refinement on top peaks
4. Multi-length search with diverse initializations
5. Symmetry-constrained optimization
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_grad_c_k(a, k):
    """Compute gradient of c[k] = conv(a,a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return grad
    for j in range(js, je + 1):
        kj = k - j
        if 0 <= kj < n:
            grad[j] += 2.0 * a[kj]
    return grad

def compute_smooth_grad(a, ac, beta=100.0, top_k=100):
    """
    Compute smooth approximation to subgradient of max_k c[k].
    Uses softmax over top-k peaks for efficiency.
    """
    n = len(a)
    num_lags = len(ac)
    k_max = min(top_k, num_lags)
    
    # Get top k peaks
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weights
    shifted = top_vals - top_vals.max()
    weights = np.exp(beta * shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_max) / k_max
    else:
        weights = weights / w_sum
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w < 1e-15:
            continue
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        if js > je:
            continue
        j_arr = np.arange(js, je + 1)
        kj_arr = k_int - j_arr
        grad[j_arr] += w * 2.0 * a[kj_arr]
    
    return grad

def amd_optimize(a0, time_limit, beta_init=200, beta_min=20, top_k_init=30, 
                 top_k_max=200, c0=5.0, verbose=False, rng_seed=42):
    """
    Accelerated Mirror Descent with smooth subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    restart_every = 5000
    
    cur_beta = beta_init
    cur_top_k = top_k_init
    see_count = 0  # count of beta/top_k adjustments
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive parameters
        progress = min(1.0, t / 50000.0)
        cur_beta = max(beta_min, beta_init * (1.0 - 0.5 * progress))
        cur_top_k = min(top_k_max, top_k_init + int(30 * progress))
        
        # Nesterov acceleration
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute smooth gradient
        ac = fast_autocorr(y)
        g = compute_smooth_grad(y, ac, beta=cur_beta, top_k=cur_top_k)
        
        # Mirror descent step (exponential family / entropy)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g / max(g.max(), 1e-15)
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart with perturbation
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.0f}, top_k={cur_top_k}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def freq_flatten(a0, time_limit, iters=5000, eta_schedule='linear', verbose=False):
    """
    Frequency-domain spectrum flattening.
    Iteratively pushes the power spectrum toward flatness.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_sum = P.sum()
        if P_sum < 1e-30:
            break
        
        # Normalize power spectrum
        P_norm = P / P_sum
        P_mean = 1.0 / len(P_norm)
        
        # Compute "excess" power (how much each bin exceeds the mean)
        excess = P_norm - P_mean
        
        # Schedule: start aggressive, become gentle
        progress = it / max(1, iters)
        if eta_schedule == 'linear':
            eta = 0.8 * (1.0 - 0.5 * progress)
        else:
            eta = 0.5
        
        # Flatten: push power from high bins to low bins
        P_target = P_norm - eta * excess
        P_target = np.maximum(P_target, 1e-30)
        
        # Preserve total power
        P_target = P_target / P_target.sum()
        
        # New spectrum: flatten magnitude, keep phase
        phase = np.angle(A)
        mag_new = np.sqrt(P_target * P_sum)  # restore scale
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqFlatten it={it} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, top_k=20, verbose=False, rng_seed=42):
    """
    Fine-tuned coordinate descent: adjust individual entries to reduce max peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        k_peaks = np.argsort(ac)[::-1][:top_k]
        
        improved = False
        
        for k in k_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find which pairs (i, k-i) contribute most to this peak
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            kj_idx = k_int - idx
            contrib = a[idx] * a[kj_idx]
            
            # Find the most contributing pair
            max_contrib_idx = np.argmax(contrib)
            i_src = idx[max_contrib_idx]
            j_src = kj_idx[max_contrib_idx]
            
            # Try to reduce by moving mass from the larger of a[i_src], a[j_src]
            # to a random position
            if a[i_src] >= a[j_src]:
                donor = i_src
            else:
                donor = j_src
            
            if a[donor] < 1e-12:
                continue
            
            # Try multiple fractions
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[donor] * frac
                
                # Choose destination: random position with low contribution to current peaks
                for trial in range(5):
                    dst = int(rng.integers(n))
                    if dst == donor:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[donor] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            d1 = int(rng.integers(n))
            d2 = int(rng.integers(n))
            if d1 != d2 and a[d1] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[d1] * frac
                a[d1] -= amt
                a[d2] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
