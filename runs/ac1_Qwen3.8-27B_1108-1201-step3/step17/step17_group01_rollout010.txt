def make_flat_spectrum_init(n, seed=0, fft_pad=2):
    rng = np.random.default_rng(seed)
    fft_size = n * fft_pad  # Zero-pad for linear autocorrelation
    # Random phase, constant amplitude
    phase = rng.uniform(0, 2*np.pi, fft_size // 2 + 1)
    phase[0] = 0  # DC component
    amp = np.ones(fft_size // 2 + 1)
    A = amp * np.exp(1j * phase)
    # Ensure conjugate symmetry
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size//2+1] = A
    A_full[fft_size//2+1:] = A[:fft_size//2][::-1].conj()
    a = np.fft.ifft(A_full).real
    a = np.maximum(a, 0)
    a = a[:n]
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

`R[i] = sum_k w_full[k+i] * a[k]` for `i = 0, 1, ..., n-1`.

The total length of the cross-correlation is `(2n-1) + n - 1 = 3n-2`. We need `N >= 3n-2`.

`R = ifft(conj(FFT(a, N)) * FFT(w_full, N))[:3n-2]`

Then `grad[i] = 2 * R[i]` for `i = 0, ..., n-1`.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient_fft(a, ac, top_k=200, beta=10.0):
    """Compute log-sum-exp subgradient via FFT-based cross-correlation."""
    n = len(a)
    ac_len = 2 * n - 1
    k_max = min(top_k, ac_len)
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    top_shifted = top_vals - top_vals[0]
    w = np.exp(beta * top_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    w_full = np.zeros(ac_len)
    w_full[top_idx] = w
    
    # Cross-correlation: R[i] = sum_k w_full[k+i] * a[k]
    N = 1 << (ac_len + n - 2 - 1).bit_length()
    W = np.fft.rfft(w_full, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(A) * W, N)
    
    grad = 2.0 * cross[:n]
    return grad

def mirror_descent_smoothed(a0, time_limit, c0=3.0, beta0=5.0, beta_end=50.0, 
                            top_k=200, restart_every=3000, verbose=False):
    """Smoothed (log-sum-exp) mirror descent with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = restart_every
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta: increase over time to focus more on the max
        progress = min(1.0, t / (time_limit * 200))
        beta_t = beta0 + (beta_end - beta0) * progress
        
        # Nesterov extrapolation
        beta_n = (t - 1.0) / (t + 1.5)
        y = x + beta_n * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute gradient
        ac = fast_autocorr(y)
        g = compute_gradient_fft(y, ac, top_k=top_k, beta=beta_t)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        score_new = 2.0 * n * M_new
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:

def make_flat_spectrum_init(n, seed=0):
    """Create initialization with flat power spectrum (random phase)."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    # Random phase, constant amplitude
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    phase[0] = 0.0
    A = np.ones(fft_size // 2 + 1) * np.exp(1j * phase)
    # Build full spectrum with conjugate symmetry
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size//2+1] = A
    A_full[fft_size//2+1:] = np.conj(A[1:fft_size//2][::-1])
    a = np.fft.ifft(A_full).real
    a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_cosine_init(n, mode=1):

def make_sinc_init(n, bandwidth=5.0):

    a = np.maximum(a, 0.0)

def make_power_law_init(n, power=1.0, center=False):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
