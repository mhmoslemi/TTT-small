```python
import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    """Numerically stable softmax."""
    z_max = z.max()
    e = np.exp(z - z_max)
    return e / e.sum()

def logsumexp_temperature(a, T):
    """Compute T * log(sum exp(ac[k]/T)) for given a and temperature T."""
    ac = fast_convolve(a)
    # Log-sum-exp for numerical stability
    ac_max = ac.max()
    val = ac_max / T + np.log(np.sum(np.exp((ac - ac_max) / T)))
```

```python
def annealed_logsumexp(a0, time_limit, T_init=None, T_min=None, n_anneal_steps=20):
    """
    Minimize the smoothed max-autocorrelation using annealed log-sum-exp
    with L-BFGS optimization in the log-parameterization.
    """
    n = len(a0)
    a = a0 / a0.sum()
    z = np.log(a + 1e-30)  # log-parameterization
    
    if T_init is None:
        ac = fast_convolve(a)
        T_init = ac.max() / 3.0
    if T_min is None:
        T_min = T_init / 10000.0
    
    best_a = a.copy()
    best_obj = float('inf')
    t0 = time.monotonic()
    
    # Annealing schedule: geometric
    T_values = np.geomspace(T_init, T_min, n_anneal_steps)
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Define objective and gradient in terms of z
        def obj_and_grad(z):
            a_local = softmax(z)
            ac = fast_convolve(a_local)
            
            # Log-sum-exp
            ac_shifted = (ac - ac.max()) / T
            S = np.sum(np.exp(ac_shifted))
            obj = ac.max() + T * np.log(S)
            
            # p (softmax over lags)
            p = np.exp(ac_shifted) / S
            
            # grad_a
            term1 = np.correlate(p, a_local, mode='valid')
            term2 = np.convolve(p, a_local, mode='full')[:n]
            grad_a = term1 + term2
            
            # grad_z
            grad_z = a_local * (grad_a - np.dot(grad_a, a_local))
            
            return obj, grad_z
        
        # Run L-BFGS
        result = minimize(
            obj_and_grad, z, method='L-BFGS-B', jac=True,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )
        
        z = result.x
        a_new = softmax(z)
        
        # Evaluate true objective
        ac = fast_convolve(a_new)
        J = 2 * n * ac.max()  # (with sum=1)
        if J < best_obj:
            best_obj = J
            best_a = a_new.copy()
        
        # Also try the unnormalized version for the actual score
        # (the score depends on the actual values, not just the shape)
        # But since the objective is scale-invariant, J = 2*n*max(ac)/sum^2
        # and with sum=1, J = 2*n*max(ac). So we're fine.
    
    return best_a, 2*n*fast_convolve(best_a).max()
```
