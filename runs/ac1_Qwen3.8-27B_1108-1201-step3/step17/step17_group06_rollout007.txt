   This is a Quadratic Program (QP) with many constraints (2n-1 of them). For n=3200, that's 6399 constraints. This could potentially be solved with a QP solver, but it might be too large.
   
   Actually, let me think about this differently. The constraint "sum_i a[i]*a[k-i] <= M" is equivalent to "a^T Q_k a <= M" where Q_k is the shift matrix. This is a non-convex constraint (since Q_k is not PSD in general... wait, actually for the full convolution, Q_k has entries that make a^T Q_k a = sum_i a[i]*a[i+k], which for non-negative a is always >= 0. But the constraint a^T Q_k a <= M is a non-convex quadratic constraint).
   
   So this is a non-convex QCQP (quadratically constrained quadratic program). Hard to solve exactly.
   
   Let me stick with the gradient-based approach.
   
   One more idea: **iterative water-filling**. 
   1. Start with a uniform distribution
   2. Identify the peak lag k*
   3. The peak is sum_i a[i]*a[k*-i]. To reduce it, we want to "de-correlate" a from its k*-shifted version.
   4. One way: reduce a[i] where a[i]*a[k*-i] is large, and increase a[j] where a[j]*a[k*-j] is small.
   5. This is like a multiplicative update: a[i] <- a[i] * exp(-eta * a[k*-i])
   
   This is actually very similar to the mirror descent approach! The subgradient of max(conv) at the peak lag k* is proportional to (a shifted by k*). So the mirror descent step a <- a * exp(-eta * a_shifted) is exactly this.
   
   So the previous approach is already doing a form of water-filling. The question is whether we can do it more efficiently.
   
   Key improvements I'll implement:
   1. **Exact gradient via log-sum-exp** (smooth, differentiable)
   2. **L-BFGS for fast local convergence**
   3. **Adaptive temperature annealing**
   4. **Multi-start with diverse initializations**
   5. **Hybrid: L-BFGS + mirror descent**
   6. **Different sequence lengths**
   
   Let me now write the code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp approximation + L-BFGS-B for precise local optimization
2. Adaptive temperature annealing for gradual sharpening
3. Multi-strategy initialization (parent, ramps, random smooth)
4. Hybrid mirror descent for exploration
5. Peak-shaving coordinate updates for final polish
6. Length sweeping for optimal n
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_conv(a)
    return float(2.0 * n * conv.max() / (s * s))

def lse_objective_and_grad(b, t, n):
    """
    Compute LSE objective and gradient in log-parameterization.
    b: log-parameters, a = exp(b) / sum(exp(b)) (softmax)
    Actually, let's use a = exp(b) directly (unnormalized) and include 
    the normalization in the objective.
    
    Since the objective is scale-invariant, we can work with a = exp(b)
    and the objective becomes 2*n*LSE(conv(a))/(sum(a))^2.
    """
    a = np.exp(b)
    s = a.sum()
    if s < 1e-30:
        return 1e10, np.zeros(n)
    # Normalize for numerical stability
    a_norm = a / s
    conv = fast_conv(a_norm)
    # Log-sum-exp
    conv_max = conv.max()
    shifted = conv / t - conv_max / t
    log_sum = conv_max / t + np.log(np.exp(shifted).sum())
    lse_val = t * log_sum
    
    # Gradient of LSE w.r.t. a_norm
    p = np.exp(shifted) / np.exp(shifted).sum()  # probability over lags
    # grad_LSE[i] = 2 * sum_j a_norm[j] * p[i+j]
    # This is cross-correlation of a_norm and p
    # Compute via FFT
    fft_size_cc = 1 << ((n - 1) + (2 * n - 1) - 1 - 1).bit_length()  # need >= 3n-3
    # Actually: cross-correlation of a (len n) and p (len 2n-1)
    # Result length: n + (2n-1) - 1 = 3n-2
    # We need indices 0 to n-1 of the cross-correlation
    # Cross-corr[i] = sum_j a[j] * p[j+i]
    # = (a_rev * p)[i + n - 1] where a_rev = a[::-1]
    a_rev = a_norm[::-1]
    fft_sz = 1 << (len(a_rev) + len(p) - 1 - 1).bit_length()
    F_a = np.fft.rfft(a_rev, fft_sz)
    F_p = np.fft.rfft(p, fft_sz)
    cc = np.fft.irfft(F_a * F_p, fft_sz)
    # cc[i + n - 1] = cross_corr[i] for i = 0, ..., n-1
    grad_lse = 2.0 * cc[n-1:2*n-1]
    
    # Gradient of full objective w.r.t. a_norm:
    # f = 2*n*lse_val (since sum=1, denominator is 1)
    # grad_f[i] = 2*n * grad_lse[i]
    grad_f = 2.0 * n * grad_lse
    
    # Gradient w.r.t. b: da_i/db_i = a_i = a_norm[i] (since a = exp(b)/s, da_i/db_j = a_i(δ_ij - a_j))
    # Actually, since the objective is scale-invariant, we can simplify.
    # Let's work directly: f(b) = 2*n*LSE(conv(a))/s^2 where a = exp(b), s = sum(a)
    # ∂f/∂b_i = ∂f/∂a_i * da_i/db_i = ∂f/∂a_i * a_i
    # But we computed grad in terms of a_norm = a/s.
    # f = 2*n*LSE(conv(a_norm)) [since scale invariant]
    # ∂f/∂a_norm_j = 2*n*grad_lse[j]
    # ∂a_norm_j/∂b_i = (δ_ij * a_i - a_i * a_j) / s = a_norm_i * (δ_ij - a_norm_j)
    # Hmm this is getting complicated. Let me just use the unnormalized version.
    
    # Simpler: since f is scale invariant, ∇_a f * a = 0 (Euler's theorem for homogeneous degree 0)
    # So the gradient in b-space is just: grad_b[i] = grad_a[i] * a_i
    # where grad_a is the gradient w.r.t. a (unnormalized)
    # But since f is scale-invariant, we can compute grad w.r.t. a_norm and project.
    
    # Actually the simplest approach: just compute the gradient numerically or use 
    # the chain rule properly.
    
    # Let me redo: f(a) = 2*n*LSE(conv(a))/(sum(a))^2
    # This is homogeneous of degree 0 in a.
    # ∂f/∂a_i = 2*n * [S^2 * ∂LSE/∂a_i - LSE * 2*S*a_i] / S^4
    #           = 2*n/S * [∂LSE/∂a_i - 2*LSE*a_i/S]
    # where S = sum(a)
    
    # For a_norm = a/S (so sum(a_norm) = 1):
    # conv(a) = S^2 * conv(a_norm)
    # LSE(conv(a)) = S^2 * LSE_conv_scaled... 
    
    # This is getting complicated. Let me just fix S=1 by normalizing.
    # Work with a_norm directly. The constraint is sum(a_norm)=1, a_norm>=0.
    # f(a_norm) = 2*n*LSE(conv(a_norm))
    # grad_f[i] = 2*n*grad_lse[i] (computed above)
    
    # To use L-BFGS-B, I'll parameterize with b where a_i = exp(b_i), 
    # and rely on the scale invariance. At each evaluation, I normalize.
    # The gradient w.r.t. b is:
    # grad_b[i] = sum_j (df/da_j) * (da_j/db_i)
    # da_j/db_i = a_j * (δ_ji - a_j) [for softmax parameterization]
    # But I'm using a = exp(b) (not softmax), so da_j/db_i = δ_ji * a_j
    # And f(exp(b)) with f scale-invariant means:
    # d/db_i f(exp(b)) = (df/da_i) * a_i
    # But since f is scale-invariant, sum_i (df/da_i) * a_i = 0
    # So the gradient in b-space is just df/da_i * a_i (evaluated at a = exp(b))
    # And we can normalize a before evaluating f.
    
    # Simplest: compute f and grad at a = exp(b) / sum(exp(b)), and the grad w.r.t. b
    # is proj(grad) * exp(b) where proj removes the component along a (since 
    # moving along the scale doesn't change f).
    
    # Actually, the cleanest: just use a directly with L-BFGS-B and bounds [0, inf).
    # The scale invariance means L-BFGS will find a direction that reduces f,
    # and the bound constraint handles non-negativity.
    
    obj = 2.0 * n * lse_val
    grad = 2.0 * n * grad_lse  # gradient w.r.t. a_norm
    
    return obj, grad

def lse_optimize(a0, t=0.01, maxiter=5000, tol=1e-10):
    """Optimize using L-BFGS-B on the log-sum-exp approximation."""
    n = len(a0)
    a0 = np.maximum(a0, 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
        s = 1.0
    a0 = a0 / s
    
    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-30:
            return 1e10
        x = x / s
        conv = fast_conv(x)
        conv_max = conv.max()
        shifted = (conv - conv_max) / t
        lse = t * (conv_max / t + np.log(np.exp(shifted).sum()))
        return 2.0 * n * lse
    
    def gradient(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-30:
            return np.zeros(n)
        x = x / s
        conv = fast_conv(x)
        conv_max = conv.max()
        shifted = (conv - conv_max) / t
        p = np.exp(shifted) / np.exp(shifted).sum()
        # Cross-correlation for gradient
        a_rev = x[::-1]
        fft_sz = 1 << (n + (2 * n - 1) - 1 - 1).bit_length()
        F_a = np.fft.rfft(a_rev, fft_sz)
        F_p = np.fft.rfft(p, fft_sz)
        cc = np.fft.irfft(F_a * F_p, fft_sz)
        grad_lse = 2.0 * cc[n-1:2*n-1]
        return 2.0 * n * grad_lse
    
    result = minimize(
        objective, a0, jac=gradient, method='L-BFGS-B',
        bounds=[(0, None)] * n,
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8, 'maxls': 50}
    )
    
    a_opt = np.maximum(result.x, 0)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt / s
    return a_opt, result

def mirror_descent_smooth(a0, time_limit, t=0.05, top_k=80, beta_w=10.0, c0=4.0,
                          restart_every=5000):
    """Mirror descent using exact LSE gradient (not subgradient approximation)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    x = a.copy()
    x_prev = a.copy()
    
    for t_iter in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t_iter > 1:
            beta_t = (t_iter - 2.0) / (t_iter + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Exact LSE gradient
        conv = fast_conv(y)
        conv_max = conv.max()
        shifted = (conv - conv_max) / t
        p = np.exp(shifted) / np.exp(shifted).sum()
        
        # Gradient via cross-correlation
        a_rev = y[::-1]
        fft_sz = 1 << (n + (2 * n - 1) - 1 - 1).bit_length()
        F_a = np.fft.rfft(a_rev, fft_sz)
        F_p = np.fft.rfft(p, fft_sz)
        cc = np.fft.irfft(F_a * F_p, fft_sz)
        g = 2.0 * cc[n-1:2*n-1]
        
        # Mirror step
        eta = c0 / (np.sqrt(t_iter) + 1.0)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t_iter + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-16:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic restart
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
    
    return best_a, best_val

def peak_shave(a0, time_limit, seed=0):
    """Direct peak shaving: iteratively reduce the maximum convolution lag."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_conv(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        
        # Compute the "pressure" at each position from the peak lag
        # pressure[i] = a[k_star - i] (contribution to peak if a[i] increases)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        
        if i_start > i_end:
            stall += 1
            continue
        
        pressure = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = k_star - i
            pressure[i] = a[j]
        
        # Also consider other high lags
        sorted_lags = np.argsort(conv)[::-1][:10]
        for k in sorted_lags[1:]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            for i in range(i_s, i_e + 1):
                j = k_i - i
                weight = conv[k_i] / M
                pressure[i] += weight * a[j]
        
        # Find source: high a[i] AND high pressure[i]
        src_score = pressure * a
        src = int(np.argmax(src_score))
        
        if a[src] < 1e-10:
            stall += 1
            continue
        
        # Find destination: low pressure, and check impact on all peaks
        dest_score = np.zeros(n)
        for k in sorted_lags[:5]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            for i in range(i_s, i_e + 1):
                j = k_i - i
                dest_score[i] += a[j]
        
        dest_score[src] = np.inf
        dst = int(np.argmin(dest_score))
        
        # Try transfers
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            amt = a[src] * frac
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-15:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.005, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
    
    return best_a, best_J

def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_bimodal(n, width=50.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.2) ** 2) + np.exp(-width * (x - 0.8) ** 2) + 0.01
    return a / a.sum()

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_chebyshev(n, alpha=0.5):
    """Chebyshev-like distribution: a_i proportional to (1 - (i/(n-1))^2)^alpha"""
    x = np.linspace(-1, 1, n)
    a = np.maximum(1 - x**2, 0) ** alpha
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_cosine(n):
    """Cosine window."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_triangular(n):
    """Triangular distribution concentrated in the middle."""
    x = np.linspace(-1, 1, n)
    a = np.maximum(1 - np.abs(x), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
```
