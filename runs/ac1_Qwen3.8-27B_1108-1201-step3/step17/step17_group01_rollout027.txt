    return g


def mirror_descent(a0, time_limit, top_k=30, beta=3.0, eta0=0.5,
                   restart_every=500, verbose=False, rng_seed=42):
    """
    Mirror descent (exponentiated gradient) for minimizing max conv(a,a).
    Normalized so sum(a)=1. Objective: 2*n*max(conv(a,a)).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    t = 0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    improvement_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        eta = eta0 / (1.0 + 0.01 * t)

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute convolution and find peaks
        conv = fast_conv(y)
        # Find top peaks (exclude k=0 and k=2n-2 as they are a[0]^2 and a[n-1]^2)
        mask = np.ones(len(conv), dtype=bool)
        mask[0] = False
        mask[-1] = False
        conv_masked = np.where(mask, conv, -1e30)
        k_max = np.argsort(conv_masked)[::-1][:cur_top_k]
        k_vals = np.maximum(conv[k_max], 1e-30)

        # Weight peaks by their magnitude (soft max)
        w = k_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(cur_top_k) / cur_top_k
        else:
            w = w / w_sum

        # Compute subgradient
        g = compute_subgradient(y, k_max, w)

        # Mirror descent update (multiplicative)
        log_a = np.log(y) - eta * g / max(g.max(), 1e-10) * eta
        # Actually use: a_new = a * exp(-eta * g / scale)
        scale = max(g.max(), 1e-10)
        log_a = np.log(y) - (eta / scale) * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            improvement_count += 1
        else:
            stall_count += 1

        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            cur_top_k = min(cur_top_k + 5, 80)
            cur_beta = max(cur_beta - 0.2, 1.5)
            eta0 = min(eta0 * 1.1, 3.0)
            # Perturb
            strength = 0.05
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}, top_k={cur_top_k}")

    return best_a, best_m


def freq_flatten(a0, time_limit, n_iters=2000, alpha=0.3, verbose=False):
    """
    Frequency-domain iterative amplitude shaping.
    Goal: make |FFT(a)|^2 as flat as possible (which makes conv(a,a) closer to uniform).
    Actually, we want A^2 (not |A|^2) to be flat. But for real non-negative a,
    we can work with the magnitude spectrum as an approximation.
    
    Alternative: directly shape the spectrum to minimize the peak of IFFT(A^2).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        # Compute A^2
        A2 = A * A
        # IFFT of A^2 gives the convolution
        conv_full = np.fft.irfft(A2, fft_size)

        # Find peaks in convolution
        conv_len = 2 * n - 1
        peaks_idx = np.argsort(conv_full[:conv_len])[::-1][:20]
        peak_vals = np.maximum(conv_full[peaks_idx], 1e-30)
        target = conv_full[:conv_len].mean()

        # In frequency domain, reduce the components that create peaks
        # Use gradient of max(conv) w.r.t. A2, which is:
        # d max(conv) / d A2[m] = exp(2*pi*i*m*k*/N) for peak lag k*
        # So in the A^2 domain, we want to subtract the "peak pattern"
        # A2_new = A2 - alpha * IFFT(peak_pattern) in freq domain

        # Simpler approach: reduce amplitude at frequencies that contribute to peaks
        # The peak at lag k* gets contribution from all frequencies, but the "excess"
        # over the mean is what we want to reduce.
        excess = conv_full[:conv_len].copy()
        excess[excess < target] = 0
        excess -= target  # Center

        # Project excess back to frequency domain to get correction
        A2_excess = np.fft.rfft(np.concatenate([excess, np.zeros(fft_size - conv_len)]), fft_size)
        A2_new = A2 - alpha * A2_excess

        # Recover A from A2_new: A_new = sqrt(A2_new) (in complex sense)
        # This is tricky for complex numbers. Use magnitude-based approach.
        mag_new = np.sqrt(np.abs(A2_new))
        phase_new = np.angle(A)
        A_new = mag_new * np.exp(1j * phase_new)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqFlat it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def spectral_shaping(a0, time_limit, verbose=False):
    """
    Direct frequency-domain optimization:
    Minimize max(IFFT(A^2)) subject to IFFT(A) >= 0, sum(IFFT(A)) = 1.
    Use iterative projection: alternate between flattening A^2 and projecting
    the time-domain signal to be non-negative with unit sum.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    # Half spectrum length
    half = fft_size // 2 + 1

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        A2 = A * A

        conv = np.fft.irfft(A2, fft_size)[:2 * n - 1]
        M_curr = conv.max()
        mean_val = conv.mean()
        if M_curr < mean_val * 1.01:
            break

        # Find the peak lags
        top_idx = np.argsort(conv)[::-1][:30]
        peak_vals = np.maximum(conv[top_idx], 1e-20)

        # Target: make convolution as flat as possible
        # In frequency: we want A^2 to have its IFFT as flat as possible
        # This means A^2 should be concentrated at DC (frequency 0)
        # But A[0] = sum(a) = 1, so A^2[0] = 1
        # The "excess" energy in A^2 at other frequencies causes non-flatness
        
        # Reduce non-DC components of A^2
        A2_reduced = A2.copy()
        decay = 0.995
        A2_reduced[1:] *= decay

        # Reconstruct A from A2_reduced
        # A_new[m] should satisfy A_new[m]^2 = A2_reduced[m]
        # For the IFFT to be real, we need A_new[N-m] = conj(A_new[m])
        # So A_new[m] = sqrt(A2_reduced[m]) with consistent phase
        mag = np.sqrt(np.abs(A2_reduced))
        # Preserve original phase patterns
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blending
            a = 0.8 * a + 0.2 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initializations(n):
    """Generate diverse initializations for length n."""
    x = np.linspace(0, 1, n)
    inits = []
    labels = []

    # Cosine window
    a = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(norm_sum1(a))
    labels.append("cos1")

    # Sinc-like
    for bw in [3, 5, 8]:
        t = (x - 0.5) * bw
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0.0)
        inits.append(norm_sum1(a))
        labels.append(f"sinc{bw}")

    # Power law (ramp)
    for p in [0.5, 1.0, 2.0]:
        a = (x + 0.1) ** p
        inits.append(norm_sum1(a))
        labels.append(f"pow{p}")

    # Inverse power law (centered)
    for p in [0.5, 1.0]:
        xc = np.abs(x - 0.5) + 0.05
        a = xc ** (-p)
        inits.append(norm_sum1(a))
        labels.append(f"inv_pow{p}")

    # Multi-hump
    for nh in [3, 5, 7]:
        a = np.zeros(n)
        for i in range(nh):
            c = (i + 1) / (nh + 1)
            a += np.exp(-((x - c) ** 2) / (2 * 0.1 ** 2))
        a += 0.01
        inits.append(norm_sum1(a))
        labels.append(f"{nh}hump")

    # Random smooth
    rng = np.random.default_rng(123)
    a = np.ones(n) * 0.5
    for f in range(1, 12):
        c = rng.normal(0, 0.1 / np.sqrt(f))
        ph = rng.uniform(0, 2 * np.pi)
        a += c * np.sin(2 * np.pi * f * x + ph)
    inits.append(norm_sum1(np.maximum(a, 0.001)))
    labels.append("rand_smooth")

    # Triangular
    a = np.maximum(1 - 2 * np.abs(x - 0.5), 0)
    inits.append(norm_sum1(a + 0.01))
    labels.append("tri")

    return inits, labels


def local_search(a0, time_limit, n_top=10, verbose=False, rng_seed=42):
    """
    Local search: try moving mass from positions that contribute most to peaks
    to positions that contribute least.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        conv = fast_conv(a)
        top_k = np.argsort(conv)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination: minimize contribution to all top peaks
            costs = np.zeros(n)
            for k2 in top_k[:min(6, len(top_k))]:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                if js2 > je2:
                    continue
                idx2 = np.arange(js2, je2 + 1)
                c2 = a[idx2] * a[k2i - idx2]
                # Cost of placing mass at position d: it would contribute a[d]*a[k2i-d]
                # approximate by a[k2i-d] if valid
                for d in range(n):
                    j = k2i - d
                    if 0 <= j < n:
                        costs[d] += a[j]

            # Prefer positions with low cost and low current value
            candidates = np.argsort(costs + a)[ :max(5, n // 20)]
            best_dst = candidates[0]

            if best_dst == src:
                if len(candidates) > 1:
                    best_dst = candidates[1]
                else:
                    continue

            # Try different fractions
            found = False
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found and stall > 20:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Larger random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LocalSearch ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = np.cos(np.pi * (x - 0.5)) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent on parent and variants ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 60:
        # Run MD on the parent
        t_per = t_phase1 / 4
        print(f"  MD on parent (n={n0}, t={t_per:.0f}s)...")
        a_out, _ = mirror_descent(best_seq, t_per, top_k=30, beta=3.0, eta0=0.5,
                                   restart_every=500, verbose=False, rng_seed=42)
        try_update(a_out, f"md_parent_n{n0}")

        # MD with different hyperparams
        remaining = time_budget - (time.monotonic() - t_global)
        t_per = min(remaining * 0.15, 150.0)
        if t_per > 30:
            print(f"  MD aggressive (n={n0}, t={t_per:.0f}s)...")
            a_out, _ = mirror_descent(best_seq, t_per, top_k=50, beta=2.0, eta0=1.0,
                                       restart_every=300, verbose=False, rng_seed=7)
            try_update(a_out, f"md_agg_n{n0}")

        # MD with gentler params
        remaining = time_budget - (time.monotonic() - t_global)
        t_per = min(remaining * 0.15, 150.0)
        if t_per > 30:
            print(f"  MD gentle (n={n0}, t={t_per:.0f}s)...")
            a_out, _ = mirror_descent(best_seq, t_per, top_k=20, beta=5.0, eta0=0.3,
                                       restart_every=1000, verbose=False, rng_seed=99)
            try_update(a_out, f"md_gentle_n{n0}")

    # === Phase 2: Try larger sequence lengths ===
    print(f"\n=== Phase 2: Larger n ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    target_lengths = []
    n_best = len(best_seq)
    # Try scaling up
    for factor in [1.5, 2.0, 3.0, 4.0]:
        new_n = int(n_best * factor)
        if new_n <= 25000 and new_n > n_best:
            target_lengths.append(new_n)
    
    # Also try some absolute targets
    for target in [8000, 12000, 16000, 20000]:
        if target > n_best and target <= 25000:
            if target not in target_lengths:
                target_lengths.append(target)
    
    target_lengths = sorted(set(target_lengths))[:4]

    for new_n in target_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t_md = min(remaining * 0.25, 200.0)
        if t_md > 30:
            a_out, _ = mirror_descent(a_long, t_md, top_k=40, beta=3.0, eta0=0.5,
                                       restart_every=400, verbose=False, rng_seed=42)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t_fs = min(remaining * 0.15, 100.0)
            if t_fs > 20:
                a_out, _ = freq_flatten(a_long, t_fs, n_iters=1000, alpha=0.2)
                try_update(a_out, f"freq_n{new_n}")

    # === Phase 3: Spectral shaping on best ===
    print(f"\n=== Phase 3: Spectral Shaping ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_out, _ = spectral_shaping(best_seq, t3, verbose=False)
        try_update(a_out, f"spectral_n{n_best}")

    # === Phase 4: Frequency domain flattening ===
    print(f"\n=== Phase 4: Freq Flattening ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = freq_flatten(best_seq, t4, n_iters=3000, alpha=0.3)
        try_update(a_out, f"freqflat_n{n_best}")

    # === Phase 5: Local search / coordinate descent ===
    print(f"\n=== Phase 5: Local Search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_out, _ = local_search(best_seq, t5, n_top=15, verbose=False, rng_seed=42)
        try_update(a_out, f"local_n{n_best}")

    # === Phase 6: Refined MD with more peaks ===
    print(f"\n=== Phase 6: Refined MD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 20:
        a_out, _ = mirror_descent(best_seq, t6, top_k=60, beta=2.5, eta0=0.8,
                                   restart_every=300, verbose=False, rng_seed=123)
        try_update(a_out, f"md_refined_n{n_best}")

    # === Phase 7: Multi-init at best length ===
    print(f"\n=== Phase 7: Multi-init ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        inits, labels = make_initializations(n_best)
        # Add parent
        inits.insert(0, norm_sum1(best_seq))
        labels.insert(0, "current_best")
        
        t_per_init = max(15.0, min(remaining * 0.08, 60.0))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per_init, remaining - 30)
            if t_avail < 10:
                break
            if idx > 5:  # Limit number of inits
                break
            print(f"  MD '{lbl}' (n={n_best}, t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent(init, t_avail, top_k=30, beta=3.0, eta0=0.5,
                                       restart_every=400, verbose=False, rng_seed=idx * 17 + 1)
            try_update(a_out, f"md_init_{lbl}")

    # === Phase 8: Final polish with local search ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.15, 120.0)
    if t8 > 15:
        a_out, _ = local_search(best_seq, t8, n_top=10, verbose=True, rng_seed=777)
        try_update(a_out, f"polish_n{n_best}")

    # === Phase 9: Final MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.20, 150.0)
    if t9 > 15:
        a_out, _ = mirror_descent(best_seq, t9, top_k=80, beta=2.0, eta0=1.0,
                                   restart_every=200, verbose=True, rng_seed=456)
        try_update(a_out, f"md_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.cos(np.pi * (x - 0.5)) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
