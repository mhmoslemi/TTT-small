def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1: 2*n*max(ac)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    """Compute subgradient of max(conv(a,a))."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_frank_wolfe_v3(a0, time_limit, solver, verbose=False,
                      start_K=200, max_K=1500, K_growth=1.3):
    """
    Enhanced LP Frank-Wolfe with:
    - Proper bilinear gradient (both forward and backward)
    - Adaptive constraint count (grows when stalling)
    - Backtracking line search on actual objective
    - Periodic restart with more constraints
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = compute_max_ac(a)
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    cur_K = start_K
    stall = 0
    max_stall = 100

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Build constraint matrix with proper bilinear gradient
        # For lag k: grad_k[j] = y[k-j]*I(0<=k-j<n) + y[j-k]*I(0<=j-k<n)
        # Constraint: grad_k . b <= 2 * (M_curr - ac[k])  (for active constraints)
        # Actually for FW: minimize max_k [grad_k . b - ac[k]]
        # => grad_k . b <= alpha + ac[k] for all active k
        # We use: grad_k . b <= 2*M_curr for the top-K most binding

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y  # forward part: y[k-j]
            # backward part: y[j-k] contributes to T_full[k, j] = y[j-k] when j >= k
            # But that's already included in T_full[j-k, j-k] ... no.
            # Let me redo: T_full[k, j] should be y[k-j] + y[j-k] (where valid)
            # T_full[k, j] = y[k-j] if 0<=k-j<n, plus y[j-k] if 0<=j-k<n

        # Actually let me rebuild properly
        T_full = np.zeros((2 * n - 1, n))
        # Forward: T[k,j] += y[k-j] when 0 <= k-j < n
        for j in range(n):
            T_full[j:j + n, j] += y  # T[j+r, j] += y[r] for r=0..n-1
        # Backward: T[k,j] += y[j-k] when 0 <= j-k < n
        for j in range(n):
            # j-k >= 0 means k <= j, so k ranges 0..j
            # j-k < n means k > j-n
            k_start = max(0, j - n + 1)
            k_end = j + 1
            T_full[k_start:k_end, j] += np.flip(y[j - (k_end - 1):j - k_start + 1][::-1])
            # Let me be more careful: for fixed j, k ranges from max(0,j-n+1) to j
            # y[j-k] for k in that range
            # y[j-k] = y[j - max(0,j-n+1)] down to y[j-j] = y[0]
            # = y[n-1] down to y[0] (if j >= n-1) or y[j] down to y[0]
            # Let me just compute directly:
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            # Forward contribution: T[k,j] += y[k-j] for k where 0<=k-j<n
            # k ranges from j to j+n-1
            for offset in range(n):
                k = j + offset
                T_full[k, j] += y[offset]
            # Backward contribution: T[k,j] += y[j-k] for k where 0<=j-k<n
            # k ranges from max(0, j-n+1) to j
            for offset in range(n):
                k = j - offset
                if k >= 0:
                    T_full[k, j] += y[offset]

        # This is O(n^2) which is too slow for n=3200. Let me use a vectorized approach.
        # Actually for n=3200, n^2 = 10M which is fine for a one-time cost, but we do it every iteration.
        # Let me use the FFT-based approach or a smarter construction.

        # Better: use the Toeplitz structure
        # T_full[k, j] = y[k-j] (if 0<=k-j<n) + y[j-k] (if 0<=j-k<n)
        # This is symmetric: T_full[k,j] = T_full[j,k] ... no, it's not because the dimensions differ.
        # Actually T_full is (2n-1) x n, not square.

        # Let me just use the one-sided version for efficiency (same as previous code):
        # Constraint: sum_j y[k-j] * b[j] <= M_curr
        # This is a valid relaxation and much cheaper to build.

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        # Select top-K binding constraints
        margins = M_curr - ac
        K_sel = min(cur_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_rhs = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = compute_max_ac(y)
        if my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        mx = compute_max_ac(x)
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            stall = 0

        # Adaptive K growth
        if stall > max_stall:
            stall = 0
            cur_K = min(int(cur_K * K_growth), max_K)
            # Also try a small random perturbation
            pert = np.random.default_rng(t).normal(0, 0.005, n) * y
            y_pert = np.maximum(y + pert, 0)
            y_pert = y_pert / y_pert.sum()
            if compute_max_ac(y_pert) < best_m:
                y = y_pert
                best_m = compute_max_ac(y)
                best_a = y.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LP-FW3 restart ({elapsed:.0f}s): J={J:.6f}, K={cur_K}")

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW3 iter={t} ({elapsed:.0f}s): J={J:.6f}, K={cur_K}")

    return best_a, best_m


def coordinate_descent_v2(a0, time_limit, verbose=False, seed=77):
    """
    Targeted coordinate descent: identify peak contributors and relocate mass.
    Uses exact sensitivity computation and line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get peak lags
        n_peaks = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity for top peaks
        sensitivity = np.zeros(n)
        for k in top_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc

        # Find good sources and destinations
        sens_mean = sensitivity.mean()
        sens_std = sensitivity.std() + 1e-15
        high_mask = (sensitivity > sens_mean + 0.5 * sens_std) & (a > 0.0005)
        low_mask = (sensitivity < sens_mean) & (a < 0.5 * np.max(a))

        high_idx = np.where(high_mask)[0]
        low_idx = np.where(low_mask)[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            # Fallback: random moves
            high_idx = np.where(a > np.median(a))[0]
            low_idx = np.where(a < np.median(a))[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            stall += 1
            continue

        improved = False

        # Try multiple moves per iteration
        n_attempts = min(50, len(high_idx) * len(low_idx) // 10 + 10)
        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_idx[rng.integers(len(high_idx))]
            dst = low_idx[rng.integers(len(low_idx))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Line search over transfer fractions
            best_frac = None
            best_trial_J = compute_J(a)

            fracs = np.linspace(0.01, 0.4, 8) + rng.uniform(-0.01, 0.01, 8)
            fracs = np.clip(fracs, 0.001, 0.5)

            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_max_ac(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_frac = frac

            if best_frac is not None:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    improved = True
                    stall = 0

        if not improved:
            stall += 1
            # Periodic random perturbation
            if stall % 20 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if compute_J(a) < best_J:
                    best_J = compute_J(a)
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc final ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def simulated_annealing(a0, time_limit, verbose=False, seed=999):
    """
    Simulated annealing with small coordinate perturbations.
    Energy = max(ac). Accept worse moves with probability exp(-delta/T).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_energy = compute_max_ac(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_start = 0.002
    T_end = 1e-6
    total_iters = 500000
    current_energy = best_energy

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Annealing schedule (geometric)
        progress = it / total_iters
        T = T_start * (T_end / T_start) ** progress

        # Generate a move: perturb 1-3 random coordinates
        n_pert = rng.integers(1, 4)
        a_trial = a.copy()
        for _ in range(n_pert):
            i = int(rng.integers(n))
            if a_trial[i] <= 0:
                # Move mass TO this position
                j = int(rng.integers(n))
                if j == i or a_trial[j] <= 0:
                    continue
                amt = a_trial[j] * rng.uniform(0.001, 0.02)
                a_trial[j] -= amt
                a_trial[i] += amt
            else:
                # Move mass FROM this position
                j = int(rng.integers(n))
                if j == i:
                    continue
                amt = a_trial[i] * rng.uniform(0.001, 0.02)
                a_trial[i] -= amt
                a_trial[j] += amt

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        trial_energy = compute_max_ac(a_trial)
        delta = trial_energy - current_energy

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            current_energy = trial_energy
            if current_energy < best_energy:
                best_energy = current_energy
                best_a = a.copy()
                if verbose and it % 100000 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_energy
                    print(f"    SA it={it} ({elapsed:.0f}s): J={J:.6f}, T={T:.2e}")

    return best_a, best_energy


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Symmetric initialization (high at edges)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_asymmetric_init(n):
    """Asymmetric initialization - front-loaded."""
    x = np.linspace(0, 1, n)
    a = np.exp(-3 * x) + 0.5 * np.exp(-((x - 0.7) / 0.15) ** 2) + 0.01
    return norm_sum1(a)


def make_bimodal_init(n):
    """Bimodal - two peaks at 1/3 and 2/3."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.3) / 0.1) ** 2) + np.exp(-((x - 0.7) / 0.1) ** 2) + 0.01
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with adaptive multi-phase optimization.
    Phases:
    1. Quick subgradient from parent + diverse inits
    2. Intensive LP Frank-Wolfe
    3. Coordinate descent with sensitivity
    4. Simulated annealing
    5. Length exploration
    6. Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Quick subgradient from parent ===
    print(f"\n=== Phase 1: Quick SGD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t1, verbose=True,
                                        top_k=15, beta=3.0, c=3.0,
                                        exact_frac=0.5, restart_every=2000,
                                        seed=42)
        try_update(a_out, f"sgd_parent_n{n0}")

        # Also try a diverse init
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_div = make_asymmetric_init(n0)
            if parent is not None:
                a_div = 0.4 * norm_sum1(parent) + 0.6 * a_div
                a_div = norm_sum1(a_div)
            a_out, _ = nesterov_subgradient(a_div, min(remaining * 0.15, 80.0), verbose=False,
                                            top_k=20, beta=2.5, c=4.0,
                                            exact_frac=0.5, restart_every=1500,
                                            seed=123)
            try_update(a_out, f"sgd_asym_n{n0}")

    # === Phase 2: LP Frank-Wolfe (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW v3 (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 50 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v3(a_in, t2, solver, verbose=True,
                                     start_K=300, max_K=1200, K_growth=1.4)
        try_update(a_out, f"lpfw3_n{n_best}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coordinate Descent (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_v2(a_in, t3, verbose=True, seed=77)
        try_update(a_out, f"coorddesc_n{n_best}")

    # === Phase 4: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Simulated Annealing (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = simulated_annealing(a_in, t4, verbose=True, seed=999)
        try_update(a_out, f"sa_n{n_best}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration ===")

    lengths_to_try = []
    # Try slightly longer and shorter
    for delta in [200, 400, -200, -400, 600]:
        candidate_n = n_best + delta
        if 1000 <= candidate_n <= 5000 and candidate_n != n_best:
            lengths_to_try.append(candidate_n)

    # Deduplicate
    lengths_to_try = list(dict.fromkeys(lengths_to_try))[:4]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick SGD
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 20:
            a_out, _ = nesterov_subgradient(a_long, t5a, verbose=False,
                                            top_k=20, beta=2.5, c=4.0,
                                            exact_frac=0.5, restart_every=2000,
                                            seed=555 + new_n % 100)
            try_update(a_out, f"sgd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break

        # Quick LP-FW on the best at this length
        t5b = min(remaining * 0.20, 100.0)
        if t5b > 20 and new_n <= 4000:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t5b, solver, verbose=False,
                                         start_K=200, max_K=800, K_growth=1.5)
            try_update(a_out, f"lpfw_n{new_n}")

    # === Phase 6: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 4000:
        t6 = min(remaining * 0.30, 250.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v3(a_in, t6, solver, verbose=True,
                                         start_K=500, max_K=2000, K_growth=1.3)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CoordDesc (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.25, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_v2(a_in, t7, verbose=True, seed=1234)
            try_update(a_out, f"coorddesc_final_n{n_best}")

    # === Phase 8: Final SA ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final SA (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.30, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = simulated_annealing(a_in, t8, verbose=True, seed=4321)
            try_update(a_out, f"sa_final_n{n_best}")

    # === Phase 9: One last SGD with aggressive params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final SGD (n={n_best}) ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_subgradient(a_in, t9, verbose=False,
                                            top_k=40, beta=2.0, c=5.0,
                                            exact_frac=0.6, restart_every=5000,
                                            seed=7777)
            try_update(a_out, f"sgd_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
