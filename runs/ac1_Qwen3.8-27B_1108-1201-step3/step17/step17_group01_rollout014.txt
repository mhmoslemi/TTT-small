2. Adaptive temperature (beta) and step size scheduling

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient_fft(a, p):
    """Compute sum_k p[k] * (a[k-i] + a[i-k]) efficiently via FFT."""
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    P = np.zeros(L)
    P[:2 * n - 1] = p
    A = np.zeros(L)
    A[:n] = a
    P_fft = np.fft.rfft(P)
    A_fft = np.fft.rfft(A)
    g = np.fft.irfft(P_fft * 2.0 * np.real(A_fft), L)[:n]
    return g


def compute_gradient_direct(a, p, top_k):
    """Direct computation of top-k weighted gradient (for verification or small k)."""
    n = len(a)
    k_max = min(top_k, len(p))
    top_idx = np.argsort(p)[::-1][:k_max]
    grad = np.zeros(n)
    for k_int in top_idx:
        wi = p[k_int]
        if wi < 1e-30:
            continue
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

        max_v = vals.max()
        exp_v = np.exp(beta * (vals - max_v))
        w = exp_v / exp_v.sum()
        p = np.zeros(len(c))
        p[top_idx] = w
    else:
        max_c = c.max()
        exp_c = np.exp(beta * (c - max_c))
        p = exp_c / exp_c.sum()
    return p


def freq_project(a, bandwidth, strength):
    """Apply soft frequency-domain band-limiting to reduce high-frequency content."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    freqs = np.arange(len(A))
    # Create a smooth cutoff
    cutoff = int(bandwidth * n / 2)
    transition = max(1, int(0.5 * bandwidth * n / 2))
    H = np.ones(len(A))
    H[cutoff:] = 0.0
    if cutoff > transition:
        H[cutoff - transition:cutoff] = np.linspace(1.0, 0.0, transition)
    else:
        H[cutoff:] = 0.0
    # Soft projection: blend toward band-limited version
    A_blended = (1.0 - strength) * A + strength * A * H
    a_new = np.fft.irfft(A_blended, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def fft_amd_optimize(a0, time_limit, beta=50.0, c0=4.0, top_k=None,
                     restart_every=5000, verbose=False, rng_seed=42,
                     use_freq_proj=False, freq_bandwidth=3.0, freq_strength=0.1):
    """FFT-based soft-max accelerated mirror descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall_count = 0

    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta
    cur_c0 = c0
    restart_count = 0

    elapsed_start = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - elapsed_start > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Compute soft-max weights
        p = soft_max_weights(ac, cur_beta, top_k)

        # Compute gradient via FFT
        g = compute_gradient_fft(y, p)

        # Normalize gradient for step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            break

        # Adaptive step size
        eta = cur_c0 / (np.sqrt(t) + 1.0)
        # Scale gradient
        g = g / g_norm * np.sqrt(n)

        # Mirror descent update (exponentiated gradient on simplex)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Optional frequency projection
        if use_freq_proj and t % 10 == 0:
            a_new = freq_project(a_new, freq_bandwidth, freq_strength)

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.01 * min(restart_count, 10))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            cur_beta = max(20.0, cur_beta * 0.85)
            cur_c0 = c0 * (1.0 + 0.05 * min(restart_count, 8))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FFT-AMD restart {restart_count} t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, c0={cur_c0:.2f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FFT-AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def freq_domain_optimize(a0, time_limit, iters=5000, eta0=0.4, verbose=False):
    """Frequency-domain iterative flattening via soft spectrum shaping."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    L = 1 << (2 * n - 1).bit_length()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, L)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Compress dynamic range of power spectrum
        eta = eta0 * (1.0 + 0.3 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqDom it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def target_level_optimize(a0, time_limit, T_init=None, c0=3.0, verbose=False):
    """Optimize using target-level hinge loss: sum max(0, c[k]-T)^2."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    L = 1 << (3 * n - 2).bit_length()

    if T_init is None:
        ac0 = fast_autocorr(a)
        T = ac0.max()
    else:
        T = T_init

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)

        # Hinge loss: only penalize lags above T
        excess = np.maximum(ac - T, 0.0)
        if excess.max() < 1e-12:
            # All below target, decrease T
            T *= 0.999
            continue

        # Gradient of sum(excess^2) = 2 * excess * dc/da
        # Weight vector for gradient
        w = 2.0 * excess
        w_sum = w.sum()
        if w_sum < 1e-30:
            continue
        p = w / w_sum

        g = compute_gradient_fft(y, p)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            break

        eta = c0 / (np.sqrt(t) + 1.0)
        g = g / g_norm * np.sqrt(n)

        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()

        # Adapt T: track the 95th percentile of the convolution
        if t % 100 == 0:
            ac_cur = fast_autocorr(x)
            T_new = np.percentile(ac_cur, 95)
            T = 0.9 * T + 0.1 * T_new

    return best_a, best_m


def make_sinc_like(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_multihump(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(a)


def make_power_law(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_cosine(n_base, mode=1)
