import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    conv = fast_conv(a_n)
    return 2.0 * n * conv.max()

def compute_pnorm_grad(a, p, conv=None):
    """
    Compute gradient of f_p(a) = (sum_k C_k^p)^(1/p) where C = conv(a,a).
    Returns grad such that grad[i] = df_p / da[i].
    """
    n = len(a)
    if conv is None:
        conv = fast_conv(a)
    
    cmax = conv.max()
    if cmax < 1e-30:
        return np.zeros(n)
    
    # Normalize to [0, 1]
    cn = conv / cmax
    # Weights w_k = cn_k^(p-1)
    w = cn ** (p - 1)
    
    # Scale factor S = (sum cn_k^p)^(1/p - 1)
    S = (np.sum(cn ** p)) ** (1.0/p - 1.0)
    
    # Compute G[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
    # Using convolution: G[i] = conv(W_rev, a)[2n-2-i]
    W_rev = w[::-1]  # reverse w (length 2n-1)
    fft_size = 1 << (3 * n - 2).bit_length()
    WR = np.fft.rfft(W_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(WR * A, fft_size)
    
    # G[i] = conv_result[2n-2-i] for i = 0, ..., n-1
    G = conv_result[n-1:2*n-1][::-1]
    
    grad = S * G
    return grad

def mirror_descent_pnorm(a0, time_limit, p_start=4, p_end=100, verbose=False, seed=42):
    """
    Nesterov-accelerated mirror descent (KL) with p-norm objective.
    p increases linearly from p_start to p_end over the course of optimization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()  # Nesterov momentum variable
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    max_stall = 5000
    c = 3.0  # step size parameter
    
    total_iters = 100000
    
    for t in range(1, total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive p: increase over time
        progress = min(t / total_iters, 1.0)
        p = p_start + (p_end - p_start) * (progress ** 0.7)
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        conv = fast_conv(y)
        g = compute_pnorm_grad(y, p, conv)
        
        # Step size
        eta = c / (np.sqrt(t) + 1.0)
        
        # Mirror step (KL projection)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best (only evaluate true J every few iterations for speed)
        if t % 50 == 0 or t < 10:
            val = eval_J(a_new)
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        
        # Restart
        if stall > max_stall:
            stall = 0
            restart_count += 1
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            c = max(c * 0.95, 0.5)
            if verbose:
                print(f"    Restart {restart_count} at t={t}: J={best_val:.6f}, p={p:.1f}")
    
    return best_a, best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. p-norm smoothed objective with Nesterov-accelerated mirror descent (KL projection)
2. Adaptive p (smooth → sharp) for progressively better max approximation
3. Pressure-based coordinate descent for local polishing
4. Multi-strategy initialization and restarts
5. Length adaptation and final refinement
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    conv = fast_conv(a_n)
    return 2.0 * n * conv.max()


def compute_pnorm_grad(a, p, conv=None):
    """
    Compute gradient of f_p(a) = (sum_k C_k^p)^(1/p) where C = conv(a,a).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    if conv is None:
        conv = fast_conv(a)

    cmax = conv.max()
    if cmax < 1e-30:
        return np.zeros(n)

    cn = np.maximum(conv / cmax, 0.0)
    # Weights for the p-norm gradient
    exponent = p - 1.0
    w = cn ** exponent

    # Scale factor S = (sum cn_k^p)^(1/p - 1)
    sum_cn_p = np.sum(cn ** p)
    if sum_cn_p < 1e-30:
        return np.zeros(n)
    S = sum_cn_p ** (1.0 / p - 1.0)

    # Compute G[i] = sum_k w[k] * a[k-i] for i = 0,...,n-1
    # via FFT-based convolution: G[i] = conv(W_rev, a)[2n-2-i]
    W_rev = w[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    WR = np.fft.rfft(W_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(WR * A, fft_size)

    G = conv_result[n - 1:2 * n - 1][::-1].copy()

    grad = S * G
    return grad


def mirror_descent_pnorm(a0, time_limit, p_start=4.0, p_end=80.0,
                         c0=3.0, max_stall=4000, verbose=False, seed=42):

    Nesterov-accelerated mirror descent with KL projection and adaptive p-norm.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0
    c = c0
    max_iters = 5000000

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive p: increase with progress
        progress = min(t / 50000.0, 1.0)
        p = p_start + (p_end - p_start) * (progress ** 0.6)

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_conv(y)
        g = compute_pnorm_grad(y, p, conv)

        g_norm = np.sqrt(np.maximum(np.sum(y * g * g), 1e-30))
        if g_norm > 1.0:
            g = g / g_norm

        # Step size
        eta = c / (np.sqrt(t) + 1.0)

        # Mirror step (KL projection onto simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best (evaluate true J periodically)
        if t % 100 == 0 or t < 20:
            val = eval_J(a_new)
            if val < best_val - 1e-15:
                best_val = val
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1

        # Restart with perturbation
        if stall > max_stall:
            stall = 0
            restart_count += 1
            scale = rng.uniform(0.005, 0.04) * (1.0 + 0.5 * restart_count)
            scale = min(scale, 0.15)
            pert = rng.normal(0, scale, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            c = max(c * 0.92, 0.5)
            if verbose and restart_count % 5 == 0:
                print(f"      MD restart {restart_count} t={t}: J={best_val:.6f} p={p:.1f} c={c:.2f}")

    # Final evaluation
    final_val = eval_J(x)
    if final_val < best_val:
        best_val = final_val
        best_a = x.copy()

    return best_a, best_val


def pressure_coord_descent(a0, time_limit, verbose=False, seed=42):
    """
    Coordinate descent using pressure from tight autocorrelation constraints.
    Transfers mass from high-pressure to low-pressure positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    num_top = min(30, 2 * n - 1)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Get top lags
        top_lags = np.argsort(conv)[::-1][:num_top]

        # Compute pressure at each position
        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        # Source: high pressure * mass
        valid_src = (a > 1e-10)
        src_scores = np.where(valid_src, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if src_scores[src] < 1e-20:
            stall += 1
            break

        # Destination: low pressure, low near-peak contribution
        dest_penalty = np.zeros(n)
        for k in top_lags[:8]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_penalty[idx] += a[k_i - idx]
        dest_penalty[src] = np.inf
        dest_penalty += 0.05 * pressure

        dst = int(np.argmin(dest_penalty))

        # Try transfer fractions
        improved = False
        for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-14:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 20 == 0 and stall < max_stall:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

    return best_a, best_J


def fine_coord_search(a0, time_limit, verbose=False, seed=77):
    """
    Fine-grained coordinate search: try small perturbations at many positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Sort positions by value to focus on significant ones
    order = np.argsort(a)[::-1]
    significant = order[:min(n, 500)]

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        M = conv.max()
        # Find top lags
        top_idxs = np.argsort(conv)[::-1][:20]

        improved = False
        # Try moving mass from high-value positions to random low-value positions
        candidates = list(significant[:50])
        rng.shuffle(candidates)

        for src in candidates:
            if a[src] < 1e-8:
                continue
            # Try several destinations
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    if a_t[src] < -1e-14:
                        break
                    a_t = np.maximum(a_t, 0.0)
                    a_t /= a_t.sum()
                    J_t = eval_J(a_t)
                    if J_t < best_J - 1e-14:
                        a = a_t
                        best_J = J_t
                        best_a = a.copy()
                        improved = True
                        break

            if time.monotonic() - t0 > time_limit:
                break

        if not improved and time.monotonic() - t0 > time_limit * 0.3:
            break

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=60.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.01
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 8):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_flat_target(n):
    """
    Initialize to approximate flat autocorrelation.
    Uses iterative spectral flattening (few steps).
    """
    a = np.ones(n) / n
    for _ in range(50):
        conv = fast_conv(a)
        # Target: all conv values equal to 1/(2n-1)
        target = 1.0 / (2 * n - 1)
        # Compute "excess" at each lag
        excess = conv - target
        # Gradient of ||excess||^2 w.r.t. a
        # d(excess_k)/da_i = a[k-i]
        # grad_i = 2 * sum_k excess_k * a[k-i]
        W = excess  # length 2n-1
        W_rev = W[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        WR = np.fft.rfft(W_rev, fft_size)
        A = np.fft.rfft(a, fft_size)
        conv_result = np.fft.irfft(WR * A, fft_size)
        G = conv_result[n - 1:2 * n - 1][::-1].copy()
        # Gradient descent step
        a -= 0.01 * G
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """
    Main optimization driver.
    Uses p-norm mirror descent + coordinate descent for local refinement.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent loaded: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial (no parent): n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"Starting optimization: n={n0}, J={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Multi-initialization p-norm mirror descent ===
    print(f"\n=== Phase 1: PN-MD Multi-Init (n={n0}) ===")

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.5))
    labels.append("ramp05")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp10")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp15")
    inits.append(make_power_ramp(n0, 2.0))
    labels.append("ramp20")
    inits.append(make_bimodal(n0, 50.0))
    labels.append("bimodal50")
    inits.append(make_bimodal(n0, 100.0))
    labels.append("bimodal100")
    inits.append(make_smooth_random(n0, seed=7))
    labels.append("smooth7")
    inits.append(make_smooth_random(n0, seed=123))
    labels.append("smooth123")
    inits.append(make_flat_target(n0))
    labels.append("flat")

    if parent is not None:
        blend = 0.5 * parent + 0.5 * make_power_ramp(n0, 1.0)
        inits.append(blend / blend.sum())
        labels.append("blend")

    # Deduplicate
    seen_sigs = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 50)], 4))
        if sig not in seen_sigs:
            seen_sigs.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    remaining = time_budget - (time.monotonic() - t_global)
    t_per = max(10.0, remaining * 0.25 / len(inits))

    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 60:
            break
        t_avail = min(t_per, time_budget - 60 - elapsed)
        if t_avail < 8:
            break
        print(f"  PN-MD [{lbl}] ({t_avail:.0f}s)...", end=" ")
        a_out, v_out = mirror_descent_pnorm(init, t_avail, p_start=4, p_end=60,
                                            c0=3.0, verbose=False, seed=100 + idx)
        print(f"J={v_out:.6f}")
        try_update(a_out, f"pn_{lbl}")

    # === Phase 2: Extended p-norm mirror descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended PN-MD (n={len(best_a)}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 20:
        a_out, v_out = mirror_descent_pnorm(best_a.copy(), t2, p_start=6, p_end=100,
                                            c0=2.5, max_stall=5000,
                                            verbose=True, seed=200)
        try_update(a_out, "pn_ext")

    # === Phase 3: Pressure coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Pressure CD (n={len(best_a)}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_out, v_out = pressure_coord_descent(best_a.copy(), t3, verbose=False, seed=314)
        try_update(a_out, "press_cd")

    # === Phase 4: PN-MD after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: PN-MD Polish (n={len(best_a)}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_out, v_out = mirror_descent_pnorm(best_a.copy(), t4, p_start=20, p_end=150,
                                            c0=2.0, max_stall=3000,
                                            verbose=True, seed=400)
        try_update(a_out, "pn_polish")

    # === Phase 5: Fine coordinate search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Fine Coord (n={len(best_a)}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_out, v_out = fine_coord_search(best_a.copy(), t5, verbose=False, seed=55)
        try_update(a_out, "fine_coord")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-300, -150, 150, 300, 600, -600]:
        nl = n_best + d
        if 1500 <= nl <= 6000:
            lengths_to_try.append(nl)
    for nl in [2000, 2500, 3000, 3500, 4000, 4500, 5000]:
        if nl not in lengths_to_try and 1500 <= nl <= 6000:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length n={nl}...", end=" ")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 10:
            a_out, v_out = mirror_descent_pnorm(a_int, t_md, p_start=6, p_end=80,
                                                c0=2.5, verbose=False, seed=nl % 1000)
            print(f"J={v_out:.6f}")
            try_update(a_out, f"len{nl}")

    # === Phase 7: Final deep optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 7: Final Deep Opt (n={n_cur}) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 15:
        a_out, v_out = mirror_descent_pnorm(best_a.copy(), t7, p_start=30, p_end=200,
                                            c0=1.5, max_stall=6000,
                                            verbose=True, seed=999)
        try_update(a_out, "final_deep")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 8: Final CD Polish ===")
        t8 = min(remaining * 0.30, 150.0)
        a_out, v_out = pressure_coord_descent(best_a.copy(), t8, verbose=False, seed=777)
        try_update(a_out, "final_cd")

    # === Phase 9: Last chance MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 9: Last MD ===")
        a_out, v_out = mirror_descent_pnorm(best_a.copy(), remaining - 5,
                                            p_start=50, p_end=300,
                                            c0=1.0, max_stall=8000,
                                            verbose=False, seed=313)
        try_update(a_out, "last_md")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(max(0.0, min(1000.0, x))) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
