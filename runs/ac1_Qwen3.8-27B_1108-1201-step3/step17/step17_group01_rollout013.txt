def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            a = a_new
            continue
        
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            # Increase damping if improving
            damping = min(1.0, damping * 1.05)
        else:
            # Reduce damping and try a smaller step
            damping = max(0.1, damping * 0.8)
            a_new2 = a + 0.1 * (a_new - a)
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-15:
                a_new2 = a_new2 / s2
                M2 = fast_autocorr(a_new2).max()
                if M2 < best_m:
                    best_m = M2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    a = a_new2  # Accept anyway for exploration
            # Reset damping
            damping = max(0.5, damping)
        
        # Anneal damping
        if it > 1000:
            damping = max(0.3, damping * 0.999)
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, 
                 restart_every=5000, verbose=False, seed=42):
    """Adaptive mirror descent with multi-peak subgradient and Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        # Multi-peak subgradient
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            
            # Adaptive restart: use combination of best and random
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            # Adapt hyperparameters
            cur_top_k = min(cur_top_k + 5, 100)
            cur_beta = max(cur_beta - 0.2, 2.0)
            gamma = c0 * (1.0 + 0.05 * min(restart_count, 10))
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart {restart_count} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    
    return best_a, best_m


def simulated_annealing_optimize(a0, time_limit, seed=42, verbose=False):
    """
    Simulated annealing in log-mirror coordinates on the simplex.
    Uses Metropolis-Hastings with adaptive temperature.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Log-parameterization: a_i = exp(theta_i) / sum(exp(theta))
    a_safe = np.maximum(a, 1e-15)
    theta = np.log(a_safe / a_safe.sum())
    
    # Initial temperature
    # Estimate typical J scale
    J0 = 2.0 * n * best_m
    T = J0 * 0.01  # Start with 1% of J as temperature scale
    T_min = J0 * 0.0001
    T_decay = 0.9999
    
    # Step size
    sigma = 0.05
    sigma_min = 0.001
    sigma_decay = 0.99999
    
    accept_count = 0
    total_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Propose: Gaussian perturbation in theta space
        dtheta = sigma * rng.standard_normal(n)
        theta_new = theta + dtheta
        
        # Convert back to simplex
        theta_shifted = theta_new - theta_new.max()
        a_new = np.exp(theta_shifted)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        # Evaluate
        M_new = fast_autocorr(a_new).max()
        J_new = 2.0 * n * M_new
        J_cur = 2.0 * n * fast_autocorr(a).max()  # This is expensive, compute once
        
        # Actually, let's avoid recomputing J_cur every time
        # Track it separately
        ...
        
        total_count += 1
        if J_new < J_cur or rng.random() < math.exp(-(J_new - J_cur) / T):
            theta = theta_new
            a = a_new
            accept_count += 1
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Anneal
        T = max(T_min, T * T_decay)
        sigma = max(sigma_min, sigma * sigma_decay)
        
        if verbose and total_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={total_count} ({elapsed:.0f}s): J_best={2.0*n*best_m:.6f}, T={T:.2e}, sigma={sigma:.4f}, acc={accept_count/total_count:.3f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=200):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp
    
    y = a.copy()
    x = a.copy()
    
    T_full = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_full[j:j + n, j] = y  # This will be updated each iteration
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update T_full
        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = y
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
