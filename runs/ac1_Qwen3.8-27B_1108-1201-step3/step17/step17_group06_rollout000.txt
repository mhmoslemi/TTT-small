import numpy as np
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """J = 2*n*max(conv)/S^2, assuming a normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def soft_gradient(a, tau):
    """
    Compute the gradient of 2n * tau * logsumexp(conv/tau) w.r.t. a.
    Returns (objective_value, gradient).
    """
    n = len(a)
    conv = fast_conv(a)  # length 2n-1
    
    # Soft-max weights
    scaled = conv / tau
    s_max = scaled.max()
    exp_scaled = np.exp(scaled - s_max)
    softmax = exp_scaled / exp_scaled.sum()
    
    # Objective: 2n * (conv.max() + tau * log(sum(exp((conv - conv.max())/tau))))
    # = 2n * tau * logsumexp(conv/tau)
    obj = 2.0 * n * tau * (s_max + np.log(exp_scaled.sum()))
    
    # Gradient: d/da_j [tau * logsumexp(conv/tau)] = sum_k softmax_k * d(conv_k)/da_j
    # d(conv_k)/da_j = 2 * a[k-j] if 0 <= k-j < n, else 0
    # So grad_j = 2 * sum_{k: 0<=k-j<n} softmax_k * a[k-j]
    # Let m = k-j: grad_j = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} softmax_{m+j} * a[m]
    # This is 2 * (a cross-correlate softmax)[j] for j = 0..n-1
    
    # Using FFT: cross-correlation of a and softmax
    # (a ⋆ s)[j] = sum_m a[m] * s[m+j]
    # = ifft(fft(a) * conj(fft(s)))
    # Need FFT size >= n + (2n-1) - 1 = 3n-1
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    S = np.fft.rfft(softmax, fft_size)
    corr = np.fft.irfft(A * np.conj(S), fft_size)
    
    # The cross-correlation (a ⋆ s)[j] = sum_m a[m]*s[m+j] is at index j in corr
    # (for j = 0, 1, ..., 2n-2)
    grad = 2.0 * corr[:n]
    
    return obj, grad


def adam_step(params, grads, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. Returns updated params and moments."""
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, m, v


def optimize_soft(a0, time_limit, tau_init=0.01, tau_final=0.001,

def interpolate_seq(a, new_n):
    old_n = len(a)

    old_x = np.linspace(0.0, 1.0, old_n)

    if s < 1e-15:
