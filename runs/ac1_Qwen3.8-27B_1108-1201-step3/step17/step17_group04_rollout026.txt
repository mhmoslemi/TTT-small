def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _norm_sum1(a):

def _subgradient_exact(ac, n):
    k = int(np.argmax(ac))
    grad = np.zeros(n)

def _mirror_step(y, g, eta):
    log_a = np.log(y + 1e-30) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def _nesterov_mirror(a0, time_limit, seed=42, verbose=False):
    """Adaptive Nesterov mirror descent with multi-scale subgradients."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_J = _compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000

    # Adaptive parameters
    top_k = 10
    beta_exp = 4.0
    c_scale = 2.0
    exact_prob = 0.5

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = _fast_autocorr(y)
        M = ac.max()
        k_peak = int(np.argmax(ac))

        # Multi-scale subgradient: mix of exact peak and soft top-K
        g = np.zeros(n)
        r = rng.random()
        if r < exact_prob:
            # Exact: target the single peak
            i_start = max(0, k_peak - n + 1)
            i_end = min(n, k_peak + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_peak - idx
            g[idx] = 2.0 * y[j_idx]
        else:
            # Soft: target top-K peaks with exponential weights
            K = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:K]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** beta_exp
            w /= w.sum()
            for wi, k_i in zip(w, top_idx):
                k_int = int(k_i)
                i_s = max(0, k_int - n + 1)
                i_e = min(n, k_int + 1)
                idx = np.arange(i_s, i_e)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * y[j_idx]

        # Adaptive step size
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            g_norm_sq = 1e-20
        # Polyak-like: use gap between current M and theoretical lower bound
        lower = 1.0 / (2 * n - 1)  # ideal flat autocorrelation value
        gap = max(M - lower, 1e-10)
        eta = c_scale * gap / g_norm_sq
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        a_new = _mirror_step(y, g, eta)
        x_prev = x
        x = a_new

        J_new = _compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptation on stall
        if stall == max_stall:
            # Perturb and continue
            sigma = 0.005 + 0.005 * (stall // max_stall)
            pert = rng.normal(0, sigma, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
            # Adapt parameters
            top_k = min(top_k + 5, 50)
            beta_exp = max(beta_exp - 0.5, 1.5)
            c_scale = min(c_scale + 0.3, 6.0)
            exact_prob = min(exact_prob + 0.05, 0.8)

        if verbose and t % 5000 == 0:
            print(f"      NM t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def _simulated_annealing(a0, time_limit, seed=99, verbose=False):
    """
    Simulated annealing with targeted mass-transfer moves.
    Uses efficient peak-affected region checking.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_J = _compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initial temperature based on typical move cost
    ac0 = _fast_autocorr(a)
    M0 = ac0.max()
    T = M0 * 0.1  # temperature (in units of peak value)
    T_min = M0 * 0.001
    cool_rate = 0.9995

    stall = 0
    max_stall = 2000

    # Precompute peak lags for targeting
    n_peaks_to_check = 20

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1

        # Compute current autocorrelation (cached between some steps)
        if step % 5 == 1 or step == 1:
            ac = _fast_autocorr(a)
            M_curr = ac.max()
            # Get top peak positions
            top_lags = np.argsort(ac)[::-1][:n_peaks_to_check]

        # Choose move type
        r = rng.random()
        if r < 0.5:
            # Targeted move: from high-contribution to low-contribution position
            k_peak = int(top_lags[0])
            i_start = max(0, k_peak - n + 1)
            i_end = min(n, k_peak + 1)
            # Pick source from peak contributors (high a value)
            contrib_idx = np.arange(i_start, i_end)
            contrib_partner = k_peak - contrib_idx
            # Score: a[i] * a[k-i] (contribution to this peak)
            scores = a[contrib_idx] * a[contrib_partner]
            # Sample source proportional to score
            total = scores.sum()
            if total < 1e-15:
                continue
            probs = scores / total
            src = int(np.random.choice(contrib_idx, p=probs))
            # Pick destination from low-autocorrelation-contribution positions
            # Use a different lag's contributions as "safe" positions
            k_safe = int(top_lags[min(len(top_lags)-1, len(top_lags)//2)])
            j_start = max(0, k_safe - n + 1)
            j_end = min(n, k_safe + 1)
            safe_idx = np.arange(j_start, j_end)
            # Prefer positions with low a value (less likely to create new peaks)
            dst_weights = np.ones(len(safe_idx)) * (1.0 + a[safe_idx] * 0.1)
            dst_weights /= dst_weights.sum()
            dst = int(np.random.choice(safe_idx, p=dst_weights))
        elif r < 0.8:
            # Random swap of mass
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst:
                continue
        else:
            # Block swap: swap two random blocks
            block_len = int(rng.integers(2, max(3, n // 100)))
            start1 = int(rng.integers(max(1, n - block_len)))
            start2 = int(rng.integers(max(1, n - block_len)))
            if abs(start1 - start2) <= block_len:
                continue
            b1 = a[start1:start1+block_len].copy()
            b2 = a[start2:start2+block_len].copy()
            a[start1:start1+block_len] = b2
            a[start2:start2+block_len] = b1
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            J = _compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                stall = 0
            else:
                # Accept with probability
                if rng.random() < np.exp(-(J - best_J) / (T * 0.01)):
                    pass
                else:
                    # Revert
                    a[start1:start1+block_len] = b1
                    a[start2:start2+block_len] = b2
                    stall += 1
            continue

        if r < 0.8 and src == dst:
            continue

        # Determine transfer amount
        src_val = a[src]
        if src_val < 1e-10:
            continue
        frac = rng.uniform(0.01, 0.15)
        amt = src_val * frac

        # Execute move
        a[src] -= amt
        a[dst] += amt
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # Evaluate
        J = _compute_J(a)
        delta = J - best_J
        if delta < 0:
            best_J = J
            best_a = a.copy()
            stall = 0
        elif rng.random() < np.exp(-delta / (T * 0.01 + 1e-20)):
            stall += 1
        else:
            # Revert
            a[src] += amt
            a[dst] -= amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall += 1

        # Cool down
        T = max(T * cool_rate, T_min)

        # Restart if stalled too long
        if stall > max_stall:
            stall = 0
            # Perturb from best
            n_pert = max(10, n // 50)
            positions = rng.choice(n, size=n_pert, replace=False)
            for p in positions:
                amt_p = a[p] * rng.uniform(0.05, 0.3)
                dst_p = int(rng.integers(n))
                a[p] -= amt_p
                a[dst_p] += amt_p
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            T = M0 * 0.05  # Reheat

        if verbose and step % 10000 == 0:
            print(f"      SA step={step} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, T={T:.2e}")

    return best_a, best_J


def _lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:

        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select binding constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix
        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for j in range(i_start, i_end):
                T_sel[ii, j] = y[k_int - j]

        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)),
                         [T_sel @ b_var <= M_curr])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        y_new = np.maximum(y_new, 0.0)

def _interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

def _peak_reduction(a0, time_limit, seed=55, verbose=False):

    best_score = float('inf')

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = _norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = _make_init_centered(n_base)
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    # === Strategy: Try multiple lengths, focus optimization on best ===
    # Start with parent length, then try longer
    n_current = len(best_seq)

    # Phase 1: Quick SGD on current best
    print(f"\n=== Phase 1: Nesterov Mirror (n={n_current}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = _nesterov_mirror(best_seq, t1, seed=42, verbose=False)
        _try_update(a_out, f"nm_n{n_current}")

    # Phase 2: Simulated annealing
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Simulated Annealing (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _simulated_annealing(a_in, t2, seed=99, verbose=False)
        _try_update(a_out, f"sa_n{len(a_in)}")

    # Phase 3: Peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Reduction (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _peak_reduction(a_in, t3, seed=55, verbose=False)
        _try_update(a_out, f"pr_n{len(a_in)}")

    # Phase 4: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_current + 2000, 8000)
    print(f"\n=== Phase 4: Longer sequence (n={n_long}) ===")
    if remaining > 200:
        a_long = _interpolate(best_seq, n_long)
        # Quick optimization on longer
        t4a = min(remaining * 0.25, 250.0)
        if t4a > 30:
            a_out, _ = _nesterov_mirror(a_long, t4a, seed=100, verbose=False)
            _try_update(a_out, f"nm_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            t4b = min(remaining * 0.3, 200.0)
            if t4b > 20:
                a_in = _norm_sum1(best_seq)
                a_out, _ = _simulated_annealing(a_in, t4b, seed=200, verbose=False)
                _try_update(a_out, f"sa_n{len(a_in)}")

    # Phase 5: LP refinement on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP Frank-Wolfe (n={n_best}) ===")
    if remaining > 60 and n_best <= 5000:
        t5 = min(remaining * 0.20, 200.0)
        if t5 > 20:
            a_in = _norm_sum1(best_seq)
            a_out, _ = _lp_frank_wolfe(a_in, t5, solver, verbose=False, top_K=200)
            _try_update(a_out, f"lpfw_n{n_best}")

    # Phase 6: Second round SGD on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SGD Round 2 (n={n_best}) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _nesterov_mirror(a_in, t6, seed=777, verbose=False)
        _try_update(a_out, f"nm2_n{n_best}")

    # Phase 7: SA on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: SA Round 2 (n={n_best}) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _simulated_annealing(a_in, t7, seed=888, verbose=False)
        _try_update(a_out, f"sa2_n{n_best}")

    # Phase 8: Try even longer if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        n_long2 = min(len(best_seq) + 2000, 10000)
        if n_long2 > len(best_seq):
            print(f"\n=== Phase 8: Longer seq n={n_long2} ===")
            a_long2 = _interpolate(best_seq, n_long2)
            t8 = min(remaining * 0.3, 300.0)
            if t8 > 30:
                a_out, _ = _nesterov_mirror(a_long2, t8, seed=333, verbose=False)
                _try_update(a_out, f"nm_n{n_long2}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 50:
                a_in = _norm_sum1(best_seq)
                a_out, _ = _peak_reduction(a_in, min(remaining*0.5, 100.0), seed=444, verbose=False)
                _try_update(a_out, f"pr2_n{len(a_in)}")

    # Phase 9: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 15:
            a_in = _norm_sum1(best_seq)
            # Use higher top_k and more exact for final polish
            a_out, _ = _nesterov_mirror(a_in, t9, seed=5555, verbose=False)
            _try_update(a_out, f"polish_n{n_best}")

    # Phase 10: One more SA burst
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t10 = min(remaining * 0.5, 100.0)
        if t10 > 15:
            a_in = _norm_sum1(best_seq)
            a_out, _ = _simulated_annealing(a_in, t10, seed=6666, verbose=False)
            _try_update(a_out, f"sa_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = _make_init_centered(n_fb)
        best_score = _eval_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
