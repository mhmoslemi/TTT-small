def build_conv_matrix(f, n):
    """Build the convolution matrix B where B[k,i] = f[k-i] if 0<=k-i<n else 0"""
    # For each k, the non-zero positions are i in [max(0,k-n+1), min(n-1,k)]
    # B[k,i] = f[k-i]
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        rows.extend([k] * count)
        cols.extend(range(i_start, i_end + 1))
        vals.extend(f[k - i_start : k - i_start + count])
    B = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return B

    else:
        w = w / w_sum

    a_rev = a[::-1]  # a_rev[i] = a[n-1-i]
    for wi, k in zip(w, top_idx):

        # a[k-i] for i in range(n) gives a[k], a[k-1], ..., a[k-n+1]
        # Valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:

            # = a[k_int - min(n-1, k_int)], ..., a[k_int - max(0, k_int-n+1)]
            j_start = k_int - i_end
            j_end = k_int - i_start
            g[i_start:i_end+1] += 2.0 * wi * a[j_start:j_end+1]
    return g

def mirror_descent_lse(a0, time_limit, beta_init=100.0, top_k_init=50, verbose=True):

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    beta = beta_init
    top_k = top_k_init
    c0 = 3.0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        conv = fast_conv(y)
        g = lse_subgradient(y, conv, beta, top_k)

        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror step (multiplicative weights)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_J - 1e-14:
            best_J = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive parameters
        if stall > 5000:
            stall = 0
            beta = max(beta * 0.8, 5.0)

            if verbose:

def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fast_conv(a)
        M = conv.max()
        top_lags = np.argsort(conv)[::-1][:20]
        
        # Compute pressure at each position
        pressure = np.zeros(n)
        for k in top_lags[:10]:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[idx] * a[k_i - idx]
        
        # Find source: position with highest pressure * a
        src_scores = pressure * a
        src = int(np.argmax(src_scores))
        if a[src] < 1e-14:
            stall += 1
            continue
        
        # Find destination: position that minimizes impact
        dest_penalty = np.zeros(n)
        for k in top_lags[:5]:
            k_i = int(k)
            for i in range(n):
                j = k_i - i
                if 0 <= j < n:
                    dest_penalty[i] += a[j]
        dest_penalty[src] = np.inf
        dst = int(np.argmin(dest_penalty))
        
        # Try transfers
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            a_t = a_t / a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.01)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
    
    return best_a, best_J

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (primal: max sum(g) s.t. B@g <= M, g>=0)
2. Log-sum-exp smoothed multiplicative weights mirror descent
3. Pressure-based coordinate descent with adaptive transfers
4. Multi-phase optimization with length exploration and restarts
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (S * S)


def build_conv_matrix(f, n, k_indices=None):
    """Build sparse matrix B[k,i] = f[k-i] if 0<=k-i<n else 0.
    If k_indices given, only include those rows."""
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    rows_list = []
    cols_list = []
    for k in k_indices:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        count = i_end - i_start + 1
        if count > 0:
            rows_list.append(np.full(count, len(rows_list) and k_int, dtype=np.int32))
            cols_list.append(np.arange(i_start, i_end + 1, dtype=np.int32))
    if not rows_list:
        return csr_matrix((len(k_indices), n))
    rows_arr = np.concatenate(rows_list)
    cols_arr = np.concatenate(cols_list)
    # Fix: rows should be the actual k values
    rows_actual = []
    for k in k_indices:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        count = i_end - i_start + 1
        if count > 0:
            rows_actual.append(np.full(count, k_int, dtype=np.int32))
    if rows_actual:
        rows_fixed = np.concatenate(rows_actual)
    else:
        rows_fixed = np.array([], dtype=np.int32)
    idx = rows_fixed - cols_arr
    vals = f[idx]
    B = csr_matrix((vals.astype(np.float64), (rows_fixed, cols_arr)), shape=(len(k_indices), n))
    return B


def solve_lp_restricted(f, n, num_lags=200):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for top-k lags, g >= 0."""
    conv = fast_conv(f)
    M = conv.max()
    if M < 1e-30:
        return None
    # Use lags where convolution is in the top num_lags
    k_idx = np.argsort(conv)[::-1][:min(num_lags, 2 * n - 1)]
    B = build_conv_matrix(f, n, k_idx)
    c = -np.ones(n)
    b_ub = np.full(len(k_idx), M)
    try:
        res = linprog(c, A_ub=B, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    if g.sum() < 1e-15:
        return None
    return g


def solve_lp_full(f, n):
    """Solve LP with all 2n-1 constraints."""
    conv = fast_conv(f)
    M = conv.max()
    if M < 1e-30:
        return None
    B = build_conv_matrix(f, n, None)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=B, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    if g.sum() < 1e-15:
        return None
    return g


def lp_optimize(a0, time_limit, use_full=False, num_lags=200, verbose=True):
    """LP-based iterative optimization with line search and restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    a = a / S if S > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(50000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 2:
            break
        # Solve LP
        if use_full and it < 10:
            g = solve_lp_full(a, n)
        else:
            g = solve_lp_restricted(a, n, num_lags)
        if g is None:
            stall += 1
            if stall > 10:
                break
            continue
        g = g / g.sum()
        # Line search
        found = False
        for t in [0.3, 0.15, 0.1, 0.5, 0.2, 0.4, 0.05, 0.25]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-15:
                continue
            a_new = a_new / S_new
            J_new = eval_J(a_new)
            if J_new < best_J - 1e-14:
                a = a_new
                best_J = J_new
                best_a = a_new.copy()
                stall = 0
                found = True
                if verbose and it % 3 == 0:
                    print(f"    LP it={it}: J={best_J:.7f}, t={t}, el={elapsed:.0f}s")
                break
        if not found:
            stall += 1
            if stall == 8:
                rng = np.random.default_rng(777 + it)
                a = a * (1.0 + rng.normal(0, 0.003, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    if verbose:
        print(f"    LP done: J={best_J:.7f}, el={time.monotonic()-t0:.0f}s")
    return best_a, best_J


def lse_subgradient(a, conv, beta, top_k):
    """Compute soft subgradient of max(conv) using log-sum-exp on top-K lags."""
    n = len(a)
    k_max = min(top_k, 2 * n - 1)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = conv[top_idx]
    max_v = top_vals.max()
    log_w = beta * (top_vals - max_v)
    exp_log = np.exp(log_w)
    w_sum = exp_log.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = exp_log / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            j_start = k_int - i_end
            j_end = k_int - i_start + 1
            g[i_start:i_end + 1] += 2.0 * wi * a[j_start:j_end]
    return g


def mirror_descent_lse(a0, time_limit, beta_init=80.0, top_k_init=50, c0_init=3.0, verbose=True):
    """Multiplicative weights mirror descent with log-sum-exp subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    a = a / S if S > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    beta = beta_init
    top_k = top_k_init
    c0 = c0_init

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit - 1:
            break
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x.copy()
        conv = fast_conv(y)
        g = lse_subgradient(y, conv, beta, top_k)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        val = eval_J(a_new)
        if val < best_J - 1e-14:
            best_J = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            stall = 0
            beta = max(beta * 0.75, 5.0)
            top_k = min(top_k + 20, 200)
            c0 = max(c0 * 0.85, 0.5)
            scale = rng.uniform(0.003, 0.015)
            x = best_a * (1.0 + rng.normal(0, scale, n))
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            x_prev = x.copy()
            if verbose:
                print(f"    MD restart t={t}: J={best_J:.7f}, b={beta:.1f}, k={top_k}")
        if verbose and t % 20000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.7f}")
    return best_a, best_J


def coordinate_descent(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent with pressure-based mass transfer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    a = a / S if S > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit - 1 and stall < 400:
        conv = fast_conv(a)
        top_lags = np.argsort(conv)[::-1][:15]
        pressure = np.zeros(n)
        for k in top_lags[:10]:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[idx] * a[k_i - idx]
        src_scores = pressure * a
        src = int(np.argmax(src_scores))
        if a[src] < 1e-14:
            stall += 1
            continue
        dest_penalty = np.zeros(n)
        for k in top_lags[:5]:
            k_i = int(k)
            for i in range(n):
                j = k_i - i
                if 0 <= j < n:
                    dest_penalty[i] += a[j]
        dest_penalty[src] = 1e30
        dst = int(np.argmin(dest_penalty))
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            a_t = a_t / a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall % 40 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.01)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
        if verbose and stall % 100 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={best_J:.7f}, stall={stall}")
    return best_a, best_J


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=60.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.01
    return a / a.sum()


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 8):
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0)
    s = a_i.sum()
    return a_i / s if s > 0 else np.ones(new_n) / new_n


def propose_candidate(budget_s=1800.0):

    parent = None

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy() / p.sum()
            print(f"Parent: n={len(parent)}, J={eval_J(parent):.7f}")
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
        n0 = len(parent)
    else:
        n0 = 3200
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)

    # === Phase 1: LP Optimization (main phase) ===
    remaining = time_budget - (time.monotonic() - t_global)

    if J_out < best_score:

    t4 = remaining * 0.15
    a_out, _ = coordinate_descent(best_a.copy(), t4, seed=123, verbose=True)

        best_score = J_out

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length Exploration (budget={remaining:.0f}s) ===")
    n_best = len(best_a)
    lengths = []
    for d in [-300, -150, 150, 300, 500]:
        nl = n_best + d
        if 1500 <= nl <= 5000:
            lengths.append(nl)
    for nl in sorted(set(lengths)):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Trying n={nl}...")
        a_int = interpolate_seq(best_a, nl)

        if t_lp > 15:

            J_out = eval_J(a_out)
            if J_out < best_score:

    # Final validation
    if best_a is not None:

        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.7f}, n={len(best_a)}, time={elapsed:.0f}s")
