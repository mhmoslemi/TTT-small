def optimize_full_grad(a0, time_limit, T_start=0.1, T_end=0.005, 
                       c0=1.0, verbose=False):
    """Mirror descent with full FFT-based gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_val = eval_J_normalized(a)
    t0 = time.monotonic()
    
    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    corr_fft_size = 1 << (size + n - 2).bit_length()
    
    step = 0
    max_steps = 10000000
    stall = 0
    max_stall = 50000  # Allow many steps before restart
    
    while time.monotonic() - t0 < time_limit and step < max_steps:
        step += 1
        progress = min(step / max_stall, 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute autocorrelation
        A_fft = np.fft.rfft(a, fft_size)
        c = np.fft.irfft(A_fft * A_fft, fft_size)[:size]
        
        # Softmax weights
        c_max = c.max()
        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(size) / size
        else:
            w = w / w_sum
        
        # Full gradient via FFT cross-correlation
        W_fft = np.fft.rfft(w, corr_fft_size)
        A_corr = np.fft.rfft(a, corr_fft_size)
        g_raw = np.fft.irfft(np.conj(W_fft) * A_corr, corr_fft_size)[:n]
        
        # Center for simplex
        g = g_raw - g_raw.mean()
        
        # Step size
        eta = c0 / (np.sqrt(step) + 1.0)
        
        # Mirror descent update
        log_r = np.log(a) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        val = eval_J_normalized(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
            a = a_new
        else:
            stall += 1
            a = a_new  # Still follow the trajectory
        
        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FG step={step} ({elapsed:.1f}s): J={best_val:.6f}, T={T:.5f}")
        
        if stall > max_stall:
            # Restart
            break
    
    return best_a, best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full FFT-based gradient (O(n log n) per step) with softmax-smoothed objective
2. Mirror descent and ADAM variants with adaptive temperature annealing

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def eval_J_norm(a):
    """Compute J assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def compute_grad_full(a, T, fft_size):
    """Compute full gradient of softmax-smoothed max autocorrelation via FFT.
    Returns (gradient, autocorrelation)."""
    n = len(a)
    size = 2 * n - 1

    # Autocorrelation via FFT
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    # Softmax weights
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n), c
    w = np.exp((c - c_max) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        g = np.zeros(n)
        return g, c
    w = w / w_sum

    # Cross-correlation: g[i] = sum_k w[k] * a[k-i]
    W = np.fft.rfft(w, fft_size)
    g = np.fft.irfft(np.conj(W) * A, fft_size)[:n]

    return g, c


def optimize_mirror(a0, time_limit, T_ratio_start=0.3, T_ratio_end=0.02,
                    c0=2.0, momentum=0.0, verbose=False):
    """Mirror descent with full FFT gradient and optional Nesterov momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_val = eval_J_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()

    step = 0
    max_steps = 20000000
    stall = 0
    max_stall = 30000
    a_prev = a.copy()

    while time.monotonic() - t0 < time_limit and step < max_steps:
        step += 1
        progress = min(step / max_stall, 1.0)
        T_ratio = T_ratio_start * (T_ratio_end / T_ratio_start) ** progress

        # Nesterov extrapolation
        if momentum > 0 and step > 1:
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        # Compute gradient
        g, c = compute_grad_full(y, T_ratio * c.max(), fft_size)

        # Center gradient for simplex
        g = g - g.mean()

        # Adaptive step size
        g_norm = np.sqrt(np.sum(g ** 2))
        if g_norm < 1e-30:
            break
        eta = c0 / (np.sqrt(step) + 1.0)
        eta = eta / (g_norm / n + 1e-10)  # Normalize by gradient magnitude

        # Mirror descent step
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Evaluate
        val = eval_J_norm(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a_prev = a
        a = a_new

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={step} ({elapsed:.1f}s): J={best_val:.6f}, "
                  f"T_r={T_ratio:.4f}, eta={eta:.4f}")

        if stall > max_stall:
            # Restart with perturbation
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            a = a_r.copy()
            a_prev = a_r.copy()
            stall = 0
            c0 = max(c0 * 0.8, 0.3)

    return best_a, best_val


def optimize_adam_log(a0, time_limit, T_ratio_start=0.3, T_ratio_end=0.01,
                      lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8,
                      verbose=False):
    """ADAM optimizer in log-space with full FFT gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_val = eval_J_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    max_steps = 10000000
    stall = 0
    max_stall = 20000

    while time.monotonic() - t0 < time_limit and step < max_steps:
        step += 1
        progress = min(step / max_stall, 1.0)
        T_ratio = T_ratio_start * (T_ratio_end / T_ratio_start) ** progress

        # Compute gradient
        g, c = compute_grad_full(a, T_ratio * c.max(), fft_size)
        g_centered = g - g.mean()

        # Log-space gradient: g_theta[i] = a[i] * (g[i] - sum(g*a))
        ga = np.dot(g_centered, a)
        g_theta = a * (g_centered - ga)
        g_theta = np.clip(g_theta, -50, 50)

        # ADAM update
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * g_theta ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        # Update in log-space
        log_a = np.log(a)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Evaluate
        val = eval_J_norm(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ADAM t={step} ({elapsed:.1f}s): J={best_val:.6f}, "
                  f"T_r={T_ratio:.4f}")

        if stall > max_stall:
            scale = rng.uniform(0.01, 0.04)
            pert = rng.normal(0, scale, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            a = a_r.copy()
            m *= 0.5
            v *= 0.5
            stall = 0
            lr = max(lr * 0.85, 0.005)

    return best_a, best_val


def peak_equalization(a0, time_limit, verbose=False):
    """Post-process: try to equalize the top peaks of autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_J = eval_J_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Compute contribution vector for the peak lag
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_star - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-30:
            break

        # Find the index with largest contribution that also has substantial mass
        local_best = np.argmax(contrib * a[idx])
        src_global = i_start + local_best
        amt = a[src_global] * 0.02

        if amt < 1e-15:
            stall += 1
            continue

        # Find destination: minimize impact on top peaks
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:top_k]

        dest_scores = np.zeros(n)
        for k in top_idx:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                j_arr = k_i - np.arange(i_s, i_e + 1)
                dest_scores[np.arange(i_s, i_e + 1)] += a[j_arr]

        dest_scores[src_global] = np.inf
        # Also avoid positions near the peak
        for di in range(-3, 4):
            pos = src_global + di
            if 0 <= pos < n:
                dest_scores[pos] *= 2.0

        dst_global = int(np.argmin(dest_scores))

        # Try the transfer
        a_t = a.copy()
        a_t[src_global] -= amt
        a_t[dst_global] += amt
        a_t = np.maximum(a_t, 0.0)
        a_t = a_t / a_t.sum()
        J_t = eval_J_norm(a_t)

        if J_t < best_J - 1e-15:
            a = a_t
            best_J = J_t
            best_a = a_t.copy()
            stall = 0
        else:
            stall += 1
            # Try random perturbation
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt_r = a[i] * rng.uniform(0.005, 0.03)
                    a_r = a.copy()
                    a_r[i] -= amt_r
                    a_r[j] += amt_r
                    a_r = np.maximum(a_r, 0.0)
                    a_r = a_r / a_r.sum()
                    J_r = eval_J_norm(a_r)
                    if J_r < best_J:
                        a = a_r
                        best_J = J_r
                        best_a = a_r.copy()
                        stall = 0
                        break

        if verbose and step % 50 == 0 if 'step' in dir() else False:
            pass

    return best_a, best_J


def make_power_ramp(n, p=1.0):

def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5

        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_chebyshev(n, degree=5):

def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def smooth_sequence(a, kernel_size=5):

    k = kernel_size // 2
    x = np.arange(-k, k + 1)
    kernel = np.exp(-x ** 2 / (2.0 * (k / 2.0) ** 2))
    kernel = kernel / kernel.sum()
    # Circular padding for boundary
    a_padded = np.concatenate([a[-k:], a, a[:k]])
    a_smooth = np.convolve(a_padded, kernel, mode='valid')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J_norm(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_n = a / s
        J = eval_J_norm(a_n)
        if J < best_score:
            best_score = J
            best_a = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J_norm(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J_norm(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)

    # === Phase 1: Full-gradient mirror descent from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FG-MD from parent (n={n0}, {remaining:.0f}s avail) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 20:
        print(f"  Running for {t1:.0f}s...")
        a_out, _ = optimize_mirror(best_a.copy(), t1, T_ratio_start=0.3,
                                   T_ratio_end=0.02, c0=2.0, momentum=0.0,
                                   verbose=True)
        try_update(a_out, "fg_md_parent")

    # === Phase 2: ADAM log-space from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: ADAM log from parent ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        print(f"  Running for {t2:.0f}s...")
        a_out, _ = optimize_adam_log(best_a.copy(), t2, T_ratio_start=0.3,
                                     T_ratio_end=0.01, lr=0.05, verbose=True)
        try_update(a_out, "adam_parent")

    # === Phase 3: Multi-init mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multi-init FG-MD ===")
    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.7))
    labels.append("ramp07")
    inits.append(make_power_ramp(n0, 1.3))
    labels.append("ramp13")
    inits.append(make_inverse_ramp(n0, 0.8))
    labels.append("inv08")
    inits.append(make_bimodal(n0))
    labels.append("bimodal")
    inits.append(make_smooth_random(n0, seed=7))
    labels.append("smooth7")
    inits.append(make_gaussian_centered(n0, 0.25))
    labels.append("gauss")
    inits.append(make_sinusoidal(n0, 5.0, 0.4))
    labels.append("sin5")

    # Blend with parent
    if parent is not None:
        for blend_p in [0.3, 0.5, 0.7]:
            b = blend_p * make_power_ramp(n0, 1.0) + (1 - blend_p) * parent
            inits.append(b / b.sum())
            labels.append(f"blend{blend_p}")

    t_per = max(15.0, remaining * 0.30 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 120:
            break
        t_avail = min(t_per, time_budget - 120 - elapsed)
        if t_avail < 10:
            break
        print(f"  MD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = optimize_mirror(init, t_avail, T_ratio_start=0.25,
                                   T_ratio_end=0.03, c0=2.5,
                                   momentum=0.3 if idx % 2 == 0 else 0.0)
        try_update(a_out, f"mi_{lbl}")

    # === Phase 4: Extended optimization on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 4: Extended FG-MD (n={n_cur}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = optimize_mirror(best_a.copy(), t4, T_ratio_start=0.15,
                                   T_ratio_end=0.01, c0=1.5, momentum=0.5,
                                   verbose=True)
        try_update(a_out, "ext_md")

    # === Phase 5: ADAM on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: ADAM on best ===")
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 15:
        a_out, _ = optimize_adam_log(best_a.copy(), t5, T_ratio_start=0.15,
                                     T_ratio_end=0.005, lr=0.03)
        try_update(a_out, "adam_best")

    # === Phase 6: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Peak Equalization ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 10:
        a_out, _ = peak_equalization(best_a.copy(), t6, verbose=True)
        try_update(a_out, "peak_eq")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-500, -300, -150, 150, 300, 500, 800]:
        nl = n_best + d
        if 1500 <= nl <= 6000:
            lengths_to_try.append(nl)
    for nl in [2000, 2500, 3000, 3500, 4000, 4500, 5000]:
        if nl not in lengths_to_try and 1500 <= nl <= 6000:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 30:
            continue
        print(f"  Length sweep: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 10:
            a_out, _ = optimize_mirror(a_int, t_md, T_ratio_start=0.2,
                                       T_ratio_end=0.03, c0=2.0)
            try_update(a_out, f"len{nl}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_a)
    print(f"\n=== Phase 8: Final Polish (n={n_final}) ===")
    t8 = min(remaining * 0.30, 300.0)
    if t8 > 20:
        # Mirror descent with very small T (sharp)
        a_out, _ = optimize_mirror(best_a.copy(), t8 * 0.5, T_ratio_start=0.08,
                                   T_ratio_end=0.005, c0=1.0, momentum=0.5,
                                   verbose=True)
        try_update(a_out, "final_md1")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = optimize_adam_log(best_a.copy(), min(remaining * 0.5, 150.0),
                                         T_ratio_start=0.08, T_ratio_end=0.003,
                                         lr=0.02)
            try_update(a_out, "final_adam")

    # === Phase 9: Smoothing + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Smoothing ===")
    if remaining > 30:
        for ks in [3, 5, 7]:
            a_smooth = smooth_sequence(best_a, ks)
            try_update(a_smooth, f"smooth{ks}")
            if remaining > 15:
                a_out, _ = optimize_mirror(a_smooth, min(remaining * 0.3, 60.0),
                                           T_ratio_start=0.1, T_ratio_end=0.02,
                                           c0=1.5)
                try_update(a_out, f"sm{ks}_md")
                remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: One final extended run ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 10: Final Extended ({remaining:.0f}s) ===")
        a_out, _ = optimize_mirror(best_a.copy(), remaining - 10,
                                   T_ratio_start=0.05, T_ratio_end=0.002,
                                   c0=0.8, momentum=0.6, verbose=True)
        try_update(a_out, "final_ext")

    # Final output
    if best_a is not None:
        final_J = eval_J_norm(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J_norm(fb):.6f}")
        return [float(x) for x in fb]
