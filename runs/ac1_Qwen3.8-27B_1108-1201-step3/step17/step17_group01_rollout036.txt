"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed log-sum-exp objective with L-BFGS-B (via scipy)
2. Softmax parameterization for simple constraints
3. Annealed smoothing parameter
4. Multiple diverse initializations and restarts
5. Frequency-domain spectrum flattening
6. Length search and coordinate descent polish
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Fast linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax(q):
    """Stable softmax."""
    q_shifted = q - q.max()
    e = np.exp(q_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(q) / len(q)
    return e / s

def smooth_obj_and_grad(q, beta, n):
    """
    Compute smoothed objective and gradient w.r.t. q.
    p = softmax(q), ac = autocorr(p)
    f = log(sum(exp(beta*ac)))/beta
    df/dq = p * (grad_p - <p, grad_p>)
    where grad_p[j] = 2 * cross_corr(w, p)[j]
    """
    p = softmax(q)
    ac = fast_autocorr(p)
    
    # Compute weights w[k] = exp(beta*ac[k]) / sum(exp(beta*ac))
    beta_ac = beta * ac
    beta_ac_shifted = beta_ac - beta_ac.max()
    w = np.exp(beta_ac_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / w_sum
    
    f = (beta_ac.max() + np.log(w_sum)) / beta  # = log(sum(exp(beta*ac)))/beta
    
    # Compute grad_p[j] = 2 * sum_k w[k] * p[k-j] * 1[0<=k-j<n]
    # This is 2 * cross_corr(w, p)[j] where cross_corr[w,p][j] = sum_k w[k]*p[k-j]
    # Using FFT: cross_corr = irfft(conj(rfft(w)) * rfft(p_padded))
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    p_padded = np.zeros(fft_size)
    p_padded[:n] = p
    
    W = rfft(w_padded)
    P = rfft(p_padded)
    cross = irfft(np.conj(W) * P, fft_size)[:n]
    grad_p = 2.0 * cross
    
    # Gradient w.r.t. q: p * (grad_p - <p, grad_p>)
    pgd = np.dot(p, grad_p)
    grad_q = p * (grad_p - pgd)
    
    return f, grad_q

def true_score(p):
    """Compute the true (unsmoothed) score."""
    n = len(p)
    ac = fast_autocorr(p)
    return 2.0 * n * ac.max()

def lbfgs_optimize(p0, time_limit, beta_start=20.0, beta_end=500.0, n_phases=5, max_iter_per_phase=500, verbose=False):
    """
    Optimize using L-BFGS-B with annealed smoothing.
    p0: initial probability vector (sum=1, non-negative)
    Returns: best p found
    """
    n = len(p0)
    best_p = p0.copy()
    best_score = true_score(p0)
    t0 = time.monotonic()
    
    # Initialize q from p0
    q = np.log(p0 + 1e-30)
    q -= q.mean()
    
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        progress = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        def obj_grad(q_x):
            f, g = smooth_obj_and_grad(q_x, beta, n)
            return f, g
        
        try:
            res = minimize(obj_grad, q, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter_per_phase, 'gtol': 1e-8,
                                  'ftol': 1e-12, 'maxfun': max_iter_per_phase * 5})
            q = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error in phase {phase}: {e}")
            break
        
        p = softmax(q)
        p = norm_sum1(p)
        sc = true_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            if verbose:
                print(f"    Phase {phase}, beta={beta:.1f}: score={sc:.6f}")
    
    return best_p, best_score

def freq_flatten(p0, time_limit, n_iters=500, eta=0.3, verbose=False):
    """
    Frequency-domain spectrum flattening.
    Iteratively modify the spectrum to be flatter, then project back to p>=0, sum=1.
    """
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_score = true_score(p)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(p, fft_size)
        P_mag = np.abs(A)
        P_mag[P_mag < 1e-30] = 1e-30
        
        # Compress the dynamic range
        P_mag_mean = P_mag.mean()
        # Target: flatten
        P_mag_new = P_mag_mean * (P_mag / P_mag_mean) ** (1.0 - eta)
        P_mag_new = np.maximum(P_mag_new, 1e-30)
        
        phase = np.angle(A)
        A_new = P_mag_new * np.exp(1j * phase)
        
        p_new = irfft(A_new, fft_size)[:n]
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-15:
            p_new = p.copy()
        else:
            p_new = p_new / s
        
        # Damped update
        p = (1.0 - 0.5) * p + 0.5 * p_new
        p = norm_sum1(p)
        
        sc = true_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            if verbose and it % 50 == 0:
                print(f"    FreqFlatten it={it}: score={sc:.6f}")
    
    return best_p, best_score

def coordinate_polish(p0, time_limit, verbose=False):
    """
    Coordinate descent: try to reduce max autocorrelation by adjusting pairs of elements.
    """
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_score = true_score(p)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(p)
        max_idx = np.argmax(ac)
        max_val = ac[max_idx]
        
        # Find elements that contribute most to ac[max_idx]
        # ac[k] = sum_i p[i]*p[k-i]
        k = int(max_idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        
        if js > je:
            stall += 1
            continue
        
        idx = np.arange(js, je + 1)
        contrib = p[idx] * p[k - idx]
        
        # Try moving mass from high contributors to reduce the peak
        top_contrib = idx[np.argmax(contrib)]
        if p[top_contrib] < 1e-10:
            stall += 1
            continue
        
        # Find a good destination: minimize increase in other ac values
        # Simple heuristic: move to a random position with low p value
        candidates = np.argsort(p)[:max(10, n // 10)]
        dst = int(candidates[rng.integers(len(candidates))])
        
        if dst == top_contrib:
            stall += 1
            continue
        
        # Try different amounts
        improved = False
        for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            amt = p[top_contrib] * frac
            if amt < 1e-12:
                continue
            p_trial = p.copy()
            p_trial[top_contrib] -= amt
            p_trial[dst] += amt
            p_trial = np.maximum(p_trial, 0.0)
            s_t = p_trial.sum()
            if s_t < 1e-15:
                continue
            p_trial = p_trial / s_t
            sc = true_score(p_trial)
            if sc < best_score - 1e-12:
                p = p_trial
                best_score = sc
                best_p = p.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and p[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = p[i] * frac
                p[i] -= amt
                p[j] += amt
                p = np.maximum(p, 0.0)
                s_t = p.sum()
                if s_t > 1e-15:
                    p = p / s_t
                    sc = true_score(p)
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                        stall = 0
            stall += 1
    
    return best_p, best_score

def make_diverse_init(n, idx):
    """Create diverse initializations."""
    x = np.linspace(0, 1, n)
    
    if idx == 0:
        # Uniform
        return np.ones(n) / n
    elif idx == 1:
        # Sine
        a = np.sin(np.pi * x) ** 2
        return norm_sum1(a)
    elif idx == 2:
        # Cosine mode 1
        a = np.cos(np.pi * (x - 0.5)) ** 2
        return norm_sum1(a)
    elif idx == 3:
        # Sinc
        t = (x - 0.5) * 8
        a = np.sinc(t) ** 2
        return norm_sum1(a)
    elif idx == 4:
        # Power law
        a = np.arange(1, n + 1, dtype=np.float64) ** (-0.5)
        return norm_sum1(a)
    elif idx == 5:
        # Multi-hump
        a = np.zeros(n)
        for h in range(3):
            c = (h + 1) / 4
            a += np.exp(-((x - c) ** 2) / 0.02)
        return norm_sum1(a)
    elif idx == 6:
        # Random smooth
        rng = np.random.default_rng(idx * 1000)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.1 / np.sqrt(f)) * np.sin(2 * np.pi * f * x)
        return norm_sum1(a)
    elif idx == 7:
        # Anti-symmetric bias (more mass at edges)
        a = np.abs(x - 0.5) + 0.1
        return norm_sum1(a)
    elif idx == 8:
        # Triangular
        a = 1.0 - np.abs(2.0 * x - 1.0)
        return norm_sum1(a)
    else:
        # Perturbed uniform
        rng = np.random.default_rng(idx * 777)
        a = 1.0 + rng.normal(0, 0.1, n)
        return norm_sum1(a)

def interpolate_to_length(p, new_n):
    """Interpolate a probability vector to a new length."""
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return norm_sum1(p_new)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed log-sum-exp objective with L-BFGS-B (scipy)
2. Softmax parameterization for non-negativity + sum-to-one constraints
3. Annealed smoothing parameter (low beta -> high beta)
4. Multiple diverse initializations and random restarts
5. Frequency-domain spectrum flattening (Gerchberg-Saxton style)
6. Adaptive length search with interpolation
7. Coordinate descent polish on the final solution
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Fast linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, clamp non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_fn(q):
    """Stable softmax."""
    q_shifted = q - q.max()
    e = np.exp(q_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(q) / len(q)
    return e / s


def smooth_obj_grad(q, beta, n):
    """
    Compute smoothed objective and its gradient w.r.t. q.
    p = softmax(q); ac = autocorr(p); f = logsumexp(beta*ac)/beta
    df/dq = p * (grad_p - <p, grad_p>)
    where grad_p[j] = 2 * cross_corr(w, p)[j]
    """
    p = softmax_fn(q)
    ac = fast_autocorr(p)

    # Smooth weights
    beta_ac = beta * ac
    beta_ac_shifted = beta_ac - beta_ac.max()
    w = np.exp(beta_ac_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / w_sum

    f = (beta_ac.max() + np.log(w_sum)) / beta

    # grad_p[j] = 2 * sum_k w[k]*p[k-j] (cross-correlation via FFT)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    p_padded = np.zeros(fft_size)
    p_padded[:n] = p

    W = rfft(w_padded)
    P = rfft(p_padded)
    cross = irfft(np.conj(W) * P, fft_size)[:n]
    grad_p = 2.0 * cross

    # Chain rule through softmax
    pgd = np.dot(p, grad_p)
    grad_q = p * (grad_p - pgd)

    return f, grad_q


def true_score(p):
    """Compute the true (unsmoothed) score: 2n * max(autocorr)."""
    n = len(p)
    ac = fast_autocorr(p)
    return 2.0 * n * ac.max()


def lbfgs_optimize(p0, time_limit, beta_start, beta_end, n_phases, max_iter, verbose):
    """
    L-BFGS-B optimization with annealed smoothing via softmax parameterization.
    Returns (best_p, best_score).
    """
    n = len(p0)
    best_p = p0.copy()
    best_score = true_score(p0)
    t0 = time.monotonic()

    q = np.log(p0 + 1e-30)
    q -= q.mean()

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        progress = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        def _obj(q_x):
            f, g = smooth_obj_grad(q_x, beta, n)
            return f

        def _grad(q_x):
            f, g = smooth_obj_grad(q_x, beta, n)
            return g

        def _obj_grad(q_x):
            return smooth_obj_grad(q_x, beta, n)

        try:
            res = minimize(_obj_grad, q, method='L-BFGS-B',
                           jac=True, args=(),
                           options={'maxiter': max_iter, 'gtol': 1e-7,
                                    'ftol': 1e-14, 'maxfun': max_iter * 10,
                                    'iprint': -1})
            q = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error phase {phase}: {e}")
            break

        p = softmax_fn(q)
        p = norm_sum1(p)
        sc = true_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            if verbose:
                print(f"    Phase {phase}, beta={beta:.1f}: score={sc:.6f}")

    return best_p, best_score


def freq_flatten(p0, time_limit, n_iters, eta, verbose):
    """
    Iterative spectrum flattening in frequency domain.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_score = true_score(p)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(p, fft_size)
        P_mag = np.abs(A)
        P_mag[P_mag < 1e-30] = 1e-30

        P_mag_mean = P_mag.mean()
        P_mag_new = P_mag_mean * (P_mag / P_mag_mean) ** (1.0 - eta)
        P_mag_new = np.maximum(P_mag_new, 1e-30)

        phase = np.angle(A)
        A_new = P_mag_new * np.exp(1j * phase)

        p_new = irfft(A_new, fft_size)[:n]
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-15:
            p_new = p.copy()
        else:
            p_new = p_new / s

        p = 0.5 * p + 0.5 * p_new
        p = norm_sum1(p)

        sc = true_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            if verbose and it % 100 == 0:
                print(f"    FreqFlatten it={it}: score={sc:.6f}")

    return best_p, best_score


def coordinate_polish(p0, time_limit, verbose, seed):
    """
    Coordinate descent: move mass between positions to reduce max autocorrelation.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_score = true_score(p)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(p)
        max_idx = int(np.argmax(ac))
        max_val = ac[max_idx]

        k = max_idx
        js = max(0, k - n + 1)
        je = min(n - 1, k)

        if js > je:
            stall += 1
            continue

        idx = np.arange(js, je + 1)
        contrib = p[idx] * p[k - idx]
        top_contrib = idx[np.argmax(contrib)]

        if p[top_contrib] < 1e-10:
            stall += 1
            continue

        candidates = np.argsort(p)[:max(5, n // 8)]
        dst = int(candidates[int(rng.integers(len(candidates)))])

        if dst == top_contrib:
            dst = int(rng.integers(n))
            if dst == top_contrib:
                stall += 1
                continue

        improved = False
        for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = p[top_contrib] * frac
            if amt < 1e-12:
                continue
            p_trial = p.copy()
            p_trial[top_contrib] -= amt
            p_trial[dst] += amt
            p_trial = np.maximum(p_trial, 0.0)
            s_t = p_trial.sum()
            if s_t < 1e-15:
                continue
            p_trial = p_trial / s_t
            sc = true_score(p_trial)
            if sc < best_score - 1e-12:
                p = p_trial
                best_score = sc
                best_p = p.copy()
                stall = 0
                improved = True
                break

        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and p[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = p[i] * frac
                p[i] -= amt
                p[j] += amt
                p = np.maximum(p, 0.0)
                s_t = p.sum()
                if s_t > 1e-15:
                    p = p / s_t
                    sc = true_score(p)
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                        stall = 0
            stall += 1

        if verbose and stall % 100 == 0:
            print(f"    CoordPolish stall={stall}: score={best_score:.6f}")

    return best_p, best_score


def make_init(n, idx):
    """Create diverse initializations by index."""
    x = np.linspace(0, 1, n)

    if idx == 0:
        return np.ones(n) / n
    elif idx == 1:
        a = np.sin(np.pi * x) ** 2
        return norm_sum1(a)
    elif idx == 2:
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0.0)
        return norm_sum1(a)
    elif idx == 3:
        t = (x - 0.5) * 10.0
        a = np.sinc(t) ** 2
        return norm_sum1(a)
    elif idx == 4:
        a = np.arange(1, n + 1, dtype=np.float64) ** (-0.5)
        return norm_sum1(a)
    elif idx == 5:
        a = np.zeros(n)
        for h in range(3):
            c = (h + 1) / 4.0
            a += np.exp(-((x - c) ** 2) / 0.02)
        return norm_sum1(a)
    elif idx == 6:
        rng = np.random.default_rng(42)
        a = np.ones(n) * 0.5
        for f in range(1, 10):
            a += rng.normal(0, 0.1 / np.sqrt(f)) * np.sin(2.0 * np.pi * f * x + rng.uniform(0, 6.28))
        return norm_sum1(a)
    elif idx == 7:
        a = np.abs(x - 0.5) + 0.1
        return norm_sum1(a)
    elif idx == 8:
        a = 1.0 - np.abs(2.0 * x - 1.0)
        return norm_sum1(a)
    elif idx == 9:
        rng = np.random.default_rng(99)
        a = 1.0 + rng.normal(0, 0.15, n)
        return norm_sum1(a)
    else:
        a = np.ones(n) * (1.0 + 0.2 * np.sin(2.0 * np.pi * 3.0 * x))
        return norm_sum1(a)


def interpolate_to_length(p, new_n):
    """Interpolate probability vector to new length."""
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return norm_sum1(p_new)


def propose_candidate():
    """Main search orchestrator. Returns list of best sequence coefficients."""
    t_global = time.monotonic()
    time_budget = 1740.0

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        if p_arr.ndim == 1 and p_arr.size > 0 and np.isfinite(p_arr).all():
            parent = norm_sum1(p_arr)
            p_score = true_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"  Could not load parent: {e}")

    if best_seq is None:
        n_default = 4000
        best_seq = make_init(n_default, 0)
        best_score = true_score(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: L-BFGS with diverse initializations ===
    print(f"\n=== Phase 1: L-BFGS diverse inits (n={n0}) ===")

    init_indices = [None, 1, 3, 7, 9]  # None means use parent
    init_labels = ["parent", "sine", "sinc", "edge", "perturbed"]

    inits = []
    labels = []
    for idx, lbl in zip(init_indices, init_labels):
        if idx is None:
            inits.append(best_seq.copy())
        else:
            inits.append(make_init(n0, idx))
        labels.append(lbl)

    remaining = time_budget - (time.monotonic() - t_global)
    t_per = max(10.0, min(remaining * 0.40 / len(inits), 150.0))

    for i, (p_init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 200:
            break
        t_avail = min(t_per, time_budget - 200 - (time.monotonic() - t_global))
        if t_avail < 10:
            break
        print(f"  L-BFGS '{lbl}' (budget={t_avail:.0f}s)...")
        p_out, sc = lbfgs_optimize(p_init, t_avail,
                                   beta_start=15, beta_end=400,
                                   n_phases=5, max_iter=400, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [lbfgs_{lbl}]: {sc:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # === Phase 2: High-beta fine-tuning on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Fine-tune high-beta (n={n_best}) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 20:
        p_out, sc = lbfgs_optimize(best_seq.copy(), t2,
                                   beta_start=200, beta_end=1200,
                                   n_phases=5, max_iter=500, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [fine_tune]: {sc:.6f}")

    # === Phase 3: Frequency domain ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Freq domain (n={n_best}) ===")
    t3 = min(remaining * 0.08, 90.0)
    if t3 > 15:
        p_out, sc = freq_flatten(best_seq.copy(), t3,
                                 n_iters=3000, eta=0.5, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [freqdom]: {sc:.6f}")
        # Follow-up L-BFGS from freq domain result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t3b = min(remaining * 0.06, 60.0)
            p_out2, sc2 = lbfgs_optimize(p_out, t3b,
                                         beta_start=300, beta_end=1000,
                                         n_phases=4, max_iter=400, verbose=False)
            if sc2 < best_score:
                best_score = sc2
                best_seq = p_out2.copy()
                print(f"  *** NEW BEST [freqdom+lbfgs]: {sc2:.6f}")

    # === Phase 4: Length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length search (base n={n_best}) ===")

    candidate_lengths = []
    for delta in [-1000, -500, 500, 1000, 1500, 2000]:
        ln = n_best + delta
        if 1500 <= ln <= 8000:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        if new_n == n_best:
            continue
        print(f"  Trying n={new_n}...")
        p_interp = interpolate_to_length(best_seq, new_n)
        sc_interp = true_score(p_interp)
        if sc_interp < best_score:
            best_score = sc_interp
            best_seq = p_interp.copy()
            print(f"  *** NEW BEST [interp_n{new_n}]: {sc_interp:.6f}")

        t4 = min(remaining * 0.20, 120.0)
        if t4 > 15:
            p_out, sc = lbfgs_optimize(p_interp, t4,
                                       beta_start=50, beta_end=600,
                                       n_phases=5, max_iter=400, verbose=False)
            if sc < best_score:
                best_score = sc
                best_seq = p_out.copy()
                print(f"  *** NEW BEST [lbfgs_n{new_n}]: {sc:.6f}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        p_out, sc = coordinate_polish(best_seq.copy(), t5,
                                      verbose=True, seed=42)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [coord]: {sc:.6f}")

    # === Phase 6: Randomized restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Restarts (n={n_best}) ===")

    rng_main = np.random.default_rng(2024)
    n_restarts = max(2, min(6, int(remaining / 80)))
    t_per_r = min(remaining * 0.60 / n_restarts, 120.0)

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        strength = 0.02 + 0.015 * r
        noise = rng_main.normal(0, strength, n_best)
        p_pert = best_seq * (1.0 + noise)
        p_pert = norm_sum1(p_pert)

        t_avail = min(t_per_r, remaining - 20)
        if t_avail < 10:
            break
        p_out, sc = lbfgs_optimize(p_pert, t_avail,
                                   beta_start=100, beta_end=900,
                                   n_phases=5, max_iter=500, verbose=False)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [restart_{r}]: {sc:.6f} (strength={strength:.3f})")

    # === Phase 7: Final high-beta L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final L-BFGS (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.40, 120.0)
        p_out, sc = lbfgs_optimize(best_seq.copy(), t7,
                                   beta_start=500, beta_end=3000,
                                   n_phases=4, max_iter=600, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [final_lbfgs]: {sc:.6f}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Final coord polish (n={n_best}) ===")
        t8 = min(remaining * 0.70, 80.0)
        p_out, sc = coordinate_polish(best_seq.copy(), t8,
                                      verbose=False, seed=777)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [final_coord]: {sc:.6f}")

    # === Phase 9: One more L-BFGS if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t9 = min(remaining * 0.70, 60.0)
        p_out, sc = lbfgs_optimize(best_seq.copy(), t9,
                                   beta_start=1000, beta_end=5000,
                                   n_phases=3, max_iter=800, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p_out.copy()
            print(f"  *** NEW BEST [ultra_lbfgs]: {sc:.6f}")

    # Final
    if best_seq is not None:
        final_sc = true_score(best_seq)
        print(f"\nFINAL: score={final_sc:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        best_seq = make_init(4000, 0)
        final_sc = true_score(best_seq)
        print(f"\nFALLBACK: score={final_sc:.6f}")

    return [float(x) for x in best_seq]
