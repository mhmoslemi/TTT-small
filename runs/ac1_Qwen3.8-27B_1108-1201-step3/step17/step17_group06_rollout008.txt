Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on softmax-parameterized simplex with FFT-based smooth gradients
2. Adaptive log-sum-exp temperature annealing
3. Pair-based coordinate descent with line search
4. Multi-start with diverse structural initializations
5. Length sweeping with interpolation
6. Periodic restarts with geometric perturbations

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def true_obj(a):
    """True objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def smooth_obj_and_grad_a(a, beta):
    """
    Compute smoothed objective and gradient w.r.t. a on the simplex.
    Returns (obj, grad_a) where grad_a has shape (n,).
    """
    n = len(a)
    c = fast_conv(a)
    # Log-sum-exp for numerical stability
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    log_sum = np.log(exp_bc.sum()) + beta * c_max
    obj = 2.0 * n * log_sum / beta

    # Softmax weights over lags
    p = exp_bc / exp_bc.sum()

    # Gradient: grad_a[i] = 4*n * sum_k p[k] * a[k-i]
    # This is the cross-correlation (p ⊛ a) evaluated at lags 0..n-1
    # Using FFT: IFFT(FFT(p) * conj(FFT(a)))[:n]
    N_corr = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N_corr)
    A = np.fft.rfft(a, N_corr)
    corr = np.fft.irfft(P * np.conj(A), N_corr)[:n]
    grad_a = 4.0 * n * corr

    return obj, grad_a


def smooth_obj_and_grad_z(z, beta):
    """
    Compute smoothed objective and gradient w.r.t. z where a = softmax(z).
    Returns (obj, grad_z).
    """
    n = len(z)
    a = softmax(z)
    obj, grad_a = smooth_obj_and_grad_a(a, beta)
    # Softmax gradient: grad_z = a * (grad_a - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return obj, grad_z

    Adam optimizer on softmax parameterization with adaptive temperature.

    n = len(a0)
    rng = np.random.default_rng(seed)

    a = np.maximum(a0.copy(), 1e-30)
    a = a / a.sum()
    z = np.log(a)

    best_a = a.copy()
    best_obj = true_obj(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    stall = 0

def coord_pair_descent(a0, time_limit, verbose=False, seed=42, max_stall=100):
    """
    Pair-based coordinate descent: transfer mass between pairs of positions.
    Uses gradient to guide pair selection.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, true_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = true_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = max_stall

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Compute gradient to guide pair selection
        c = fast_conv(a)
        # Use top lags to compute approximate gradient
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        # Softmax weights on top lags
        top_vals = c[top_idx]
        beta_w = 20.0
        w = np.exp(beta_w * (top_vals - top_vals.max()))
        w = w / w.sum()

        # Approximate gradient from top lags
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                j_idx = k_i - idx
                g[idx] += 2.0 * wi * a[j_idx]

        # Select source: high gradient + high mass
        src_score = g * a
        # Select destination: low gradient
        dst_score = -g

        # Try multiple random pairs guided by gradient
        n_tries = 30
        improved = False

        for _ in range(n_tries):
            # Weighted random selection
            src_weights = np.maximum(src_score, 1e-30)
            dst_weights = np.maximum(dst_score - dst_score.min() + 1e-10, 1e-10)

            src = int(rng.choice(n, p=src_weights / src_weights.sum()))
            dst = int(rng.choice(n, p=dst_weights / dst_weights.sum()))

            if src == dst or a[src] < 1e-12:
                continue

            # Try multiple transfer fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                a_t /= a_t.sum()
                obj_t = true_obj(a_t)
                if obj_t < best_obj - 1e-14:
                    a = a_t
                    best_obj = obj_t
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Occasional random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    obj_new = true_obj(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD ({elapsed:.0f}s): obj={best_obj:.8f}, stall={stall}")

    return best_a, best_obj

def make_smooth_random(n, seed=0, n_freq=8):

    rng = np.random.default_rng(seed)

        phase = rng.uniform(0, 2 * np.pi)

    s = a.sum()
    if s < 1e-15:

    return a / s

def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length, returns normalized."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

            print(f"Parent loaded: n={len(parent)}, obj={pobj:.8f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        obj = true_obj(a)
        if obj < best_score:
            best_score = obj
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: obj={obj:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = true_obj(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = true_obj(best_a)
        print(f"Initial (no parent): n={len(best_a)}, obj={best_score:.8f}")

    n_best_init = len(best_a)

    # === Phase 1: Adam optimization from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam from parent (n={n_best_init}), budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = adam_optimize(best_a.copy(), t1, beta_start=5.0, beta_end=50.0,
                                  lr=0.008, restart_every=4000, verbose=True, seed=1)
        try_update(a_out, "adam_parent")

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord Descent (n={len(best_a)}), budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, _ = coord_pair_descent(best_a.copy(), t2, verbose=True, seed=7, max_stall=80)
        try_update(a_out, "cd1")

    # === Phase 3: Adam again after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Adam after CD, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = adam_optimize(best_a.copy(), t3, beta_start=10.0, beta_end=80.0,
                                  lr=0.005, restart_every=5000, verbose=True, seed=2)
        try_update(a_out, "adam_cd")

    # === Phase 4: Multi-init Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-init Adam, budget={remaining:.0f}s ===")

    n_cur = len(best_a)
    inits = []
    labels = []

    inits.append(make_power_ramp(n_cur, 0.5))
    labels.append("ramp0.5")
    inits.append(make_power_ramp(n_cur, 1.5))
    labels.append("ramp1.5")
    inits.append(make_bimodal(n_cur, 30.0))
    labels.append("bimodal30")
    inits.append(make_triangular(n_cur, 0.5, 0.4))
    labels.append("triangular")
    inits.append(make_smooth_random(n_cur, seed=7))
    labels.append("smooth7")
    inits.append(make_log_ramp(n_cur))
    labels.append("logramp")
    inits.append(make_asymmetric(n_cur, 0.5, 0.5))
    labels.append("asym")

    # Blend with parent
    if parent is not None and len(parent) == n_cur:
        inits.append(0.5 * parent + 0.5 * make_power_ramp(n_cur, 1.0))
        labels.append("blend_ramp")
        inits.append(0.7 * parent + 0.3 * make_bimodal(n_cur, 40.0))
        labels.append("blend_bimod")

    t_per_init = max(15.0, remaining * 0.25 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 60:
            break
        t_avail = min(t_per_init, time_budget - 60 - elapsed)
        if t_avail < 10:
            break
        print(f"  Multi-init [{lbl}] ({t_avail:.0f}s)...", end=" ")
        a_out, obj_out = adam_optimize(init.copy(), t_avail, beta_start=5.0,
                                        beta_end=40.0, lr=0.008, restart_every=3000,
                                        verbose=False, seed=100 + idx)
        improved = try_update(a_out, f"mi_{lbl}")
        print(f"obj={obj_out:.8f}{' [NEW BEST]' if improved else ''}")

    # === Phase 5: Coordinate descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: CD on best (n={len(best_a)}), budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_out, _ = coord_pair_descent(best_a.copy(), t5, verbose=True, seed=99, max_stall=60)
        try_update(a_out, "cd2")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep, budget={remaining:.0f}s ===")

    n_best_now = len(best_a)
    lengths_to_try = []
    for delta in [-300, -150, 150, 300, 500, 800]:
        nl = n_best_now + delta
        if 2000 <= nl <= 6000:
            lengths_to_try.append(nl)
    for nl in [2500, 3000, 3500, 4000, 4500, 5000]:
        if nl not in lengths_to_try and 2000 <= nl <= 6000:
            lengths_to_try.append(nl)

    lengths_to_try = sorted(set(lengths_to_try))
    # Limit to top 5 most promising
    lengths_to_try = lengths_to_try[:6]

    for nl in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best_now) < 50:
            continue
        print(f"  Length {nl}...", end=" ")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.4, 80.0)
        if t_md > 10:
            a_out, obj_out = adam_optimize(a_int, t_md, beta_start=5.0, beta_end=40.0,
                                            lr=0.008, restart_every=3000, verbose=False, seed=nl % 1000)
            improved = try_update(a_out, f"len{nl}")
            print(f"obj={obj_out:.8f}{' [NEW BEST]' if improved else ''}")

    # === Phase 7: Final Adam polish (longer, higher temperature range) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_a)
    print(f"\n=== Phase 7: Final Adam Polish (n={n_final}), budget={remaining:.0f}s ===")
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 30:
        a_out, _ = adam_optimize(best_a.copy(), t7, beta_start=15.0, beta_end=100.0,
                                  lr=0.003, restart_every=6000, verbose=True, seed=777)
        try_update(a_out, "final_adam")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final CD (n={len(best_a)}), budget={remaining:.0f}s ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 15:
        a_out, _ = coord_pair_descent(best_a.copy(), t8, verbose=True, seed=314, max_stall=50)
        try_update(a_out, "final_cd")

    # === Phase 9: One more Adam if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 9: Extra Adam, budget={remaining:.0f}s ===")
        t9 = remaining - 15
        a_out, _ = adam_optimize(best_a.copy(), t9, beta_start=20.0, beta_end=120.0,
                                  lr=0.002, restart_every=8000, verbose=False, seed=999)
        try_update(a_out, "extra_adam")

    # Final output
    if best_a is not None:
        final_obj = true_obj(best_a)
        best_score = min(best_score, final_obj)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: obj={best_score:.8f}, n={len(best_a)}, time={elapsed:.0f}s")
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in best_a])
            print(f"evaluate_sequence confirms: {eval_score:.8f}")
        except Exception as e:
            print(f"Warning: evaluate_sequence failed: {e}")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, obj={true_obj(fb):.8f}")
        return [float(x) for x in fb]
