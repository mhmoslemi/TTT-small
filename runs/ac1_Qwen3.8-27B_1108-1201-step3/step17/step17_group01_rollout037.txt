    p = np.exp(c / T - c.max())

        k_min = max(0, i)
        k_max = min(m-1, i+n-1)
        if k_min > k_max:

        k_range = np.arange(k_min, k_max+1)

        grad[i] = np.sum(contrib) / T
    return grad

2. FFT-based gradient computation in O(n log n) per iteration

4. Temperature annealing from soft to hard max

6. Final coordinate-level peak shaving

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_softmax_gradient(a, c, T):
    """Compute gradient of log-sum-exp(c/T) w.r.t. a."""
    n = len(a)
    m = len(c)  # 2n-1
    # Softmax weights with temperature
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    p = np.exp(c_scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(m) / m
    else:
        p = p / p_sum
    
    # C[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # Computed via FFT cross-correlation
    N = 1 << (m + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(P * np.conj(A), N)
    C = R[:n]
    
    # Correction for the diagonal term (where k = 2i)
    two_i = 2 * np.arange(n)
    corr = np.zeros(n)
    valid = two_i < m
    corr[valid] = p[two_i[valid]] * a[valid]
    
    grad = (2.0 * C - corr) / T
    return grad

def lse_mirror_descent(a0, time_limit, T0_frac=0.05, T_min_frac=0.001, 
                        gamma=5.0, restart_every=5000, rng_seed=42, verbose=False):
    """Soft-max mirror descent with Nesterov acceleration and temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c0 = fast_autocorr(a)
    best_m = c0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    c_max0 = c0.max()
    T = max(T0_frac * c_max0, 1e-10)
    T_min = max(T_min_frac * c_max0, 1e-12)
    
    x = a.copy()
    x_prev = a.copy()
    
    stall = 0
    max_stall = restart_every
    restart_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        c = fast_autocorr(y)
        
        # Compute gradient
        g = compute_softmax_gradient(y, c, T)
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_y = np.log(y) - eta * g
        max_log = log_y.max()
        x_new = np.exp(log_y - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        x_prev = x
        x = x_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Temperature annealing
        T = max(T * 0.9999, T_min)
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.02 + 0.01 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            T = max(T0_frac * best_m, T * 0.5)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LSE-MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, T={T:.6f}, restarts={restart_count}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({elapsed:.0f}s): J={J:.6f}, T={T:.6f}, stall={stall}")
    
    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=False, rng_seed=42):
    """Iteratively reduce the peak of the autocorrelation by redistributing mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M_curr = c.max()
        
        # Find top peaks
        n_peaks = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k)
            # Find positions that contribute most to this peak
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the position with maximum contribution
            src = idx[np.argmax(contrib)]
            
            if a[src] < 1e-12:
                continue
            
            # Find a good destination: minimize increase in other peaks
            # Compute the "cost" of placing mass at each position
            # Cost at position d: sum over all peaks of a[peak - d] + a[d - peak]
            costs = np.zeros(n)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                for d in range(n):
                    j1 = k2i - d
                    j2 = d - k2i
                    cost_val = 0.0
                    if 0 <= j1 < n:
                        cost_val += a[j1]
                    if 0 <= j2 < n and j2 != j1:
                        cost_val += a[j2]
                    costs[d] += cost_val
            
            # Add a small penalty for the current peak
            costs[src] += M_curr
            
            # Find the best destination
            # Sort by cost, pick from the lowest-cost positions
            sorted_costs = np.argsort(costs)
            candidates = sorted_costs[:max(1, n // 10)]
            
            # Pick the candidate that reduces the peak the most
            best_dst = None
            best_improvement = 0
            for dst in candidates:
                dst_int = int(dst)
                if dst_int == src:
                    continue
                # Try moving a small fraction
                for frac in [0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        best_m = M_trial
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = min(0.5, rng.uniform(0.01, 0.1))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_multihump(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(a)

def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (softmax) mirror descent with Nesterov acceleration
2. FFT-based O(n log n) gradient computation via cross-correlation
3. Temperature annealing from soft max to hard max
4. Spectral equalization (magnitude compression)
5. Peak-shaving coordinate descent for final polish
6. Multi-initialization and multi-length search

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_softmax_gradient(a, c, T):
    """Compute gradient of log-sum-exp(c/T) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    m = len(c)
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    p = np.exp(c_scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(m) / m
    else:
        p = p / p_sum
    N = 1 << (m + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(P * np.conj(A), N)
    C = R[:n]
    two_i = 2 * np.arange(n)
    corr = np.zeros(n)
    valid = two_i < m
    corr[valid] = p[two_i[valid]] * a[valid]
    grad = (2.0 * C - corr) / T
    return grad


def lse_mirror_descent(a0, time_limit, T0_frac=0.05, T_min_frac=0.005,
                       gamma=5.0, restart_every=5000, rng_seed=42,
                       symmetrize=False, verbose=False):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c0 = fast_autocorr(a)
    best_m = c0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    c_max0 = c0.max()
    T = max(T0_frac * c_max0, 1e-10)
    T_min = max(T_min_frac * c_max0, 1e-12)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        if symmetrize:
            y = 0.5 * (y + y[::-1])
            y = y / y.sum()

        c = fast_autocorr(y)
        g = compute_softmax_gradient(y, c, T)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30)) - eta * g
        max_log = log_y.max()
        x_new = np.exp(log_y - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        if symmetrize:
            x_new = 0.5 * (x_new + x_new[::-1])
            x_new = x_new / x_new.sum()

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        T = max(T * 0.99995, T_min)

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.02 + 0.01 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            if symmetrize:
                x_r = 0.5 * (x_r + x_r[::-1])
                x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            T = max(T0_frac * best_m * 0.5, T * 0.3)

            if verbose:

def spectral_equalize(a, alpha=0.3):
    """Compress spectral magnitude to flatten autocorrelation."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    safe_mag = np.maximum(mag, 1e-30)
    phase = A / safe_mag
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    mag_new = mag * (mag / mag_mean) ** (-alpha)
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def peak_shaving(a0, time_limit, verbose=False, rng_seed=42):

        improved = False

        for ki in range(n_peaks):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(top_idx[ki])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue

            costs = np.zeros(n)

                k2i = int(k2)

                    j2 = d - k2i
                    cv = 0.0
                    if 0 <= j1 < n:
                        cv += a[j1]
                    if 0 <= j2 < n and j2 != j1:
                        cv += a[j2]
                    costs[d] += cv
            costs[src] += M_curr

            sorted_idx = np.argsort(costs)
            candidates = sorted_idx[:max(1, n // 8)]
            found = False
            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.03, 0.08, 0.15]:
                    amt = a[src] * frac

                        continue

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
