import numpy as np

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def build_sparse_T(a):
    """Build sparse matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # T[k,j] = a[k-j] for k=j to j+n-1
        k_range = np.arange(j, j + n)
        val_range = np.arange(n)  # a[0], a[1], ..., a[n-1]
        rows.extend(k_range.tolist())
        cols.extend([j] * n)
        vals.extend(a.tolist())
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T

def build_sparse_T_fast(a):
    """Build sparse matrix T more efficiently using banded structure."""
    n = len(a)
    # For each column j, the non-zeros are at rows j, j+1, ..., j+n-1 with values a[0], a[1], ..., a[n-1]
    # This is a "constant diagonal" structure shifted for each column
    
    # More efficient: build column by column
    row_indices = []
    col_indices = []
    data = []
    
    for j in range(n):
        start_row = j
        row_indices.append(np.arange(start_row, start_row + n))
        col_indices.append(np.full(n, j))
        data.append(a)
    
    rows = np.concatenate(row_indices)
    cols = np.concatenate(col_indices)
    vals = np.concatenate(data)
    
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T

def nesterov_subgradient_v2(a0, time_limit, verbose=False, top_k=20, beta=2.5, c=3.0, 
                              exact_frac=0.3, restart_every=2000, seed=42):
    """
    Improved Nesterov subgradient with better step size schedule and
    adaptive peak selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    mu = 1.0  # running average of best J
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Subgradient
        if rng.random() < exact_frac:
            k = int(np.argmax(ac))
            g = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            g[idx] = 2.0 * a[j_idx]
        else:
            g = np.zeros(n)
            k_max = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            # Use softmax weighting
            top_vals = ac[top_idx]
            scale = top_vals.max()
            w = np.exp((top_vals - scale) / (0.1 * scale + 1e-10))
            w = w / w.sum()
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * a[j_idx]
        
        # Step size: use diminishing with Polyak-like adjustment
        g_norm = np.dot(g, g)
        if g_norm < 1e-30:
            eta = 2.0 / (t + 1.0)
        else:
            # Polyak step: (f(x) - f*) / ||g||^2
            # f* lower bound: 1.0 (theoretical minimum of J/n * n / (2n-1))
            # Actually lower bound on M when sum=1 is 1/(2n-1) (if possible)
            # J lower bound is 2n/(2n-1) ~ 1
            # So M_lower = 1/(2n) approximately
            M_lower = 1.0 / n  # loose lower bound
            gap = max(M - M_lower, 1e-10)
            eta = c * gap / (g_norm + 1e-20)
            eta = min(eta, 2.0 / (t + 1.0))
            eta = max(eta, 1e-10)
        
        # Mirror descent step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        
        # Restart
        if stall > restart_every:
            stall = 0
            sigma = 0.02
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD v2 t={t} ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J

def spectral_init(n, mode='cosine'):
    """Create initialization based on spectral considerations."""
    if mode == 'cosine':
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
        # Add some structure to reduce autocorrelation peaks
        a = a + 0.01 * np.exp(-((x - 0.5) / 0.1) ** 2)
    elif mode == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.1) / 0.15) ** 2) + np.exp(-((x - 0.9) / 0.15) ** 2) + 0.01
    elif mode == 'triodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.05) / 0.1) ** 2) + np.exp(-((x - 0.5) / 0.15) ** 2) + np.exp(-((x - 0.95) / 0.1) ** 2) + 0.005
    elif mode == 'fourier':
        # Try to create a signal whose power spectrum is flat
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 13 * x) + 0.1 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Wait, I have a problem. The `try_update` function is a closure that captures `best_seq` and `best_score`. The rules say "Don't use any lambda functions" and "Make all helper functions top level and have no closures from function nesting."

Let me restructure to avoid closures. I'll use a mutable container (list or dict) to hold the best sequence and score.

Also, I need to be more careful about the LP construction. Building the full sparse matrix T for n=3200 means creating a 6399x3200 matrix with ~10M entries. This might be slow to construct repeatedly. Let me precompute it or use a more efficient approach.

Actually, for the LP, I don't need to explicitly build T. I can express the constraint differently:
T_sel @ b <= M

where T_sel is the subset of rows. But building T is still needed.

Alternative: I can express the LP constraint as:
For each selected lag k: sum_j a_{k-j} * b_j <= M (where a_{k-j} = 0 if k-j not in [0,n))

This is a linear constraint in b. I can build it as a scipy sparse matrix once and reuse it.

Actually, the issue is that T depends on a (the current point), so I need to rebuild it each iteration. But the structure (which entries are non-zero) is fixed - only the values change.

Let me precompute the structure:
- For column j, the non-zero rows are j, j+1, ..., j+n-1
- The values are a[0], a[1], ..., a[n-1]

So I can precompute the row and column indices once, and just update the values each iteration.

def precompute_T_structure(n):
    """Precompute row/col indices for the autocorrelation matrix."""
    rows = []
    cols = []
    for j in range(n):
        row_start = j
        rows.append(np.arange(row_start, row_start + n))

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LP Frank-Wolfe with precomputed sparse structure and line search
2. Active-set LP polish for multi-peak equalization
3. Nesterov subgradient descent with adaptive peak weighting
4. Spectral initialization and multi-length search
Key innovations: Precomputed T-matrix structure for fast LP construction,
line-search step sizes in FW, and explicit multi-peak equalization steps.

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_solver():
    """Find best available LP solver."""
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def precompute_T_structure(n):
    """Precompute row/col indices for the autocorrelation convolution matrix."""
    rows_list = []
    cols_list = []
    for j in range(n):
        rows_list.append(np.arange(j, j + n))
        cols_list.append(np.full(n, j, dtype=np.int32))
    rows = np.concatenate(rows_list).astype(np.int32)
    cols = np.concatenate(cols_list)
    return rows, cols


def make_T_sparse(a, rows, cols):
    """Create sparse T matrix from precomputed structure and current a values."""
    n = len(a)
    vals = np.tile(a.astype(np.float64), n)
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def lp_frank_wolfe(a0, time_limit, solver, rows, cols, verbose=False,
                   top_K=3000, max_iters=500):
    """
    LP Frank-Wolfe with precomputed sparse structure and line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select binding constraints
        K_sel = min(top_K, 2 * n - 1)
        if K_sel < (2 * n - 1):
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]
        else:
            binding_idx = np.arange(2 * n - 1)

        # Build sparse T
        T = make_T_sparse(y, rows, cols)
        T_sel = T[binding_idx, :].tocsr()

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Retry with fewer constraints
            K2 = max(500, K_sel // 3)
            binding_idx2 = np.argsort(M_curr - ac)[:K2]
            T_sel2 = T[binding_idx2, :].tocsr()
            b_var2 = cp.Variable(n, nonneg=True)
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_sel2 @ b_var2 <= M_curr])
            try:
                prob2.solve(solver=solver, verbose=False)
                if prob2.status not in ('optimal', 'optimal_inaccurate'):
                    continue
                b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Line search: try several step sizes
        best_alpha = 2.0 / (t + 2.0)
        best_trial_m = M_curr

        for alpha in [2.0 / (t + 2.0), 1.0 / (t + 2.0), 3.0 / (t + 2.0), 1.0 / (t + 1.0)]:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_trial = y_trial.sum()
            if s_trial < 1e-15:
                continue
            y_trial = y_trial / s_trial
            trial_m = fast_autocorr(y_trial).max()
            if trial_m < best_trial_m:
                best_trial_m = trial_m
                best_alpha = alpha

        # Nesterov update with found alpha
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            if verbose and (t % 5 == 0 or my < 1e-4):
                elapsed = time.monotonic() - t0
                print(f"    LP-FW iter={t} ({elapsed:.0f}s): max_conv={best_m:.8f}, J={2 * n * best_m:.6f}")

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

    return best_a, best_m


def active_set_polish(a0, time_limit, solver, verbose=False, n_active=15, max_iters=100):
    """
    Active-set LP polish: minimize the maximum over a few active peaks
    via a small LP for the step direction.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Select active peaks
        K = min(n_active, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K]

        # Build gradient vectors for active peaks
        # grad_k[j] = 2*a[k-j] if 0<=k-j<n else 0
        G = np.zeros((K, n))
        for ii, k in enumerate(top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                exc = k_int - j
                if 0 <= exc < n:
                    G[ii, j] = 2.0 * a[exc]

        # LP: min t s.t. G @ d <= t - ac[active], sum(d)=0, d >= -a*0.8
        d_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            G @ d_var <= t_var - ac[top_idx],
            cp.sum(d_var) == 0,
            d_var >= -a * 0.8,
            d_var <= a * 0.8
        ]

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        d_val = np.asarray(d_var.value, dtype=np.float64)
        t_val = float(t_var.value)

        # Line search on the step
        best_frac = 0.0
        best_trial_m = M_curr
        for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 1.0]:
            a_trial = a + frac * d_val
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            trial_m = fast_autocorr(a_trial).max()
            if trial_m < best_trial_m - 1e-15:
                best_trial_m = trial_m
                best_frac = frac

        if best_frac > 0:
            a = a + best_frac * d_val
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            curr_m = fast_autocorr(a).max()
            if curr_m < best_m - 1e-15:
                best_m = curr_m
                best_a = a.copy()
                if verbose and (iteration % 5 == 0):
                    elapsed = time.monotonic() - t0
                    print(f"    ActiveSet iter={iteration} ({elapsed:.0f}s): max_conv={best_m:.8f}, J={2 * n * best_m:.6f}")
        else:
            # No improvement, try smaller active set
            if K > 5:
                n_active = max(5, K // 2)
            else:
                break

    return best_a, best_m


def nesterov_sgd(a0, time_limit, verbose=False, top_k=20, c=3.0,
                 exact_frac=0.3, restart_every=3000, seed=42):
    """Nesterov subgradient descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        # Subgradient
        if rng.random() < exact_frac:
            k = int(np.argmax(ac))
            g = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            g[idx] = 2.0 * a[j_idx]
        else:
            g = np.zeros(n)
            k_max = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = ac[top_idx]
            scale = top_vals.max()
            if scale > 1e-15:
                w = np.exp((top_vals - scale) / (0.05 * scale + 1e-10))
                w = w / w.sum()
            else:
                w = np.ones(k_max) / k_max
            for wi, kk in zip(w, top_idx):
                k_int = int(kk)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * a[j_idx]

        g_norm = np.dot(g, g)
        if g_norm < 1e-30:
            eta = 2.0 / (t + 1.0)
        else:
            M_lower = 1.0 / n
            gap = max(M - M_lower, 1e-10)
            eta = c * gap / (g_norm + 1e-20)
            eta = min(eta, 2.0 / (t + 1.0))
            eta = max(eta, 1e-10)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.02
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J
