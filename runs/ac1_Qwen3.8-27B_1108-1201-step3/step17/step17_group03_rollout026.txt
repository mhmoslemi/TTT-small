    elif family == 'cubic':
        a = np.arange(n, 0, -1, dtype=np.float64) ** 3
    elif family == 'sqrt':
        a = np.sqrt(np.arange(n, 0, -1, dtype=np.float64))
    elif family == 'sine':
        a = np.sin(np.pi * np.linspace(0, 1, n))
    elif family == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return a / a.sum()

def lp_frank_wolfe_v2(a0, time_limit, verbose=True, top_K=500, margin_frac=1e-3):
    """
    LP-based Frank-Wolfe for minimizing max autocorrelation.
    Key insight: the LP subproblem uses the linearization of r(k) at the current point.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()  # for accelerated FW
    
    for t in range(1, 100000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select active constraints
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        
        if n_active > top_K:
            # Select the top_K most violated
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix: for each active lag k, the constraint is
        # sum_m a[k-m] * b[m] <= (M_curr + ac[k]) / 2
        # The matrix row for lag k is: [a[k-0], a[k-1], ..., a[k-(n-1)]] (with a[x]=0 for x<0 or x>=n)
        # This is a Toeplitz-like matrix.
        
        # For efficiency, we'll build the matrix directly
        T = np.zeros((K, n))
        for row, k in enumerate(active_idx):
            k_int = int(k)
            # b[m] is multiplied by a[k-m]
            # Valid m: 0 <= m < n and 0 <= k-m < n
            # i.e., max(0, k-n+1) <= m <= min(n-1, k)
            m_start = max(0, k_int - n + 1)
            m_end = min(n, k_int + 1)
            for m in range(m_start, m_end):
                T[row, m] = y[k_int - m]
        
        # RHS
        rhs = (M_curr + ac[active_idx]) / 2.0
        
        # Solve LP: maximize sum(b) subject to T @ b <= rhs, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Accelerated FW
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten margin
            margin_frac = max(margin_frac * 0.9, 1e-5)
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            J = 2.0 * n * best_m
            print(f"    LP-FW v2 iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")
    
    return best_a, best_m
