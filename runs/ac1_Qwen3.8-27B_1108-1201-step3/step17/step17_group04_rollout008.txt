_global_solver = None

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_mirror_descent(a0, time_limit, top_k=15, beta=3.0, c=3.0,

    for t in range(1, 200000000):

        y = np.maximum(y, 1e-15)

        lower_bound = 1.0 / n

        eta = min(eta, 2.0 / (t + 1.0))

        log_a = np.log(y) - eta * g

def bisection_peak_reduction(a0, time_limit, seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]

        # Compute sensitivity for top peaks
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii, jj in zip(idx, j_idx):
                sensitivity[ii] += a[jj]
                sensitivity[jj] += a[ii]

        # Find high-sensitivity positions with mass
        threshold_hi = sensitivity.mean() + 0.5 * sensitivity.std()
        threshold_lo = sensitivity.mean() - 0.5 * sensitivity.std()
        high_sens = np.where((sensitivity > threshold_hi) & (a > 0.0005))[0]
        low_sens = np.where((sensitivity < threshold_lo) & (a < 0.01))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            if stall % 50 == 0:
                _random_perturb(a, rng, n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
            continue

        improved = False

        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            # Bisection to find optimal transfer amount
            lo = 0.0
            hi = src_val * 0.5
            best_amt = lo
            best_trial_J = best_J

            for _bis in range(12):
                mid = (lo + hi) / 2.0
                if mid < 1e-12:
                    break
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_amt = mid
                    lo = mid
                else:
                    hi = mid

            if best_amt > 1e-10 and best_trial_J < best_J - 1e-14:
                a[src] -= best_amt
                a[dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_J


def _random_perturb(a, rng, n):
    n_pert = max(3, n // 200)
    for _ in range(n_pert):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and a[i] > 1e-8:
            amt = a[i] * rng.uniform(0.01, 0.05)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver

            prob.solve(solver=s, verbose=False)

        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_mirror_descent(a0, time_limit, top_k=15, beta=3.0, c=3.0,
                            exact_frac=0.4, restart_every=3000, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c
    cur_exact_frac = exact_frac

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        use_exact = rng.random() < cur_exact_frac
        if use_exact:
            g = get_subgradient(y, ac, exact=True)
        else:
            g = get_subgradient(y, ac, exact=False, top_k=cur_top_k, beta=cur_beta)

        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(3, (t // restart_every) % 6)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_c = min(cur_c + 0.3, 8.0)
            cur_exact_frac = min(cur_exact_frac + 0.03, 0.7)

    return best_a, best_J

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:12]

        sensitivity = np.zeros(n)
        n_peaks_use = min(6, len(top_lags))
        for k in top_lags[:n_peaks_use]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii, jj in zip(idx, j_idx):
                sensitivity[ii] += a[jj]
                sensitivity[jj] += a[ii]

        mean_s = sensitivity.mean()
        std_s = sensitivity.std()
        if std_s < 1e-15:
            std_s = 1e-15
        threshold_hi = mean_s + 0.3 * std_s
        threshold_lo = mean_s - 0.3 * std_s
        high_sens = np.where((sensitivity > threshold_hi) & (a > 0.0003))[0]
        low_sens = np.where((sensitivity < threshold_lo))[0]

        if len(high_sens) == 0:
            high_sens = np.where(a > np.median(a))[0]
        if len(low_sens) == 0:
            low_sens = np.where(a < np.median(a))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            if stall % 30 == 0:
                n_pert = max(3, n // 100)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.01, 0.05)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
            continue

        improved = False

        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            fracs = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.4]
            best_amt = -1.0
            best_trial_J = best_J

            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_amt = amt

            if best_amt > 0 and best_trial_J < best_J - 1e-14:
                a[src] -= best_amt
                a[dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_J


def lp_frank_wolfe_cg(a0, time_limit, solver, top_K=250):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        margins = M_curr - ac
        K_sel = min(top_K + t * 2, 2 * n - 1)
        active_idx = np.argsort(margins)[:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ri in range(K_sel):
            k_int = int(active_idx[ri])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[ri, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, candidate_a, label, t_global):
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: Nesterov SGD from parent
    print(f"\n=== Phase 1: Nesterov SGD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.13, 180.0)
    if t1 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t1,
                                           top_k=15, beta=3.0, c=3.0,
                                           exact_frac=0.35, restart_every=2000, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_p1", t_global)

    # Phase 1b: SGD from a slightly perturbed parent
    remaining = time_budget - (time.monotonic() - t_global)
    t1b = min(remaining * 0.08, 100.0)
    if t1b > 15:
        rng_init = np.random.default_rng(0)
        a_pert = norm_sum1(best_seq) + rng_init.normal(0, 0.005, n0)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = nesterov_mirror_descent(a_pert, t1b,
                                           top_k=20, beta=2.5, c=4.0,
                                           exact_frac=0.4, restart_every=1500, seed=100)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_pert_p1b", t_global)

    # Phase 2: Bisection peak reduction
    print(f"\n=== Phase 2: Peak Reduction (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.22, 380.0)
    if t2 > 30:
        a_out, _ = bisection_peak_reduction(norm_sum1(best_seq), t2, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakred_p2", t_global)

    # Phase 3: LP Frank-Wolfe with constraint generation
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW CG (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30 and n_best <= 4000:
        a_out, _ = lp_frank_wolfe_cg(norm_sum1(best_seq), t3, solver, top_K=250)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_p3", t_global)

    # Phase 4: Fine SGD with higher top-K after LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Fine SGD (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t4,
                                           top_k=30, beta=2.0, c=5.0,
                                           exact_frac=0.5, restart_every=5000, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_fine_p4", t_global)

    # Phase 5: Second round of peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak Reduction 2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_out, _ = bisection_peak_reduction(norm_sum1(best_seq), t5, seed=333)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakred2_p5", t_global)

    # Phase 6: Multi-start
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-start (n={n_best}) ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 30:
        rng = np.random.default_rng(123)
        n_starts = 3
        t_per = max(20.0, t6 / n_starts)
        for si in range(n_starts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            a_pert = norm_sum1(best_seq) + rng.normal(0, 0.008, n_best)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_out, _ = nesterov_mirror_descent(a_pert, t_per,
                                               top_k=20, beta=2.5, c=4.0,
                                               exact_frac=0.5, restart_every=3000,
                                               seed=500 + si * 7)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"multistart_{si}", t_global)

    # Phase 7: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length Exploration ===")
    for delta in [200, -200, 400]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        new_n = n_best + delta
        if new_n < 1500 or new_n > 5000:
            continue
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t7 = min(remaining * 0.15, 120.0)
        if t7 > 20:
            a_out, _ = nesterov_mirror_descent(a_long, t7,
                                               top_k=15, beta=3.0, c=3.0,
                                               exact_frac=0.4, restart_every=2000, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"len_n{new_n}", t_global)

    # Phase 8: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 3500:
        t8 = min(remaining * 0.25, 150.0)
        if t8 > 20:
            a_out, _ = lp_frank_wolfe_cg(norm_sum1(best_seq), t8, solver, top_K=300)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 9: Final peak reduction polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Polish (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.4, 120.0)
        if t9 > 10:
            a_out, _ = bisection_peak_reduction(norm_sum1(best_seq), t9, seed=1234)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "polish_final", t_global)

    # Phase 10: One last SGD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final SGD ===")
        t10 = min(remaining * 0.5, 80.0)
        if t10 > 10:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t10,
                                               top_k=40, beta=2.0, c=5.0,
                                               exact_frac=0.6, restart_every=8000, seed=4321)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
