                    partner = k2i - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]
            
            # Find good destinations
            far_mask = np.abs(np.arange(n) - src) > n // 10
            candidates = np.where(far_mask & (sens < sens.mean()))[0]
            if len(candidates) == 0:
                candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            best_dst = candidates[np.argmin(sens[candidates])]
            if best_dst == src:
                continue
            
            for frac in [0.03, 0.01, 0.05, 0.015, 0.02, 0.04, 0.005, 0.025]:
                amt = src_val * frac
                if amt < 1e-14:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[best_dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-14:
                    continue
                a_t = a_t / s_t
                M_t = fast_autocorr(a_t).max()
                if M_t < best_m - 1e-14:
                    a = a_t
                    best_m = M_t
                    best_a = a.copy()
                    improvement = True
                    break
        
        if improvement:
            stall = 0
            moves += 1
            if verbose and moves % 30 == 0:
                print(f"    Coord ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, moves={moves}")
        else:
            stall += 1
            if stall % 5 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, 2.0 * n * best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    sf = f_interp.sum()
    if sf < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / sf
