def lp_remez_step(a, max_cons=50, r=0.1, solver=None):
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            coeffs = 2.0 * a[k - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k])

    constraints.append(a_var <= a * (1.0 + r) + r * 0.01)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), a.copy()
    return a_new / s, a.copy()

    """Optimize using trigonometric parameterization: a[i] = (1/n)(1 + sum alpha_m cos(2*pi*m*i/n + phi_m))."""
    n = len(a0)
    rng = np.random.default_rng(77)
    a = norm1(a0)
    best_a = a.copy()
    best_obj = eval_obj(a)
    t0 = time.monotonic()

    # Start with parameters fit to current a
    M = min(15, n // 10)
    # Fit initial parameters via DFT
    A = np.fft.rfft(a * n - 1.0)  # a*n - 1 has zero mean
    alpha = np.zeros((M, 2))
    for m in range(1, M + 1):
        if m < len(A):
            alpha[m - 1, 0] = 2.0 * np.real(A[m]) / M
            alpha[m - 1, 1] = 2.0 * np.imag(A[m]) / M

    # Gradient-free optimization (Nelder-Mead style with restarts)
    x_scale = 0.02
    for restart in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        if restart > 0:
            # Perturb parameters
            alpha_pert = alpha + rng.normal(0, x_scale * 0.5, alpha.shape)

        def build_seq(par):
            a_tmp = np.ones(n)
            for m in range(M):
                a_tmp += 2.0 * par[2 * m] * np.cos(2.0 * np.pi * (m + 1) * np.arange(n) / n + par[2 * m + 1])
            a_tmp = np.maximum(a_tmp, 0.0)
            s = a_tmp.sum()
            if s < 1e-15:
                return np.ones(n) / n
            return a_tmp / s

        # Simple coordinate descent on parameters
        par_init = alpha_pert.copy()
        par = par_init.copy()
        a_cur = build_seq(par)
        obj_cur = eval_obj(a_cur)

        for coord in range(len(par)):
            for step_frac in [0.05, 0.02, 0.01, 0.005, 0.002, -0.002, -0.005, -0.01, -0.02, -0.05]:
                par_try = par.copy()
                par_try[coord] += step_frac * x_scale
                a_try = build_seq(par_try)
                obj_try = eval_obj(a_try)
                if obj_try < obj_cur - 1e-10:
                    par = par_try
                    a_cur = a_try
                    obj_cur = obj_try
                    break

        if obj_cur < best_obj - 1e-10:
            best_obj = obj_cur
            best_a = a_cur.copy()
            alpha = par.copy()
            x_scale *= 0.8

        if verbose and restart % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Trig restart={restart} ({elapsed:.1f}s): obj={best_obj:.8f}")

    return best_a, best_obj

def local_search(a0, time_limit, verbose=False, rng_seed=555):
    """Random local search: try small random perturbations."""
    n = len(a0)
    a = norm1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        # Pick 2-5 random coordinates and perturb them
        n_pert = int(rng.integers(2, 6))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial = a.copy()
        for i in idx:
            a_trial[i] *= rng.uniform(0.85, 1.15)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        m_trial = fast_conv(a_trial).max()
        if m_trial < best_m - 1e-15:
            a = a_trial
            best_m = m_trial
            best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LS ({elapsed:.1f}s): J={2.0*n*best_m:.8f}")

    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=111):
    """Coordinate descent: try moving mass between pairs of coordinates."""
    n = len(a0)
    a = norm1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 50)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_trials = min(200, n // 10)
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    best_n = 0

    def try_update(a, label):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            best_n = len(a_norm)
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2600
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_a = p / p.sum()
            best_seq = parent_a.copy()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: NMD on parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.22, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: NMD n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = nesterov_mirror_descent(norm1(best_seq), t1, top_k=20, c0=3.5, verbose=True, rng_seed=42)
        try_update(a_out, f"NMD_{n_base}")

    # === Phase 2: Remez on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
        a_out, _ = remez_polish(norm1(best_seq), t2, solver=solver, r_init=0.08, verbose=True)
        try_update(a_out, f"Remez_{len(best_seq)}")

    # === Phase 3: NMD at n=3000 ===
    n3 = 3000
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: NMD n={n3} (budget={t3:.0f}s) ===")
        a_interp = interpolate_sequence_safe(best_seq, n3)
        a_out, _ = nesterov_mirror_descent(a_interp, t3, top_k=20, c0=3.0, verbose=True, rng_seed=100)
        try_update(a_out, f"NMD_{n3}")

    # === Phase 4: Remez at n=3000 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 150.0)
    if t4 > 30:
        print(f"\n=== Phase 4: Remez n={best_n} (budget={t4:.0f}s) ===")
        a_out, _ = remez_polish(norm1(best_seq), t4, solver=solver, r_init=0.06, verbose=True)
        try_update(a_out, f"Remez_{best_n}")

    # === Phase 5: NMD at n=3500 ===
    n5 = 3500
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 30:
        print(f"\n=== Phase 5: NMD n={n5} (budget={t5:.0f}s) ===")
        a_interp = interpolate_sequence_safe(best_seq, n5)
        a_out, _ = nesterov_mirror_descent(a_interp, t5, top_k=20, c0=3.0, verbose=True, rng_seed=200)
        try_update(a_out, f"NMD_{n5}")

    # === Phase 6: Trigonometric parameterization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 180.0)
    if t6 > 30 and best_n <= 3500:
        print(f"\n=== Phase 6: Trig opt n={best_n} (budget={t6:.0f}s) ===")
        a_out, _ = trig_param_optimize(norm1(best_seq), t6, verbose=True)
        try_update(a_out, f"Trig_{best_n}")

    # === Phase 7: NMD at n=4000 ===
    n7 = 4000
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.10, 180.0)
    if t7 > 30:
        print(f"\n=== Phase 7: NMD n={n7} (budget={t7:.0f}s) ===")
        a_interp = interpolate_sequence_safe(best_seq, n7)
        a_out, _ = nesterov_mirror_descent(a_interp, t7, top_k=25, c0=3.0, verbose=True, rng_seed=300)
        try_update(a_out, f"NMD_{n7}")

    # === Phase 8: Remez on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.08, 140.0)
    if t8 > 20:
        print(f"\n=== Phase 8: Remez final n={best_n} (budget={t8:.0f}s) ===")
        a_out, _ = remez_polish(norm1(best_seq), t8, solver=solver, r_init=0.04, verbose=True)
        try_update(a_out, f"Remez_final_{best_n}")

    # === Phase 9: NMD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.08, 140.0)
    if t9 > 20:
        print(f"\n=== Phase 9: NMD polish n={best_n} (budget={t9:.0f}s) ===")
        a_out, _ = nesterov_mirror_descent(norm1(best_seq), t9, top_k=10, c0=1.5, verbose=False, rng_seed=500)
        try_update(a_out, f"NMD_polish_{best_n}")

    # === Phase 10: Local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    t10 = min(remaining * 0.05, 80.0)
    if t10 > 15:
        print(f"\n=== Phase 10: Local search n={best_n} (budget={t10:.0f}s) ===")
        a_out, _ = local_search(norm1(best_seq), t10, verbose=True, rng_seed=666)
        try_update(a_out, f"LS_{best_n}")

    # === Phase 11: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t11 = min(remaining * 0.05, 80.0)
    if t11 > 15 and best_n <= 3500:
        print(f"\n=== Phase 11: CD n={best_n} (budget={t11:.0f}s) ===")
        a_out, _ = coordinate_descent(norm1(best_seq), t11, verbose=True, rng_seed=777)
        try_update(a_out, f"CD_{best_n}")

    # === Phase 12: Final Remez + NMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 12: Final push n={best_n} (budget={remaining:.0f}s) ===")
        t12a = min(remaining * 0.4, 100.0)
        if t12a > 15:
            a_out, _ = nesterov_mirror_descent(norm1(best_seq), t12a, top_k=8, c0=1.0, verbose=False, rng_seed=888)
            try_update(a_out, f"NMD_last_{best_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = remez_polish(norm1(best_seq), remaining - 5, solver=solver, r_init=0.02, verbose=False)
            try_update(a_out, f"Remez_last_{best_n}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]

def interpolate_sequence_safe(f, new_n):
    """Safe wrapper for interpolation."""
    try:
        return interpolate_seq(f, new_n)
    except Exception:
        return make_init(new_n, 'flat')
