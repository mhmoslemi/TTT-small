    A = np.fft.rfft(a, f1)
    c = np.fft.irfft(A * A, f1)[:L]
    cs = c - c.max()
    ev = np.exp(tau * cs)
    p = ev / ev.sum()
    # R_cross[i] = sum_{j=0}^{n-1} a[j] * p[j+i]  (linear cross-correlation)
    need = n + L - 1
    f2 = 1 << (need - 1).bit_length()
    Af = np.fft.rfft(a, f2)
    Pf = np.fft.rfft(p, f2)
    R = np.fft.irfft(np.conj(Af) * Pf, f2)[:n]
    g = 2.0 * R
    return g, c, p


def surrogate_value(c, tau):
    cmax = c.max()
    return cmax + (1.0 / tau) * float(np.log(np.exp(tau * (c - cmax)).sum()))


def _verify_grad_once():
    a = np.array([0.31, 0.17, 0.42, 0.09, 0.23, 0.35, 0.12])
    tau = 7.0
    g, c, p = smooth_grad(a, tau)
    n = len(a)
    L = 2 * n - 1
    gd = np.zeros(n)
    for i in range(n):
        for j in range(n):
            gd[i] += a[j] * p[j + i]
    gd = 2.0 * gd
    err = np.max(np.abs(g - gd))
    print("  grad self-check relerr=%.3e" % (err / max(1.0, np.max(np.abs(g)))))


def parametric_inits(n, extra):
    """Return list of (label, normalized candidate) at length n."""
    out = []
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    manu = {
        "sine": np.sin(np.pi * x) ** 2,
        "sine3": np.maximum(np.sin(np.pi * x) ** 2 + 0.05 * np.sin(3 * np.pi * x) ** 2, 0.0),
        "pow1": i + 1.0,
        "ipow1": (n - i) + 1.0,
        "pow2": (i + 1.0) ** 2,
        "tri": 1.0 - np.abs(2 * x - 1.0),
        "beta": np.maximum(i + 1.0, 0.5) ** 1.0 * np.maximum(n - i, 0.5) ** 1.0,
        "bell": np.exp(-((x - 0.35) ** 2) / 0.08) + np.exp(-((x - 0.65) ** 2) / 0.08),
        "sg": 1.0 / (1.0 + np.exp(-12.0 * (x - 0.4))),
        "flip": np.exp(-x) + np.exp((x - 1.0) * 2.0),
    }
    for k, v in manu.items():
        v = np.maximum(v, 0.0)
        if v.sum() > 1e-12:
            out.append((k, v / v.sum()))
    for c in extra:
        c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
        if c.sum() > 1e-12:
            out.append(("parent", c / c.sum()))
    return out


def smooth_optimize(a0, time_limit, tau0, tau1, verbose_idx, verbose):
    """Nesterov mirror descent on smoothed peak surrogate; track true max(c)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(1000 + verbose_idx)

    x = a.copy()
    x_prev = a.copy()
    timestep = 1.0
    stall = 0
    it = 0
    total_iters_target = max(2000, int(time_limit * 250))  # rough

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / float(total_iters_target))
        tau = tau0 + (tau1 - tau0) * frac

        # Nesterov extrapolation
        beta = (timestep - 1.0) / (timestep + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-14)
        y = y / y.sum()

        g, c, p = smooth_grad(y, tau)
        Sy = surrogate_value(c, tau)
        My = c.max()
        if My < best_m - 1e-15:
            best_m = My
            best_a = y.copy()

        # mirror descent step with line search on surrogate
        eta = 1.2
        logy = np.log(y)
        accepted = False
        for _ in range(10):
            xc = np.exp(logy - eta * g)
            xc = np.maximum(xc, 1e-14)
            xc = xc / xc.sum()
            _, cc, _ = smooth_grad(xc, tau)
            Sc = surrogate_value(cc, tau)
            if Sc <= Sy + 1e-12:
                accepted = True
                break
            eta *= 0.5
        if not accepted:
            # small randomized perturbation to leave a stuck point
            xc = x + rng.normal(0.0, 0.01, n) * x
            xc = np.maximum(xc, 1e-12)
            xc = xc / xc.sum()
        x_prev = x
        x = xc

        cx_val = fast_autocorr(x).max()
        if cx_val < best_m - 1e-15:
            best_m = cx_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            stall = 0
            xr = best_a + rng.normal(0.0, 0.02, n) * best_a
            xr = np.maximum(xr, 1e-12)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr.copy()

        timestep += 1.0
        if verbose and it % 400 == 0:
            el = time.monotonic() - t0
            print("    smooth[%d] (%.0fs): J=%.6f tau=%.0f" % (verbose_idx, el, 2.0 * n * best_m, tau))

    el = time.monotonic() - t0
    if verbose:
        print("    smooth[%d] done (%.0fs): J=%.6f iters=%d" % (verbose_idx, el, 2.0 * n * best_m, it))
    return best_a, best_m


def lp_frank_wolfe_cutting(a0, time_limit, solver, verbose_idx, initial_K=300, max_K=1200, verbose=True):
    """Bilinear Frank-Wolfe with adaptive cutting-plane constraint management."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    M_curr = ac.max()
    constraint_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    iter_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[j]) for j in keep)
            constraint_list = sorted(constraint_set)
        K = len(constraint_list)
        if K == 0:
            break

        Cm = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                Cm[jj, idx] = y[ki - idx]

        gv = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        prob = cp.Problem(cp.Minimize(Mv), [cp.sum(gv) == 1, Cm @ gv <= Mv])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(constraint_list[j]) for j in range(initial_K))
            continue
        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue
        gdir = np.maximum(np.asarray(gv.value, dtype=np.float64), 0.0)
        gs = gdir.sum()
        if gs < 1e-15:
            continue
        gdir = gdir / gs

        # constraint management: add most-violated lags of (y*gdir)
        size = 2 * n - 1
        fft = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft)
        Gf = np.fft.rfft(gdir, fft)
        conv_ys = np.fft.irfft(Yf * Gf, fft)[:size]
        viol = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(viol) > 0:
            amount = conv_ys[viol] - M_curr
            for k in viol[np.argsort(amount)[::-1][:30]]:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        ynew = (1.0 - alpha) * y + alpha * gdir
        ynew = np.maximum(ynew, 0.0)
        ys = ynew.sum()
        y = ynew / ys if ys > 1e-15 else y.copy()

        iter_count += 1
        if verbose and t % 25 == 0:
            el = time.monotonic() - t0
            print("    LP[%d] (%.0fs): J=%.6f |S|=%d t=%d" % (verbose_idx, el, 2.0 * n * best_m, len(constraint_set), t))

    el = time.monotonic() - t0
    if verbose:
        print("    LP[%d] done (%.0fs): J=%.6f iters=%d" % (verbose_idx, el, 2.0 * n * best_m, iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, seed, verbose):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        topk = np.argsort(ac)[::-1][:np.minimum(25, len(ac))]
        improved = False
        for k in topk[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            injk = np.zeros(n, dtype=bool)
            for k2 in topk[:6]:
                k2i = int(k2)
                injk[max(0, k2i - n + 1):min(n, k2i + 1)] = True
            cand = np.where(~ink if False else ~inkjk)[0]
            if len(cand) == 0:
                cand = np.arange(n)
            scores = np.zeros(n)
            for k2 in topk[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            dst = cand[np.argmin(scores[cand])]
            if dst == src:
                continue
            for frac in [0.01, 0.02, 0.04, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                if at.sum() < 1e-15:
                    continue
                at = at / at.sum()
                Mt = fast_autocorr(at).max()
                if Mt < best_m - 1e-16:
                    a = at
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            j = int(rng.integers(n))
            kk = int(rng.integers(n))
            if j != kk and a[j] > 1e-10:
                frac = rng.uniform(0.004, 0.02)
                amt = a[j] * frac
                a[j] -= amt
                a[kk] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1
            else:
                stall += 1
        if verbose and stall % 100 == 0:
            print("    coord (%.0fs): J=%.6f stall=%d" % (time.monotonic() - t0, 2.0 * n * best_m, stall))
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    return fi / ss if ss > 1e-15 else np.ones(new_n) / new_n


def _check_update(state, a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    an = a / s
    sc = evaluate_safe(an)
    if np.isfinite(sc) and sc < state["score"]:
        state["score"] = sc
        state["seq"] = an.copy()
        print("  *** NEW BEST[" + label + "]: %.6f  n=%d  t=%.0fs" % (sc, len(an), time.monotonic() - state["t0"]))


def propose_candidate(budget_s=1800.0):
    """Orchestrate the multi-stage search and return the best list of floats."""
    t0 = time.monotonic()
    budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: %.0fs" % budget)
    _verify_grad_once()

    state = {"score": float("inf"), "seq": None, "t0": t0}

    # ---- base length + extra initializations (parent) ----
    n0 = 3000
    extra = []
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            extra.append(p)
            pn = norm_sum1(p)
            sc = evaluate_safe(pn)
            if np.isfinite(sc):
                state["score"] = sc
                state["seq"] = pn.copy()
            print("Parent n=%d score=%.6f" % (len(p), sc))
    except Exception:
        pass

    if state["seq"] is None:
        a = np.sin(np.pi * np.linspace(0, 1, n0)) ** 2
        a = norm_sum1(a)
        state["seq"] = a
        state["score"] = evaluate_safe(a)
        print("Default init n=%d score=%.6f" % (len(a), state["score"]))

    # ---- build diverse initializations at n0, rank by score ----
    cands = parametric_inits(n0, extra)
    ranked = []
    for lab, c in cands:
        sc = evaluate_safe(c)
        if np.isfinite(sc):
            ranked.append((sc, lab, c))
    ranked.sort(key=lambda z: z[0])
    print(" %d parametric inits, best J=%.6f" % (len(ranked), ranked[0][0]))

    # ---- Phase 1: smooth-optimize top inits ----
    rem = budget - (time.monotonic() - t0)
    top_inits = [c[2] for c in ranked[:3]]
    n_top = len(top_inits)
    per = rem * 0.45 / n_top
    for idx, ain in enumerate(top_inits):
        rem = budget - (time.monotonic() - t0)
        if rem < 30:
            break
        budget_this = min(per, rem * 0.9)
        print(" Phase1 smooth init#%d (n=%d, budget=%.0fs)" % (idx, len(ain), budget_this))
        aout, _ = smooth_optimize(ain, budget_this, tau0=15.0, tau1=200000.0,
                                  verbose_idx=idx, verbose=True)
        _check_update(state, aout, "smooth%d_n%d" % (idx, len(ain)))

    # ---- Phase 2: LP-FW polish best ----
    rem = budget - (time.monotonic() - t0)
    nb = len(state["seq"])
    if rem > 60 and nb <= 3300:
        tl = min(rem * 0.30, 420.0)
        print(" Phase2 LP polish n=%d budget=%.0fs" % (nb, tl))
        aout, _ = lp_frank_wolfe_cutting(state["seq"], tl, solver, 99,
                                         initial_K=300, max_K=1200, verbose=True)
        _check_update(state, aout, "lp_n%d" % nb)

    # ---- Phase 3: smooth again from LP result (different tunables) ----
    rem = budget - (time.monotonic() - t0)
    nb = len(state["seq"])
    if rem > 60:
        tl = min(rem * 0.16, 200.0)
        print(" Phase3 smooth n=%d budget=%.0fs" % (nb, tl))
        aout, _ = smooth_optimize(state["seq"], tl, tau0=40.0, tau1=400000.0,
                                  verbose_idx=55, verbose=True)
        _check_update(state, aout, "smoothP3_n%d" % nb)

    # ---- Phase 4: coordinate polish ----
    rem = budget - (time.monotonic() - t0)
    nb = len(state["seq"])
    if rem > 30:
        tl = min(rem * 0.12, 120.0)
        print(" Phase4 coord n=%d budget=%.0fs" % (nb, tl))
        aout, _ = coordinate_polish(state["seq"], tl, seed=314, verbose=True)
        _check_update(state, aout, "coord_n%d" % nb)

    # ---- Phase 5: length sweep ----
    rem = budget - (time.monotonic() - t0)
    base = len(state["seq"])
    for new_n in [base + 500, base - 400, base + 1100]:
        rem = budget - (time.monotonic() - t0)
        if rem < 90 or new_n < 800:
            continue
        print(" Phase5 length n=%d (rem=%.0fs)" % (new_n, rem))
        a_long = interpolate_sequence(state["seq"], new_n)
        tl1 = min(rem * 0.35, 160.0)
        aout, _ = smooth_optimize(a_long, tl1, tau0=30.0, tau1=300000.0,
                                  verbose_idx=77, verbose=True)
        _check_update(state, aout, "smooth_n%d" % new_n)
        rem = budget - (time.monotonic() - t0)
        if rem > 60 and new_n <= 3300:
            tl2 = min(rem * 0.35, 180.0)
            aout, _ = lp_frank_wolfe_cutting(state["seq"], tl2, solver, 77,
                                             initial_K=250, max_K=1000, verbose=True)
            _check_update(state, aout, "lp_n%d" % len(state["seq"]))

    # ---- Final LP polish on global best ----
    rem = budget - (time.monotonic() - t0)
    nb = len(state["seq"]) if state["seq"] is not None else 0
    if rem > 50 and nb <= 3300:
        tl = min(rem * 0.5, 240.0)
        print(" Final LP n=%d budget=%.0fs" % (nb, tl))
        aout, _ = lp_frank_wolfe_cutting(state["seq"], tl, solver, 42,
                                         initial_K=400, max_K=1400, verbose=True)
        _check_update(state, aout, "lpfinal_n%d" % nb)

    if state["seq"] is not None:
        fs = evaluate_safe(state["seq"])
        if np.isfinite(fs):
            state["score"] = fs
        print("FINAL score=%.6f n=%d t=%.0fs" % (state["score"], len(state["seq"]), time.monotonic() - t0))
        return [float(x) for x in state["seq"]]

    # fallback
    fb = np.sin(np.pi * np.linspace(0, 1, n0)) ** 2
    fb = norm_sum1(fb)
    print("FALLBACK score=%.6f" % evaluate_safe(fb))
    return [float(x) for x in fb]
