def fast_autocorr(a):
    """Compute full convolution conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def exact_peak_subgradient(a, beta=20.0):
    """
    Compute subgradient of max(conv(a,a)) focusing on the peak lag(s).
    If one lag dominates, use exact gradient. Otherwise softmax over top lags.
    """
    n = len(a)
    ac = fast_autocorr(a)
    max_val = ac.max()
    
    # Find lags within epsilon of the max
    threshold = max_val - 1e-10 * max(max_val, 1e-30)
    peak_mask = ac >= threshold
    
    if peak_mask.sum() <= 3:
        # Use exact (or near-exact) subgradient at peak lags
        peak_indices = np.where(peak_mask)[0]
        # Equal weight among peak lags
        w = np.zeros(len(ac))
        w[peak_indices] = 1.0 / len(peak_indices)
    else:
        # Softmax over top lags
        top_k = min(50, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        shifted = top_vals - top_vals.max()
        w_top = np.exp(beta * shifted)
        w_top = w_top / w_top.sum()
        w = np.zeros(len(ac))
        w[top_idx] = w_top
    
    # Compute gradient: grad[j] = 2 * sum_m w[m] * a[m-j] for valid m-j
    # Using FFT
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev_padded = np.zeros(fft_size)
    a_rev_padded[:n] = a[::-1]
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    # Correction for the self-term (diagonal)
    # Actually, the factor of 2 is correct for the subgradient of the max
    # because c[k] = sum_i a[i]*a[k-i] and d(c[k])/d(a[j]) = 2*a[k-j] (for j != k-j)
    # The FFT computation gives sum_m w[m] * a[m-j], and we multiply by 2.
    # But for the self-term (m-j = j, i.e., m = 2j), we should only multiply by 1.
    # For simplicity and since this is a subgradient (not exact gradient), the factor of 2 is fine.
    
    return grad, max_val


def subgradient_optimize(a0, time_limit, verbose=True, seed=42):
    """
    Subgradient method with backtracking line search on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)

    t0 = time.monotonic()

    x = a.copy()
    iterations = 0
    step = 1.0 / np.sqrt(n)
    restart_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        
        # Compute subgradient
        g, M_curr = exact_peak_subgradient(x)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Backtracking line search
        # We want to minimize f(x) = max_k conv(x,x)[k]
        # Take step in direction -g, project onto simplex
        found_alpha = None
        alpha = step / np.sqrt(iterations)
        for _ in range(30):
            x_trial = x - alpha * g
            # Project onto simplex (sum=1, a>=0)
            x_trial = simplex_project(x_trial)
            M_trial = fast_autocorr(x_trial).max()
            if M_trial <= M_curr - 1e-14 * alpha * g_norm:
                found_alpha = alpha
                break
            alpha *= 0.7
        
        if found_alpha is None:
            # Use smallest alpha
            x_trial = x - alpha * g
            x_trial = simplex_project(x_trial)
            M_trial = fast_autocorr(x_trial).max()
            found_alpha = alpha
        
        x = x_trial
        step *= 0.95  # Gradually reduce step
        
        if M_trial < best_m - 1e-12:
            best_m = M_trial
            best_a = x.copy()
        
        # Periodic restart
        if iterations % 5000 == 0:
            restart_count += 1
            # Perturb around best
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = simplex_project(x_r)
            x = x_r
            step = 1.0 / np.sqrt(n)
        
        if verbose and iterations % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SG iter={iterations} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def simplex_project(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def make_power_law_init(n, exponent, side='right', seed=0):
    """Create a power-law initialization. Mass concentrated on one side."""
    i = np.arange(n)
    if side == 'right':
        # a[i] ∝ (i+1)^(-exponent) - decays from left to right
        a = (i + 1.0) ** (-exponent)
    else:
        # a[i] ∝ (n-i)^(-exponent) - decays from right to left
        a = (n - i) ** (-exponent)
    return norm_sum1(a)


def make_exp_decay_init(n, rate, side='right', seed=0):
    """Create an exponential decay initialization."""
    i = np.arange(n)
    if side == 'right':
        a = np.exp(-rate * i / n)
    else:
        a = np.exp(-rate * (n - 1 - i) / n)
    return norm_sum1(a)


def make_shifted_gaussian_init(n, center, width, seed=0):
    """Create a shifted Gaussian initialization."""
    i = np.arange(n)
    a = np.exp(-0.5 * ((i - center * n) / (width * n))**2)
    return norm_sum1(a)


def peak_shaving_descent(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent that specifically targets reducing the peak lag.
    When peak is at lag k, move mass from high-contributing positions to
    low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))
        
        # Find positions contributing to peak
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_lag - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            continue
            
        # Top contributors
        sorted_contrib = np.argsort(contrib)[::-1]
        
        improved = False
        for si in sorted_contrib[:5]:
            src = int(idx[si])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # For each source position, find best destination
            # We want to move mass to a position d that:
            # 1. Doesn't increase any other convolution values
            # 2. Reduces the peak
            # d should be far from positions that pair with it to create high values
            
            # Compute sensitivity: for each position d, how much would moving
            # mass there increase the convolution?
            n_top = min(20, 2*n-1)
            top_lags = np.argsort(ac)[::-1][:n_top]
            
            sens = np.zeros(n)
            for k2 in top_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 > e2:
                    continue
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]
            
            # Best destination: low sensitivity, not equal to src
            candidates = np.where((sens < sens.mean()) & (np.arange(n) != src))[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Also prefer positions that are far from src (to reduce the peak contribution)
            dist = np.abs(candidates.astype(float) - src)
            scores = sens[candidates] - 0.1 * dist / n
            best_dst = int(candidates[np.argmin(scores)])
            
            # Try transfer
            for frac in [0.01, 0.02, 0.005, 0.03, 0.015]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation every 5 stalls
            if stall % 5 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
        
        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PS stall={stall} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True, top_K=500):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    prev_y = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select binding constraints
        K_sel = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]
        T = np.zeros((K_sel, n))
        for col_idx, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            js = np.arange(j_start, j_end + 1)
            T[col_idx, js] = y[m_int - js]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        prev_y = y
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
