def compute_smoothed_grad(a, b, tau):
    """Compute gradient of tau * log(sum exp(b_k/tau)) w.r.t. a."""
    n = len(a)
    # Compute softmax weights
    b_max = b.max()
    log_weights = (b - b_max) / tau
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    weights /= weights.sum()
    
    # g_i = 2 * sum_j a_j * weights[i+j]
    # = 2 * (a * q)[2n-2-i] where q = weights[::-1]
    size_q = 2 * n - 1
    fft_size = 1 << (n + size_q - 1 - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(weights[::-1], fft_size)
    conv = np.fft.irfft(A * Q, fft_size)
    
    g = conv[n-1 : 2*n-1][::-1] * 2.0
    return g, weights

    n = len(a)

    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(b[active])[::-1]][:max_active]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        cp.sum(a_var) == 1,
        cp.norm1(a_var - a) <= rho,
    ]
    for k in active:
        k_i = int(k)
        j_start = max(0, k_i - n + 1)
        j_end = min(n - 1, k_i)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        coeffs = 2.0 * a[k_i - idx]
        constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), M
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), M
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), M
    a_new = a_new / s
    return a_new, float(M_var.value)

def remez_loop(a0, time_limit, solver=None, verbose=False, seed=99):
    """Iterative LP Remez with adaptive trust region and line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rho = 0.12
    no_improve = 0
    rng = np.random.default_rng(seed)
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        a_cand, _ = lp_remez_step(a, max_active=50, rho=rho, solver=solver)
        accepted = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1.0 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                rho = min(0.3, rho * 1.08)
                break
        if not accepted:
            no_improve += 1
            rho = max(0.01, rho * 0.7)
            if no_improve > 15:
                idx = rng.choice(n, size=min(n // 5, 500), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                rho = 0.1
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Remez t={t} ({elapsed:.1f}s): J={J:.8f}, rho={rho:.4f}")
    return best_a, best_m

def coord_descent_polish(a0, time_limit, verbose=False, seed=555):
    """Random pair coordinate descent with bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = max(30, n // 80)
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    if kind == 'triangle':
        a = np.maximum(1.0 - np.abs(t), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(np.pi * t)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * t ** 2)
    elif kind == 'quartic':
        a = np.maximum(1.0 - t ** 4, 0.0)
    elif kind == 'beta':
        a = np.maximum(1.0 - t ** 2, 0.0) ** 1.5
    elif kind == 'sech':
        a = 1.0 / np.cosh(3.0 * t)
    elif kind == 'sinusoid':
        a = np.abs(np.sin(np.pi * x)) ** 0.8
    else:
        a = np.maximum(1.0 - np.abs(t), 0.0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def cosine_shape_optimize(a0, n_terms=60, time_limit=60, verbose=False):
    """Optimize sequence shape using cosine basis parameterization."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    M = min(n_terms, n)
    C = np.zeros((n, M + 1))
    for m in range(M + 1):
        C[:, m] = np.cos(m * np.pi * x)
    c0, _, _, _ = np.linalg.lstsq(C, a, rcond=None)
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv(a).max()
    t_start = time.monotonic()
    lr = 0.05
    momentum = 0.9
    v = np.zeros(M + 1)
    for t in range(5000):
        if time.monotonic() - t_start > time_limit:
            break
        a_curr = np.maximum(C @ c0, 1e-15)
        a_curr = a_curr / a_curr.sum()
        b = fast_conv(a_curr)
        M_peak = b.max()
        J_curr = 2.0 * n * M_peak
        if J_curr < best_J:
            best_J = J_curr
            best_a = a_curr.copy()
        active_thresh = M_peak * 0.999
        active = b >= active_thresh
        if not active.any():
            active[int(np.argmax(b))] = True
        w = np.where(active, b, 0.0)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum
        # Gradient via same FFT trick
        size_w = 2 * n - 1
        fft_size = 1 << (n + size_w - 2).bit_length()
        A = np.fft.rfft(a_curr, fft_size)
        W = np.fft.rfft(w[::-1], fft_size)
        conv = np.fft.irfft(A * W, fft_size)
        grad_a = conv[n - 1:2 * n - 1][::-1] * 2.0
        grad_c = C.T @ grad_a
        g_norm = np.linalg.norm(grad_c)
        if g_norm < 1e-15:
            break
        v = momentum * v - lr * grad_c / g_norm
        c0 = c0 + v
        c0[0] = max(c0[0], 0.01)
        if t % 200 == 0 and verbose:
            print(f"      CosShape t={t}: J={best_J:.8f}, c0={c0[0]:.4f}")
    return best_a

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'raised_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Smoothed mirror descent from parent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: Smoothed MD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = smoothed_mirror_descent(norm_sum1(best_seq), t1, verbose=True, seed=42)
        try_update(a_out, f"SMD_parent_n{n0}")

    # === Phase 2: LP Remez on best (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_loop(norm_sum1(best_seq), t2, solver=solver, verbose=True, seed=99)
        try_update(a_out, f"Remez_n{len(best_seq)}")

    # === Phase 3: SMD with different temperature (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: SMD restart (budget={t3:.0f}s) ===")
    if t3 > 30:
        # Perturb in log space
        a_pert = best_seq * (1.0 + np.random.default_rng(7).normal(0, 0.05, n0))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = smoothed_mirror_descent(a_pert, t3, verbose=False, seed=777)
        try_update(a_out, f"SMD_perturb_n{n0}")

    # === Phase 4: Cosine shape optimization (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: Cosine shape (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out = cosine_shape_optimize(norm_sum1(best_seq), n_terms=50, time_limit=t4)
        try_update(a_out, f"CosShape_n{n0}")
        # Then run SMD from the cosine result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = smoothed_mirror_descent(a_out, min(remaining - 5, 60), verbose=False, seed=555)
            try_update(a_out2, f"SMD_from_cos_n{n0}")

    # === Phase 5: Multi-init SMD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5_total = min(remaining * 0.15, 180.0)
    if t5_total > 60:
        print(f"\n=== Phase 5: Multi-init SMD (budget={t5_total:.0f}s) ===")
        inits = [('sech', 111), ('gaussian', 222), ('beta', 333)]
        t_per = t5_total / len(inits)
        for kind, sd in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = make_init(n0, kind)
            a_out, _ = smoothed_mirror_descent(a_init, min(t_per, remaining - 5), verbose=False, seed=sd)
            try_update(a_out, f"SMD_{kind}_n{n0}")

    # === Phase 6: Alternate length (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n0 + 300
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        t6a = min(remaining * 0.5, 150.0)
        if t6a > 30:
            a_out, _ = smoothed_mirror_descent(a_alt, t6a, verbose=True, seed=888)
            try_update(a_out, f"SMD_n{n_alt}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = remez_loop(norm_sum1(best_seq), min(remaining - 5, 80), solver=solver, verbose=False, seed=444)
            try_update(a_out, f"Remez_{len(best_seq)}")

    # === Phase 7: Final polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        # LP Remez with small trust region
        t7a = min(remaining * 0.5, 100.0)
        if t7a > 15:
            a_out, _ = remez_loop(norm_sum1(best_seq), t7a, solver=solver, verbose=False, seed=1234)
            try_update(a_out, f"Remez_final_{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        # SMD with very small step
        if remaining > 15:
            a_out, _ = smoothed_mirror_descent(norm_sum1(best_seq), min(remaining - 5, 60), verbose=False, seed=4321)
            try_update(a_out, f"SMD_final_{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        # CD polish
        if remaining > 10:
            a_out, _ = coord_descent_polish(norm_sum1(best_seq), min(remaining - 3, 60), seed=999)
            try_update(a_out, f"CD_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
