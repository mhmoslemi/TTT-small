1. Start with a random non-negative sequence
2. Compute its FFT
3. In the frequency domain, "whiten" the power spectrum (make it as flat as possible)
4. Transform back, clip to non-negative, normalize
5. Repeat

This is similar to the "Gershberg-Saxton" or "error reduction" algorithm used in opticalee and astronomy.

Specifically:
1. a = initial (non-negative, sum=1)
2. A = FFT(a, N) where N >= 2n-1
3. P = |A|^2
4. c = IFFT(P) (real part, first 2n-1 elements)
5. c_target = 1/(2n-1) * ones(2n-1)  (flat target)
6. c_new = c + alpha * (c_target - c)  (move toward flat)
7. c_new = max(c_new, 0)  (autocorrelation must be non-negative for non-negative a)
8. P_new = FFT(c_new, N)  (this should be real and non-negative for a valid autocorrelation)
9. If P_new has negative values, clip them to 0
10. a_new = IFFT(sqrt(P_new) * phase(A))  (keep original phase, modify amplitude)
11. a_new = max(a_new, 0)  (non-negativity)
12. a_new = a_new / sum(a_new)  (normalize)
13. Set a = a_new, go to step 2

This is a Gerchberg-Saxton-style iteration. It alternates between:
- Frequency domain: enforce flat autocorrelation (by modifying the power spectrum)
- Time domain: enforce non-negativity (by clipping)

This might converge to a good local minimum.

import numpy as np

_global_solver = None

def get_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def evaluate_safe(a):
    try:

def smoothed_obj_grad(a, T):
    """
    Compute f_T(a) = (2n/T)*log(sum_k exp(T*c[k])) and its gradient.
    Assumes sum(a) = 1.
    Gradient: g[i] = 4n * sum_k w[k] * a[k-i] (valid index only)
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    S = exp_Tc.sum()
    w = exp_Tc / S
    obj = (2.0 * n / T) * (Tc_max + np.log(S))
    b = np.correlate(w, a, mode='valid')
    grad = 4.0 * n * b
    return obj, grad

def freq_depeak(a0, time_limit, gamma=0.02, verbose=False):
    """Frequency-domain targeted de-peak: reduce amplitude at peak autocorrelation frequency."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    N = 1 << (2 * n - 1).bit_length()
    n_bins = N // 2 + 1

    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, N)[:2 * n - 1]
        top_k = np.argsort(c)[::-1][:3]
        for k_max in top_k:
            bin_idx = int(k_max) % N
            if bin_idx < n_bins:
                A[bin_idx] *= (1.0 - gamma)
        a_new = np.fft.irfft(A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 50 == 0:
            print(f"    FreqDepeak it={it} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J

def nesterov_smoothed(a0, time_limit, T_start, T_end, verbose=False, seed=42):
    """
    Nesterov-accelerated projected gradient with adaptive softmax-smoothed objective.
    Temperature increases from T_start to T_end over the course of optimization.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    v = np.zeros(n)
    step = 1.0 / n
    mu = 0.85
    stall = 0
    total_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        total_iters += 1

        # Adaptive temperature: exponential schedule
        frac = min(1.0, t / 50000.0)
        T = T_start * (T_end / T_start) ** frac

        # Nesterov extrapolation
        beta_t = mu
        y = x + beta_t * (x - x_prev)
        y = norm_sum1(y)

        # Compute smoothed gradient at y
        f_y, g_y = smoothed_obj_grad(y, T)
        g_norm = np.linalg.norm(g_y)
        if g_norm < 1e-18:
            break

        # Backtracking line search (from x, using gradient at y)
        alpha = step
        found = False
        for _ in range(25):
            x_new = x - alpha * g_y
            x_new = norm_sum1(x_new)
            f_xnew = compute_J(x_new)
            if f_xnew <= f_y - 0.3 * alpha * g_norm * g_norm / n:
                found = True
                break
            alpha *= 0.5
        if not found:
            x_new = x - alpha * g_y
            x_new = norm_sum1(x_new)

        x_prev = x
        x = x_new
        f_x = compute_J(x)

        if f_x < best_J - 1e-12:
            best_J = f_x
            best_a = x.copy()
            stall = 0
            step = min(step * 1.02, 4.0 / n)
        else:
            stall += 1
            if stall % 10 == 0:
                step *= 0.7

        # Restart if stalled
        if stall > 3000:
            stall = 0
            sigma = 0.005 + 0.003 * (stall // 3000)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = norm_sum1(x_r)
            x = x_r
            x_prev = x_r.copy()
            step = 1.0 / n
            mu = 0.7 + 0.1 * rng.random()
            if verbose:
                print(f"    NS restart ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, T={T:.0f}, mu={mu:.2f}")

        if verbose and t % 20000 == 0:
            print(f"    NS t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, T={T:.0f}, step={step:.2e}")

    # Final exact evaluation
    best_J_final = compute_J(best_a)
    if best_J_final < best_J:
        best_J = best_J_final
    return best_a, best_J

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=800):
    """LP Frank-Wolfe with Nesterov acceleration and adaptive constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_fw = a.copy()

    for t in range(1, 500):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix (sparse for efficiency)
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g_dir

        sy = y_new.sum()
        if sy < 1e-15:

            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)

            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y.copy()
        y = y_new
        x_fw = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x_fw).max()
        if mx < best_m:
            best_m = mx
            best_a = x_fw.copy()

        if verbose and t % 5 == 0:
            print(f"    LP-FW it={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m

                        if J_trial < best_J - 1e-14:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
                            improved = True
                            break

        if verbose:
            print(f"    CoordPolish ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")

    return best_a, best_J

def make_symmetric_init(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)
