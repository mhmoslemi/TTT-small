    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def init_uniform(n):
    return np.ones(n) / n


def init_sine(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def init_hann(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a[0] = 0.001
    a[-1] = 0.001
    return norm_sum1(a)


def init_triangle(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2 * x - 1)
    return norm_sum1(a)


def init_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def init_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def init_spectral(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def init_random_positive(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.exponential(1.0, n)
    return norm_sum1(a)


def compute_soft_subgrad(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_md(a0, time_limit, top_k=60, beta=5.0, c0=5.0,
                 restart_every=5000, adaptive=True, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_k = top_k
    cur_beta = beta

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgrad(y, ac, top_k=cur_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if adaptive and stall > restart_every // 2:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_k = min(cur_k + 5, 2 * n - 1)

        if stall > restart_every:
            stall = 0
            noise_scale = 0.02 + 0.03 * (stall > restart_every * 3)
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_k = min(cur_k + 15, 2 * n - 1)
            cur_beta = max(cur_beta - 0.5, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, k={cur_k}, b={cur_beta:.1f}")

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={2.0*n*best_m:.6f}")

    return best_a, best_m


def fw_lp(a0, time_limit, solver, verbose=True, n_constraints_start=200,
          n_constraints_max=None, use_all=False, nesterov=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy() if nesterov else None
    max_constraints = 2 * n - 1 if n_constraints_max is None else min(n_constraints_max, 2 * n - 1)
    cur_n_con = n_constraints_start

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = max(ac_y.max(), 1e-15)

        if use_all or cur_n_con >= max_constraints:
            binding_idx = np.arange(2 * n - 1)
        else:
            margins = M_curr - ac_y
            binding_idx = np.argsort(margins)[:cur_n_con]

        # Build constraint matrix efficiently
        # (y * b)[k] = sum_j y_j * b_{j+k} for k < n, and sum_j y_{j+k} * b_j for k >= n
        # Actually: conv(y, b)[k] = sum_j y_j * b_{j+k}
        # We need: for each k in binding_idx, sum_j y_j * b_{j+k} <= M_curr
        # This is: (T_k) @ b <= M_curr where T_k[j] = y_j if 0 <= j+k < n else 0
        # i.e., T_k[j] = y[j] for j in [max(0,k-n+1), min(n-1, n-1-k)]... 
        # Actually simpler: conv(y,b)[k] = sum_{j} y[j] * b[j+k] where j+k is in [0,n-1]
        # = sum_{j=max(0,k-n+1)}^{min(n-1-k+n-1, n-1)} ... 
        # Let me just use: for lag k, valid j are those where 0<=j<n and 0<=j+k<n
        # i.e., j in [max(0, -k), min(n-1, n-1-k)] for k>=0
        # For k < n: j in [0, n-1-k], so row is y[0:n-k] padded
        # For k >= n: j in [k-n+1, n-1], so row is zeros + y[k-n+1:n]

        # Build sparse-ish representation (dense is fine for n <= 5000)
        # Actually for speed, let's just form the matrix directly
        n_con = len(binding_idx)
        # T[i, j] = y[j] if (j + binding_idx[i]) < n else 0
        # More precisely: conv(y,b)[k] = sum_{j=0}^{n-1} y[j]*b[j+k] where j+k < n
        # So the coefficient of b[m] in constraint k is y[m-k] if 0<=m-k<n, i.e., k<=m<n+k
        # Wait, let me be more careful.
        # np.convolve(y, b)[k] = sum_j y[j] * b[k-j] (full convolution)
        # No: np.convolve(a, b)[n] = sum_m a[m] * b[n-m]
        # So np.convolve(y, b)[k] = sum_j y[j] * b[k-j] where 0<=j<n and 0<=k-j<n
        # = sum_j y[j] * b[k-j] for j in [max(0,k-n+1), min(k, n-1)]
        # The coefficient of b[m] (where m = k-j, so j = k-m) is y[k-m] if 0<=k-m<n
        # i.e., coefficient of b[m] in row k is y[k-m] if 0 <= k-m < n, i.e., k-n < m <= k
        
        # For LP: minimize 0 subject to T @ b <= M_sel, b >= 0, maximize sum(b)
        # T[i, m] = y[binding_idx[i] - m] if 0 <= binding_idx[i] - m < n else 0
        
        # This is a banded matrix! For each row i (lag k), the non-zero entries are
        # columns m in [max(0, k-n+1), min(n-1, k)]
        
        # For efficiency with CVXPY, let's just build it as a dense matrix for n up to ~5000
        # Actually that's 8000 x 5000 = 40M entries which is 320MB... too much.
        # Let's use the structure differently.
        
        # Alternative: use the fact that each constraint is a dot product of y (shifted) with b.
        # We can express this as: for lag k, the constraint is y_shifted[k] @ b <= M
        # where y_shifted[k][m] = y[m-k] if valid.
        
        # For CVXPY, let's just form the product y @ b via the convolution structure.
        # Actually, the simplest approach: just use the full matrix for moderate n.
        # For n <= 3000, the matrix is at most 6000 x 3000 = 18M floats = 144MB. OK.
        # For larger n, we subsample constraints.
        
        if n <= 4000:
            T = np.zeros((n_con, n))
            for i, k in enumerate(binding_idx):
                ki = int(k)
                # constraint: sum_j y[j]*b[ki-j] <= M for valid j
                # b[m] appears when m = ki - j, i.e., j = ki - m
                # valid: 0 <= ki-m < n => ki-n < m <= ki
                m_start = max(0, ki - n + 1)
                m_end = min(n - 1, ki)
                if m_start <= m_end:
                    T[i, m_start:m_end+1] = y[m_end - (m_start - (ki - n + 1)):][::-1] if False else y[ki - np.arange(m_start, m_end+1)]
            # Wait, let me redo this.
            # T[i, m] should = y[j] where j = ki - m (the index in y)
            # So T[i, m] = y[ki - m] for m in [max(0, ki-n+1), min(n-1, ki)]
            T = np.zeros((n_con, n))
            for i, k in enumerate(binding_idx):
                ki = int(k)
                m_start = max(0, ki - n + 1)
                m_end = min(n - 1, ki)
                if m_start <= m_end:
                    ms = np.arange(m_start, m_end + 1)
                    T[i, m_start:m_end+1] = y[ki - ms]
        else:
            # For large n, use fewer constraints
            cur_n_con = min(cur_n_con, 500)
            binding_idx = binding_idx[:cur_n_con]
            n_con = len(binding_idx)
            T = np.zeros((n_con, n))
            for i, k in enumerate(binding_idx):
                ki = int(k)
                m_start = max(0, ki - n + 1)
                m_end = min(n - 1, ki)
                if m_start <= m_end:
                    ms = np.arange(m_start, m_end + 1)
                    T[i, m_start:m_end+1] = y[ki - ms]

        M_sel = M_curr * np.ones(n_con)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            cur_n_con = max(50, cur_n_con // 2)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if nesterov:
            if t == 1:
                x = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
                x = x_new

        y = y_new
        iter_count_real = t

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if nesterov and x is not None:
            mx = fast_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}, n_con={len(binding_idx)}")

        # Grow constraints
        if not use_all and t % 10 == 0:
            cur_n_con = min(cur_n_con + 30, max_constraints)

        # After enough iterations, switch to all constraints
        if not use_all and t > 100 and cur_n_con < max_constraints:
            cur_n_con = min(cur_n_con * 1.5, max_constraints)
            if cur_n_con >= max_constraints and n <= 3000:
                use_all = True

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    T_start = 0.01
    T_end = 0.0001
    n_iter = 50000
    top_k = 10

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature
        frac = it / n_iter
        T = T_start * (T_end / T_start) ** frac

        # Generate perturbation: pick a few random pairs and redistribute
        n_moves = rng.integers(1, 6)
        a_new = a.copy()
        for _ in range(n_moves):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac_move = rng.uniform(0.001, 0.02)
            amt = a_new[i] * frac_move
            if amt < 1e-15:
                continue
            a_new[i] -= amt
            a_new[j] += amt
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        delta = M_new - best_m

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA it={it} ({elapsed:.0f}s): J={J:.6f}, T={T:.6f}")

    return best_a, best_m


def peak_descent(a0, time_limit, verbose=True, rng_seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_peaks = min(8, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_peaks]

        # Sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            best_local = np.argmax(contrib)
            src = idx[best_local]
            partner = (ki - idx)[best_local]

            if a[src] < 1e-12:
                continue

            # Choose which to reduce
            if sens[src] >= sens[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner

            # Find best destination
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]

            if dst == reduce_idx:
                dst = candidates[np.argmin(cand_sens[1:])] + (1 if np.argmin(cand_sens) > 0 else 0)
                if dst == reduce_idx:
                    continue

            found = False
            for frac in [0.01, 0.005, 0.02, 0.003, 0.03, 0.05]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-10:
                    frac = rng.uniform(0.003, 0.015)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random kick
            n_pert = min(3, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

            return

            return

        sc = evaluate_safe(a_norm)

    t_phase1 = min(remaining * 0.20, 350.0)

        labels.append("u2.5")

        inits.append(init_hann(n0))
        labels.append("hann")

        if parent is not None:
            for mix in [0.3, 0.7]:
                blend = mix * norm_sum1(parent) + (1 - mix) * init_u_shape(n0, 2.0)
                inits.append(norm_sum1(blend))
                labels.append(f"blend_{mix}")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 40)], 2))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  MD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_md(init, t_avail, top_k=50, beta=5.0, c0=5.0,
                                    restart_every=4000, adaptive=True,
                                    verbose=False, rng_seed=42 + idx * 7)
            try_update(a_out, f"md_{lbl}")

    # === Phase 2: LP Frank-Wolfe (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP FW (n={n_best}) ===")
    t2 = min(remaining * 0.22, 400.0)
    if t2 > 50 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        n_con_start = min(300, 2 * n_best - 1)
        n_con_max = min(2000, 2 * n_best - 1) if not use_all else 2 * n_best - 1
        a_out, _ = fw_lp(a_in, t2, solver, verbose=True,
                         n_constraints_start=n_con_start,
                         n_constraints_max=n_con_max,
                         use_all=use_all, nesterov=True)
        try_update(a_out, "fw_main")

    # === Phase 3: MD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MD polish (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_md(a_in, t3, top_k=80, beta=7.0, c0=4.0,
                                restart_every=6000, adaptive=True,
                                verbose=False, rng_seed=200)
        try_update(a_out, "md_polish")

    # === Phase 4: LP FW round 2 with all constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP FW round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2000)
        n_con_start = min(500, 2 * n_best - 1)
        n_con_max = 2 * n_best - 1 if use_all else min(3000, 2 * n_best - 1)
        a_out, _ = fw_lp(a_in, t4, solver, verbose=True,
                         n_constraints_start=n_con_start,
                         n_constraints_max=n_con_max,
                         use_all=use_all, nesterov=True)
        try_update(a_out, "fw_r2")

    # === Phase 5: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Simulated Annealing (n={n_best}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = simulated_annealing(a_in, t5, verbose=True, rng_seed=456)
        try_update(a_out, "sa")

    # === Phase 6: Peak descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Peak Descent (n={n_best}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_descent(a_in, t6, verbose=True, rng_seed=789)
        try_update(a_out, "peak")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length Exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_seq(best_seq, new_n)

        t7a = min(remaining * 0.35, 180.0)
        if t7a > 25:
            a_out, _ = nesterov_md(a_long, t7a, top_k=50, beta=5.0, c0=4.0,
                                    restart_every=5000, adaptive=True,
                                    verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 3000:
            t7b = min(remaining * 0.35, 120.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                a_in_long = interpolate_seq(a_in, new_n)
                use_all = (new_n <= 2000)
                a_out, _ = fw_lp(a_in_long, t7b, solver, verbose=False,
                                 n_constraints_start=min(200, 2*new_n-1),
                                 n_constraints_max=min(1500, 2*new_n-1),
                                 use_all=use_all, nesterov=True)
                try_update(a_out, f"fw_n{new_n}")

    # === Phase 8: Final LP on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t8 = min(remaining * 0.45, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 2500)
            n_con_start = min(400, 2 * n_best - 1)
            n_con_max = 2 * n_best - 1 if use_all else min(3500, 2 * n_best - 1)
            a_out, _ = fw_lp(a_in, t8, solver, verbose=True,
                             n_constraints_start=n_con_start,
                             n_constraints_max=n_con_max,
                             use_all=use_all, nesterov=True)
            try_update(a_out, "fw_final")

    # === Phase 9: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final Polish (n={n_best}) ===")
        t9a = min(remaining * 0.4, 60.0)
        if t9a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_md(a_in, t9a, top_k=100, beta=8.0, c0=3.0,
                                    restart_every=3000, adaptive=True,
                                    verbose=False, rng_seed=999)
            try_update(a_out, "md_final")

        remaining = time_budget - (time.monotonic() - t_global)
        t9b = min(remaining * 0.4, 40.0)
        if t9b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_descent(a_in, t9b, verbose=False, rng_seed=1234)
            try_update(a_out, "peak_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = init_sine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
