import numpy as np

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def compute_subgradient_topk(a, conv, top_k=100, beta=8.0):
    """Compute soft subgradient of max(conv) at a using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_end, i_start - 1, -1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def compute_subgradient_full(a, conv, beta=4.0):
    """Compute subgradient using all lags with softmax weighting (slower but more complete)."""
    n = len(a)
    conv_len = 2 * n - 1
    vals = np.maximum(conv, 1e-30)
    w = vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(conv_len) / conv_len
    else:
        w = w / w_sum

    g = np.zeros(n)
    for k in range(conv_len):
        wi = w[k]
        if wi < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_end, i_start - 1, -1)
            j_idx = k - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def projection_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1.0
    rho = mu > cv / np.arange(1, n + 1)
    if not rho.any():
        return np.ones(n) / n
    rho_hat = np.max(np.where(rho)[0]) + 1
    theta = (mu[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def gradient_projection(a0, time_limit, momentum=0.9, verbose=False, seed=0):
    """Gradient descent with simplex projection and momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    vel = np.zeros(n)
    eta = 0.005
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        # Use top-200 lags for gradient
        g = compute_subgradient_topk(a, conv, top_k=200, beta=6.0)
        # Remove mean (for the sum constraint)
        g = g - g.mean()

        vel = momentum * vel - eta * g
        a_new = projection_simplex(a + vel)

        J_new = 2.0 * n * fast_conv(a_new).max()
        if J_new < best_J - 1e-15:
            a = a_new
            best_J = J_new
            best_a = a.copy()
            stall = 0
            eta = min(eta * 1.05, 0.05)
        else:
            stall += 1
            eta = eta * 0.9
            if stall % 50 == 0:
                # Random perturbation
                pert = rng.normal(0, 0.01, n) * a
                a_p = a + pert
                a_p = np.maximum(a_p, 0.0)
                sp = a_p.sum()
                if sp > 1e-10:
                    a_p /= sp
                    J_p = eval_J(a_p)
                    if J_p < best_J:
                        a = a_p
                        best_J = J_p
                        best_a = a.copy()
                        stall = 0
                        eta = max(eta * 1.5, 0.005)

        if verbose and stall % 100 == 0:
            print(f"    GP ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}, eta={eta:.4f}")

    return best_a, best_J


def mirror_descent(a0, time_limit, top_k=80, beta_w=10.0, c0=4.0,
                   restart_every=5000, verbose=False, use_backtrack=True):
    """Nesterov-accelerated mirror descent with backtracking and adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w
    cur_c0 = c0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Subgradient
        conv = fast_conv(y)
        if t % 2000 == 0:
            g = compute_subgradient_full(y, conv, beta=cur_beta * 0.5)
        else:
            g = compute_subgradient_topk(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        # Step size with backtracking
        eta = cur_c0 / (np.sqrt(t) + 1.0)
        max_backtracks = 5

        for bt in range(max_backtracks):
            log_r = np.log(y) - eta * g
            log_r -= log_r.max()
            a_new = np.exp(log_r)
            a_new = np.maximum(a_new, 1e-30)
            a_new /= a_new.sum()

            val_new = eval_J(a_new)

            if not use_backtrack or val_new < best_val or bt == 0:
                break
            eta *= 0.5

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        if val_new < best_val - 1e-16:
            best_val = val_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart logic
        if stall > restart_every:
            stall = 0
            pert_type = rng.integers(3)
            if pert_type == 0:
                # Geometric perturbation
                scale = rng.uniform(0.005, 0.03)
                pert = rng.normal(0, scale, n) * best_a
                x_r = best_a + pert
            elif pert_type == 1:
                # Multiplicative perturbation
                log_pert = rng.normal(0, rng.uniform(0.05, 0.15), n)
                x_r = best_a * np.exp(log_pert)
            else:
                # Localized perturbation
                idx = rng.integers(0, n, size=max(1, n // 10))
                x_r = best_a.copy()
                x_r[idx] *= rng.uniform(0.7, 1.3, size=len(idx))
            x_r = np.maximum(x_r, 1e-12)
            sr = x_r.sum()
            if sr > 1e-10:
                x_r /= sr
            else:
                x_r = best_a.copy()
            x = x_r.copy()
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 10, 250)
            cur_beta = max(cur_beta - 0.5, 3.0)
            cur_c0 = max(cur_c0 * 0.92, 0.8)
            if verbose:
                print(f"    MD restart t={t}: J={best_val:.6f}, k={cur_top_k}, b={cur_beta:.1f}, c={cur_c0:.2f}")

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with pressure-based mass transfer and try-optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Compute pressures for top lags
        n_lags = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        # Source: highest pressure * mass
        valid = (pressure > 1e-15) & (a > 1e-12)
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        # Destination: minimize impact
        dest_scores = np.zeros(n)
        for k in top_lags[:6]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            for i in range(i_s, i_e + 1):
                j = k_i - i
                if 0 <= j < n:
                    dest_scores[i] += a[j]
        dest_scores[src] = np.inf
        dest_scores += 0.15 * pressure
        dst = int(np.argmin(dest_scores))

        # Try multiple transfer fractions
        improved = False
        for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.08, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t /= st
            J_t = 2.0 * n * fast_conv(a_t).max()
            if J_t < best_J - 1e-15:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random multi-perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.003, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 200 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_ramp(n, p=1.0, direction=0):
    """Power-law ramp. direction=0: increasing, 1: decreasing, 2: asymmetric."""
    x = np.arange(1, n + 1, dtype=np.float64)
    if direction == 0:
        a = x ** p
    elif direction == 1:
        a = x ** (-p)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(p * (x - 0.3)) * np.exp(-2.0 * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal_asym(n, peak1=0.15, peak2=0.75, w1=40.0, w2=25.0, ratio=1.5):
    """Asymmetric bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = ratio * np.exp(-w1 * (x - peak1) ** 2) + np.exp(-w2 * (x - peak2) ** 2) + 0.01
    return a / a.sum()


def make_exponential(n, alpha=0.5, direction=0):
    """Exponential decay."""
    x = np.arange(n, dtype=np.float64)
    if direction == 0:
        a = np.exp(-alpha * x / max(1, n // 10))
    else:
        a = np.exp(-alpha * (n - 1 - x) / max(1, n // 10))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
        a += coeff * 0.5 * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_poly(n, degree=3, seed=0):
    """Random polynomial (positive on [0,1])."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for d in range(1, degree + 1):
        coeff = rng.normal(0, 0.3 / (d + 1))
        a += coeff * x ** d
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 1e-15)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    parent_n = 3200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            parent_n = len(parent)
            pJ = eval_J(parent)
            print(f"Parent: n={parent_n}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
        print(f"Starting from parent: n={parent_n}, J={best_score:.6f}")
    else:
        n0 = 3000
        best_a = make_ramp(n0, 1.0, direction=0)
        best_score = eval_J(best_a)
        parent_n = n0
        print(f"Initial (no parent): n={n0}, J={best_score:.6f}")

    n_cur = len(best_a)

    # === Phase 1: Multi-init mirror descent (diverse asymmetric starts) ===
    print(f"\n=== Phase 1: MD Multi-Init (n={n_cur}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
        # Blend parent with other shapes
        b1 = 0.6 * parent + 0.4 * make_ramp(n_cur, 1.5, 0)
        inits.append(b1 / b1.sum())
        labels.append("parent_ramp")
        b2 = 0.7 * parent + 0.3 * make_exponential(n_cur, 2.0, 0)
        inits.append(b2 / b2.sum())
        labels.append("parent_exp")

    inits.append(make_ramp(n_cur, 0.5, 0))
    labels.append("ramp0.5")
    inits.append(make_ramp(n_cur, 1.0, 0))
    labels.append("ramp1.0")
    inits.append(make_ramp(n_cur, 1.5, 0))
    labels.append("ramp1.5")
    inits.append(make_ramp(n_cur, 0.7, 2))
    labels.append("asym_ramp")
    inits.append(make_exponential(n_cur, 1.0, 0))
    labels.append("exp1.0")
    inits.append(make_exponential(n_cur, 2.0, 1))
    labels.append("exp2.0r")
    inits.append(make_bimodal_asym(n_cur, 0.1, 0.8, 50, 30, 2.0))
    labels.append("bimodal")
    inits.append(make_smooth_random(n_cur, seed=7))
    labels.append("smooth7")
    inits.append(make_poly(n_cur, degree=4, seed=3))
    labels.append("poly4")

    # Deduplicate
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n_cur // 50)], 3))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    t_per = max(12.0, remaining * 0.30 / max(1, len(inits)))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 120:
            break
        t_avail = min(t_per, time_budget - 120 - elapsed)
        if t_avail < 8:
            break
        print(f"  MD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = mirror_descent(init, t_avail, top_k=80, beta_w=10.0, c0=4.0,
                                   use_backtrack=True)
        try_update(a_out, f"md_{lbl}")

    # === Phase 2: Extended MD on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended MD (n={len(best_a)}, J={best_score:.6f}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_out, _ = mirror_descent(best_a.copy(), t2, top_k=120, beta_w=7.0,
                                  c0=3.5, restart_every=7000, use_backtrack=True)
        try_update(a_out, "md_ext")

    # === Phase 3: Gradient projection ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Gradient Projection (n={len(best_a)}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = gradient_projection(best_a.copy(), t3, momentum=0.85, verbose=True, seed=13)
        try_update(a_out, "grad_proj")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent (n={len(best_a)}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 15:

    # === Phase 5: MD after CD (different params) ===

    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        a_out, _ = mirror_descent(best_a.copy(), t5, top_k=150, beta_w=5.0,
                                   c0=3.0, restart_every=8000, use_backtrack=True)
        try_update(a_out, "md_after_cd")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = sorted(set([
        max(1000, n_best - 600),
        max(1000, n_best - 300),
        n_best + 200,
        n_best + 500,
        n_best + 1000,
        2000, 2500, 3000, 3500, 4000, 4500, 5000,
    ]))

    for nl in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        if abs(nl - n_best) <= 30:
            continue
        print(f"  Length sweep: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 10:
            a_out, _ = mirror_descent(a_int, t_md, top_k=100, beta_w=8.0, c0=3.5,
                                       use_backtrack=True)
            try_update(a_out, f"len{nl}_md")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and len(best_a) != nl:
            t_gp = min(remaining * 0.10, 40.0)
            if t_gp > 10:
                a_cur = interpolate_seq(best_a, nl)
                a_out, _ = gradient_projection(a_cur, t_gp, momentum=0.8, seed=nl)
                try_update(a_out, f"len{nl}_gp")

    # === Phase 7: Final MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_f = len(best_a)
    print(f"\n=== Phase 7: Final MD Polish (n={n_f}, J={best_score:.6f}) ===")
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 20:
        a_out, _ = mirror_descent(best_a.copy(), t7, top_k=200, beta_w=4.0,
                                   c0=2.5, restart_every=12000, use_backtrack=True,
                                   verbose=True)
        try_update(a_out, "final_md")

    # === Phase 8: Final gradient projection ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final GP (n={len(best_a)}) ===")
    t8 = min(remaining * 0.20, 180.0)
    if t8 > 15:
        a_out, _ = gradient_projection(best_a.copy(), t8, momentum=0.9,
                                        verbose=True, seed=999)
        try_update(a_out, "final_gp")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final CD (n={len(best_a)}) ===")
    t9 = min(remaining * 0.20, 150.0)
    if t9 > 10:
        a_out, _ = coordinate_descent(best_a.copy(), t9, verbose=True, seed=314)
        try_update(a_out, "final_cd")

    # === Phase 10: One more MD if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 10: Extra MD ===")
        a_out, _ = mirror_descent(best_a.copy(), remaining - 8,
                                   top_k=250, beta_w=3.0, c0=2.0,
                                   restart_every=15000, use_backtrack=True)
        try_update(a_out, "extra_md")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_ramp(n_fb, 1.0, 0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
