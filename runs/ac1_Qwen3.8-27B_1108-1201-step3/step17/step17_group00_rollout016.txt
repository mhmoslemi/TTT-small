def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_local(a):

def lp_fw_step(y, constraint_set, solver, max_K=3000):
    """
    Perform one LP-FW step. Returns (g_dir, M_opt, success).
    y: current point (sum=1, non-negative)
    constraint_set: set of lag indices to include
    """
    n = len(y)
    S = sorted(constraint_set)
    if len(S) > max_K:
        # Prune: calculate margins
        ac = fast_autocorr(y)
        M = ac.max()
        margins = np.array([M - ac[k] for k in S if 0 <= k < len(ac)])
        # Keep the tightest ones
        valid_S = np.array(S)
        valid_margins = np.array([M - ac[k] for k in valid_S if 0 <= k < len(ac)])
        # This might not align, let me redo
        valid_S = np.array([k for k in S if 0 <= k < len(ac)])
        if len(valid_S) == 0:
            return None, M, False
        margins = M - ac[valid_S]
        keep = np.argsort(margins)[:max_K]
        S = sorted(int(valid_S[i]) for i in keep)
    
    K = len(S)
    if K == 0:
        return None, fast_autocorr(y).max(), False
    
    # Build constraint matrix
    # Row k: (A_k)_i = y[k-i] if 0 <= k-i < n, else 0
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(S):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, fast_autocorr(y).max(), False
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, fast_autocorr(y).max(), False
    
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, fast_autocorr(y).max(), False
    
    M_opt = float(M_var.value) if M_var.value is not None else fast_autocorr(y).max()
    g_dir = g_val / g_sum
    return g_dir, M_opt, True

            no_improve = 0

            continue

def power_sum_smoothed_amd(a0, time_limit, verbose=True):
    """Accelerated mirror descent with power-sum smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    x_prev = a.copy()
    
    p = 20.0  # Power for smoothing
    eta0 = 5.0  # Initial step size
    restart_every = 3000
    
    stall = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Power-sum gradient
        # grad_j = sum_k ac_k^(p-1) / sum_m ac_m^p * (a_{k-j} + a_{j-k})
        # Use only top-K lags for efficiency
        K = min(50, len(ac))
        top_idx = np.argsort(ac)[::-1][:K]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        weights = top_vals ** (p - 1)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(K) / K
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            k_int = int(k)
            # grad_j += wi * (y_{k-j} + y_{j-k})
            # y_{k-j} is non-zero when 0 <= k-j < n, i.e., k-n < j < k
            lo1 = max(0, k_int - n + 1)
            hi1 = min(n, k_int + 1)
            if lo1 < hi1:
                idx = np.arange(lo1, hi1)
                grad[idx] += wi * y[k_int - idx]
            
            lo2 = max(0, k_int - n + 1)  # same range since it's symmetric
            # Actually for the second term y_{j-k}, non-zero when 0 <= j-k < n, i.e., k < j < k+n
            lo2 = max(0, k_int)
            hi2 = min(n, k_int + n)
            if lo2 < hi2:
                idx = np.arange(lo2, hi2)
                grad[idx] += wi * y[idx - k_int]
        
        # Step size with decay
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent update (exponentiated gradient)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive p: increase as we converge
        if t % 1000 == 0:
            p = min(p * 1.5, 200.0)
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = np.maximum(best_a + pert, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, p={p:.0f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    AMD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_M

                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
        
        # Strategy 2: Random perturbation
        if not improved or stall > 50:
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t < 1e-15:
                    continue
                a = a / s_t
                M_new = fast_autocorr(a).max()
                if M_new < best_M - 1e-16:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

def interpolate_to_length(a, new_n):
    """Interpolate sequence to a new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / s

def parametric_initialization(n, time_budget, best_init, best_score):
    """Try parametric families for initialization."""
    t0 = time.monotonic()
    i = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    
    best_a = best_init.copy()
    best_val = best_score
    
    candidates = []
    # Power functions
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i + 1.0) ** p)
        candidates.append((n - i) ** p)
    # Sigma function
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for s in [5, 10, 20, 50, 100]:
            candidates.append(1.0 / (1.0 + np.exp(-s * (x - c))))
    # Sine
    for f in [1, 2, 3]:
        candidates.append(np.maximum(np.sin(np.pi * f * x) ** 2, 0.01))
    # Two-peak
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sig in [0.05, 0.1, 0.15]:
                candidates.append(np.exp(-((x-c1)**2)/(2*sig*sig)) + np.exp(-((x-c2)**2)/(2*sig*sig)))
    # Exponential
    for r in [0.5, 0.7, 0.9]:
        candidates.append(r ** i)
        candidates.append(r ** (n - 1 - i))
    
    for cand in candidates:
        if time.monotonic() - t0 > time_budget:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        val = 2.0 * n * fast_autocorr(cand).max()
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    
    return best_a, best_val
