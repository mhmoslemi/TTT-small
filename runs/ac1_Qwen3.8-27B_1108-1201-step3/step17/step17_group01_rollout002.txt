import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum to 1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=50, temperature=4.0):
    """Compute weighted subgradient of max(conv(a,a)) using softmax over top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weights with temperature
    max_v = top_vals.max()
    if max_v < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = np.exp(temperature * (top_vals - max_v) / max_v)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad

def fista_mirror(a0, time_limit, top_k=50, temperature=4.0, gamma=3.0,
                  restart_every=2000, verbose=False, lower_bound=None):
    """FISTA with KL mirror descent for minimizing max(conv(a,a)) on the simplex.
    
    Uses Polyak step size when lower_bound is provided.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    if lower_bound is None:
        lower_bound = 1.0  # conservative
    
    x = a.copy()
    x_prev = a.copy()
    momentum = 0.0
    
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # FISTA momentum
        if t > 1:
            momentum = (1 + np.sqrt(1 + 4 * momentum * momentum)) / 2
            y = x + ((momentum - 1) / momentum) * (x - x_prev)
        else:
            y = x.copy()
        np.maximum(y, 1e-15, out=y)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=top_k, temperature=temperature)
        
        # Step size: Polyak or sqrt schedule
        cur_f = ac.max()
        if lower_bound < cur_f:
            g_norm_sq = np.dot(g, g)
            if g_norm_sq > 1e-20:
                eta = (cur_f - lower_bound) / g_norm_sq
                eta = min(eta, 1.0)  # Cap step size
            else:
                eta = gamma / (np.sqrt(t) + 1.0)
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror step (exponential / KL)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
            # FISTA restart if stalled too long
            if stall > restart_every:
                # Reset momentum
                x_prev = x.copy()
                momentum = 0.0
                stall = 0
                # Small perturbation
                pert = rng.normal(0, 0.01, n) * best_a
                x = best_a + pert
                x = np.maximum(x, 1e-10)
                x = x / x.sum()
                x_prev = x.copy()
                top_k = min(top_k + 5, 100)
                temperature = max(temperature - 0.2, 1.5)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, top_k={top_k}, temp={temperature:.1f}")
    
    return best_a, best_m

def peak_surgery(a0, time_limit, verbose=False, rng_seed=42):
    """Targeted peak reduction by moving mass from high-contribution positions to low-contribution ones."""

    a = norm_sum1(a0)
    best_a = a.copy()

    t0 = time.monotonic()

    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find positions contributing to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Source: position with highest contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Compute "cost" of each position (how much it contributes to all top peaks)
            costs = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                i_s2 = max(0, k2i - n + 1)
                i_e2 = min(n - 1, k2i)
                if i_s2 > i_e2:
                    continue
                idx2 = np.arange(i_s2, i_e2 + 1)
                c2 = a[idx2] * a[k2i - idx2]
                costs[idx2] += c2
            
            # Total contribution at each position
            # For position i, its total contribution across all top peaks is:
            # sum over k: a[i]*a[k-i] (if valid)
            # = a[i] * sum over k: a[k-i] (where k-i is valid)
            # This is approximately a[i] * (sum of a in a window around i)
            
            # Find best destination: minimize increase in max peak
            # Heuristic: positions with low "cost" are good destinations
            candidate_mask = np.ones(n, dtype=bool)
            candidate_mask[src] = False
            candidate_candidates = np.where(candidate_mask)[0]
            
            # Sort by cost (ascending) and pick among the lowest-cost, lowest-value positions
            # Use a composite score
            score = costs + 0.1 * a  # prefer low cost AND low current value

            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:

                        break
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not found and stall > 20:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:

                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Surgery ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def spectral_equalize(a0, time_limit, n_iters=500, alpha=0.5, verbose=False):
    """Frequency-domain phase optimization to flatten the spectrum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Target: flatten the power spectrum
        # Compress dynamic range
        eta = alpha * (1.0 + 0.5 * it / max(1, n_iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)
        
        # Update: keep phase, adjust magnitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
