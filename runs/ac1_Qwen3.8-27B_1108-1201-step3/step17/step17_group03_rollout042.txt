Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient coordinate descent with line search
2. LP-based Frank-Wolfe for periodic large moves
3. Spectral diversification for escaping local minima
4. Multi-length exploration

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")

            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe evaluation using the actual evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Fast coordinate descent that identifies peak contributions and
    moves mass from high-contribution to low-contribution elements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        
        # Find the top peaks
        # Focus on the global max and nearby peaks
        peak_k = int(np.argmax(ac))
        
        # Compute contribution of each element to the peak
        # c[peak_k] = sum_i a[i] * a[peak_k - i]
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        
        # Also consider the weighted sum of top few peaks
        # For efficiency, focus on the main peak
        
        # Find elements with high contribution (to decrease)
        # and elements with low contribution (to increase)
        n_decrease = min(5, len(contrib))
        n_increase = min(5, len(contrib))
        
        top_contrib = np.argsort(contrib)[::-1][:n_decrease]
        low_contrib = np.argsort(contrib)[:n_increase]
        
        improved = False
        
        # Try moving mass from high-contrib to low-contrib
        for ti in top_contrib:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[ti])
            if a[src] < 1e-12:
                continue
            
            for tj in low_contrib:
                if tj == ti:
                    continue
                dst = int(idx[tj])
                
                # Line search: find best fraction to move
                best_frac = 0
                best_trial_m = M
                
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_frac = frac
                
                if best_frac > 0:
                    amt = a[src] * best_frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            # Try random moves
            src = int(rng.integers(n))
            if a[src] > 1e-10:
                dst = int(rng.integers(n))
                if dst != src:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = best_a.copy()  # Revert
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD final ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

If we move amount `amt` from `a[src]` to `a[dst]`, the change in c[k] is:

This is O(n) per lag, but we only need to check the peak lag and a few nearby lags.

Define f(a) = max_k c[k] where c = conv(a,a). This is a convex function (max of linear functions of a⊗a... wait, it's not linear in a).

Actually, c[k] = sum_i a[i]*a[k-i] is quadratic in a. So f(a) = max_k (quadratic form) is NOT convex.

But we can use the Dinkelbach parameterization: minimize f(a) subject to sum(a)=1, a>=0.

Or: for a target value T, check if there exists a with f(a) <= T. This is a feasibility problem:
- a >= 0
- sum(a) = 1
- for all k: sum_i a[i]*a[k-i] <= T

This is a non-convex feasibility problem (quadratic constraints).

However, if we fix all variables except one, say a[j], then each constraint becomes a quadratic in a[j]:
sum_i a[i]*a[k-i] = a[j]*(a[k-j] + a[j-k]) + (terms not involving a[j]) <= T

This is linear in a[j]! So for a fixed set of other variables, the feasible range of a[j] is an interval.

This suggests a coordinate-wise approach:
1. Fix all variables except a[j]
2. Compute the feasible range of a[j] (intersection of all constraint intervals)
3. Set a[j] to the value in its feasible range that minimizes the max violation
4. Repeat for all j

This is essentially a "cyclic coordinate descent" on the peak constraint!

Let me formalize:
- Given a[0..n-1] with a[j] fixed for i ≠ j
- For each lag k, the constraint c[k] <= T becomes:
  a[j] * (a[k-j] + a[j-k]) + c_k_without_j <= T
  where c_k_without_j = c[k] - a[j]*(a[k-j] + a[j-k]) (the part not involving a[j])
  
  If a[k-j] + a[j-k] > 0: a[j] <= (T - c_k_without_j) / (a[k-j] + a[j-k])
  If a[k-j] + a[j-k] < 0: a[j] >= (T - c_k_without_j) / (a[k-j] + a[j-k])
  If a[k-j] + a[j-k] = 0: need c_k_without_j <= T (no constraint on a[j] from this k)

So the feasible range of a[j] is:
a[j] ∈ [max(0, lower_bounds), min(upper_bounds)]

where upper_bounds and lower_bounds come from the various k's.

Then the optimal a[j] for minimizing the peak is:
- If we want to minimize max_k c[k], and c[k] is linear in a[j] for fixed other variables...
- The peak as a function of a[j] is: max_k (c_k_without_j + a[j]*(a[k-j] + a[j-k]))
- This is the max of linear functions of a[j], hence convex in a[j]
- The minimum is where the subgradient contains 0, i.e., where the maximum is "flattest"

This is a 1D convex optimization problem! We can solve it with bisection or ternary search.

        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=5e-4, min_margin_frac=1e-5):
    """
    Frank-Wolfe with LP subproblem to find direction that reduces max autocorrelation.
    Adaptively tightens the active constraint margin.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Select active constraints: lags where ac is close to max
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        n_active = (margins <= threshold).sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(margins <= threshold)[0]

        K = len(active_idx)

        # Build constraint matrix: for each active lag k, constraint is
        # sum_i y[i] * b[k-i] <= M_curr  (linear in b)
        # T[row, i] = y[k - i] for valid i
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_sel @ b_var <= M_curr])

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_mom = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten margin on improvement
            margin_frac = max(margin_frac * 0.9, min_margin_frac)

        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.1e}")

    return best_a, best_m


def greedy_peak_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Greedy coordinate descent: at each step, find the (src, dst) pair
    that most reduces the global peak of the autocorrelation.
    Uses fast O(1) peak change estimation for candidate screening,
    then full FFT verification.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    verify_every = 5

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        if stall % verify_every == 0:
            ac = fast_autocorr(a)
            M = ac.max()
            if M < best_m - 1e-14:
                best_m = M
                best_a = a.copy()
                stall = 0
            peak_k = int(np.argmax(ac))
        else:
            peak_k = int(np.argmax(fast_autocorr(a)))

        # Contributions to the peak lag
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        if len(idx) == 0:
            stall += 1
            continue

        # contrib[i] = a[idx[i]] * a[peak_k - idx[i]]
        paired = a[peak_k - idx]
        contrib = a[idx] * paired

        # Find best move: reduce from high contrib, add to low contrib
        n_cand = min(8, len(contrib))
        src_ranks = np.argsort(contrib)[::-1][:n_cand]
        dst_ranks = np.argsort(contrib)[:n_cand]

        best_improvement = 0
        best_move = None

        for si in src_ranks:
            src = int(idx[si])
            if a[src] < 1e-12:
                continue
            for di in dst_ranks:
                dst = int(idx[di])
                if dst == src:
                    continue

                # Fast estimate: delta_peak = -amt*paired[src] + amt*paired_dst
                # where paired_dst = a[peak_k - dst]
                paired_src_val = a[peak_k - src] if 0 <= peak_k - src < n else 0.0
                paired_dst_val = a[peak_k - dst] if 0 <= peak_k - dst < n else 0.0

                if paired_src_val <= paired_dst_val:
                    continue

                # Try fractions
                for frac in [0.1, 0.3, 0.5]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    # Quick check: only if this clearly helps
                    est_delta = -amt * paired_src_val + amt * paired_dst_val
                    if est_delta >= -1e-14:
                        continue
                    if -est_delta > best_improvement:
                        best_improvement = -est_delta
                        best_move = (src, dst, amt)

        if best_move is not None:
            src, dst, amt = best_move
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = best_a.copy()
                stall += 1
                continue
            a = a / s_a

            # Verify with full autocorrelation
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
            else:
                # Revert
                a[dst] -= amt
                a[src] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall += 1
        else:
            # No good move found: try random perturbation
            n_pert = min(5, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    break
                else:
                    a[j] -= amt
                    a[i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
            else:
                stall += 1

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GPD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    GPD final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def multi_peak_descent(a0, time_limit, verbose=True, rng_seed=99):
    """
    Coordinate descent that considers multiple peaks simultaneously.
    Uses weighted subgradient to balance reduction across top peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    top_k = 10
    beta_exp = 4.0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0

        # Weighted subgradient from top peaks
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_exp
        w = w / w.sum()

        # Compute gradient direction: for each element, its "damage" weighted by peaks
        damage = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    damage[i] += 2.0 * wi * a[j]

        # Move mass from high-damage to low-damage
        n_move = min(5, n)
        src_idx = np.argsort(damage)[::-1][:n_move]
        dst_idx = np.argsort(damage)[:n_move]

        improved = False
        for si in src_idx:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(si)
            if a[src] < 1e-12:
                continue
            for di in dst_idx:
                dst = int(di)
                if dst == src:
                    continue
                for frac in [0.05, 0.15, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a < 1e-15:
                        break
                    a = a / s_a
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                    else:
                        a[dst] -= amt
                        a[src] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random restart from best with perturbation
                pert = rng.normal(0, 0.01, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
            else:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MPD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MPD final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def spectral_perturb(a, strength=0.02, rng=None):
    """Perturb the phase of the FFT to get a new diverse starting point."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A) + rng.normal(0, strength, len(np.angle(A)))
    A_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def triangle_perturb(a, rng, strength=0.05):
    """Add a small triangular bump at a random location."""
    n = len(a)
    center = int(rng.integers(n // 4, 3 * n // 4))
    width = max(1, n // 20)
    bump = np.zeros(n)
    for i in range(max(0, center - width), min(n, center + width + 1)):
        bump[i] = 1.0 - abs(i - center) / width
    a_new = a + strength * strength * bump * a.sum()
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_limit = float(budget_s) - 45.0  # Buffer
    solver = get_solver()
    print(f"Total time budget: {time_limit:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = np.maximum(p, 0.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_fb = 3600
        best_seq = make_init(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW main optimization ===
    remaining = time_limit - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW main (n={n0}) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out, _ = lp_frank_wolfe(best_seq, t1, solver, verbose=True,
                                   top_K=250, margin_frac=2e-3,
                                   min_margin_frac=5e-5)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 2: Multi-peak coordinate descent ===
    remaining = time_limit - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-peak CD (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 30:
        a_out, _ = multi_peak_descent(best_seq, t2, verbose=True, rng_seed=77)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "mpd1")

    # === Phase 3: LP-FW with tighter margins ===
    remaining = time_limit - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP-FW tight (n={len(best_seq)}) ===")
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and n_cur <= 5000:
        a_out, _ = lp_frank_wolfe(best_seq, t3, solver, verbose=True,
                                   top_K=350, margin_frac=5e-4,
                                   min_margin_frac=1e-5)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 4: Greedy peak descent ===
    remaining = time_limit - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Greedy peak descent (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = greedy_peak_descent(best_seq, t4, verbose=True, rng_seed=55)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "gpd")

    # === Phase 5: Try longer sequences ===
    remaining = time_limit - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    longer_targets = sorted(set([n_cur + 500, n_cur + 1000, n_cur + 2000]))
    longer_targets = [lt for lt in longer_targets if lt <= 6000]
    print(f"\n=== Phase 5: Longer sequences ===")

    for new_n in longer_targets[:2]:
        remaining = time_limit - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"  Trying n={new_n} (remaining {remaining:.0f}s)")
        a_long = interpolate_seq(best_seq, new_n)

        # Quick LP-FW on longer sequence
        t5a = min(remaining * 0.25, 200.0)
        if t5a > 40 and new_n <= 5000:
            a_out, _ = lp_frank_wolfe(a_long, t5a, solver, verbose=False,
                                       top_K=200, margin_frac=2e-3,
                                       min_margin_frac=1e-4)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

        remaining = time_limit - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.15, 100.0)
            if t5b > 20:
                a_out, _ = greedy_peak_descent(best_seq, t5b, verbose=False, rng_seed=33)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"gpd_n{len(best_seq)}")

    # === Phase 6: Spectral diversification + re-optimize ===
    remaining = time_limit - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Spectral diversify (n={len(best_seq)}) ===")
    rng = np.random.default_rng(2024)
    n_trials = 3
    time_per_trial = min(remaining * 0.10, 120.0)

    for trial in range(n_trials):
        remaining = time_limit - (time.monotonic() - t_global)
        if remaining < time_per_trial + 30:
            break

        # Different perturbation strengths
        strengths = [0.01, 0.03, 0.05]
        a_pert = spectral_perturb(best_seq, strength=strengths[trial % 3], rng=rng)

        # Then optimize the perturbed version
        t6 = min(remaining * 0.12, 120.0)
        if t6 > 20:
            n_pert = len(a_pert)
            if n_pert <= 5000:
                a_out, _ = lp_frank_wolfe(a_pert, t6, solver, verbose=False,
                                           top_K=150, margin_frac=5e-3,
                                           min_margin_frac=1e-4)
            else:
                a_out, _ = greedy_peak_descent(a_pert, t6, verbose=False, rng_seed=trial + 100)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"spec{trial}")

    # === Phase 7: Final LP polish ===
    remaining = time_limit - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_cur}) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 30 and n_cur <= 5000:
        a_out, _ = lp_frank_wolfe(best_seq, t7, solver, verbose=True,
                                   top_K=400, margin_frac=2e-4,
                                   min_margin_frac=1e-6)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final coordinate descent polish ===
    remaining = time_limit - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final CD polish (n={len(best_seq)}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 15:
            a_out, _ = multi_peak_descent(best_seq, t8, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_init(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 20:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
