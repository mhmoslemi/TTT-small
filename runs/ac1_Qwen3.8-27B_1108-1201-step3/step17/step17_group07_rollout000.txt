def compute_subgrad_full(y, ac, threshold_frac=0.9):
    """Compute subgradient using all lags above threshold. O(n * n_active)."""
    n = len(y)
    M = ac.max()
    threshold = M * threshold_frac
    active = np.where(ac >= threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= len(active)
    return g

def compute_topk_subgrad(y, ac, top_k):
    """Compute subgradient using top-k lags. O(n*k)."""
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def lp_remez_single(a, active_set, max_cons, r, solver=None):
    """Single LP Remez step with given active set."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    if len(active_set) > max_cons:
        active_set = np.sort(active_set)[-max_cons:]
    if len(active_set) == 0:
        active_set = np.array([int(np.argmax(b))])
    
    # Use scipy linprog for speed
    # Variables: a_0, ..., a_{n-1}, M (last variable)
    # Minimize M
    # Constraints:
    #   sum(a) = 1
    #   For each active k: sum_j coeffs_j * a_j - M <= b[k]
    #   a_j >= 0
    #   a_j <= a_j * (1+r)
    
    n_vars = n + 1  # a_0..a_{n-1}, M
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality constraints: A_ub @ x <= b_ub
    rows_ub = []
    b_ub = []
    
    for k in active_set:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            row = np.zeros(n_vars)
            row[idx] = coeffs
            row[-1] = -1.0  # -M
            rows_ub.append(row)
            b_ub.append(b[k_int])
    
    # Upper bounds: a_j <= a_j*(1+r)
    for j in range(n):
        row = np.zeros(n_vars)
        row[j] = 1.0
        rows_ub.append(row)
        b_ub.append(a[j] * (1.0 + r))
    
    A_ub = np.array(rows_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_j >= 0, M unbounded below (but we minimize it)
    bounds = [(0.0, None)] * n + [(-np.inf, None)]
    
    try:
        result = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        try:
            result = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds, method='highs')
        except Exception:
            return a.copy()
    
    if result.status != 0:
        return a.copy()
    
    a_new = np.maximum(result.x[:n], 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def get_active_set(b, max_b, n_total, top_count=50):
    """Get active set: lags near the maximum."""
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > top_count:
        # Keep the top ones
        order = np.argsort(b)[-top_count:]
        active = np.sort(order)
    return active

    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 80000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Active set: near-max lags + shadow lags
        active = get_active_set(b, max_b, n, top_count=40)
        # Add shadow constraints
        shadow_thresh = max_b * shadow_frac
        shadow = np.where(b >= shadow_thresh)[0]
        all_constraints = np.unique(np.concatenate([active, shadow]))
        if len(all_constraints) > 80:
            all_constraints = np.sort(all_constraints)[-80:]
        
        a_cand = lp_remez_single(a, all_constraints, max_cons=80, r=r)
        
        # Line search with exact objective
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = max(fast_conv(a_trial))
            if m_trial < best_m - 1e-20:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.2, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.002, r * 0.6)
            if no_improve > 15:
                # Perturb to escape local minimum
                rng = np.random.default_rng(t * 7 + 3)
                idx = rng.choice(n, size=min(n // 4, 300), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = max(r, 0.05)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.5f}, no_imp={no_improve}")
    
    return best_a, best_m

def remez_tight_polish(a0, time_limit, r_start=0.01, verbose=False):
    """Tight LP Remez for final polishing with very small trust regions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_start
    no_improve = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Use only the top few lags for tight polish
        active = get_active_set(b, max_b, n, top_count=30)
        
        a_cand = lp_remez_single(a, active, max_cons=30, r=r)
        
        accepted = False
        for t_frac in [1.0, 0.7, 0.4, 0.2]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = max(fast_conv(a_trial))
            if m_trial < best_m - 1e-20:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.05, r * 1.02)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.0005, r * 0.7)
            if no_improve > 20:
                break
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    TightRemez t={t} ({elapsed:.1f}s): J={J:.10f}, r={r:.6f}")
    
    return best_a, best_m

def pair_coordinate_descent(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Bisection to find best transfer amount
            lo, hi = 0.0, a[i] * 0.5
            best_frac = 0.0
            best_m_local = best_m
            
            for _ in range(8):
                mid = (lo + hi) / 2
                amt = a[i] * mid
                if amt < 1e-15:
                    hi = mid
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    hi = mid
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = max(fast_conv(a_trial))
                if m_trial < best_m_local - 1e-18:
                    best_m_local = m_trial
                    best_frac = mid
                    a = a_trial
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 0:
                best_m = best_m_local
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def single_coordinate_descent(a0, time_limit, verbose=False, rng_seed=456):
    """Single coordinate descent: try adjusting one coordinate at a time."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 20)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_coords = min(300, n)
        
        for _ in range(n_coords):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            if a[i] < 1e-14:
                continue
            
            # Try scaling a[i] by factor
            for factor in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = max(fast_conv(a_trial))
                if m_trial < best_m - 1e-18:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):

    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.10f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'raised_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: NMD from parent (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)

        return [1.0]
    
    return [float(x) for x in best_seq]
