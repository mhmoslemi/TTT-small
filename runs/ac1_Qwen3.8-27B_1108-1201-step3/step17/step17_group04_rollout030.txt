            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            T_sel[row_idx, i_start:i_end] = y[i_start:i_end]
            # Also the other side for symmetry in convolution
            # Actually for linear conv: C_k = sum_j y_j * y_{j+k}
            # dC_k/db_i = y_{i+k} (if valid) + y_{i-k} (if valid)
            # But we want T @ b to approximate C when b = y
            # So T[k, i] = y_{i+k} + y_{i-k} (boundary handled)
            # Actually the standard formulation: C_k = sum_i y_i * y_{i+k}
            # dC_k/db_i = y_{i+k} + y_{i-k}
            # And T[k,i] * b_i should sum to C_k when b = y
            # sum_i T[k,i] * y_i = sum_i (y_{i+k} + y_{i-k}) * y_i
            # = sum_i y_i * y_{i+k} + sum_i y_i * y_{i-k}
            # = C_k + C_{-k} = C_k + C_k (by symmetry of autocorr) ... no
            # Actually C_k = sum_i y_i * y_{i+k} and C_{-k} = C_k for autocorrelation
            # Wait, linear conv: (y*y)[k] = sum_i y[i]*y[k-i] for k=0..2n-2
            # This is symmetric: (y*y)[k] = (y*y)[2n-2-k]
            # d(y*y)[k]/dy[i] = y[k-i] + y[i-k] (when valid)
            # Let's just use the correct Jacobian
            
        # Redo with correct Jacobian
        T_sel = np.zeros((K_sel, n))
        for row_idx, k in enumerate(binding_idx):
            k_int = int(k)
            # (conv)[k] = sum_i a[i]*a[k-i] where both in [0,n-1]
            # d/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            for j in range(n):
                val = 0.0
                ki = k_int - j
                if 0 <= ki < n:
                    val += y[ki]
                jk = j - k_int
                if 0 <= jk < n:
                    val += y[jk]
                T_sel[row_idx, j] = val

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel] if False else [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        prev_y = y
        y = y_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_symmetric_init(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)

def make_power_law_init(n, power=0.5):
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)

def make_gaussian_edges(n, sigma=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)

def genetic_combine(pool, time_limit, n_sgd_iters=2000, verbose=False):
    """Combine pairs from pool and polish."""
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    best_a = pool[0][0].copy()
    best_J = pool[0][1]
    n = len(best_a)

    while time.monotonic() - t0 < time_limit:
        if len(pool) < 2:
            break
        # Pick two random solutions
        i1 = int(rng.integers(len(pool)))
        i2 = int(rng.integers(len(pool)))
        if i1 == i2:
            continue
        a1, _ = pool[i1]
        a2, _ = pool[i2]
        # Ensure same length
        if len(a1) != len(a2):
            a2 = interpolate_sequence(a2, len(a1))
        t = rng.uniform(0.2, 0.8)
        candidate = t * a1 + (1 - t) * a2
        candidate = norm_sum1(candidate)
        # Brief SGD polish
        a_out, J_out = nesterov_sgd(candidate, min(5.0, time_limit - (time.monotonic() - t0)),
                                     top_k=20, beta=2.5, c=4.0, exact_frac=0.5,
                                     restart_every=1000, seed=int(rng.integers(100000)))
        if J_out < best_J:
            best_J = J_out
            best_a = a_out.copy()
            # Add to pool
            pool.append((a_out.copy(), J_out))
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Genetic ({elapsed:.0f}s): J={best_J:.6f}")
        # Keep pool size bounded
        if len(pool) > 10:
            pool.sort(key=lambda x: x[1])
            pool = pool[:10]

    return best_a, best_J

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Nesterov SGD from multiple inits (25% time) ===
    print(f"\n=== Phase 1: Nesterov SGD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_symmetric_init(n0))
        labels.append("symmetric")
        inits.append(make_power_law_init(n0, power=0.5))
        labels.append("power0.5")
        inits.append(make_power_law_init(n0, power=1.0))
        labels.append("power1.0")
        inits.append(make_gaussian_edges(n0, sigma=0.12))
        labels.append("gauss_edges")
        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_symmetric_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.05:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(15.0, t_phase1 / n_inits)
        pool = []

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  SGD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, J_out = nesterov_sgd(init, t_avail, verbose=False,
                                         top_k=15, beta=3.0, c=3.0,
                                         exact_frac=0.4, restart_every=2000,
                                         seed=42 + idx)
            try_update(a_out, f"sgd_{lbl}")
            pool.append((a_out.copy(), J_out))

    # === Phase 2: Coordinate Descent (25% time) ===
    print(f"\n=== Phase 2: Coordinate Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_optimize(a_in, t2, n_coords_per_batch=300,
                                          n_grid=50, verbose=True, seed=99)
        try_update(a_out, f"coord_n{len(a_in)}")
        if len(pool) > 0:
            pool.append((a_out.copy(), compute_J(a_out)))

    # === Phase 3: LP Frank-Wolfe (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, top_K=500, verbose=True)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: More Coordinate Descent (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent round 2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_optimize(a_in, t4, n_coords_per_batch=500,
                                          n_grid=60, verbose=True, seed=200)
        try_update(a_out, f"coord2_n{len(a_in)}")

    # === Phase 5: Genetic combination (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Genetic recombination ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20 and len(pool) >= 2:
        pool.sort(key=lambda x: x[1])
        a_out, _ = genetic_combine(pool, t5, verbose=True)
        try_update(a_out, "genetic")

    # === Phase 6: Multi-length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length ===")
    lengths_to_try = [n_best + 400, n_best + 800, n_best - 400]
    lengths_to_try = [l for l in lengths_to_try if 1000 <= l <= 5000]

    for new_n in lengths_to_try[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.3, 120.0)
        if t6a > 15:
            a_out, _ = nesterov_sgd(a_long, t6a, verbose=False,
                                     top_k=20, beta=2.5, c=4.0,
                                     exact_frac=0.5, restart_every=2000, seed=555)
            try_update(a_out, f"sgd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.3, 80.0)
            if t6b > 10:
                a_out, _ = coord_descent_optimize(norm_sum1(best_seq) if len(norm_sum1(best_seq)) == new_n else a_long,
                                                   t6b, n_coords_per_batch=200,
                                                   n_grid=40, verbose=False, seed=333)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 7: Final polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 30:
        t7a = min(remaining * 0.5, 200.0)
        if t7a > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_optimize(a_in, t7a, n_coords_per_batch=400,
                                              n_grid=50, verbose=True, seed=777)
            try_update(a_out, f"coord_final_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.5, 100.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = nesterov_sgd(a_in, t7b, verbose=False,
                                         top_k=30, beta=2.0, c=5.0,
                                         exact_frac=0.6, restart_every=5000, seed=4321)
                try_update(a_out, f"sgd_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
