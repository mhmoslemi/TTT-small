    Solve LP: maximize sum(b) subject to (a*b)_k <= M for k in constraint_indices, b >= 0.

        j_min = max(0, ki - n + 1)
        j_max = min(n - 1, ki)
        js = np.arange(j_min, j_max + 1)
        A_ub[idx, js] = a[ki - js]
    
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_best_response_optimize(a0, time_limit, top_k=100, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    blend = 0.5
    stall = 0
    rng = np.random.default_rng(123)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_aa = fast_conv(a)
        M = conv_aa.max()
        k_total = len(conv_aa)
        
        if top_k < k_total:
            top_idx = np.argsort(conv_aa)[::-1][:top_k]
        else:
            top_idx = np.arange(k_total)
        
        b = solve_lp_best_response(a, M, top_idx)
        if b is None:
            blend *= 0.5
            if blend < 0.001:
                break
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        b = b / b_sum
        # Add small uniform for diversity
        b = 0.9 * b + 0.1 * np.ones(n) / n
        b = b / b.sum()
        
        a_new = (1 - blend) * a + blend * b
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        val = eval_J(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            stall = 0
            blend = max(blend * 0.995, 0.02)
            if verbose and it % 20 == 0:
                print(f"  LP it={it} ({time.monotonic()-t0:.0f}s) J={best_val:.6f} blend={blend:.3f}")
        else:
            stall += 1
            blend *= 0.5
            if stall > 50 and blend < 0.01:
                # Perturb and reset
                a = best_a + rng.normal(0, 0.01, n) * best_a
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                blend = 0.3
                stall = 0
            if blend < 0.0005:
                break
    
    return best_a, best_val


def mirror_descent_optimize(a0, time_limit, top_k=80, beta_w=10.0, c0=4.0, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        conv = fast_conv(y)
        k_max = min(top_k, len(conv))
        top_idx_s = np.argsort(conv)[::-1][:k_max]
        top_vals = np.maximum(conv[top_idx_s], 1e-30)
        w = top_vals ** beta_w
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx_s):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n - 1, ki)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                g[idx] += 2.0 * wi * y[ki - idx]
        
        eta = c0 / (np.sqrt(t) + 1.0)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        val = eval_J(a_new)
        if val < best_val - 1e-16:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-12)
            x /= x.sum()
            x_prev = x.copy()
            stall = 0
            c0 = max(c0 * 0.9, 1.0)
    
    return best_a, best_val


def make_init_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_init_bimodal(n, w=50.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-w * (x - 0.15) ** 2) + np.exp(-w * (x - 0.85) ** 2) + 0.001
    return a / a.sum()
