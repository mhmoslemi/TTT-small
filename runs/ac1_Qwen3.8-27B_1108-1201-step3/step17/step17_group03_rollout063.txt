def fast_conv(a):
    """Fast full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_score(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2 * len(a) * c.max() / (s * s))


def interp_to_length(a, new_n):
    """Interpolate sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return result / rs

        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # d(c[k])/d(a[i]) = a[k-i] + (if k-i != i then a[i] contributes to c[k] via a[k-i]*a[i])
        # Actually c[k] = sum_i a[i]*a[k-i], so d(c[k])/d(a[i]) = a[k-i]
        # But also if i' = k-i is valid, a[i] appears as a[k-i'] = a[i], contributing a[i']
        # Wait, c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # d(c[k])/d(a[i]) = a[k-i] (for valid k-i)
        # But a[i] also appears when the index j in the sum satisfies k-j = i, i.e., j = k-i
        # So d(c[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i]? No...
        # c[k] = sum_j a[j] * a[k-j]
        # d/d(a[i]) of sum_j a[j]*a[k-j]:
        # = a[k-i] (from j=i term) + a[i] (from j=k-i term, if k-i != i)
        # = a[k-i] + a[i] if k-i != i
        # = 2*a[k-i] if k-i == i (i.e., k = 2i)
        # Actually more carefully:
        # d/d(a[i]) sum_j a[j]*a[k-j] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] + a[k-i] = 2*a[k-i] if both i and k-i are in [0,n-1]
        # = a[k-i] if only one of them is valid
        # Hmm, this is getting complicated. Let me just use the simpler formula:
        # For the full convolution c[k] = sum_{j} a[j] * a[k-j] (valid indices)
        # The gradient with respect to a[i] is: a[k-i] (from the j=i term in the sum)
        # PLUS a[k-i] (from the j term where k-j=i, i.e., j=k-i)
        # So if both i and k-i are valid: grad += 2 * a[k-i]
        # If only i is valid (k-i out of range): grad += a[k-i]... but that's 0
        # Actually, I think the correct gradient is:
        # d(c[k])/d(a[i]) = a[k-i] if 0 <= k-i < n, else 0
        # PLUS a[k-i] if 0 <= k-i < n AND k-i != i (to avoid double counting)
        
        # Let me just use the simple version: grad[i] += a[k-i] for valid k-i
        # This gives a subgradient of the max (up to a factor of 2 for the symmetric case)
        for i_val in idx:
            j_val = k_int - i_val
            if 0 <= j_val < n:
                grad[i_val] += 2.0 * wi * a[j_val]
    
    return grad


def fw_lp_optimize(a0, time_limit, solver, top_K=100, margin_frac=1e-3, 
                   verbose=True, min_margin=1e-5):
    """
    Frank-Wolfe with LP subproblem for minimizing max(conv(a,a)).
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    current_margin = margin_frac
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(y)
        M = c.max()
        if M < 1e-16:
            break
        
        # Select active constraints
        margins = M - c
        threshold = current_margin * M
        n_active = (margins <= threshold).sum()
        
        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:min(top_K, len(c))]
        else:
            active_idx = np.where(margins <= threshold)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix (linearized)
        # Constraint: sum_i y[k-i] * b[i] <= M for each active k
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]
        
        # Solve LP: maximize sum(b) subject to T_sel @ b <= M, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints_list = [T_sel @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints_list)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check improvements
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            current_margin = max(current_margin * 0.85, min_margin)
        
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={current_margin:.2e}")
        
        # Early stopping if no improvement in many iterations
        if t > 100 and my > best_m * (1 + 0.01) and mx > best_m * (1 + 0.01):
            if t % 200 == 0:
                # Try perturbation
                rng = np.random.default_rng(t)
                pert = rng.normal(0, 0.01, n) * best_a
                x_pert = best_a + pert
                x_pert = np.maximum(x_pert, 1e-10)
                x_pert = x_pert / x_pert.sum()
                x = x_pert
                y = x_pert.copy()
    
    return best_a


def mirror_descent(a0, time_limit, c0=3.0, top_k=50, beta_power=3,
                   restart_every=3000, verbose=True):
    """
    Mirror descent (in log space) with Nesterov acceleration.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute subgradient
        c = fast_conv(y)
        g = subgradient_max_conv(y, c, top_k=top_k)
        
        # Mirror descent step in log space
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Backtracking line search
        base_m = c.max()
        accepted = False
        for ls in range(8):
            log_step = np.log(y) - eta * g
            log_step -= log_step.max()
            a_new = np.exp(log_step)
            a_new = np.maximum(a_new, 1e-30)
            sa = a_new.sum()
            a_new = a_new / sa
            m_trial = fast_conv(a_new).max()
            if m_trial < base_m - 1e-20:
                x_prev = x.copy()
                x = a_new
                accepted = True
                break
            eta *= 0.5
        
        if not accepted:
            x_prev = x.copy()
            x = y.copy()
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a


def peak_rebalance(a0, time_limit, verbose=True, rng_seed=42):
    """
    Targeted peak rebalancing: move mass from high-contributing positions
    to low-contributing positions to flatten the convolution.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        c = fast_conv(a)
        M = c.max()
        
        # Find the peak lag
        peak_k = int(np.argmax(c))
        
        # Find which positions contribute most to this peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total = contrib.sum()
        
        if total < 1e-20:
            stall += 1
            continue
        
        # Try to reduce the peak by moving mass
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        
        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            best_move = None
            best_new_m = M
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                
                # Try multiple random destinations
                for _ in range(40):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)
            
            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakRebal ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a


def spectral_perturb(a, strength=0.02, rng=None):
    """Perturb in the frequency domain to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def multi_peak_flatting(a0, time_limit, verbose=True):
    """
    Directly target flattening multiple peaks simultaneously.
    Uses a weighted sum of top peaks as the objective.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Use a soft-max of top peaks as smooth surrogate
    top_k = 10
    sharpness = 5.0
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        c = fast_conv(y)
        
        # Soft max gradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        max_val = top_vals[0]
        
        if max_val < 1e-15:
            break
        
        # Soft weights
        w = np.exp(sharpness * (top_vals / max_val - 1.0))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror step
        eta = 2.0 / (np.sqrt(t) + 1.0)
        log_step = np.log(y) - eta * grad
        log_step -= log_step.max()
        a_new = np.exp(log_step)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x.copy()
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiPeak t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a


def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = eval_score(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2 * len(a) * c.max() / (s * s))


def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return result / rs


def make_u_shape(n, alpha=1.5):
    x = np.linspace(0.01, 0.99, n)

def compute_gradient(a, c, top_k=50, power=3):
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 1e-30)
    max_val = top_vals[0]
    w = np.exp(power * (top_vals / max_val - 1.0))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def mirror_descent_opt(a0, time_limit, c0=3.0, top_k=60, power=3,

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        c = fast_conv(y)
        g = compute_gradient(y, c, top_k=top_k, power=power)

                x_prev = x.copy()
                x = a_new
                accepted = True
                break
            eta *= 0.5

        if not accepted:
            x_prev = x.copy()
            x = y.copy()

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new

            stall = 0
        else:

            pert = rng.normal(0, 0.015, n) * best_a

def peak_rebalance(a0, time_limit, verbose=True, rng_seed=42, n_trials=30):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 600:
        c = fast_conv(a)
        M = c.max()
        peak_k = int(np.argmax(c))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total = contrib.sum()

        if total < 1e-20:
            stall += 1
            continue

        n_top = min(15, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_move = None
            best_new_m = M

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:

            J = 2.0 * n * best_m
            print(f"    PeakRebal ({elapsed:.0f}s): J={J:.8f}, stall={stall}")

    return best_a


def spectral_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

    if parent is not None and parent.sum() > 0.01:
        best_seq = normalize(parent)
        best_score = eval_score(best_seq)
    else:
        n_fb = 3600
        best_seq = make_u_shape(n_fb, alpha=1.5)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Quick mirror descent to get a feel ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD warmup (n={n0}) ===")
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        a_out = mirror_descent_opt(best_seq, t1, c0=2.5, top_k=50, power=3,
                                   restart_every=2000, verbose=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "md_warm")

    # === Phase 2: Main LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 40 and n_best <= 5500:
        a_out = fw_lp_optimize(best_seq, t2, solver, top_K=100,
                               margin_frac=3e-3, verbose=True, min_margin=1e-5)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Peak rebalance ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Rebalance (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_out = peak_rebalance(best_seq, t3, verbose=True, rng_seed=42, n_trials=25)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "rebal1")

    # === Phase 4: Targeted gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Targeted GD (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 15:
        a_out = targeted_gradient_descent(best_seq, t4, verbose=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "tgd1")

    # === Phase 5: LP with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW tight (n={n_best}) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30 and n_best <= 5500:
        a_out = fw_lp_optimize(best_seq, t5, solver, top_K=150,
                               margin_frac=5e-4, verbose=True, min_margin=1e-6)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 6: Spectral diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Spectral diversify (n={len(best_seq)}) ===")
    rng = np.random.default_rng(77)
    n_best = len(best_seq)
    t6_per = min(remaining * 0.06, 80.0)
    if t6_per > 20:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 20:
                break
            a_pert = spectral_perturb(best_seq, strength=0.008, rng=rng)
            a_out = fw_lp_optimize(a_pert, t6_per, solver, top_K=100,
                                   margin_frac=2e-3, verbose=False, min_margin=1e-5)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"spectr{trial}")

    # === Phase 7: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 400, n_best + 800]))
    longer_lengths = [l for l in longer_lengths if l <= 5500 and remaining > 150]

    for new_n in longer_lengths[:1]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 7: Extend to n={new_n} ===")
        a_long = interp_to_length(best_seq, new_n)

        t7a = min(remaining * 0.12, 150.0)
        if t7a > 20:
            a_out = fw_lp_optimize(a_long, t7a, solver, top_K=100,
                                   margin_frac=2e-3, verbose=False, min_margin=1e-5)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.06, 80.0)
            if t7b > 15:
                a_out = peak_rebalance(best_seq, t7b, verbose=False, rng_seed=new_n, n_trials=20)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"rebal_n{new_n}")

    # === Phase 8: Final tight LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.20, 250.0)
    if t8 > 30 and n_best <= 5500:
        a_out = fw_lp_optimize(best_seq, t8, solver, top_K=200,
                               margin_frac=1e-4, verbose=True, min_margin=1e-7)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final peak rebalance ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final rebalance (n={len(best_seq)}) ===")
    t9 = min(remaining * 0.20, 180.0)
    if t9 > 10:
        a_out = peak_rebalance(best_seq, t9, verbose=True, rng_seed=999, n_trials=30)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "final_rebal")

    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_u_shape(n_fb, alpha=1.5)
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
