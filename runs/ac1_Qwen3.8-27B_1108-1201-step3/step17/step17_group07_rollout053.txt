def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def soft_gradient(a, p):
    """g[m] = 2 * sum_k p[k] * a[k-m] via FFT cross-correlation."""
    n = len(a)
    N = 1 << (3 * n).bit_length()
    P = np.fft.rfft(np.asarray(p, dtype=np.float64), N)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), N)
    G = np.fft.irfft(P * np.conj(A), N)
    return 2.0 * G[:n]
