            break
        budget_used += int(600)  # Allow time for each Frank-Wolfe round
        print(f"  Iteration {i+1}: Optimization of {n0} degrees active")
        # Call Frank-Wolf with dimensionally-reduced constraints

    print(f"Final candidate: score={best_score:.6f} with sequence length={L}")
    print(f"Budget used: {budget_used}s/{time_budget}s")
    print("Returning the best sequence found.")
    return np.array([best_seq[i] for i in range(min(len(best_seq), 15))]) if best_seq else float('inf')
