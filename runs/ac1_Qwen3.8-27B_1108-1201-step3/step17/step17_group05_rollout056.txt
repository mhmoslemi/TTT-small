import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(a):
    """Project onto probability simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_subgrad(a, beta=50.0, top_k=200):
    """Compute softmax-weighted subgradient of max autocorrelation."""
    n = len(a)
    ac = fast_autocorr(a)
    # Use top-k lags for efficiency
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    max_val = top_vals[0]
    w = np.exp(beta * (top_vals - max_val))
    w = w / w.sum()
    
    # Build sparse weight vector
    w_full = np.zeros(len(ac))
    for i in range(k):
        w_full[top_idx[i]] = w[i]
    
    # grad[j] = 2 * sum_m w[m] * a[m-j] (valid range)
    # Efficient computation via convolution
    total_len = n + len(w_full) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_full)] = w_full
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad, ac.max()

                               top_k=200, verbose=False):
    """Nesterov-accelerated mirror descent with softmax subgradient."""
    n = len(a0)
    x = normalize_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        frac = min(1.0, t / 5000.0)
        beta = beta_init + (beta_final - beta_init) * frac

        y = x + m * (x - x_prev)

        log_new -= log_new.max()
        x_new = np.exp(log_new)
        x_new = project_simplex(x_new)
        
        x_prev = x.copy()
        x = x_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_M:

            best_a = x.copy()

            print(f"    ND t={t}: J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_M

def lp_direction(a, M, top_K=300):
    """Solve LP to find direction b that keeps conv(a,b) <= M."""
    n = len(a)
    ac = fast_autocorr(a)
    k = min(top_K, 2*n-1)
    top_idx = np.argsort(ac)[::-1][:k]
    
    # Build constraint matrix: T[m,j] = a[m-j] if valid
    T = np.zeros((k, n))
    for i, m in enumerate(top_idx):
        m = int(m)
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        T[i, js] = a[m - js]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(k), cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

        for _ in range(20):

    w = np.exp(beta * (top_vals - max_val))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / w_sum
    w_full = np.zeros(len(ac))
    w_full[top_idx] = w
    total_len = n + len(w_full) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_full)] = w_full
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()


def nesterov_softmax_descent(a0, time_limit, beta_start=30.0, beta_end=250.0,
                              top_k=200, c0=5.0, verbose=False, seed=42):
    """Nesterov-accelerated mirror descent on simplex with adaptive softmax subgradient."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 15000.0)
        beta = beta_start + (beta_end - beta_start) * frac

        m = (t - 1.0) / (t + 2.0)
        y = x + m * (x - x_prev)
        y = norm_sum1(y)

        g, M_curr = softmax_subgrad(y, beta=beta, top_k=top_k)

        step = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - step * g
        log_new -= log_new.max()
        x_new = np.exp(log_new)
        x_new = norm_sum1(x_new)

        x_prev = x.copy()
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            stall = 0
            scale = 0.01 + 0.03 * (stall / 8000.0)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x = norm_sum1(x_r)
            x_prev = x.copy()

        if verbose and t % 3000 == 0:
            J = 2.0 * n * best_M
            print(f"    NSD t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_a, best_M


def lp_direction(a, active_lags, M, solver):
    """Solve LP to find direction b with conv(a,b)[m] <= M for active lags."""
    n = len(a)
    k = len(active_lags)
    if k == 0:
        return None

    T = np.zeros((k, n))
    for i in range(k):
        m = int(active_lags[i])
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        T[i, js] = a[m - js]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(k), cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def lp_fw_optimize(a0, time_limit, solver, top_K=250, verbose=False, seed=42):
    """LP Frank-Wolfe with adaptive constraint selection and backtracking line search."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for t in range(1, 1000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(x)
        M_curr = ac.max()

        # Adaptive K: start smaller, grow over time
        k = min(max(100, top_K // 2 + top_K // 2 * (t // 50)), top_K)
        active = np.argsort(ac)[::-1][:k]

        direction = lp_direction(x, active, M_curr, solver)
        if direction is None:
            k2 = max(50, k // 2)
            active2 = np.argsort(ac)[::-1][:k2]
            direction = lp_direction(x, active2, M_curr, solver)
            if direction is None:
                if t > 20:
                    break
                continue

        # Backtracking line search
        alpha = 2.0 / (t + 2.0)
        x_new = None
        for _ in range(25):
            x_trial = (1.0 - alpha) * x + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            s = x_trial.sum()
            if s < 1e-15:
                break
            x_trial = x_trial / s
            M_trial = fast_autocorr(x_trial).max()
            if M_trial < M_curr - 1e-15:
                x_new = x_trial
                break
            alpha *= 0.5

        if x_new is None:
            no_improve += 1
            if no_improve > 30:
                pert = rng.normal(0, 0.01, n)
                x_trial = x + pert
                x_trial = norm_sum1(x_trial)
                M_trial = fast_autocorr(x_trial).max()
                if M_trial < best_M:
                    x = x_trial
                    best_M = M_trial
                    best_a = x.copy()
                    no_improve = 0
            continue

        x = x_new
        M_new = fast_autocorr(x).max()
        if M_new < best_M:
            best_M = M_new
            best_a = x.copy()
            no_improve = 0

        if verbose and t % 5 == 0:
            J = 2.0 * n * best_M
            print(f"    LP-FW t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, K={k}")

    return best_a, best_M


def pairwise_swap_optimize(a0, time_limit, verbose=False, seed=42):
    """Fast pairwise swap local search targeting peak autocorrelation lags."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:25]
        improved = False

        for lag_val in top_lags[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(lag_val)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k - js]
            if contrib.max() < 1e-20:
                continue

            src = js[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations by their contribution to top lags
            sens = np.zeros(n)
            for lag2 in top_lags[:8]:
                k2 = int(lag2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                for d in range(s2, e2 + 1):
                    partner = k2 - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]

            # Try random candidates from low-sensitivity region
            for _ in range(40):
                dst = int(rng.integers(n))
                if dst == src or sens[dst] >= sens.mean():
                    continue
                for frac in [0.01, 0.02, 0.05, 0.005]:
                    amt = src_val * frac
                    a_new = a.copy()
                    a_new[src] -= amt
                    a_new[dst] += amt
                    a_new = np.maximum(a_new, 0.0)
                    s = a_new.sum()
                    if s < 1e-15:
                        continue
                    a_new = a_new / s
                    M_new = fast_autocorr(a_new).max()
                    if M_new < best_M - 1e-14:
                        a = a_new
                        best_M = M_new
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            no_improve = 0
            if verbose and no_improve == 0 and best_a.sum() > 0:
                J = 2.0 * n * best_M
                if J < 1.505:
                    print(f"    PS ({time.monotonic()-t0:.0f}s): J={J:.6f}")
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.01
                    a[i] -= amt
                    a[j] += amt
                    a = norm_sum1(a)
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        no_improve = 0

    return best_a, best_M


def spectral_flat_init(n, seed=0):
    """Initialize with flat power spectrum (CAZAC-like) for real non-negative sequence."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    # Generate random phases
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Flat magnitude
    A = np.ones(fft_size // 2 + 1) * np.exp(1j * phases)
    # Enforce Hermitian symmetry for real signal
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size // 2 + 1] = A
    for i in range(1, fft_size // 2):
        A_full[fft_size - i] = np.conj(A[i])
    # IFFT to get time domain
    x = np.fft.ifft(A_full).real[:n]
    # Make non-negative: shift and clip
    x = x - x.min() + 0.1
    x = np.maximum(x, 0.0)
    return norm_sum1(x)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining Nesterov softmax descent, LP Frank-Wolfe,
    and pairwise swap refinement with spectral initialization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = spectral_flat_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Nesterov Softmax Descent (multiple inits) ===
    print(f"\n=== Phase 1: Nesterov Softmax Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 350.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Spectral flat init
        inits.append(spectral_flat_init(n0, seed=7))
        labels.append("spec_flat")

        # Symmetric smooth
        x = np.linspace(0, 1, n0)
        a_sym = 1.0 + 0.1 * np.sin(2 * np.pi * 3 * x) + 0.05 * np.sin(2 * np.pi * 7 * x)
        a_sym = np.maximum(a_sym, 0.0)
        inits.append(norm_sum1(a_sym))
        labels.append("smooth")

        # Blend
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * spectral_flat_init(n0, seed=42)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 90:
                break
            t_avail = min(t_per, time_budget - 90 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  NSD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = nesterov_softmax_descent(init, t_avail,
                                                 beta_start=30.0, beta_end=200.0,
                                                 top_k=200, c0=5.0, verbose=False,
                                                 seed=42 + idx)
            try_update(a_out, f"nsd_{lbl}")

    # === Phase 2: LP Frank-Wolfe (main power) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Frank-Wolfe (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.40, 550.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t2, solver, top_K=250, verbose=True, seed=123)
        try_update(a_out, f"lp_fw_n{n_best}")

    # === Phase 3: Another NSD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: NSD Polish ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_softmax_descent(a_in, t3,
                                             beta_start=80.0, beta_end=300.0,
                                             top_k=100, c0=3.0, verbose=False,
                                             seed=777)
        try_update(a_out, f"nsd_polish_n{n_best}")

    # === Phase 4: LP Frank-Wolfe Round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW Round 2 ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t4, solver, top_K=200, verbose=True, seed=456)
        try_update(a_out, f"lp_fw2_n{n_best}")

    # === Phase 5: Pairwise swap refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Pairwise Swap ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_swap_optimize(a_in, t5, verbose=True, seed=99)
        try_update(a_out, f"swap_n{n_best}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 100, n_best + 100, n_best + 200]
    candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000 and l != n_best]

    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 6: Length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 120.0)
        if t6a > 20:
            a_out, _ = nesterov_softmax_descent(a_long, t6a,
                                                 beta_start=40.0, beta_end=200.0,
                                                 top_k=150, c0=4.0, verbose=False,
                                                 seed=333)
            try_update(a_out, f"nsd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 4000:
            t6b = min(remaining * 0.40, 120.0)
            if t6b > 20:
                a_out, _ = lp_fw_optimize(norm_sum1(a_long), t6b, solver,
                                           top_K=200, verbose=False, seed=555)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final LP polish if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 4500:
        print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
        t7 = min(remaining * 0.6, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_optimize(a_in, t7, solver, top_K=150, verbose=False, seed=888)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = spectral_flat_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
