def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:

def compute_J(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (a.sum() ** 2)

def soft_max(ac, temp):
    """Soft maximum using log-sum-exp."""
    m = ac.max()
    return m + temp * np.log(np.sum(np.exp((ac - m) / temp)))

def soft_max_grad(ac, temp):
    """Gradient of soft max with respect to ac."""
    m = ac.max()
    weights = np.exp((ac - m) / temp)
    weights /= weights.sum()
    return weights  # d(soft_max)/d(ac[k]) = weights[k]

2. Multi-scale temperature annealing for global-to-local transition
3. B-spline and Gaussian initializations informed by frequency-domain analysis
4. Peak-shaving coordinate refinement
5. Multi-start with diverse initializations and length exploration
Key idea: Use mirror descent (exponentiated gradient) which naturally handles the simplex
constraint, combined with Nesterov acceleration and adaptive log-sum-exp temperature
for a smooth-yet-accurate approximation of the max-convolution objective.

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Non-negative normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_obj(a):
    """Compute objective 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def evaluate_safe(a):

def soft_max_vals(ac, temp):
    """Compute soft-max weights (softmax of ac/temp)."""
    m = ac.max()
    w = np.exp((ac - m) / temp)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(ac)
        w[np.argmax(ac)] = 1.0
    else:
        w = w / sw
    return w


def nesterov_exp_gradient(a0, time_limit, temp0=0.01, lr0=0.5,
