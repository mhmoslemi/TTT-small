            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: cvxpy {s}")
                return
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        if len(a_arr) == 0:
            return float('inf')
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_cosine_init(n, alpha=2.0):
    """Create a cosine-window initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** alpha
    a[np.cos(np.pi * x) < 0] = 0  # Ensure non-negative
    return norm_sum1(a)

def make_hann_init(n):
    """Create a Hann window initialization."""
    x = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    return norm_sum1(a)

def make_smooth_init(n, seed=0, freqs=None):
    """Create a smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    if freqs is None:
        freqs = [1, 2, 3, 4, 5]
    for f in freqs:
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_symmetric(a):
    """Symmetrize a sequence."""
    return 0.5 * (a + a[::-1])

def solve_fw_lp(y, n):
    """Solve the LP oracle for one FW step.
    
    minimize M
    subject to: conv(b, y)[k] <= M for all k
                sum(b) = 1
                b >= 0
    """
    global _global_use_scipy, _global_solver
    
    # Build T matrix: T[k, i] = y[k-i] if 0 <= k-i < n
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = y
    
    if _global_use_scipy:
        # Variables: [b_0, ..., b_{n-1}, M]
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M
        
        A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(-np.inf, np.inf)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        
        if res.status != 0:
            return None
        b_val = np.maximum(res.x[:n], 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            return None
        return b_val / b_sum
    
    else:
        # cvxpy fallback
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=_global_solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            return None
        return b_val / b_sum

def nesterov_fw(a0, time_limit, verbose=True, max_iters=2000):
    """Nesterov-accelerated Frank-Wolfe with LP oracle."""
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()
    prev_y = y.copy()
    
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP oracle
        b_star = solve_fw_lp(y, n)
        if b_star is None:
            # Fallback: use subgradient step
            ac = fast_autocorr(y)
            k_star = np.argmax(ac)
            g = np.zeros(n)
            for i in range(n):
                j = int(k_star) - i
                if 0 <= j < n:
                    g[i] = y[j]
                if 2 * i == int(k_star):
                    g[i] += y[i]
            b_star = np.zeros(n)
            b_star[np.argmin(g)] = 1.0
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        prev_y = y
        y = y_new
        x = x_new
        
        # Evaluate
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    FW done ({elapsed:.0f}s): J={J:.6f}, {t} iters")
    
    return best_a, best_m

def compute_soft_subgrad(a, beta=8.0, top_k=100):
    """Compute soft subgradient of max(conv(a,a))."""
    n = len(a)
    ac = fast_autocorr(a)
    
    if top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = np.zeros(len(ac))
        top_vals = ac[top_idx]
        shifted = top_vals - top_vals.max()
        w_top = np.exp(beta * shifted)
        w_top /= w_top.sum()
        for wi, idx in zip(w_top, top_idx):
            w[int(idx)] = wi
    else:
        shifted = ac - ac.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # Use FFT
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, ac.max()

def mirror_descent(a0, time_limit, verbose=True, beta_init=8.0, beta_min=3.0,
                   c0=5.0, restart_every=5000, seed=42):
    """Mirror descent with adaptive soft-subgradient and Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta_init
    stall = 0
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Soft subgradient
        if iterations < 100 or iterations % 50 == 0:
            g, M_curr = compute_soft_subgrad(y, beta=cur_beta, top_k=100)
        else:
            g, M_curr = compute_soft_subgrad(y, beta=cur_beta, top_k=50)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x.copy()
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        cur_beta = max(beta_min, beta_init - 0.002 * iterations)
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if rng.random() < 0.2:
                x = make_smooth_init(n, seed=int(rng.integers(10000)))
                x_prev = x.copy()
        
        if verbose and iterations % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={iterations} ({elapsed:.0f}s): J={J:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    MD done ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
