def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Compute conv(a, b) using FFT."""
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    B = np.fft.rfft(b.astype(np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_soft_subgrad(a, beta=10.0):

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=2000):
    """
    LP-based Frank-Wolfe with backtracking line search.
    Solves: max sum(b) s.t. conv(a, b)[k] <= M for all k, b >= 0, sum(b) = 1
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_nesterov = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Build constraint matrix T where T[k, j] = y[k-j] if 0 <= k-j < n
        n_constraints = 2 * n - 1
        T = np.zeros((n_constraints, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(n_constraints),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_nesterov = y_new + beta_t * (y_new - y)
        x_nesterov = np.maximum(x_nesterov, 0.0)
        sx = x_nesterov.sum()
        if sx < 1e-15:
            x_nesterov = y_new.copy()
        else:
            x_nesterov = x_nesterov / sx
        
        y = y_new
        
        # Backtracking line search on the Nesterov point
        # Try to improve by moving from y toward x_nesterov
        # Actually, let's just evaluate both and keep the best
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_nesterov).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nesterov.copy()
            # If Nesterov point is better, use it as new y
            y = best_a.copy()
            x_nesterov = best_a.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def targeted_peeling(a0, time_limit, verbose=True, seed=42):
    """
    Targeted coordinate descent that identifies and reduces the worst
    contributing pairs for peak lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top lags
        n_top = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improvement_found = False
        
        for k in top_lags[:10]:
            k_int = int(k)
            # Find contributing pairs
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the most contributing pair
            max_idx = np.argmax(contrib)
            src = int(idx[max_idx])
            dst_partner = k_int - src
            
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Try to move mass from src to a location that reduces the peak
            # Score destinations by their "danger" (how much they'd contribute to peaks)
            n_sens = min(10, n_top)
            danger = np.zeros(n)
            for k2 in top_lags[:n_sens]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    if 0 <= partner < n:
                        danger[d] += a[partner]
            
            # Find safe destinations (low danger, far from src)
            danger[src] = np.inf
            candidates = np.argsort(danger)[:min(50, n // 10)]
            
            for dst in candidates[:20]:
                dst = int(dst)
                if dst == src:
                    continue
                
                # Try different transfer amounts
                for frac in [0.01, 0.005, 0.02, 0.003, 0.015]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improvement_found = True
                        stall = 0
                        break
                if improvement_found:
                    break
            if improvement_found:
                break
        
        if not improvement_found:
            stall += 1
            # Random kick
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Peel ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_smooth_init(n, seed=0):
    """Create a smooth symmetric initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    a_sym = 0.5 * (a + a[::-1])
    return norm_sum1(a_sym)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Mirror descent with adaptive soft-subgradient (coarse optimization)
2. LP Frank-Wolfe with top-K constraints and Nesterov acceleration (medium refinement)
3. Targeted coordinate peeling on worst lags (fine-tuning - main workhorse)
4. Multi-length exploration and final polish
Key: FFT-based convolution, adaptive beta schedule, danger-based destination selection.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  Using solver: ECOS (fallback)")
    return _global_solver


def fast_autocorr(a):
    """Compute conv(a, a) using FFT. Input must be float64 numpy array."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_soft_subgrad(a, beta=10.0, top_k=0):
    """Compute soft subgradient of max(conv(a,a)). Returns (grad, max_val)."""
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    
    if top_k > 0 and top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        shifted = top_vals - top_vals.max()
        w_top = np.exp(beta * shifted)
        ws = w_top.sum()
        if ws < 1e-30:
            w_top = np.ones(top_k) / top_k
        else:
            w_top = w_top / ws
        w = np.zeros(len(ac))
        for i in range(top_k):
            w[int(top_idx[i])] = w_top[i]
    else:
        shifted = ac - ac_max
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / ws
    
    # Mask tiny weights
    mask = w > 1e-12
    if mask.sum() < 1:
        w = np.ones_like(ac) / len(ac)
    else:
        w = np.where(mask, w, 0.0)
        w = w / w.sum()
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid (k-j)
    # = 2 * conv(a[::-1], w)[n-1 : 2n-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv_result = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad, ac_max


def mirror_descent(a0, time_limit, beta_init=10.0, beta_min=4.0,
                   c0=5.0, restart_every=4000, seed=42, verbose=False):
    """Nesterov-accelerated mirror descent on simplex with adaptive soft-subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta_init
    stall_count = 0
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        
        # Nesterov momentum
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient (use top-k for speed after warmup)
        top_k = 0 if iterations < 300 else min(64, 2 * n - 1)
        g, M_curr = compute_soft_subgrad(y, beta=cur_beta, top_k=top_k)
        
        # Mirror descent step (exponential family projection)
        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30))
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x.copy()
        x = a_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive beta: increase sharpness
        cur_beta = max(beta_min, beta_init * (0.9997 ** iterations))
        
        # Restart if stalled
        if stall_count > restart_every:
            stall_count = 0
            noise = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                print(f"    MD restart iter={iterations}: J={2.0*n*best_m:.6f}")
        
        if verbose and iterations % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={iterations} ({elapsed:.0f}s): J={2.0*n*best_m:.6f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=600, max_iters=3000):
    """LP-based Frank-Wolfe with top-K constraints and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_nv = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select top-K most binding constraints
        K = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix: T[m, j] = y[m-j] if 0 <= m-j < n
        T = np.zeros((K, n))
        for col_i, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            T[col_i, js] = y[m_int - js]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_nv = y_new + beta_t * (y_new - y)
        x_nv = np.maximum(x_nv, 0.0)
        sx = x_nv.sum()
        if sx < 1e-15:
            x_nv = y_new.copy()
        else:
            x_nv = x_nv / sx
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_nv).max()
        if mx < best_m:
            best_m = mx
            best_a = x_nv.copy()
            y = best_a.copy()
            x_nv = best_a.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={t} ({elapsed:.0f}s): J={2.0*n*best_m:.6f}")
    
    return best_a, best_m


def targeted_peeling(a0, time_limit, verbose=True, seed=42):
    """
    Targeted coordinate descent: identifies worst contributing pairs for peak lags
    and moves mass to safe destinations. The main fine-tuning workhorse.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500
    total_moves = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        
        # Get top lags
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        # Compute danger field: for each position d, how much would adding mass there
        # affect the top lags?
        danger = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            s = max(0, k_int - n + 1)
            e = min(n - 1, k_int)
            for d in range(s, e + 1):
                partner = k_int - d
                if 0 <= partner < n:
                    danger[d] += a[partner]
        
        improvement_found = False
        
        # For each top lag, try to reduce its worst contributor
        for k in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Get top contributing pairs
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            
            for ci in top_contrib_idx:
                src = int(idx[ci])
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Try moving to safe destinations
                danger[src] = np.inf
                # Sort by danger (ascending) and take safe ones
                safe_dists = np.argsort(danger)[:min(40, n // 5)]
                
                for dst in safe_dists[:15]:
                    dst = int(dst)
                    if dst == src:
                        continue
                    
                    # Try multiple transfer fractions (smaller = more delicate)
                    for frac in [0.003, 0.005, 0.002, 0.008, 0.004, 0.01]:
                        amt = src_val * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] = src_val - amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-13:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improvement_found = True
                            total_moves += 1
                            stall = 0
                            break
                    if improvement_found:
                        break
                if improvement_found:
                    break
            if improvement_found:
                break
        
        if not improvement_found:
            stall += 1
            # Periodic random perturbation to escape local minima
            if stall in (30, 80, 150, 250, 400):
                n_pert = min(20, n // 100)
                srcs = rng.choice(n, size=n_pert, replace=False)
                for src in srcs:
                    src = int(src)
                    if a[src] < 1e-12:
                        continue
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and (stall % 100 == 0 or total_moves % 500 == 0):
            elapsed = time.monotonic() - t0
            print(f"    Peel ({elapsed:.0f}s): J={2.0*n*best_m:.6f}, moves={total_moves}, stall={stall}")
    
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_smooth_init(n, seed=0):
    """Create a smooth symmetric initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    a_sym = 0.5 * (a + a[::-1])
    return norm_sum1(a_sym)


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Mirror descent (coarse, 20% time)
    Phase 2: LP Frank-Wolfe (medium, 20% time)
    Phase 3: Targeted peeling (fine, 40% time) - main workhorse
    Phase 4: Length exploration + final polish (20% time)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        best_seq = make_smooth_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Mirror Descent (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 50:
        print(f"\n=== Phase 1: Mirror Descent (t={t1:.0f}s) ===")
        
        inits = [norm_sum1(best_seq)]
        labels = ["parent"]
        if parent is not None:
            blend = 0.7 * norm_sum1(parent) + 0.3 * make_smooth_init(n0, seed=5)
            inits.append(norm_sum1(blend))
            labels.append("blend")
        
        t_per = t1 / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  MD '{lbl}' ({t_avail:.0f}s)...")
            a_out, _ = mirror_descent(init, t_avail, verbose=False,
                                       beta_init=10.0, beta_min=5.0,
                                       c0=5.0, restart_every=4000,
                                       seed=100 + idx)
            try_update(a_out, f"md_{lbl}")
    
    # === Phase 2: LP Frank-Wolfe (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 350.0)
    n_best = len(best_seq)
    if t2 > 50 and n_best <= 4000:
        print(f"\n=== Phase 2: LP-FW (n={n_best}, t={t2:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=600)
        try_update(a_out, f"lpfw_n{n_best}")
    
    # === Phase 3: Targeted Peeling (40% time) - main fine-tuning ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.40, 600.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Targeted Peeling (n={len(best_seq)}, t={t3:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = targeted_peeling(a_in, t3, verbose=True, seed=42)
        try_update(a_out, f"peel_n{len(best_seq)}")
    
    # === Phase 4: Length exploration + final polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best + 300, n_best + 600, n_best - 200]
    candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000]
    
    explored = 0
    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        explored += 1
        print(f"\n=== Phase 4.{explored}: Length n={new_n} ===")
        a_long = interpolate_seq(best_seq, new_n)
        
        t4a = min(remaining * 0.25, 150.0)
        if t4a > 20:
            a_out, _ = mirror_descent(a_long, t4a, verbose=False,
                                       beta_init=10.0, beta_min=5.0,
                                       c0=4.0, restart_every=3000,
                                       seed=777 + explored)
            try_update(a_out, f"md_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 3500:
            t4b = min(remaining * 0.30, 150.0)
            if t4b > 20:
                a_out, _ = lp_frank_wolfe(a_long, t4b, solver, verbose=False, top_K=500)
                try_update(a_out, f"lp_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t4c = min(remaining * 0.40, 150.0)
            if t4c > 20:
                a_out, _ = targeted_peeling(a_long, t4c, verbose=False, seed=300 + explored)
                try_update(a_out, f"peel_n{new_n}")
    
    # === Final: Extended peeling on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Final: Extended Peeling (n={n_best}, t={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = targeted_peeling(a_in, remaining * 0.8, verbose=True, seed=555)
        try_update(a_out, f"peel_final_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent(a_in, remaining * 0.8, verbose=False,
                                       beta_init=15.0, beta_min=8.0,
                                       c0=2.0, restart_every=10000,
                                       seed=1234)
            try_update(a_out, f"md_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_smooth_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
