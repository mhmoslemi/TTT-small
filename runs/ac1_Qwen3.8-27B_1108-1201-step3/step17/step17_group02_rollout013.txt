"""

5. Coordinate descent refinement

import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution of a with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max_gradient_fft(a, beta):
    """
    Compute gradient of log(sum_k exp(beta*b_k)) where b = conv(a,a).
    Returns (gradient, smooth_max_value).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_autocorr(a)  # length 2n-1
    
    # Numerical stability
    b_max = b.max()
    shifted = beta * (b - b_max)
    # Clip for numerical safety
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    Z = exp_b.sum()
    if Z < 1e-300:
        # All exp are underflowed; use uniform
        w = np.ones(2 * n - 1) / (2 * n - 1)
        smooth_max = b_max / beta  # degenerate
    else:
        w = exp_b / Z
        smooth_max = b_max + np.log(Z) / beta
    
    # grad_j = 2*beta * sum_{m=0}^{n-1} a[m] * w[j+m]
    # = 2*beta * conv(a_rev, w)[j+n-1]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:conv_len]
    
    grad = 2.0 * beta * conv_result[n - 1:2 * n - 1]
    
    return grad, smooth_max

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_initialization(n, kind='parabolic', alpha=1.0):
    """Create various U-shaped initializations."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'parabolic':
        a = alpha * (x - 0.5)**2 + 0.01
    elif kind == 'quartic':
        a = alpha * (x - 0.5)**4 + 0.01
    elif kind == 'cosine_u':
        a = np.cos(np.pi * x)**2 + 0.01  # zero at edges... wait
        # cos(pi*0) = 1, cos(pi) = -1, so cos^2 is 1 at edges, 0 in middle. That's U-shaped!
    elif kind == 'sin_centered':
        # High in middle, low at edges (inverted U) - not what we want
        a = np.sin(np.pi * x)**2 + 0.01
    elif kind == 'exp_u':
        a = np.exp(alpha * (x - 0.5)**2 / 0.25) + 0.001
    elif kind == 'power_u':
        a = ((x - 0.5) * 2)**alpha + 0.01  # (2x-1)^alpha for alpha > 0
    else:
        a = np.ones(n)
    return norm_sum1(np.maximum(a, 0.0))

def smooth_max_mirror_descent(a0, time_limit, beta_start=0.5, beta_end=200.0,
                               c0=2.0, use_nesterov=True, restart_every=5000,
                               verbose=False, rng_seed=42):
    """
    Mirror descent with smooth max approximation of the autocorrelation peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2 * restart_every
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(step / (restart_every * 10), 1.0)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        
        # Nesterov extrapolation
        if use_nesterov and step > 1:
            gamma = (step - 1.0) / (step + 1.5)
            y = x + gamma * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
        else:
            y = x.copy()
        
        # Gradient
        grad, _ = smooth_max_gradient_fft(y, beta)
        
        # Mirror descent update (multiplicative)
        eta = c0 / (np.sqrt(step) + 1.0)
        # Clip gradient to avoid extreme updates
        grad_max = np.abs(grad).max()
        if grad_max > 1e10:
            grad = grad / grad_max * 1e10
        log_a = np.log(y) - eta * grad
        log_a = log_a - log_a.max()
        x_new = np.exp(log_a)
        x_new = np.maximum(x_new, 1e-15)
        s = x_new.sum()
        if s < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s
        
        x_prev = x
        x = x_new
        
        # Evaluate
        M = fast_autocorr(x).max()
        if M < best_M - 1e-15:
            best_M = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SMD step={step} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")
        
        if stall > max_stall:
            # Restart with perturbation
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            # Reset progress
            step = 0
    
    # Final check on x and x_prev too
    for candidate in [x, x_prev]:
        M_c = fast_autocorr(candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = candidate.copy()
    
    return best_a, best_M

def lp_frank_wolfe(a0, time_limit, solver=None, top_K=500, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    if solver is None:
        solver = _get_solver()
    
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation of current y
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < 1e-16:
            break
        
        # Select top K constraints (most binding)
        if top_K >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:top_K]
        
        # Build constraint matrix: T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        # For the selected lags
        T_sel = np.zeros((len(binding_idx), n))
        for idx, k in enumerate(binding_idx):
            ki = int(k)
            for j in range(n):
                partner = ki - j
                if 0 <= partner < n:
                    T_sel[idx, j] = y[partner]
        
        M_sel = M_curr * np.ones(len(binding_idx))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, top_k=20, verbose=False, rng_seed=123):
    """Coordinate descent: move mass from high-contributing positions to low-sensitivity ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:top_k]
        
        # Sensitivity map
        sens = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            for j in range(n):
                partner = ki - j
                if 0 <= partner < n:
                    sens[j] += a[partner]
        
        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            # Find the pair contributing most to this peak
            best_contrib = -1
            best_src = -1
            for j in range(n):
                partner = ki - j
                if 0 <= partner < n:
                    c = a[j] * a[partner]
                    if c > best_contrib:
                        best_contrib = c
                        best_src = j
            
            if best_src < 0 or best_contrib < 1e-20:
                continue
            
            # Find destination: lowest sensitivity
            min_sens = np.inf
            best_dst = -1
            for d in range(n):
                if d != best_src and sens[d] < min_sens:
                    min_sens = sens[d]
                    best_dst = d
            
            if best_dst < 0:
                continue
            
            # Try to move fraction of mass
            for frac in [0.01, 0.005, 0.02, 0.05, 0.002]:
                amt = a[best_src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[best_src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth-max mirror descent (SMD) with adaptive beta schedule
2. Nesterov acceleration, restarts, and two-phase beta annealing
3. LP-based Frank-Wolfe polishing with adaptive constraint selection
4. Coordinate descent refinement targeting peak-contributing pairs
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_gradient_fft(a, beta):
    """
    Compute gradient of log(sum_k exp(beta*b_k)) where b = conv(a,a).
    Returns (gradient, smooth_max_value). O(n log n) via FFT.
    """
    n = len(a)
    b = fast_autocorr(a)

    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    Z = exp_b.sum()
    if Z < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        smooth_max = b_max
    else:
        w = exp_b / Z
        smooth_max = b_max + np.log(Z) / beta

    # grad_j = 2*beta * sum_{m=0}^{n-1} a[m] * w[j+m]
    # = 2*beta * conv(a_rev, w)[j + n - 1]
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:conv_len]

    grad = 2.0 * beta * conv_result[n - 1:2 * n - 1]
    return grad, smooth_max


def norm_sum1(a):
    """Normalize to non-negative with sum = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def make_initialization(n, kind='parabolic', alpha=1.0):
    """Create various U-shaped initializations."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'parabolic':
        a = alpha * (2.0 * x - 1.0) ** 2 + 0.01
    elif kind == 'quartic':
        a = alpha * (2.0 * x - 1.0) ** 4 + 0.01
    elif kind == 'cosine_u':
        a = np.cos(np.pi * x) ** 2 + 0.01
    elif kind == 'exp_u':
        a = np.exp(alpha * (x - 0.5) ** 2 / 0.1) + 0.001
    elif kind == 'power2':
        a = np.abs(2.0 * x - 1.0) ** 2 + 0.01
    elif kind == 'power3':
        a = np.abs(2.0 * x - 1.0) ** 3 + 0.01
    elif kind == 'power4':
        a = np.abs(2.0 * x - 1.0) ** 4 + 0.01
    elif kind == 'linear_u':
        a = np.abs(2.0 * x - 1.0) + 0.05
    else:
        a = np.ones(n)
    return norm_sum1(np.maximum(a, 0.0))


def smooth_max_md(a0, time_limit, beta_start=0.5, beta_end=300.0,
                  c0=2.0, use_nesterov=True, restart_every=4000,
                  verbose=False, rng_seed=42, two_phase=True):
    """
    Mirror descent with smooth max of autocorrelation peak.
    Two-phase: slow beta ramp then fast ramp.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = restart_every * 2
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)

        if two_phase:
            # Phase 1 (first 40%): beta from beta_start to beta_start*50
            # Phase 2 (remaining): beta from beta_start*50 to beta_end
            if progress < 0.4:
                p1 = progress / 0.4
                beta = beta_start * (50.0) ** p1
            else:
                p2 = (progress - 0.4) / 0.6
                beta = beta_start * 50.0 * (beta_end / (beta_start * 50.0)) ** p2
        else:
            beta = beta_start * (beta_end / beta_start) ** progress

        # Nesterov extrapolation
        if use_nesterov and step > 1:
            gamma = (step - 1.0) / (step + 1.5)
            y = x + gamma * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        # Compute gradient
        grad, _ = smooth_max_gradient_fft(y, beta)

        # Scale step size with beta to keep updates reasonable
        grad_norm = np.abs(grad).max()
        if grad_norm > 1e12:
            grad = grad / grad_norm * 1e12

        # Mirror descent (multiplicative) update
        eta = c0 / (np.sqrt(step % restart_every + 1) + 1.0) / max(1.0, beta / 10.0)
        log_a = np.log(y) - eta * grad
        log_a = log_a - log_a.max()
        x_new = np.exp(log_a)
        x_new = np.maximum(x_new, 1e-15)
        s = x_new.sum()
        if s < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s

        x_prev = x
        x = x_new

        # Evaluate
        M = fast_autocorr(x).max()
        if M < best_M - 1e-15:
            best_M = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if verbose and step % 10000 == 0:
            J = 2.0 * n * best_M
            print(f"    SMD step={step} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

        if stall > max_stall:
            stall = 0
            # Restart with perturbation from best
            sigma = 0.03 + 0.03 * (step // (max_stall * 10))
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

    # Final check
    for candidate in [x, x_prev]:
        M_c = fast_autocorr(candidate).max()
        if M_c < best_M:
            best_M = M_c
            best_a = candidate.copy()

    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, top_K=500, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top constraints
        if top_K >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:top_K]

        # Build constraint matrix
        T_sel = np.zeros((len(binding_idx), n))
        for idx, k in enumerate(binding_idx):
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                T_sel[idx, j] = y[ki - j]

        M_sel = M_curr * np.ones(len(binding_idx))

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, top_k=25, verbose=False, rng_seed=123):
    """Coordinate descent: move mass from peak-contributing to low-sensitivity positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:top_k]

        # Sensitivity: how much each position contributes to top peaks
        sens = np.zeros(n)
        for k in top_idx[:12]:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            for j in range(j_start, j_end):
                sens[j] += a[ki - j]

        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)

            # Find most contributing pair
            best_contrib = -1.0
            best_src = -1
            for j in range(j_start, j_end):
                c = a[j] * a[ki - j]
                if c > best_contrib:
                    best_contrib = c
                    best_src = j

            if best_src < 0 or best_contrib < 1e-20:
                continue

            # Find best destination (lowest sensitivity, not in peak region)
            min_sens = np.inf
            best_dst = -1
            for d in range(n):
                if d != best_src and sens[d] < min_sens:
                    min_sens = sens[d]
                    best_dst = d

            if best_dst < 0:
                continue

            for frac in [0.005, 0.01, 0.02, 0.003, 0.05]:
                amt = a[best_src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[best_src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random kick
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0 - 1.0 / old_n, old_n)
    new_x = np.linspace(0.0, 1.0 - 1.0 / new_n, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    Phase 1: Multi-init SMD (40% time)
    Phase 2: LP Frank-Wolfe (15%)
    Phase 3: Coordinate descent (10%)
    Phase 4: High-beta SMD polish (10%)
    Phase 5: LP round 2 (15%)
    Phase 6: Length exploration + final polish (10%)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = norm_sum1(parent)
            p_score = evaluate_safe(p_norm)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = p_norm
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_initialization(n_base, 'parabolic', 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init SMD ===
    print(f"\n=== Phase 1: SMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.40, 650.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_initialization(n0, 'parabolic', 1.0))
        labels.append("parab1")

        inits.append(make_initialization(n0, 'parabolic', 2.0))
        labels.append("parab2")

        inits.append(make_initialization(n0, 'quartic', 1.0))
        labels.append("quart")

        inits.append(make_initialization(n0, 'cosine_u'))
        labels.append("cosu")

        inits.append(make_initialization(n0, 'exp_u', 1.5))
        labels.append("expu")

        inits.append(make_initialization(n0, 'power3'))
        labels.append("pow3")

        inits.append(make_initialization(n0, 'power4'))
        labels.append("pow4")

        inits.append(make_initialization(n0, 'linear_u'))
        labels.append("linu")

        # Blend parent with shapes
        if parent is not None:
            p_norm = norm_sum1(parent)
            blend1 = 0.6 * p_norm + 0.4 * make_initialization(n0, 'parabolic', 1.5)
            inits.append(norm_sum1(blend1))
            labels.append("blend1")
            blend2 = 0.7 * p_norm + 0.3 * make_initialization(n0, 'power3')
            inits.append(norm_sum1(blend2))
            labels.append("blend2")

        # De-duplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 20)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  SMD '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = smooth_max_md(
                init, t_avail, beta_start=0.5, beta_end=400.0,
                c0=2.5, use_nesterov=True, restart_every=3000,
                verbose=False, rng_seed=42 + idx, two_phase=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"    *** NEW BEST [smd_{lbl}]: {sc:.6f}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        top_K = min(400, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, top_K=top_K, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"    *** NEW BEST [lp_main]: {sc:.6f}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, top_k=25, verbose=False, rng_seed=123)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"    *** NEW BEST [coord1]: {sc:.6f}")

    # === Phase 4: High-beta SMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SMD high-beta (n={n_best}) ===")
    t4 = min(remaining * 0.10, 170.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_max_md(
            a_in, t4, beta_start=50.0, beta_end=600.0,
            c0=1.0, use_nesterov=True, restart_every=5000,
            verbose=False, rng_seed=200, two_phase=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"    *** NEW BEST [smd_high]: {sc:.6f}")

    # === Phase 5: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP r2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        top_K = min(600, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, top_K=top_K, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"    *** NEW BEST [lp_r2]: {sc:.6f}")

    # === Phase 6: Length exploration + final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    trial_lengths = []
    for delta in [-1000, -500, 500, 1000, 2000, -2000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 80]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 150.0)
        if t7a > 20:
            a_out, _ = smooth_max_md(
                a_interp, t7a, beta_start=1.0, beta_end=300.0,
                c0=2.0, use_nesterov=True, restart_every=3000,
                verbose=False, rng_seed=new_n % 1000, two_phase=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"    *** NEW BEST [smd_n{new_n}]: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 5000:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                top_K = min(500, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, top_K=top_K, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"    *** NEW BEST [lp_n{new_n}]: {sc:.6f}")

    # === Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Final Polish (n={n_best}) ===")
        t_final = min(remaining * 0.5, 120.0)
        if t_final > 20:
            a_in = norm_sum1(best_seq)
            top_K = min(800, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t_final, solver, top_K=top_K, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"    *** NEW BEST [lp_final]: {sc:.6f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_cd = min(remaining * 0.6, 50.0)
            if t_cd > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t_cd, top_k=40, verbose=False, rng_seed=789)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"    *** NEW BEST [coord_final]: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_initialization(n_fb, 'parabolic', 1.0)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
