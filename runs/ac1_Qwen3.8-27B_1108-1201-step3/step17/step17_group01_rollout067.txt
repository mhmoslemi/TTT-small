import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient(a, ac, n_peaks=80, temperature=1.0):
    """Compute softmax-weighted multi-peak subgradient."""
    n = len(a)
    k_max = min(n_peaks, len(ac))
    # Use softmax weighting over all lags for smoother gradient
    ac_clipped = np.minimum(ac, ac.max())
    log_ac = np.log(np.maximum(ac_clipped, 1e-30))
    exp_vals = np.exp((log_ac - log_ac.max()) / temperature)
    w = exp_vals / exp_vals.sum()
    # Focus on top peaks but with smooth falloff
    grad = np.zeros(n)
    # Compute contribution from top peaks
    top_idx = np.argsort(ac)[::-1][:k_max]
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-10:
            continue
        # d c[ki]/d a[j] = 2*a[ki-j] for valid j
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo > j_hi:
            continue
        idx = np.arange(j_lo, j_hi + 1)
        pair_idx = ki - idx
        grad[idx] += 2.0 * wi * a[pair_idx]
    return grad


def grad_optimize(a0, time_limit, n_peaks=80, temperature0=3.0, c0=5.0,
                  restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_temp = temperature0
    restart_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_gradient(y, ac, n_peaks=n_peaks, temperature=cur_temp)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = ac.max()
        # Recompute from x to be safe
        M_x = fast_autocorr(x).max()
        M_eff = min(M_new, M_x)
        if M_eff < best_m - 1e-15:
            best_m = M_eff
            if M_x <= M_new:
                best_a = x.copy()
            else:
                best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adapt temperature: decrease over time to focus on hardest peaks
        cur_temp = max(0.8, temperature0 * (1.0 - 0.1 * min(stall_count / 2000.0, 1.0)))

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.01 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Vary parameters
            if restart_count % 3 == 0:
                n_peaks = min(n_peaks + 10, 150)
            cur_temp = max(0.8, temperature0 - 0.2 * restart_count)
            gamma = c0 * (1.0 + 0.05 * min(restart_count, 5))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    GRAD restart t={t} ({elapsed:.0f}s): J={J:.6f}, peaks={n_peaks}, temp={cur_temp:.2f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GRAD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, alpha0=0.3, iters=5000, verbose=False):
    """Gerchberg-Saxton flat spectrum optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Progressive: start with aggressive flattening, then refine
        alpha = alpha0 + (1.0 - alpha0) * (it / max(1, iters))
        P_target = P_mean * (P / np.maximum(P_mean, 1e-30)) ** (1.0 - alpha)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target / np.maximum(P, 1e-30))
        A_new = A * mag_new
        # But cap the modification to avoid instability
        mag_ratio = mag_new
        mag_ratio = np.clip(mag_ratio, 0.7, 1.4)
        A_new = A * mag_ratio

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped: take a smaller step
            blend = 0.3
            a = (1 - blend) * a + blend * a_new
            a = norm_sum1(a)
            M_curr = fast_autocorr(a).max()
            if M_curr < best_m:
                best_m = M_curr
                best_a = a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def peak_reduction_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Direct coordinate moves targeting peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Get top peaks
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            # Find the pair (i, ki-i) with maximum contribution
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo > j_hi:
                continue
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # The position with max contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find a good destination: position not heavily involved in other top peaks
            costs = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                j2_lo = max(0, k2i - n + 1)
                j2_hi = min(n - 1, k2i)
                valid = np.arange(j2_lo, j2_hi + 1)
                if len(valid) > 0:
                    costs[valid] += a[k2i - valid]

            # Pick destination with lowest cost among positions with low value
            candidates = np.argsort(costs + 10 * a)[:n // 3]
            best_dst = candidates[np.argmin(a[candidates])]
            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            # Try moving a fraction of mass
            found = False
            for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Try random small move from src
                if a[src] > 1e-10:
                    dst = int(rng.integers(n))
                    if dst != src:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def make_power_law_ramp(n, power=0.5):
    """Increasing ramp: a[i] = (i+1)^power, normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_mild_ramp(n, center_frac=0.3):
    """1 + c*i/n shape, normalized."""
    x = np.linspace(0, 1, n)
    a = 1.0 + center_frac * x
    return norm_sum1(a)


def make_exponential_ramp(n, rate=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return norm_sum1(a)


def make_alg_ramp(n, alpha=1.5):
    """a[i] = 1/(n-i+alpha) type: increasing towards the right."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (n - i + alpha)
    return norm_sum1(a)


def make_two_phase(n, frac=0.6, alpha=0.3):
    """Flat on left, ramp on right."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    mask = x > frac
    a[mask] = 1.0 + alpha * (x[mask] - frac) / (1.0 - frac)
    return norm_sum1(a)

    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_descent(a_in, t3, verbose=False, rng_seed=42)
        try_update(a_out, f"peakred_{n_best}")

    # === Phase 4: Aggressive gradient with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Aggressive gradient (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = grad_optimize(a_in, t4, n_peaks=120, temperature0=1.5,
                                  c0=6.0, restart_every=3000, verbose=True,
                                  rng_seed=7)
        try_update(a_out, f"grad_agg_{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length search ===")

    lengths_to_try = []
    for delta in [-500, -1000, 500, 1000, 2000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 10000:
            lengths_to_try.append(new_n)
    lengths_to_try = sorted(set(lengths_to_try))

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5a = min(remaining * 0.30, 150.0)
        if t5a > 20:
            a_out, _ = grad_optimize(a_long, t5a, n_peaks=80, temperature0=3.0,
                                      c0=5.0, restart_every=5000, verbose=False,
                                      rng_seed=42)
            try_update(a_out, f"grad_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.20, 80.0)
            if t5b > 15:
                a_out, _ = spectral_flatten(a_long, t5b, alpha0=0.4, verbose=False)
                try_update(a_out, f"spectral_n{new_n}")

    # === Phase 6: Final gradient polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.40, 200.0)
        if t6 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = grad_optimize(a_in, t6, n_peaks=150, temperature0=1.0,
                                      c0=3.0, restart_every=8000, verbose=True,
                                      rng_seed=123)
            try_update(a_out, f"grad_final_{n_best}")

    # === Phase 7: Another peak reduction pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.30, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_reduction_descent(a_in, t7, verbose=False, rng_seed=777)
            try_update(a_out, f"peakred_final_{n_best}")

    # === Phase 8: Final spectral + grad combo ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8a = min(remaining * 0.40, 80.0)
        t8b = (remaining - t8a) * 0.8
        if t8a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_flatten(a_in, t8a, alpha0=0.5, verbose=False)
            try_update(a_out, f"spectral_final_{n_best}")
        if t8b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = grad_optimize(a_in, t8b, n_peaks=100, temperature0=1.5,
                                      c0=4.0, restart_every=4000, verbose=False,
                                      rng_seed=555)
            try_update(a_out, f"grad_very_final_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_power_law_ramp(n_fb, power=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
