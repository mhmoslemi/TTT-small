import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient(a, ac, top_k=50, sharpness=4.0):
    """Compute weighted subgradient focusing on top peaks with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = np.exp(sharpness * (top_vals - top_vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n - 1, ki)
        if s > e:
            continue
        # g[s:e+1] += 2 * wi * a[ki-s : ki-e-1 : -1]
        indices = ki - np.arange(s, e + 1)
        g[s:e + 1] += 2.0 * wi * a[indices]
    return g


def amd_optimize(a0, time_limit, top_k=60, sharpness=5.0, c0=4.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent (multiplicative weights)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    cur_sharpness = sharpness
    cur_top_k = top_k

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_gradient(y, ac, top_k=cur_top_k, sharpness=cur_sharpness)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameters
        if stall_count > restart_every // 3:
            cur_sharpness = min(cur_sharpness * 1.05, 15.0)
            cur_top_k = min(cur_top_k + 3, 200)

        if stall_count > restart_every:
            stall_count = 0
            # Perturb from best
            noise_scale = min(0.05, 0.02 * (1.0 + stall_count / 1000.0))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_sharpness = max(cur_sharpness - 1.0, 3.0)
            cur_top_k = min(cur_top_k + 10, 250)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=None, mode="m_min"):
    """
    LP-based Frank-Wolfe.
    mode='m_min': minimize M subject to a_old * a <= M, sum(a)=1, a>=0
    mode='fw': maximize sum(b) subject to a_old * b <= M, sum(b)=1, b>=0
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if top_K is None:
        top_K = min(2000, 2 * n - 1)

    K_sel = min(top_K, 2 * n - 1)
    dual_vars = np.zeros(2 * n - 1)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive constraint selection: use dual variables to pick most important constraints
        constraint_margin = M_curr - ac
        # Blend: use top-K by margin, but also include constraints with high dual values
        top_by_margin = np.argsort(constraint_margin)[:K_sel]
        
        # Also include constraints with highest dual values
        if t > 5:
            top_by_dual = np.argsort(dual_vars)[::-1][:max(50, K_sel // 5)]
            combined = np.unique(np.concatenate([top_by_margin, top_by_dual]))
            binding_idx = combined[:K_sel]
        else:
            binding_idx = top_by_margin

        # Build constraint matrix: T[i, j] = y[k - j] for constraint i (peak at k_i)
        # a_old * b at lag k = sum_j y[k-j] * b[j]
        T_sel = np.zeros((len(binding_idx), n))
        for ii, k in enumerate(binding_idx):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            if s <= e:
                indices = ki - np.arange(s, e + 1)
                T_sel[ii, s:e + 1] = y[indices]

        if mode == "m_min":
            a_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [T_sel @ a_var <= M_var, cp.sum(a_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
        else:
            b_var = cp.Variable(n, nonneg=True)
            M_val = M_curr
            constraints = [T_sel @ b_var <= M_val, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        if mode == "m_min":
            a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
            M_lp = float(M_var.value)
            # Update dual variables
            if prob.dual_values is not None:
                try:
                    dv = np.asarray(prob.dual_values, dtype=np.float64)
                    # Extract duals for the T_sel @ a_var <= M_var constraints
                    n_duals = len(dv)
                    if n_duals >= len(binding_idx):
                        # The first len(binding_idx) duals correspond to our constraints
                        for ii, k in enumerate(binding_idx):
                            dual_vars[int(k)] = max(0.0, float(dv[ii]))
                except Exception:
                    pass
        else:
            a_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

        a_sum = a_val.sum()
        if a_sum < 1e-15:
            continue
        g_dir = a_val / a_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Gradually increase K
        if t % 15 == 0 and K_sel < 2 * n - 1:
            K_sel = min(K_sel + 100, 2 * n - 1)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=123, verbose=False):
    """Sensitivity-guided coordinate descent with multi-step moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100
    n_top = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            partner = ki - idx
            sens[idx] += a[partner]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination outside peak regions
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]
            if dst == src:
                dst = candidates[np.argmin(cand_sens[candidates != dst if len(candidates) > 1 else candidates])]
                if dst == src:
                    continue

            found = False
            for frac in [0.02, 0.01, 0.005, 0.03, 0.05, 0.08, 0.15, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found and stall > 50:
                # Random exploration
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.005, 0.05)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Multi-point perturbation
            n_pert = min(8, n)
            pert = rng.normal(0, 0.015, n) * a
            a_new = a + pert
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-15:
                a_new = a_new / a_new.sum()
                M_new = fast_autocorr(a_new).max()
                if M_new < best_m:
                    a = a_new
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.12) / width) ** 2) + np.exp(-((x - 0.88) / width) ** 2) + 0.003
    return norm_sum1(a)


def make_w_shape(n, center_width=0.1):
    """W-shaped: high at edges, dip in center, small bumps."""
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - 0.05) / 0.1) ** 2) + np.exp(-((x - 0.95) / 0.1) ** 2) +
         np.exp(-((x - 0.3) / center_width) ** 2) * 0.3 +
         np.exp(-((x - 0.7) / center_width) ** 2) * 0.3 + 0.002)
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.2 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def peak_pair_reduction(a0, time_limit, rng_seed=77, verbose=False):
    """Targeted peak reduction via redistribution from most-contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 150
    n_peaks = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            sens[idx] += a[partner]

        improved = False

        for k in top_k[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            best_i = np.argmax(contrib)
            src = idx[best_i]
            src_part = partner[best_i]

            imp_src = sens[src]
            imp_part = sens[src_part]
            reduce_idx = src if imp_src >= imp_part else src_part
            keep_idx = src_part if imp_src >= imp_part else src

            if a[reduce_idx] < 1e-12:
                continue

            exclude = {reduce_idx, keep_idx}
            candidates = [d for d in range(n) if d not in exclude]
            if not candidates:
                candidates = list(range(n))
            cand_arr = np.array(candidates)
            cand_sens = sens[cand_arr]
            dst = cand_arr[np.argmin(cand_sens)]

            found = False
            for frac in [0.03, 0.015, 0.008, 0.05, 0.1, 0.2]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 40:
                j = int(rng.integers(n))
                if j != reduce_idx:
                    frac = rng.uniform(0.005, 0.04)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            pert = rng.normal(0, 0.02, n) * a
            a_new = a + pert
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-15:
                a_new = a_new / a_new.sum()
                M_new = fast_autocorr(a_new).max()
                if M_new < best_m:
                    a = a_new
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakPair ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3500
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init AMD ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.28, 500.0)

    if t_phase1 > 100:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")

        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")

        inits.append(make_bimodal(n0, width=0.12))
        labels.append("bimodal")

        inits.append(make_w_shape(n0, center_width=0.08))
        labels.append("wshape")

        inits.append(spectral_init(n0, n_modes=40, seed=42))
        labels.append("spectral")

        inits.append(spectral_init(n0, n_modes=15, seed=7))
        labels.append("spectral2")

        if parent is not None:
            blend = 0.4 * norm_sum1(parent) + 0.6 * make_u_shape(n0, 2.5)
            inits.append(norm_sum1(blend))
            labels.append("blend_u")

            blend2 = 0.7 * norm_sum1(parent) + 0.3 * make_bimodal(n0, 0.1)
            inits.append(norm_sum1(blend2))
            labels.append("blend_b")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 40)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(30.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, sharpness=6.0, c0=5.0,
                                     restart_every=4000, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW (M-min mode) on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW M-min (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 450.0)
    if t2 > 50 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        top_K = min(2500, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=top_K, mode="m_min")
        try_update(a_out, "lp_mmin")

    # === Phase 3: AMD refinement on updated best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AMD refinement (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, top_k=100, sharpness=8.0, c0=3.0,
                                 restart_every=6000, verbose=False, rng_seed=200)
        try_update(a_out, "amd_refine")

    # === Phase 4: LP-FW round 2 (standard FW mode) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW standard (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        top_K = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=top_K, mode="fw")
        try_update(a_out, "lp_fw")

    # === Phase 5: Peak-pair local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Peak-pair (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_pair_reduction(a_in, t5, verbose=False, rng_seed=77)
        try_update(a_out, "peakpair")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coord descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=False, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: LP-FW final push ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LP-FW final (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        top_K = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=top_K, mode="m_min")
        try_update(a_out, "lp_final")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 2000, -1000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.35, 180.0)
        if t8a > 30:
            a_out, _ = amd_optimize(a_long, t8a, top_k=60, sharpness=6.0, c0=4.0,
                                     restart_every=5000, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4000:
            t8b = min(remaining * 0.30, 120.0)
            if t8b > 25:
                top_K = min(2000, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_long, t8b, solver, verbose=False, top_K=top_K, mode="m_min")
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final local polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.4, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_pair_reduction(a_in, t9, verbose=False, rng_seed=999)
            try_update(a_out, "peakpair_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t9b = min(remaining * 0.5, 40.0)
            if t9b > 8:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t9b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
