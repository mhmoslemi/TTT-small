```python
import numpy as np
import time

def fast_autocorr(a):
    # FFT-based autocorrelation
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_J(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

# Coordinate descent
def coord_descent(a, time_limit, verbose=False, seed=42):
    n = len(a)
    a = norm_sum1(a).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = len(range(n))  # one full sweep without improvement
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        # Random order for this sweep
        order = rng.permutation(n)
        
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try several candidate values
            current_val = a[i]
            candidates = [0.0, current_val * 0.5, current_val * 0.1, 0.01 * a.mean()]
            # Also try increasing
            candidates.append(current_val * 1.5)
            candidates.append(current_val * 2.0)
            # Clip to reasonable range
            candidates = [max(0.0, min(c, 1.0)) for c in candidates]
            # Remove duplicates
            seen = set()
            unique_candidates = []
            for c in candidates:
                if c not in seen:
                    seen.add(c)
                    unique_candidates.append(c)
            
            best_cand = current_val
            best_local_J = best_J  # Current is always an option
            
            for c in unique_candidates:
                if abs(c - current_val) < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[i] = c
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_t = compute_J(a_trial)
                if J_t < best_local_J - 1e-12:
                    best_local_J = J_t
                    best_cand = c
            
            if best_cand != current_val:
                a[i] = best_cand
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = norm_sum1(best_a)
                    break
                a = a / s
                if best_local_J < best_J - 1e-12:
                    best_J = best_local_J
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    
    return best_a, best_J
```

When I change a[i] from x0 to x:

```python
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0

    best_a = a.copy()
    best_J = J
    best_S = S
    best_ac = ac.copy()

    max_sweeps_no_improve = 10

        order = rng.permutation(n)
        improved_this_sweep = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            x0 = a[i]
            if x0 < 1e-15:
                # Try adding some mass
                candidates = [1e-4, 1e-3, 5e-3]

                candidates = [0.0, x0 * 0.5, x0 * 0.1, x0 * 0.8, x0 * 1.2, x0 * 1.5]
                # Add some absolute values
                avg = S / n
                candidates.extend([avg * 0.5, avg * 1.5, avg])
            
            best_x = x0
            best_trial_J = J
            
            for x in candidates:
                if x < -1e-15:
                    continue
                x = max(0.0, x)
                if abs(x - x0) < 1e-15:
                    continue
                
                # Compute new ac and J
                delta = x - x0
                S_new = S + delta
                
                if S_new < 1e-15:
                    continue
                
                # Update ac
                # Affected lags: k in [i, i+n-1] intersect [0, 2n-2]
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)

                #   ac_new[k] = ac[k] + 2*delta*a[k-i]
                # For k = 2i (if in range):
                #   ac_new[2i] = ac[2i] + delta*(x + x0) = ac[2i] + (x^2 - x0^2)

                # Affected lags: ac[k] + 2*delta*a[k-i] (or + x^2-x0^2 for k=2i)
                
                # To avoid creating a new array, compute max on the fly
                max_unaffected = 0.0
                max_affected = 0.0
                
                # This is O(n) per candidate, which is fine
                
                for k in range(2*n-1):
                    if k_lo <= k <= k_hi:
                        if k == 2*i and 2*i < 2*n-1:
                            val = ac[k] + (x*x - x0*x0)
                        else:
                            # a[k-i] is valid since k-i is in [k_lo-i, k_hi-i] = [0, n-1]
                            val = ac[k] + 2.0 * delta * a[k-i]
                        if val > max_affected:
                            max_affected = val
                    else:
                        if ac[k] > max_unaffected:
                            max_unaffected = ac[k]
                
                max_ac_new = max(max_unaffected, max_affected)
                J_trial = 2.0 * n * max_ac_new / (S_new * S_new)
                
                if J_trial < best_trial_J - 1e-14:
                    best_trial_J = J_trial
                    best_x = x
            
            if best_x != x0:
                x0_old = a[i]
                delta = best_x - x0_old
                a[i] = best_x
                S += delta
                
                # Update ac
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)

                    else:
                        ac[k] += 2.0 * delta * a[k-i]

                    best_S = S

    return best_a
```
