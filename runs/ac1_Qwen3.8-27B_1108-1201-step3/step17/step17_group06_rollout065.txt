    t = t0.copy()
    t0_time = time.monotonic()
    
    def obj_and_grad(t):
        # Compute a from t using softmax
        t_shifted = t - t.max()
        exp_t = np.exp(t_shifted)
        a = exp_t / exp_t.sum()
        
        # Compute convolution
        conv = fast_conv(a)
        
        # Soft max
        m = conv.max()
        shifted = beta * (conv - m)
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        w = exp_shifted / sum_exp
        S = m + np.log(sum_exp) / beta
        
        # Gradient of S w.r.t. a: dS/da_i = 2 * sum_j a[j] * w[i+j]
        # Compute using FFT
        n_conv = 2*n - 1
        fft_size = 1 << (n_conv - 1).bit_length()
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, fft_size)
        W = np.fft.rfft(w, fft_size)
        conv_result = np.fft.irfft(A_rev * W, fft_size)[:n_conv]
        # result[i] = conv_result[n-1+i] for i=0,...,n-1
        dS_da = 2.0 * conv_result[n-1:n-1+n]
        
        # J = 2n * S
        J = 2.0 * n * S
        
        # dJ/dt_i = a_i * (dJ/da_i - sum_j dJ/da_j * a_j)
        dJ_da = 2.0 * n * dS_da
        sum_dJ_da_a = np.dot(dJ_da, a)
        grad_t = a * (dJ_da - sum_dJ_da_a)
        
        return J, grad_t
    
    result = minimize(obj_and_grad, t, method='L-BFGS-B',
                      jac=True,

    a = np.exp(t_shifted)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B in log-domain with log-sum-exp smoothing and beta annealing
2. FFT-based gradient computation via cross-correlation
3. Nesterov mirror descent with adaptive top-K subgradients
4. Coordinate descent for final polish
5. Multi-start with diverse initializations
6. Length variation search
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    conv = fast_conv(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def softmax_obj_grad(t, beta, n):
    t_sh = t - t.max()
    et = np.exp(t_sh)
    a = et / et.sum()
    
    conv = fast_conv(a)
    m = conv.max()
    sh = beta * (conv - m)
    es = np.exp(sh)
    se = es.sum()
    w = es / se
    S = m + np.log(se) / beta
    
    # dS/da_i = 2 * sum_j a[j] * w[i+j]
    # = 2 * conv(reverse(a), w)[n-1+i]
    n_conv = 2 * n - 1
    fft_sz = 1 << (3 * n - 3).bit_length()
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, fft_sz)
    W = np.fft.rfft(w, fft_sz)
    cross = np.fft.irfft(Ar * W, fft_sz)
    dS_da = 2.0 * cross[n-1 : n-1+n]
    
    J = 2.0 * n * S
    dJ_da = 2.0 * n * dS_da
    sda = np.dot(dJ_da, a)
    grad = a * (dJ_da - sda)
    return J, grad

def lgbfsm(t0, n, beta, maxiter=5000):
    def fg(t):
        return softmax_obj_grad(t, beta, n)
    res = minimize(fg, t0, jac=True, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'maxfun': maxiter*5,
                            'ftol': 1e-14, 'gtol': 1e-10})
    tf = res.x
    ts = tf - tf.max()
    a = np.exp(ts)
    a /= a.sum()
    return a

def to_log(a):
    a = np.maximum(a, 1e-15)
    t = np.log(a)
    return t - t.max()

def mirror_descent(a0, time_limit, top_k=100, beta_w=10.0, c0=4.0):

    a = np.maximum(a0, 0.0).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    ck = top_k
    cb = beta_w
    cc = c0
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t > 1:
            bt = (t - 2.0) / (t + 1.0)
            y = x + bt * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y /= y.sum()
        conv = fast_conv(y)
        cl = 2*n - 1
        km = min(ck, cl)
        ti = np.argsort(conv)[::-1][:km]
        tv = np.maximum(conv[ti], 1e-30)
        w = tv ** cb
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(km) / km
        else:
            w /= ws
        g = np.zeros(n)

            is_ = max(0, ki - n + 1)
            ie = min(n-1, ki)
            if is_ <= ie:
                idx = np.arange(is_, ie+1)
                g[idx] += 2.0 * wi * y[ki - idx]
        eta = cc / (np.sqrt(t) + 1.0)
        lr = np.log(y) - eta * g
        lr -= lr.max()
        an = np.exp(lr)
        an = np.maximum(an, 1e-30)
        an /= an.sum()
        at = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - at) * x + at * an
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        val = eval_J(an)
        if val < best_val - 1e-15:
            best_val = val
            best_a = an.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8000:
            stall = 0
            sc = rng.uniform(0.005, 0.03)

            x = xr.copy()
            x_prev = xr.copy()
            ck = min(ck + 20, 250)
            cb = max(cb - 0.5, 2.0)

    return best_a, best_val

            idx = np.arange(is_, ie+1)
            pr[idx] += a[idx] * a[ki - idx]
        improved = False
        sc = np.argsort(-pr * a)[:5]
        for src in sc:
            if a[src] < 1e-12:
                continue
            dp = np.zeros(n)
            for k in tl[:5]:
                ki = int(k)
                is_ = max(0, ki-n+1)
                ie = min(n-1, ki)

            dp[src] = np.inf
            dst = int(np.argmin(dp))
            if dst == src:

            for fr in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
                am = a[src] * fr
                if am < 1e-15:

                at = a.copy()
                at[src] -= am
                at[dst] += am
                at = np.maximum(at, 0.0)
                at /= at.sum()
                Jt = eval_J(at)
                if Jt < best_J - 1e-14:
                    a = at
                    best_J = Jt
                    best_a = a.copy()

                    a[j] += am

                    Jn = eval_J(a)
                    if Jn < best_J:

                        best_a = a.copy()
                        stall = 0
                        break

def make_ramp(n, p=1.0):
    x = np.arange(1, n+1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_exp(n, alpha=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(alpha * x)
    return a / a.sum()

def make_sine(n, p=1.0, phase=0.0):
    x = np.linspace(0, 1, n)
    a = (1.0 + np.sin(np.pi * x + phase)) ** p
    a = np.maximum(a, 0.01)
    return a / a.sum()

def make_bimodal(n, c1=0.2, c2=0.8, w=20.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-w*(x-c1)**2) + 0.5*np.exp(-w*(x-c2)**2) + 0.01
    return a / a.sum()

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        a += rng.normal(0, 0.08) * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0.01)
    return a / a.sum()

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    ai = np.interp(nx, ox, a)
    ai = np.maximum(ai, 0.0)
    s = ai.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return ai / s
