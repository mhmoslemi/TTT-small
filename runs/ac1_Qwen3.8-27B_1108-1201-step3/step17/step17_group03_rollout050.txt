"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using non-symmetric power-law
initializations combined with LP Frank-Wolfe optimization and coordinate
descent. Key insight: non-symmetric sequences (ramps) can achieve J ~ 4/3,
far below the J=2 barrier for symmetric sequences.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_ramp(n, power=1.0):
    """Create a power-law ramp: a[i] proportional to (i+1)^power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return norm_sum1(a)


def make_ramp_desc(n, power=1.0):
    """Descending ramp."""
    i = np.arange(n, 0, -1, dtype=np.float64)
    a = i ** power
    return norm_sum1(a)


def make_two_sided(n, power=1.0, alpha=0.3):
    """Two-sided: rise then fall, asymmetric."""
    i = np.arange(n, dtype=np.float64)
    center = n * alpha
    a = (i + 1) ** power * (n - i) ** power
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3, max_iters=50000):
    """LP Frank-Wolfe: iteratively solve LP subproblem and move toward solution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        # Build constraint matrix: T[row, col] = a[k - col] for valid indices
        # conv(a, b)[k] = sum_i a[i] * b[k-i]
        # For fixed k, this is sum_i b[i] * a[k-i] (when k-i is valid)
        # So T[row, i] = a[k - i] if 0 <= k-i < n else 0
        
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_i = int(k_int)
            # b[j] appears with coefficient a[k_i - j] when 0 <= k_i - j < n
            # i.e., j in [max(0, k_i-n+1), min(n-1, k_i)]
            j_start = max(0, k_i - n + 1)
            j_end = min(n, k_i + 1)
            for j in range(j_start, j_end):
                T_sel[row, j] = a[k_i - j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with diminishing step size
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sa

        m_new = fast_conv(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            # Tighten margin when improving
            margin_frac = max(margin_frac * 0.85, 1e-5)

        # Also check the LP solution directly
        m_lp = fast_conv(g_dir).max()
        if m_lp < best_m:
            best_m = m_lp
            best_a = g_dir.copy()

        a = a_new

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-crushing coordinate descent: move mass from high-contributing
    positions to random positions to reduce the convolution peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Find contributing pairs
        j_start = max(0, peak_k - n + 1)
        j_end = min(n, peak_k + 1)
        js = np.arange(j_start, j_end)
        contrib = a[js] * a[peak_k - js]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        # Move mass from top contributors
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(js[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 40
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDesc final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def nesterov_mirror(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                    restart_every=3000, verbose=False):
    """Nesterov mirror descent with backtracking for the nonconvex problem."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        # Soft subgradient: weight top peaks
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            idx = np.arange(j_start, j_end)
            grad[idx] += 2.0 * wi * y[k_int - idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        base_m = ac.max()
        accepted = False
        for _ in range(6):
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_conv(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5

        if not accepted:
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 80)
            cur_beta = max(cur_beta - 0.5, 2.5)

    return best_a, best_m

    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord descent (n={n_best}) ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 15:
        a_out, _ = coordinate_descent(best_seq, t8, verbose=True, rng_seed=777)
        try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_ramp(n, 1.0)
        best_score = evaluate_safe(best_seq)

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
