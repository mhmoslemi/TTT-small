def lse_gradient(a, p, N):
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(A) * P, N)

```python
"""
Search for a non-negative step-function height sequence minimizing

    2 * n * max(np.convolve(a, a)) / (sum(a))**2.

The main optimizer is a fast annealed log-sum-exp mirror-descent method.
Instead of sampling only the largest convolution lags, this implementation
uses the exact gradient of a smooth surrogate

    F_tau(a) = (1/tau) * logsumexp(tau * conv(a, a))

and evaluates that gradient in O(N log N) using FFT:
if p = softmax(tau * conv(a,a)), then
    grad_a F_tau = 2 * cross_corr(a, p)[:n].

This gives a smooth, dense, and relatively accurate version of the
max-gradient, much faster than dense O(n^2) gradient evaluation.
The search is combined with LP-based Remez polishing, symmetrization,
restart perturbations, and interpolation to a larger length for exploration.
"""

import numpy as np
import time

try:
    import cvxpy as cp
    HAVE_CP = True
except Exception:
    HAVE_CP = False

_global_solver = None


def get_solver():
    """Find a usable CVXPY solver, or return None if unavailable."""
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    if not HAVE_CP:
        return None

    candidates = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _global_solver = s
                return s
        except Exception:
            pass

    _global_solver = cp.SCS
    return cp.SCS


def conv_and_fft(a):
    """
    Return (conv(a,a), FFT(a), N) for linear self-convolution.
    N is a power of two large enough to avoid circular aliasing.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64), np.array([], dtype=np.complex128), 1
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:size]
    if np.min(b) < 0.0:
        b = np.maximum(b, 0.0)
    return b, A, N


def fast_conv(a):
    """Linear self-convolution via FFT."""
    b, _, _ = conv_and_fft(a)
    return b


def evaluate_safe(a):
    """Call the externally provided evaluate_sequence, defensively."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        if not np.all(np.isfinite(arr)):
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.sum() < 0.01:
            arr = arr.reshape(-1) * (1.0 / max(1e-15, arr.sum()))
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def norm_sum1(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, a.size)
    return a / s


def make_init(n, kind="triangle"):
    """Construct a smooth non-negative initialization of length n."""
    if n <= 0:
        return np.array([1.0])
    x = np.linspace(0.0, 1.0, int(n))
    if kind == "raised_cos":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == "gaussian":
        a = np.exp(-5.0 * ((x - 0.5) / 0.25) ** 2)
    elif kind == "flat":
        a = np.ones_like(x)
    elif kind == "quartic":
        z = 2.0 * x - 1.0
        a = np.maximum(1.0 - z ** 4, 0.0)
    else:  # triangle
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 1e-12)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    f = np.asarray(f, dtype=np.float64).ravel()
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([1.0])
    old_n = f.size
    if old_n == new_n:
        return norm_sum1(f)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f = f / s

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    s2 = out.sum()
    if s2 < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / s2


def symmetrize(a):
    """Symmetrize a sequence: (a + reverse(a)) / 2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    return norm_sum1(0.5 * (a + a[::-1]))


def perturb_sequence(a, scale, rng, kind):
    """Create a randomized perturbation of a normalized sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1:
        return norm_sum1(a)

    if kind == 0:
        a2 = a * np.exp(scale * rng.normal(0.0, 1.0, n))
    elif kind == 1:
        a2 = a.copy()
        k = min(max(1, n // 4), 800)
        idx = rng.choice(n, size=k, replace=False)
        a2[idx] *= rng.uniform(0.5, 1.5, size=k)
    elif kind == 2:
        a2 = a.copy()
        lo = max(1, n // 4)
        hi = min(n - 1, 3 * n // 4)
        if hi <= lo:
            center = n // 2
        else:
            center = int(rng.integers(lo, hi + 1))
        width = max(3, n // 30)
        i0 = max(0, center - width)
        i1 = min(n, center + width + 1)
        a2[i0:i1] *= rng.uniform(0.7, 1.3, size=max(1, i1 - i0))
    else:
        a2 = 0.5 * a + 0.5 * a[::-1]

    a2 = np.maximum(a2, 1e-300)
    s = a2.sum()
    if s < 1e-15:
        return norm_sum1(a)
    return a2 / s


def consider_candidate(seq, current_best, current_score, label):
    """Evaluate a candidate exactly and update the best if improved."""
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return current_best, current_score

    if arr.size == 0:
        return current_best, current_score

    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 1e-12:
        return current_best, current_score
    arr = arr / s

    sc = evaluate_safe(arr)
    if np.isfinite(sc):
        if current_best is None or not np.isfinite(current_score) or sc < current_score - 1e-15:
            print(f"  *** NEW BEST [{label}]: {sc:.10f}  n={arr.size}")
            return arr.copy(), sc

    if current_best is None:
        print(f"  initialize [{label}]: score={sc if np.isfinite(sc) else 'inf'}  n={arr.size}")
        return arr.copy(), sc

    return current_best, current_score


def lse_mirror_descent(
    a0,
    time_limit,
    tau_start=None,
    tau_max=None,
    c0=2.0,
    verbose=False,
    seed=11,
    max_stall=12000,
):
    """
    Annealed log-sum-exp mirror descent.

    Uses exact O(N log N) gradients for the smooth surrogate
    (1/tau) logsumexp(tau * conv(a,a)).
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    if n == 0:
        return np.array([1.0]), float("inf")

    a = norm_sum1(a0)
    b = fast_conv(a)
    best_a = a.copy()
    best_m = float(np.max(b)) if b.size else 0.0

    t0 = time.monotonic()
    t = 0
    stall = 0
    cur_max_stall = int(max(max_stall, 1000))
    c = float(c0)
    rng = np.random.default_rng(int(seed + n))

    if tau_start is None:
        tau_start = 25000.0 * float(n) / 2600.0
    if tau_max is None:
        tau_max = max(500000.0, float(tau_start) * 40.0)

    tau = float(tau_start)
    tau_inc = max(1e-9, (float(tau_max) - float(tau_start)) / 150000.0)
    fracs = (1.0, 0.6, 0.35, 0.18, 0.09)

    while time.monotonic() - t0 < float(time_limit):
        t += 1
        tau = min(float(tau_max), tau + tau_inc)

        b, A, N = conv_and_fft(a)
        if b.size == 0:
            break
        cur_m = float(np.max(b))
        if cur_m < best_m - 1e-14:
            best_m = cur_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > cur_max_stall:
            kind = int(rng.integers(4))
            scale = 0.005 + 0.035 * float(rng.random())
            a = perturb_sequence(best_a, scale, rng, kind)
            c = max(0.25, c * 0.75)
            tau = max(float(tau_start), tau * 0.5)
            cur_max_stall = max(1500, int(cur_max_stall * 0.75))
            stall = 0
            continue

        tb = tau * b
        tb = tb - np.max(tb)
        p = np.exp(tb)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = p / ps

        P = np.fft.rfft(p, N)
        corr = np.fft.irfft(np.conj(A) * P, N)
        g = 2.0 * corr[:n]
        g = np.maximum(g, 0.0)
        gmax = float(np.max(g))

        if not np.isfinite(gmax) or gmax < 1e-300:
            stall += 100
            continue

        base_eta = c / ((np.sqrt(float(t)) + 1.0) * gmax)
        accepted = False
        last_cand = None
        last_m = None

        for frac in fracs:
            if time.monotonic() - t0 > float(time_limit):
                break
            step = base_eta * frac
            if step <= 0.0:
                continue

            cand = a * np.exp(-step * g)
            cand = np.maximum(cand, 1e-300)
            s = cand.sum()
            if s < 1e-15:
                continue
            cand = cand / s

            b_new = fast_conv(cand)
            m_new = float(np.max(b_new))
            last_cand = cand
            last_m = m_new

            if m_new < cur_m - 1e-14:
                a = cand
                accepted = True
                break

        if not accepted and last_cand is not None:
            a = last_cand

        if verbose and (t % 5000 == 0):
            elapsed = time.monotonic() - t0
            print(
                f"    LSE-MD t={t} ({elapsed:.1f}s) tau={tau:.3g} "
                f"best_score={2.0 * n * best_m:.10f} n={n} stall={stall}"
            )

    return best_a, best_m


def lp_remez_single(a, max_cons=80, r=0.2, solver=None):
    """
    One LP Remez step.

    Linearize the largest convolution lags around current a and solve:
      min M
      s.t. sum(a_new)=1,
           a_new >= 0,
           a_new <= trust,
           C a_new <= M + rhs for active lags.
    """
    if not HAVE_CP:
        return None

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return None

    b = fast_conv(a)
    if b.size == 0:
        return None

    K = int(max_cons)
    if K <= 0:
        return None

    if K >= b.size:
        active = np.arange(b.size)
    else:
        active = np.argpartition(b, -K)[-K:]
        amax = int(np.argmax(b))
        if amax not in active:
            active[0] = amax

    m = active.size
    C = np.zeros((m, n), dtype=np.float64)
    rhs = np.zeros(m, dtype=np.float64)

    for row, k in enumerate(active):
        k = int(k)
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j1 < j0:
            continue
        idx = np.arange(j0, j1 + 1)
        vals = 2.0 * a[k - idx]
        C[row, idx] = vals
        row_dot = float(np.dot(vals, a[idx]))
        rhs[row] = row_dot - b[k]

    avar = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    cons = [
        cp.sum(avar) == 1.0,
        C @ avar <= M + rhs,
        M >= 0.0,
    ]

    if r is not None and float(r) > 0.0:
        upper = a * (1.0 + float(r))
        upper = np.maximum(upper, 0.02 / float(n))
        cons.append(avar <= upper)

    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    a_new = np.asarray(avar.value, dtype=np.float64).ravel()
    if a_new.size != n or not np.all(np.isfinite(a_new)):
        return None

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s


def lp_remez_optimize(
    a0,
    time_limit,
    max_cons=80,
    r_init=0.2,
    solver=None,
    verbose=False,
):
    """LP Remez with line search, adaptive trust region, and perturb restarts."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a = norm_sum1(a0)
    b = fast_conv(a)
    cur_m = float(np.max(b)) if b.size else 0.0
    best_a = a.copy()
    best_m = cur_m

    if not HAVE_CP:
        return best_a, best_m

    t0 = time.monotonic()
    r = max(1e-4, float(r_init))
    no_imp = 0
    it = 0
    rng = np.random.default_rng(9991 + len(a))
    fracs = (1.0, 0.6, 0.35, 0.18)

    while time.monotonic() - t0 < float(time_limit):
        it += 1

        b = fast_conv(a)
        cur_m = float(np.max(b)) if b.size else 0.0
        if cur_m < best_m - 1e-14:
            best_m = cur_m
            best_a = a.copy()

        if it > 3 and cur_m > best_m * 1.03 + 1e-12:
            a = best_a.copy()
            cur_m = best_m

        if time.monotonic() - t0 > float(time_limit):
            break

        cand = lp_remez_single(a, max_cons=max_cons, r=r, solver=solver)
        accepted = False

        if cand is not None:
            for frac in fracs:
                if time.monotonic() - t0 > float(time_limit):
                    break
                trial = (1.0 - frac) * a + frac * cand
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                m = float(np.max(fast_conv(trial)))

                if m < cur_m - 1e-14:
                    a = trial
                    cur_m = m
                    best_a = trial.copy()
                    best_m = m
                    accepted = True
                    no_imp = 0
                    r = max(0.05, r * 0.9)
                    break

        if not accepted:
            no_imp += 1
            r = min(0.6, max(r * 1.25, 0.02))

            if no_imp > 15:
                if no_imp > 45:
                    a = best_a.copy()
                    r = 0.2
                else:
                    kind = int(rng.integers(4))
                    a = perturb_sequence(a, 0.025, rng, kind)
                    r = 0.3
                no_imp = 0

        if verbose and (it % 5 == 0):
            elapsed = time.monotonic() - t0
            print(
                f"    Remez it={it} ({elapsed:.1f}s) "
                f"best_score={2.0 * len(a) * best_m:.10f} n={len(a)} r={r:.4f} no_imp={no_imp}"
            )

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search driver.

    Phases:
      1. Annealed LSE mirror descent from the parent.
      2. LP Remez polish.
      3. Sharper LSE mirror descent.
      4. LP Remez polish.
      5. Final LSE.
      6. Larger-length interpolation and search.
      7. Final push on the globally best sequence.
    """
    t_global = time.monotonic()
    try:
        budget = min(float(budget_s) - 30.0, 1770.0)
    except Exception:
        budget = 1770.0

    if budget <= 30.0:
        return [1.0]

    best_seq = None
    best_score = float("inf")

    # Load previous construction if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            best_seq, best_score = consider_candidate(p, None, float("inf"), "parent")
    except Exception:
        pass

    if best_seq is None:
        base = make_init(2600, "triangle")
        best_seq, best_score = consider_candidate(base, None, float("inf"), "default")

    n0 = len(best_seq)
    print(f"Initial best: score={best_score:.10f} n={n0}")

    # Cheap symmetrization and shape checks.
    best_seq, best_score = consider_candidate(
        symmetrize(best_seq), best_seq, best_score, "sym"
    )
    for kind in ("raised_cos", "gaussian", "quartic"):
        init = make_init(n0, kind)
        best_seq, best_score = consider_candidate(
            init, best_seq, best_score, f"init_{kind}"
        )

    solver = get_solver()

    # Phase 1: broad annealed LSE mirror descent.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 120.0:
        tA = min(remaining * 0.25, 520.0)
        a_out, _ = lse_mirror_descent(
            best_seq,
            tA,
            tau_start=None,
            tau_max=None,
            c0=2.0,
            verbose=False,
            seed=101,
            max_stall=12000,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "LSE_phase1")

    # Phase 2: LP Remez polish.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 80.0 and HAVE_CP:
        tB = min(remaining * 0.10, 180.0)
        a_out, _ = lp_remez_optimize(
            best_seq,
            tB,
            max_cons=100,
            r_init=0.25,
            solver=solver,
            verbose=False,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "Remez_phase2")

    # Phase 3: sharper LSE mirror descent.
    remaining = budget - (time.monotonic() - t_global)
    ncur = len(best_seq)
    if remaining > 90.0:
        tC = min(remaining * 0.16, 300.0)
        tau_c = max(80000.0, 80000.0 * float(ncur) / 2600.0)
        a_out, _ = lse_mirror_descent(
            best_seq,
            tC,
            tau_start=tau_c,
            tau_max=max(600000.0, tau_c * 20.0),
            c0=1.4,
            verbose=False,
            seed=202,
            max_stall=9000,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "LSE_phase3")

    # Phase 4: LP Remez polish.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 70.0 and HAVE_CP:
        tD = min(remaining * 0.08, 130.0)
        a_out, _ = lp_remez_optimize(
            best_seq,
            tD,
            max_cons=120,
            r_init=0.1,
            solver=solver,
            verbose=False,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "Remez_phase4")

    # Phase 5: very sharp LSE mirror descent.
    remaining = budget - (time.monotonic() - t_global)
    ncur = len(best_seq)
    if remaining > 70.0:
        tE = min(remaining * 0.12, 220.0)
        tau_e = max(250000.0, 250000.0 * float(ncur) / 2600.0)
        a_out, _ = lse_mirror_descent(
            best_seq,
            tE,
            tau_start=tau_e,
            tau_max=max(1200000.0, tau_e * 8.0),
            c0=0.8,
            verbose=False,
            seed=303,
            max_stall=7000,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "LSE_phase5")

    # Phase 6: small LP polish.
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 50.0 and HAVE_CP:
        tF = min(remaining * 0.05, 80.0)
        a_out, _ = lp_remez_optimize(
            best_seq,
            tF,
            max_cons=150,
            r_init=0.04,
            solver=solver,
            verbose=False,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "Remez_phase6")

    # Phase 7: explore a larger length.
    remaining = budget - (time.monotonic() - t_global)
    n_alt = min(4096, max(len(best_seq) + 300, int(len(best_seq) * 1.5)))
    if n_alt > len(best_seq) + 100 and remaining > 240.0:
        a_alt = interpolate_sequence(best_seq, n_alt)
        best_seq, best_score = consider_candidate(a_alt, best_seq, best_score, "alt_interp")

        t_alt = min(remaining * 0.55, 280.0)
        a_out, _ = lse_mirror_descent(
            a_alt,
            t_alt,
            tau_start=None,
            tau_max=None,
            c0=2.0,
            verbose=False,
            seed=404,
            max_stall=12000,
        )
        best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "LSE_alt")

        remaining = budget - (time.monotonic() - t_global)
        if remaining > 60.0 and HAVE_CP:
            t_alt_lp = min(remaining * 0.35, 110.0)
            a_out, _ = lp_remez_optimize(
                best_seq,
                t_alt_lp,
                max_cons=110,
                r_init=0.15,
                solver=solver,
                verbose=False,
            )
            best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "Remez_alt")

    # Phase 8: final push on the best sequence.
    remaining = budget - (time.monotonic() - t_global)
    nfin = len(best_seq)
    if remaining > 45.0:
        tF1 = min(remaining * 0.55, remaining - 12.0)
        if tF1 > 10.0:
            tau_f = max(400000.0, 400000.0 * float(nfin) / 2600.0)
            a_out, _ = lse_mirror_descent(
                best_seq,
                tF1,
                tau_start=tau_f,
                tau_max=max(1500000.0, tau_f * 6.0),
                c0=0.55,
                verbose=False,
                seed=505,
                max_stall=6000,
            )
            best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "LSE_final")

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 35.0 and HAVE_CP:
        tL = min(remaining - 6.0, 90.0)
        if tL > 10.0:
            a_out, _ = lp_remez_optimize(
                best_seq,
                tL,
                max_cons=160,
                r_init=0.03,
                solver=solver,
                verbose=False,
            )
            best_seq, best_score = consider_candidate(a_out, best_seq, best_score, "Remez_final")

    # Final symmetrization check.
    best_seq, best_score = consider_candidate(
        symmetrize(best_seq), best_seq, best_score, "final_sym"
    )

    if best_seq is None:
        return [1.0]

    arr = np.asarray(best_seq, dtype=np.float64).ravel()
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if not np.all(np.isfinite(arr)) or s < 0.01:
        return [1.0]
    arr = arr / s

    # One last exact validation. If it fails, return a trivial valid sequence.
    sc = evaluate_safe(arr)
    if not np.isfinite(sc):
        return [1.0]

    return [float(x) for x in arr]
```
