Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full softmax subgradient (avoiding top-K approximation)
2. Nesterov mirror descent with temperature annealing
3. Adam-projected gradient descent for local refinement
4. Super-Gaussian and multi-scale initialization
5. Adaptive length sweeping with interpolation
6. Two-phase CD: pressure-based (global) + pair-search (local)

import numpy as np
import time
from scipy import signal

# --- Core computation ---

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)

def softmax_gradient(a, tau=20.0):
    """Compute gradient of smoothed max of conv(a,a)."""
    n = len(a)
    conv = fast_conv(a)
    max_c = conv.max()
    exp_c = np.exp(tau * (conv - max_c))
    p = exp_c / exp_c.sum()
    # g[j] = 2 * sum_k p[k] * a[k-j] for 0 <= k-j < n
    # = 2 * correlate(p, a, mode='valid')
    grad = 2.0 * signal.correlate(p, a, mode='valid')
    return grad, conv, p

def mirror_descent_full(a0, time_limit, tau0=50.0, tau_end=5.0, c0=3.0,
                        restart_every=5000, n_iters_max=5000000):
    """Mirror descent with full softmax subgradient and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, t / (2 * restart_every))
        tau = tau0 * (tau_end / tau0) ** progress
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Full softmax gradient
        g, conv, p = softmax_gradient(y, tau=tau)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror step
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.02)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            c0 = max(c0 * 0.92, 1.0)
    
    return best_a, best_val

        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with perturbation
        if stall > restart_every:
            stall = 0
            total_restarts += 1
            scale = rng.uniform(0.003, 0.015)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            c0 = max(c0 * 0.95, 0.8)
            if verbose:
                print(f"      MD restart {total_restarts} t={t}: J={best_val:.6f}, tau={tau:.1f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={t} ({elapsed:.0f}/{time_limit:.0f}s): J={best_val:.6f}, tau={tau:.1f}")
    
    return best_a, best_val

def adam_projected(a0, time_limit, lr=0.05, tau=30.0, batch=1, verbose=False):
    """Adam optimizer with full softmax gradient, projected to simplex."""
    n = len(a0)
    a = normalize_simplex(a0).copy()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        tau_cur = max(tau * 0.9995**t, 3.0)  # Slowly decrease tau
        
        g, conv, p = softmax_gradient(a, tau=tau_cur)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new /= s
        
        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                # Reset momentum
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
                lr = max(lr * 0.8, 0.001)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t} ({elapsed:.0f}/{time_limit:.0f}s): J={best_val:.6f}, tau={tau_cur:.1f}")
    
    return best_a, best_val

def pressure_cd(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent using full pressure (softmax) information."""
    n = len(a0)
    a = normalize_simplex(a0).copy()
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        tau = 30.0
        g, conv, p = softmax_gradient(a, tau=tau)
        M = conv.max()
        
        # Pressure: gradient scaled by current mass
        pressure = g * a
        
        # Source: highest pressure * mass
        src_scores = np.where(a > 1e-15, pressure, -np.inf)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Destination: lowest pressure (best place to add mass)
        dst_scores = g.copy()
        dst_scores[src] = np.inf
        # Also penalize positions near the source (to avoid oscillation)
        for offset in range(-3, 4):
            ni = src + offset
            if 0 <= ni < n and ni != src:
                dst_scores[ni] += 0.5 * abs(g[src])
        dst = int(np.argmin(dst_scores))
        
        # Try multiple transfer fractions
        improved = False
        for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t.copy()
                best_J = J_t
                best_a = a_t.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.001, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD ({elapsed:.0f}/{time_limit:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full softmax subgradient via FFT-based cross-correlation (replaces top-K approximation)
2. Nesterov mirror descent with temperature annealing
3. Adam-projected gradient descent for local refinement
4. Low-dimensional Gaussian-mixture shape search (scipy Powell)
5. Multi-shape initialization (super-Gaussian, sine-power, flat-top, parent)
6. Adaptive length sweeping with interpolation + re-optimization
7. Pressure-based coordinate descent with multi-fraction transfer
"""

import numpy as np
import time
from scipy import signal
from scipy.optimize import minimize as scipy_minimize


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_gradient(a, tau=50.0):
    """Compute gradient of smoothed max of conv(a,a) w.r.t. a using full softmax.
    
    gradient[j] = 2 * sum_k p[k] * a[k-j] for valid indices.
    Returns (gradient, conv, softmax_weights).
    """
    n = len(a)
    conv = fast_conv(a)
    max_c = conv.max()
    exp_c = np.exp(tau * (conv - max_c))
    p = exp_c / exp_c.sum()
    grad = 2.0 * signal.correlate(p, a, mode='valid', method='fft')
    return grad, conv, p


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def make_supergaussian(n, p=4.0):
    """Super-Gaussian centered shape."""
    x = np.linspace(0, 1, n)
    a = np.exp(-np.abs(2 * x - 1) ** p)
    return a / a.sum()


def make_sine_power(n, q=2.0):
    """Sine power shape."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** q
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_flat_top(n):
    """Flat-top (Blackman-Harris) window."""
    x = np.linspace(0, 1, n)
    a = (0.36358 - 0.48917 * np.cos(2 * np.pi * x)
         + 0.13659 * np.cos(4 * np.pi * x)
         - 0.01168 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def make_asymmetric(n, skew=0.2):
    """Asymmetric Gaussian with shift."""
    x = np.linspace(0, 1, n)
    center = 0.5 + skew
    width = 0.25 * (1.0 - abs(skew))
    a = np.exp(-((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_power_ramp(n, p=1.0):
    """Power ramp."""
    x = np.linspace(0.01, 1.0, n)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, sep=0.7):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    c1 = (1 - sep) / 2
    c2 = 1 - c1
    a = np.exp(-((x - c1) / 0.12) ** 2) + np.exp(-((x - c2) / 0.12) ** 2) + 0.01
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, normalize to simplex."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    return normalize_simplex(a_i)


def mirror_descent_fast(a0, time_limit, tau_start=80.0, tau_end=5.0,
                        c0=3.0, restart_every=3000, verbose=False):
    """Nesterov mirror descent with full softmax subgradient and temperature annealing."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_c0 = c0
    cur_restarts = 0

    for t in range(1, 500000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        progress = min(1.0, t / (3.0 * restart_every))
        tau = tau_start * (tau_end / tau_start) ** progress

        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        g, conv, p = softmax_gradient(y, tau=tau)
        eta = cur_c0 / (np.sqrt(t) + 1.0)

        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            cur_restarts += 1
            scale = rng.uniform(0.003, 0.015)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_c0 = max(cur_c0 * 0.94, 0.6)
            if verbose:
                print(f"      MD restart {cur_restarts} t={t}: J={best_val:.6f}")

        if verbose and t % 5000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}/{time_limit:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def adam_projected(a0, time_limit, lr=0.04, tau_start=50.0, verbose=False):
    """Adam optimizer with full softmax gradient, projected to simplex."""
    n = len(a0)
    a = normalize_simplex(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    t = 0
    stall = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        tau_cur = max(tau_start * 0.9998 ** t, 3.0)

        g, conv, p = softmax_gradient(a, tau=tau_cur)

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new /= s

        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
                cur_lr = max(cur_lr * 0.7, 0.001)

        if verbose and t % 3000 == 0:
            print(f"      Adam t={t} ({time.monotonic()-t0:.0f}/{time_limit:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def pressure_cd(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent using full softmax pressure information."""
    n = len(a0)
    a = normalize_simplex(a0).copy()
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        g, conv, p = softmax_gradient(a, tau=25.0)
        pressure = g * a

        src_scores = np.where(a > 1e-15, pressure, -np.inf)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        dst_scores = g.copy()
        dst_scores[src] = np.inf
        nearest = max(0, src - 4)
        farthest = min(n, src + 5)
        for ni in range(nearest, farthest):
            if ni != src:
                dst_scores[ni] += 0.3 * abs(g[src])
        dst = int(np.argmin(dst_scores))

        improved = False
        for frac in [0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.35]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t.copy()
                best_J = J_t
                best_a = a_t.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.04)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 40 == 0:
            print(f"      CD ({time.monotonic()-t0:.0f}/{time_limit:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def gaussian_mixture_obj(params, n):
    """Objective for low-dim Gaussian mixture shape search."""
    n_gauss = len(params) // 3
    a = np.full(n, params[-1])
    for k in range(n_gauss):
        w = params[3 * k]
        c = params[3 * k + 1]
        s = max(params[3 * k + 2], 0.01)
        x = np.linspace(0, 1, n)
        a += w * np.exp(-((x - c) / s) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 100.0
    a = a / s
    return eval_J(a)


def lowdim_shape_search(n, time_limit, verbose=False):
    """Search for good shape using low-dim Gaussian mixture parameterization."""
    t0 = time.monotonic()
    best_x0 = None
    best_val = np.inf
    rng = np.random.default_rng(555)

    for n_gauss in [1, 2, 3, 4]:
        if time.monotonic() - t0 > time_limit:
            break
        for trial in range(3):
            if time.monotonic() - t0 > time_limit:
                break
            params = []
            for k in range(n_gauss):
                params.append(rng.uniform(0.1, 2.0))
                params.append(rng.uniform(0.1, 0.9))
                params.append(rng.uniform(0.05, 0.3))
            params.append(rng.uniform(0.01, 0.1))

            res = scipy_minimize(gaussian_mixture_obj, params, args=(n,),
                                  method='Powell', options={'maxiter': 200,
                                  'xatol': 1e-6, 'fatol': 1e-7})
            if res.fun < best_val:
                best_val = res.fun
                best_x0 = res.x
                if verbose:
                    print(f"        LG ({n_gauss}g, trial {trial}): J={best_val:.6f}")

    if best_x0 is None:
        return None
    n_gauss = (len(best_x0) - 1) // 3
    x = np.linspace(0, 1, n)
    a = np.full(n, best_x0[-1])
    for k in range(n_gauss):
        w = best_x0[3 * k]
        c = best_x0[3 * k + 1]
        s = max(best_x0[3 * k + 2], 0.01)
        a += w * np.exp(-((x - c) / s) ** 2)
    a = np.maximum(a, 0.0)
    return normalize_simplex(a)


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = normalize_simplex(p.copy())
            print(f"Parent: n={len(parent)}, J={eval_J(parent):.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        best_a = make_supergaussian(3000, 4.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)

    # Helper to check and update best
    def check_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = normalize_simplex(a)
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Low-dim shape search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Low-dim shape (n={n0}) ===")
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 15:
        a_lm = lowdim_shape_search(n0, t1, verbose=True)
        check_update(a_lm, "lowdim")

    # === Phase 2: MD multi-init with full softmax ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD Multi-Init (n={n0}, rem={remaining:.0f}s) ===")

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent)
        labels.append("parent")
    inits.append(make_supergaussian(n0, 4.0))
    labels.append("sg4")
    inits.append(make_supergaussian(n0, 8.0))
    labels.append("sg8")
    inits.append(make_sine_power(n0, 2.0))
    labels.append("sin2")
    inits.append(make_sine_power(n0, 4.0))
    labels.append("sin4")
    inits.append(make_flat_top(n0))
    labels.append("flat")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp1")
    inits.append(make_asymmetric(n0, 0.2))
    labels.append("asym")
    inits.append(make_bimodal(n0, 0.6))
    labels.append("bimod")

    if parent is not None:
        b1 = 0.5 * parent + 0.5 * make_supergaussian(n0, 4.0)
        inits.append(normalize_simplex(b1))
        labels.append("blend1")
        b2 = 0.7 * parent + 0.3 * make_sine_power(n0, 3.0)
        inits.append(normalize_simplex(b2))
        labels.append("blend2")

    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 50)], 4))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)

    n_inits = len(u_inits)
    time_per = max(20.0, remaining * 0.30 / n_inits)

    for idx, (init, lbl) in enumerate(zip(u_inits, u_labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 120:
            break
        t_avail = min(time_per, time_budget - 120 - elapsed)
        if t_avail < 15:
            break
        if lbl in ("parent", "blend1"):
            t_avail = min(t_avail * 1.5, time_budget - 120 - elapsed)
        print(f"  MD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = mirror_descent_fast(init, t_avail, tau_start=80.0, tau_end=5.0,
                                        c0=3.0, restart_every=2500)
        check_update(a_out, f"md_{lbl}")

    # === Phase 3: Extended MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Extended MD (n={len(best_a)}, rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.14, 220.0)
    if t3 > 30:
        a_out, _ = mirror_descent_fast(best_a, t3, tau_start=50.0, tau_end=3.0,
                                        c0=2.5, restart_every=4000, verbose=True)
        check_update(a_out, "md_ext")

    # === Phase 4: Adam refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Adam (n={len(best_a)}, rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = adam_projected(best_a, t4, lr=0.03, tau_start=40.0, verbose=True)
        check_update(a_out, "adam")

    # === Phase 5: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: CD (n={len(best_a)}, rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 140.0)
    if t5 > 15:
        a_out, _ = pressure_cd(best_a, t5, verbose=True, seed=77)
        check_update(a_out, "cd1")

    # === Phase 6: MD after CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: MD after CD (rem={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_out, _ = mirror_descent_fast(best_a, t6, tau_start=30.0, tau_end=3.0,
                                        c0=2.0, restart_every=3000)
        check_update(a_out, "md_post_cd")

    # === Phase 7: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length Sweep (rem={remaining:.0f}s) ===")
    n_best = len(best_a)
    lengths = sorted(set([n_best - 400, n_best - 200, n_best + 200, n_best + 400,
                          2000, 2500, 3000, 3500, 4000, 4500, 5000, 5500]))
    lengths = [nl for nl in lengths if 1500 <= nl <= 7000 and abs(nl - n_best) > 100]

    for nl in lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  n={nl}...", end=" ")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 70.0)
        if t_md > 15:
            a_out, _ = mirror_descent_fast(a_int, t_md, tau_start=60.0, tau_end=5.0,
                                            c0=3.0, restart_every=2000)
            check_update(a_out, f"len{nl}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25 and len(best_a) != nl:
            t_cd = min(remaining * 0.12, 35.0)
            if t_cd > 10:
                a_cur = interpolate_seq(best_a, nl)
                a_out, _ = pressure_cd(a_cur, t_cd, seed=nl)
                check_update(a_out, f"len{nl}_cd")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Polish (n={len(best_a)}, rem={remaining:.0f}s) ===")
    if remaining > 60:
        t8a = min(remaining * 0.45, 200.0)
        a_out, _ = mirror_descent_fast(best_a, t8a, tau_start=30.0, tau_end=2.0,
                                        c0=2.0, restart_every=5000, verbose=True)
        check_update(a_out, "final_md")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t8b = min(remaining * 0.35, 120.0)
        a_out, _ = pressure_cd(best_a, t8b, verbose=True, seed=314)
        check_update(a_out, "final_cd")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_out, _ = mirror_descent_fast(best_a, remaining - 15, tau_start=20.0, tau_end=2.0,
                                        c0=1.5, restart_every=6000)
        check_update(a_out, "extra_md")

    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        return [float(x) for x in best_a]
    else:
        fb = make_supergaussian(3000, 4.0)
        print(f"\nFALLBACK: J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
