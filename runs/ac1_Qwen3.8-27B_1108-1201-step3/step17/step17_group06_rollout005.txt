    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)

def softmax_stable(x, T):
    x_shifted = (x - x.max()) / T
    ex = np.exp(x_shifted)
    s = ex.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return ex / s

def smooth_grad(a, T):
    """Gradient of 2n*T*log(sum exp(c/T)) w.r.t. a, assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    p = softmax_stable(c, T)
    corr = np.correlate(a, p, mode='full')
    G = corr[2*n-2 : n-2 : -1]
    return 4.0 * n * G, c

def md_lse(a0, T_init, T_final, n_steps, lr_base, time_limit, verbose=False, t0=None):
    """
    Mirror descent with log-sum-exp smoothed subgradient and temperature annealing.
    """
    if t0 is None:
        t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_val = eval_J(a)
    
    for t in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature annealing (geometric)
        frac = t / max(n_steps - 1, 1)
        T = T_init * (T_final / T_init) ** frac
        
        # Learning rate (decreases with T)
        lr = lr_base * (T / T_init + 0.1)
        
        # Compute gradient
        grad, c = smooth_grad(a, T)
        
        # Multiplicative update
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        a = a_new
        
        # Track best
        if t % 50 == 0 or t == n_steps - 1:
            val = eval_J(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                if verbose:
                    print(f"    MD-LSE t={t} T={T:.4f}: J={best_val:.6f}")
    
    return best_a, best_val

def local_search(a0, time_limit, n_trials=10000, verbose=False, t0=None):
    """
    Local search: try moving mass from one position to another.
    """
    if t0 is None:
        t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_val = eval_J(a)
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Compute current autocovolution to find "hot" positions
        c = fast_conv(a)
        c_max = c.max()
        
        # Find source: position with high value that contributes to peaks
        # Find destination: position that would reduce peaks
        
        # Try random moves
        improved = False
        for _ in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue
            frac = rng.uniform(0.01, 0.2)
            amt = a[i] * frac
            a_try = a.copy()
            a_try[i] -= amt
            a_try[j] += amt
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            val = eval_J(a_try)
            if val < best_val - 1e-12:
                a = a_try
                best_a = a.copy()
                best_val = val
                improved = True
                break
        
        if not improved:
            stall += 1
        else:
            stall = 0
    
    return best_a, best_val

def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_bimodal(n, centers=(0.2, 0.8), width=50.0):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.01
    for c in centers:
        a += np.exp(-width * (x - c) ** 2)
    return a / a.sum()

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_cosine_modulated(n, freq=3.0):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
