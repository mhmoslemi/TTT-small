import math

def _fast_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def _fast_autocorr(a):
    """conv(a, a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    w_full /= w_sum  # normalize to sum = 1

    # g[m] = 2 * sum_k w_full[k] * a[k-m] for valid k-m
    # = 2 * (a crosscorr w_full_rev) shifted
    # Use: g[m] = 2 * conv(a, w_rev)[2n-2-m]
    w_rev = w_full[::-1]
    total_len = n + len(w_rev) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_rev)] = w_rev
    conv_result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_pad), fft_size)
    # g[m] = 2 * conv_result[2n-2-m] for m = 0..n-1
    g = 2.0 * conv_result[n-1:2*n-1][::-1]
    return g, M

def _subgradient_topk(a, top_k=50):
    """Compute subgradient using only top-k lags (faster for large n)."""
    n = len(a)
    ac = _fast_autocorr(a)
    M = ac.max()
    k_max = min(top_k, len(ac))
    # Select top-k lags
    idx = np.argsort(ac)[::-1][:k_max]
    vals = ac[idx]
    # Weight by relative closeness to max
    thresh = M - 0.1 * (M - vals.min()) if vals.min() < M else M
    sel = vals >= thresh
    if not sel.any():
        sel = np.zeros(k_max, dtype=bool)
        sel[np.argmax(vals)] = True
    w_full = np.zeros(len(ac))
    for i in range(k_max):
        if sel[i]:
            w_full[int(idx[i])] = vals[i]  # weight by value
    w_sum = w_full.sum()
    if w_sum < 1e-30:
        w_full[int(idx[0])] = 1.0
        w_sum = 1.0
    w_full /= w_sum

    w_rev = w_full[::-1]
    total_len = n + len(w_rev) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_rev)] = w_rev
    conv_result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_pad), fft_size)
    g = 2.0 * conv_result[n-1:2*n-1][::-1]
    return g, M

def _mirror_descent_nesterov(a0, time_limit, seed=42, use_topk=False, top_k=80):
    """
    Nesterov-accelerated mirror descent (multiplicative weights) on the simplex.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    iterations = 0
    stall = 0
    best_stall_at = best_M

    # Step size: eta_t = c / (sqrt(t) + gamma)
    c_param = 4.0
    gamma_fallback = 1.0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        iterations += 1

        # Nesterov extrapolation
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        if use_topk or n > 5000:
            g, M_curr = _subgradient_topk(y, top_k=top_k)
        else:
            g, M_curr = _subgradient(y)

        # Mirror descent step (multiplicative weights)
        eta = c_param / (np.sqrt(iterations) + gamma_fallback)
        # Log-space update: exp(log(y) - eta * g)
        log_y = np.log(np.maximum(y, 1e-30))
        # Scale gradient to avoid overflow
        g_scaled = g / (np.abs(g).max() + 1e-15)
        log_new = log_y - eta * g_scaled
        log_new -= log_new.max()
        new_a = np.exp(log_new)
        new_a = np.maximum(new_a, 1e-30)
        sn = new_a.sum()
        if sn < 1e-15:
            new_a = y.copy()
        else:
            new_a = new_a / sn

        x_prev = x
        x = new_a

        # Evaluate
        M_new = _fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled too long
        if stall > 3000:
            stall = 0
            # Perturb around best
            noise = rng.normal(0, 0.1, n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Occasionally try random restart
            if rng.random() < 0.2:
                x = _make_random_smooth(n, rng, seed=iterations)
                x_prev = x.copy()

    return best_a, best_M

def _make_random_smooth(n, rng, seed=0):
    """Create a random smooth non-negative sequence."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        a += rng.normal(0, 0.1) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    # Add asymmetry
    a += rng.normal(0, 0.15) * np.sin(np.pi * x + rng.uniform(-1, 1))
    a = np.maximum(a, 0.0)
    return _normalize(a)

def _lp_frank_wolfe_cutting_plane(a0, time_limit, max_constr=200, verbose=True):
    """
    LP-based Frank-Wolfe with cutting plane.
    At each step, solve: min t s.t. conv(a_t, b)[k] <= t, b >= 0, sum(b) = 1
    but only enforce a subset of constraints (the most binding ones).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a0.copy(), _fast_autocorr(a0).max()

    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_nest = a.copy()
    prev_y = a.copy()

    # Try to find a working solver
    solver = None
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            xv = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(xv)), [cp.sum(xv) == 1])
            p.solve(solver=s, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit - 5:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select constraints: top binding lags
        K_select = min(max_constr, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_select]

        # Build constraint matrix T (K_select x n): T[i, j] = y[top_idx[i] - j] if valid
        T = np.zeros((K_select, n))
        for i, k_val in enumerate(top_idx):
            k_int = int(k_val)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            js = np.arange(j_start, j_end + 1)
            if len(js) > 0:
                T[i, js] = y[k_int - js]

        # LP: min t s.t. T @ b <= t, b >= 0, sum(b) = 1
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [
            cp.sum(b_var) == 1,
            T @ b_var <= t_var * np.ones(K_select),
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False, max_iters=5000)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_val = b_val / b_sum

        # Nesterov FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_val
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        prev_y = y
        y = y_new
        x_nest = x_new

        # Track best
        my = _fast_autocorr(y).max()
        if my < best_M:
            best_M = my
            best_a = y.copy()
        mx = _fast_autocorr(x_nest).max()
        if mx < best_M:
            best_M = mx
            best_a = x_nest.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.8f}")

    return best_a, best_M

def _coordinate_descent(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent: find pairs (i,j) where moving mass reduces the peak.
    Uses sensitivity analysis to prioritize moves.
    """
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300
    moves_made = 0

    while time.monotonic() - t0 < time_limit - 5 and stall < max_stall:
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(20, 2 * n - 1)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        # For each top lag, try to reduce it by moving mass away from contributing positions
        for k_idx in range(min(8, n_top)):
            if time.monotonic() - t0 > time_limit - 5:
                break
            k_int = int(top_lags[k_idx])
            # Find positions j that contribute to this lag
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            js = np.arange(j_start, j_end + 1)
            if len(js) == 0:
                continue
            contrib = a[js] * a[k_int - js]
            if contrib.max() < 1e-20:
                continue

            # Source: position with highest contribution
            src = js[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue

            # Compute sensitivity: for each destination d, how much does moving mass there affect peak lags?
            # sens[d] = sum over top lags k2 of a[k2-d] (the partner value)
            sens = np.zeros(n)
            for k2_idx in range(min(12, n_top)):
                k2 = int(top_lags[k2_idx])
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                for d in range(s2, e2 + 1):
                    partner = k2 - d
                    sens[d] += a[partner]

            # Find low-sensitivity destinations far from source
            dist = np.abs(np.arange(n) - src).astype(float)
            score = sens + 0.01 * (1.0 / (1.0 + dist / (n / 10.0)))
            # Exclude positions too close to source
            mask = dist > n // 16
            if mask.sum() > 0:
                candidates = np.where(mask)[0]
                best_dst = candidates[np.argmin(score[candidates])]
            else:
                best_dst = np.argmin(sens)

            if best_dst == src:
                continue

            # Try moving small fractions
            found = False
            for frac in [0.05, 0.02, 0.08, 0.01, 0.1, 0.03, 0.04, 0.015]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-14:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    moves_made += 1
                    break

            if found:
                break  # Restart lag iteration after a successful move

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 5 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = _fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and moves_made % 100 == 0 and moves_made > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, moves={moves_made}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}, moves={moves_made}")

    return best_a, best_M

def _make_parametric_init(n, type_id=0, seed=0):
    """Create diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    if type_id == 0:
        # Smooth asymmetric bump: Gaussian shifted off-center
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
        # Add slight asymmetry
        a *= (1.0 + rng.uniform(-0.3, 0.3) * (x - 0.5))

    elif type_id == 1:
        # Sinusoidal modulation of uniform
        freq = rng.uniform(1, 4)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.sin(2 * np.pi * freq * x + phase)

    elif type_id == 2:
        # Power-law ramp (asymmetric)
        power = rng.uniform(0.5, 3.0)
        flip = rng.random() < 0.5
        if flip:
            a = x ** power
        else:
            a = (1 - x) ** power
        # Add smooth modulation
        a *= (1.0 + 0.2 * np.sin(2 * np.pi * rng.uniform(1, 5) * x))

    elif type_id == 3:
        # Two-peak structure
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w1 = rng.uniform(0.05, 0.15)
        w2 = rng.uniform(0.05, 0.15)
        a = 0.7 * np.exp(-0.5 * ((x - c1) / w1) ** 2) + 0.3 * np.exp(-0.5 * ((x - c2) / w2) ** 2)

    elif type_id == 4:
        # Cosine window with asymmetry
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a *= (1.0 + rng.uniform(-0.4, 0.4) * x)

    elif type_id == 5:
        # Superposition of multiple sinusoids
        a = np.ones(n)
        for f in range(1, 6):
            a += rng.normal(0, 0.2 / f) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))

    else:
        # Exponential ramp with noise
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(rate * (x - 0.5))
        a += 0.1 * rng.standard_normal(n)

    a = np.maximum(a, 0.0)
    return _normalize(a)

def _interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = _normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0.0)
    return _normalize(a_interp)

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase optimization:
    Phase 1: Parametric initialization search
    Phase 2: Nesterov mirror descent (multiple restarts)
    Phase 3: LP-based Frank-Wolfe with cutting plane
    Phase 4: Mirror descent polish
    Phase 5: Coordinate descent refinement
    Phase 6: Length exploration
    Phase 7: Final polish
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        if len(a) == 0 or not np.isfinite(a).all():
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = 2.0 * len(a_norm) * _fast_autocorr(a_norm).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = _normalize(parent)
            p_score = 2.0 * len(p_norm) * _fast_autocorr(p_norm).max()
            print(f"Parent loaded: n={len(p_norm)}, score={p_score:.6f}")
            best_seq = p_norm
            best_score = p_score
    except (NameError, Exception):
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = _make_parametric_init(n_base, 0, seed=1)
        best_score = _eval(best_seq)
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Parametric search for better initialization ===
    print(f"\n=== Phase 1: Parametric Search ===")
    remaining = time_budget - (time.monotonic() - t_start)
    t_phase1 = min(remaining * 0.08, 80.0)

    if t_phase1 > 20:
        for type_id in range(5):
            for seed in [type_id * 100, type_id * 100 + 42, type_id * 100 + 99]:
                elapsed = time.monotonic() - t_start
                if elapsed > time_budget - 60:
                    break
                init = _make_parametric_init(n0, type_id, seed=seed)
                # Quick mirror descent polish (short)
                a_out, _ = _mirror_descent_nesterov(init, min(8.0, t_phase1 / 15.0), seed=seed, use_topk=True)
                try_update(a_out, f"param_{type_id}_{seed}")

    # === Phase 2: Nesterov Mirror Descent with multiple initializations ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_start)
    t_phase2 = min(remaining * 0.30, 450.0)

    if t_phase2 > 60:
        # Use best of: parent, parametric results
        inits = []
        labels = []
        if parent is not None:
            inits.append(_normalize(parent))
            labels.append("parent")
        inits.append(_normalize(best_seq))
        labels.append("current_best")
        inits.append(_make_parametric_init(n0, 2, seed=77))
        labels.append("ramp")
        inits.append(_make_parametric_init(n0, 5, seed=123))
        labels.append("multi_sin")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        seen_sigs = set()
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 2))
            if sig not in seen_sigs:
                seen_sigs.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase2 / (len(inits) + 1))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed = time.monotonic() - t_start
            if elapsed > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_start))
            if t_avail < 10:
                break
            print(f"  MD [{lbl}] (t_avail={t_avail:.0f}s)...")
            a_out, _ = _mirror_descent_nesterov(init, t_avail, seed=100 + idx, use_topk=(n0 > 4000), top_k=100)
            try_update(a_out, f"md_{lbl}")

    # === Phase 3: LP-based Frank-Wolfe with Cutting Plane ===
    print(f"\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = time_budget - (time.monotonic() - t_start)
    t_phase3 = min(remaining * 0.25, 350.0)

    if t_phase3 > 40 and n0 <= 5000:
        a_in = _normalize(best_seq)
        a_out, _ = _lp_frank_wolfe_cutting_plane(a_in, t_phase3, max_constr=150, verbose=True)
        try_update(a_out, f"lp_fw_n{n0}")

    # === Phase 4: Mirror Descent Polish ===
    print(f"\n=== Phase 4: MD Polish ===")
    remaining = time_budget - (time.monotonic() - t_start)
    t_phase4 = min(remaining * 0.15, 200.0)

    if t_phase4 > 20:
        a_in = _normalize(best_seq)
        a_out, _ = _mirror_descent_nesterov(a_in, t_phase4, seed=999, use_topk=(n0 > 4000), top_k=120)
        try_update(a_out, f"md_polish_n{n0}")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = time_budget - (time.monotonic() - t_start)
    t_phase5 = min(remaining * 0.20, 250.0)

    if t_phase5 > 15:
        a_in = _normalize(best_seq)
        a_out, _ = _coordinate_descent(a_in, t_phase5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n0}")

    # === Phase 6: Length Exploration ===
    print(f"\n=== Phase 6: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-400, -200, 200, 400, 600, 800]:
        cl = n_best + delta
        if 1500 <= cl <= 6000:
            candidate_lengths.append(cl)

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = _interpolate_sequence(best_seq, new_n)

        t_md = min(remaining * 0.3, 80.0)
        if t_md > 15:
            a_out, _ = _mirror_descent_nesterov(a_long, t_md, seed=new_n % 1000, use_topk=True, top_k=80)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_start)
        t_cd = min(remaining * 0.3, 60.0)
        if t_cd > 10:
            a_out, _ = _coordinate_descent(a_long, t_cd, verbose=False, seed=new_n % 500)
            try_update(a_out, f"coord_n{new_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_start)
    n_final = len(best_seq)

    if remaining > 30:
        # Short MD polish
        t_md2 = min(remaining * 0.4, 60.0)
        if t_md2 > 10:
            a_in = _normalize(best_seq)
            a_out, _ = _mirror_descent_nesterov(a_in, t_md2, seed=2024, use_topk=True, top_k=60)
            try_update(a_out, f"md_final_n{n_final}")

        # Short coordinate descent
        remaining = time_budget - (time.monotonic() - t_start)
        t_cd2 = min(remaining * 0.4, 60.0)
        if t_cd2 > 10:
            a_in = _normalize(best_seq)
            a_out, _ = _coordinate_descent(a_in, t_cd2, verbose=False, seed=777)
            try_update(a_out, f"coord_final_n{n_final}")

    # === Final validation ===
    if best_seq is not None:
        final_score = _eval(best_seq)
        if final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = _make_parametric_init(n_fb, 0, seed=1)
        best_score = _eval(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
