def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 assuming a is normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def smooth_max(obj, temperature):
    """Log-sum-exp approximation of max with given temperature."""
    m = obj.max()
    if temperature < 1e-10:
        return m
    shifted = (obj - m) / temperature
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    return m + temperature * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad(obj, temperature):
    """Gradient of log-sum-exp smooth max."""
    m = obj.max()
    if temperature < 1e-10:
        # Hard max
        g = np.zeros_like(obj)
        g[obj == m] = 1.0 / (obj == m).sum()
        return g
    shifted = (obj - m) / temperature
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    return exp_shifted / exp_shifted.sum()

def J_smooth(a, temperature):
    """Smooth version of J with log-sum-exp."""
    n = len(a)
    conv = fast_conv(a)
    sm = smooth_max(conv, temperature)
    return 2.0 * n * sm

def J_smooth_grad(a, temperature):

    n = len(a)
    conv = fast_conv(a)
    w = smooth_max_grad(conv, temperature)

        j_idx = k - idx
        g[idx] += 2.0 * w[k] * a[j_idx]
    return 2.0 * n * g

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Use bisection on the Lagrange multiplier
    v_sorted = np.sort(v)[::-1]
    cssum = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n+1) > (cssum - 1))[0][-1]
    theta = (cssum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

            idx = np.arange(i_start, i_end + 1)
            j_idx = k - idx
            grad[idx] += 2.0 * g[k] * a[j_idx]
        grad *= 2.0 * n
        
        # Gradient step on simplex (negate gradient, project)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        J_new = eval_J(a_new)
        if J_new < best_J:

            best_a = a_new.copy()
        
        a = a_new
        
        # Decrease temperature

        if verbose and (int((time.monotonic() - t0) * 10) % 50 == 0):
            print(f"    Smooth t={time.monotonic()-t0:.1f}s: J={best_J:.6f}, temp={temperature:.4f}")

def evolve_population(population, time_limit, verbose=False, seed=42):

            idxs = rng.integers(0, len(population), size=3)
            vals = [scores[i] for i in idxs]
            return idxs[np.argmin(vals)]
        
        # Create offspring

            p2 = population[select()]

                pert = rng.normal(0, 0.02, n) * child

            else:
                child /= child_sum
            
            new_population.append(child)
        
        # Evaluate new population
        new_scores = [eval_J(ind) for ind in new_population]
        
        # Elitism: keep best from old and new

        pop_size = len(population)

def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if parent is not None:
        n0 = len(parent)
        best_a = parent.copy()
        best_score = eval_J(parent)

        best_score = eval_J(best_a)
        print(f"Initial: n={n0}, J={best_score:.6f}")
