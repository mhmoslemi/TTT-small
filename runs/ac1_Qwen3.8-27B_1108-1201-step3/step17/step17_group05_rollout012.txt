Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected subgradient descent with exact max and backtracking line search
2. LP-based local search for active-set refinement
3. Random local search polish
Key improvement over previous: exact subgradient (no soft-max), 
backtracking for step size, and LP-based active-set optimization.

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Clip tiny negative values from FFT numerical error
    result = np.maximum(result, 0.0)
    return result

def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, max_lags):
    """
    Compute the subgradient of max_k conv(a,a)[k] at point a,
    given the set of max lags.
    
    grad[i] = (2/|K|) * sum_{k in K} a[k-i] (where 0 <= k-i < n)
    """
    n = len(a)
    g = np.zeros(n)
    m = len(max_lags)
    for k in max_lags:
        k_int = int(k)
        # grad[i] += 2 * a[k-i] for valid i
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            g[i_start:i_end+1] += 2.0 * a[k_int - i_end:k_int - i_start:-1]
    g /= m
    return g

def subgradient_descent(a0, time_limit, verbose=True, seed=42):
    """
    Projected subgradient descent with exact max and backtracking.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    b = fast_autocorr(a)
    best_M = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()
    
    iters = 0
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        t_it = iters
        
        # Nesterov extrapolation
        beta_t = (t_it - 1.0) / (t_it + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution and find max lags
        b = fast_autocorr(y)
        M = b.max()
        max_idx = np.where(b >= M * (1 - 1e-8))[0]
        
        # Compute subgradient
        g = compute_subgradient(y, max_idx)
        
        # Backtracking line search
        eta = 0.5 / (np.sqrt(t_it) + 1.0)
        improved = False
        for _ in range(30):
            a_new = project_simplex(y - eta * g)
            M_new = fast_autocorr(a_new).max()
            if M_new < M - 1e-14:
                improved = True
                break
            eta *= 0.7
        
        if not improved:
            a_new = project_simplex(y - eta * g)
        
        x_prev = x
        x = a_new
        
        # Evaluate
        M_x = fast_autocorr(x).max()
        if M_x < best_M - 1e-14:
            best_M = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Also check x in Nesterov
        if M_x < best_M:
            best_M = M_x
            best_a = x.copy()
            stall = 0
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Perturb best solution
            scale = 0.01
            pert = rng.normal(0, scale, n)
            x_new = best_a + pert
            x_new = np.maximum(x_new, 0)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new.copy()
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SGD iter={iters} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected subgradient with exact max, backtracking, and Nesterov momentum
2. Active-set LP refinement (first-order direction from LP on binding lags)
3. Random pair-swap local search for final polish
Different from previous: uses exact (not soft) subgradient, LP-based active-set
optimization, and a different overall structure.

import numpy as np
import time
import cvxpy as cp

# --- Core utilities ---

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def proj_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_for_lag(a, k):
    """Gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        # g[i] = 2*a[k-i] for i in [i_start, i_end]
        # a[k-i] for i from i_start to i_end is a[k-i_end] down to a[k-i_start]
        g[i_start:i_end + 1] = 2.0 * a[k - i_end:k - i_start - 1:-1]
    return g

    a = proj_simplex(a0.astype(np.float64))
    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    momentum = 0.9
    
    iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov step
        beta_t = (iters - 1.0) / (iters + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute conv and find max lags
        b = fast_conv(y)
        M = b.max()
        
        # Find lags within 0.1% of max
        thresh = M * 0.999
        max_lags = np.where(b >= thresh)[0]
        
        # Average subgradient over max lags
        g = np.zeros(n)
        for k in max_lags:
            g += subgrad_for_lag(y, int(k))
        g /= len(max_lags)

def get_lp_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x >= 0])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lp_refine(a0, solver, time_limit, verbose=True):
    """
    Active-set LP refinement.
    At each step, identify binding constraints (lags at max),
    solve an LP for the best first-order descent direction,
    and take a step in that direction.
    """
    n = len(a0)
    a = proj_simplex(a0.astype(np.float64))
    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    t0 = time.monotonic()
    
    iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            break
        
        # Find active lags (within 0.01% of max)
        thresh = M * 0.9999
        active = np.where(b >= thresh)[0]
        m = len(active)
        
        if m == 0:
            break
        
        # Build LP:
        # minimize t
        # subject to: b_k + grad_k . delta <= t for k in active
        #              sum(delta) = 0
        #              delta_i >= -a_i
        #              norm(delta, 2) <= rho  (approximate with L-inf for LP)
        
        # For LP, replace L2 constraint with L-inf: |delta_i| <= rho
        rho = 0.01  # max change per coordinate
        
        delta = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(delta) == 0,
            delta >= -a,
            delta <= rho,
            -delta <= rho  # |delta_i| <= rho
        ]
        
        for k in active[:50]:  # limit to 50 constraints max
            k_int = int(k)
            gk = subgrad_for_lag(a, k_int)
            constraints.append(b[k_int] + gk @ delta <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        delta_val = np.asarray(delta.value, dtype=np.float64)
        delta_val = np.clip(delta_val, -a + 1e-15, 1e10)
        delta_val = np.maximum(delta_val, -a + 1e-15)
        
        # Ensure sum is 0
        delta_val -= delta_val.mean()
        
        # Try the step
        a_new = a + delta_val
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        M_new = fast_conv(a_new).max()
        if M_new < M - 1e-15:
            a = a_new
        else:
            # Try smaller step
            for frac in [0.5, 0.1, 0.01]:
                a_try = a + frac * delta_val
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                M_try = fast_conv(a_try).max()
                if M_try < M - 1e-15:
                    a = a_try
                    break
            else:
                stall += 100  # force break soon
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-refine iter={iters} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_M

def local_search(a0, time_limit, verbose=True, seed=0):
    """Random pair-swap local search."""
    n = len(a0)
    a = proj_simplex(a0.astype(np.float64))
    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iters = 0
    stall = 0
    max_stall = 5000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iters += 1
        
        # Pick random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Try random swap amounts
        best_delta = 0
        best_M_local = best_M
        
        for scale in [0.001, 0.0005, 0.002, 0.0001, 0.005]:
            for direction in [1, -1]:
                delta = direction * scale * min(a[i], a[j])
                if delta == 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                M_try = fast_conv(a_try).max()
                if M_try < best_M_local:
                    best_M_local = M_try
                    best_delta = delta
        
        if best_delta != 0:
            a[i] -= best_delta
            a[j] += best_delta
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_M_local < best_M:
                best_M = best_M_local
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LocalSearch iter={iters} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_M

def make_init(n, seed=0, type='smooth'):
    rng = np.random.default_rng(seed)
    if type == 'uniform':
        a = np.ones(n)
    elif type == 'cosine':
        x = np.arange(n) / (n - 1)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x)
    elif type == 'edge':
        a = np.ones(n)
        a[:n//10] = 0.3
        a[-n//10:] = 0.3
    elif type == 'smooth':
        x = np.arange(n) / (n - 1)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.05) * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    return proj_simplex(a)
