def fast_conv(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J_normalized(a):
    """Compute J = 2*n*max(conv) assuming a is normalized to sum=1."""
    n = len(a)
    if n == 0:
        return float('inf')
    conv = fast_conv(a)
    return float(2.0 * n * conv.max())


def compute_smooth_gradient(a, conv, T, top_k=100):
    """Compute gradient of (2n/T)*log(sum exp(T*b[k])) using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = conv[top_idx]

    # Softmax weights
    shifted = T * (top_vals - top_vals.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            a_vals = a[k_int - idx]
            grad[idx] += 4.0 * n * wi * a_vals

    return grad


def compute_exact_subgradient(a, conv, top_k=50, sharpness=20.0):

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_optimizer(n):
    """Create Adam optimizer state."""
    return {
        'm': np.zeros(n),
        'v': np.zeros(n),
        't': 0,
        'beta1': 0.9,
        'beta2': 0.999,
        'eps': 1e-8
    }


def adam_step(opt_state, grad, lr):
    """Perform one Adam step, return updated parameters direction."""
    opt_state['t'] += 1
    t = opt_state['t']
    b1 = opt_state['beta1']
    b2 = opt_state['beta2']
    eps = opt_state['eps']

    opt_state['m'] = b1 * opt_state['m'] + (1 - b1) * grad
    opt_state['v'] = b2 * opt_state['v'] + (1 - b2) * grad * grad

    m_hat = opt_state['m'] / (1 - b1 ** t)
    v_hat = opt_state['v'] / (1 - b2 ** t)

    return lr * m_hat / (np.sqrt(v_hat) + eps)


def smooth_gradient_descent(a0, time_limit, T_start=30.0, T_end=3000.0,
                            top_k_start=40, top_k_end=250,
                            lr=0.01, restart_every=5000, seed=42, verbose=False):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_val = eval_J_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    opt = adam_optimizer(n)
    total_iters = max(total_iters, 1)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Progress fraction [0, 1]
        progress = min(1.0, t / max(1, int(time_limit * 50)))  # ~50 iters/sec estimate
        # Temperature schedule: exponential ramp
        T_cur = T_start * (T_end / T_start) ** progress
        # Top-K schedule
        k_cur = int(top_k_start + (top_k_end - top_k_start) * progress)
        # Learning rate decay
        lr_cur = lr / (1.0 + 0.01 * t)

        conv = fast_conv(a)
        grad = compute_smooth_gradient(a, conv, T_cur, top_k=k_cur)

        # Adam step in log-space for better non-negativity handling
        # Actually, let's use projected gradient in linear space
        a_new = a - lr_cur * grad
        a_new = project_simplex(a_new)

        # Occasionally add small noise for exploration
        if t % 200 == 0:
            noise_scale = 0.001 * (1 - progress)
            a_new += noise_scale * rng.normal(0, 1, n)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

        val = eval_J_normalized(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()

        a = a_new

        # Restart logic
        if t % restart_every == 0 and progress > 0.3:
            # Perturb from best
            scale = 0.01 * (1 - progress) + 0.002
            pert = best_a + scale * rng.normal(0, 1, n)
            pert = np.maximum(pert, 1e-15)
            pert = pert / pert.sum()
            if eval_J_normalized(pert) < best_val * 1.01:
                a = pert
            opt = adam_optimizer(n)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD t={t} ({elapsed:.0f}s): J={best_val:.6f}, T={T_cur:.1f}, k={k_cur}")

    return best_a, best_val


def pressure_coordinate_descent(a0, time_limit, seed=42, verbose=False, max_stall=200):
    """Coordinate descent with pressure-based mass transfer and annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J_normalized(a)
    t0 = time.monotonic()

    stall = 0
    temp = 0.01  # annealing temperature

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Identify hot lags (near-maximum)
        threshold = M * 0.95
        hot_lags = np.where(conv >= threshold)[0]
        if len(hot_lags) == 0:
            hot_lags = np.array([int(np.argmax(conv))])

        # Compute position pressure from hot lags
        pressure = np.zeros(n)
        for k in hot_lags[:30]:

            i_s = max(0, k_i - n + 1)

                    best_J = J_t

                    print(f"      FineTune ({elapsed:.0f}s): J={best_J:.6f}, no improvement")
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      FineTune ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def make_asymmetric_power(n, p=1.5):
    """Asymmetric power law: strong ramp shape."""
    x = np.linspace(0, 1, n)
    a = (x + 0.01) ** p
    return a / a.sum()


def make_double_ramp(n, p1=1.0, p2=2.0, mid=0.3):
    """Two-segment ramp with different slopes."""
    x = np.linspace(0, 1, n)
    a = np.where(x < mid, (x / mid) ** p1, 1.0 + ((x - mid) / (1 - mid)) ** p2 * 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_log_ramp(n):
    """Logarithmic ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.log(x + 1)
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main search driver with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J_normalized(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    if parent is None:
        n0 = 3200
        parent = make_asymmetric_power(n0, 1.5)
        print(f"Generated initial: n={n0}, J={eval_J_normalized(parent):.6f}")

    best_a = parent.copy()
    best_J = eval_J_normalized(best_a)
    print(f"Starting best: J={best_J:.6f}, n={len(best_a)}")

    def try_update(a, label):
        nonlocal best_a, best_J
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J_normalized(a)
        if J < best_J - 1e-15:
            best_J = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_a)

    # === Phase 1: Smooth gradient descent (warm-up to sharp) ===
    print(f"\n=== Phase 1: Smooth GD (n={n0}) ===")
    t1 = min(remaining_time() * 0.35, 500.0)
    if t1 > 30:
        print(f"  Running smooth GD for {t1:.0f}s...")
        a_out, _ = smooth_gradient_descent(best_a.copy(), t1,
                                           T_start=20.0, T_end=5000.0,
                                           top_k_start=30, top_k_end=200,
                                           lr=0.005, restart_every=4000, seed=7, verbose=True)
        try_update(a_out, "smooth_gd_1")

    # === Phase 2: Continued smooth GD from best ===
    print(f"\n=== Phase 2: Extended Smooth GD ===")
    remaining = remaining_time()
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        print(f"  Running extended GD for {t2:.0f}s...")
        a_out, _ = smooth_gradient_descent(best_a.copy(), t2,
                                           T_start=200.0, T_end=10000.0,
                                           top_k_start=80, top_k_end=300,
                                           lr=0.003, restart_every=6000, seed=42, verbose=True)
        try_update(a_out, "smooth_gd_2")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Pressure CD ===")
    remaining = remaining_time()
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        print(f"  Running CD for {t3:.0f}s...")
        a_out, _ = pressure_coordinate_descent(best_a.copy(), t3, seed=99, verbose=True, max_stall=300)
        try_update(a_out, "coord_cd")

    # === Phase 4: Fine-tune lags ===
    print(f"\n=== Phase 4: Fine-tune ===")
    remaining = remaining_time()
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        print(f"  Running fine-tune for {t4:.0f}s...")
        a_out, _ = fine_tune_lags(best_a.copy(), t4, seed=314, verbose=True)
        try_update(a_out, "fine_tune")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration ===")
    remaining = remaining_time()
    n_best = len(best_a)

    # Try nearby lengths
    length_offsets = [-200, -100, -50, -10, 10, 50, 100, 200]
    lengths_to_try = []
    for d in length_offsets:
        nl = n_best + d
        if 1000 <= nl <= 6000:
            lengths_to_try.append(nl)
    # Also try some round numbers
    for nl in [2000, 2500, 3000, 3500, 4000, 4500, 5000]:
        if nl not in lengths_to_try and 1000 <= nl <= 6000:
            lengths_to_try.append(nl)

    lengths_to_try.sort(key=lambda x: abs(x - n_best))

    for nl in lengths_to_try[:8]:  # Try up to 8 different lengths
        remaining = remaining_time()
        if remaining < 30:
            break
        if abs(nl - len(best_a)) <= 5:
            continue
        print(f"  Trying n={nl} (remaining={remaining:.0f}s)...")
        a_int = interpolate_seq(best_a, nl)
        t_sub = min(remaining * 0.30, 80.0)
        if t_sub > 10:
            # Quick smooth GD
            a_out, _ = smooth_gradient_descent(a_int, t_sub * 0.6,
                                               T_start=100.0, T_end=5000.0,
                                               top_k_start=50, top_k_end=150,
                                               lr=0.004, restart_every=3000, seed=nl, verbose=False)
            try_update(a_out, f"len{nl}_gd")
            remaining = remaining_time()
            if remaining > 15 and len(best_a) != nl:
                t_cd = min(remaining * 0.30, 40.0)
                if t_cd > 8:
                    a_cur = interpolate_seq(best_a, nl)
                    a_out, _ = pressure_coordinate_descent(a_cur, t_cd, seed=nl, verbose=False, max_stall=100)
                    try_update(a_out, f"len{nl}_cd")

    # === Phase 6: Final gradient descent polish ===
    print(f"\n=== Phase 6: Final GD Polish (n={len(best_a)}) ===")
    remaining = remaining_time()
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 15:
        print(f"  Running final GD for {t6:.0f}s...")
        a_out, _ = smooth_gradient_descent(best_a.copy(), t6,
                                           T_start=500.0, T_end=20000.0,
                                           top_k_start=100, top_k_end=350,
                                           lr=0.002, restart_every=8000, seed=2718, verbose=True)
        try_update(a_out, "final_gd")

    # === Phase 7: Final coordinate descent ===
    print(f"\n=== Phase 7: Final CD ===")
    remaining = remaining_time()
    t7 = min(remaining * 0.25, 150.0)
    if t7 > 10:
        print(f"  Running final CD for {t7:.0f}s...")
        a_out, _ = pressure_coordinate_descent(best_a.copy(), t7, seed=31415, verbose=True, max_stall=400)
        try_update(a_out, "final_cd")

    # === Phase 8: Ultra-fine coordinate descent ===
    print(f"\n=== Phase 8: Ultra-fine CD ===")
    remaining = remaining_time()
    t8 = min(remaining * 0.30, 100.0)
    if t8 > 8:
        a_out, _ = fine_tune_lags(best_a.copy(), t8, seed=99999, verbose=True)
        try_update(a_out, "ultra_fine")

    # === Phase 9: One more GD if time permits ===
    remaining = remaining_time()
    if remaining > 20:
        print(f"\n=== Phase 9: Extra GD ===")
        a_out, _ = smooth_gradient_descent(best_a.copy(), remaining - 5,
                                           T_start=1000.0, T_end=50000.0,
                                           top_k_start=150, top_k_end=400,
                                           lr=0.001, restart_every=10000, seed=8675, verbose=False)
        try_update(a_out, "extra_gd")

    # Final
    if best_a is not None:
        final_J = eval_J_normalized(best_a)
        best_J = min(best_J, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_J:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3200
        fb = make_asymmetric_power(n_fb, 1.5)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J_normalized(fb):.6f}")
        return [float(x) for x in fb]
