```python
def adam_softmax_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, 
                          softmax_beta=50.0, verbose=False):
    """
    Adam optimizer on the softmax-parameterized soft-max objective.
    a = exp(s) / sum(exp(s)), optimize over s.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    s = np.log(a + 1e-30)  # Initialize s from a
    
    # Center s (invariance: s + c gives same a)
    s -= s.mean()
    
    m_s = np.zeros(n)  # First moment
    v_s = np.zeros(n)  # Second moment
    t = 0
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute current a from s
        s_centered = s - s.max()
        a = np.exp(s_centered)
        a = a / a.sum()
        
        # Compute gradient of f_beta(a) w.r.t. a
        b = fast_conv(a)
        scaled = softmax_beta * b
        scaled_max = scaled.max()
        p = np.exp(scaled - scaled_max)
        p = p / p.sum()
        grad_a = softmax_gradient_fft(a, p)
        
        # Gradient w.r.t. s: grad_s = (grad_a - a * sum(grad_a))
        # (Jacobian of softmax: da_j/ds_k = a_j*(delta_jk - a_k))
        # So d f / d s_k = sum_j (df/da_j) * a_j * (delta_jk - a_k)
        # = a_k * (df/da_k - sum_j (df/da_j) * a_j)
        grad_s = a * (grad_a - np.dot(grad_a, a))
        
        # Adam update
        m_s = beta1 * m_s + (1 - beta1) * grad_s
        v_s = beta2 * v_s + (1 - beta2) * grad_s ** 2
        m_hat = m_s / (1 - beta1 ** t)
        v_hat = v_s / (1 - beta2 ** t)
        s -= lr * m_hat / (np.sqrt(v_hat) + eps)
        s -= s.mean()  # Re-center
        
        # Track best
        M_new = fast_conv(a).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a.copy()
    
    return best_a, best_m
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max Nesterov mirror descent with FFT-based dense gradient (O(n log n))
2. Full-constraint LP Remez method for near-optimal polishing
3. Adam optimizer with log-parameterization for diversity
4. Coordinate descent with pair moves for micro-optimization
5. Multi-length exploration
Key improvement: softmax-weighted gradient via FFT cross-correlation gives a smooth,
dense gradient (using all lags) vs the previous top-k subgradient approximation.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_gradient_fft(a, p):
    """
    Compute grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] for j=0,...,n-1.
    p has length 2n-1. Uses FFT-based convolution in O(n log n).
    """
    n = len(a)
    p_flip = p[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_flip, fft_size)
    conv = np.fft.irfft(A * P, fft_size)
    C = conv[n - 1: 2 * n - 1][::-1]
    return 2.0 * C

def softmax_nesterov(a0, time_limit, beta_start=5.0, beta_max=500.0,
                     beta_growth=1.003, c0=4.0, verbose=False):
    """
    Soft-max Nesterov mirror descent with FFT gradient.
    Minimizes f_beta(a) = (1/beta)*logsumexp(beta * (a*a)) via mirror descent.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    beta = beta_start
    c = c0
    stall = 0
    max_stall = 4000
    rng = np.random.default_rng(54321 + n)

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta = min(beta * beta_growth, beta_max)

        mu = (t - 1.0) / (t + 1.5)
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        b = fast_conv(y)

        scaled = beta * b
        scaled_max = scaled.max()
        p = np.exp(scaled - scaled_max)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(b))
            p[int(np.argmax(b))] = 1.0
        else:
            p = p / p_sum

        grad = softmax_gradient_fft(y, p)

        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            strategy = int(rng.integers(4))
            if strategy == 0:
                x_r = best_a * (1.0 + rng.normal(0, 0.015, n))
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                x_r = best_a * (1.0 + rng.normal(0, 0.003, n))
            else:
                x_r = 0.75 * best_a + 0.25 * make_init(n, 'raised_cos')
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
            max_stall = max(400, int(max_stall * 0.6))
            c = max(0.3, c * 0.65)
            beta = max(beta_start * 0.3, beta * 0.3)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftMaxNMD t={t} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_m

def adam_log_opt(a0, time_limit, softmax_beta=50.0, lr=0.005, verbose=False):
    """
    Adam optimizer on log-parameterized soft-max objective.
    Parameterization: a = exp(s) / sum(exp(s)), optimize s freely.
    """
    n = len(a0)
    a_init = norm_sum1(a0).copy()
    s = np.log(a_init + 1e-30)
    s -= s.mean()

    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a_init.copy()
    best_m = fast_conv(a_init).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 20000
    rng = np.random.default_rng(99999 + n)

    while time.monotonic() - t0 < time_limit:
        t += 1

        s_centered = s - s.max()
        a = np.exp(s_centered)
        a = a / a.sum()

        b = fast_conv(a)
        scaled = softmax_beta * b
        scaled_max = scaled.max()
        p = np.exp(scaled - scaled_max)
        p = p / p.sum()
        grad_a = softmax_gradient_fft(a, p)

        # Gradient w.r.t. s via chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_s = a * (grad_a - dot)

        m_s = beta1 * m_s + (1 - beta1) * grad_s
        v_s = beta2 * v_s + (1 - beta2) * (grad_s * grad_s)
        m_hat = m_s / (1.0 - beta1 ** t)
        v_hat = v_s / (1.0 - beta2 ** t)
        s -= lr * m_hat / (np.sqrt(v_hat) + eps)
        s -= s.mean()

        M_new = fast_conv(a).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Reset Adam moments and perturb
            m_s = np.zeros(n)
            v_s = np.zeros(n)
            s = np.log(best_a + 1e-30)
            s += rng.normal(0, 0.01, n)
            s -= s.mean()
            stall = 0
            max_stall = max(5000, int(max_stall * 0.6))
            lr = max(0.0001, lr * 0.7)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AdamLog t={t} ({elapsed:.1f}s): J={J:.8f}, lr={lr:.6f}")

    return best_a, best_m

def lp_remez_step(a, max_cons, r, solver):
    """
    Single LP Remez step. Linearizes top max_cons lag constraints.
    If max_cons=0, uses all 2n-1 constraints.
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    if max_cons > 0 and max_cons < len(b):
        active = np.argsort(b)[::-1][:max_cons]
    else:
        active = np.arange(len(b))

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var + b[k_int])

    if r > 0:
        constraints.append(a_var >= a * (1.0 - r))
        constraints.append(a_var <= a * (1.0 + r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_loop(a0, time_limit, solver, r_init=0.05, max_cons_init=0, verbose=False):
    """
    Iterative LP Remez with adaptive trust region and constraint count.
    Starts with all constraints (max_cons=0) and small r for accuracy.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    max_cons = max_cons_init
    rng = np.random.default_rng(777 + n)

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        a_cand = lp_remez_step(a, max_cons, r, solver)

        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            a_trial = (1.0 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                r = min(0.15, r * 1.03)
                break

        if not accepted:
            no_improve += 1
            r = max(0.001, r * 0.6)
            if no_improve > 25:
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                a[idx] *= rng.uniform(0.85, 1.15, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                max_cons = min(200, max_cons + 10) if max_cons > 0 else 0

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    RemezLoop t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.5f}")

    return best_a, best_m

def cd_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent with random pair moves and multiple trial fractions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 40)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(300, n // 2)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            best_frac = None
            best_m_trial = best_m
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3, 0.5]:
                amt = a[i] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m_trial - 1e-16:
                    best_m_trial = m_trial
                    best_frac = frac

            if best_frac is not None:
                amt = a[i] * best_frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_m_trial
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        fs = evaluate_safe(best_seq)

            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
