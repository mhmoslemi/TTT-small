def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=100, beta=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_full_subgradient(a, ac, top_k=200, beta=3.0):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Softmax weighting
    max_v = top_vals[0]
    exp_vals = np.exp(beta * (top_vals - max_v))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

    cur_beta = beta
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = min(100, max(10, int((time_limit - (time.monotonic() - t0)) * 10)))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            iteration += 1
            beta_t = (iteration - 1.0) / (iteration + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            ac = fast_autocorr(y)
            if adapt_temp:
                cur_beta = max(2.0, beta * (1.0 - 0.3 * np.sin(np.pi * iteration / (2.0 * restart_every))))
            g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

            if use_backtrack:
                eta = gamma / (np.sqrt(iteration) + 1.0)
                base_m = ac.max()
                accepted = False
                for _bt in range(8):
                    log_a = np.log(y) - eta * g
                    max_log = log_a.max()
                    a_new = np.exp(log_a - max_log)
                    a_new = np.maximum(a_new, 1e-30)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        break
                    a_new = a_new / s_new
                    m_trial = fast_autocorr(a_new).max()
                    if m_trial < base_m - 1e-18:
                        x_prev = x.copy()
                        x = a_new
                        accepted = True
                        break
                    else:
                        eta *= 0.5
                if not accepted:
                    log_a = np.log(y) - eta * g
                    max_log = log_a.max()
                    a_new = np.exp(log_a - max_log)
                    a_new = np.maximum(a_new, 1e-30)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        a_new = y.copy()
                    else:
                        a_new = a_new / s_new
                    x_prev = x.copy()
                    x = a_new
            else:
                eta = gamma / (np.sqrt(iteration) + 1.0)
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = y.copy()
                else:
                    a_new = a_new / s_new
                x_prev = x.copy()
                x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert_strength = 0.02 + 0.02 * np.random.random()
            pert = rng.normal(0, pert_strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 200)
            cur_beta = max(cur_beta - 0.3, 1.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart ({elapsed:.0f}s): J={J:.6f}")

        if verbose and iteration % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD iter={iteration} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=400,
                   margin_frac=1e-3, min_K=50):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    t = 0

    while time.monotonic() - t0 < time_limit:
        # Batch of LP iterations
        batch = min(5, max(1, int((time_limit - (time.monotonic() - t0)) / 5.0)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            ac = fast_autocorr(y)
            M_curr = ac.max()
            if M_curr < 1e-16:
                break

            # Dynamic constraint selection
            margins = M_curr - ac
            threshold = margin_frac * M_curr
            active_mask = margins <= threshold
            n_active = active_mask.sum()

            # Adaptive K: use more constraints as we converge
            K_target = min(top_K, max(min_K, n_active))
            if n_active > K_target:
                active_idx = np.argsort(margins)[:K_target]
            elif n_active < max(20, min_K):
                active_idx = np.argsort(margins)[:K_target]
            else:
                active_idx = np.where(active_mask)[0]
                if len(active_idx) > K_target:
                    active_idx = active_idx[:K_target]

            K = len(active_idx)
            if K < 5:
                active_idx = np.argsort(margins)[:max(50, min_K)]
                K = len(active_idx)

            # Build constraint matrix (sparse rows)
            T_sel = np.zeros((K, n))
            for row, k_int in enumerate(active_idx):
                k_i = int(k_int)
                i_start = max(0, k_i - n + 1)
                i_end = min(n, k_i + 1)
                idx = np.arange(i_start, i_end)
                T_sel[row, idx] = y[k_i - idx]

            b_var = cp.Variable(n, nonneg=True)
            constraints = [T_sel @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                continue
            g_dir = b_val / b_sum

            # Accelerated Frank-Wolfe
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy

            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx

            y = y_new
            x = x_new

            my = fast_autocorr(y).max()
            if my < best_m:
                best_m = my
                best_a = y.copy()
                # Tighten margin as we improve
                margin_frac = max(margin_frac * 0.85, 1e-5)

            mx = fast_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def lp_frank_wolfe_dense(a0, time_limit, solver, verbose=True, n_constraints=2000):
    """LP-FW using a fixed set of stratified constraints for better coverage."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    t = 0
    n_full = 2 * n - 1

    # Stratified constraint selection: uniform spacing + top peaks
    base_indices = np.linspace(0, n_full - 1, min(n_constraints, n_full)).astype(int)

    while time.monotonic() - t0 < time_limit:
        batch = min(5, max(1, int((time_limit - (time.monotonic() - t0)) / 5.0)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            ac = fast_autocorr(y)
            M_curr = ac.max()
            if M_curr < 1e-16:
                break

            # Combine stratified + top peaks
            top_idx = np.argsort(ac)[::-1][:min(200, n_full)]
            all_idx = np.unique(np.concatenate([base_indices, top_idx]))
            K = len(all_idx)

            T_sel = np.zeros((K, n))
            for row, k_int in enumerate(all_idx):
                k_i = int(k_int)
                i_start = max(0, k_i - n + 1)
                i_end = min(n, k_i + 1)
                idx = np.arange(i_start, i_end)
                T_sel[row, idx] = y[k_i - idx]

            b_var = cp.Variable(n, nonneg=True)
            constraints = [T_sel @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                continue
            g_dir = b_val / b_sum

            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy

            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx

            y = y_new
            x = x_new

            my = fast_autocorr(y).max()
            if my < best_m:
                best_m = my
                best_a = y.copy()

            mx = fast_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-dense iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=777):
    """Targeted peak reduction via mass transfer from high-contribution positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Find top peaks
        n_peaks = min(10, len(ac))
        peak_indices = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for peak_k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(peak_k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find top contributors
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]

            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Try moving mass to random destinations
                best_new_m = M_curr
                best_move = None

                n_trials = 40
                for frac in [0.005, 0.01, 0.02, 0.04, 0.06]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    for _ in range(n_trials // 5):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)

                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s < 1e-15:
                        a = best_a.copy()
                    else:
                        a = a / s
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-16:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            if a[i] > 1e-10:
                j = int(rng.integers(n))
                if i != j:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m - 1e-16:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.02, rng=None):
    """Perturb in frequency domain to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.1)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'beta_3_3':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 3 * (1 - x) ** 3)
    elif family == 'gaussian':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.exp(-4 * (x - 0.5) ** 2))
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

    best_score = float('inf')

    try:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)

        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)

                                    top_K=500, margin_frac=2e-3, min_K=100)

    # === Phase 3: LP-FW with stratified constraints ===

        a_in = norm_sum1(best_seq)

        a_out, _ = lp_frank_wolfe_dense(a_in, t3, solver, verbose=True,
                                          n_constraints=n_cons)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_strat")

    # === Phase 4: Peak crushing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak-crush (n={n_best}) ===")

        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush(a_in, t4, verbose=True, rng_seed=42)

                                     use_backtrack=True, adapt_temp=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

    rng = np.random.default_rng(1234)

    if t7_per > 30:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t7_per + 30:
                break
            strength = 0.01 + 0.01 * trial
            a_pert = freq_perturb(best_seq, strength=strength, rng=rng)
            a_out, _ = amd_optimize(a_pert, t7_per, top_k=80, beta=4.0, c0=3.5,
                                     restart_every=1500, verbose=False,
                                     use_backtrack=True, adapt_temp=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 8: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t8, top_k=150, beta=2.5, c0=2.0,
                                 restart_every=2000, verbose=False,
                                 use_backtrack=True, adapt_temp=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "final_amd")

    # === Phase 9: Final peak crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final peak-crush (n={n_best}) ===")
        t9 = min(remaining * 0.30, 150.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_crush(a_in, t9, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
