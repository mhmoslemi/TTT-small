def fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fft_autocorr(a / s)
    return 2.0 * n * ac.max()


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def logsumexp_soft_max(ac, top_k=100, beta=8.0):
    """Approximate max(ac) using log-sum-exp on top-k elements."""
    k = min(top_k, len(ac))
    top = np.partition(ac, len(ac) - k)[len(ac) - k:]
    top = top / beta
    m = top.max()
    if m < 1e-30:
        return 1.0 / beta
    return float(m + np.log(np.sum(np.exp(top - m)))) * beta / beta


def compute_lse_subgradient(a, ac, top_k=100, beta=8.0):
    """Compute subgradient of the LSE-softened max autocorrelation."""
    n = len(a)
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    # Soft weights via softmax with temperature beta
    w = np.exp(top_vals / beta)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def water_filling_cd(a0, time_limit, n_peaks=5, verbose=True, rng_seed=77,
                     max_stall=500):
    """Multi-peak water-filling coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    # Temperature for simulated annealing acceptance
    T = 0.001
    T_min = 0.0001
    T_decay = 0.9995

    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            break

        ac = fft_autocorr(a)
        M_curr = ac.max()
        avg_ac = ac.mean()

        # Target the top n_peaks lags
        top_lags = np.argsort(ac)[::-1][:n_peaks]
        all_improved = False

        for peak_k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(peak_k)
            peak_val = ac[peak_k]
            if peak_val <= avg_ac * 1.01:
                continue  # This peak is already near average

            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[peak_k - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue

            # Find top contributors
            n_top = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]

            found_move = False
            for li in top_local:
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                best_delta = 0.0
                best_dst = -1
                best_amt = 0.0

                # Try several fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    # Try randomized destinations
                    n_trials = 20
                    for _ in range(n_trials):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        # Measure improvement: reduction in top peaks
                        ac_trial = fft_autocorr(a_trial)
                        trial_metric = ac_trial[top_lags].max()
                        if trial_metric < M_curr + T:
                            delta = M_curr - ac_trial.max()
                            if delta > best_delta:
                                best_delta = delta
                                best_dst = dst
                                best_amt = amt

                if best_dst >= 0 and best_delta > T:
                    a[src] -= best_amt
                    a[best_dst] += best_amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fft_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    all_improved = True
                    found_move = True
                    break

            if found_move:
                break

        if not all_improved:
            stall += 1
            T *= T_decay
            if T < T_min:
                T = T_min

            # Random kick
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fft_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    WF-CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}, T={T:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    WF-CD final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=300,
                   margin_frac=1e-3, use_nesterov=True):
    """LP Frank-Wolfe with adaptive constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy() if use_nesterov else None
    margin = margin_frac

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fft_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select constraints: all within margin, or top-K if too many
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix T_sel @ b <= M_curr
        # For each lag k: sum_i b[i] * y[k-i] <= M_curr (with bounds)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False, timeout=10)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if use_nesterov:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
            x = x_new

        y = y_new

        # Track best
        my = fft_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin = max(margin * 0.85, 1e-5)

        if use_nesterov:
            mx = fft_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin:.1e}")

    return best_a, best_m


def spectral_smooth(a, time_limit, verbose=True, rng_seed=314):
    """Spectral smoothing: minimize variance of autocorrelation via gradient flow."""
    n = len(a)
    a = norm_sum1(a0 if (a0 := a) is not None else a)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Minimize: 0.9 * LSE_max + 0.1 * variance_penalty
    eta = 0.1
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fft_autocorr(a)
        M = ac.max()

        # Gradient of LSE approximation to max
        top_k = 60
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = np.exp(top_vals / 5.0)
        w = w / w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx_arr = np.arange(i_start, i_end)
            j_idx = ki - idx_arr
            g[idx_arr] += 2.0 * wi * a[j_idx]

        # Gradient of variance
        # d/d a[i] sum_k ac[k]^2 = 2 * sum_k ac[k] * (a[i+k] + a[i-k])
        # = 2 * (ac_even * a)[i] where ac_even is the even extension
        # Compute via FFT
        ac_even = np.zeros(2 * n - 1)
        ac_even[:n] = ac[:n]
        ac_even[n:] = ac[:n-1][::-1]
        fft_size = 1 << (2 * n - 1).bit_length()
        C = np.fft.rfft(ac_even, fft_size)
        A = np.fft.rfft(a, fft_size)
        var_g = np.fft.irfft(C * A, fft_size)[:n]
        g += 0.1 * 2.0 * var_g

        # Mirror descent step
        eta_t = eta / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-15)) - eta_t * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-20)
        a_new = a_new / a_new.sum()

        M_new = fft_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()

        a = a_new

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def batch_cd(a0, time_limit, n_peaks=8, verbose=True, rng_seed=271):
    """Batch coordinate descent: adjust multiple coordinates simultaneously."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break

        ac = fft_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Collect all top contributor positions across all peaks
        src_set = set()
        for pk in top_lags:
            pk = int(pk)
            i_start = max(0, pk - n + 1)
            i_end = min(n, pk + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[pk - idx]
            top_local = np.argsort(contrib)[::-1][:5]
            for li in top_local:
                src_set.add(int(idx[li]))

        # Try batch moves: reduce top sources, spread to random destinations
        src_list = list(src_set)
        if not src_list:
            stall += 1
            continue

        improved = False
        # Try reducing all top sources by a small fraction simultaneously
        for total_frac in [0.003, 0.007, 0.015, 0.03]:
            a_trial = a.copy()
            total_moved = 0.0
            for src in src_list:
                amt = a_trial[src] * total_frac
                a_trial[src] -= amt
                total_moved += amt

            # Distribute to random destinations
            n_dst = int(rng.integers(3, len(src_list) + 1))
            dsts = rng.choice(n, size=n_dst, replace=False)
            per_dst = total_moved / n_dst
            for dst in dsts:
                a_trial[int(dst)] += per_dst

            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fft_autocorr(a_trial).max()

            if M_trial < best_m:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            # Single coordinate escape
            src = int(rng.choice(src_list)) if src_list else int(rng.integers(n))
            dst = int(rng.integers(n))
            if src != dst and a[src] > 1e-10:
                amt = a[src] * rng.uniform(0.002, 0.01)
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fft_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    BatchCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'tau':
        # Tau function: sin(pi*x)/(pi*x) squared
        x = np.linspace(0.001, 0.999, n)
        t = np.sin(np.pi * x) / (np.pi * x)
        return norm_sum1(t ** 2)
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    sc = compute_score(a)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a / s, sc
    return best_seq, best_score

        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = compute_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
