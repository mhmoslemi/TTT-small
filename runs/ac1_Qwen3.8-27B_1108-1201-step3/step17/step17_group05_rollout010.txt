def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute full linear convolution a*b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def make_hann_window(n, center_shift=0.0, alpha=1.0):
    """Create a Hann-like window with optional shift and generalized alpha."""
    x = np.arange(n)
    t = (x - center_shift) / (n - 1)
    # Generalized cosine window
    w = 0.5 * (1 + np.cos(2 * np.pi * t)) ** alpha
    w = np.maximum(w, 0.0)
    return norm_sum1(w)


def make_asymmetric_window(n, skew=0.1, seed=0):
    """Create an asymmetric window by shifting a Hann window."""
    shift = int(skew * n)
    x = np.arange(n)
    center = (n - 1) / 2.0 + shift
    t = (x - center) / (n - 1)
    w = 0.5 * (1 + np.cos(2 * np.pi * np.abs(t)))
    w = np.maximum(w, 0.0)
    return norm_sum1(w)


def make_tukey_window(n, r=0.5, center_shift=0.0):
    """Tukey (tapered cosine) window."""
    x = np.arange(n) + center_shift
    t = x / (n - 1)
    w = np.ones(n)
    # Left taper
    left_mask = t < r / 2
    w[left_mask] = 0.5 * (1 + np.cos(np.pi * (2 * t[left_mask] / r - 1)))
    # Right taper
    right_mask = t > 1 - r / 2
    w[right_mask] = 0.5 * (1 + np.cos(np.pi * (2 * t[right_mask] / r - 1)))
    w = np.maximum(w, 0.0)
    return norm_sum1(w)


def make_random_smooth(n, seed=0, freq_scale=5.0):
    """Random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f in range(1, int(freq_scale) + 1):
        coeff = rng.normal(0, 0.1 / f)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def fw_lp_step(a, M, K, solver):
    """
    Solve the LP subproblem: max sum(b) s.t. conv(a,b) <= M (top K constraints),
    b >= 0, sum(b) = 1.
    Returns the direction b/sum(b) or None if infeasible.
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Select top-K most binding constraints
    k_sel = min(K, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    
    # Build constraint matrix: T[m,j] = a[m-j] if 0<=m-j<n else 0
    T = np.zeros((k_sel, n))
    for col_idx, m in enumerate(top_idx):
        m_int = int(m)
        j_start = max(0, m_int - n + 1)
        j_end = min(n - 1, m_int)
        js = np.arange(j_start, j_end + 1)
        if len(js) > 0:
            T[col_idx, js] = a[m_int - js]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(k_sel),
                   cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum


def fw_lp_step_full(a, M, solver):
    """LP step with all constraints (for smaller n)."""
    n = len(a)
    ac = fast_autocorr(a)
    n_cons = len(ac)
    
    T = np.zeros((n_cons, n))
    for m in range(n_cons):
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        if len(js) > 0:
            T[m, js] = a[m - js]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(n_cons),
                   cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum


def frank_wolfe_as(a0, time_limit, solver, K=300, use_full=False, verbose=True, seed=42):
    """
    Frank-Wolfe with Away-Steps (FW-AS) and adaptive step size.
    """
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    y = x.copy()  # extrapolation point
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_x = y.copy()
        
        # LP step: find s (toward direction)
        s = None
        if use_full and n <= 2000:
            s = fw_lp_step_full(y, M_curr, solver)
        else:
            s = fw_lp_step(y, M_curr, K, solver)
        
        if s is None:
            # Try with slightly relaxed M
            s = fw_lp_step(y, M_curr * 1.01, K, solver)
            if s is None:
                continue
        
        # Away step: find the coordinate to move away from
        # The away vertex is the unit vector e_i where i maximizes (g . e_i) 
        # among the support of y, where g is the subgradient
        # Approximate g by the sensitivity of the max peak
        
        # Find peak lag
        k_peak = int(np.argmax(ac))
        
        # Gradient at peak: grad[i] = 2 * a[k_peak - i] if valid
        grad = np.zeros(n)
        for i in range(n):
            partner = k_peak - i
            if 0 <= partner < n:
                grad[i] = 2.0 * a if False else 0  # placeholder
        
        # Better: compute gradient properly
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # d/dx_i of (x*x)[k] = sum_j (delta(i,j)*x[k-j] + x[j]*delta(i,k-j))
        # = x[k-i] + x[k-i] (if both valid and i != k-i)
        # = 2*x[k-i] if k-i valid and i != k-i
        # = x[k-i] if i == k-i (i.e., 2i=k)
        
        partner = k_peak - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * y[partner[valid]]
        # Fix diagonal: where arange(n) == partner, i.e., 2*i == k_peak
        diag_idx = k_peak // 2
        if 2 * diag_idx == k_peak and 0 <= diag_idx < n:
            grad[diag_idx] = y[diag_idx]  # was 2*y[diag_idx], should be y[diag_idx]
        
        # Away vertex: the i that maximizes grad[i]/y[i] (most "unwanted" mass)
        # among indices where y[i] > 0
        support = y > 1e-10
        if support.any():
            ratios = np.zeros(n)
            ratios[support] = grad[support] / y[support]
            i_away = int(np.argmax(ratios))
        else:
            i_away = 0
        
        # Determine step size with away step
        # x_new = (1-alpha)*y + alpha*s - alpha_away*x_away
        # where x_away = e_{i_away}
        
        # Try several alpha values
        best_alpha = 2.0 / (t + 2.0)
        best_delta_away = 0.0
        best_obj = fast_autocorr(y).max()
        
        for alpha_try in [best_alpha, 0.5 * best_alpha, 1.5 * best_alpha, 0.1, 0.05, 0.3]:
            for delta_a in [0.0, 0.3 * alpha_try, 0.5 * alpha_try, 0.7 * alpha_try]:
                alpha_a = min(delta_a, best_alpha)
                x_trial = (1.0 - alpha_try - alpha_a) * y + alpha_try * s
                x_trial[i_away] -= alpha_a * y[i_away]
                # Actually, the away step formulation:
                # x_new = (1 - alpha) * y + alpha * s - beta * e_i
                # with alpha + beta <= 1
                x_trial = (1.0 - alpha_try - alpha_a) * y + alpha_try * s
                x_trial[i_away] -= alpha_a
                x_trial = np.maximum(x_trial, 0.0)
                sx = x_trial.sum()
                if sx < 1e-15:
                    continue
                x_trial = x_trial / sx
                m_trial = fast_autocorr(x_trial).max()
                if m_trial < best_obj - 1e-14:
                    best_obj = m_trial
                    best_alpha = alpha_try
                    best_delta_away = alpha_a
        
        # Apply best step
        if best_obj < fast_autocorr(y).max() - 1e-14:
            y_new = (1.0 - best_alpha - best_delta_away) * y + best_alpha * s
            y_new[i_away] -= best_delta_away
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
            
            m_new = fast_autocorr(y_new).max()
            if m_new < best_m:
                best_m = m_new
                best_x = y_new.copy()
            
            y = y_new
        else:
            # No improvement, try plain FW step
            alpha_fw = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha_fw) * y + alpha_fw * s
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new / sy
                m_new = fast_autocorr(y_new).max()
                if m_new < best_m:
                    best_m = m_new
                    best_x = y_new.copy()
                y = y_new
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-AS iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_x, best_m


def pairwise_coord_descent(a0, time_limit, verbose=True, seed=42, n_peaks=5):
    """
    Pairwise coordinate descent: for each top peak, try moving mass
    between pairs of indices to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300
    peels = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        top_k = min(n_peaks, len(ac))
        peak_lags = np.argsort(ac)[::-1][:top_k]
        
        improved = False
        
        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find indices contributing to this peak
            js = np.arange(max(0, k_int - n + 1), min(n, k_int + 1))
            partners = k_int - js
            
            # Contribution of each pair (i, k-i) to this peak
            # a[i] * a[k-i]
            contribs = a[js] * a[partners]
            
            # Find the most contributing source index
            max_idx = np.argmax(contribs)
            src = int(js[max_idx])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: an index that minimizes impact on all top peaks
            # and is far from src
            sens = np.zeros(n)
            for k2 in peak_lags:
                k2i = int(k2)
                p = k2i - np.arange(n)
                valid = (p >= 0) & (p < n)
                sens[valid] += a[p[valid]]
            
            # Score: low sensitivity, prefer far from src
            dist = np.abs(np.arange(n) - src).astype(float)
            dist[dist == 0] = 1
            score = sens / (1.0 + dist / n)
            
            # Exclude src and very close indices
            mask = score < np.median(score)
            mask[src] = False
            if not mask.any():
                mask = np.ones(n, dtype=bool)
                mask[src] = False
                mask[np.abs(np.arange(n) - src) < n // 10] = False
            
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]
            
            best_dst = candidates[np.argmin(score[candidates])]
            
            # Try multiple fractions
            found = False
            for frac in [0.01, 0.005, 0.02, 0.002, 0.03, 0.008, 0.015, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if found:
                peels += 1
                if verbose and peels % 100 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    PD ({elapsed:.0f}s): J={J:.6f}, peels={peels}")
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PD stall={stall} ({elapsed:.0f}s): J={J:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PD final ({elapsed:.0f}s): J={J:.6f}, peels={peels}")
    
    return best_a, best_m


def gradient_descent_peak(a0, time_limit, verbose=True, seed=0):
    """
    Projected gradient descent specifically targeting the peak.
    Uses the exact subgradient of max(conv(a,a)).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # Subgradient: gradient at peak lag
        k_peak = int(np.argmax(ac))
        grad = np.zeros(n)
        partner = k_peak - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        grad[valid] = 2.0 * a[partner[valid]]
        # Fix diagonal
        if 2 * (k_peak // 2) == k_peak:
            idx = k_peak // 2
            if 0 <= idx < n:
                grad[idx] = a[idx]
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Line search
        best_step = 0.01 / (np.sqrt(it + 1) + 1)
        best_obj = M
        
        for step in [best_step, 2 * best_step, 0.5 * best_step, 4 * best_step, 0.1 * best_step, 0.01]:
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_obj - 1e-14:
                best_obj = m_trial
                best_step = step
        
        if best_step == 0.01 / (np.sqrt(it + 1) + 1) and best_obj < M:
            a = a - best_step * grad
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            # Try with a smaller step
            small_step = 0.001 / (np.sqrt(it + 1) + 1)
            a_trial = a - small_step * grad
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < M:
                    a = a_trial
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Grad iter={it} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining FW-AS, gradient descent,
    pairwise coordinate descent, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        best_seq = make_hann_window(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init (Hann): n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Diverse initializations + quick gradient descent ===
    print(f"\n=== Phase 1: Diverse Inits + Gradient Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)
    
    if t_phase1 > 30:
        # Create diverse initializations
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        
        # Hann window (symmetric)
        inits.append(make_hann_window(n0, center_shift=0.0))
        labels.append("hann_sym")
        
        # Slightly asymmetric windows
        for skew in [-0.05, 0.05, -0.1, 0.1, 0.03, -0.03]:
            inits.append(make_asymmetric_window(n0, skew=skew, seed=hash(str(skew)) % 1000))
            labels.append(f"asym_{skew:+.2f}")
        
        # Tukey windows with different tapers
        for r in [0.3, 0.5, 0.7]:
            inits.append(make_tukey_window(n0, r=r))
            labels.append(f"tukey_r{r}")
        
        # Random smooth
        for s in [1, 7, 42]:
            inits.append(make_random_smooth(n0, seed=s, freq_scale=4.0))
            labels.append(f"rand_{s}")
        
        # Generalized Hann (alpha variations)
        for alpha in [0.7, 1.5, 2.0]:
            inits.append(make_hann_window(n0, alpha=alpha))
            labels.append(f"hann_a{alpha}")
        
        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 100)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        inits = unique_inits
        labels = unique_labels
        print(f"  {len(inits)} diverse initializations")
        
        # Quick gradient descent on each (short time)
        t_per = max(5.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 3:
                break
            a_out, _ = gradient_descent_peak(init, t_avail, verbose=False)
            try_update(a_out, f"grad_{lbl}")
    
    # === Phase 2: FW-AS on best found so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FW-AS (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 500.0)
    
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        K = min(300, 2 * n_best - 1)
        a_out, _ = frank_wolfe_as(a_in, t2, solver, K=K, use_full=(n_best <= 2000),
                                   verbose=True, seed=42)
        try_update(a_out, f"fw_as_n{n_best}")
    
    # === Phase 3: Gradient descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Gradient Polish ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_descent_peak(a_in, t3, verbose=False)
        try_update(a_out, f"grad_polish_n{n_best}")
    
    # === Phase 4: Pairwise coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Pairwise CD ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 180.0)
    
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_coord_descent(a_in, t4, verbose=True, seed=42, n_peaks=5)
        try_update(a_out, f"pwc_d_n{n_best}")
    
    # === Phase 5: Second FW-AS round ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: FW-AS Round 2 ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.20, 300.0)
    
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        K = min(400, 2 * n_best - 1)
        a_out, _ = frank_wolfe_as(a_in, t5, solver, K=K, use_full=(n_best <= 2000),
                                   verbose=True, seed=123)
        try_update(a_out, f"fw_as_r2_n{n_best}")
    
    # === Phase 6: Fine length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Fine Length Sweep (base n={n_best}) ===")
    
    # Try small perturbations in length
    length_offsets = [-10, -5, 5, 10, -20, 20, -50, 50, -100, 100]
    
    for offset in length_offsets:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        
        new_n = n_best + offset
        if new_n < 500 or new_n > 10000:
            continue
        
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick gradient descent
        t6 = min(remaining * 0.08, 60.0)
        if t6 > 5:
            a_out, _ = gradient_descent_peak(a_interp, t6, verbose=False)
            try_update(a_out, f"grad_n{new_n}")
        
        # Quick FW if time permits
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4000:
            t6b = min(remaining * 0.10, 80.0)
            if t6b > 20:
                a_out, _ = frank_wolfe_as(a_interp, t6b, solver, K=min(300, 2*new_n-1),
                                           use_full=False, verbose=False, seed=77)
                try_update(a_out, f"fw_n{new_n}")
    
    # === Phase 7: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final PWCD (n={n_best}) ===")
    
    if remaining > 20:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_coord_descent(a_in, t7, verbose=True, seed=99, n_peaks=8)
            try_update(a_out, f"pwcd_final_n{n_best}")
    
    # === Phase 8: Final FW if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    if remaining > 40 and n_best <= 5000:
        print(f"\n=== Phase 8: Final FW (n={n_best}) ===")
        t8 = min(remaining * 0.50, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            K = min(500, 2 * n_best - 1)
            a_out, _ = frank_wolfe_as(a_in, t8, solver, K=K, use_full=False,
                                       verbose=True, seed=2024)
            try_update(a_out, f"fw_final_n{n_best}")
    
    # === Phase 9: Very fine length tweak ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    if remaining > 30:
        print(f"\n=== Phase 9: Ultra-fine length (n={n_best}) ===")
        for offset in [-1, 1, -2, 2, -3, 3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 10:
                break
            new_n = n_best + offset
            if new_n < 500 or new_n > 10000:
                continue
            a_interp = interpolate_sequence(best_seq, new_n)
            sc = evaluate_safe(a_interp)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_interp.copy()
                print(f"  *** Ultra-fine n={new_n}: {sc:.6f}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_hann_window(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
