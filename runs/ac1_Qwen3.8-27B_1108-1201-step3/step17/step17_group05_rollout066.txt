Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated subgradient with
soft-max smoothing, partner-based coordinate descent, and multi-scale

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def make_exp_ramp(n, rate):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(rate * i)
    return norm_sum1(a)


def make_waterfall(n, alpha):
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** alpha
    return norm_sum1(a)


def make_cosine_taper(n):

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / max(n - 1, 1)))

def make_smooth_asym(n, seed, freqs):

        a += coeff * np.sin(2 * np.pi * f * i + phase)
    a += 0.08 * rng.normal(0, 1, n)

def softmax_subgrad(a, beta, top_k):
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    shifted = top_vals - top_vals.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_use) / k_use
    else:
        w = w / ws
    w_full = np.zeros(2 * n - 1)
    for wi, idx in zip(w, top_idx):
        w_full[int(idx)] = wi
    total_len = n + len(w_full) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_full)] = w_full
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g, c.max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
