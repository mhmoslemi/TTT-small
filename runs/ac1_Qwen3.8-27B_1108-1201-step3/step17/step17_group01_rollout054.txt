def fast_autocorr(a):

def compute_soft_subgradient(a, ac, beta=5.0, top_frac=1.0):
    """Soft subgradient using softmax weights over top fraction of peaks."""
    n = len(a)
    m = len(ac)
    k_use = max(10, int(m * top_frac))
    k_use = min(k_use, m)
    
    top_idx = np.argsort(ac)[::-1][:k_use]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Softmax weighting
    scaled = beta * top_vals / max(top_vals.max(), 1e-30)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_use) / k_use
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        j_idx = np.arange(i_start, i_end)
        grad[i_start:i_end] += 2.0 * wi * a[k_int - j_idx]
    return grad

def compute_hard_subgradient(a, ac, top_k=30, beta=8.0):
    """Hard subgradient focused on top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-15:

        k_int = int(k)
        i_start = max(0, k_int - n + 1)

        j_idx = np.arange(i_start, i_end)

def amd_optimize(a0, time_limit, mode="soft", c0=4.0, restart_every=6000,
                 top_k=40, beta_init=6.0, beta_min=2.0, verbose=False):
    """Accelerated Mirror Descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_beta = beta_init
    cur_top_k = top_k
    cur_top_frac = 0.5 if mode == "soft" else 1.0
    restart_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        if mode == "soft":
            g = compute_soft_subgradient(y, ac, beta=cur_beta, top_frac=cur_top_frac)
        else:
            g = compute_hard_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Normalize gradient for numerics
        g_norm = g.sum()
        if g_norm < 1e-30:
            break

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g / g_norm
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameter updates

            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

            # Adapt parameters

            else:
                cur_top_k = min(cur_top_k + 15, 120)

            if verbose:

def spectral_flatten(a0, time_limit, iters=2000, eta0=0.7, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Compress dynamic range
        eta = eta0 * (0.5 + 0.5 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update with occasional reset
            if it % 50 == 0:
                a = 0.7 * a + 0.3 * a_new
            else:
                a = 0.9 * a + 0.1 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Spectral it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initializer(n, kind="cosine"):
    x = np.linspace(0, 1, n)
    if kind == "cosine":

    elif kind == "sinc":
        a = np.sinc((x - 0.5) * 5.0) ** 2
    elif kind == "gaussian":
        a = np.exp(-((x - 0.5) ** 2) / (0.15 ** 2))
    elif kind == "bilateral":
        a = 1.0 / (0.1 + np.abs(x - 0.5) ** 2)
    elif kind == "multihump":
        a = np.zeros(n)
        for c in [0.2, 0.5, 0.8]:
            a += np.exp(-((x - c) ** 2) / (0.08 ** 2))
        a += 0.05
    elif kind == "random_smooth":
        rng = np.random.default_rng(42 + n)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.1 / f) * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0.01)
    else:
        a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)

    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_flatten(a_in, t3, iters=3000, eta0=0.7, verbose=False)
        try_update(a_out, f"spectral_n{n0}")

    # === Phase 4: AMD after spectral ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, mode="soft", c0=3.5,
                                restart_every=4000, verbose=False)
        try_update(a_out, f"amd_post_spectral_n{n0}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=False, rng_seed=42)
        try_update(a_out, f"coord_n{n0}")

    # === Phase 6: Chebyshev parameterization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = chebyshev_optimize(a_in, t6, n_cheb=25, verbose=False)
        try_update(a_out, f"cheb_n{n0}")

    # === Phase 7: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length (base n={n_best}) ===")

    length_candidates = [n_best + 500, n_best + 1000, n_best + 2000,
                         max(1500, n_best - 1000), max(1000, n_best - 2000)]
    length_candidates = [l for l in length_candidates if 500 <= l <= 8000]

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t7a = min(remaining * 0.25, 120.0)
        if t7a > 20:
            a_out, _ = amd_optimize(a_long, t7a, mode="soft", c0=4.0,
                                    restart_every=4000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.15, 60.0)
            if t7b > 15:
                a_out, _ = spectral_flatten(a_long, t7b, iters=500, eta0=0.6)
                try_update(a_out, f"spectral_n{new_n}")

    # === Phase 8: Fresh starts at various lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Fresh starts ===")
    fresh_lengths = [2000, 4000, 5000]
    fresh_kinds = ["sinc", "multihump", "bilateral"]

    for new_n, kind in zip(fresh_lengths, fresh_kinds):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        a_fresh = make_initializer(new_n, kind)
        t8 = min(remaining * 0.15, 100.0)
        if t8 < 15:
            continue
        print(f"  Fresh n={new_n}, kind={kind} (t={t8:.0f}s)...")
        a_out, _ = amd_optimize(a_fresh, t8, mode="soft", c0=4.0,
                                restart_every=4000, verbose=False)
        try_update(a_out, f"fresh_{kind}_n{new_n}")

    # === Phase 9: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
    if remaining > 30:
        a_in = norm_sum1(best_seq)

        t9a = min(remaining * 0.30, 150.0)
        if t9a > 20:
            a_out, _ = amd_optimize(a_in, t9a, mode="soft", c0=3.0,
                                    restart_every=8000, verbose=False)
            try_update(a_out, f"final_amd_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.25, 80.0)
            if t9b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t9b, verbose=False, rng_seed=777)
                try_update(a_out, f"final_coord_n{n_best}")

    # === Phase 10: One more AMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final AMD variant (n={n_best}) ===")
        t10 = min(remaining * 0.5, 100.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t10, mode="hard", c0=3.0,
                                    restart_every=10000, top_k=80, beta_init=5.0,
                                    verbose=False)
            try_update(a_out, f"final_hard_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_initializer(n_fb, "cosine")
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
