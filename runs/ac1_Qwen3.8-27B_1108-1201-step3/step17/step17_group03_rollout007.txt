```python
def lp_frank_wolfe(a0, time_limit, solver, verbose=True, 
                    initial_K=500, max_K=2000,
                    initial_margin=1e-3, final_margin=1e-6,
                    n_iters_max=50000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(x)
        M_curr = ac.max()
        
        # Adaptive margin and K
        progress = min(t / n_iters_max, 1.0)
        margin = initial_margin * (final_margin / initial_margin) ** progress
        K = min(int(initial_K + (max_K - initial_K) * progress), 2*n-1)
        
        # Select active constraints
        margins = M_curr - ac
        threshold = margin * M_curr
        n_active = (margins <= threshold).sum()
        
        if n_active >= K:
            # Use exactly K most constrained
            active_idx = np.argsort(margins)[:K]
        else:
            # Use all active + some more
            active_all = np.where(margins <= threshold)[0]
            if len(active_all) < K:
                # Add the next most constrained
                remaining = np.setdiff1d(np.arange(2*n-1), active_all)
                extra = np.argsort(margins[remaining])[:K - len(active_all)]
                active_idx = np.concatenate([active_all, remaining[extra]])
            else:
                active_idx = active_all
        
        K_actual = len(active_idx)
        
        # Build constraint matrix
        T = build_constraint_matrix(x, active_idx, n)
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # FW update with extrapolation
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * x + alpha * g_dir
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        
        # Evaluate
        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()
        
        x_prev = x
        x = x_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K_actual}, margin={margin:.1e}")
    
    return best_a, best_m
```

```python
        contrib = a[idx] * a[peak_k - idx]

        src = int(idx[np.argmax(contrib)])

        best_dst = -1
```

The LP is:
```
maximize: sum(b)
subject to: (x*b)_k <= M for k in active_set
            b >= 0
```

where x is the current iterate and M = max(x*x).

The optimal `b*` is the FW direction. We then update:
```
x_new = (1-alpha) * x + alpha * b*/sum(b*)
```

The key is that `b*` should be "far" from `x` in the direction that reduces the max autocorrelation. If the LP is too easy (few constraints), `b*` might not be very different from `x`.

Using more constraints makes the LP harder, so `b*` is more constrained and potentially more useful.

```python
def line_search(x, g_dir, ac_x, M, n, max_evals=20):
    """Find best t in [0,1] to minimize max((x+t*(g-x))*(x+t*(g-x)))"""
    # Bisection on t
    t_lo, t_hi = 0.0, 1.0
    best_t = 0.0
    best_val = M
    
    for _ in range(max_evals):
        t_mid = (t_lo + t_hi) / 2
        x_mid = (1 - t_mid) * x + t_mid * g_dir
        x_mid = np.maximum(x_mid, 0)
        s = x_mid.sum()
        if s < 1e-15:
            t_hi = t_mid
            continue
        x_mid = x_mid / s
        val = fast_autocorr(x_mid).max()
        if val < best_val:
            best_val = val
            best_t = t_mid
        
        # Check derivative sign by evaluating at t_mid +/- small
        eps = 1e-4
        x_plus = (1 - min(t_mid+eps, 1)) * x + min(t_mid+eps, 1) * g_dir
        x_plus = np.maximum(x_plus, 0)
        sp = x_plus.sum()
        if sp > 1e-15:
            x_plus = x_plus / sp
            val_plus = fast_autocorr(x_plus).max()
        else:
            val_plus = val
        
        if val_plus > val:
            t_lo = t_mid
        else:
            t_hi = t_mid
    
    return best_t, best_val
```

```python
def build_constraint_matrix_fast(a, active_idx, n):
    K = len(active_idx)
    # Use broadcasting
    # T[row, col] = a[k - col] where k = active_idx[row], col = k - ...
    # For each row r with k = active_idx[r]:
    #   valid cols: max(0, k-n+1) to min(k, n-1)
    #   T[r, j] = a[k-j] for j in valid range
    
    T = np.zeros((K, n))
    for r in range(K):
        k = int(active_idx[r])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1) + 1
        T[r, j_start:j_end] = a[k - j_end + 1:k - j_start + 1][::-1]
    return T
```

Wait, let me be more careful. For row r with lag k:
- j ranges from `j_start = max(0, k-n+1)` to `j_end = min(k, n-1)`
- T[r, j] = a[k-j]

So `T[r, j_start:j_end+1] = a[k-j_end : k-j_start+1]`

Let me verify: j = j_start: T[r, j_start] = a[k - j_start]
j = j_end: T[r, j_end] = a[k - j_end]

The slice `a[k-j_end : k-j_start+1]` gives elements at indices k-j_end, k-j_end+1, ..., k-j_start.
But we want T[r, j_start] = a[k-j_start], T[r, j_start+1] = a[k-j_start-1], ..., T[r, j_end] = a[k-j_end].

So it's reversed: `T[r, j_start:j_end+1] = a[k-j_end : k-j_start+1][::-1]`

Let me verify with a small example:
a = [a0, a1, a2, a3], n=4
k = 2: j ranges from max(0, 2-3) = 0 to min(2, 3) = 2
T[0] = a[2-0] = a[2]
T[1] = a[2-1] = a[1]
T[2] = a[2-2] = a[0]

Using the formula: a[k-j_end : k-j_start+1] = a[2-2: 2-0+1] = a[0:3] = [a0, a1, a2]
Reversed: [a2, a1, a0] ✓

Great, so the formula is:
```python
T[r, j_start:j_end+1] = a[k-j_end : k-j_start+1][::-1]
```

This is O(1) per row (just a slice), so the whole construction is O(K) slice operations, which is very fast.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP Frank-Wolfe
with adaptive constraint selection, extrapolation, and multi-length exploration.
"""

import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def build_T(a, active_idx, n):
    K = len(active_idx)
    T = np.zeros((K, n), dtype=np.float64)
    for r in range(K):
        k = int(active_idx[r])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        seg = a[k - j_end:k - j_start + 1][::-1]
        T[r, j_start:j_end + 1] = seg
    return T


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, fn)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_new / ss


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'beta_3_3':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 3 * (1 - x) ** 3)
    elif family == 'uniform':
        return np.ones(n, dtype=np.float64) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def lp_fw(a0, time_limit, solver, verbose=True,
          K_start=300, K_end=1500,
          margin_start=5e-3, margin_end=1e-5,

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x_prev = a.copy()

        M_curr = ac.max()
        if M_curr < 1e-18:

        p_smooth = progress ** 1.5  # ease-in
        margin = margin_start * (margin_end / margin_start) ** p_smooth
        K = int(K_start + (K_end - K_start) * p_smooth)
        K = min(K, 2 * n - 1)

        # Select active constraints
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        elif n_active >= 10:

                    active_idx = np.concatenate([active_idx, remaining[extra_idx]])

            x_new = (1.0 - alpha) * y + alpha * g_dir

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 40
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_new_m:
                        best_new_m = Mt
                        best_move = (src, dst, amt)

            if best_move is not None:
                s_i, d_i, amt = best_move
                a[s_i] -= amt
                a[d_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush done ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m
