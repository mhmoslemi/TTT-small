def fast_autocorr(a):
    """Full linear autocorrelation (a*a) via rfft.  length = 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    ac = fast_autocorr(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss


def make_parametric(n, family="beta"):
    x = np.linspace(0.0, 1.0, n)
    if family == "beta":
        v = (x * (1.0 - x)) ** 1.5 + 1e-6
    elif family == "sine":
        v = np.sin(np.pi * x) ** 2 + 1e-9
    elif family == "skew":
        v = x ** 0.25 * (1.0 - x) ** 1.25 + 1e-6
    else:
        v = np.ones(n)
    return norm_sum1(v)

def soft_subgradient(y, ac, temp=0.5):
    """Return d max-soft of ac w.r.t. y, shape (n,).  temp controls focus."""
    n = len(y)
    m = len(ac)
    # softmax over lags
    w = np.exp(-ac / max(temp, 1e-12))
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(m) / m
    else:
        w = w / ws
    g = np.zeros(n)
    # for each lag k with weight wk, d ac[k]/dy[i] = y[k-i] when 0<=k-i<n
    for i in range(n):
        # k range where k-i in [0,n-1]  ->  k in [i, i+n-1]
        k0 = i
        k1 = min(m, i + n)
        if k1 > k0:
            # dy[i] appears paired with y[k-i] for k in [i, i+n-1]
            idx = np.arange(k0, k1)
            partner = idx - i  # 0..(<n)
            g[i] = float(np.dot(w[idx], y[partner]))
    return g


def soft_subgradient_fast(y, ac, top_k=200, beta=4.0):
    """Focused sub-gradient on the top-k lags (much faster, stronger)."""
    n = len(y)
    k_max = min(top_k, len(ac))
    top = np.argpartition(ac, -k_max)[-k_max:]
    tv = ac[top]
    w = tv ** beta
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws
    g = np.zeros(n)
    for wi, k in zip(w, top):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i1 > i0:
            idx = np.arange(i0, i1)
            g[idx] += wi * y[k - idx]
    return g

def sens_vector(a, src):
    """dc/d a[src] over all lags k:  = a[k-src] for 0<=k-src<n else 0."""

    i1 = src + n
    i1 = min(m, i1)
    if i1 > i0:
        v[i0:i1] = a[0:(i1 - i0)]
    return v

def freq_perturb(a, strength=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    ph = np.angle(A) + rng.normal(0, strength, len(A))
    an = np.fft.irfft(mag * np.exp(1j * ph), fft_size)[:n]
    an = np.maximum(an, 0.0)
    s = an.sum()
    return an / s if s > 1e-15 else a.copy()

def _record(a, best):

        return best
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    return best

            best = (parent.copy(), ps if np.isfinite(ps) else float("inf"))
    except Exception as e:
        print("Parent load failed:", e)

    if best[0] is None:
        a = make_parametric(3600, "beta")
        best = (a, evaluate_safe(a))
        print(f"Default init n={len(a)} score={best[1]:.6f}")

    n0 = len(best[0])
    rng = np.random.default_rng(2024)

    def remaining():
        return budget - (time.monotonic() - t_global)

    # ---- Phase 1: sub-gradient polish (light) ----
    print("\n[Phase 1] AMD sub-gradient polish")
    if remaining() > 60:
        t1 = min(remaining() * 0.12, 150.0)
        out = amd_optimize(best[0], t1, top_k=90, beta=4.0, c0=3.0, seed=3)
        best = _record(out, best)
        print(f"  after P1 score={best[1]:.6f}")

    # ---- Phase 2: local search on parent length ----
    print("\n[Phase 2] targeted local search (n=%d)" % n0)
    if remaining() > 60:
        t2 = min(remaining() * 0.30, 520.0)
        out = local_search(best[0], t2, verbose=True, seed=11)
        best = _record(out, best)
        print(f"  after P2 score={best[1]:.6f}")

    # ---- Phase 3: LP-FW global step + re-local-search ----
    print("\n[Phase 3] LP-FW + local search (n=%d)" % len(best[0]))
    if remaining() > 120 and len(best[0]) <= 4600:
        t3a = min(remaining() * 0.10, 160.0)
        lw = lp_frank_wolfe(best[0], t3a, top_K=300, margin_frac=2e-3, verbose=True)
        cand = _record(lw, best)
        if remaining() > 60:
            t3b = min(remaining() * 0.14, 240.0)
            ls = local_search(cand[0], t3b, verbose=False, seed=23)
            cand = _record(ls, cand)
        best = cand
        print(f"  after P3 score={best[1]:.6f}")

    # ---- Phase 4: multi-length exploration ----
    print("\n[Phase 4] length exploration")
    base = len(best[0])
    lens = sorted(set([base - 200, base, base + 200, base + 500, base + 900]))
    lens = [L for L in lens if 200 <= L <= 5000]
    for L in lens[:5]:
        if remaining() < 110:
            break
        print(f"  -- length {L}")
        aL = interpolate_sequence(best[0], L)
        tLa = min(remaining() * 0.10, 150.0)
        out = local_search(aL, tLa, verbose=False, seed=5 + L)
        cand = _record(out, (None, float("inf")))
        if cand[1] < best[1] and remaining() > 60:
            tLb = min(remaining() * 0.08, 120.0)
            out2 = amd_optimize(cand[0], tLb, top_k=100, beta=4.0, c0=3.0, seed=L)
            cand = _record(out2, cand)
        best = _record(best[0], best)  # keep
        best = (cand[0], cand[1]) if (cand[1] < best[1] or best[0] is None) else best
        print(f"     best score={best[1]:.6f} (n={len(best[0])})")

    # ---- Phase 5: symmetry-break restarts + heavy local search ----
    print("\n[Phase 5] diversify + local search")
    for trial in range(3):
        if remaining() < 90:
            break
        a_p = freq_perturb(best[0], strength=0.02, rng=rng) if trial % 2 == 0 \
            else make_parametric(len(best[0]), "skew")
        # blend with current best to keep quality
        a_p = 0.5 * norm_sum1(a_p) + 0.5 * best[0]
        a_p = norm_sum1(a_p)
        t5 = min(remaining() * 0.14, 220.0)
        out = local_search(a_p, t5, verbose=False, seed=101 + trial)
        best = _record(out, best)
        print(f"  P5 trial {trial} score={best[1]:.6f}")

    # ---- Phase 6: final heavy local grind on global best ----
    print("\n[Phase 6] final grind")
    if remaining() > 60:
        t6 = remaining() * 0.70
        t6 = min(t6, 480.0)
        out = local_search(best[0], t6, verbose=True, seed=777)
        best = _record(out, best)
        print(f"  final score={best[1]:.6f}")

    # ---- final validation ----
    a = best[0]
    if a is None:
        a = make_parametric(3600, "beta")
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = make_parametric(3600, "beta")
    final_score = evaluate_safe(a)
    print(f"\nFINAL score={final_score:.6f} n={len(a)} "
          f"time={time.monotonic()-t_global:.0f}s")

    rem = float(budget_s) - (time.monotonic() - t_global)
    if rem < 25:
        print("WARNING: low remaining time, returning")
    return [float(x) for x in a]
