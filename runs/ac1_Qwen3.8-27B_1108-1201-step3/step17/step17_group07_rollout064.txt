def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, w, N):
    n = len(a)
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(w, N)
    corr = np.fft.irfft(W_f * np.conj(A_f), N)
    return 2.0 * corr[:n]

def lp_remez_single(a, active_set, solver):
    n = len(a)
    b = fast_conv(a)

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active_set:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(solver=cp.SCS, verbose=False)
    except Exception:
        return a.copy(), prob.status

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), prob.status

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy(), prob.status
    return a_new / s_new, prob.status

def remez_method(a0, time_limit, solver=None, max_iter=500, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    a = a / s

    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    b = fast_conv(a)
    active_set = [int(np.argmax(b))]

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)

        max_active = max(b[k] for k in active_set)
        thresh = max_active * 0.999
        new_active = [k for k in range(len(b)) if b[k] >= thresh]
        if len(new_active) > 25:
            new_active = sorted(new_active, key=lambda k: -b[k])[:25]

            new_active = [int(np.argmax(b))]
        active_set = new_active

        a_cand, status = lp_remez_single(a, active_set, solver)
        if status not in ('optimal', 'optimal_inaccurate'):
            break

        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1.0 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                break

            if m_full < best_m - 1e-18:
                a = a_cand
                best_m = m_full
                best_a = a.copy()
            else:
                break

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez it={it} ({elapsed:.1f}s): J={J:.8f}, |K|={len(active_set)}")

def coordinate_descent(a0, time_limit=30.0, seed=999):

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            for frac in [0.001, 0.005, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    else:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to_n(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
