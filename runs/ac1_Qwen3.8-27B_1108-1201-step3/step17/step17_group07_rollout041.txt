import cvxpy as cp
import time

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, b, active_lags):
    """Compute subgradient of max(b[k] for k in active_lags).
    grad b_k[j] = 2*a[k-j] if 0 <= k-j < n else 0
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        k = int(k)
        # g[j] += 2 * a[k-j] for j where 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[k - idx]
    g /= len(active_lags)
    return g

def simplex_projection(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Binary search for the threshold
    lo, hi = v.min(), v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.maximum(v - mid, 0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:

    proj = np.maximum(v - (lo + hi) / 2, 0)
    # Fix rounding

    if s > 0:

    return proj

def lp_remez(a0, max_iter=100, r=0.1, solver=None, verbose=False):
    """LP Remez step."""
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    
    b = fast_conv(a)
    max_b = b.max()
    
    # Get active constraints
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > 40:
        active = np.argsort(b)[::-1][:40]
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k])
    
    # Trust region
    constraints.append(a_var <= a * (1 + r))
    constraints.append(a_var >= a * (1 - r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-5.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines: (1) Multiplicative weight descent with exponentially-weighted multi-lag subgradient,
(2) Projected gradient with backtracking line search, (3) LP Remez with adaptive trust region,
(4) Frank-Wolfe for diversity, (5) Multi-length exploration.
Key innovation: Exponential-weighted subgradient emphasizes critical lags more strongly,
and adaptive method switching avoids stagnation.

import numpy as np
import cvxpy as cp
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_weighted_subgradient(a, b, active_lags, weights):
    """Compute weighted subgradient of sum(w_i * b[k_i])."""
    n = len(a)
    m = len(active_lags)
    g = np.zeros(n)
    for idx in range(m):
        k = int(active_lags[idx])
        w = weights[idx]
        if w < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx_arr = np.arange(j_start, j_end + 1)
            g[idx_arr] += w * 2.0 * a[k - idx_arr]
    total_w = weights.sum()
    if total_w > 1e-15:
        g /= total_w
    return g


def simplex_projection(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.maximum(v - mid, 0.0)
        s = proj.sum()
        if abs(s - target_sum) < 1e-15:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.maximum(v - (lo + hi) / 2.0, 0.0)
    s = proj.sum()
    if s > 1e-15:
        proj *= target_sum / s
    else:
        proj = np.full(n, target_sum / n)
    return proj


def make_init(n, kind='triangle'):
    """Create initial sequence of given type and length."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-5.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(3.0 * (x - 0.5) * 2.0)
    elif kind == 'bump':
        a = np.exp(-10.0 * (x - 0.5) ** 2) + 0.1 * np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def mwd_optimize(a0, time_limit, seed=42, verbose=False):
    """
    Multiplicative weight descent with Nesterov acceleration
    and exponentially-weighted multi-lag subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    c = 4.0
    stall = 0
    max_stall = 1500
    temperature = 0.01

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Compute convolution
        b = fast_conv(y)
        M = b.max()

        # Find active lags with exponential weights
        threshold = M * 0.998
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 60:
            # Keep top 60 by value
            active = np.argsort(b)[::-1][:60]

        # Exponential weights: emphasize lags closer to max
        b_active = b[active]
        weights = np.exp(-(M - b_active) / (M * temperature))
        # Normalize
        weights = weights / weights.sum()

        # Compute weighted subgradient
        g = compute_weighted_subgradient(y, b, active, weights)

        # Adaptive step size with backtracking
        eta = c / (np.sqrt(t) + 1.0)

        # Multiplicative update
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            # Slowly decrease temperature
            temperature = max(0.001, temperature * 0.999)
        else:
            stall += 1

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = max(200, int(max_stall * 0.65))
            c = max(0.3, c * 0.65)
            temperature = 0.01

            strategy = int(rng.integers(4))
            if strategy == 0:
                # Gaussian perturbation
                x = best_a * (1.0 + rng.normal(0, 0.015, n))
            elif strategy == 1:
                # Random subset rescale
                x = best_a.copy()
                n_pert = max(10, n // 8)
                idx = rng.choice(n, size=n_pert, replace=False)
                x[idx] *= rng.uniform(0.6, 1.5, len(idx))
            elif strategy == 2:
                # Frank-Wolfe style move
                j = int(rng.integers(n))
                t_fw = rng.uniform(0.05, 0.2)
                e_j = np.zeros(n)
                e_j[j] = 1.0
                x = (1.0 - t_fw) * best_a + t_fw * e_j
            else:
                # Blend with a new initialization
                kind = ['raised_cos', 'gaussian', 'sech'][int(rng.integers(3))]
                new_init = make_init(n, kind)
                x = 0.7 * best_a + 0.3 * new_init

            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"    MWD t={t} ({elapsed:.1f}s): obj={obj:.8f}, stall={stall}, temp={temperature:.4f}")

    return best_a, best_m


def pgd_optimize(a0, time_limit, seed=77, verbose=False):
    """
    Projected gradient descent with backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    c = 2.0
    stall = 0
    max_stall = 500

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        M = b.max()

        # Active lags with exponential weights
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]

            active = np.argsort(b)[::-1][:40]

        # Backtracking line search

        improved = False

                a = a_new
                improved = True
                c = min(5.0, c * 1.1)

        if not improved:
            stall += 1
            c = max(0.05, c * 0.8)
            if stall > max_stall:

                idx = rng.choice(n, size=n_pert, replace=False)

                stall = 0
                max_stall = max(200, int(max_stall * 0.7))

        if verbose and t % 5000 == 0:

def frank_wolfe_optimize(a0, time_limit, seed=123, verbose=False):

        b = fast_conv(a)

            active = np.argsort(b)[::-1][:30]

        b_active = b[active]
        weights = np.exp(-(M - b_active) / (M * 0.003))
        weights = weights / weights.sum()

        g = compute_weighted_subgradient(a, b, active, weights)

        # Linear minimization oracle over simplex: pick argmin g
        j_min = int(np.argmin(g))

        # Try several step sizes (exact line search approximation)
        best_local_m = M
        best_local_frac = 0.0
        for frac in [2.0 / (t + 2), 1.0 / (t + 1), 0.5 * 2.0 / (t + 2), 0.1]:
            a_new = (1.0 - frac) * a + frac * np.zeros(n)
            a_new[j_min] = frac
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:

            a_new = a_new / s
            m_new = fast_conv(a_new).max()
            if m_new < best_local_m:
                best_local_m = m_new
                best_local_frac = frac

        if best_local_frac > 0 and best_local_m < best_m - 1e-18:
            a = (1.0 - best_local_frac) * a + best_local_frac * np.zeros(n)
            a[j_min] = best_local_frac
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = best_local_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Random restart perturbation
            n_pert = max(5, n // 8)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= rng.uniform(0.4, 2.0, len(idx))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall = 0
            max_stall = max(300, int(max_stall * 0.7))

        if verbose and t % 10000 == 0:

def lp_remez_single(a, r=0.1, max_cons=50):
    """Single LP Remez step with linearized constraints."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    # Get active constraints
    threshold = max_b * 0.998
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            b_k_val = b[k_int]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b_k_val)

    # Trust region (box constraints relative to current point)
    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * (1.0 - r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def remez_optimize(a0, time_limit, r_init=0.08, verbose=False):
    """Iterative LP Remez with adaptive trust region and line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(int(time.time()))

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        a_cand = lp_remez_single(a, r=r, max_cons=50)
        if a_cand is None or np.allclose(a_cand, a):
            no_improve += 1
            continue

        # Line search
        accepted = False
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1.0 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                best_m = m_trial
                best_a = a_trial.copy()
                a = a_trial
                accepted = True
                r = min(0.4, r * 1.15)
                no_improve = 0
                break

        if not accepted:
            no_improve += 1
            r = max(0.01, r * 0.7)
            if no_improve > 12:
                # Perturb
                n_pert = max(5, n // 6)
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = 0.08

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): obj={obj:.8f}, r={r:.4f}, no_imp={no_improve}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_polish(a0, time_limit, seed=999):
    """Coordinate descent with random pair moves for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = max(30, n // 80)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(150, n // 4)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
            max_stall = max(30, int(max_stall * 1.2))
        else:
            stall += 1

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Combines multiple optimization methods
    with adaptive time allocation and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting optimization. n={n0}, initial score={best_score:.8f}")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: MWD from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 1: MWD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = mwd_optimize(best_seq, t1, seed=42, verbose=True)
        check_update(a_out, f"MWD_parent_n{n0}")

    # === Phase 2: PGD (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: PGD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = pgd_optimize(best_seq, t2, seed=77, verbose=True)
        check_update(a_out, f"PGD_n{len(best_seq)}")

    # === Phase 3: LP Remez (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: LP Remez n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        try:
            a_out, _ = remez_optimize(best_seq, t3, r_init=0.10, verbose=True)
            check_update(a_out, f"Remez_n{len(best_seq)}")
        except Exception as e:
            print(f"    Remez error: {e}")

    # === Phase 4: MWD polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: MWD polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = mwd_optimize(best_seq, t4, seed=777, verbose=False)
        check_update(a_out, f"MWD_polish_n{len(best_seq)}")

    # === Phase 5: Frank-Wolfe for diversity (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: FW n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = frank_wolfe_optimize(best_seq, t5, seed=314, verbose=False)
        check_update(a_out, f"FW_n{len(best_seq)}")

    # === Phase 6: Alternate length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        # Try a few nearby lengths
        for n_try in [n0 + 100, n0 - 100, n0 + 200]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 500:
                continue
            print(f"\n=== Phase 6: n={n_try} (budget={remaining:.0f}s) ===")
            a_interp = interpolate_sequence(best_seq, n_try)
            t6 = min(remaining * 0.4, 120.0)
            if t6 > 20:
                a_out, _ = mwd_optimize(a_interp, t6, seed=555, verbose=False)
                check_update(a_out, f"MWD_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = pgd_optimize(a_interp, remaining - 5, seed=222, verbose=False)
                check_update(a_out, f"PGD_n{n_try}")

    # === Phase 7: Final LP Remez polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 7: Final Remez n={len(best_seq)} (budget={remaining:.0f}s) ===")
        try:
            a_out, _ = remez_optimize(best_seq, remaining - 5, r_init=0.03, verbose=False)
            check_update(a_out, f"Remez_final_n{len(best_seq)}")
        except Exception:
            pass

    # === Phase 8: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 8: Coord polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
        a_out, _ = coordinate_polish(best_seq, remaining - 3, seed=999)
        check_update(a_out, f"CD_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
