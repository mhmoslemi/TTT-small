Wait: M_k[i,j] = 1 iff j = k-i. M_k[j,i] = 1 iff i = k-j. These are the same condition! So M_k is symmetric. ✓

And a^T M_k a = sum_{i,j} a[i] M_k[i,j] a[j] = sum_i a[i] * a[k-i] (summing over j=k-i). ✓

So f_k(a) = a^T M_k a where M_k is symmetric but not PSD.

The function f_k is convex iff M_k is PSD. Since M_k is not PSD in general, f_k is not convex.

However, on the non-negative orthant (a≥0), f_k(a) = sum_i a[i]*a[k-i] ≥ 0. But non-negativity doesn't imply convexity.

Let me check: is f_k convex on a≥0? Take n=3, k=1: f_1(a) = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]. This is convex on a≥0? The Hessian is [[0,2],[2,0]] (in the (a[0],a[1]) subspace), which has eigenvalues ±2. So it's NOT convex!

This means the problem min max_k f_k(a) s.t. a≥0, sum(a)=1 is NOT a convex optimization problem. This explains why the previous approaches used heuristic methods (mirror descent, LP heuristics) rather than exact convex optimization.

def fast_convolution(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, b, K=50, beta=5.0):
    """Compute weighted subgradient of max(b) using top K peaks."""
    n = len(a)
    K = min(K, len(b))
    top_idx = np.argpartition(b, -K)[-K:]
    top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
    top_vals = np.maximum(b[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for ki in range(K):
        k = int(top_idx[ki])
        wi = 2.0 * w[ki]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wi * a[k - idx]
    return grad

def mirror_descent(a0, time_limit, K_init=30, beta_init=5.0, c0=3.0,
                   restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent for minimizing max convolution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    b = fast_convolution(a)
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_K = K_init
    cur_beta = beta_init
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        theta = (t - 1.0) / (t + 1.0)
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        b_y = fast_convolution(y)
        g = compute_subgradient(y, b_y, K=cur_K, beta=cur_beta)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_x = np.log(y) - eta * g
        log_x -= log_x.max()  # numerical stability
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        # Track best
        b_x = fast_convolution(x)
        M = b_x.max()
        if M < best_m - 1e-12:
            best_m = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart logic
        if stall >= restart_every:
            stall = 0
            # Lévy perturbation
            pert = rng.standard_t(df=3, size=n) * 0.05 * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            # Adapt parameters
            cur_K = min(cur_K + 15, 150)
            cur_beta = max(cur_beta - 0.5, 2.0)
            c0 = c0 * 1.1
            
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.0f}s): J={J:.6f}, K={cur_K}, beta={cur_beta:.1f}, stall={stall}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, n_peaks=20, verbose=False, rng_seed=42):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    b = fast_convolution(a)
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_convolution(a)
        M = b.max()
        if M < best_m - 1e-12:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        n_top = min(n_peaks, 2*n-1)
        top_k = np.argpartition(b, -n_top)[-n_top:]
        top_k = top_k[np.argsort(b[top_k])[::-1]]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            # For this peak, the contributing indices are i and k-i
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            
            # Find the pair with max product
            idx = np.arange(i_lo, i_hi + 1)
            products = a[idx] * a[k - idx]
            max_j = idx[np.argmax(products)]
            
            src = int(max_j)
            dst = int(k - max_j)
            
            if src == dst:
                # Single element contributes (a[src]^2)
                # Try reducing a[src]
                old_val = a[src]
                if old_val < 1e-12:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = old_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    # Redistribute to random positions far from peaks
                    n_redist = max(1, n // 10)
                    targets = rng.choice(n, n_redist, replace=False)
                    a_trial[targets] += amt / n_redist
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_convolution(a_trial).max()
                    if M_trial < M - 1e-14:
                        a = a_trial
                        improved = True
                        break
            else:
                # Two elements contribute: a[src]*a[dst]
                # Try transferring mass from the larger to a distant position
                if a[src] >= a[dst]:
                    source, target_peak = src, dst
                else:
                    source, target_peak = dst, src
                
                old_val = a[source]
                if old_val < 1e-12:
                    continue
                
                # Find a good destination: low contribution to all top peaks
                best_dest = -1
                best_score = float('inf')
                # Sample some candidate destinations
                n_candidates = min(50, n)
                candidates = rng.choice(n, n_candidates, replace=False)
                for d in candidates:
                    # Estimate the increase in max convolution if we add mass at d
                    # (rough estimate: max over top peaks of a[d's contribution])
                    cost = 0.0
                    for k2 in top_k[:5]:
                        k2i = int(k2)
                        j = k2i - d
                        if 0 <= j < n:
                            cost += a[j]
                    if cost < best_score:
                        best_score = cost
                        best_dest = int(d)
                
                if best_dest < 0 or best_dest == source:
                    continue
                
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = old_val * frac
                    a_trial = a.copy()
                    a_trial[source] -= amt
                    a_trial[best_dest] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_convolution(a_trial).max()
                    if M_trial < M - 1e-14:
                        a = a_trial
                        improved = True
                        break
        
        if not improved:
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_convolution(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, top_K=200, verbose=False):

    for s in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    y = a.copy()
    x_momentum = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = b_y.max()
        
        # Select binding constraints
        K_sel = min(top_K, 2*n - 1)
        binding = np.argpartition(b_y, -K_sel)[-K_sel:]
        
        # Build constraint matrix (sparse would be better but dense for now)
        # T @ v <= M_curr, where T[i,j] = y[j] if i-j is valid
        T = np.zeros((K_sel, n))
        for ii, k in enumerate(binding):
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                T[ii, j_lo:j_hi+1] = y[j_lo:j_hi+1]
        
        v = cp.Variable(n, nonneg=True)
        constraints = [T @ v <= M_curr + 1e-10]
        prob = cp.Problem(cp.Maximize(cp.sum(v)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)

            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):

        s_v = v_val.sum()
        if s_v < 1e-15:
            continue
        v_dir = v_val / s_v
        
        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * v_dir
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        
        # Nesterov
        if t > 1:
            theta = (t - 1) / (t + 2)
            x_momentum = y_new + theta * (y_new - y)
            x_momentum = np.maximum(x_momentum, 0.0)
            x_momentum = x_momentum / x_momentum.sum()
        
        y = y_new
        
        # Track best
        for candidate in [y, x_momentum]:
            M_cand = fast_convolution(candidate).max()

        if verbose and t % 10 == 0:

def make_init(n, init_type='cosine'):
    """Generate initialization."""
    x = np.linspace(0, 1, n)
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'cosine':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif init_type == 'sinc':
        a = np.sinc((x - 0.5) * 6.0) ** 2
    elif init_type == 'power':
        a = np.linspace(0.1, 1.0, n)
    elif init_type == 'bump':
        a = np.exp(-10 * (x - 0.5) ** 2) + 0.01
    elif init_type == 'multibump':
        a = np.zeros(n)
        for c in [0.2, 0.5, 0.8]:
            a += np.exp(-20 * (x - c) ** 2)
        a += 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return normalize(a)

def random_smooth_init(n, seed, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):

        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return normalize(a)
