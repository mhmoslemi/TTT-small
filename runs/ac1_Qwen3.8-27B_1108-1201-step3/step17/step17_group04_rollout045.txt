def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):

def convolve_matrix(y):
    """Build the convolution matrix T such that T @ b = conv(y, b)."""
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def get_subgradient(a, ac, exact=True, top_k=10, beta=3.0):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def build_conv_matrix_efficient(y):
    """Build convolution matrix T where T[b] = conv(y, b) using structured approach."""
    n = len(y)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(y[k - i])
    T = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def lp_frank_wolfe_improved(a0, time_limit, solver, verbose=False,
                             use_all_constraints=True, max_iters=5000):
    """
    Improved LP Frank-Wolfe with backtracking line search.
    Uses the LP: max sum(b) s.t. b>=0, conv(y,b) <= M
    with adaptive constraint selection and backtracking steps.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute structure of constraints
    n_constraints = 2 * n - 1

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix (dense for small n, sparse for large)
        if n <= 4000:
            T = convolve_matrix(a)
        else:
            T = build_conv_matrix_efficient(a)

        # Select constraints: use all if feasible, else top-K
        if use_all_constraints and n_constraints <= 7000:
            binding_idx = np.arange(n_constraints)
        else:
            K_sel = min(2000, n_constraints)
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T[binding_idx, :]
        M_vec = M_curr * np.ones(len(binding_idx))

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = 0.0
        best_trial_m = best_m
        base_m = ac.max()

        for trial in range(10):
            a_new = (1.0 - alpha) * a + alpha * g_dir
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_new = fast_autocorr(a_new).max()
            if m_new < best_trial_m:
                best_trial_m = m_new
                best_alpha = alpha
                best_a_new = a_new.copy()
            alpha *= 0.5
            if alpha < 1e-6:
                break

        if best_alpha > 0 and best_trial_m < best_m - 1e-14:
            a = best_a_new
            best_m = best_trial_m
            best_a = a.copy()
        else:
            # Standard FW step as fallback
            a_new = (1.0 - 2.0 / (t + 2.0)) * a + (2.0 / (t + 2.0)) * g_dir
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, alpha={best_alpha:.4f}")

    return best_a, best_m


def nesterov_mirror_descent(a0, time_limit, verbose=False,
                            top_k=15, beta=3.0, c=3.0,
                            exact_frac=0.5, restart_every=3000,
                            seed=42):
    """Nesterov-accelerated mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        # Adaptive subgradient
        use_exact = rng.random() < exact_frac
        if use_exact:
            g = get_subgradient(y, ac, exact=True)
        else:
            g = get_subgradient(y, ac, exact=False, top_k=cur_top_k, beta=cur_beta)

        # Adaptive step size
        lower_bound = 1.0 / n
        gap = max(M - lower_bound, 1e-10)
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        # Mirror descent step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_c = min(cur_c + 0.3, 6.0)
            exact_frac = min(exact_frac + 0.03, 0.7)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SGD restart ({elapsed:.0f}s): J={best_J:.6f}")

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def peak_targeted_optimization(a0, time_limit, verbose=False, seed=123):
    """
    Iteratively target the peak of the convolution and solve a local
    optimization to reduce it while maintaining sum constraint.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))

        # Find positions that contribute to the peak
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n, peak_lag + 1)
        contrib_idx = np.arange(i_start, i_end)
        contrib_vals = a[contrib_idx] * a[peak_lag - contrib_idx]

        # Try to reduce contribution by moving mass from high-contribution
        # positions to positions that don't contribute to the peak
        high_contrib = contrib_idx[np.argsort(contrib_vals)[::-1][:n // 4]]
        # Find "safe" positions (not contributing to top 5 peaks)
        top5_lags = np.argsort(ac)[::-1][:5]
        dangerous = set()
        for k in top5_lags:
            ks = max(0, int(k) - n + 1)
            ke = min(n, int(k) + 1)
            dangerous.update(range(ks, ke))
        safe_positions = np.array([i for i in range(n) if i not in dangerous])

        if len(safe_positions) == 0 or len(high_contrib) == 0:
            stall += 1
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            src = high_contrib[rng.integers(len(high_contrib))]
            if a[src] < 1e-10:
                continue
            dst = safe_positions[rng.integers(len(safe_positions))]
            if src == dst:
                continue

            # Try several transfer amounts
            for frac in [0.001, 0.003, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Escape: random perturbation
            if stall % 30 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.01, 0.05)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    J_new = compute_J(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakOpt ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def softmax_smooth_minimize(a0, time_limit, verbose=False, seed=555):
    """
    Minimize a softmax-smoothed version of the max convolution,
    with increasing temperature.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    T = 2.0  # Temperature (1/T controls sharpness)
    eta = 0.1
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        # Softmax weights
        T_ac = ac / T
        T_ac = T_ac - T_ac.max()
        p = np.exp(T_ac)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones_like(ac) / len(ac)
        else:
            p = p / p_sum

        # Gradient of smooth max
        grad = np.zeros(n)
        # Only process top contributions for efficiency
        top_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        for k in top_idx:
            pk = p[int(k)]
            if pk < 1e-10:
                continue

            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * pk * a[j_idx]

        # Mirror descent step
        eta_t = eta / (1.0 + 0.001 * t)
        log_a = np.log(a + 1e-30) - eta_t * (grad + 1e-10)
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()

def make_symmetric_init(n):
    """Symmetric initialization (high at edges)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    return norm_sum1(a)


def make_power_law_init(n, power=0.7):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_bimodal_init(n, sigma=0.15):
    """Bimodal: two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / sigma) ** 2) + np.exp(-((x - 0.9) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_spectral_init(n, flatness=0.5):
    """
    Spectral initialization: create a flat-ish magnitude spectrum,
    then take inverse FFT and take absolute value.
    """
    # Create a target spectrum that's relatively flat
    fft_size = 1 << (2 * n - 1).bit_length()
    freqs = np.fft.rfftfreq(fft_size)
    # Shape: slightly elevated at certain frequencies to avoid zero mean
    target_mag = 1.0 + flatness * np.exp(-((freqs - 0.5) / 0.1) ** 2)
    # Random phase
    rng = np.random.default_rng(42)
    phase = rng.uniform(0, 2 * np.pi, len(target_mag))
    spectrum = target_mag * np.exp(1j * phase)
    # Inverse FFT
    a_full = np.fft.irfft(spectrum, fft_size)
    # Take the first n values, make non-negative
    a = np.abs(a_full[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_symmetric_init(n)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t2, solver, verbose=True,
                                           use_all_constraints=(n_best <= 3500),
                                           max_iters=5000)
        try_update(a_out, f"lpfw_main_n{n_best}")

    # === Phase 3: Peak-targeted optimization ===
    print(f"\n=== Phase 3: Peak Targeted ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_targeted_optimization(a_in, t3, verbose=True, seed=123)
        try_update(a_out, f"peak_n{len(a_in)}")

    # === Phase 4: LP-FW second round ===
    print(f"\n=== Phase 4: LP-FW Round 2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 280.0)
    if t4 > 40 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t4, solver, verbose=True,
                                           use_all_constraints=(n_best <= 3500),
                                           max_iters=5000)
        try_update(a_out, f"lpfw_r2_n{n_best}")

    # === Phase 5: Softmax smooth minimize ===
    print(f"\n=== Phase 5: SoftMax Minimize ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_smooth_minimize(a_in, t5, verbose=False, seed=555)
        try_update(a_out, f"softmax_n{len(a_in)}")

    # === Phase 6: SGD polish with different params ===
    print(f"\n=== Phase 6: SGD Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t6, verbose=False,
                                           top_k=30, beta=2.0, c=5.0,
                                           exact_frac=0.6, restart_every=5000,
                                           seed=999)
        try_update(a_out, f"sgd_polish_n{len(a_in)}")

    # === Phase 7: Try longer sequences ===
    print(f"\n=== Phase 7: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for delta in [400, 800]:
        new_n = n_best + delta
        if new_n > 5000:
            break
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.15, 100.0)
        if t7a > 20:
            a_out, _ = nesterov_mirror_descent(a_long, t7a, verbose=False,
                                               top_k=15, beta=3.0, c=3.0,
                                               exact_frac=0.5, restart_every=2000,
                                               seed=777)
            try_update(a_out, f"sgd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3500:
            a_in = norm_sum1(a_out)
            t7b = min(remaining * 0.10, 60.0)
            if t7b > 20:
                a_out2, _ = lp_frank_wolfe_improved(a_in, t7b, solver, verbose=False,
                                                     use_all_constraints=True, max_iters=2000)
                try_update(a_out2, f"lpfw_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final LP-FW on best ===
    print(f"\n=== Phase 8: Final LP-FW ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 3500:
        t8 = min(remaining * 0.4, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_improved(a_in, t8, solver, verbose=True,
                                               use_all_constraints=True, max_iters=5000)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 9: Final SGD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_mirror_descent(a_in, t9, verbose=False,
                                               top_k=40, beta=2.0, c=5.0,
                                               exact_frac=0.7, restart_every=8000,
                                               seed=4321)
            try_update(a_out, f"sgd_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
