where `conv(a, a)` is the full linear convolution of the sequence with itself.

The key insight is that this is related to the "sidelobe" problem in signal processing. We want a sequence whose autocorrelation has a flat peak (all peaks equal), which would minimize the ratio.

**Key observations:**
- The objective is `2n * M / S^2` where M = max autocorrelation and S = sum
- Normalizing to sum=1, we minimize `2n * max(ac)`
- This is equivalent to minimizing the peak sidelobe level (PSL) of the autocorrelation

**Different approaches to consider:**

1. **Proximal gradient with adaptive step sizes** - Use the exact subgradient of the max function (not soft-max) and use proximal methods.

2. **Newton-type methods** - The max of linear functions can be handled with active set methods. If we identify which lag is the maximum, we can do gradient descent on that specific constraint.

3. **Alternating minimization** - Alternate between fixing which lag is the max and optimizing the sequence.

4. **Spectral methods** - Use the fact that the autocorrelation is the IFFT of |FFT(a)|^2. We can work in the frequency domain.

5. **Active set LP** - At the optimum, some subset of lags will be exactly at the maximum. We can identify this active set and solve the resulting LP/QP exactly.

6. **Riemannian optimization** - Work on the simplex directly with Riemannian gradient descent.

7. **ADMM (Alternating Direction Method of Multipliers)** - Split the problem into parts that are easy to solve.

**Approach: Active Set + QP**

At the optimum, there exists some set of lags S such that `ac(m) = M` for m in S and `ac(m) < M` for m not in S. We can:
1. Start with an initial guess
2. Identify the "active" lags (those closest to max)
3. Solve the QP: min M subject to ac(m) <= M for m in S, a >= 0, sum(a) = 1
4. Check if any inactive lag exceeds M, add it to S
5. Repeat

This is essentially a cutting-plane method for the outer max.

**Spectral initialization:**
We want ac(m) to be as flat as possible. In the frequency domain, this means |FFT(a)|^2 should be as close to constant as possible (since flat autocorrelation = flat power spectrum).

So we want a sequence whose power spectrum is flat - i.e., a "white" sequence. But with non-negativity constraint.

One approach: Start with a random sequence, take its FFT, set |FFT| = constant, take IFFT, project onto non-negative simplex. Iterate.

This is like a Gerchberg-Saxton algorithm for "flat spectrum" design.

Let me code this up:

```python
def flat_spectrum_init(n, n_iter=100, seed=42):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    
    for it in range(n_iter):
        # FFT
        A = np.fft.rfft(a, 2*n-1)
        # Set flat spectrum
        target_mag = (np.sum(np.abs(A)**2) / len(A)) ** 0.5
        A_flat = target_mag * A / np.abs(A)  # preserve phase
        # IFFT
        a_new = np.fft.irfft(A_flat, 2*n-1)[:n]
        # Project onto non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a = a_new
    
    return a
```

"""
SLP-QP optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses sequential linear programming with trust-region QP subproblems,
active-set reduction, Nesterov gradient polishing, and spectral initialization.
"""

import numpy as np
import time
import cvxpy as cp

# Global solver cache
_solver = None

def _get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x)==1, x<=0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except:
            continue
    _solver = cp.SCS
    return _solver

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fs = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A*A, fs)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, float), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1.0
    rho = np.nonzero(mu * np.arange(1, n+1) > cv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    tau = (mu[rho] + 1.0/(rho+1.0))
    return np.maximum(v - tau, 0.0)

def build_C(a_old):
    n = len(a_old)
    C = np.zeros((2*n-1, n))
    for k in range(n):
        C[k:k+n, k] = 2.0 * a_old
    return C

def slp_qp_step(a_old, lam, solver):
    n = len(a_old)
    ac_old = fast_autocorr(a_old)
    C = build_C(a_old)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    obj = cp.Minimize(M_var + lam/2.0 * cp.sum_squares(a_var - a_old))
    cons = [C @ a_var - ac_old <= M_var, cp.sum(a_var) == 1]
    prob = cp.Problem(obj, cons)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    a_new = np.maximum(np.asarray(a_var.value, float), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None, None
    a_new /= s
    
    # Get dual variables for active set identification
    try:
        y = np.asarray(cons[0].dual_value, float)
    except:
        y = None
    
    return a_new, y

def slp_optimize(a_init, time_limit, solver, verbose=True, label=""):

    best_a = a.copy()

def nesterov_gradient_polish(a_init, time_limit, verbose=True, label=""):
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    n = len(a)
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = 0.1 / np.sqrt(n)
    beta = 5.0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_ratio = (it - 1.0) / (it + 2.0)
        y = x + t_ratio * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute soft subgradient
        ac = fast_autocorr(y)
        M_curr = ac.max()
        shifted = ac - M_curr
        w = np.exp(beta * shifted)
        w /= w.sum()
        
        # Gradient: sum_m w[m] * 2*a[m-k] for valid
        # Use FFT
        ac_len = len(ac)  # 2n-1
        fs = 1 << (n + ac_len - 1 - 1).bit_length()
        y_rev = np.zeros(fs)
        y_rev[:n] = y[::-1]
        w_pad = np.zeros(fs)
        w_pad[:ac_len] = w
        conv = np.fft.irfft(np.fft.rfft(y_rev) * np.fft.rfft(w_pad), fs)
        g = 2.0 * conv[n-1:2*n-1]
        
        # Nesterov step
        x_prev = x.copy()
        step = eta * g
        x_new = y - step
        x_new = project_simplex(x_new)
        x = x_new
        
        M_x = fast_autocorr(x).max()
        if M_x < best_M:
            best_M = M_x
            best_a = x.copy()
        
        # Adapt eta
        if M_x < M_curr - 1e-12:
            eta = min(eta * 1.05, 1.0)
        else:
            eta = eta * 0.9
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"  Grad {label} iter={it} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.6e}, eta={eta:.4f}")
    
    return best_a, best_M

def coordinate_polish(a_init, time_limit, verbose=True, label=""):
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    n = len(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        stall += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:20]
        
        improved = False
        for m in top_lags[:8]:
            mi = int(m)
            # Find the pair (j, mi-j) that contributes most to ac[mi]
            js = max(0, mi - n + 1)
            je = min(n - 1, mi)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[mi - idx]
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Try moving to a low-sensitivity position
            sens = np.zeros(n)
            for m2 in top_lags[:10]:
                m2i = int(m2)
                s2 = max(0, m2i - n + 1)
                e2 = min(n-1, m2i)
                for d in range(s2, e2+1):
                    partner = m2i - d
                    if 0 <= partner < n:
                        sens[d] += a[partner]
            
            # Find destination
            candidates = np.where((sens < np.median(sens)) & (np.abs(np.arange(n) - src) > n//10))[0]
            if len(candidates) == 0:
                candidates = np.where(sens < np.mean(sens))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dist = np.abs(candidates - src).astype(float)
            dist[dist==0] = n
            scores = sens[candidates] / dist
            dst = candidates[np.argmin(scores)]
            
            if dst == src:
                continue
            
            # Try fractions
            for frac in [0.05, 0.02, 0.01, 0.1, 0.03, 0.08]:
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0)
                a_t /= a_t.sum()
                Mt = fast_autocorr(a_t).max()
                if Mt < best_M - 1e-14:
                    a = a_t
                    best_M = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    Mt = fast_autocorr(a).max()
                    if Mt < best_M:
                        best_M = Mt
                        best_a = a.copy()
                        stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_M

def spectral_init(n, n_iter=200, seed=42):
    """Initialize with flat power spectrum (white noise in frequency domain)."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    
    for it in range(n_iter):
        # FFT to frequency domain
        A = np.fft.rfft(a, 2*n-1)
        # Set flat magnitude
        target_mag = np.sqrt(np.sum(np.abs(A)**2) / len(A))
        # Preserve phase, set magnitude
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_flat = target_mag * phase
        # IFFT
        a_new = np.fft.irfft(A_flat, 2*n-1)[:n]
        # Project
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a = a_new
    
    return a

import numpy as np
import time
import cvxpy as cp

_solver_cache = None

def _get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print(f"  Using solver: SCS (fallback)")
    return _solver_cache


def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1.0
    cond = mu * np.arange(1, n + 1) > cv
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    tau = (mu[rho] + 1.0 / (rho + 1.0))
    return np.maximum(v - tau, 0.0)


def build_C(a_old):
    """Build the linearized constraint matrix C where C[m,k] = 2*a_old[m-k] if valid."""
    n = len(a_old)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    two_a = 2.0 * a_old
    for k in range(n):
        C[k:k + n, k] = two_a
    return C


def slp_qp_solve(a_old, lam, solver):
    """Solve one SLP-QP subproblem. Returns (a_new, dual_values) or (None, None)."""
    n = len(a_old)
    ac_old = fast_autocorr(a_old)
    C = build_C(a_old)
    
    a_var = cp.Variable(n, nonneg=True)
    a_var.value = a_old
    M_var = cp.Variable()
    M_var.value = float(ac_old.max())
    
    obj = cp.Minimize(M_var + (lam / 2.0) * cp.sum_squares(a_var - a_old))
    cons = [C @ a_var - ac_old <= M_var, cp.sum(a_var) == 1]
    prob = cp.Problem(obj, cons)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None, None
    a_new /= s
    
    # Try to extract dual values
    try:
        dual = np.asarray(cons[0].dual_value, dtype=np.float64)
    except Exception:
        dual = None
    
    return a_new, dual


def slp_optimize(a_init, time_limit, solver, verbose=True, label=""):
    """Main SLP-QP optimization loop."""
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = float(fast_autocorr(a).max())
    n = len(a)
    t0 = time.monotonic()
    
    lam = 5.0 / np.sqrt(n)
    stall = 0
    max_stall = 80
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        a_new, dual = slp_qp_solve(a, lam, solver)
        
        if a_new is None:
            stall += 1
            lam *= 1.5
            if stall > 20:
                break
            continue
        
        M_new = float(fast_autocorr(a_new).max())
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adapt lambda based on step size
        step_norm = float(np.linalg.norm(a_new - a))
        if step_norm < 1e-10:
            lam *= 0.5
        elif step_norm > 2.0 / np.sqrt(n):
            lam *= 2.0
        else:
            # Slight decrease to encourage progress
            lam *= 0.95
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"  SLP[{label}] it={it} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.6e}, lam={lam:.2e}, stall={stall}")
        
        if stall > max_stall:
            if verbose:
                print(f"  SLP[{label}] stalled at it={it}, J={2.0*n*best_M:.6f}")
            break
    
    return best_a, best_M


def nesterov_grad_optimize(a_init, time_limit, verbose=True, label=""):
    """Nesterov-accelerated gradient descent with soft subgradient."""
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = float(fast_autocorr(a).max())
    n = len(a)
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = 0.5 / np.sqrt(n)
    beta = 6.0
    stall = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_ratio = (it - 1.0) / (it + 2.0)
        y = x + t_ratio * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
