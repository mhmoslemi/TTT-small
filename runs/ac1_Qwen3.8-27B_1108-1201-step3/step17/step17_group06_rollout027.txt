def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * conv.max()

    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_beta_dist(n, alpha=2.0, beta=1.0):
    x = np.linspace(0, 1, n)
    a = x**alpha * (1-x)**beta + 1e-10
    return a / a.sum()

def make_exp_ramp(n, rate=1.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return a / a.sum()

def build_conv_matrix(a, n):
    """Build sparse constraint matrix C where C[k,i] = a[k-i] if 0<=k-i<n."""
    # C is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i_arr = np.arange(i_start, i_end + 1)
            ki = k - i_arr  # k-i values, all in [0, n-1]
            rows.extend([k] * len(i_arr))
            cols.extend(i_arr.tolist())
            vals.extend(a[ki].tolist())
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C

def solve_lp(a, M, timeout=5.0):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0, sum(b) = 1."""
    n = len(a)
    C = build_conv_matrix(a, n)
    
    # Minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: C @ b <= M * ones
    A_ub = C
    b_ub = np.full(2*n - 1, M)
    
    # Equality: sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x
    return None

def lp_based_descent(a0, time_limit, verbose=False):
    """LP-based block coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        
        # Solve LP
        b = solve_lp(a, M, timeout=10.0)
        
        if b is not None and b.sum() > 1e-10:
            b = b / b.sum()
            b = np.maximum(b, 0.0)
            
            # Try full update
            J_b = eval_J(b)
            if J_b < best_J - 1e-12:
                a = b.copy()
                best_J = J_b
                best_a = a.copy()
                stall = 0
            else:
                # Try blending
                improved = False
                for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.8, 0.9]:
                    blend = (1-t)*a + t*b
                    blend = np.maximum(blend, 0.0)

                    J_blend = eval_J(blend)
                    if J_blend < best_J - 1e-12:
                        a = blend
                        best_J = J_blend
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if not improved:
                    stall += 1
                    if stall % 50 == 0 and verbose:
                        print(f"  LP stall={stall}, J={best_J:.6f}")
        else:
            stall += 1
            if verbose and stall % 10 == 0:
                print(f"  LP infeasible, stall={stall}")
        
        if verbose and (time.monotonic() - t0) % 30 < 1:
            print(f"  LP ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J
