Wait, I realize there's an issue with my Nesterov FW. In the standard Nesterov FW, the LP is solved at the extrapolated point y^{t+1/2}, not at x^t or y^t. Let me fix this.

Standard Nesterov FW (from Lan 2013, or Beck & Tetruashvili 2013):
- Initialize x^1 = y^1 = a^0
- For t = 1, 2, ...:
  - beta_t = (t-1) / (t+2)  [or (t-1)/sqrt(12t+3) for a different rate]
  - y^{t+1} = x^t + beta_t * (x^t - x^{t-1})  (for t > 1; y^1 = x^1)
  - Find s^t = argmin_{s in C} <g^t, s> where g^t is a subgradient at y^t
    (or for our case: s^t = argmin_{s in simplex} max_k (y^t * s)_k)
  - gamma_t = 2 / (t+2)
  - x^{t+1} = (1 - gamma_t) * y^t + gamma_t * s^t

So the LP should be solved at y^t (the extrapolated point), and the convex combination is between y^t and s^t.

1. Compute autocorrelation
2. Find the top K peaks
3. For each peak, find the most contributing pair (i, j) with i+j = k (or |i-j| = k depending on convention)
4. Try to reduce one of the pair and add to a low-impact position

Actually, let me think about this more carefully. The autocorrelation is:
(a*a)_k = sum_i a_i * a_{i+k} for k >= 0 (with appropriate bounds)

To reduce (a*a)_k, we can:
- Reduce a_i for some i in the sum
- Reduce a_{i+k} for some i in the sum
- Add the reduced amount to a position that has low contribution to all top peaks

The "sensitivity" of position p is approximately: sum over top peaks k of a_{p-k} + a_{p+k} (the derivative of the peak w.r.t. a_p).

            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac_y = fast_autocorr(y)
        M_y = ac_y.max()
        if M_y < 1e-15:
            break

        T = build_conv_matrix(y)
        c_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(c_var)), [T @ c_var <= M_y])

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            del T
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            del T
            continue

        c_val = np.maximum(np.asarray(c_var.value, dtype=np.float64), 0.0)
        del T
        c_sum = c_val.sum()
        if c_sum < 1e-15:
            s_dir = y.copy()
        else:
            s_dir = c_val / c_sum

        gamma = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - gamma) * y + gamma * s_dir
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            x = s_dir.copy()
        else:
            x = x / sx

        m_x = fast_autocorr(x).max()
        if m_x < best_m:
            best_m = m_x
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  FW t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.3e}")

    return best_a, best_m


def gradient_fw_fallback(a0, time_limit, verbose=True):
    """Gradient-based FW using subgradient of max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    c0 = 3.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        if t == 1:
            y = x.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()
        top_k = np.argsort(ac)[::-1][:20]
        top_vals = ac[top_k]
        w = (top_vals - M + 1.0) ** 8
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_k)) / len(top_k)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_k):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            grad[idx] += 2.0 * wi * y[partner]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y) - eta * grad
        log_y -= log_y.max()
        a_new = np.exp(log_y)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        s_dir = a_new
        gamma = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - gamma) * y + gamma * s_dir
        x = np.maximum(x, 0.0)
        x = x / x.sum()

        m_x = fast_autocorr(x).max()
        if m_x < best_m:
            best_m = m_x
            best_a = x.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  GradFW t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def local_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 300
    npeaks = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:npeaks]

        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:6]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[s:e + 1] = True
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k[:5]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-10:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(sens[candidates])]
            if dst == src:
                continue

            found = False
            for frac in [0.004, 0.008, 0.002, 0.01, 0.006]:
                amt = a[src] * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                M_t = fast_autocorr(a_t).max()
                if M_t < best_m - 1e-15:
                    a = a_t
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 80:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-8:
                    amt = a[src] * rng.uniform(0.002, 0.01)
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                    else:
                        a = best_a.copy()

        if not improved:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.003, 0.015)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
            if not improved:
                a = best_a.copy()

        if verbose and stall % 60 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Polish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    a_start = norm_sum1(best_seq)

    # === Phase 1: Main Nesterov FW on parent ===
    print(f"\n=== Phase 1: Nesterov FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_p1 = min(remaining * 0.55, 950.0)
    if t_p1 > 60:
        a_out, _ = nesterov_fw(a_start, t_p1, solver, verbose=True)
        try_update(a_out, "fw_main")

    # === Phase 2: Local polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Local polish (n={len(best_seq)}) ===")
    t_p2 = min(remaining * 0.12, 200.0)
    if t_p2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_polish(a_in, t_p2, verbose=True)
        try_update(a_out, "polish_1")

    # === Phase 3: Second FW pass on polished ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FW round 2 (n={n_best}) ===")
    t_p3 = min(remaining * 0.20, 350.0)
    if t_p3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_fw(a_in, t_p3, solver, verbose=True)
        try_update(a_out, "fw_r2")

    # === Phase 4: Polish again ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Polish round 2 ===")
    t_p4 = min(remaining * 0.08, 120.0)
    if t_p4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_polish(a_in, t_p4, verbose=True)
        try_update(a_out, "polish_2")

    # === Phase 5: Explore nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")
    trial_ns = []
    for delta in [300, -300, 600, -600]:
        nn = n_best + delta
        if 2000 <= nn <= 6000:
            trial_ns.append(nn)
    trial_ns = list(dict.fromkeys(trial_ns))

    for nn in trial_ns[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={nn}...")
        a_long = interpolate_sequence(best_seq, nn)
        t_exp = min(remaining * 0.35, 250.0)
        if t_exp > 30:
            a_out, _ = nesterov_fw(a_long, t_exp, solver, verbose=False)
            try_update(a_out, f"fw_n{nn}")
            if np.isfinite(evaluate_safe(a_out)):
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    a_out2, _ = local_polish(a_out, min(remaining * 0.3, 80.0), verbose=False)
                    try_update(a_out2, f"polish_n{nn}")

    # === Phase 6: Final gradient-based refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Gradient FW (n={n_best}) ===")
    t_p6 = min(remaining * 0.3, 200.0)
    if t_p6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_fw_fallback(a_in, t_p6, verbose=True)
        try_update(a_out, "gradfw")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish ===")
    t_p7 = min(remaining * 0.4, 100.0)
    if t_p7 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_polish(a_in, t_p7, verbose=True)
        try_update(a_out, "polish_final")

    # === Final ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
