So the inner problem (min over a) is convex only if Q_p is PSD for all p. Is Q_p PSD?

Q_p = sum_k p[k] T_k where T_k is the shift matrix. T_k has 1s on the k-th subdiagonal and superdiagonal (for the symmetric version). The sum of such matrices with non-negative weights... 

For k=0: T_0 = I (identity), which is PSD.
For k>0: T_k has 1s on positions (i, i+k) and (i+k, i). This matrix is NOT PSD in general (it has negative eigenvalues).

For example, for n=2, k=1: T_1 = [[0, 1], [1, 0]] which has eigenvalues 1 and -1. Not PSD.

def fast_conv(a):
    """Fast full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_topk_subgrad(y, ac, top_k):
    """Compute subgradient using top-k lags. O(n*k)."""
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=25, c0=4.0, verbose=False):

def lp_remez_step(a, active_set, r, solver):
    """Single LP Remez step with given active set and trust region radius."""
    n = len(a)
    b = fast_conv(a)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_set:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None, b.max()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, b.max()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None, b.max()
    
    a_new = a_new / s
    m_new = fast_conv(a_new).max()
    return a_new, m_new

    """LP Remez with adaptive trust region and growing active set."""
    n = len(a0)

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    active_size = min(10, max_active)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Build active set: lags where b[k] is close to max
        threshold = max_b * (0.999 - 0.001 * min(t, 100) / 100)  # Tighten over time
        active = np.where(b >= threshold)[0]
        if len(active) > active_size:
            active = np.argsort(b)[::-1][:active_size]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        a_cand, m_cand = lp_remez_step(a, active, r, solver)
        
        if a_cand is not None:
            # Line search
            accepted = False
            for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                a_trial = (1 - frac) * a + frac * a_cand
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    accepted = True
                    no_improve = 0
                    r = min(0.2, r * 1.05)
                    active_size = min(max_active, active_size + 1)
                    break
            
            if not accepted:
                no_improve += 1
                r = max(0.002, r * 0.6)
                if no_improve > 30:
                    # Escape: perturb and reset
                    rng = np.random.default_rng(t)
                    idx = rng.choice(n, size=min(n//4, 500), replace=False)
                    a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    no_improve = 0
                    r = 0.05
        else:
            no_improve += 1
            r = max(0.002, r * 0.7)
            if no_improve > 30:
                rng = np.random.default_rng(t + 1000)
                idx = rng.choice(n, size=min(n//4, 500), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = 0.05
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.5f}, active={len(active)}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Aggressive coordinate descent with random pair moves and 1D optimization."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(50, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            
            # Try several transfer fractions
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.25]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-14:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Periodic perturbation
            if stall % 10 == 0:
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.85, 1.15, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='raised_cos'):
    """Create initial sequences of various shapes."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sine':
        a = np.sin(np.pi * x)
    else:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def smoothness_filter(a, sigma=2.0):
    """Apply a gentle smoothing to the sequence."""
    n = len(a)
    if n < 5:
        return a.copy()
    kernel_size = max(5, int(2 * sigma * 3) | 1)  # odd
    if kernel_size >= n:
        kernel_size = n if n % 2 == 1 else n - 1
    kernel = np.exp(-0.5 * (np.arange(-kernel_size//2, kernel_size//2 + 1) / sigma) ** 2)
    kernel = kernel / kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s

def compute_topk_subgrad(y, ac, top_k):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sine':
        a = np.sin(np.pi * x)
    else:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-scale NMD+Remez with warm-started LP,
    aggressive Remez polish, CD, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 25.0, 1775.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        best_seq = make_init(2000, 'raised_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Multi-scale coarse-to-fine ===
    scales = [200, 500, 1000, 2000]
    if n_parent not in scales:
        scales.append(n_parent)
    if n_parent > 2000:
        scales.append(n_parent)
    scales = sorted(set(s for s in scales if s >= 200 and s <= 4000))

    elapsed_init = time.monotonic() - t_global
    print(f"\n=== Phase 1: Multi-scale (scales={scales}, elapsed={elapsed_init:.1f}s) ===")

    for scale_n in scales:
        rem = time_budget - (time.monotonic() - t_global)
        if rem < 50:
            break
        # Interpolate best to this scale
        a_scale = interpolate_sequence(best_seq, scale_n)
        print(f"  Scale n={scale_n}")

        # NMD
        t_nmd = min(rem * 0.4, 150.0)
        if t_nmd > 15:
            a_out, _ = nesterov_mirror_descent(a_scale, t_nmd, top_k=20, c0=3.0)
            _try_update(a_out, f"NMD_{scale_n}")

        # Remez
        rem = time_budget - (time.monotonic() - t_global)
        t_rz = min(rem * 0.4, 120.0)
        if t_rz > 10:
            a_out, _ = remez_method(norm_sum1(best_seq), t_rz, solver=solver,
                                    r_init=0.06, max_active=40)
            _try_update(a_out, f"Remez_{scale_n}")

    # === Phase 2: Focused NMD on best ===
    rem = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(rem * 0.25, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Focused NMD n={n_best} (budget={t2:.0f}s) ===")
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t2, top_k=30, c0=5.0, verbose=True)
        _try_update(a_out, f"NMD_focus_{n_best}")

    # === Phase 3: Aggressive Remez (most important) ===
    rem = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(rem * 0.30, 350.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Aggressive Remez n={n_best} (budget={t3:.0f}s) ===")
        a_out, _ = remez_method(norm_sum1(best_seq), t3, solver=solver,
                                r_init=0.04, max_active=70, verbose=True)
        _try_update(a_out, f"Remez_agg_{n_best}")

    # === Phase 4: CD polish ===
    rem = time_budget - (time.monotonic() - t_global)
    t4 = min(rem * 0.10, 80.0)
    if t4 > 10:
        print(f"\n=== Phase 4: CD polish n={len(best_seq)} (budget={t4:.0f}s) ===")
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t4, rng_seed=42)
        _try_update(a_out, f"CD_{len(best_seq)}")

    # === Phase 5: Alternative lengths ===
    rem = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if rem > 120:
        print(f"\n=== Phase 5: Length exploration ===")
        n_alts = sorted(set([n_best - 200, n_best + 200, n_best + 400, n_best + 600]))
        n_alts = [n for n in n_alts if 800 <= n <= 4000 and n != n_best]
        t_per = min(rem * 0.2, 80.0)
        for n_alt in n_alts:
            rem = time_budget - (time.monotonic() - t_global)
            if rem < 30:
                break
            a_alt = interpolate_sequence(best_seq, n_alt)
            t_use = min(t_per, rem - 10)
            if t_use > 25:
                a_out, _ = nesterov_mirror_descent(a_alt, t_use * 0.45, top_k=20, c0=3.0)
                _try_update(a_out, f"NMD_{n_alt}")
                rem = time_budget - (time.monotonic() - t_global)
                if rem > 15:
                    a_out, _ = remez_method(norm_sum1(best_seq), t_use * 0.4, solver=solver,
                                            r_init=0.04, max_active=40)
                    _try_update(a_out, f"Remez_{n_alt}")

    # === Phase 6: Final Remez squeeze ===
    rem = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    if rem > 25:
        print(f"\n=== Phase 6: Final squeeze n={n_final} (budget={rem-5:.0f}s) ===")
        t6 = rem - 5
        a_out, _ = remez_method(norm_sum1(best_seq), t6, solver=solver,
                                r_init=0.015, max_active=100, verbose=True)
        _try_update(a_out, f"Remez_final_{n_final}")
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 12:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), rem - 8, rng_seed=999)
            _try_update(a_out, f"CD_final_{n_final}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
