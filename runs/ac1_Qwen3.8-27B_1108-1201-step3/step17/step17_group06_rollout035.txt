Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed projected gradient descent with adaptive log-sum-exp temperature
2. Euclidean simplex projection (O(n log n))
3. Pair-coordinate descent for local refinement

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 assuming a is normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def simplex_project(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    ar = np.arange(1, n + 1, dtype=np.float64)
    rho_candidates = np.where(u * ar > (cv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_smooth_grad(a, T, top_k=None):
    """Compute smoothed gradient of max(conv(a,a)) using softmax weights.
    Returns (gradient, conv)."""
    n = len(a)
    conv = fast_conv(a)

    if top_k is not None and top_k < len(conv):
        idx = np.argsort(conv)[::-1][:top_k]
        conv_m = np.zeros_like(conv)
        conv_m[idx] = conv[idx]
    else:
        conv_m = conv

    # Compute softmax weights
    cmax = conv_m.max()
    exp_vals = np.exp((conv_m - cmax) / T)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv_m)
        w[conv_m == cmax] = 1.0 / np.sum(conv_m == cmax)
    else:
        w = exp_vals / w_sum

    # grad[i] = sum_j a[j] * w[i+j] for valid indices
    # Using FFT: conv(a, flip(w))[2n-2-i]
    n_w = len(w)  # 2n-1
    conv_len = n + n_w - 1  # 3n-2
    N = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(np.flip(w), N)
    cz = np.fft.irfft(A * W, N)[:conv_len]
    grad = cz[2 * n - 2 - np.arange(n)]
    return grad, conv


def compute_hard_grad(a, top_k=50):

    n = len(a)
    conv = fast_conv(a)
    idx = np.argsort(conv)[::-1][:top_k]

    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / top_k

    n_w = len(w)
    conv_len = n + n_w - 1
    N = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(np.flip(w), N)
    cz = np.fft.irfft(A * W, N)[:conv_len]
    grad = cz[2 * n - 2 - np.arange(n)]
    return grad, conv


def smoothed_pgdescent(a0, time_limit, T_init=0.005, T_min=0.0005,
                       eta_init=0.005, top_k_mode=False, top_k=80,
                       restart_every=3000, verbose=False, seed=42):
    """Smoothed projected gradient descent with adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient
        if top_k_mode:
            grad, conv = compute_hard_grad(a, top_k=top_k)
        else:
            grad, conv = compute_smooth_grad(a, T=T)

        # Adaptive step size with backtracking
        eta_t = eta
        improved = False
        for backtrack in range(8):
            a_new = simplex_project(a - eta_t * grad, 1.0)
            val = eval_J(a_new)
            if val < best_val - 1e-16:
                best_val = val
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
                improved = True
                break
            eta_t *= 0.5

        if not improved:
            stall += 1
            # Occasionally try a random direction if stuck
            if stall > 50 and stall % 25 == 0:
                for _ in range(5):
                    d = rng.normal(0, 1, n)
                    d = simplex_project(a + 0.01 * d, 1.0) - a
                    a_try = simplex_project(a + 0.005 * d, 1.0)
                    v_try = eval_J(a_try)
                    if v_try < best_val - 1e-16:
                        a = a_try
                        best_a = a.copy()
                        best_val = v_try
                        stall = 0
                        break

        # Temperature annealing
        if t % 200 == 0:
            T = max(T * 0.97, T_min)

        # Restart logic
        if stall > restart_every:
            stall = 0
            # Structured perturbation: modify the shape envelope
            pert_type = rng.integers(4)
            if pert_type == 0:
                # Gaussian noise scaled by current value
                noise = rng.normal(0, 0.03, n) * best_a
                a = simplex_project(best_a + noise, 1.0)
            elif pert_type == 1:
                # Shift the "center of mass"
                shift = rng.uniform(0.05, 0.15)
                x = np.linspace(0, 1, n)
                envelope = np.exp(-50 * (x - 0.5) ** 2)
                a_pert = best_a * (1 + shift * (envelope / max(envelope.max(), 1e-10) - 1))
                a = simplex_project(a_pert, 1.0)
            elif pert_type == 2:
                # Frequency-domain perturbation
                N = 1 << (n - 1).bit_length()
                A = np.fft.rfft(best_a, N)
                mag = np.abs(A)
                ph = np.angle(A)
                ph += rng.normal(0, 0.1, len(ph))
                A_new = mag * np.exp(1j * ph)
                a_new = np.fft.irfft(A_new, N)[:n]
                a = simplex_project(a_new, 1.0)
            else:
                # Power-law reshaping
                p_new = rng.uniform(0.5, 2.5)
                x = np.linspace(0, 1, n)
                power_shape = (x + 0.01) ** p_new
                a_pert = best_a * (power_shape / power_shape.mean())
                a = simplex_project(a_pert, 1.0)

            T = min(T * 1.5, T_init * 0.5)
            eta = eta_init * 0.9

        if verbose and t % 5000 == 0:
            print(f"    SPDGD t={t}: J={best_val:.6f}, T={T:.5f}, eta={eta:.5f}, stall={stall}")

    return best_a, best_val


def pair_coord_descent(a0, time_limit, verbose=False, seed=42):
    """Pair coordinate descent: try moving mass between pairs of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Find top lags and identify "hot" positions
        top_lags = np.argsort(conv)[::-1][:20]

        # Compute pressure: how much each position contributes to peaks
        pressure = np.zeros(n)
        for k in top_lags[:10]:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            for i in range(i_start, i_end + 1):
                j = ki - i
                pressure[i] += a[i] * a[j]

        # Source: position with high pressure AND significant mass
        src_scores = pressure * a
        top_srcs = np.argsort(src_scores)[::-1][:20]

        # Try moving mass from high-pressure sources to low-pressure destinations
        improved = False
        for src_idx in range(min(5, len(top_srcs))):
            src = int(top_srcs[src_idx])
            if a[src] < 1e-12:
                continue

            # Destination: low pressure, not adjacent to source
            dest_scores = pressure.copy()
            dest_scores[src] = np.inf
            # Also penalize positions near source (they'll increase nearby conv)
            for dk in range(3):
                if src - dk >= 0:
                    dest_scores[src - dk] += 1e-6
                if src + dk < n:
                    dest_scores[src + dk] += 1e-6

            top_dsts = np.argsort(dest_scores)[:15]

            for dst_idx in range(min(5, len(top_dsts))):
                dst = int(top_dsts[dst_idx])
                if dst == src:
                    continue
                if abs(dst - src) < 2:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    a_t /= a_t.sum()
                    J_t = eval_J(a_t)
                    if J_t < best_J - 1e-16:
                        a = a_t
                        best_J = J_t
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                # Random perturbation to escape
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.01, 0.1)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 50 == 0:
            print(f"    PCD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_gaussian_envelope(n, center=0.5, width=0.3):
    """Gaussian envelope."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_exponential_decay(n, rate=2.0):
    """Exponential decay from center."""
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * (x - 0.5) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_bimodal(n, c1=0.25, c2=0.75, width=0.08):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) / width) ** 2) + np.exp(-((x - c2) / width) ** 2) + 0.01
    return a / a.sum()


def make_v_shape(n, center=0.5, sharpness=3.0):
    """V-shaped (minimum at center)."""
    x = np.linspace(0, 1, n)
    a = np.abs(x - center) ** sharpness + 0.01
    return a / a.sum()


def make_arcosh(n, p=1.0):
    """Arcosh-like shape (high at edges, low at center)."""
    x = np.linspace(0.01, 0.99, n)
    a = np.cosh(p * (x - 0.5) * 8) ** 2
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3200
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"Starting optimization with n={n0}, J={best_score:.6f}")

    # === Phase 1: Smoothed PGD on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smoothed PGD on parent (n={n0}) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = smoothed_pgdescent(best_a.copy(), t1, T_init=0.005,
                                       T_min=0.0003, eta_init=0.003,
                                       top_k_mode=False, restart_every=2000,
                                       verbose=True, seed=1)
        try_update(a_out, "spgd_parent")

    # === Phase 2: Hard subgradient PGD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Hard-subgrad PGD (n={len(best_a)}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = smoothed_pgdescent(best_a.copy(), t2, T_init=0.003,
                                       T_min=0.0002, eta_init=0.003,
                                       top_k_mode=True, top_k=60,
                                       restart_every=2500,
                                       verbose=True, seed=2)
        try_update(a_out, "hardpgd")

    # === Phase 3: Pair coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Pair CD (n={len(best_a)}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_out, _ = pair_coord_descent(best_a.copy(), t3, verbose=True, seed=100)
        try_update(a_out, "pair_cd1")

    # === Phase 4: More PGD cycles with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: PGD with different params ===")
    configs = [
        (0.008, 0.0005, 0.004, False, 100, 1500, 3),
        (0.003, 0.0002, 0.002, True, 40, 2000, 4),
        (0.010, 0.001, 0.006, False, 0, 3000, 5),
    ]
    for ci, (T_i, T_m, e_i, tk_m, tk, re, sd) in enumerate(configs):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t_avail = min(remaining * 0.12, 180.0)
        if t_avail < 20:
            break
        print(f"  Config {ci}: T_init={T_i}, top_k_mode={tk_m}")
        a_out, _ = smoothed_pgdescent(best_a.copy(), t_avail,
                                       T_init=T_i, T_min=T_m,
                                       eta_init=e_i, top_k_mode=tk_m,
                                       top_k=tk, restart_every=re,
                                       verbose=False, seed=sd)
        try_update(a_out, f"pgd_cfg{ci}")

    # === Phase 5: Multi-init exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-Init (n={n0}) ===")

    n_current = len(best_a)
    inits_configs = []

    if parent is not None:
        # Blend parent with different shapes
        for p_blend, shape_fn, shape_kwargs in [
            (0.7, make_power_ramp, {'p': 0.5}),
            (0.7, make_power_ramp, {'p': 1.5}),
            (0.6, make_gaussian_envelope, {'center': 0.5, 'width': 0.4}),
            (0.6, make_bimodal, {'c1': 0.3, 'c2': 0.7, 'width': 0.1}),
            (0.5, make_v_shape, {'center': 0.5, 'sharpness': 2.0}),
        ]:
            shape = shape_fn(n0, **shape_kwargs)
            b = p_blend * parent + (1 - p_blend) * shape
            b = b / b.sum()
            inits_configs.append((b, f"blend_{shape_fn.__name__}"))

    inits_configs.append((make_power_ramp(n0, 0.3), "ramp0.3"))
    inits_configs.append((make_power_ramp(n0, 0.7), "ramp0.7"))
    inits_configs.append((make_gaussian_envelope(n0, 0.5, 0.4), "gauss"))
    inits_configs.append((make_bimodal(n0, 0.2, 0.8, 0.12), "bimodal_wide"))
    inits_configs.append((make_v_shape(n0, 0.5, 2.0), "vshape"))
    inits_configs.append((make_smooth_random(n0, seed=7), "smooth7"))
    inits_configs.append((make_arcosh(n0, 1.5), "arcosh"))

    # Deduplicate
    seen_sigs = set()
    unique_inits = []
    for init, lbl in inits_configs:
        sig = tuple(np.round(init[::max(1, n0 // 50)], 4))
        if sig not in seen_sigs:
            seen_sigs.add(sig)
            unique_inits.append((init, lbl))

    t_per = max(15.0, remaining * 0.25 / max(len(unique_inits), 1))
    for idx, (init, lbl) in enumerate(unique_inits):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t_avail = min(t_per, remaining - 30)
        if t_avail < 15:
            break
        J_init = eval_J(init)
        print(f"  Init [{lbl}]: J={J_init:.6f}, t={t_avail:.0f}s")
        a_out, _ = smoothed_pgdescent(init, t_avail, T_init=0.005,
                                       T_min=0.0003, eta_init=0.003,
                                       top_k_mode=False, restart_every=2000,
                                       verbose=False, seed=10 + idx)
        try_update(a_out, f"init_{lbl}")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_a)
    lengths = []
    for d in [-300, -150, 150, 300, 500, -500]:
        nl = n_best + d
        if 1500 <= nl <= 5000:
            lengths.append(nl)
    for nl in [2000, 2500, 3000, 3500, 4000]:
        if nl not in lengths and 1500 <= nl <= 5000:
            lengths.append(nl)

    for nl in sorted(set(lengths)):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        if abs(nl - n_best) <= 30:
            continue
        print(f"  Length n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 120.0)
        if t_md > 15:
            a_out, _ = smoothed_pgdescent(a_int, t_md, T_init=0.005,
                                           T_min=0.0003, eta_init=0.003,
                                           top_k_mode=False, restart_every=2000,
                                           verbose=False, seed=nl % 1000)
            try_update(a_out, f"len{nl}")

    # === Phase 7: Final refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_a)
    print(f"\n=== Phase 7: Final Refinement (n={n_final}) ===")

    # PGD with very fine temperature
    t7a = min(remaining * 0.30, 250.0)
    if t7a > 30:
        a_out, _ = smoothed_pgdescent(best_a.copy(), t7a, T_init=0.002,
                                       T_min=0.0001, eta_init=0.002,
                                       top_k_mode=False, restart_every=5000,
                                       verbose=True, seed=999)
        try_update(a_out, "final_fine")

    # Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    t7b = min(remaining * 0.25, 150.0)
    if t7b > 20:
        a_out, _ = pair_coord_descent(best_a.copy(), t7b, verbose=True, seed=555)
        try_update(a_out, "final_pair_cd")

    # Another PGD
    remaining = time_budget - (time.monotonic() - t_global)
    t7c = min(remaining * 0.30, 200.0)
    if t7c > 20:
        a_out, _ = smoothed_pgdescent(best_a.copy(), t7c, T_init=0.001,
                                       T_min=0.00005, eta_init=0.001,
                                       top_k_mode=True, top_k=30,
                                       restart_every=8000,
                                       verbose=False, seed=777)
        try_update(a_out, "final_hard")

    # === Phase 8: Last moment polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 8: Last Polish ===")
        t8 = remaining - 10
        a_out, _ = pair_coord_descent(best_a.copy(), t8, verbose=False, seed=313)
        try_update(a_out, "last_polish")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3200
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
