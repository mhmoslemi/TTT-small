def spectral_equalize(a, n_iter=20, eps=1e-10):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a

def lp_fw_step(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized direction b/sum(b) or None.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def lp_fw_full_step(y, M, active_lags):

        j_end = min(n - 1, k_int)

    b_val = np.maximum(res.x, 0.0)
    b_sum = b_val.sum()

    return b_val / b_sum

        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        # Select active constraints
        if use_full_constraints and n <= 2000:
            active_lags = np.arange(2 * n - 1)
        else:
            active_lags = np.argsort(ac)[::-1][:min(top_k, 2 * n - 1)]
        
        # Solve LP for direction
        if use_full_constraints and n <= 3000:
            direction = lp_fw_full_step(y, M_curr, active_lags)
        else:
            direction = lp_fw_step(y, M_curr, active_lags, solver)
        
        if direction is None:
            continue
        
        # Backtracking line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = alpha
        best_val = M_curr
        
        for _ in range(10):
            x_trial = (1.0 - alpha) * y + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            sx = x_trial.sum()
            if sx < 1e-15:
                alpha *= 0.5
                continue
            x_trial = x_trial / sx
            val = fast_autocorr(x_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
        
        x_new = (1.0 - best_alpha) * y + best_alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
        
        if t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.8f}, M={best_m:.8e}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, rng_seed=42):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find good destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(100, len(candidates))]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d = k2i - dst
                    if 0 <= d < n:
                        sens += a[d]
                    d2 = dst - k2i
                    if 0 <= d2 < n and d2 != d:
                        sens += a[d2]
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if found:
                break
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
        else:
            stall = 0
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
