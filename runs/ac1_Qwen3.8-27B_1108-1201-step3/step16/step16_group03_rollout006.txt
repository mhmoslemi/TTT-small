"""

1. Annealed log-sum-exp gradient descent in log-space (natural gradient)
2. Basin hopping with diverse perturbations for global exploration
3. Coordinate descent with line search for local polish
4. Multi-length exploitation

import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lse_max(conv_vals, temp):
    """Compute log-sum-exp approximation to max."""
    m = conv_vals.max()
    return m + np.log(np.sum(np.exp((conv_vals - m) * temp))) / temp

def lse_weights(conv_vals, temp):
    """Compute softmax weights for the log-sum-exp."""
    shifted = (conv_vals - conv_vals.max()) * temp
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()

def grad_conv(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    Result: grad[j] = 2 * sum_k weights[k] * a[k-j] (with valid bounds).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1
    # We need: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m] for j=0..n-1
    # This is cross-correlation of a and weights at lags 0..n-1.
    # Using FFT: cross[lag] = sum_m a[m] * w[m+lag]
    # FFT method: IFFT(conj(FFT(a_padded)) * FFT(w_padded))[lag]
    fft_size = 1 << (n + size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    cross = np.fft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * cross[:n]

def natural_grad(grad_a, a):
    """Convert gradient in a-space to natural gradient (log-space)."""
    # In log-space: dx_i = a_i * (grad_a_i - sum_j grad_a_j * a_j)
    return a * (grad_a - np.dot(grad_a, a))

def annealed_gradient_descent(a0, time_limit, t_start=1.0, t_end=200.0, 
                                beta=0.3, lr0=0.5, verbose=False, seed=42):
    """
    Annealed gradient descent on log-sum-exp smoothed objective.
    Optimizes in log-space with natural gradient.
    """
    n = len(a0)
    x = np.log(np.maximum(a0, 1e-15))  # log-space parameterization
    best_a = a0.copy()
    best_M = fast_conv(a0).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    n_steps = 0
    stall = 0
    max_stall = 500
    
    for step in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current a from x
        log_a = x - x.max()
        a = np.exp(log_a)
        a = a / a.sum()
        
        # Compute conv and smooth max
        ac = fast_conv(a)
        M_curr = ac.max()
        
        # Temperature scheduling: anneal from t_start to t_end
        progress = min(1.0, step / 5000.0)
        temp = t_start + (t_end - t_start) * progress
        
        # Compute weights and gradient
        w = lse_weights(ac, temp)
        g_a = grad_conv(a, w)
        g_x = natural_grad(g_a, a)
        
        # Line search with backtracking
        lr = lr0
        best_trial_M = M_curr
        best_trial_x = x.copy()
        
        a_curr = a.copy()
        for _ in range(20):
            x_trial = x - lr * g_x
            # Convert back to a
            log_at = x_trial - x_trial.max()
            a_trial = np.exp(log_at)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial_x = x_trial.copy()
            if M_trial < M_curr * (1.0 - beta * lr * np.abs(g_x).max() * 0.5):
                break
            lr *= 0.5
            if lr < 1e-10:
                break
        
        x = best_trial_x
        n_steps += 1
        
        # Track best
        # Recompute a from updated x
        log_a_new = x - x.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        M_new = fast_conv(a_new).max()
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stuck
        if stall > max_stall:
            x = np.log(np.maximum(best_a, 1e-15))
            x += rng.normal(0, 0.01, n)
            stall = 0
        
        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      AGD step={step} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.1f}")
    
    return best_a, best_M

def coordinate_descent(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with 2-point moves and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        # Find top peaks
        top_k = np.argsort(ac)[::-1][:5]
        improved = False
        
        for k_idx in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k_idx)
            # Find the pair (i, j) with i+j=k that contributes most
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k - idx]
            top_pairs = idx[np.argsort(contrib)[::-1][:3]]
            
            for src in top_pairs:
                if a[src] < 1e-15:
                    continue
                # Find best destination
                # Try moving mass from src to a random set of positions
                n_try = min(100, n)
                candidates = rng.integers(0, n, size=n_try)
                
                best_score = best_M
                best_move = None
                for dst in candidates:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15 or a_trial[dst] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_score - 1e-15:
                            best_score = M_trial
                            best_move = (src, dst, frac)
                
                if best_move is not None:
                    src, dst, frac = best_move
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_M = fast_conv(a).max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
            if not improved:
                stall += 1
        else:
            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_M

def diverse_perturbation(a, rng, strength, n_peaks=10):
    """Create a diverse perturbation of a."""
    n = len(a)
    ac = fast_conv(a)
    # Perturb away from peaks
    peak_mask = np.zeros(n, dtype=bool)
    for k in np.argsort(ac)[::-1][:n_peaks]:
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        peak_mask[s:e] = True
    
    # Create perturbation that's larger away from peaks
    noise = rng.normal(0, 1, n)
    noise[~peak_mask] *= 2.0  # Emphasize non-peak positions
    perturbed = a + strength * noise
    perturbed = np.maximum(perturbed, 0.0)
    ps = perturbed.sum()
    if ps < 1e-15:
        return a.copy()
    return perturbed / ps

I want: grad[j] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-j] (where a[m] = 0 for m < 0 or m >= n)

Let me define: for m = 0, ..., n-1, a[m] is defined. For other m, a[m] = 0.
For k = 0, ..., 2n-2, w[k] is defined.

grad[j] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-j]

Let m = k-j, so k = m+j. As k goes from 0 to 2n-2, m goes from -j to 2n-2-j.
But a[m] is non-zero only for m = 0, ..., n-1.
So: grad[j] = 2 * sum_{m=max(0,-j)}^{min(n-1, 2n-2-j)} w[m+j] * a[m]
= 2 * sum_{m=0}^{n-1} w[m+j] * a[m] (since j >= 0, -j <= 0, and 2n-2-j >= n-1 for j <= n-1)

Wait, 2n-2-j >= n-1 iff n-1 >= j, which is true since j < n.
And max(0, -j) = 0 for j >= 0.

So: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]

This is the cross-correlation of the sequence a (length n) and w (length 2n-1) at lag j.

Now, the standard definition of cross-correlation:
(a ⋆ w)[lag] = sum_m a[m] * w[m + lag]

(This is the "non-conjugate" cross-correlation, appropriate for real signals.)

In the frequency domain:
FFT(a ⋆ w) = conj(FFT(a)) * FFT(w)

But this is for circular cross-correlation. For linear cross-correlation, we need to zero-pad.

If a has length n and w has length 2n-1, the linear cross-correlation has length n + (2n-1) - 1 = 3n-2.
We need lags 0 to n-1, which is within this range.

For FFT computation:
- Pad a to length N >= 3n-2
- Pad w to length N
- Compute: cross = IFFT(conj(FFT(a_padded)) * FFT(w_padded))
- The first N elements give the circular cross-correlation, which equals the linear cross-correlation for lags 0 to N-1 (with the second half being the negative lags).

Wait, actually for real signals, the circular cross-correlation of length N is:
cross[k] = sum_{m=0}^{N-1} a_padded[m] * w_padded[(m+k) mod N]

For the linear cross-correlation (without wraparound):
lin_cross[k] = sum_{m=0}^{n-1} a[m] * w[m+k] (valid when 0 <= m+k < 2n-1)

If N >= n + (2n-1) - 1 = 3n-2, then for k = 0, ..., n-1:
lin_cross[k] = sum_{m=0}^{n-1} a[m] * w[m+k]
and since m+k < n + n = 2n <= 3n-2 <= N for k < n, there's no wraparound.

So: lin_cross[k] = circ_cross[k] for k = 0, ..., n-1.

Therefore: grad[j] = 2 * circ_cross[j] = 2 * IFFT(conj(FFT(a_padded)) * FFT(w_padded))[j]
