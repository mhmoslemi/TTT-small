import numpy as np
import time


def fast_conv(a):
    """Full linear autocorrelation conv(a, a) via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (max(size, 2) - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize1(v):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def simplex_project(v, target=1.0):
    """Euclidean projection of v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target)
    cand = np.where(cond)[0]
    if cand.size == 0:
        return np.full(n, target / n)
    rho = cand[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_weights(C, beta):
    size = len(C)
    Cs = C - C.max()
    e = np.exp(beta * Cs)
    s = e.sum()
    if s < 1e-300:
        return np.ones(size) / size
    return e / s


def smooth_grad_fft(a, w, beta):
    """Gradient of f_beta via FFT cross-correlation. w has length 2n-1."""
    n = len(a)
    fft_size = 1 << (3 * n - 3).bit_length()
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(W * A, fft_size)
    g = c[2 * n - 2 : n - 2 : -1].copy()
    return (2.0 / beta) * g


def smooth_grad_direct(a, w, beta):
    """O(n^2) reference implementation of the same gradient."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        aj = a[j]
        if aj != 0.0:
            g += aj * w[j:j + n]
    return (2.0 / beta) * g


def logsumexp_beta_C(C, beta):
    return (1.0 / beta) * np.log(np.exp(beta * (C - C.max())).sum() + 1e-300) + C.max()


def nesterov_smooth(a0, time_limit, beta_start=2.0, beta_end=40.0,
                    use_fft=True, verbose=False, rng_seed=0):
    """Nesterov momentum grad descent on the softmax smooth-max surrogate."""
    a = normalize1(a0)
    n = len(a)
    C = fast_conv(a)
    best_a = a.copy()
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    a_prev = a.copy()
    stall = 0
    max_stall = 600
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / 2500.0)
        beta = beta_start + (beta_end - beta_start) * (frac * frac)

        mu = it / (it + 15.0)
        y = a + mu * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        Cy = fast_conv(y)
        w = _softmax_weights(Cy, beta)
        if use_fft:
            g = smooth_grad_fft(y, w, beta)
        else:
            g = smooth_grad_direct(y, w, beta)
        gn2 = float(np.dot(g, g))
        f_cur = logsumexp_beta_C(Cy, beta)

        # Armijo backtracking
        eta = 2.0 * beta
        ym = y
        for _ in range(60):
            trial = y - eta * g
            trial = simplex_project(trial, 1.0)
            Ctr = fast_conv(trial)
            f_new = logsumexp_beta_C(Ctr, beta)
            if f_new <= f_cur - 1e-4 * eta * gn2 - 1e-16:
                ym = trial
                break
            eta *= 0.5
            if eta < 1e-12:
                break

        a_prev = a.copy()
        a = ym

        Ca = fast_conv(a)
        m = Ca.max()
        if m < best_m - 1e-14:
            best_m = m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            ns = 0.004 + 0.01 * min(1.0, max_stall / 1200.0)
            noise = rng.normal(0.0, ns, n) * best_a
            ar = simplex_project(best_a + noise, 1.0)
            a = ar
            a_prev = ar.copy()
            max_stall = max(250, max_stall // 2)
            if verbose:

                      (it, time.monotonic() - t0, 2.0 * n * best_m, beta))

def coord_polish(a0, time_limit, use_fft=True, verbose=False, rng_seed=1):
    """Peak-targeted coordinate descent using gradient sensitivity field."""
    a = normalize1(a0)
    n = len(a)
    C = fast_conv(a)
    best_a = a.copy()
    best_m = C.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_conv(a)
        M = C.max()
        w = _softmax_weights(C, 20.0)
        if use_fft:
            sens = smooth_grad_fft(a, w, 20.0)   # ~ how C rises per unit at each pos
        else:
            sens = smooth_grad_direct(a, w, 20.0)

        k_peak = int(np.argmax(C))
        i0 = max(0, k_peak - n + 1)
        i1 = min(n, k_peak + 1)
        idx = np.arange(i0, i1)
        jdx = k_peak - idx
        prod = a[idx] * a[jdx]
        if prod.size == 0:
            break
        mp = int(np.argmax(prod))
        src_i = idx[mp]
        dst_j = jdx[mp]
        src = src_i if a[src_i] >= a[dst_j] else dst_j

        # cold destinations = low sensitivity
        order = np.argsort(sens)
        cand = [int(c) for c in order if abs(c - src) > n // 20]
        if len(cand) < 10:
            cand = [int(c) for c in order]
        cand = cand[:40]

        improved = False
        for dst in cand:
            if dst == src:
                continue
            for frac in (0.001, 0.003, 0.01, 0.03):
                amt = a[src] * frac
                if amt < 1e-16:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                mt = fast_conv(at).max()
                if mt < best_m - 1e-14:
                    a = at
                    best_m = mt
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # escape: a couple of random hot->random moves
            for _ in range(max(2, n // 300)):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.04)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s2 = a.sum()
                    if s2 > 1e-15:
                        a = a / s2
                        mt = fast_conv(a).max()
                        if mt < best_m:
                            best_m = mt
                            best_a = a.copy()
                            improved = True
            if not improved:
                if verbose and stall % 50 == 0:
                    print("    coord stall=%d J=%.6f" %
                          (stall, 2.0 * n * best_m))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    f = normalize1(f)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, f)
    return normalize1(fi)


def _selftest_gradient():
    """Cross-check FFT gradient against direct on a tiny vector."""
    rng = np.random.default_rng(123)
    n = 9
    a = normalize1(rng.random(n))
    C = fast_conv(a)
    beta = 5.0
    w = _softmax_weights(C, beta)
    g1 = smooth_grad_fft(a, w, beta)
    g2 = smooth_grad_direct(a, w, beta)
    ok = np.allclose(g1, g2, rtol=1e-7, atol=1e-10)
    print("  gradient self-test (fft vs direct): %s" % ("PASS" if ok else "FAIL"))
    return ok
