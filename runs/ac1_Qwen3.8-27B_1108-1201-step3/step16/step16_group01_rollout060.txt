import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation (convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_subgradient(a, ac, alpha=2.0, threshold=0.01):
    """Compute smooth subgradient with power-law weighting over all significant lags."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Weight by (ac[k]/M)^alpha for significant lags
    weights = np.zeros(len(ac))
    significant = ac > threshold * M
    weights[significant] = (ac[significant] / M) ** alpha
    wsum = weights.sum()
    if wsum < 1e-30:
        # Just use the argmax
        k_max = int(ac.argmax())
        weights[k_max] = 1.0
        wsum = 1.0
    else:
        weights /= wsum
    
    g = np.zeros(n)
    for k in np.where(significant)[0]:
        k_int = int(k)
        w = weights[k_int]
        if w < 1e-10:
            continue
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += w * a[j_idx]
    
    # The subgradient of max_k conv[k] is 2*(a * 1_{k*}) where k* is the argmax
    # For the smooth version, it's 2*sum_k w_k * a_reversed_at_k
    g *= 2.0
    return g

def amd_optimize(a0, time_budget, alpha=2.0, c0=3.0, verbose=True):
    """AMD with smooth subgradient."""
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_val = fast_autocorr(best).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    
    rng = np.random.default_rng(0)
    
    for it in range(1, 10**8):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Nesterov extrapolation
        beta = (it - 1.0) / (it + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = smooth_subgradient(y, ac, alpha=alpha)
        
        # Step size
        eta = c0 / (np.sqrt(it) + 1.0) * np.sqrt(n)
        
        # Mirror update (log-space)
        log_y = np.log(y) - eta * g
        log_y -= log_y.max()
        y_new = np.exp(log_y)
        y_new = np.maximum(y_new, 1e-30)
        sy_new = y_new.sum()
        if sy_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy_new
        
        x_prev = x
        x = y_new
        
        M = fast_autocorr(x).max()
        if M < best_val - 1e-15:
            best_val = M
            best = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Restart with perturbation
            noise = rng.normal(0, 0.01, n) * best
            x_re = np.maximum(best + noise, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            stall = 0
            alpha = alpha * 1.1  # Make subgradient more peaky
            if alpha > 10:
                alpha = 2.0
            c0 = c0 * 1.15
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_val
            print(f"  AMD it={it} ({elapsed:.1f}s): obj={obj:.6f}, alpha={alpha:.1f}")
    
    return best, 2.0 * n * best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Asymmetric (decreasing) initialization from "equal-lag" shape analysis
2. AMD with smooth power-law weighted subgradient and adaptive alpha
3. Peak-priority coordinate descent targeting top lags
4. Multi-length exploration (2000-8000)
5. Spectral rebalancing polish in frequency domain
6. Structured restarts using shape parameter sweeps
Key insight: asymmetric (front-loaded) sequences reduce center convolution peak.
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_obj(a):
    """Compute 2*n*max(conv)/(sum)^2 for a given sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_conv(a)
    return 2.0 * len(a) * ac.max() / (s * s)


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_subgradient(a, ac, alpha=3.0, threshold_frac=0.05):
    """
    Smooth subgradient with power-law weighting.
    weight_k = (ac[k]/M)^alpha for ac[k] > threshold_frac * M
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)

    sig = ac > threshold_frac * M
    n_sig = sig.sum()
    if n_sig == 0:
        k_max = int(ac.argmax())
        g = np.zeros(n)
        k_int = k_max
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] = 2.0 * a[j_idx]
        return g

    weights = np.zeros_like(ac)
    weights[sig] = (ac[sig] / M) ** alpha
    wsum = weights.sum()
    if wsum < 1e-30:
        k_max = int(ac.argmax())
        weights[k_max] = 1.0
        wsum = 1.0
    else:
        weights /= wsum

    g = np.zeros(n)
    sig_indices = np.where(sig)[0]
    for k in sig_indices:
        k_int = int(k)
        w = weights[k_int]
        if w < 1e-8:
            continue
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += w * a[j_idx]

    g *= 2.0
    return g


def exact_subgradient_topk(a, ac, top_k=5):
    """Subgradient from top-k lags only (for final polish)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    top_idx = np.argsort(ac)[::-1][:top_k]
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    g /= top_k
    return g


def mirror_update(y, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(y, 1e-18)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def amd_optimize(a0, time_budget, alpha_init=3.0, c0=3.0, verbose=True, seed=42):
    """
    AMD with smooth subgradient, adaptive alpha, and structured restarts.
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 4000
    alpha = alpha_init
    c_current = c0
    it_count = 0

    while time.monotonic() - t0 < time_budget:
        if stall > max_stall:
            stall = 0
            # Structured restart: try different shapes
            restart_type = int(rng.integers(3))
            if restart_type == 0:
                # Gaussian perturbation
                noise = rng.normal(0, 0.008, n) * best_a
                x = np.maximum(best_a + noise, 1e-10)
                x = x / x.sum()
            elif restart_type == 1:
                # Shape change: blend with a power-law
                p = rng.uniform(0.5, 3.0)
                shape = np.maximum(1.0 - np.arange(n) / n, 0.01) ** p
                shape = shape / shape.sum()
                blend = rng.uniform(0.2, 0.5)
                x = (1 - blend) * best_a + blend * shape
                x = x / x.sum()
            else:
                # Reverse and blend (try increasing vs decreasing)
                rev = best_a[::-1]
                blend = rng.uniform(0.3, 0.6)
                x = (1 - blend) * best_a + blend * rev
                x = x / x.sum()
            
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            alpha = alpha_init * (1.0 + 0.2 * (max_stall // 4000))
            alpha = min(alpha, 8.0)
            c_current = c_current * 1.1
            if verbose:
                print(f"    AMD restart ({time.monotonic()-t0:.1f}s): alpha={alpha:.1f}, c={c_current:.1f}, type={restart_type}")

        it_count += 1
        # Nesterov extrapolation
        beta_t = (it_count - 1.0) / (it_count + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_conv(y)
        g = smooth_subgradient(y, ac_y, alpha=alpha, threshold_frac=0.05)
        
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            a_new = x.copy()
        else:
            eta = c_current / (np.sqrt(it_count) + 1.0)
            a_new = mirror_update(y, g, eta)

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Slowly decrease alpha to include more lags
        if it_count % 5000 == 0:
            alpha = max(1.5, alpha * 0.97)

        if verbose and it_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"    AMD it={it_count} ({elapsed:.1f}s): obj={obj:.6f}, alpha={alpha:.2f}, stall={stall}")

    return best_a, 2.0 * n * best_m


def peak_coord_descent(a0, time_budget, verbose=True, seed=123):
    """
    Peak-priority coordinate descent: identify top peaks, find contributing
    positions, and transfer mass to reduce those peaks.
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_budget and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        # Get top peaks
        n_peaks = min(8, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_peaks]

        improved = False

        for k in top_peaks[:6]:
            if time.monotonic() - t0 > time_budget:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            if len(idx) == 0:
                continue

            contrib = a[idx] * a[j_idx]
            if contrib.max() < 1e-20:
                continue

            # Find the dominant pair
            max_local = np.argmax(contrib)
            src_i = idx[max_local]
            dst_j = j_idx[max_local]

            # Source is the larger element
            if a[src_i] >= a[dst_j]:
                src = src_i
            else:
                src = dst_j

            if a[src] < 1e-12:
                continue

            # Compute peak sensitivity for each position
            peak_sens = np.zeros(n)
            for k2 in top_peaks[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                j2 = k2i - idx2
                peak_sens[idx2] += a[j2]

            # Find low-sensitivity destinations
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) > 100:
                candidates = rng.choice(candidates, size=100, replace=False)
            elif len(candidates) == 0:
                candidates = np.arange(n)

            if len(candidates) == 0:
                continue

            dst_sorted = candidates[np.argsort(peak_sens[candidates])[:15]]

            for dst in dst_sorted[:8]:
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Escape with structured perturbation
            n_perturb = max(5, n // 200)
            # Preferentially perturb "hot" positions
            hot = set()
            for k2 in top_peaks[:4]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                hot.update(range(s2, e2))
            
            hot_list = list(hot)
            if len(hot_list) > n_perturb:
                hot_list = list(rng.choice(hot_list, size=n_perturb, replace=False))
            
            for i in hot_list:
                if time.monotonic() - t0 > time_budget:
                    break
                if a[i] > 1e-10:
                    j = int(rng.integers(n))
                    if j != i:
                        frac = rng.uniform(0.02, 0.06)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0.0)
                                a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"    CoordDescent ({elapsed:.1f}s): obj={obj:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_m


def spectral_polish(a0, time_budget, verbose=True):
    """
    Frequency-domain polishing: flatten |A(w)|^2 to reduce autocorrelation peaks.
    Uses gradient of max autocorrelation expressed in frequency domain.
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_budget:
            break

        beta_t = (it - 1.0) / (it + 2.0)
        y = x + beta_t * (x - x_prev)

        y = y / y.sum()

        # Use top-3 lag subgradient for sharper convergence
        ac_y = fast_conv(y)
        g = exact_subgradient_topk(y, ac_y, top_k=3)
        
        eta = 2.0 / (np.sqrt(it) + 1.0)

        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.1f}s): obj={obj:.6f}")

def make_powerlaw_init(n, p=2.0):
    """Create a decreasing power-law initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.maximum(1.0 - x / n, 0.01) ** p
    a = a / a.sum()
    return a


def make_exp_init(n, alpha=2.0):
    """Create an exponential decay initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * x / n)
    a = a / a.sum()
    return a


def make_equal_lag_init(n):
    """
    Compute the 'equal-lag' shape: the decreasing sequence where
    early convolution lags are all approximately equal.
    """
    # Solve recursively: conv[k] = T for k = 0, 1, 2, ...
    # a[0]^2 = T
    # 2*a[0]*a[1] = T
    # 2*a[0]*a[2] + a[1]^2 = T
    # 2*a[0]*a[3] + 2*a[1]*a[2] = T
    # etc.
    T = 1.0  # Will normalize later
    a = np.zeros(n)
    a[0] = np.sqrt(T)
    for k in range(1, n):
        # conv[k] = sum_{i} a[i]*a[k-i] = T
        # a[k]*a[0] + 2*sum_{i=1}^{k-1, i<=n-1-k...} ... 
        # Actually: conv[k] = a[0]*a[k] + a[1]*a[k-1] + ... + a[k]*a[0]
        #           = 2*a[0]*a[k] + 2*sum_{i=1}^{floor((k-1)/2)} a[i]*a[k-i] + (if k even) a[k/2]^2
        # So: a[k] = (T - 2*sum_{i=1}^{...} a[i]*a[k-i] - (diag term)) / (2*a[0])
        
        # More precisely: conv[k] = sum_{i=0}^{k} a[i]*a[k-i] (for k < n)
        # = a[0]*a[k] + a[k]*a[0] + sum_{i=1}^{k-1} a[i]*a[k-i]
        # = 2*a[0]*a[k] + sum_{i=1}^{k-1} a[i]*a[k-i]
        # (when k < n, all indices are valid)
        
        if k >= n:
            break
        
        cross_sum = 0.0
        for i in range(1, k):
            if i < n and k - i < n:
                cross_sum += a[i] * a[k - i]
        
        a[k] = (T - cross_sum) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.0

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return make_powerlaw_init(n, 2.0)
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            a_out, _ = peak_coord_descent(normalize(best_seq), t4, verbose=True, seed=3)
            try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Spectral polish (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Spectral polish ({remaining:.0f}s remaining) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        t5 = min(remaining * 0.25, 150.0)
        if t5 > 15:
            a_out, _ = spectral_polish(normalize(best_seq), t5, verbose=True)
            try_update(a_out, f"spectral_n{n_best}")
    
    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length ({remaining:.0f}s remaining) ===")
    
    if best_seq is not None and remaining > 100:
        n_best = len(best_seq)
        # Try longer
        for n_try in [n_best + 800, n_best + 1600, 6000, 8000]:
            if n_try <= n_best:
                continue
            if time.monotonic() - t_global > time_budget - 60:
                break
            t6 = min(remaining * 0.2, 120.0)
            if t6 > 20:
                a_interp = interpolate_seq(best_seq, n_try)
                a_out, _ = amd_optimize(normalize(a_interp), t6, alpha_init=3.0, c0=3.0, verbose=False, seed=10)
                try_update(a_out, f"amd_long_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
        
        # Try shorter
        for n_try in [n_best - 400, 2000, 1500]:
            if n_try >= n_best or n_try < 500:
                continue
            if time.monotonic() - t_global > time_budget - 60:
                break
            t6 = min(remaining * 0.15, 80.0)
            if t6 > 15:
                a_interp = interpolate_seq(best_seq, n_try)
                a_out, _ = amd_optimize(normalize(a_interp), t6, alpha_init=3.0, c0=3.0, verbose=False, seed=11)
                try_update(a_out, f"amd_short_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
    
    # === Phase 7: Final AMD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final AMD polish ({remaining:.0f}s remaining) ===")
    
    if best_seq is not None and remaining > 30:
        n_best = len(best_seq)
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 20:
            a_out, _ = amd_optimize(normalize(best_seq), t7, alpha_init=2.0, c0=2.0, verbose=True, seed=20)
            try_update(a_out, f"amd_polish_n{n_best}")
    
    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final coord ({remaining:.0f}s remaining) ===")
    
    if best_seq is not None and remaining > 20:
        n_best = len(best_seq)
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 10:
            a_out, _ = peak_coord_descent(normalize(best_seq), t8, verbose=True, seed=30)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 9: Ultra-fine polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Ultra-fine polish ({remaining:.0f}s remaining) ===")
    
    if best_seq is not None and remaining > 15:
        n_best = len(best_seq)
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 10:
            # Use very peaked subgradient for final convergence
            a_out, _ = spectral_polish(normalize(best_seq), t9, verbose=False)
            try_update(a_out, f"ultra_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n_fb = 2400
        a_fb = make_powerlaw_init(n_fb, 2.0)
        print(f"Fallback: n={n_fb}")
        return [float(x) for x in a_fb]

    return [float(x) for x in best_seq]
