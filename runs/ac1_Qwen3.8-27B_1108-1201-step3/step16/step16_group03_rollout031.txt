import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_subgradient(a, B, beta):
    """Compute smoothed subgradient using softmax weights on peaks."""
    n = len(a)
    # Compute softmax weights
    max_B = B.max()
    exp_B = np.exp(beta * (B - max_B))
    p = exp_B / exp_B.sum()
    
    # Compute gradient: g[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    g = np.zeros(n)
    for k in range(len(B)):
        if p[k] < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * p[k] * a[j]
    return g

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_smoothing_gradient(a, B, beta):
    """Compute the smoothed subgradient using softmax weights."""
    n = len(a)
    max_B = B.max()
    # Clip to avoid overflow
    scaled = beta * (B - max_B)
    scaled = np.maximum(scaled, -50)  # Prevent underflow
    exp_B = np.exp(scaled)
    p = exp_B / exp_B.sum()
    
    # g[i] = 2 * sum_j a[j] * p[i+j] for valid indices
    g = np.zeros(n)
    for j in range(n):
        aj = a[j]
        if aj < 1e-15:
            continue
        # i ranges from 0 to min(n-1, 2n-2-j)
        end = min(n, 2 * n - 1 - j)
        if end > 0:
            g[:end] += 2.0 * aj * p[j:j+end]
    return g

def softmax_subgrad_descent(a0, time_limit, beta_start=5.0, beta_end=50.0, 
                             c0=3.0, seed=42, verbose=True):
    """
    Nesterov accelerated mirror descent with softmax-smoothed subgradient
    and backtracking line search.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta = c0
    stall = 0
    max_stall = 2000
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        t = total_iters
        
        # Adaptive beta
        progress = min(t / 5000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Nesterov momentum
        beta_mom = (t - 1.0) / (t + 1.5) if t > 0 else 0.0
        y = x + beta_mom * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution and gradient
        B = fast_conv(y)
        M_curr = B.max()
        g = compute_smoothing_gradient(y, B, beta)
        
        # Backtracking line search
        eta_trial = eta
        found = False
        for bt in range(20):
            log_a = np.log(y) - eta_trial * g
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                eta_trial *= 0.5
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < M_curr - 1e-12:
                found = True
                break
            eta_trial *= 0.5
        
        if not found:
            # Use a very small step
            eta_trial = max(eta * 0.5**20, 1e-8)
            log_a = np.log(y) - eta_trial * g
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial / a_trial.sum()
        
        x_prev = x
        x = a_trial
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive step size
        if found:
            eta = min(eta * 1.05, 50.0)  # Slightly increase if line search succeeded
        else:
            eta = max(eta * 0.7, 0.5)
        
        # Restart if stuck
        if stall > max_stall:
            x_restarted = best_a + rng.normal(0, 0.01, n) * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            eta = c0
        
        total_iters += 1
        if verbose and total_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    iter={total_iters} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, eta={eta:.3f}, stall={stall}")
    
    return best_a, best_M

def coordinate_descent(a0, time_limit, seed=42, verbose=True):

            i = int(rng.integers(n))

                continue

            # Exact 1D: minimize max(conv) along the line a_i - t, a_j + t
            # Use golden section or just try several values
            best_t = 0
            best_M_trial = M_curr

                    continue
                a_trial = a.copy()

                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0.0)

                if s_t < 1e-15:

                    best_M_trial = M_trial
                    best_t = t
            
            if best_t > 0 and best_M_trial < best_M - 1e-15:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0.0)

                best_M = best_M_trial

        J = 2.0 * n * best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
