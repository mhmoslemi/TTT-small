import numpy as np
import time

def fast_autocorr(a):
    """Fast autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def nesterov_mirror_descent(a0, time_limit, verbose=True, top_k=20, beta=3.0):
    """Nesterov-accelerated mirror descent for minimizing max autocorrelation."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 2.0  # base step size parameter
    stall = 0
    max_stall = 2000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        beta_t = (t - 1) / (t + 2)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac = fast_autocorr(y)
        k_top = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_top]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        # Softmax weighting
        logits = top_vals * beta
        logits -= logits.max()
        w = np.exp(logits)
        w /= w.sum()
        
        # Subgradient g[i] = sum_r w_r * 2*y[k_r - i]
        g = np.zeros(n)
        for wr, kr in zip(w, top_idx):
            kr = int(kr)
            for i in range(n):
                j = kr - i
                if 0 <= j < n:
                    g[i] += 2.0 * wr * y[j]
        
        # Mirror descent update (multiplicative weights)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_new = np.log(y) - eta * g
        log_new -= log_new.max()  # for numerical stability
        a_new = np.exp(log_new)
        a_new = normalize(a_new)
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with perturbation if stuck
        if stall > max_stall:
            stall = 0
            # Perturb around best
            noise = rng.normal(0, 0.01, n) * best_a
            x_new = best_a + noise
            x_new = normalize(x_new)
            x = x_new
            x_prev = x_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  NMD t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a

def fast_autocorr(a):
    n = len(a)

def kkt_local_refine(a0, time_limit, verbose=True):
    """Iteratively solve KKT conditions for the active peak set."""
    n = len(a0)

        if len(active_peaks) == 0:
            break
        if len(active_peaks) > 10:
            # Keep only top 10
            top_order = np.argsort(ac[active_peaks])[::-1][:10]
            active_peaks = active_peaks[top_order]

        peak_vals = ac[active_peaks]
        peak_vals = np.maximum(peak_vals, 1e-15)
        lam = peak_vals / peak_vals.sum()

        eta = 1.0

                if i_max > i_min:

                    w_i[idx] += lr * a[kr - idx]

            log_a = -eta * w_i + np.log(np.maximum(a, 1e-15))
            log_a -= log_a.max()
            a_new = np.exp(log_a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Nesterov Mirror Descent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 1: Nesterov Mirror Descent, n={n0} ===")
        t1 = min(remaining * 0.35, 500.0)
        a_out, _ = nesterov_mirror_descent(norm_sum := normalize_nonneg(best_seq), t1,
                                           top_k=30, beta=4.0, gamma=3.0, verbose=True)
        try_update(a_out, "nmd_p1")

    # === Phase 2: LP-FW (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW, n={n_cur} ===")
        t2 = min(remaining * 0.25, 400.0)
        a_out, _ = lp_frank_wolfe(normalize_nonneg(best_seq), t2, solver, verbose=True)
        try_update(a_out, f"lp_p2_n{n_cur}")

    # === Phase 3: Aggressive Coordinate Descent (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord Descent, n={n_cur} ===")
        t3 = min(remaining * 0.15, 250.0)
        a_out, _ = aggressive_coord_descent(normalize_nonneg(best_seq), t3, verbose=True)
        try_update(a_out, f"coord_p3_n{n_cur}")

    # === Phase 4: KKT Local Refinement (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: KKT Refine, n={n_cur} ===")
        t4 = min(remaining * 0.10, 150.0)
        a_out, _ = kkt_local_refine(normalize_nonneg(best_seq), t4, verbose=True)
        try_update(a_out, f"kkt_p4_n{n_cur}")

    # === Phase 5: Multi-n exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: Multi-n exploration ===")
        test_ns = []
        for delta in [200, 400, -200, 600, -400]:
            nt = n_best + delta
            if 1200 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            # Quick NMD at new length
            t5a = min(remaining * 0.3, 120.0)
            a_out, _ = nesterov_mirror_descent(normalize_nonneg(a_inter), t5a,
                                               top_k=20, beta=4.0, gamma=3.0, verbose=False)
            try_update(a_out, f"nmd_n{nt}")
            # LP-FW polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t5b = min(remaining * 0.4, 100.0)
                a_out, _ = lp_frank_wolfe(normalize_nonneg(a_inter), t5b, solver, verbose=False)
                try_update(a_out, f"lp_n{nt}")

    # === Phase 6: Final combined polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final polish, n={n_best} ===")

        # NMD with different params
        t6a = min(remaining * 0.3, 150.0)
        a_out, _ = nesterov_mirror_descent(normalize_nonneg(best_seq), t6a,
                                           top_k=20, beta=6.0, gamma=5.0,
                                           verbose=False, rng_seed=777)
        try_update(a_out, "nmd_p6a")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.4, 150.0)
            a_out, _ = aggressive_coord_descent(normalize_nonneg(best_seq), t6b,
                                                verbose=True, rng_seed=555)
            try_update(a_out, "coord_p6b")

        # LP-FW final
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6c = min(remaining * 0.5, 100.0)
            a_out, _ = lp_frank_wolfe(normalize_nonneg(best_seq), t6c, solver, verbose=True)
            try_update(a_out, "lp_p6c")

        # KKT final
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6d = min(remaining * 0.5, 60.0)
            a_out, _ = kkt_local_refine(normalize_nonneg(best_seq), t6d, verbose=True)
            try_update(a_out, "kkt_p6d")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
