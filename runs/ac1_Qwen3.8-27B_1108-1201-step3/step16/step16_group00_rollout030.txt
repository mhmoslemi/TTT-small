Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LP Frank-Wolfe
with Nesterov acceleration, adaptive constraint selection, and coordinate descent polishing.
Key improvements: efficient constraint matrix construction via convolution structure,

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver

def fast_conv(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_constraint_matrix(y, lags):
    """Build the K x n constraint matrix for selected lags.
    A[k_idx, j] = y[k - j] if 0 <= k - j < n, else 0
    where k = lags[k_idx]
    """
    n = len(y)
    K = len(lags)
    A = np.zeros((K, n))
    for idx, k in enumerate(lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            A[idx, js] = y[k - js]
    return A


def lp_frank_wolfe_fast(a0, time_limit, solver, verbose=True, K_start=50, K_end=300, verbose_every=10):
    """FFT-accelerated LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Nesterov state
    x = y.copy()
    y_prev = y.copy()
    
    total_iters = 0
    lp_fail_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive K
        progress = min(1.0, t / 500.0)
        K = int(K_start + (K_end - K_start) * progress)
        K = min(K, 2*n - 1)
        
        # Compute current autocorrelation
        ac = fast_conv(y)
        M = ac.max()
        if M < 1e-16:
            break
        
        # Select top-K lags (most binding constraints)
        top_idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix
        A_np = build_constraint_matrix(y, top_idx)
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A_np @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 20:
                # Restart
                y = best_a.copy()
                x = y.copy()
                y_prev = y.copy()
                lp_fail_count = 0
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            continue
        
        lp_fail_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        d = b_val / b_sum  # direction (sums to 1)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * d
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Check improvement
        ac_new = fast_conv(y_new)
        M_new = ac_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y_new.copy()
        
        # Nesterov momentum
        if t == 1:
            x = d.copy()
        else:
            beta_t = (np.sqrt(t + 1) - 1) / (np.sqrt(t + 1) + 1)
            x = y_new + beta_t * (y_new - y)
            x = np.maximum(x, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                x = y_new.copy()
            else:
                x = x / sx
        
        # Use x for evaluation (it might be better than y)
        ac_x = fast_conv(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
        
        y_prev = y
        y = y_new
        total_iters += 1
        
        if verbose and total_iters % verbose_every == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={total_iters} ({elapsed:.0f}s): J={J:.6f}, K={K}, M={M:.6f}")
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Aggressive coordinate descent for local optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M = ac.max()
        
        # Find the top few peak lags
        n_peaks = min(10, len(ac))
        peak_lags = np.argsort(ac)[::-1][:n_peaks]
        
        # For each peak lag, find the "hottest" pair and try to cool it
        improved = False
        
        for k in peak_lags[:5]:
            ki = int(k)
            # Get contributing pairs
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            
            idx = np.arange(i_start, i_end)
            # Contribution of each pair: a[i] * a[ki - i]
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            
            # Source: the position in idx with the largest contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = ki - src
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: position that minimizes increase in peak
            # Simple heuristic: move to a position far from all peak lags
            # Score each position by its "exposure" to peaks
            scores = np.zeros(n)
            for k2 in peak_lags:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                for pos in range(s2, e2):
                    # If we add mass at pos, it affects lag k2 through a[pos]*a[k2-pos]
                    j2 = k2i - pos
                    if 0 <= j2 < n:
                        scores[pos] += a[j2]
            
            # Best destination has lowest score
            best_dst = np.argmin(scores)
            if best_dst == src:
                best_dst = (best_dst + 1) % n
            
            # Try moving various fractions
            for frac in [0.1, 0.05, 0.02, 0.01, 0.005]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if a.max() > 1e-10:
                i = int(np.argmax(a))
                j = int(rng.integers(n))
                if i != j:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 20 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric_init(n, param_type="ramp", **kwargs):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)
    if param_type == "ramp":
        power = kwargs.get("power", 1.0)
        a = (x + 0.01) ** power
    elif param_type == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif param_type == "gaussian":
        mu = kwargs.get("mu", 0.5)
        sigma = kwargs.get("sigma", 0.2)
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif param_type == "bimodal":
        a = np.exp(-20*(x-0.2)**2) + np.exp(-20*(x-0.8)**2) + 0.01
    elif param_type == "sinusoidal":
        a = np.sin(np.pi * x) ** 2
    elif param_type == "hyperbolic":
        a = 1.0 / (1.0 + 10.0 * x)
    else:
        a = np.ones(n)
    return norm_sum1(np.maximum(a, 0.01 * a.max()))

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a, b=None):
    """Fast linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safe wrapper around evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_constraint_matrix(y, lags):
    """Build K x n matrix where A[idx, j] = y[k - j] if 0 <= k-j < n, else 0."""
    n = len(y)
    K = len(lags)
    A = np.zeros((K, n), dtype=np.float64)
    for idx in range(K):
        k = int(lags[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            A[idx, js] = y[k - js]
    return A


def lp_frank_wolfe_fast(a0, time_limit, solver, verbose=True, K_start=50, K_end=300, verbose_every=20, rng_seed=0):
    """LP Frank-Wolfe with Nesterov acceleration and adaptive K."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x_momentum = y.copy()
    y_prev = y.copy()
    luc = float(best_m)
    iter_count = 0
    lp_fail = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / max(1, t * 0.0))
        frac = min(1.0, t / 800.0)
        K = int(K_start + (K_end - K_start) * frac)
        K = min(K, 2 * n - 1)

        ac = fast_conv(y)
        M = ac.max()
        if M < 1e-16:
            break

        # Select top-K lags
        top_idx = np.argsort(ac)[::-1][:K]

        # Occasionally include some random lags for diversity
        if t > 100 and K < 2 * n - 1:
            n_rand = max(1, K // 20)
            all_lags = np.arange(2 * n - 1)
            remaining = np.setdiff1d(all_lags, top_idx)
            if len(remaining) > n_rand:
                rand_lags = rng.choice(remaining, size=n_rand, replace=False)
                top_idx = np.concatenate([top_idx[:K - n_rand], rand_lags])

        A_np = build_constraint_matrix(y, top_idx)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A_np @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail += 1
            if lp_fail > 30:
                y = best_a.copy()
                x_momentum = y.copy()
                y_prev = y.copy()
                luc = float(fast_conv(y).max())
                lp_fail = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail += 1
            if lp_fail > 30:
                y = best_a.copy()
                x_momentum = y.copy()
                y_prev = y.copy()
                lp_fail = 0
            continue

        lp_fail = 0
        b_val = np.asarray(b_var.value, dtype=np.float64) if b_var.value is not None else np.zeros(n)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue

        d = b_val / b_sum

        # Proper FW step size using luc
        luc_val = float(np.dot(y - d, ac - np.dot(y, ac)))
        if luc_val > 1e-15:
            alpha = 2.0 * luc / (luc + luc_val)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * d
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum
        beta_t = (np.sqrt(t + 1) - 1) / (np.sqrt(t + 1) + 1)
        x_momentum = y_new + beta_t * (y_new - y_prev)
        x_momentum = np.maximum(x_momentum, 0.0)
        sx = x_momentum.sum()
        if sx < 1e-15:
            x_momentum = y_new.copy()
        else:
            x_momentum = x_momentum / sx

        # Update luc
        luc = float(np.dot(x_momentum - d, ac - np.dot(x_momentum, ac)))

        # Check improvement at y_new
        ac_yn = fast_conv(y_new)
        M_yn = ac_yn.max()
        if M_yn < best_m - 1e-15:
            best_m = M_yn
            best_a = y_new.copy()

        # Check improvement at x_momentum
        ac_x = fast_conv(x_momentum)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x_momentum.copy()
            y = x_momentum.copy()

        y_prev = y.copy()
        y = y_new
        iter_count += 1

        # Periodic restart if stuck
        if iter_count % 3000 == 0:
            ac_check = fast_conv(y)
            M_check = ac_check.max()
            if M_check > best_m * 1.001:
                y = best_a + 0.01 * rng.normal(0, 1, n) * best_a
                y = np.maximum(y, 1e-10)
                y = y / y.sum()
                y_prev = y.copy()
                x_momentum = y.copy()

        if verbose and iter_count % verbose_every == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.7f}, K={K}")

    return best_a, best_m


def coordinate_descent_aggressive(a0, time_limit, verbose=True, rng_seed=42):
    """Aggressive coordinate descent with sensitivity-aware moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M = ac.max()

        peak_lags = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in peak_lags[:8]:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue

            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the pair with maximum contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute sensitivity: for each position, how "exposed" is it to peaks?
            scores = np.zeros(n)
            for k2 in peak_lags[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                pos_range = np.arange(s2, e2)
                partners = k2i - pos_range
                valid = (partners >= 0) & (partners < n)
                pos_range = pos_range[valid]
                partners = partners[valid]
                scores[pos_range] += a[partners]

            # Best destination: lowest sensitivity, not the source
            scores[src] = np.inf
            dst = int(np.argmin(scores))

            # Try multiple transfer fractions
            for frac in [0.15, 0.10, 0.05, 0.03, 0.01]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial.copy()
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Escalating random perturbation
            if stall % 10 == 0:
                n_moves = min(stall // 10, 5)
                for _ in range(n_moves):
                    i = int(np.argmax(a))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.02, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break

        if verbose and stall % 30 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.7f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_smooth_init(n, param_type="ramp", seed=0):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if param_type == "ramp":
        a = (x + 0.05) ** 1.0
    elif param_type == "ramp2":
        a = (x + 0.05) ** 2.0
    elif param_type == "sinusoidal":
        a = np.sin(np.pi * x) ** 2
    elif param_type == "bimodal":
        a = np.exp(-30 * (x - 0.15) ** 2) + np.exp(-30 * (x - 0.85) ** 2) + 0.01
    elif param_type == "uniform":
        a = np.ones(n)
    elif param_type == "hyperbolic":
        a = 1.0 / (0.5 + 5.0 * x)
    elif param_type == "smooth_rand":
        a = np.ones(n) * 0.5
        for freq in [1, 2, 3, 4, 5, 7]:
            coeff = rng.normal(0, 0.08)
            phase = rng.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    return norm_sum1(a)

    for delta in [-300, -100, 100, 300, 500, 700]:
        nn = n_best + delta
        if 2000 <= nn <= 5500 and nn != n_best:
            candidate_lengths.append(nn)
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    for new_n in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"\n  n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_sequence(best_seq, new_n)

        t_lp = min(remaining * 0.20, 100.0)
        if t_lp > 10:
            a_out, _ = lp_frank_wolfe_fast(a_interp, t_lp, solver, verbose=False,
                                            K_start=40, K_end=150, verbose_every=50)
            a_norm = norm_sum1(a_out)
            sc = evaluate_safe(a_norm)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_norm.copy()
                print(f"  *** NEW BEST [n{new_n}]: {sc:.7f}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_cd = min(remaining * 0.06, 40.0)
            if t_cd > 5:
                a_out, _ = coordinate_descent_aggressive(norm_sum1(best_seq), t_cd, verbose=False)
                a_norm = norm_sum1(a_out)
                sc = evaluate_safe(a_norm)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_norm.copy()
                    print(f"  *** NEW BEST [coord_n{new_n}]: {sc:.7f}")

    # === Phase 6: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n{'='*60}")
    print(f"Phase 6: Final LP-FW on n={n_best} (budget={min(remaining*0.30, 250):.0f}s)")
    print(f"{'='*60}")
    t6 = min(remaining * 0.30, 250.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_fast(a_in, t6, solver, verbose=True,
                                        K_start=100, K_end=500, verbose_every=15)
        a_norm = norm_sum1(a_out)
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [final_lpfw_n{n_best}]: {sc:.7f}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n{'='*60}")
    print(f"Phase 7: Final coord polish on n={n_best} (budget={min(remaining*0.25, 80):.0f}s)")
    print(f"{'='*60}")
    t7 = min(remaining * 0.25, 80.0)
    if t7 > 10:
        a_out, _ = coordinate_descent_aggressive(norm_sum1(best_seq), t7, verbose=True, rng_seed=777)
        a_norm = norm_sum1(a_out)
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [final_coord_n{n_best}]: {sc:.7f}")

    # === Final validation ===
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            if final_score < best_score:
                best_score = final_score
            print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        else:
            print(f"\nWARNING: Final evaluation returned inf, using internal score={best_score:.7f}")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3200
        best_seq = make_smooth_init(n_fb, "sinusoidal")
        best_score = evaluate_safe(best_seq)
        print(f"FB: score={best_score:.7f}")

    return [float(x) for x in best_seq]
