def lp_optimize(a0, time_limit, solver, verbose=True):
    """LP-based successive linearization for peak minimization."""
    n = len(a0)
    a_cur = norm_sum1(a0)
    best_a = a_cur.copy()
    best_M = fast_autocorr(a_cur).max()
    t0 = time.monotonic()
    iter_count = 0
    K = 200  # number of constraints to use
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a_cur)
        M_cur = c.max()
        
        if M_cur < best_M - 1e-15:
            best_M = M_cur
            best_a = a_cur.copy()
        
        # Select top-K constraints
        K_use = min(K, 2 * n - 1)
        binding_idx = np.argsort(c)[::-1][:K_use]
        
        # Build constraint matrix R where R[idx, i] = a_cur[k-i] for valid i
        # Then constraint is: 2 * (R @ d) <= M_new - c[binding_idx]
        R = np.zeros((K_use, n))
        for idx, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_end > i_start:
                R[idx, i_start:i_end] = a_cur[k_int - (np.arange(i_start, i_end))]
        
        # Solve LP:
        # min M_new
        # s.t. sum(d) = 0
        #      a_cur + d >= 0
        #      2*R@d + c[binding_idx] <= M_new
        #      M_new <= M_cur (don't increase)
        
        d = cp.Variable(n)
        M_new = cp.Variable()
        
        constraints = [
            cp.sum(d) == 0,
            a_cur + d >= 0,
            2.0 * (R @ d) + c[binding_idx] <= M_new,
            M_new <= M_cur,
        ]
        
        prob = cp.Problem(cp.Minimize(M_new), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        d_val = np.asarray(d.value, dtype=np.float64)
        M_LP = M_new.value
        
        # Check improvement
        if M_LP < M_cur * (1 - 1e-10):
            # Take the step
            a_trial = a_cur + d_val
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            
            if M_trial < M_cur:
                a_cur = a_trial
                iter_count += 1
                # Gradually increase K
                if iter_count % 20 == 0:
                    K = min(K + 50, 800)
            else:
                # Linearization was too optimistic; try smaller step
                improved = False
                for alpha in [0.5, 0.1, 0.05]:
                    a_trial = a_cur + alpha * d_val
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_cur * (1 - 1e-12):
                        a_cur = a_trial
                        iter_count += 1
                        improved = True
                        break
                if not improved:
                    # Stuck; try to restart K
                    K = max(100, K // 2)
                    if iter_count > 0 and (iter_count % 50 == 0):
                        break
        else:
            break
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.8f}, K={K}")
    
    return best_a, best_M

def coord_descent(a0, time_limit, verbose=True):
    """Targeted coordinate descent focused on reducing peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_autocorr(a)
        M_cur = c.max()
        
        # Get top peaks
        n_peaks = min(10, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        
        for k in peak_idx[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            indices = np.arange(i_start, i_end)
            j_indices = k_int - indices
            
            # Find the pair with maximum product
            products = a[indices] * a[j_indices]
            max_pair_idx = np.argmax(products)
            src_i = indices[max_pair_idx]
            src_j = j_indices[max_pair_idx]
            
            # Try to move mass from the larger of a[src_i], a[src_j] to a random position
            src = src_i if a[src_i] >= a[src_j] else src_j
            if a[src] < 1e-12:
                continue
            
            # Find a good destination: one that doesn't increase other peaks
            # Score each position by its contribution to top peaks
            scores = np.zeros(n)
            for k2 in peak_idx[:3]:
                k2_int = int(k2)
                s2 = max(0, k2_int - n + 1)
                e2 = min(n, k2_int + 1)
                for i in range(s2, e2):
                    j2 = k2_int - i
                    scores[i] += a[j2]
                    if j2 != i:
                        scores[j2] += a[i]
            
            # Choose destination with lowest score
            dst = np.argmin(scores)
            if dst == src:
                dst = (src + n // 2) % n
            
            # Try moving some mass
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-14:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    """Interpolate a sequence from its current length to new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return norm_sum1(f_interp)
