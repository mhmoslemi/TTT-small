def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, w):
    """Compute cross-correlation: result[i] = sum_j a[j] * w[i+j] for i=0..n-1.
    w has length 2n-1."""
    n = len(a)
    size = 4 * n
    a_fft = np.fft.rfft(a, size)
    w_fft = np.fft.rfft(w, size)
    return np.fft.irfft(a_fft * np.conj(w_fft), size)[:n]


def compute_smooth_gradient(a, conv_a, eps):
    """Compute gradient of eps*log(sum_k exp(conv_a[k]/eps)) w.r.t. a."""
    # Soft weights
    max_conv = conv_a.max()
    logits = (conv_a - max_conv) / eps
    weights = np.exp(logits)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones_like(conv_a) / len(conv_a)
    else:
        weights = weights / w_sum
    # Gradient: 2 * cross_correlation(a, weights)
    grad = 2.0 * fast_cross_corr(a, weights)
    return grad, weights


def compute_hard_gradient(a, conv_a, top_k=1):
    """Compute hard subgradient: top-k peak gradients."""
    n = len(a)
    k_max = min(top_k, len(conv_a))
    top_idx = np.argsort(conv_a)[::-1][:k_max]
    top_vals = np.maximum(conv_a[top_idx], 1e-30)
    w = top_vals ** 2.0
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += 2.0 * wi * a[j]
    return grad

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smooth_mirror_descent(a0, time_limit, eps_start=None, eps_end=None,

        eps = eps_start * (eps_end / eps_start) ** progress

        if grad_norm < 1e-15:

        eta = c0 / (grad_norm * (1.0 + 0.1 * t / total_iters))
        # Cap step size

            x = x_restarted

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth (logsumexp) Nesterov mirror descent with adaptive epsilon and FFT-based gradients
2. LP-based Frank-Wolfe with warm-start direction reuse
3. Stochastic coordinate descent with analytic peak perturbation
4. Adaptive hyperparameter schedules and periodic restarts
5. Multi-length exploration via interpolation
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, w):
    """Cross-correlation: result[i] = sum_j a[j]*w[i+j] for i=0..n-1. w has length 2n-1."""
    n = len(a)
    size = 4 * n
    a_fft = np.fft.rfft(a, size)
    w_fft = np.fft.rfft(w, size)
    return np.fft.irfft(a_fft * np.conj(w_fft), size)[:n]


def compute_smooth_gradient(a, conv_a, eps):
    """Gradient of eps*log(sum exp(conv_a/eps)) w.r.t. a. Returns (grad, weights)."""
    n = len(a)
    max_c = conv_a.max()
    logits = (conv_a - max_c) / eps
    weights = np.exp(logits)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones_like(conv_a) / len(conv_a)
    else:
        weights = weights / w_sum
    grad = 2.0 * fast_cross_corr(a, weights)
    return grad, weights


def compute_topk_gradient(a, conv_a, k, beta=2.0):
    """Weighted sum of top-k peak gradients."""
    n = len(a)
    kk = min(k, len(conv_a))
    top_idx = np.argsort(conv_a)[::-1][:kk]
    top_vals = np.maximum(conv_a[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(kk) / kk
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        i_s = max(0, ki - n + 1)
        i_e = min(n, ki + 1)
        for i in range(i_s, i_e):
            j = ki - i
            if 0 <= j < n:
                grad[i] += 2.0 * wi * a[j]
    return grad


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smooth_mirror_descent(a0, time_limit, eps_start=None, eps_end=None,
                          c0=3.0, verbose=True, seed=42, max_stall=2000,
                          total_iters=100000, top_k_switch=8000, beta_topk=2.0):
    """
    Nesterov-accelerated mirror descent with logsumexp-smoothed objective.
    Uses FFT for efficient gradient computation. Adaptive epsilon schedule.
    Switches to top-k hard gradient in later phase.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv_init = fast_conv(a)
    best_m = conv_init.max()
    init_peak = best_m

    if eps_start is None:
        eps_start = init_peak / 4.0
    if eps_end is None:
        eps_end = init_peak / 3000.0

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Geometric epsilon decay
        progress = min(t / total_iters, 1.0)
        eps = eps_start * (eps_end / eps_start) ** progress

        # Nesterov momentum
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute convolution
        conv_y = fast_conv(y)

        # Choose gradient type: smooth early, hard later
        if t < top_k_switch:
            grad, _ = compute_smooth_gradient(y, conv_y, eps)
        else:
            top_k = 1 + (t - top_k_switch) // 5000
            top_k = min(top_k, 5)
            grad = compute_topk_gradient(y, conv_y, top_k=top_k, beta=beta_topk)

        # Adaptive step size
        grad_max = np.abs(grad).max()
        if grad_max < 1e-15:
            grad_max = 1e-15
        eta = c0 / (grad_max * (1.0 + 0.05 * t / total_iters))
        eta = min(eta, 5.0 / (grad_max + 1e-15))

        # Mirror descent (multiplicative update)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart with perturbation
        if stall > max_stall:
            noise_scale = 0.001 + 0.002 * (stall / max_stall)
            x_r = best_a + rng.normal(0, noise_scale, n) * np.maximum(best_a, 0.01)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            M_r = fast_conv(x_r).max()
            if M_r < best_m:
                best_m = M_r
                best_a = x_r.copy()
            x = x_r
            x_prev = x_r
            stall = 0

        # Large random jump every 20000 iterations
        if t % 20000 == 0:
            x_j = best_a + rng.normal(0, 0.01, n) * np.maximum(best_a, 0.01)
            x_j = np.maximum(x_j, 1e-8)
            x_j = x_j / x_j.sum()
            M_j = fast_conv(x_j).max()
            if M_j < best_m:
                best_m = M_j
                best_a = x_j.copy()
                x = x_j
                x_prev = x_j

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            mode = "smooth" if t < top_k_switch else f"topk{min(1+(t-top_k_switch)//5000, 5)}"
            print(f"    SMD t={t} ({elapsed:.1f}s): J={J:.6f}, eps={eps:.1e}, mode={mode}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):

            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:

        need_lp = (t == 1) or (time.monotonic() - last_lp_time > 0.8 and t % 2 == 0)

            for j in range(n):
                T[j:j + n, j] = y
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M_curr]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                cached_direction = None
                continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                cached_direction = None
                continue
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                cached_direction = None
                continue
            cached_direction = b_val / b_sum
            last_lp_time = time.monotonic()
            lp_count += 1

        if cached_direction is None:
            # Fallback: negative subgradient of top peak
            k_star = int(np.argmax(ac))
            direction = np.zeros(n)
            i_s = max(0, k_star - n + 1)
            i_e = min(n, k_star + 1)
            for i in range(i_s, i_e):

                if 0 <= j < n:
                    direction[i] = -2.0 * y[j]

                direction = y.copy()

            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)

            trial_m = fast_conv(trial).max()

                best_eval = trial_m
                best_alpha = alpha
                best_trial = trial

        y_new = best_trial.copy()

        # Nesterov momentum
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx if sx > 1e-15 else y_new

        y = y_new
        x_mom = x_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_conv(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}, lps={lp_count}")

            for k2 in top_peaks[:3]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 200)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            found_for_k = False
            for src in srcs[:4]:
                if time.monotonic() - t0 > time_limit or found_for_k:
                    break
                src = int(src)
                if a[src] < 1e-14:
                    continue

                # Find best destination: minimize a[k-dst]
                best_dst = -1
                best_val = float('inf')
                for dst in chosen:
                    dst = int(dst)
                    if dst == src:
                        continue
                    val = a[k_int - dst] if 0 <= k_int - dst < n else 0.0
                    if val < best_val:
                        best_val = val
                        best_dst = dst

                if best_dst < 0:
                    continue
                dst = best_dst

                for frac in [0.001, 0.003, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-14:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found_for_k = True
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape when stuck
            if stall > 100:
                n_pert = rng.integers(2, 6)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.003, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with adaptive time allocation.
    Primary: smooth gradient Nesterov mirror descent.
    Secondary: LP Frank-Wolfe, coordinate descent.
    Tries multiple n values and optimization configs.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    # === Phase 1: Primary smooth mirror descent (long) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: SMD n={n0} (primary) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_mirror_descent(a_in, t1, c0=4.0, verbose=True, seed=42,
                                         total_iters=80000, top_k_switch=50000,
                                         max_stall=1500)
        try_update(a_out, f"smd1_n{n0}")

    # === Phase 2: SMD with different hyperparameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SMD variant n={len(best_seq)} ===")
    t2 = min(remaining * 0.18, 300.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        # Different epsilon schedule: faster decay
        conv_in = fast_conv(a_in)
        ep = conv_in.max()
        a_out, _ = smooth_mirror_descent(a_in, t2, eps_start=ep / 2.0, eps_end=ep / 5000.0,
                                         c0=5.0, verbose=True, seed=7,
                                         total_iters=60000, top_k_switch=30000,
                                         max_stall=1200)
        try_update(a_out, f"smd2_n{len(best_seq)}")

    # === Phase 3: SMD with slower decay, more smooth phase ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: SMD slow-decay n={len(best_seq)} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        conv_in = fast_conv(a_in)
        ep = conv_in.max()
        a_out, _ = smooth_mirror_descent(a_in, t3, eps_start=ep / 8.0, eps_end=ep / 1000.0,
                                         c0=3.0, verbose=True, seed=123,
                                         total_iters=50000, top_k_switch=35000,
                                         max_stall=1000)
        try_update(a_out, f"smd3_n{len(best_seq)}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP n={len(best_seq)} ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 50:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True)
        try_update(a_out, f"lp_n{len(best_seq)}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord n={len(best_seq)} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 200:
        deltas = [-200, 200, -100, 100]
        for delta in deltas:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            n_try = n_best + delta
            if n_try < 1000 or n_try > 4000 or n_try == n_best:
                continue
            print(f"\n=== Phase 6: n={n_try} ===")
            a_interp = interpolate_sequence(best_seq, n_try)
            a_in = norm_sum1(a_interp)

            t6 = min(remaining * 0.35, 150.0)
            if t6 > 30:
                a_out, _ = smooth_mirror_descent(a_in, t6, c0=4.0, verbose=False,
                                                 seed=55 + n_try % 100,
                                                 total_iters=40000, top_k_switch=25000,
                                                 max_stall=1000)
                try_update(a_out, f"smd_n{n_try}")

            # Quick coord polish on the new length
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t6b = min(remaining * 0.2, 60.0)
                if t6b > 15:
                    a_in2 = norm_sum1(best_seq)
                    a_interp2 = interpolate_sequence(a_in2, n_try)
                    a_out2, _ = coordinate_polish(a_interp2, t6b, verbose=False, seed=77)
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 7: Final SMD refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 3000 and remaining > 60:
        print(f"\n=== Phase 7: Final SMD n={n_best} ===")
        t7 = min(remaining * 0.4, 120.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            conv_in = fast_conv(a_in)
            ep = conv_in.max()
            a_out, _ = smooth_mirror_descent(a_in, t7, eps_start=ep / 20.0,
                                             eps_end=ep / 10000.0,
                                             c0=6.0, verbose=False, seed=999,
                                             total_iters=40000, top_k_switch=20000,
                                             max_stall=800)
            try_update(a_out, f"smd_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
