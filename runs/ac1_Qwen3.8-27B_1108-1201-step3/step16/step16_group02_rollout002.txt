            if prob.status in ('optimal', 'optimal_inaccurate'):
                _GLOBAL_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _GLOBAL_SOLVER


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _active_set_subgradient(a, ac, tol_frac=0.02, power=3.0):
    """
    Compute exact subgradient of f(a) = max_k ac[k] using active set.
    Active set: lags where ac[k] >= (1 - tol_frac) * ac.max()
    Gradient of q_k(a) = sum_i a[i]*a[i+k] w.r.t. a[j]:
      if k==0: 2*a[j]
      if k>0: a[j-k] (if valid) + a[j+k] (if valid)
    Weights: proportional to ac[k]^power over active set.
    """
    n = len(a)
    ac_max = ac.max()
    if ac_max <= 0:
        return np.zeros(n)

    threshold = ac_max * (1.0 - tol_frac)
    active = np.where(ac >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(ac)])

    # Weights
    w = np.maximum(ac[active], 1e-30) ** power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum

    # Compute gradient
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        if k_int == 0:
            grad += wi * 2.0 * a
        else:
            # a[j-k] for j = k, ..., n-1
            j_start = k_int
            grad[j_start:] += wi * a[:n - k_int]
            # a[j+k] for j = 0, ..., n-1-k
            j_end = n - k_int
            grad[:j_end] += wi * a[k_int:]

    return grad


def _nesterov_active_descent(a0, time_limit, tol_frac_init=0.05, tol_frac_final=0.005,
                              c_step=2.0, verbose=True):
    """Nesterov accelerated descent with exact active-set subgradient and backtracking."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    ac0 = _fast_autocorr(a)
    best_m = ac0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000

    # Geometric ergodic average
    geo_avg = np.zeros(n)
    geo_weight = 0.0
    decay = 0.995  # geometric decay factor

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive tolerance: tighten over time
        frac = min(1.0, t / 10000.0)
        tol_frac = tol_frac_init * (1 - frac) + tol_frac_final * frac

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient at y
        ac_y = _fast_autocorr(y)
        g = _active_set_subgradient(y, ac_y, tol_frac=tol_frac, power=3.0)

        # Adaptive step size: eta = c / (sqrt(t) + 1)
        eta = c_step / (np.sqrt(t) + 1.0)

        # Mirror descent step (exponential update)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Update geometric ergodic average
        w_t = (1.0 - decay) * decay ** t
        geo_weight += w_t
        geo_avg = (geo_avg * geo_weight * decay ** 0 + x * w_t) / geo_weight if geo_weight > 0 else x.copy()

        M_new = _fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Check geometric average periodically
        if t % 500 == 0:
            M_geo = _fast_autocorr(geo_avg).max()
            if M_geo < best_m:
                best_m = M_geo
                best_a = geo_avg.copy()

        # Restart if stalled
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c_step = max(0.5, c_step * 0.6)
            geo_avg = x_r.copy()
            geo_weight = 1.0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NAD t={t} ({elapsed:.1f}s): J={J:.8f}, tol={tol_frac:.5f}, stall={stall_count}")

    return best_a, best_m


def _lp_frank_wolfe(a0, time_limit, solver, max_constraints=250, verbose=True):
    """LP Frank-Wolfe with progressive constraint tightening and warm-starting."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    threshold_frac = 0.99  # start loose, tighten over time
    b_warm = None

    for t in range(1, 30000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        # Progressive tightening
        threshold_frac = 0.99 - 0.02 * min(1.0, t / 2000.0)
        threshold = M_curr * threshold_frac
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = _fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = _fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.8f}, active={len(active_lags)}")

    return best_a, best_m


def _equalization_polish(a0, time_limit, top_k=30, verbose=True):
    """
    Water-filling style equalization: try to make top-k autocorrelation lags equal.
    At each step, identify the lag with max value and the lag with min value among top-k,
    then move mass to reduce the gap.
    """
    n = len(a0)
    a = _norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(424242)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 50:
        ac = _fast_autocorr(a)
        # Get top-k lags
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        delta = top_vals[0] - top_vals[-1]

        if delta < 1e-10:
            break

        # Find a pair (i, j) such that moving mass from i to j reduces the max lag
        # and increases the min lag
        best_improvement = 0
        best_move = None

        # Strategy: the peak lag k_max needs to be reduced.
        # Moving mass from position p to q changes ac[k] by -delta*(a[p-k]+a[p+k]) + delta*(a[q-k]+a[q+k])
        # We want this to be negative for the max lag and positive for min lags.

        n_tries = min(20, n // 2)
        peak_lag = int(np.argmax(ac))

        for _ in range(n_tries):
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if p == q or a[p] < 1e-12:
                continue

            # Estimate effect on peak lag
            # Change in ac[peak_lag] when moving eps from p to q:
            # -eps * (a[p-peak_lag] + a[p+peak_lag]) + eps * (a[q-peak_lag] + a[q+peak_lag])
            def ac_change(pos, lag):
                c = 0.0
                if 0 <= pos - lag < n:
                    c += a[pos - lag]
                if 0 <= pos + lag < n:
                    c += a[pos + lag]
                return c

            eff_peak = -(ac_change(p, peak_lag) - ac_change(q, peak_lag))
            if eff_peak >= 0:
                continue

            # Check a few trial fractions
            for frac in [0.001, 0.005, 0.01, 0.02]:
                eps = a[p] * frac
                if eps < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[p] -= eps
                a_trial[q] += eps
                if a_trial[p] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = _fast_autocorr(a_trial).max()
                if m_trial < best_m - best_improvement - 1e-14:
                    best_improvement = best_m - m_trial
                    best_move = (p, q, eps)

        if best_move is not None:
            p, q, eps = best_move
            a[p] -= eps
            a[q] += eps
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            best_m = _fast_autocorr(a).max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if p != q and a[p] > 1e-8:
                eps = a[p] * 0.01
                a[p] -= eps
                a[q] += eps
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    EQ ({elapsed:.1f}s): J={J:.8f}, stall={stall}, gap={delta:.2e}")

    return best_a, best_m


def _coordinate_descent_golden(a0, time_limit, verbose=True, rng_seed=2024):
    """
    Coordinate descent with golden-section line search.
    For each coordinate pair (i, j), find optimal transfer amount via golden section.
    """
    n = len(a0)
    a = _norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 3
    golden = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618

    sweep = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        sweep_improved = False

        # Randomize coordinate order
        coord_order = np.arange(n)
        rng.shuffle(coord_order)

        for i in coord_order:
            if time.monotonic() - t0 > time_limit:
                    break
            if a[i] < 1e-12:
                continue

            # Pick 3 random destinations
            dests = rng.integers(0, n, size=3)
            best_trial_m = best_m
            best_trial = None

            for dst in dests:
                if dst == i:
                    continue
                # Golden section search for optimal transfer from i to dst
                # f(eps) = max_autocorr(a - eps*e_i + eps*e_dst) normalized
                lo = 0.0
                hi = a[i] * 0.5  # don't transfer more than half

                if hi < 1e-15:
                    continue

                # Evaluate at golden section points
                c = hi * (1.0 - golden)
                d = hi * golden

                def _eval_at(eps):
                    a_t = a.copy()
                    a_t[i] -= eps
                    a_t[dst] += eps
                    if a_t[i] < 0 or a_t[dst] < 0:
                        return float('inf')
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        return float('inf')
                    return _fast_autocorr(a_t / s_t).max()

                fc = _eval_at(c)
                fd = _eval_at(d)

                # 3 iterations of golden section
                for _ in range(3):
                    if fc < fd:
                        hi = d
                        d = c
                        fd = fc
                        c = hi * (1.0 - golden)
                        fc = _eval_at(c)
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + golden * (hi - lo)
                        d = min(d, hi)
                        fd = _eval_at(d)

                best_eps = (lo + hi) / 2.0
                fb = _eval_at(best_eps)
                if fc < fb:
                    best_eps = c
                    fb = fc
                if fd < fb:
                    best_eps = d
                    fb = fd

                if fb < best_trial_m - 1e-14:
                    best_trial_m = fb
                    a_trial = a.copy()
                    a_trial[i] -= best_eps
                    a_trial[dst] += best_eps
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    best_trial = a_trial.copy()

            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                sweep_improved = True

        if sweep_improved:
            stall = 0
        else:
            stall += 1
            # Perturb
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if p != q and a[p] > 1e-8:
                eps = a[p] * 0.02
                a[p] -= eps
                a[q] += eps
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = _fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0

        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD-Golden sweep={sweep} ({elapsed:.1f}s): J={J:.8f}, stall={stall}")

    return best_a, best_m


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global

    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")

        pass

    if best_seq is None:

    remaining = time_budget - (time.monotonic() - t_global)

                                             tol_frac_final=0.003, c_step=3.0, verbose=True)

    # === Phase 2: LP Frank-Wolfe (25% time) ===

                                             tol_frac_final=0.001, c_step=1.5, verbose=False)

                                                 tol_frac_final=0.002, c_step=2.5, verbose=True)

                a_in6 = _norm_sum1(best_seq)
                a_out, _ = _lp_frank_wolfe(a_in6, t6b, solver, max_constraints=200, verbose=False)
                try_update(a_out, f"LP_final_{len(a_out)}")

    # === Phase 7: Final equalization + CD at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        a_in = _norm_sum1(best_seq)

        t7a = min(remaining * 0.5, 80.0)
        if t7a > 15:
            a_out, _ = _equalization_polish(a_in, t7a, top_k=15, verbose=False)
            try_update(a_out, f"EQ_final_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t7b = min(remaining * 0.6, 50.0)
            if t7b > 10:
                a_in7 = _norm_sum1(best_seq)
                a_out, _ = _coordinate_descent_golden(a_in7, t7b, verbose=False, rng_seed=313)
                try_update(a_out, f"CD_final_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
