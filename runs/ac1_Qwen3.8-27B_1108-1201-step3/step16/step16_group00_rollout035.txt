def interpolate_sequence(f, new_n):

        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        if use_backtrack:
            # Backtracking line search
            eta = gamma / (np.sqrt(t) + 1.0)
            best_trial_m = fast_autocorr(y).max()
            for _ in range(5):
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new / s_new
                m_trial = fast_autocorr(a_new).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    # Accept this step
                    x_prev = x
                    x = a_new
                    break
                else:
                    eta *= 0.5
            else:
                # All backtracking failed, take the smallest step
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = y.copy()
                else:
                    a_new = a_new / s_new
                x_prev = x
                x = a_new
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   adaptive_margin=1e-4):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive constraint selection: include constraints within margin
        margins = M_curr - ac
        threshold = adaptive_margin * M_curr
        active_idx = np.where(margins <= threshold)[0]
        if len(active_idx) > top_K * 3:
            # Too many, use top-K
            active_idx = np.argsort(margins)[:top_K * 3]
        elif len(active_idx) < 10:
            active_idx = np.argsort(margins)[:top_K]

        # Build constraint matrix for active constraints
        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")

    return best_a, best_m


def peak_crush_coordinate(a0, time_limit, verbose=True, rng_seed=123):
    """Targeted coordinate descent focusing on the maximum peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Find contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        # Focus on top contributors
        n_top = min(10, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            src_partner = peak_k - src
            if src == src_partner:
                continue

            # Try moving mass from src to various destinations
            # Score destinations by how much they'd increase other peaks
            # Use a quick approximation: check the top-5 other peaks
            other_peaks = np.argsort(ac)[::-1][:5]
            other_peaks = [pk for pk in other_peaks if pk != peak_k]

            best_frac = None
            best_new_m = M_curr
            for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue

                # Try several random destinations
                n_trials = 20
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_frac = (src, dst, amt)

            if best_frac is not None:
                src_i, dst_i, amt = best_frac
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    """Apply a small perturbation in the frequency domain for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Perturb the phase (not the magnitude) to preserve roughly the same energy distribution
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_beta_family(n, alpha, beta):
    """Beta-family initialization: a[i] = x^alpha * (1-x)^beta."""
    x = np.linspace(0.01, 0.99, n)
    a = x ** alpha * (1 - x) ** beta
    return norm_sum1(a)


def make_parametric(n, family='sine', **kwargs):
    """Various parametric initializations."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_sym':
        return make_beta_family(n, kwargs.get('alpha', 1.0), kwargs.get('alpha', 1.0))
    elif family == 'beta_asym':
        return make_beta_family(n, kwargs.get('alpha', 0.5), kwargs.get('beta', 2.0))
    elif family == 'power':
        x = np.linspace(0.01, 1, n)
        p = kwargs.get('p', 1.0)
        return norm_sum1(x ** p)
    elif family == 'gaussian':
        x = np.linspace(0, 1, n)
        sigma = kwargs.get('sigma', 0.2)
        return norm_sum1(np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2)))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

**Approach: Direct optimization of the max using bisection on M**

We want to find the minimum M such that there exists a >= 0, sum(a) = 1, with (a*a)[k] <= M for all k.

One more thought: the previous approach used `top_k=50` for the soft subgradient. I'm using `top_k=60` with `beta=4.0`. Let me think about whether this is the right choice.

The soft subgradient approximates the subgradient of max_k c[k] by:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 5:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.8, 1e-5)

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def peak_crush_coordinate(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        n_top = min(15, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 30
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'beta_3_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 3 * (1 - x) ** 2)
    elif family == 'beta_2_3':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 3)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Extended AMD with backtracking ===
    print(f"\n=== Phase 1: AMD backtrack (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t1, top_k=60, beta=4.0, c0=3.5,
                                restart_every=4000, verbose=True, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_bt")

    # === Phase 2: LP Frank-Wolfe (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 450.0)
    if t2 > 40 and n_best <= 4500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=250,
                                   margin_frac=2e-3)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Peak-crushing coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak-crush (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush_coordinate(a_in, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "peakcrush1")

    # === Phase 4: Second LP with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30 and n_best <= 4500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=350,
                                   margin_frac=5e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 400, n_best + 800, n_best + 1500]))
    longer_lengths = [l for l in longer_lengths if l <= 5500 and remaining > 200]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 5: Extend to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.20, 200.0)
        if t5a > 30:
            a_out, _ = amd_optimize(a_long, t5a, top_k=60, beta=4.0, c0=3.0,
                                     restart_every=4000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 4000:
            t5b = min(remaining * 0.15, 150.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=True, top_K=200,
                                           margin_frac=2e-3)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 6: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Freq-diversify (n={n_best}) ===")
    rng = np.random.default_rng(99)
    t6_per = min(remaining * 0.08, 100.0)
    if t6_per > 30:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 30:
                break
            a_pert = freq_perturb(best_seq, strength=0.015, rng=rng)
            a_out, _ = amd_optimize(a_pert, t6_per, top_k=50, beta=5.0, c0=4.0,
                                     restart_every=2000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 7: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30 and n_best <= 4500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=400,
                                   margin_frac=2e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final peak-crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final peak-crush (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.30, 180.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_crush_coordinate(a_in, t8, verbose=True, rng_seed=555)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety: ensure we return before budget
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
