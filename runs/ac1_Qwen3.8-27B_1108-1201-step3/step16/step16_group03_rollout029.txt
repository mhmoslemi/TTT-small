def fast_conv(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(a, peak_indices=None):
    """Build the convolution matrix T where (T @ b)[k] = (a*b)[k].
    If peak_indices is given, only build those rows."""
    n = len(a)
    if peak_indices is None:
        idx = np.arange(2 * n - 1)
    else:
        idx = peak_indices
    T = np.zeros((len(idx), n))
    for row, k in enumerate(idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        T[row, js:je] = a[k - js:k - js + (je - js)]
    return T


def nesterov_md(a0, time_limit, verbose=False, seed=42, c0=5.0):
    """Nesterov accelerated mirror descent for minimizing max(conv(a,a)) on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 5000
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution and find peak
        ac = fast_conv(y)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))
        
        # Subgradient: 2 * B_k* * y (gradient of (a*a)[k*] at a=y)
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n, k_star + 1)
        for j in range(js, je):
            jj = k_star - j
            if 0 <= jj < n:
                grad[j] += 2.0 * y[jj]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic restart
        if stall > max_stall:
            x_r = best_a + rng.normal(0, 0.003, n) * best_a
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            stall = 0
            gamma = c0  # reset step size
        
        # Multi-peak subgradient every 5000 steps
        if t % 5000 == 0:
            top_k = np.argsort(ac)[::-1][:3]
            grad_multi = np.zeros(n)
            for k in top_k:
                k_i = int(k)
                js2 = max(0, k_i - n + 1)
                je2 = min(n, k_i + 1)
                for j in range(js2, je2):
                    jj = k_i - j
                    if 0 <= jj < n:
                        grad_multi[j] += 2.0 * y[jj]
            eta2 = gamma / (np.sqrt(t) + 1.0) * 0.5
            log_a2 = np.log(y) - eta2 * grad_multi
            max_log2 = log_a2.max()
            a_new2 = np.exp(log_a2 - max_log2)
            a_new2 = np.maximum(a_new2, 1e-30)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                M_chk = fast_conv(a_new2).max()
                if M_chk < best_m:
                    best_m = M_chk
                    best_a = a_new2.copy()
                    x = a_new2
                    x_prev = a_new2
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [NMD] t={t} ({elapsed:.0f}s): max_conv={best_m:.8f}, score={2*n*best_m:.6f}")
    
    return best_a, best_m


def lp_fw(a0, time_limit, solver, verbose=True, full_first=20, reduced_k=200, switch_to_full_at=0.8):
    """LP-based Frank-Wolfe with adaptive constraint set."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_conv(y).max()
    t0 = time.monotonic()
    
    iter_count = 0
    n_iters_planned = 500  # rough estimate
    
    b_var = cp.Variable(n, nonneg=True)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        use_full = (iter_count < full_first) or (elapsed_frac > switch_to_full_at)
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if use_full:
            T = build_conv_matrix(y)
        else:
            top_idx = np.argsort(ac)[::-1][:reduced_k]
            T = build_conv_matrix(y, top_idx)
        
        # LP: min t s.t. T @ b <= t, sum(b) = 1, b >= 0
        t_var = cp.Variable()
        constraints = [cp.sum(b_var) == 1, T @ b_var <= t_var]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            iter_count += 1
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            iter_count += 1
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            iter_count += 1
            continue
        direction = b_val / b_sum
        
        # FW step with standard step size
        alpha = 2.0 / (iter_count + 2.0)
        y_new = (1.0 - alpha) * y + alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        iter_count += 1
        
        # Track best
        M_new = fast_conv(y).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y.copy()
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [LP-FW] iter={iter_count} ({elapsed:.0f}s): "
                  f"max_conv={best_m:.8f}, score={2*n*best_m:.6f}, full={use_full}")
    
    return best_a, best_m

                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    j2 = k2i - dst
                    if 0 <= j2 < n:
                        sens += a[j2]
                if sens < best_score_dst:
                    best_score_dst = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            dst = best_dst
            
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_conv(a_t).max()
                if Mt < best_m - 1e-16:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 100:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    [Coord] ({elapsed:.0f}s): score={2*n*best_m:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = f / f.sum()  # ensure normalized
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, shape='uniform'):
    """Create initial sequence of given length and shape."""
    x = np.linspace(0, 1, n)
    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif shape == 'triangular':
        a = 1 - np.abs(2 * x - 1)
    elif shape == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif shape == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif shape == 'flat_top':
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s
