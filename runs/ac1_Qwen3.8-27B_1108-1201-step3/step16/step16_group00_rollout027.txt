This means coordinate descent on a single coordinate requires checking O(n) lag values, which is O(n) per coordinate, O(n^2) per full sweep. For n=3000, that's 9 million operations per sweep, which might be OK.

But finding the optimal a_i is the hard part. J(a_i) = 2n * max_k c_k(a_i) / (T + a_i)^2 where c_k(a_i) is piecewise linear in a_i.

This is a piecewise function that we can optimize by finding the breakpoints. But that's complex. A simpler approach: use a 1D line search (golden section or ternary search) for each coordinate.

Actually, for the coordinate descent, a simpler approach that worked well in the previous code was:
- Pick a "hot" coordinate (one that contributes to the peak)
- Move a small fraction of its mass to a "cold" coordinate
- Check if J improved

Let me design the algorithm:

```
Phase 1: Multi-initialization with mirror descent
  - Try several initializations (power ramps, bimodal, etc.)
  - Run mirror descent with Nesterov acceleration on each
  - Keep the best

Phase 2: Coordinate descent polish

  - Use sensitivity analysis to pick good moves

Phase 3: Length interpolation and re-optimization
  - Interpolate the best sequence to different lengths
  - Quick re-optimization at each length
  - Keep the overall best
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with FFT-based subgradients
2. Adaptive top-K subgradient selection
3. Coordinate descent with sensitivity-based mass transfer
4. Multi-length exploration and interpolation

import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def subgradient(a, conv, top_k=100):
    """Compute weighted subgradient of max(conv) at a."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    # Use exponential weighting to focus on the largest
    beta = 10.0
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # (conv)_k = sum_j a_j * a[k-j]
        # d(conv)_k / da_i = a[k-i] (if 0 <= k-i < n) + a[k-i] (if i != k-i and 0 <= k-i < n)
        # Actually: conv_k = sum_j a_j * a_{k-j}
        # d conv_k / da_i: 
        #   term j=i: a_i * a_{k-i} -> derivative is a_{k-i} (if 0<=k-i<n)
        #   term j=k-i: a_{k-i} * a_i -> derivative is a_{k-i} (if 0<=k-i<n and k-i != i)
        # So if k != 2i: d conv_k / da_i = 2 * a[k-i] (if 0<=k-i<n)
        #     if k = 2i: d conv_k / da_i = 2 * a[i] = 2*a[k-i] (same formula!)
        # Wait: if k=2i, then k-i=i, so the term is a_i * a_i = a_i^2, derivative = 2*a_i = 2*a[k-i].
        # If k != 2i and 0<=k-i<n: two terms a_i*a[k-i] and a[k-i]*a_i, derivative = 2*a[k-i].
        # If k-i < 0 or k-i >= n: no contribution.
        #
        # So in all cases: d conv_k / da_i = 2 * a[k-i] if 0 <= k-i < n, else 0.
        
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    
    return g

def mirror_descent(a0, time_limit, top_k=100, beta_w=10.0, verbose=False):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    
    x = a.copy()  # Nesterov extrapolation point
    x_prev = a.copy()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 5000
    
    c0 = 4.0  # initial step size parameter
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        conv = fast_conv(y)
        g = subgradient(y, conv, top_k=top_k)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent step (KL projection)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Evaluate
        val = eval_J(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic restart
        if stall > max_stall:
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r /= x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
            top_k = min(top_k + 20, 200)
            beta_w = max(beta_w - 2.0, 3.0)
            
            if verbose:
                print(f"    MD restart t={t}: J={eval_J(best_a):.6f}")
        
        if verbose and t % 5000 == 0:
            print(f"    MD t={t}: J={eval_J(best_a):.6f}, stall={stall}")
    
    return best_a, eval_J(best_a)

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with mass transfer from hot to cold positions."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        
        # Find hot positions (contributing to peak lags)
        n_top = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_top]
        
        # Compute "hotness" of each position
        hotness = np.zeros(n)
        for k in top_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                contrib = a[idx] * a[k_int - idx]
                hotness[idx] += contrib / (M + 1e-30)
        
        # Find the hottest position
        src = int(np.argmax(hotness))
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Find cold positions (low contribution to peaks)
        # Score each position by how much it contributes to peak lags
        scores = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    scores[i] += a[j]
        
        # Find best destination: low score, far from source
        candidate_scores = scores.copy()
        candidate_scores[src] = np.inf
        dst = int(np.argmin(candidate_scores))
        
        if dst == src:
            stall += 1
            continue
        
        # Try to move mass from src to dst
        found = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
            J_trial = eval_J(a_trial)
            if J_trial < best_J - 1e-16:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                found = True
                stall = 0
                break
        
        if not found:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    J_new = eval_J(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_J

def make_power_ramp(n, p=1.0):
    """Create a power-law ramp: a_i proportional to (i+1)^p."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_bimodal(n):
    """Create a bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-100 * (x - 0.15) ** 2) + np.exp(-100 * (x - 0.85) ** 2) + 0.01
    return a / a.sum()

def make_smooth_random(n, seed=0):
    """Create a smooth random function."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()

def interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with FFT-based subgradients and adaptive top-K
2. Multi-strategy initialization (power ramps, bimodal, smooth random, parent)
3. Coordinate descent with pressure-based mass transfer
4. Coarse-to-fine length sweeping
5. Periodic restarts with adaptive parameter tuning

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 assuming a is normalized to sum=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def compute_subgradient(a, conv, top_k=80, beta_w=10.0):
    """Compute soft subgradient of max(conv) at a using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent(a0, time_limit, top_k=80, beta_w=10.0, c0=4.0,
                   restart_every=6000, verbose=False):
    """Nesterov-accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Subgradient
        conv = fast_conv(y)
        g = compute_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        # Step size (Polylak-type)
        eta = c0 / (np.sqrt(t) + 1.0)

        # Mirror step (KL projection onto simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-16:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart logic
        if stall > restart_every:
            stall = 0
            # Geometric perturbation
            scale = rng.uniform(0.005, 0.03)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 15, 200)
            cur_beta = max(cur_beta - 1.0, 3.0)
            c0 = max(c0 * 0.9, 1.0)
            if verbose:
                print(f"    MD restart t={t}: J={best_val:.6f}, k={cur_top_k}, b={cur_beta:.1f}")

        if verbose and t % 10000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with pressure-based mass transfer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Compute position pressures from top lags
        n_lags = min(40, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        pressure = np.zeros(n)
        for k in top_lags[:15]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        # Find source (highest pressure with significant mass)
        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        # Find destination: minimize impact on peaks
        dest_scores = np.zeros(n)
        for k in top_lags[:8]:
            k_i = int(k)
            for i in range(n):
                j = k_i - i
                if 0 <= j < n:
                    dest_scores[i] += a[j]

        dest_scores[src] = np.inf
        # Also penalize positions that are already hot
        dest_scores += 0.1 * pressure

        dst = int(np.argmin(dest_scores))

        # Try multiple transfer fractions
        improved = False
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-15:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-16:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 30 == 0:
                # Random perturbation to escape
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.02)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 100 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp: a_i proportional to (i+1)^p."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 10):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric_ramp(n, p=1.0, shift=0.3):
    """Asymmetric ramp with peak shifted."""
    x = np.linspace(0, 1, n)
    a = ((x + shift / (1 - 2 * shift)) ** p) * ((1 - x) ** 0.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

        result = [float(x) for x in best_a]
        return result
    else:
        # Fallback
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
