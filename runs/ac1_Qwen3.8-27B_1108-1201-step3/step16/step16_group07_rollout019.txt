def normalize_to(a, n):

def build_toeplitz_row(a, k, n):
    """Build the k-th row of the convolution matrix: (a * b)_k = sum_j a[j] * b[k-j] (with valid indices)."""
    # (conv(a,b))[k] = sum_j a[j] * b[k-j] where 0<=j<n and 0<=k-j<n
    # So b[k-j] is involved when j is in [max(0, k-n+1), min(k, n-1)]
    # Let i = k-j, so i is in [max(0, k-n+1), min(k, n-1)] reversed... 
    # Actually: (a*b)[k] = sum_{j} a[j] * b[k-j]
    # For this to be valid: 0 <= j < n and 0 <= k-j < n
    # So: max(0, k-n+1) <= j <= min(k, n-1)
    # The coefficient of b[i] (where i = k-j) is a[k-i]
    # So the row has entries a[k-i] for i in [max(0,k-n+1), min(k, n-1)]
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return np.zeros(n)
    row = np.zeros(n)
    # For i in [lo, hi], coefficient is a[k-i]
    idx = np.arange(lo, hi + 1)
    row[lo:hi+1] = a[k - idx]
    return row


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    if size == 0:
        return np.ones(n) / n, None
    
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            i_idx = np.arange(lo, hi + 1)
            T[idx, lo:hi+1] = a_fixed[k - i_idx]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def compute_dual_gradient(a, duals, active_set, n):
    """Compute gradient from dual variables: grad_j = sum_k lambda_k * a[k-j] (for valid k-j)."""
    grad = np.zeros(n)
    for w, k in zip(duals, active_set):
        if w <= 0:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            i_idx = np.arange(lo, hi + 1)
            grad[lo:hi+1] += w * a[k - i_idx]
    return grad

        y = y_new

        # Track best
        score_y = float(fast_autocorr(y).max())
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = float(fast_autocorr(a).max())
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Dual-based refinement (every 100 iterations)
        if t % 100 == 0 and duals is not None and len(duals) > 0:
            duals_clipped = np.maximum(duals, 0.0)
            if duals_clipped.sum() > 1e-15:
                duals_norm = duals_clipped / duals_clipped.sum()
                grad = compute_dual_gradient(y, duals_norm, active_set, n)
                # Apply a small step in the dual gradient direction
                support = np.where(y > 0.001 * y.max())[0]
                if len(support) > 1:
                    grad_centered = grad - np.mean(grad[support])
                    step_size = 0.01 * norm / max(np.linalg.norm(grad_centered), 1e-15)
                    a_d = a - step_size * grad_centered
                    a_d = np.maximum(a_d, 0.0)
                    s_d = a_d.sum()
                    if s_d > 1e-15:
                        a_d = a_d * norm / s_d
                        score_d = float(fast_autocorr(a_d).max())
                        if score_d < best_score:
                            best_score = score_d
                            best_a = a_d.copy()
                            a = a_d.copy()
                            stall_count = 0
        
        # Restart if stalled
        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None  # Force rebuild
            stall_count = 0
        
        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def spectral_smoothing(a0, time_limit, seed=0, verbose=False):
    """
    Spectral domain optimization: smooth the power spectrum to reduce autocorrelation peaks.
    The idea: sharp features in the time domain correspond to broad frequency content.
    By smoothing in frequency, we get a smoother time-domain signal.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Strategy: iteratively modify the spectrum
    # The autocorrelation is IFFT(|A|^2). We want IFFT(|A|^2) to be flat (except at k=0).
    # Equivalently, we want |A|^2 to be as "concentrated" as possible (narrow in time) 
    # which means... hmm, this is circular.
    
    # Alternative: use the fact that (a*a)_k = IFFT(|A|^2)_k
    # We want max_k (a*a)_k to be small.
    # The DC component is fixed: (a*a)_0 = sum a_j^2 = (1/n) sum |A[m]|^2 (by Parseval)
    # Actually (a*a)_0 = sum a_j^2, and by Parseval sum|A[m]|^2 = n*sum a_j^2
    # So (a*a)_0 = (1/n) * sum|A[m]|^2
    
    # Key: we want to minimize max over k of IFFT(|A|^2)_k
    # This is equivalent to making IFFT(|A|^2) as flat as possible
    # IFFT(|A|^2) is the autocorrelation, and its sum is sum_k (a*a)_k = (a*a)_0 * n ... no
    # sum_k (a*a)_k = sum_k sum_j a_j a_{j+k} = sum_j a_j sum_k a_{j+k} = sum_j a_j * sum(a) = (sum a)^2
    # So sum of autocorrelation = (sum a)^2 = 2n (in our normalization)
    # If it were perfectly flat: each value = 2n/(2n-1) ≈ 1
    # The peak (a*a)_0 = sum a_j^2 must be at least (sum a)^2/n = 2n/n = 2 (by Cauchy-Schwarz)
    # So the best possible peak is 2, giving score = 2*n*2/(2n) = 2... wait that's the uniform case.
    # Hmm, I'm confusing myself. Let me just do the optimization directly.
    
    # Approach: iterative proportional fitting (IPF) style
    # 1. Compute autocorrelation
    # 2. Identify peaks
    # 3. Modify the spectrum to reduce those peaks
    # 4. Project back to non-negative time domain
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Get spectrum
        fft_size = 1 << (2 * n - 1 + 1).bit_length()
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # The autocorrelation is IFFT(power)[:2n-1]
        # We want to reduce peaks. 
        # Strategy: where autocorrelation is high, the corresponding frequency components
        # should be reduced. Where it's low, they can be increased.
        
        # Simple approach: apply a "water-filling" on the autocorrelation
        # Redistribute power from high peaks to low ones
        target = np.ones(2 * n - 1) * (2.0 * n) / (2 * n - 1)  # Flat target
        # But (a*a)_0 is always >= 2 (by C-S), so we can't make it flat
        # Instead, minimize the excess above the mean
        mean_conv = conv.mean()
        excess = np.maximum(conv - mean_conv, 0)
        
        # Gradient of max(conv) w.r.t. a is the subgradient at the argmax
        k_max = np.argmax(conv)
        
        # Subgradient of (a*a)_{k_max} w.r.t. a:
        # d/d(a[j]) (a*a)_{k_max} = a[k_max - j] + a[j - k_max] (for valid indices, times 2 if same)
        # Actually: (a*a)_k = sum_j a_j * a_{j+k}
        # d/d(a_j) = a_{j+k} + a_{j-k} (with valid index handling)
        # Wait: (a*a)_k = sum_i a_i * a_{i+k}
        # d/d(a_j) = a_{j+k} (from i=j term) + a_{j-k} (from i+k=j term, i.e., i=j-k)
        # So subgrad_j = a_{j+k} + a_{j-k} (where defined)
        
        k = k_max
        subgrad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j + k < n:
                val += a[j + k]
            if 0 <= j - k < n:
                val += a[j - k]
            # For k=0: d/d(a_j) sum_i a_i^2 = 2*a_j
            if k == 0:
                val = 2.0 * a[j]
            subgrad[j] = val
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-15:
            break
        subgrad = subgrad / sg_norm
        
        # Step size: decrease over time
        step = 0.3 / (1.0 + 0.01 * it)
        
        # Update
        a_new = a - step * subgrad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new
        
        score_new = float(fast_autocorr(a_new).max())
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a smaller step
            a_try = a - step * 0.1 * subgrad
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try * norm / s_try
                score_try = float(fast_autocorr(a_try).max())
                if score_try < best_score:
                    best_score = score_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    step *= 0.9
                    if step < 1e-12:
                        break
    
    return best_a, best_score


def block_coordinate_descent(a0, time_limit, seed=0, verbose=False):
    """
    Block coordinate descent: update blocks of B coordinates at a time.
    For each block, solve a small-scale optimization to minimize the max autocorrelation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    block_size = max(8, n // 200)  # ~20 blocks
    n_blocks = n // block_size
    no_improve = 0
    max_no_improve = 3
    
    for sweep in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve >= max_no_improve:
            break
        
        improved = False
        conv = fast_autocorr(a)
        
        # Randomize block order
        block_order = rng.permutation(n_blocks)
        
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            start = bi * block_size
            end = min(start + block_size, n)
            block_indices = np.arange(start, end)
            
            # For this block, try to find better values
            # The change in autocorrelation when we change block[b] to new_val:
            # For each k, (a*a)_k changes based on the overlap of the block with position k
            
            # Simple approach: try a few perturbation directions
            current_block = a[start:end].copy()
            best_block = current_block.copy()
            best_block_score = conv.max()
            
            # Generate candidate blocks
            candidates = [current_block.copy()]
            
            # Smooth version
            if end - start > 1:
                smoothed = np.convolve(current_block, np.ones(3)/3.0, mode='same')
                candidates.append(np.maximum(smoothed, 0))
            
            # Shifted version (if not at boundary)
            shift = max(1, (end - start) // 4)
            if shift < end - start:
                shifted = np.roll(current_block, shift, axis=0)
                candidates.append(np.maximum(shifted, 0))
            
            # Random perturbation
            noise = rng.standard_normal(end - start) * 0.1 * current_block.std()
            candidates.append(np.maximum(current_block + noise, 0))
            
            # Gradient-based candidate
            # Subgradient of max autocorrelation
            k_max = np.argmax(conv)
            sg = np.zeros(end - start)
            for j in range(end - start):
                idx = start + j
                val = 0.0
                if 0 <= idx + k_max < n:
                    val += a[idx + k_max]
                if 0 <= idx - k_max < n:
                    val += a[idx - k_max]
                if k_max == 0:
                    val = 2.0 * a[idx]
                sg[j] = val
            
            if np.linalg.norm(sg) > 1e-15:
                grad_dir = sg / np.linalg.norm(sg)
                candidates.append(np.maximum(current_block - 0.05 * current_block.std() * grad_dir, 0))
                candidates.append(np.maximum(current_block + 0.05 * current_block.std() * grad_dir, 0))
            
            # Evaluate each candidate
            for cand in candidates:
                if cand.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[start:end] = cand
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_trial = a_trial * norm / s_trial
                score_trial = float(fast_autocorr(a_trial).max())
                if score_trial < best_block_score:
                    best_block_score = score_trial
                    best_block = cand.copy()
            
            if np.any(np.abs(best_block - current_block) > 1e-10):
                a[start:end] = best_block
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = float(fast_autocorr(a).max())
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            
        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD sweep={sweep} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def fine_tune_cd(a0, time_limit, seed=0, verbose=False):
    """
    Fine-tuning coordinate descent with careful step selection.
    Focuses on the coordinates that most affect the peak.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5
    
    for sweep in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve >= max_no_improve:
            break
        
        improved = False
        conv = fast_autocorr(a)
        
        # Prioritize coordinates near the peak contribution
        k_max = np.argmax(conv)
        # Coordinates that most affect (a*a)_{k_max} are those where a[j+k_max] or a[j-k_max] is large
        importance = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j + k_max < n:
                val += a[j + k_max]
            if 0 <= j - k_max < n:
                val += a[j - k_max]
            if k_max == 0:
                val = 2.0 * a[j]
            importance[j] = val
        
        # Sort by importance (most important first) but add some randomness
        order = np.argsort(-importance + 0.1 * importance.std() * rng.standard_normal(n))
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            if aj < 1e-10:
                continue
                
            s_others = norm - aj
            
            # Try various values
            typical = norm / n
            candidates = [aj * 0.7, aj * 0.85, aj * 1.0, aj * 1.15, aj * 1.3, 0.0, typical]
            candidates = list(set([max(0.0, min(c, 10.0 * typical)) for c in candidates]))
            
            best_x = aj
            best_obj = conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                lo = max(0, j)
                hi = min(size, j + n)
                new_conv[lo:hi] += 2.0 * delta * a[max(0, lo-j):hi-j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                
                scale = norm / (s_others + x) if (s_others + x) > 1e-15 else 1.0
                new_max = new_conv.max() * scale * scale
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = float(fast_autocorr(a).max())
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            
        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def make_structured_init(n, seed=42):
    """Create a structured initialization using incommensurate frequencies."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    # Use golden ratio for incommensurability
    phi = (1 + np.sqrt(5)) / 2
    a = np.ones(n) * 0.3
    
    # Add several frequency components
    n_freqs = 8
    for i in range(n_freqs):
        freq = (i + 1) * phi % 1.0  # Incommensurate
        freq = max(0.1, freq)  # Avoid DC
        amplitude = 0.15 / (i + 1)
        phase = rng.random() * 2 * np.pi
        a += amplitude * np.cos(2 * np.pi * freq * n * x + phase)
    
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * 0.5
    a = a / a.sum()
    return a

                best_seq = a_out.copy()
                print(f"  *** Phase 7 improved: {best_score:.8f}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
