    top_vals = c[top_idx]

    p = p / p.sum()
    
    # Compute subgradient: g_j = 2 * sum_k p_k * a_{k-j}
    g = np.zeros(n)

        k_int = int(k)
        # a_{k-j} is nonzero when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k_int - n + 1)

        # a_{k-j} = a[k-j]
        for j in range(j_start, j_end):

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, top_k=20, beta=10.0):
    """Compute a subgradient of max(conv(a,a)) using softmax over top peaks.
    Returns (gradient, max_val, top_indices)."""
    n = len(a)
    c = fast_autocorr(a)
    M = c.max()
    
    if M < 1e-30:
        return np.zeros(n), M, np.array([0])
    
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    
    # Softmax weights over top peaks
    p = np.exp(beta * (top_vals - top_vals[0]))
    p = p / p.sum()
    
    # Compute subgradient: g_j = 2 * sum_k p_k * a_{k-j}
    g = np.zeros(n)
    for pk, k in zip(p, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        j_range = np.arange(j_start, j_end)
        a_vals = a[k_int - j_range]
        g[j_start:j_end] += 2.0 * pk * a_vals
    
    return g, M, top_idx

def lbfgsb_refine(a0, time_limit, verbose=True):
    """L-BFGS-B refinement using softmax parameterization."""
    n = len(a0)
    a_init = norm_sum1(a0)
    best_a = a_init.copy()
    best_m = fast_autocorr(a_init).max()
    t0 = time.monotonic()
    
    # Parameterize: a = softmax(z), start from z = log(a_init)
    z0 = np.log(a_init + 1e-30)
    
    def objective(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_autocorr(a)
        return 2.0 * n * c.max()
    
    def gradient(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_autocorr(a)
        k_star = int(np.argmax(c))
        # Gradient of max(conv) wrt a: 2 * a_{k*-j}
        grad_a = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        j_range = np.arange(j_start, j_end)
        grad_a[j_start:j_end] = 2.0 * a[k_star - j_range]
        
        # Chain rule: d/dz
        # da_i/dz_j = a_i * (delta_ij - a_j)
        # df/dz_j = sum_i df/da_i * a_i * (delta_ij - a_j)
        # = a_j * (df/da_j - sum_i df/da_i * a_i)
        df_da_dot_a = np.dot(grad_a, a)
        grad_z = a * (grad_a - df_da_dot_a)
        return 2.0 * n * grad_z
    
    result = minimize(objective, z0, method='L-BFGS-B',
                      jac=gradient,
                      options={'maxiter': 500, 'maxfun': 2000,
                               'ftol': 1e-12, 'gtol': 1e-8})
    
    z_opt = result.x
    a_opt = np.exp(z_opt - z_opt.max())
    a_opt = a_opt / a_opt.sum()
    a_opt = np.maximum(a_opt, 0)
    
    M_opt = fast_autocorr(a_opt).max()
    if M_opt < best_m:
        best_m = M_opt
        best_a = a_opt.copy()
    
    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    L-BFGS-B ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

                break

            contrib = a[j_range] * a[pk_int - j_range]
            
            if contrib.max() < 1e-30:

            for _ in range(3):
                src = j_range[np.argmax(contrib)]

                sens = np.zeros(n)

                    for pos in range(ts, te + 1):
                        sens[pos] += a[tp_int - pos]

                candidates = np.argsort(sens)[:min(100, n // 5)]
                
                for dst in candidates[:20]:
                    if dst == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-18:

                        a_trial = a.copy()

                        a_trial[dst] += amt

                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-18:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                
                if improved:
                    break
                # Update contrib for next attempt
                contrib = a[j_range] * a[pk_int - j_range]
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()

    if verbose:

        print(f"    PeakBal ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diverse_init(n, seed=0):
    """Generate a diverse initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    # Try different shapes
    shapes = [
        np.sin(np.pi * x) ** 2,
        np.sin(np.pi * x) ** 4,
        1.0 + 0.3 * np.cos(2 * np.pi * x),
        np.exp(-((x - 0.5) ** 2) / 0.05),
        0.5 + 0.5 * np.abs(np.sin(np.pi * x * 3)),
        np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.1),
    ]
    
    idx = seed % len(shapes)
    f = shapes[idx].astype(np.float64)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f


def lp_fw_single_step(a, solver_name):
    """One LP-FW step using cvxpy."""
    import cvxpy as cp
    n = len(a)
    a = norm_sum1(a)
    c = fast_autocorr(a)
    M = c.max()
    
    if M < 1e-30:
        return a
    
    # Build constraint matrix T: T[k,j] = a[j] if j <= k <= j+n-1
    # T is (2n-1) x n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a[j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        return a
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return a
    
    return b_val / b_sum


def get_solver():
    """Get a working LP solver."""
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining fast FW, L-BFGS-B, coordinate descent,
    and multi-n exploration.
    """
    global _global_best_seq, _global_best_score, _global_t_start
    
    _global_t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    
    best_seq = None
    best_score = float('inf')
    
    solver = None
    try:
        solver = get_solver()
    except Exception:
        pass
    
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - _global_t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # === Initialization ===
    # Try parent first
    init_a = None
    n_choice = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            init_a = p / p.sum()
            n_choice = len(init_a)
            p_score = evaluate_safe(init_a)
            best_seq = init_a.copy()
            best_score = p_score
            print(f"Parent: n={n_choice}, score={p_score:.6f}")
    except Exception:
        pass
    
    if init_a is None:
        n_choice = 2400
        init_a = np.sin(np.pi * np.linspace(0, 1, n_choice)) ** 2
        init_a = init_a / init_a.sum()
        best_seq = init_a.copy()
        best_score = evaluate_safe(init_a)
        print(f"Default init: n={n_choice}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Fast FW (main workhorse, 45% of time) ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 60:
        t1 = min(remaining * 0.45, 700.0)
        print(f"\n=== Phase 1: Fast FW n={n0} (t={t1:.0f}s) ===")
        a_out, _ = fast_fw(norm_sum1(best_seq), t1, verbose=True)
        try_update(a_out, f"fastfw_n{n0}")
    
    # === Phase 2: L-BFGS-B refinement ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 40:
        t2 = min(remaining * 0.15, 200.0)
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: L-BFGS-B n={n_cur} (t={t2:.0f}s) ===")
        a_out, _ = lbfgsb_refine(norm_sum1(best_seq), t2, verbose=True)
        try_update(a_out, f"lbfgs_n{n_cur}")
    
    # === Phase 3: Peak-balancing CD ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 30:
        t3 = min(remaining * 0.12, 150.0)
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: PeakBal CD n={n_cur} (t={t3:.0f}s) ===")
        a_out, _ = peak_balance_cd(norm_sum1(best_seq), t3, verbose=True)
        try_update(a_out, f"peakbal_n{n_cur}")
    
    # === Phase 4: Fast FW round 2 (with different hyperparams) ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 60:
        t4 = min(remaining * 0.2, 250.0)
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Fast FW round 2 n={n_cur} (t={t4:.0f}s) ===")
        a_out, _ = fast_fw(norm_sum1(best_seq), t4, verbose=True)
        try_update(a_out, f"fastfw2_n{n_cur}")
    
    # === Phase 5: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: Multi-n exploration ===")
        test_ns = []
        for delta in [200, 400, -200, -400, 600]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)
        
        for nt in test_ns[:4]:
            remaining = time_budget - (time.monotonic() - _global_t_start)
            if remaining < 40:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.2, 100.0)
            a_out, _ = fast_fw(norm_sum1(a_inter), t_per, verbose=False)
            try_update(a_out, f"multin_n{nt}")
    
    # === Phase 6: LP-FW for a few iterations if solver available ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 30 and solver is not None:
        n_cur = len(best_seq)
        print(f"\n=== Phase 6: LP-FW steps n={n_cur} ===")
        a_cur = norm_sum1(best_seq).copy()
        t6 = min(remaining * 0.15, 120.0)
        t0 = time.monotonic()
        best_lp_a = a_cur.copy()
        best_lp_m = fast_autocorr(a_cur).max()
        y = a_cur.copy()
        
        while time.monotonic() - t0 < t6:
            b_dir = lp_fw_single_step(y, solver)
            if b_dir is None:
                break
            alpha = 2.0 / (len([1]) + 2.0)  # fixed small step
            alpha = 0.1
            y_new = (1 - alpha) * y + alpha * b_dir
            y_new = np.maximum(y_new, 0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new / sy
            y = y_new
            M_new = fast_autocorr(y).max()
            if M_new < best_lp_m:
                best_lp_m = M_new
                best_lp_a = y.copy()
        
        try_update(best_lp_a, f"lpfw_n{n_cur}")
    
    # === Phase 7: Final L-BFGS-B ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 30:
        n_cur = len(best_seq)
        t7 = min(remaining * 0.2, 100.0)
        print(f"\n=== Phase 7: Final L-BFGS-B n={n_cur} ===")
        a_out, _ = lbfgsb_refine(norm_sum1(best_seq), t7, verbose=True)
        try_update(a_out, f"final_lbfgs_n{n_cur}")
    
    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - _global_t_start)
    if remaining > 20:
        n_cur = len(best_seq)
        t8 = min(remaining * 0.3, 60.0)
        print(f"\n=== Phase 8: Final polish n={n_cur} ===")
        a_out, _ = peak_balance_cd(norm_sum1(best_seq), t8, verbose=False)
        try_update(a_out, f"final_polish_n{n_cur}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-_global_t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
