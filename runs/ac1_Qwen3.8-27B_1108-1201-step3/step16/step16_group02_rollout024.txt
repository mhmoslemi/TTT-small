def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgradient_topk(a, ac, top_k, beta):
    """Compute subgradient using top-k lags with power weighting.
    grad[j] = sum_k w_k * (a[k-j] + a[j-k]) where a[.]=0 outside [0,n-1]
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        if wi < 1e-12:
            continue
        # First term: a[k-j] for j in [max(0,k-n+1), min(k,n-1)]
        j_start1 = max(0, k_int - n + 1)
        j_end1 = min(k_int, n - 1) + 1
        if j_start1 < j_end1:
            j_vals = np.arange(j_start1, j_end1)
            grad[j_vals] += wi * a[k_int - j_vals]
        # Second term: a[j-k] for j in [max(k,0), min(k+n-1, n-1)]
        j_start2 = max(k_int, 0)
        j_end2 = min(k_int + n - 1, n - 1) + 1
        if j_start2 < j_end2:
            j_vals = np.arange(j_start2, j_end2)
            grad[j_vals] += wi * a[j_vals - k_int]
    return grad

def compute_subgradient_single_max(a, ac):

    n = len(a)

def mirror_descent_step(a, grad, eta):
    """Mirror descent step with entropy mirror."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=200, verbose=True):

    n = len(a0)

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Select active constraints
        threshold = M_curr * 0.95
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            j_start1 = max(0, k_int - n + 1)
            j_end1 = min(k_int, n - 1)
            if j_start1 <= j_end1:
                j_vals = np.arange(j_start1, j_end1 + 1)
                T[row_idx, j_vals] += y[k_int - j_vals]
            j_start2 = max(k_int, 0)
            j_end2 = min(k_int + n - 1, n - 1)
            if j_start2 <= j_end2:
                j_vals = np.arange(j_start2, j_end2 + 1)
                T[row_idx, j_vals] += y[j_vals - k_int]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (iter_count + 2.0)

            y_new = y.copy()
        else:

        else:

            if sx < 1e-15:

        x = x_new

        if mx < best_m:

            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue

            # Try moving mass from i to j
            for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2025)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label, n_override=None):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            n_show = n_override if n_override else len(a)
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={n_show}, t={elapsed:.0f}s)")

    # Load parent sequence
    n_base = 2400
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Subgradient Mirror Descent on parent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 1: SMD parent n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, _ = subgrad_mirror_descent(best_seq, t1, verbose=True, rng_seed=1,
                                          init_eta_scale=0.12, decay_rate=0.9987)
        try_update(a_out, "SMD_parent")

        # Second run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.10, 150.0)
        if t1b > 30:
            a_in = norm_sum1(best_seq)
            a_out2, _ = subgrad_mirror_descent(a_in, t1b, verbose=False, rng_seed=7,
                                               init_eta_scale=0.20, decay_rate=0.9990)
            try_update(a_out2, "SMD_parent_v2")

    # === Phase 2: LP Frank-Wolfe (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP-FW n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, max_constraints=250, verbose=True)
        try_update(a_out, "LP_main")

    # === Phase 3: SMD polish after LP (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: SMD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgrad_mirror_descent(a_in, t3, verbose=True, rng_seed=99,
                                          init_eta_scale=0.08, decay_rate=0.9992)
        try_update(a_out, "SMD_polish")

    # === Phase 4: Try alternative lengths (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Alt lengths (budget={remaining:.0f}s) ===")
    if remaining > 100:
        n_alt_list = [n_base - 200, n_base + 200, n_base + 400]
        time_per_alt = remaining / (len(n_alt_list) + 1)

        for n_alt in n_alt_list:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            if n_alt <= 500 or n_alt > 5000:
                continue
            t_alt = min(time_per_alt, 200.0)
            if t_alt < 30:
                continue
            print(f"  Trying n={n_alt} (budget={t_alt:.0f}s)")
            a_alt = interpolate_to_length(best_seq, n_alt)
            a_out, _ = subgrad_mirror_descent(a_alt, t_alt, verbose=False, rng_seed=200,
                                              init_eta_scale=0.12, decay_rate=0.9987)
            try_update(a_out, f"SMD_n{n_alt}", n_override=n_alt)

    # === Phase 5: Coordinate descent polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.5, 100.0)
    print(f"\n=== Phase 5: CD polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=555)
        try_update(a_out, "CD_polish")

    # === Phase 6: Final LP at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.6, 120.0)
    print(f"\n=== Phase 6: Final LP n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, max_constraints=150, verbose=False)
        try_update(a_out, "LP_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
