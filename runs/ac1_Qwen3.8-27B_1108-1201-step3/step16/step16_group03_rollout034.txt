            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def subgradient_at_peak(y, k):
    """Exact subgradient of c_k = sum_j y_j * y_{k-j} w.r.t. y."""
    n = len(y)
    k = int(k)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * y[j]
    return grad


def smooth_max_subgradient(y, ac, temperature):
    """
    Compute subgradient using smooth max approximation.
    temperature controls the sharpness: lower = harder max.
    Returns gradient of the smooth max of ac.
    """
    n = len(y)
    max_val = ac.max()
    # Softmax weights over the convolution values
    shifted = (ac - max_val) / temperature
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback to hard max
        top_k = int(np.argmax(ac))
        return subgradient_at_peak(y, top_k)
    w = w / w_sum
    # Only use top entries for efficiency
    top_indices = np.where(w > 1e-6)[0]
    if len(top_indices) == 0:
        top_indices = np.array([int(np.argmax(ac))])
    grad = np.zeros(n)
    for idx in top_indices:
        grad += w[idx] * subgradient_at_peak(y, int(idx))
    return grad


def nesterov_smooth(a0, time_limit, verbose=True, seed=42,
                     temp_init=0.01, temp_final=0.001, temp_decay_rate=0.9995,
                     c0=5.0):
    """
    Nesterov accelerated mirror descent with adaptive-temperature smooth max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 5000
    temperature = temp_init
    best_temp = temp_init

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: cool down over time, but reset if stuck
        temperature *= temp_decay_rate
        if temperature < temp_final:
            temperature = temp_final

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute convolution and subgradient
        ac = fast_conv(y)
        g = smooth_max_subgradient(y, ac, temperature)

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate Nesterov iterate too
        y_mom = x + beta_t * (x - x_prev)
        y_mom = np.maximum(y_mom, 1e-15)
        sy_mom = y_mom.sum()
        if sy_mom > 1e-15:
            y_mom = y_mom / sy_mom
            M_mom = fast_conv(y_mom).max()
            if M_mom < best_m - 1e-15:
                best_m = M_mom
                best_a = y_mom.copy()
                stall = 0
                continue

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with structured perturbation
        if stall > max_stall:
            # Try different perturbation strategies
            strategy = rng.integers(4)
            if strategy == 0:
                # Gaussian perturbation proportional to current values
                x_restarted = best_a * (1.0 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                # Swap-based: randomly swap pairs of values
                x_restarted = best_a.copy()
                for _ in range(n // 50):
                    i, j = rng.integers(0, n, 2)
                    if i != j:
                        x_restarted[i], x_restarted[j] = x_restarted[j], x_restarted[i]
            elif strategy == 2:
                # Block swap: swap two contiguous blocks
                x_restarted = best_a.copy()
                block_len = max(1, n // 20)
                i = rng.integers(0, n - block_len)
                j = rng.integers(0, n - block_len)
                if abs(i - j) > block_len:
                    tmp = x_restarted[i:i+block_len].copy()
                    x_restarted[i:i+block_len] = x_restarted[j:j+block_len]
                    x_restarted[j:j+block_len] = tmp
            else:
                # Exponential tilt
                tilt = rng.normal(0, 0.1)
                x_restarted = best_a * np.exp(tilt * np.linspace(-1, 1, n))

            x_restarted = np.maximum(x_restarted, 1e-10)
            s_r = x_restarted.sum()
            if s_r > 1e-15:
                x_restarted = x_restarted / s_r
            x = x_restarted
            x_prev = x_restarted
            stall = 0
            temperature = max(temperature * 2, temp_init * 0.5)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temperature:.5f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe_lazy(a0, time_limit, solver, verbose=True, top_K=15):
    """
    LP-based Frank-Wolfe with lazy constraints (only top-K active).
    Much faster per iteration than full LP.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    iter_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Get top-K active constraints
        top_idx = np.argsort(ac)[::-1][:top_K]
        top_idx = np.sort(top_idx)  # Sort for consistency

        # Build constraint matrix T for only these K constraints
        # T[k, j] = y[j] if k-j is in [0, n-1], else contribution from y[k-j]
        # (T @ b)[k] = sum_j b[j] * y[k-j] = (b * y)[k]
        T = np.zeros((len(top_idx), n))
        for ci, k in enumerate(top_idx):
            k_int = int(k)
            for j in range(n):
                jj = k_int - j
                if 0 <= jj < n:
                    T[ci, j] = y[jj]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Line search with golden section
        best_alpha = 0.0
        best_eval = M_curr

        # Try a grid of alphas
        alphas = np.linspace(0.0, 1.0, 20)
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        if best_alpha < 1e-10:
            # No improvement; try larger top_K or break
            if top_K < 40:
                top_K = min(top_K + 5, 40)
            iter_count += 1
            if iter_count % 50 == 0:
                break
            continue

        y = (1.0 - best_alpha) * y + best_alpha * direction
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-lazy iter={t} ({elapsed:.1f}s): J={J:.6f}, K={top_K}")

    return best_a, best_m


def coordinate_polish_aggressive(a0, time_limit, verbose=True, seed=42):
    """
    Aggressive coordinate descent with paired moves and fine epsilon search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    epsilons = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        # Get top peaks
        top_peaks = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        # Sample coordinate pairs: mix of peak-contributing and random
        n_pairs = min(300, n * 2)
        pairs = []

        # Pairs from top peaks
        for k in top_peaks[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) < 2:
                continue
            contrib = a[idx] * a[k_int - idx]
            top_src = idx[np.argmax(contrib)]
            # Find destination outside peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) > 0:
                dst = candidates[rng.integers(len(candidates))]
                if dst != top_src:
                    pairs.append((int(top_src), int(dst)))

        # Random pairs
        while len(pairs) < n_pairs:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                pairs.append((i, j))

        # Try each pair
        for (src, dst) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Find optimal epsilon by checking where the max peak is reduced
            # Try moving epsilon from src to dst
            best_frac = 0
            best_M = best_m

            for frac in epsilons:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_frac = frac

            if best_frac > 0:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                best_m = best_M
                best_a = a.copy()
                improved = True
                stall = 0

            # Also try reverse direction (dst -> src)
            if a[dst] > 1e-10:
                best_frac_r = 0
                best_M_r = best_m
                for frac in epsilons:
                    amt = a[dst] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[dst] -= amt
                    a_trial[src] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M_r:
                        best_M_r = M_trial
                        best_frac_r = frac
                if best_frac_r > 0:
                    amt = a[dst] * best_frac_r
                    a[dst] -= amt
                    a[src] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                    best_m = best_M_r
                    best_a = a.copy()
                    improved = True
                    stall = 0

        if not improved:
            stall += 1
            # Try random perturbation to escape
            if stall > 100:
                n_perturb = max(5, n // 50)
                idx = rng.permutation(n)[:n_perturb]
                a_perturbed = a * (1.0 + rng.normal(0, 0.02, n_perturb))
                a_perturbed[idx] = a_perturbed[rng.permutation(n_perturb)[:n_perturb]]
                a_perturbed = np.maximum(a_perturbed, 1e-10)
                a_perturbed = a_perturbed / a_perturbed.sum()
                M_pert = fast_conv(a_perturbed).max()
                if M_pert < best_m:
                    a = a_perturbed
                    best_m = M_pert
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord aggressive ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def zoom_in_init(n_target, time_budget, rng):
    """
    Create a good initial sequence by optimizing at small n, then upscaling.
    """
    # Start at small n
    n_small = max(50, n_target // 16)
    # Initialize with a "U-shaped" profile (large at edges, small in middle)
    x = np.linspace(0, 1, n_small)
    a_small = 0.5 + 0.5 * np.cos(np.pi * x)  # U-shaped
    a_small = np.maximum(a_small, 0.01)
    a_small = a_small / a_small.sum()

    # Quick optimization at small n
    t0 = time.monotonic()
    t_opt = min(time_budget * 0.3, 30.0)
    if t_opt > 5:
        a_small, _ = nesterov_smooth(a_small, t_opt, verbose=False, seed=123,
                                     temp_init=0.005, temp_final=0.001)

    # Upscale to target n
    a_target = interpolate_sequence(a_small, n_target)
    return a_target


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation and diverse strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1700.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
            parent_loaded = True
    except Exception:
        pass

    if best_seq is None:
        # Create default initialization
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = 0.5 + 0.5 * np.cos(np.pi * x)
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 0: Quick zoom-in initialization to potentially find a better starting point ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200 and not parent_loaded:
        print("\n=== Phase 0: Zoom-in init ===")
        t_zoom = min(remaining * 0.1, 100.0)
        a_zoom = zoom_in_init(n0, t_zoom, rng)
        # Quick polish
        a_zoom, _ = nesterov_smooth(a_zoom, min(t_zoom * 0.5, 50.0), verbose=False, seed=77)
        try_update(a_zoom, f"zoom_n{n0}")

    # === Phase 1: Main Nesterov smooth-max optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: NSMD main n={len(best_seq)} ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 50:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth(a_in, t1, verbose=True, seed=42,
                                   temp_init=0.01, temp_final=0.0005,
                                   temp_decay_rate=0.9997, c0=6.0)
        try_update(a_out, f"nsmd_main_n{len(best_seq)}")

    # === Phase 2: LP Frank-Wolfe lazy ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-lazy n={n_cur} ===")
    if n_cur <= 3000:
        t2 = min(remaining * 0.20, 250.0)
        if t2 > 60:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_lazy(a_in, t2, solver, verbose=True, top_K=15)
            try_update(a_out, f"lp_lazy_n{n_cur}")

    # === Phase 3: Aggressive coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Coord aggressive n={n_cur} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish_aggressive(a_in, t3, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 4: Second NSMD pass with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: NSMD pass 2 n={n_cur} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        # Perturb slightly before second pass
        a_pert = a_in * (1.0 + rng.normal(0, 0.005, n_cur))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        a_out, _ = nesterov_smooth(a_pert, t4, verbose=True, seed=137,
                                   temp_init=0.005, temp_final=0.0003,
                                   temp_decay_rate=0.9998, c0=7.0)
        try_update(a_out, f"nsmd2_n{n_cur}")

    # === Phase 5: Coordinate polish after pass 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord 2 n={n_cur} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish_aggressive(a_in, t5, verbose=True, seed=99)
        try_update(a_out, f"coord2_n{n_cur}")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length search around n={n_best} ===")
    if remaining > 150:
        for delta in [200, -200, 400, -400]:
            n_try = n_best + delta
            if n_try < 800 or n_try > 4000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={n_try}...")
            a_long = interpolate_sequence(best_seq, n_try)
            # Quick optimization
            t6 = min(remaining * 0.35, 120.0)
            if t6 > 20:
                a_out, _ = nesterov_smooth(a_long, t6, verbose=False, seed=55 + n_try % 100,
                                           temp_init=0.008, temp_final=0.001, c0=5.0)
                try_update(a_out, f"nsmd_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    a_out2, _ = coordinate_polish_aggressive(a_out, min(remaining * 0.4, 60.0),
                                                             verbose=False, seed=77)
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 7: Final LP polish if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 100:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_lazy(a_in, t7, solver, verbose=True, top_K=20)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 40:
        t8 = min(remaining * 0.7, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish_aggressive(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final NSMD micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Micro NSMD n={n_best} ===")
        t9 = min(remaining * 0.6, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_smooth(a_in, t9, verbose=False, seed=271,
                                       temp_init=0.003, temp_final=0.0002,
                                       temp_decay_rate=0.9999, c0=8.0)
            try_update(a_out, f"micro_nsmd_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, elapsed={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
