```
def next_pow2(m):
    if m <= 1: return 1
    return 1 << (m - 1).bit_length()

def convolve_autosq(a, fft_size=None):
    n = len(a)
    if fft_size is None:
        fft_size = next_pow2(2*n-1)
    af = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(af * af, fft_size)[:2*n-1]
    # numerical negatives
    if c.min() < 0: c = np.maximum(c,0)
    return c
```

```
def project_simplex(v):
    n = len(v)
    if n == 0: return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    # find rho
    # use np.nonzero
    idx = np.nonzero(u * np.arange(1, n+1) > (css - 1.0))[0]
    if len(idx)==0: return np.ones(n)/n
    rho = idx[-1]
    theta = (css[rho] - 1.0)/(rho+1.0)
    return np.maximum(v - theta, 0.0)
```

```
def project_simplex(v):
    n=len(v)
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u * np.arange(1,n+1) > (cssv - 1.0))[0]
    if len(rho)==0:
        return np.ones(n)/n
    rho=int(rho[-1])
    theta=(cssv[rho]-1.0)/(rho+1.0)
    return np.maximum(v-theta,0.0)
```

```python
"""
Search for a non-negative coefficient sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

Strategy:
- Warm-start from the provided parent sequence when available.
- Create several structured square-root / tapered initializations as
  alternatives to avoid bad local minima.
- Use exact coordinate-pair descent.  Each pair move is optimized in one
  dimension by directly minimizing the maximum of all updated autocorrelation
  coefficients, using only O(n) operations per move.
- Use a short accelerated projected-gradient phase with a full softmax-smoothed
  max autocorrelation objective when the starting point is not already good.
- Optionally use L-BFGS-B in a softmax parameterization for multi-coordinate
  polishing.
- Try a modestly longer length after optimizing the primary length.
- Always reserve time and return the best sequence found.
"""

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_sum(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def conv_square(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if fft_size is None:
        fft_size = next_pow2(2 * n - 1)
    af = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(af * af, fft_size)[:2 * n - 1]
    return np.maximum(c, 0.0)


def smooth_value_max(a, beta, fft_size=None):
    n = len(a)
    if n == 0:
        return 0.0, 0.0
    if fft_size is None:
        fft_size = next_pow2(2 * n - 1)
    af = np.fft.rfft(a, fft_size)
    c = np.maximum(np.fft.irfft(af * af, fft_size)[:2 * n - 1], 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc) or maxc <= 0.0:
        return 0.0, 0.0
    beta = max(1.0, float(beta))
    z = beta * (c - maxc)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return maxc, maxc
    val = maxc + math.log(s) / beta
    return val, maxc


def smooth_value_grad_max(a, beta, fft_size=None):
    n = len(a)
    if n == 0:
        return 0.0, np.array([], dtype=np.float64), 0.0
    if fft_size is None:
        fft_size = next_pow2(2 * n - 1)
    af = np.fft.rfft(a, fft_size)
    c = np.maximum(np.fft.irfft(af * af, fft_size)[:2 * n - 1], 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc) or maxc <= 0.0:
        return 0.0, np.zeros(n, dtype=np.float64), 0.0
    beta = max(1.0, float(beta))
    z = beta * (c - maxc)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return maxc, 2.0 * a, maxc
    w = e / s
    val = maxc + math.log(s) / beta
    fw = np.fft.rfft(w, fft_size)
    h = np.fft.irfft(fw * np.conj(af), fft_size)[:n]
    g = 2.0 * h
    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), g, maxc


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def resample_to_length(a, n):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    old_n = a.size
    if old_n == 0:
        return np.ones(n, dtype=np.float64) / n
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(n, dtype=np.float64) / n
    if old_n == n:
        return a / s
    f = a / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, n)
    g = np.interp(x_new, x_old, f)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if not np.isfinite(gs) or gs <= 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return g / gs


def sqrt_base_coeff(n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    s = np.zeros(n, dtype=np.float64)
    s[0] = 1.0
    for k in range(1, n):
        if k > 1:
            corr = float(np.dot(s[1:k], s[k - 1:0:-1]))
        else:
            corr = 0.0
        val = 0.5 * (1.0 - corr)
        if not np.isfinite(val) or val < 0.0:
            val = 0.0
        s[k] = val
    return s

def run_apg(a0, time_budget, beta_start=20.0, beta_end=300.0, step0=0.35,
            max_iters=200000, verbose=True, seed=1):
    a0 = np.asarray(a0, dtype=np.float64).reshape(-1)
    if a0.size == 0 or float(time_budget) <= 0.0:
        return a0

    n = a0.size
    a = normalize_sum(a0)
    fft_size = next_pow2(2 * n - 1)
    t0 = time.monotonic()
    best = a.copy()
    best_max = float(np.max(conv_square(a, fft_size)))

    x = a.copy()
    old_x = x.copy()
    y = x.copy()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 >= time_budget:
            break

        frac = min(1.0, it / 20000.0)
        beta = beta_start + (beta_end - beta_start) * frac

        f_y, g, max_y = smooth_value_grad_max(y, beta, fft_size)
        if not np.isfinite(f_y) or not np.all(np.isfinite(g)):
            y = x.copy()
            continue

        if max_y < best_max - 1e-15:
            best_max = max_y
            best = y.copy()

        f_x, max_x = smooth_value_max(x, beta, fft_size)
        base_f = min(f_y, f_x)
        fs = step0
        accepted = False

        for _ in range(4):
            cand = project_simplex(y - fs * g)
            f_new, max_new = smooth_value_max(cand, beta, fft_size)
            if np.isfinite(f_new) and (
                f_new < base_f - 1e-13 or
                (f_new < base_f + 1e-10 and max_new < min(max_x, max_y) - 1e-15)
            ):
                accepted = True
                old_x = x
                x = cand
                if max_new < best_max - 1e-15:
                    best_max = max_new
                    best = x.copy()
                y = project_simplex(x + 0.5 * (x - old_x))
                stall = 0
                break
            fs *= 0.5

        if not accepted:
            stall += 1
            y = x.copy()
            if stall % 700 == 0:
                noise = rng.normal(0.0, 1e-4, n) * x
                cand = project_simplex(x + noise)
                _, max_new = smooth_value_max(cand, beta, fft_size)
                if np.isfinite(max_new) and max_new < best_max * 1.02:
                    old_x = x
                    x = cand
                    y = x.copy()
                    if max_new < best_max - 1e-15:
                        best_max = max_new
                        best = x.copy()
                    stall = 0

        if verbose and it % 2500 == 0:
            print(f"     APG it={it} beta={beta:.1f} score={2.0 * n * best_max:.6f}")

    _, best_max = smooth_value_max(best, 1.0, fft_size)
    if best_max <= 0.0:
        best_max = float(np.max(conv_square(best, fft_size)))
    return best


def eval_pair_loss(c, l, q, d):
    return float(np.max(c + l * float(d) + q * float(d) * float(d)))


def pair_best_step(c, l, q, lo, hi, ngrid=49, golden_iters=20):
    lo = float(lo)
    hi = float(hi)
    if hi - lo < 1e-18:
        return 0.0, float(np.max(c))

    d = np.linspace(lo, hi, int(ngrid), dtype=np.float64)
    dq = c[None, :] + l[None, :] * d[:, None] + q[None, :] * (d[:, None] * d[:, None])
    vals = dq.max(axis=1)
    idx = int(np.argmin(vals))
    best_d = float(d[idx])
    best_val = float(vals[idx])

    if idx > 0 and idx < len(d) - 1:
        left = float(d[idx - 1])
        right = float(d[idx + 1])
    elif idx == 0:
        left = lo
        right = float(d[1])
    else:
        left = float(d[-2])
        right = hi

    if left > right:
        left, right = right, left
    if right - left < 1e-18:
        return best_d, best_val

    gr = 0.6180339887498949
    c1 = right - gr * (right - left)
    c2 = left + gr * (right - left)
    f1 = eval_pair_loss(c, l, q, c1)
    f2 = eval_pair_loss(c, l, q, c2)

    for _ in range(int(golden_iters)):
        if right - left < 1e-18:
            break
        if f1 < f2:
            right = c2
            c2 = c1
            f2 = f1
            c1 = right - gr * (right - left)
            f1 = eval_pair_loss(c, l, q, c1)
        else:
            left = c1
            c1 = c2
            f1 = f2
            c2 = left + gr * (right - left)
            f2 = eval_pair_loss(c, l, q, c2)

    mid = 0.5 * (left + right)
    fmid = eval_pair_loss(c, l, q, mid)
    if fmid < best_val:
        best_val = fmid
        best_d = float(mid)
    return best_d, best_val


def compute_source_scores(a, c, n, top_count=24):
    m = 2 * n - 1
    top_count = min(max(1, int(top_count)), m)
    k_list = np.argsort(c)[-top_count:]
    src = np.zeros(n, dtype=np.float64)
    for kk in k_list:
        k = int(kk)
        ist = max(0, k - n + 1)
        ien = min(n - 1, k) + 1
        if ist >= ien:
            continue
        idx = np.arange(ist, ien)
        src[ist:ien] += a[idx] * a[k - idx]
    return src


def make_pairs(n, src, rng, n_src=50, n_dst=120, random_pairs=100, pair_limit=350):
    n_src = min(max(1, int(n_src)), n)
    n_dst = min(max(1, int(n_dst)), n)
    random_pairs = max(0, int(random_pairs))
    pair_limit = max(64, int(pair_limit))

    if not np.all(np.isfinite(src)) or float(src.sum()) <= 0.0:
        top_src = rng.integers(0, n, size=n_src)
        bottom_dst = rng.integers(0, n, size=n_dst)
    else:
        top_src = np.argsort(-src)[:n_src]
        bottom_dst = np.argsort(src)[:n_dst]

    pairs = []
    for i in top_src:
        i = int(i)
        sz = min(8, int(bottom_dst.size))
        if sz <= 0:
            continue
        choices = rng.choice(bottom_dst, size=sz, replace=False)
        for j in choices:
            j = int(j)
            if j != i:
                pairs.append((i, j))
        if len(pairs) >= pair_limit:
            break

    if random_pairs > 0:
        if np.all(np.isfinite(src)) and float(src.sum()) > 0.0:
            weights = src / float(src.sum())
            weights = np.nan_to_num(weights, nan=0.0, posinf=0.0, neginf=0.0)
            if float(weights.sum()) <= 0.0:
                weights = None
            else:
                weights = weights / float(weights.sum())
        else:
            weights = None

        if weights is None:
            extra_sources = rng.integers(0, n, size=random_pairs)
        else:
            extra_sources = rng.choice(n, size=random_pairs, p=weights)

        for i in extra_sources:
            i = int(i)
            j = int(bottom_dst[int(rng.integers(bottom_dst.size))])
            if j != i:
                pairs.append((i, j))

        for _ in range(min(50, random_pairs)):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.append((i, j))

    return pairs[:pair_limit]

            if i == j:
                continue
            if a[i] <= 0.0 and a[j] <= 0.0:
                continue

            l = np.zeros(m, dtype=np.float64)
            q = np.zeros(m, dtype=np.float64)

            l[j:j + n] += 2.0 * a
            l[i:i + n] -= 2.0 * a

            if 2 * i < m:
                q[2 * i] += 1.0
            if 2 * j < m:
                q[2 * j] += 1.0
            ksum = i + j
            if ksum < m:
                q[ksum] -= 2.0

            lo = -float(a[j])
            hi = float(a[i])
            if hi - lo < 1e-18:
                continue

            d, val = pair_best_step(c, l, q, lo, hi,
                                    ngrid=ngrid, golden_iters=golden_iters)
            d = float(d)
            if d > hi:
                d = hi
            if d < lo:
                d = lo
            if abs(d) < 1e-18:
                continue

            if val < cur - 1e-15:
                c = c + l * d + q * d * d
                a[i] = a[i] - d
                a[j] = a[j] + d

                if a[i] < 0.0:
                    a[i] = 0.0
                if a[j] < 0.0:
                    a[j] = 0.0

                if c.min() < 0.0:
                    c = np.maximum(c, 0.0)

                cur = float(c.max())
                if not np.isfinite(cur):
                    a = normalize_sum(a)
                    c = conv_square(a)
                    cur = float(c.max())

                chunk_improved = True
                no_improve = 0
                accepted += 1

                if 2.0 * n * cur < best_score - 1e-13:
                    best_score = float(2.0 * n * cur)
                    best_a = a.copy()

                if accepted % 100 == 0:
                    ss = float(a.sum())
                    if not np.isfinite(ss) or ss <= 1e-12 or abs(ss - 1.0) > 1e-8:
                        a = normalize_sum(a)
                    c = conv_square(a)
                    cur = float(c.max())
                    if 2.0 * n * cur < best_score:
                        best_score = float(2.0 * n * cur)
                        best_a = a.copy()
            else:
                no_improve += 1

        if not chunk_improved:
            no_improve += 8

        if verbose and time.monotonic() - last_print > 30.0:
            print(f"     pair t={time.monotonic() - t0:.0f}s score={best_score:.6f} cur={2.0*n*cur:.6f}")
            last_print = time.monotonic()

    best_a = normalize_sum(best_a)
    cmax = float(np.max(conv_square(best_a)))
    final_est = float(2.0 * n * cmax)
    if np.isfinite(final_est) and final_est < best_score:
        best_score = final_est
    return best_a


def lbfgs_fun_grad(p):
    state = _LBFGS_STATE
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.nan_to_num(p, nan=0.0, posinf=60.0, neginf=-60.0)
    a = softmax_positive(p)
    size = int(state["fft_size"])
    beta = max(1.0, float(state["beta"]))
    val, g, _ = smooth_value_grad_max(a, beta, size)
    if not np.isfinite(val):
        return 0.0, np.zeros_like(p)
    ga = g - a * float(np.dot(g, a))
    ga = np.nan_to_num(ga, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), ga


def lbfgs_callback(p):
    _LBFGS_STATE["last_p"] = np.asarray(p, dtype=np.float64).copy()
    state = _LBFGS_STATE
    if time.monotonic() - float(state["start"]) > float(state["limit"]):
        raise RuntimeError("lbfgs time limit")


def run_lbfgs(a0, time_budget, beta=200.0, maxiter=220):
    a0 = np.asarray(a0, dtype=np.float64).reshape(-1)
    if a0.size == 0 or float(time_budget) <= 0.0 or not HAVE_SP_OPT:
        return a0

    a = normalize_sum(a0)
    n = a.size
    p0 = np.log(np.maximum(a, 1e-300))
    p0 -= float(np.max(p0))

    _LBFGS_STATE["beta"] = float(beta)
    _LBFGS_STATE["n"] = n
    _LBFGS_STATE["fft_size"] = next_pow2(2 * n - 1)
    _LBFGS_STATE["start"] = time.monotonic()
    _LBFGS_STATE["limit"] = float(time_budget)
    _LBFGS_STATE["last_p"] = p0.copy()

    bounds = [(-60.0, 60.0) for _ in range(n)]
    popt = p0
    try:
        res = sp_opt.minimize(
            lbfgs_fun_grad,
            p0,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "maxfun": max(1000, int(maxiter) * 25),
                "ftol": 1e-14,
                "gtol": 1e-8,
                "disp": False,
                "callback": lbfgs_callback,
            },
        )
        if hasattr(res, "x") and res.x is not None:
            popt = np.asarray(res.x, dtype=np.float64)
        else:
            popt = _LBFGS_STATE["last_p"]
    except Exception:
        popt = _LBFGS_STATE["last_p"]

    if popt is None:
        popt = p0
    popt = np.nan_to_num(popt, nan=0.0, posinf=60.0, neginf=-60.0)
    a2 = softmax_positive(popt)
    return normalize_sum(a2)


def uniform_fallback(n=2400):
    n = int(max(1, n))
    return [1.0 / n] * n


def time_remaining(deadline):
    return max(0.0, float(deadline) - time.monotonic())


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total = min(float(budget_s), 1800.0) - 90.0
    if total < 60.0:
        total = max(30.0, min(float(budget_s), 1800.0) - 10.0)
    deadline = t_start + total

    parent = get_parent_sequence()

    if parent is not None:
        if len(parent) < 2000:
            parent = resample_to_length(parent, 2400)
        base_n = min(3600, max(1200, len(parent)))
    else:
        base_n = 2400

    if parent is not None:
        parent = resample_to_length(parent, base_n)

    candidates = []

    if parent is not None:
        candidates.append(normalize_sum(parent))

    base = sqrt_base_coeff(base_n)
    for gamma in (0.0, 0.5, 1.0, 1.5, 2.5, -0.5):
        candidates.append(tapered_sqrt_candidate(base, gamma))

    best_a = np.ones(base_n, dtype=np.float64) / base_n
    best_score = float("inf")

    for cand in candidates:
        best_a, best_score = consider_candidate(cand, best_a, best_score)

    print(f"initial best score={best_score:.6f} n={len(best_a)}")

    # Optional short smoothed projected-gradient phase when the initial score
    # is not already close to the target.
    if best_score > 1.515 or parent is None:
        rem = time_remaining(deadline)
        if rem > 80.0:
            budget = min(80.0, rem * 0.20)
            a_out = run_apg(best_a, budget, beta_start=25.0, beta_end=350.0,
                            step0=0.30, seed=7, verbose=False)
            best_a, best_score = consider_candidate(a_out, best_a, best_score)
            print(f"after APG score={best_score:.6f} n={len(best_a)}")

    # Main primary-length pair descent.
    rem = time_remaining(deadline)
    if rem > 120.0:
        budget = min(700.0, rem * 0.42)
        a_out = pair_descent(best_a, budget, verbose=True, seed=101,
                             ngrid=45, golden_iters=18, pair_limit=350)
        best_a, best_score = consider_candidate(a_out, best_a, best_score)
        print(f"after primary pair score={best_score:.6f} n={len(best_a)}")

    # Multi-coordinate smoothed polish.
    rem = time_remaining(deadline)
    if HAVE_SP_OPT and rem > 160.0 and best_score > 1.500:
        budget = min(110.0, rem * 0.12)
        a_out = run_lbfgs(best_a, budget, beta=220.0, maxiter=220)
        best_a, best_score = consider_candidate(a_out, best_a, best_score)
        print(f"after L-BFGS score={best_score:.6f} n={len(best_a)}")

    # Try a modestly longer length if still worth attempting.
    rem = time_remaining(deadline)
    if rem > 350.0 and best_score > 1.5020:
        old_n = int(len(best_a))
        new_n = min(old_n + 600, 3600)
        if new_n > old_n:
            work_a = resample_to_length(best_a, new_n)
            work_a, _ = consider_candidate(work_a, np.array([], dtype=float), float("inf"))
            print(f"longer init n={new_n} est_score={estimated_score(work_a):.6f}")

            rem = time_remaining(deadline)
            if estimated_score(work_a) > 1.52 and rem > 80.0:
                budget = min(70.0, rem * 0.18)
                work_a = run_apg(work_a, budget, beta_start=25.0, beta_end=350.0,
                                 step0=0.30, seed=17, verbose=False)

            rem = time_remaining(deadline)
            if rem > 150.0:
                budget = min(650.0, rem * 0.45)
                work_a = pair_descent(work_a, budget, verbose=True, seed=202,
                                      ngrid=45, golden_iters=18, pair_limit=320)

            best_a, best_score = consider_candidate(work_a, best_a, best_score)
            print(f"after longer length score={best_score:.6f} n={len(best_a)}")

            rem = time_remaining(deadline)
            if HAVE_SP_OPT and rem > 120.0:
                budget = min(80.0, rem * 0.15)
                work_a = run_lbfgs(work_a, budget, beta=250.0, maxiter=180)
                best_a, best_score = consider_candidate(work_a, best_a, best_score)
                print(f"after longer L-BFGS score={best_score:.6f} n={len(best_a)}")

    # Final polish on the retained best sequence.
    rem = time_remaining(deadline) - 30.0
    if rem > 90.0:
        budget = max(30.0, min(500.0, rem * 0.75))
        a_out = pair_descent(best_a, budget, verbose=True, seed=303,
                             ngrid=49, golden_iters=22, pair_limit=350)
        best_a, best_score = consider_candidate(a_out, best_a, best_score)

    # Final safety check using the actual evaluator.
    best_a = normalize_sum(best_a)
    est = estimated_score(best_a)
    exact = exact_score(best_a)

    if not np.isfinite(exact) or not np.isfinite(est):
        best_a = np.ones(base_n, dtype=np.float64) / base_n
        exact = exact_score(best_a)

    print(f"FINAL estimated={est:.6f} exact={exact:.6f} n={len(best_a)} elapsed={time.monotonic()-t_start:.0f}s")

    if not np.isfinite(exact):
        return uniform_fallback(base_n)

    return [float(x) for x in np.asarray(best_a, dtype=np.float64)]
```
