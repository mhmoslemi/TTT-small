And c[0] = λ^2/4 = (2/(2n-1))^2/4 = 1/(2n-1)^2.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Reduced LP-Frank-Wolfe with Nesterov acceleration (top-K peak constraints)
2. Fast FFT-based autocorrelation
3. Adaptive peak tracking
4. Direct subgradient descent with exponential weighting
5. Peak-targeted coordinate descent for final polish
6. Multi-start exploration
"""

import numpy as np
import time
import scipy.fft as sp_fft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = sp_fft.rfft(a.astype(np.float64), fft_size)
    return sp_fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def reduced_lp_frank_wolfe(a0, time_limit, top_k=10, verbose=True, solver_name=None):
    """LP-Frank-Wolfe with reduced constraints (only top K peaks)."""
    import cvxpy as cp
    n = len(a0)
    a = norm_seq(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if solver_name is None:
        solver_name = get_solver()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    t = 0
    max_iters = 100000
    
    # Adaptive top_k: start small, grow
    k_current = min(5, top_k)
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive K
        if t % 200 == 0:
            k_current = min(top_k, 5 + t // 200)
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Get top K peak indices
        k_eff = min(k_current, len(ac))
        peak_idx = np.argsort(ac)[::-1][:k_eff]
        
        # Build constraint vectors for these peaks
        # For peak at index k: c[k] = sum_j a[j] * a[k-j]
        # Linearized: sum_j a[k-j] * b[j] <= c[k](y)
        M_vec = np.array([ac[int(pi)] for pi in peak_idx])
        
        # Build T matrix (K x n) where T[i, j] = y[k_i - j] if valid
        T = np.zeros((k_eff, n))
        for i, pi in enumerate(peak_idx):
            pi_int = int(pi)
            j_start = max(0, pi_int - n + 1)
            j_end = min(n - 1, pi_int)
            T[i, j_start:j_end+1] = y[pi_int - j_end : pi_int - j_start : -1]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} K={k_eff} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m


def nesterov_subgradient(a0, time_limit, top_k=30, beta=4.0, verbose=True, rng_seed=42):
    """Nesterov-accelerated mirror descent with peak-weighted subgradient."""
    n = len(a0)
    a = norm_seq(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0
    stall_count = 0
    max_stall = 5000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            k_int = int(ki)
            # d c[k] / d a[i] = a[k-i] + a[i-k] (each term valid only if index in range)
            # For efficiency, iterate over valid i
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            # Term 1: a[k-i] for i in valid range
            term1 = y[k_int - idx]  # a[k-i]
            # Term 2: a[i-k] for i where i-k >= 0, i.e., i >= k
            mask2 = idx >= k_int
            term2 = np.zeros_like(idx, dtype=np.float64)
            if mask2.any():
                idx2 = idx[mask2]
                term2[mask2] = y[idx2 - k_int]
            grad[idx] += wi * (term1 + term2)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.03, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSGD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = norm_seq(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_idx = np.argmax(ac)
        k_int = int(peak_idx)
        
        # Compute sensitivity: d c[k] / d a[i] for the peak
        sens = np.zeros(n)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        idx = np.arange(i_start, i_end + 1)
        term1 = a[k_int - idx]
        mask2 = idx >= k_int
        term2 = np.zeros(n)
        if mask2.any():
            term2[idx[mask2]] = a[idx[mask2] - k_int]
        sens[idx] = term1 + term2
        
        # Also consider top 5 peaks
        top_peaks = np.argsort(ac)[::-1][:5]
        total_sens = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            i_s = max(0, pk_int - n + 1)
            i_e = min(n - 1, pk_int)
            idx2 = np.arange(i_s, i_e + 1)
            t1 = a[pk_int - idx2]
            m2 = idx2 >= pk_int
            t2 = np.zeros(len(idx2))
            if m2.any():
                t2[m2] = a[idx2[m2] - pk_int]
            total_sens[idx2] += t1 + t2
        
        # Try to move mass from high-sensitivity to low-sensitivity positions
        improved = False
        # Sort by sensitivity (descending) for source, ascending for destination
        src_sorted = np.argsort(total_sens)[::-1][:10]
        dst_sorted = np.argsort(total_sens)[:10]
        
        for src in src_sorted:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in dst_sorted:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordPD stall={stall} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interp_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.
