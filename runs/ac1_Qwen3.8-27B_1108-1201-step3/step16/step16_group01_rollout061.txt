def exact_subgradient(a, conv, threshold_frac=0.999):
    """Compute exact subgradient from active set of lags."""
    n = len(a)
    M = conv.max()
    if M < 1e-30:
        return np.zeros(n)
    
    active = np.where(conv >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(conv.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    return g


def mirror_update(y, g, eta):
    """Mirror descent update on the simplex."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax(x):
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-30:
        return np.ones_like(x) / len(x)
    return exp_x / s


def lse_conv_value_and_grad_a(a, beta):
    """
    Compute LSE value of conv(a) and its gradient w.r.t. a.
    Returns (lse_value, grad_a) where grad_a = d(LSE)/d(a).
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_c = conv.max()
    log_w = beta * (conv - max_c)
    exp_w = np.exp(log_w)
    w = exp_w / exp_w.sum()
    lse_val = max_c + np.log(exp_w.sum()) / beta

    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_full = np.fft.irfft(A * W, fft_size)
    grad_a = 2.0 * conv_full[n - 1: 2 * n - 1]
    return lse_val, grad_a


def softmax_obj_grad(x, beta, scale):
    """
    Objective and gradient in softmax parameterization.
    f(x) = scale * LSE(beta, conv(softmax(x)))
    """
    a = softmax(x)
    n = len(a)
    lse_val, grad_a = lse_conv_value_and_grad_a(a, beta)
    obj = scale * lse_val

    centered = grad_a - np.dot(a, grad_a)
    grad_x = scale * a * centered
    return obj, grad_x

def exact_mirror_descent(a0, time_limit, c0=3.0, threshold_frac=0.999,
                         verbose=True, rng_seed=42):
    """
    Mirror descent with exact active-set subgradient and backtracking line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    scale = 2.0 * n

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and exact subgradient
        conv = fast_autocorr(y)
        M_curr = conv.max()
        active = np.where(conv >= threshold_frac * M_curr)[0]
        if len(active) == 0:
            active = np.array([int(conv.argmax())])

        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * y[j_idx]

        # Step size with backtracking
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1.0
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n) / gn

        # Backtracking on exact objective
        obj_curr = scale * M_curr
        a_new = mirror_update_func(y, g, eta)
        M_new = fast_autocorr(a_new).max()
        obj_new = scale * M_new

        cnt = 0
        while obj_new > obj_curr * (1 - 1e-10) and cnt < 15:
            eta *= 0.5
            a_new = mirror_update_func(y, g, eta)
            M_new = fast_autocorr(a_new).max()
            obj_new = scale * M_new
            cnt += 1

        x_prev = x
        x = a_new

        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.003, n) * best_a
            x_restarted = np.maximum(best_a + noise, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = scale * best_M
            print(f"    Exact-MD t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M


def mirror_update_func(y, g, eta):
    """Mirror descent update on simplex."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    scale = 2.0 * n

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_autocorr(a)
        M_curr = conv.max()
        peak_lags = np.argsort(conv)[::-1][:min(20, len(conv))]

        # Compute peak sensitivity for each position
        peak_sens = np.zeros(n)
        for k in peak_lags[:12]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]

        # Find hot positions (high sensitivity, high value)
        hot_score = peak_sens * a
        n_hot = min(50, n // 10)
        hot_positions = np.argsort(hot_score)[::-1][:n_hot]

        # Find cold positions
        cold_positions = np.argsort(peak_sens)[:n_hot]

        improved = False
        n_trials = min(len(hot_positions), 30)

        for src in hot_positions[:n_trials]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Try moving to various cold positions
            for dst in cold_positions[:20]:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

            if not improved:
                # Random transfer
                dst_r = int(rng.integers(n))
                if dst_r != src and a[src] > 1e-8:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_r] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() > 1e-15:
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autonocorr_safe(a_trial)
                        if M_trial < best_M:
                            a = a_trial
                            best_M = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0

        if not improved:
            stall += 1
            # Larger random perturbation
            n_pert = max(5, n // 50)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = scale * best_M
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M


def fast_autonocorr_safe(a):
    """Safe autocorrelation computation."""
    try:
        return fast_autocorr(a).max()
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lse_anneal_lbfgsb(a0, time_limit, beta_levels=None, verbose=True):
    """
    More aggressive LSE annealing with L-BFGS-B.
    Uses more beta levels and longer runs at high beta.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    scale = 2.0 * n

    if beta_levels is None:
        beta_levels = [20, 50, 100, 200, 400, 800, 1500, 3000]

    x = np.log(np.maximum(a, 1e-15))

    for level, beta in enumerate(beta_levels):
        if time.monotonic() - t0 > time_limit:
            break

        remaining = time_limit - (time.monotonic() - t0)
        # Allocate more time to higher beta levels
        time_frac = 0.5 + 0.5 * (level / max(len(beta_levels) - 1, 1))
        level_time = min(remaining * time_frac, 120.0)
        if level_time < 5:
            level_time = remaining * 0.8
        if level_time < 3:
            break

        t_level = time.monotonic()

        def fun_and_grad(x_var):
            try:
                a_var = softmax(x_var)
                lse_val, grad_a = lse_conv_value_and_grad_a(a_var, beta)
                obj = scale * lse_val
                if not np.isfinite(obj):
                    return 1e10, np.zeros_like(x_var)
                centered = grad_a - np.dot(a_var, grad_a)
                grad_x = scale * a_var * centered
                if not np.all(np.isfinite(grad_x)):
                    grad_x = np.zeros_like(x_var)
                return obj, grad_x
            except Exception:
                return 1e10, np.zeros_like(x_var)

        n_scipy = 0
        try:
            max_iter = max(100, int(level_time * 20))
            result = scipy_minimize(
                fun_and_grad, x, jac=True, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'maxfun': max_iter * 4,
                    'ftol': 1e-15,
                    'gtol': 1e-10,
                    'maxls': 50,
                    'maxcor': 25,
                },
            )
            n_scipy = getattr(result, 'nit', 0)
            if result.x is not None:
                a_new = norm_sum1(softmax(result.x))
                M_new = fast_autocorr(a_new).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a_new.copy()
                x = result.x.copy()
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at beta={beta}: {e}")

        # Try a second run with different initial perturbation
        if time.monotonic() - t_level < level_time * 0.6:
            x_pert = x + rng_perturb(x, 0.01)
            try:
                result2 = scipy_minimize(
                    fun_and_grad, x_pert, jac=True, method='L-BFGS-B',
                    options={
                        'maxiter': max(50, int(level_time * 10)),
                        'maxfun': max(200, int(level_time * 40)),
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 30,
                        'maxcor': 15,
                    },
                )
                if result2.x is not None:
                    a_new = norm_sum1(softmax(result2.x))
                    M_new = fast_autocorr(a_new).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a_new.copy()
                        x = result2.x.copy()
            except Exception:
                pass

        if verbose:
            elapsed = time.monotonic() - t0
            J = scale * best_M
            print(f"    LSE-anneal beta={beta} ({elapsed:.1f}s): J={J:.6f}, iters={n_scipy}")

    return best_a, best_M


def rng_perturb(x, scale):
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
    return rng.normal(0, scale, len(x))


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Load parent, establish baseline
    2. LSE-smoothed L-BFGS-B with beta annealing (primary workhorse)
    3. Exact active-set mirror descent
    4. Coordinate descent polish
    5. Second round of LSE from improved point
    6. Multi-length exploration
    7. Final polishing
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LSE + L-BFGS-B (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-LBFGS n={n0} ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_out, _ = lse_anneal_lbfgsb(norm_sum1(best_seq), t1, verbose=True)
        try_update(a_out, f"lse1_n{len(a_out)}")

    # === Phase 2: Exact mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Exact-MD n={n_best} ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 20:
        a_out, _ = exact_mirror_descent(norm_sum1(best_seq), t2, c0=3.0,
                                        threshold_frac=0.999, verbose=True, rng_seed=1)
        try_update(a_out, f"exact_md_n{len(a_out)}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t3, verbose=True, rng_seed=4)
        try_update(a_out, f"coord_n{len(a_out)}")

    # === Phase 4: LSE again (higher beta, from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE-LBFGS round 2 n={n_best} ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 20:
        a_out, _ = lse_anneal_lbfgsb(norm_sum1(best_seq), t4,
                                     beta_levels=[200, 500, 1000, 2000, 5000],
                                     verbose=True)
        try_update(a_out, f"lse2_n{len(a_out)}")

    # === Phase 5: Exact MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Exact-MD polish n={n_best} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_out, _ = exact_mirror_descent(norm_sum1(best_seq), t5, c0=2.0,
                                        threshold_frac=0.9995, verbose=True, rng_seed=2)
        try_update(a_out, f"exact_md2_n{len(a_out)}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-length ===")
        for n_try in [n_best + 200, n_best - 200, 2000, 2800]:
            if n_try <= 500 or n_try > 4000:
                continue
            if n_try == n_best:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={n_try}")
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(remaining * 0.3, 120.0)
            if t_sub > 20:
                a_out, _ = lse_anneal_lbfgsb(norm_sum1(a_interp), t_sub,
                                             beta_levels=[50, 100, 300, 600],
                                             verbose=False)
                try_update(a_out, f"lse_ml_n{n_try}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} ===")
    if remaining > 20:
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 10:
            a_out, _ = coordinate_descent(norm_sum1(best_seq), t7, verbose=True, rng_seed=7)
            try_update(a_out, f"coord_final_n{len(a_out)}")

    # === Phase 8: Final exact MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Final exact-MD n={n_best} ===")
        t8 = min(remaining * 0.40, 100.0)
        if t8 > 10:
            a_out, _ = exact_mirror_descent(norm_sum1(best_seq), t8, c0=1.5,
                                            threshold_frac=0.9999, verbose=False, rng_seed=8)
            try_update(a_out, f"final_md_n{len(a_out)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
