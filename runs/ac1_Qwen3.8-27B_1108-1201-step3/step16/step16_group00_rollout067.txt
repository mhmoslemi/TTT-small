def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=8000,

def build_corr_matrix(y):
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=None,
                   use_line_search=True):
    """LP Frank-Wolfe with Nesterov acceleration, line search, margin constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    K_base = top_K if top_K is not None else min(2 * n - 1, max(300, n))
    b_var = cp.Variable(n, nonneg=True)
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        T_current = build_corr_matrix(y)
        constraint_vals = T_current @ y
        margins = M_curr - constraint_vals
        K_actual = min(K_base + (t // 10) // 2, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_actual]
        T_sel = T_current[binding_idx, :]

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_sel @ b_var <= M_curr])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        if use_line_search:
            alpha_best = 2.0 / (t + 2.0)
            obj_best = fast_autocorr((1 - alpha_best) * y + alpha_best * g_dir).max()
            for trial_alpha in (0.3, 0.5, 0.7, 0.9, 1.0):
                cand = np.maximum((1 - trial_alpha) * y + trial_alpha * g_dir, 0.0)
                sc = cand.sum()
                if sc > 1e-15:
                    cand = cand / sc
                    obj_trial = fast_autocorr(cand).max()
                    if obj_trial < obj_best:
                        obj_best = obj_trial
                        alpha_best = trial_alpha
            alpha = alpha_best
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x = np.maximum(y_new + beta_t * (y_new - y), 0.0)
            sx = x.sum()
            if sx < 1e-15:
                x = y_new.copy()
            else:
                x = x / sx

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish_2point(a0, time_limit, verbose=False, rng_seed=123):

            k2i = int(k2)

                d2 = d - k2i

                if 0 <= d2 < n:
                    add += a[d2]

            if time.monotonic() - t0 > time_limit:

    return best_a, best_m


def make_linear_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_bimodal(n, lo=0.08, hi=0.92, width=40.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - lo) ** 2) + np.exp(-width * (x - hi) ** 2) + 0.005
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in (1, 2, 3, 4, 5, 6):
        coeff = rng.normal(0, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine_bowl(n, m=2.0):
    # raised-cosine "bowl" concentrates mass near edges -> flatter convolution
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 - np.cos(np.pi * x * m))
    a = a + 1e-3
    return norm_sum1(a)
