import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def subgradient_at_k_fast(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a efficiently."""
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2: hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def dual_guided_polish(a0, time_limit, solver, verbose=False):
    """
    Use LP dual variables to guide subgradient updates.
    At optimum, weighted subgradient should be constant on support.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 400
    step = 0.5 / n ** 0.3

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get active set
        tol = 0.05
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 100):]

        # Solve LP to get duals
        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue

        # If we have duals, use them for weighting
        if duals is not None and len(duals) == len(active_set):
            # Duals tell us which constraints are "tightest"
            # Use duals as weights for the peaks
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            # Fallback: use exponential weighting
            weights = np.exp(-5.0 * (M_curr - conv[active_set]) / max(M_curr * 0.02, 1e-10))
            weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k_fast(a, int(k), n)

        # On the support, we want grad to be constant
        support_thresh = 0.005 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            break

        # Direction: reduce where grad is high, increase where grad is low
        # But maintain sum constraint
        grad_mean = np.mean(grad[support])
        direction = grad - grad_mean
        # Slightly bias towards reducing the max
        direction[support] -= 0.1 * np.std(grad[support])

        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 1.5)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish iter={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def multi_subgradient(a0, time_limit, verbose=False, seed=42):
    """Multi-peak subgradient with adaptive step size and diversity."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0 / n ** 0.35
    patience = 0
    max_patience = 600
    rng = np.random.default_rng(seed)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Pick peaks with diversity
        n_peaks = max(3, min(10, n // 80))
        top_indices = np.argsort(conv)[-n_peaks:]

        # Add a random near-max for exploration
        if t % 15 == 0:
            threshold = M_curr * (0.85 + 0.1 * rng.random())
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices.astype(int), k_rand)

        weights = np.exp(-8.0 * (M_curr - conv[top_indices.astype(int)]) / max(M_curr * 0.02, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k_fast(a, int(k), n)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad * n ** 0.35
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.004, 2.5)
        else:
            patience += 1
            step *= 0.985
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False, seed=0):
    """Fast coordinate descent with efficient autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 4

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)

        # Randomize order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm - aj

            # Sensitivity: how (a*a)[k] changes when a[j] changes by delta
            # d(a*a)[k]/da[j] = 2*a[k-j] (if 0<=k-j<n)
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Try candidate values
            candidates = [0.0, aj * 0.3, aj * 0.6, aj * 1.0, aj * 1.4, aj * 1.8, aj * 2.5]
            typical = norm / n
            candidates += [typical * 0.5, typical, typical * 1.5, typical * 3.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.5)) for c in candidates]))

            best_x = aj
            best_obj = fast_autocorr(a).max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                # Update autocorrelation
                # (a*a)[k] changes by 2*(x-aj)*a[k-j] for k in [j, j+n-1]
                # Except at k=2j where we need x^2-aj^2 instead of 2*(x-aj)*a[j]
                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    # Correction: we added 2*delta*a[j] but should add x^2-aj^2
                    # Difference: x^2-aj^2 - 2*delta*a[j] = (x-aj)^2
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    # Verify with full computation
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = new_score * np.ones_like(conv)  # Invalidate cached conv
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        # Revert
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_and_optimize(best_seq, time_limit, solver, rng, seed):
    """Perturb the best sequence and optimize the perturbed version."""
    n = len(best_seq)
    scale = 0.01 + 0.015 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.2)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = perturbed / s

    a_norm = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=False, seed=seed)
    return a_out


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence. Returns (best_seq, best_score, improved)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
