def compute_subgradient(a, ac, top_k=80, beta=4.0):
    """Compute weighted subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        wi_f = float(wi)
        if wi_f < 1e-12:
            continue
        k_int = int(k)
        i1s = max(0, k_int - n + 1)
        i1e = min(n - 1, k_int)
        if i1s <= i1e:
            idx = np.arange(i1s, i1e + 1)
            grad[idx] += wi_f * a[k_int - idx]
        i2s = max(0, k_int)
        i2e = min(n - 1, k_int + n - 1)
        if i2s <= i2e:
            idx = np.arange(i2s, i2e + 1)
            grad[idx] += wi_f * a[idx - k_int]
    return 2.0 * grad

def greedy_alpha(y, d, n_alphas=20):
    """Find best alpha in [0,1] along direction d from y."""
    best_alpha = 0.5
    best_M = fast_autocorr(y).max()
    alphas = np.linspace(0.05, 1.0, n_alphas)
    for alpha in alphas:
        trial = (1.0 - alpha) * y + alpha * d

        M_trial = fast_autocorr(trial).max()
        if M_trial < best_M:
            best_M = M_trial
            best_alpha = alpha
    return best_alpha, best_M

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=250, verbose=True):

            active_lags = np.argsort(ac)[::-1][:max_constraints]
