"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gaussian-shaped initialization (key insight: centered Gaussian gives ~1.1-1.3)
2. Log-sum-exp smooth objective with Adam optimizer and FFT-based gradients
3. Beta annealing for progressive peak sharpening
4. LP-based Frank-Wolfe polishing
5. Coordinate descent final polish
6. Multi-width exploration

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def gaussian_init(n, sigma_frac=0.5):
    """Create a Gaussian-shaped sequence centered in the middle."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-(x - center)**2 / (2 * sigma**2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def compute_gradient(a, beta, fft_size=None):
    """Compute gradient of log-sum-exp objective using FFT."""
    n = len(a)
    conv = fast_conv(a)
    # Softmax weights
    beta_conv = beta * conv
    beta_conv_max = beta_conv.max()
    exp_bc = np.exp(beta_conv - beta_conv_max)
    p = exp_bc / exp_bc.sum()
    
    # Compute gradient via FFT
    # grad[i] = sum_k p[k] * (a[k-i] + a[i-k])
    # = cross_correlation(p, a) + convolution(a, p)
    
    if fft_size is None:
        fft_size = 1 << (3 * n - 2 + 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    
    # Cross-correlation: sum_k p[k] * a[k-i] for i=0,...,n-1
    # = ifft(fft(p) * conj(fft(a)))
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    
    # Convolution: sum_k a[k] * p[k-i]... 
    # Actually: h'[i] = sum_j a[j] * p[i-j] = convolution(a, p)
    conv_ap = np.fft.irfft(A * P, fft_size)[:n]
    
    grad = corr + conv_ap
    return grad, p

def adam_logsumexp(a0, time_limit, beta_init=1.0, beta_max=200.0, 
                   lr_init=0.01, verbose=True, seed=42):
    """
    Adam optimizer on log-sum-exp objective with beta annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    best_obj = 2.0 * n * best_m  # since sum=1
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    # Annealing schedule
    n_anneal_steps = 50000
    current_beta = beta_init
    
    stall = 0
    max_stall = 2000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Anneal beta
        if t < n_anneal_steps:
            frac = t / n_anneal_steps
            current_beta = beta_init + (beta_max - beta_init) * frac**2
        else:
            current_beta = min(current_beta * 1.01, 500.0)
        
        # Compute gradient
        grad, p = compute_gradient(a, current_beta)
        
        # Adam update
        lr = lr_init / (np.sqrt(t) + 1.0)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex (sum=1, a>=0)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        
        # Track best
        M_new = fast_conv(a).max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stuck
        if stall > max_stall:
            # Perturb and restart
            noise = rng.normal(0, 0.01, n)
            a_new = best_a + noise * best_a
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            m = np.zeros(n)
            v = np.zeros(n)
            stall = 0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE t={t} ({elapsed:.1f}s): beta={current_beta:.1f}, obj={best_obj:.6f}")
    
    return best_a, best_obj / (2.0 * n)  # return (a, max_b) since sum=1
