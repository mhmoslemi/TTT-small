from scipy.fft import rfft, irfft

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, fft_size)[:size]

    """Subgradient of (a*a)[k] w.r.t. a"""
    # (a*a)[k] = sum_i a[i]*a[k-i] (valid indices only)
    # d/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0
    # (including the case j = k-j, which gives 2*a[j], correct for d/d(a[j]) of a[j]^2)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo >= hi:
        return np.zeros(n)
    grad = np.zeros(n)
    # For j in [lo, hi): k-j is in [0, n-1]
    # grad[j] = 2 * a[k-j]
    idx = np.arange(lo, hi)
    grad[lo:hi] = 2.0 * a[k - idx]
    return grad

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return irfft(A * A, fft_size)[:size]

def subgrad_conv_k(a, k, n):
    """Subgradient of (a*a)[k] w.r.t. a. Result has length n."""
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo >= hi:
        return np.zeros(n)
    grad = np.zeros(n)
    idx = np.arange(lo, hi)
    grad[lo:hi] = 2.0 * a[k - idx]
    return grad

def solve_lp(a, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a)
    k_arr = np.asarray(active_set, dtype=int)
    size = len(k_arr)
    
    # Build constraint matrix T: T[row, j] = a[k - j] if 0 <= k-j < n
    T = np.zeros((size, n))
    for row, k in enumerate(k_arr):
        lo = max(0, int(k) - n + 1)
        hi = min(n, int(k) + 1)
        if lo < hi:
            T[row, lo:hi] = a[int(k) - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, duals

def normalize(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def lp_fictitious_play(a0, time_limit, solver, verbose=True, seed=0):

    stall_count = 0
    n_iters = 0
    
    for t in range(1, 1000000):

            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Adaptive active set

            # Start with wide tolerance, tighten over time
            tol = 0.10 * (1.0 - progress) + 0.01 * progress
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= threshold)[0]
            # Always include top peaks
            n_top = max(10, n // 40)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap size

            if len(active_set) > max_active:

                active_set = active_set[keep]
        
        b, duals = solve_lp(y, M_curr, active_set, solver)
        
        if b is None:
            # Retry with fewer constraints
            top_k = np.argsort(conv_y)[-max(20, n // 20):]
            b, duals = solve_lp(y, M_curr, top_k, solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, n)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, n)
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Restart if stalled
        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def subgradient_anneal(a0, time_limit, verbose=True, seed=0):
    """Subgradient method with geometric annealing of temperature."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    conv0 = fast_autocorr(a)
    M0 = conv0.max()
    t_temp = M0 * 0.05  # initial temperature
    cooling = 0.995
    
    step = 0.3 / np.sqrt(n)
    patience = 0
    max_patience = 300
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Update temperature
        t_temp *= cooling
        if t_temp < M_curr * 0.001:
            t_temp = M_curr * 0.001
        
        # Softmax weights
        shifted = (conv - M_curr) / t_temp
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = weights / w_sum
        
        # Compute weighted subgradient (use top peaks for efficiency)
        n_peaks = max(20, n // 20)
        top_idx = np.argsort(weights)[-n_peaks:]
        top_weights = weights[top_idx]
        tw_sum = top_weights.sum()
        if tw_sum > 1e-15:
            top_weights = top_weights / tw_sum
        else:
            top_weights = np.ones(len(top_idx)) / len(top_idx)
        
        grad = np.zeros(n)
        for w, k in zip(top_weights, top_idx):
            grad += w * subgrad_conv_k(a, int(k), n)
        
        # Center the gradient (to maintain sum constraint approximately)
        grad_mean = grad.mean()
        grad_centered = grad - grad_mean
        
        # Update
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, n)
        
        score = fast_autocorr(a_new).max()
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a_new.copy()
            patience = 0
            step = min(step * 1.02, 2.0 / np.sqrt(n))
        else:
            patience += 1
            step *= 0.98
            if step < 1e-10:
                break
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):

    rng = np.random.default_rng(seed)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            
            # Try a few candidate values for a[j]
            candidates = [0.0, aj * 0.5, aj * 1.5]
            typical = norm / n
            candidates += [typical * 0.5, typical, typical * 2.0]
            candidates = sorted(set([max(0.0, c) for c in candidates]))
            
            best_x = aj
            best_local = fast_autocorr(a).max()
            
            for x in candidates:

                old_j = a[j]
                a[j] = x
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_j
                    continue
                a = a * norm / s
                sc = fast_autocorr(a).max()
                if sc < best_local:
                    best_local = sc
                    best_x = x
                a[j] = old_j
                s = a.sum()
                a = a * norm / s
            
            if best_x != aj:
                a[j] = best_x
                s = a.sum()
                a = a * norm / s
                sc = fast_autocorr(a).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()

                    a = best_a.copy()
        
        if not improved:

            if no_improve >= max_no_improve:

        else:
            no_improve = 0
    
    return best_a, best_score
