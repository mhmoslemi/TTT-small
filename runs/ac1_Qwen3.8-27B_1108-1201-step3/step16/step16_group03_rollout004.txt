This can be computed via FFT: FFT(w) * conj(FFT(a)) → IFFT → take the appropriate part. Or just FFT(w) * FFT(reflected a).

Actually, let me define it more carefully. Let `w_ext` be `w` padded to some length, and `a_ext` be `a` padded. Then `sum_k w_k * a[k-j]` is the (2n-2-j)-th element of `conv(w_ext, a_ext)`... this is getting complicated. Let me just use direct computation or FFT.

For efficiency, I'll use FFT. The computation is:
- `cross[j] = sum_k w_k * a[k-j]` for `j = 0, ..., n-1`

This is `cross = ifft(fft(w) * fft(a_reflected))` where `a_reflected` is `a` reversed and properly padded. Or more simply:

`cross[j] = sum_k w[k] * a[k-j]` where `a[x] = 0` if `x < 0` or `x >= n`.

This is the cross-correlation of `w` and `a`, which can be computed as:
`cross = ifft(fft(w, N) * conj(fft(a, N)))[:n]` for appropriate N >= 2n-1.

**Another idea: Use the fact that for the optimal solution, the peaks should all be equal (KKT conditions).**

At the optimum of `min max(c[k])` subject to `a >= 0, sum(a) = 1`:
- If `a[i] > 0` for all i (interior point), then the gradient of the Lagrangian is zero
- The active constraints are the peaks: `c[k] = M` for k in the active set
- The subgradient condition: `sum_{k in active} lambda_k * dc_k/da = mu * 1` (where mu is the Lagrange multiplier for the sum constraint)

This suggests that at the optimum, the weighted sum of the peak gradients should be constant.

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lse_objective(z, beta, n):
    """Compute log-sum-exp of convolution for softmax parametrization."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = fast_conv(a)
    max_c = c.max()
    if max_c < 1e-300:
        return 0.0, np.zeros(n)
    # Stable log-sum-exp
    shifted = beta * (c - max_c)
    log_sum = max_c * beta + np.log(np.sum(np.exp(shifted)))
    val = (2.0 * n / beta) * log_sum
    # Gradient
    w = np.exp(shifted)
    w = w / w.sum()  # softmax(beta * c)
    # cross[j] = sum_k w[k] * a[k-j]
    # This is cross-correlation
    fft_size = 1 << (2 * n - 1 + n - 1).bit_length()  # need at least 3n-2 for safety
    # Actually, w has length 2n-1, a has length n
    # cross[j] = sum_k w[k] * a[k-j], j = 0..n-1
    # This is conv(w, flip(a))[n-1-j]... let me just do it directly with FFT
    W = np.fft.rfft(w, 2 * fft_size)
    A = np.fft.rfft(a, 2 * fft_size)
    # cross[j] = sum_k w[k] * a[k-j] = (w * flip(a))[??]
    # Let's think: if we define b = flip(a) = a[n-1], a[n-2], ..., a[0]
    # Then conv(w, b)[j] = sum_k w[k] * b[j-k] = sum_k w[k] * a[n-1-(j-k)] = sum_k w[k] * a[n-1-j+k]
    # Hmm, this is getting confusing. Let me use a different approach.
    # cross[j] = sum_k w[k] * a[k-j]
    # Let m = k-j, then k = m+j, sum_m w[m+j] * a[m]
    # This is the cross-correlation R_wa[j]
    # R_wa = ifft(conj(fft(w)) * fft(a)) ... no
    # Actually R_wa[j] = sum_k w[k] * conj(a[k-j]) for complex, but for real:
    # R_wa[j] = sum_k w[k] * a[k-j]
    # In terms of convolution: R_wa = conv(w, a_flipped) where a_flipped[i] = a[-i] (with appropriate indexing)
    # More precisely, R_wa[j] = (w * a~)[j] where a~[i] = a[-i]
    # With FFT: FFT(R_wa) = FFT(w) * conj(FFT(a)) (for the cross-correlation)
    
    # Let me just use: R[j] = sum_k w[k] * a[k-j]
    # Pad both to length >= (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W_pad = np.zeros(N)
    W_pad[:len(w)] = w
    A_pad = np.zeros(N)
    A_pad[:n] = a
    # R[j] for j=0..n-1: this is the correlation
    # Using the property: ifft(fft(w) * conj(fft(a)))[j] = sum_k w[k] * conj(a[k-j])
    # For real a, this is sum_k w[k] * a[k-j] = R[j]
    R = np.fft.irfft(np.fft.rfft(W_pad) * np.conj(np.fft.rfft(A_pad)))[:n]
    
    dc_dz = 4.0 * n * a * (R - np.dot(w, c))
    return val, dc_dz

def lse_objective_neg(z, beta, n):
    """Negative of LSE objective for minimization."""
    val, grad = lse_objective(z, beta, n)
    return -val, -grad

def softmax_params(a):
    """Convert normalized non-negative a to softmax parameters z."""
    a = np.maximum(a, 1e-30)
    return np.log(a)

def try_lse_optimize(a0, time_limit, beta_start=5.0, beta_end=100.0, n_betas=5):
    """Optimize using L-BFGS with increasing beta."""
    n = len(a0)
    a = norm_sum1(a0)
    z = softmax_params(a)
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        # L-BFGS optimization
        try:
            result = minimize(
                fun=lambda z: lse_objective_neg(z, beta, n)[0],
                x0=z,
                jac=lambda z: lse_objective_neg(z, beta, n)[1],
                method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-6, 'maxfun': 2000}
            )
            z_new = result.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            m_new = fast_conv(a_new).max()
            if m_new < best_val:
                best_val = m_new
                best_a = a_new.copy()
                z = z_new
            elapsed = time.monotonic() - t0
            print(f"      LSE beta={beta:.1f}: M={m_new:.8f} (best={best_val:.8f}), t={elapsed:.1f}s")
        except Exception as e:
            print(f"      LSE error at beta={beta:.1f}: {e}")
    
    return best_a, best_val

def subgradient_nesterov(a0, time_limit, top_k=3, c0=5.0, seed=42):
    """Nesterov accelerated subgradient descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Soft subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 3.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * y[j]
        
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            x = best_a + rng.normal(0, 0.01, n) * best_a
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, seed=123):
    """Coordinate descent with targeted moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(10, len(ac))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            
            # Try transferring to random locations
            for _ in range(20):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18 or a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
