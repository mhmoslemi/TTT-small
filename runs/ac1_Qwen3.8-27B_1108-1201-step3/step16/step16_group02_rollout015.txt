        t_stage = t_remaining * time_frac

        def func_grad(u):
            val, grad = smooth_obj_and_grad(u, tau, n)
            return val, grad

        # Run L-BFGS
        result = minimize(
            func_grad, u0, method='L-BFGS-B',
            jac=True,  # func_grad returns (f, g)
            options={'maxiter': 2000, 'maxfun': 5000, 'iprint': -1,
                     'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )

        u0 = result.x
        a_new = _softmax(u0)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        m_new = fast_conv(a_new).max()
        if m_new < best_obj:
            best_obj = m_new
            best_a = a_new.copy()

        elapsed = time.monotonic() - t0
        if verbose:
            print(f"    L-BFGS stage {stage_idx} ({elapsed:.0f}s): tau={tau}, obj={best_obj:.8f}, score={2*n*best_obj:.6f}")

        # Check time
        if time.monotonic() - t0 > time_limit - 30:
            break

    return best_a, best_obj

def lp_fw_full(y, solver, max_iters=200, line_search=True, verbose=True):
    """LP Frank-Wolfe with all 2n-1 constraints."""
    n = len(y)
    a_curr = norm_sum1(y)
    best_a = a_curr.copy()
    best_m = fast_conv(a_curr).max()
    t0 = time.monotonic()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit_global:
            break

        # Build constraint matrix
        # A_mat[k, j] = 2 * y[k-j] if 0 <= k-j < n else 0
        A_mat = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                A_mat[k, lo:hi] = 2.0 * y[k - lo:k - hi + 1][::-1]

        # LP: min t s.t. A_mat @ a <= t, sum(a) = 1, a >= 0
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [
            A_mat @ a_var <= t_var,
            cp.sum(a_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        sg = g.sum()
        if sg < 1e-15:
            continue
        g = g / sg

        # Step size
        if line_search:
            # Try a few alpha values
            best_alpha = 0
            best_trial = fast_conv(a_curr).max()
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:
                trial = (1 - alpha) * a_curr + alpha * g
                trial = np.maximum(trial, 0)
                st = trial.sum()
                if st < 1e-15:
                    continue
                trial = trial / st
                m_trial = fast_conv(trial).max()
                if m_trial < best_trial:
                    best_trial = m_trial
                    best_alpha = alpha
            alpha = best_alpha
            if alpha == 0:
                alpha = 2.0 / (t + 2)
        else:
            alpha = 2.0 / (t + 2)

        a_new = (1 - alpha) * a_curr + alpha * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        a_curr = a_new
        m_curr = fast_conv(a_curr).max()
        if m_curr < best_m:
            best_m = m_curr
            best_a = a_curr.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, alpha={alpha:.3f}")

    return best_a, best_m


def nesterov_mirror(a0, time_limit, verbose=True):
    """Nesterov mirror descent with top-k subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        M = c.max()

        # Top-k subgradient
        top_k = min(80, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = np.maximum(c[top_idx], 1e-30)
        w = top_vals ** 4.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n, k_int + 1)
            if lo < hi:
                grad[lo:hi] += 2.0 * wi * y[k_int - lo:k_int - hi + 1][::-1]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        x_new = np.exp(log_a - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x
        x = x_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            stall = 0
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.5)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD t={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")

    return best_a, best_m


def local_search_polish(a0, time_limit, verbose=True):
    """Coordinate-wise local search with bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 20:
        improved = False
        # Random coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            a_i = a[i]
            if a_i < 1e-12:
                continue

            # Try transferring to random destinations
            n_tries = min(6, n)
            dsts = rng.integers(0, n, size=n_tries)
            best_frac = 0
            best_dst = -1
            best_trial_m = best_m

            for dst in dsts:
                if dst == i:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = a_i * frac
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[dst] += amt
                    if a_t[i] < -1e-15:
                        continue
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    m_t = fast_conv(a_t).max()
                    if m_t < best_trial_m - 1e-16:
                        best_trial_m = m_t
                        best_frac = frac
                        best_dst = dst

            if best_dst >= 0:
                amt = a_i * best_frac
                a[i] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = a / a.sum()

        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _softmax(u):
