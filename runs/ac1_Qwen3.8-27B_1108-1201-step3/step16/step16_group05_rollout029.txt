`A[k]` is the DFT of a real signal, so `A[N-k] = conj(A[k])`.
`(A*A)[N-k] = A[N-k] * A[N-k] = conj(A[k]) * conj(A[k]) = conj(A[k]^2) = conj((A*A)[k])`.

Yes! `A*A` IS conjugate-symmetric (for real input). So `irfft(A*A, N)` is well-defined and gives the linear convolution `conv(a, a)`.

def fast_conv_self(a):
    """Compute full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv_self(a)
    return 2.0 * n * c.max() / (s * s)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_constraint_matrix(a, lags):
    """Build the constraint matrix T where (T @ b)[i] = (a*b)[lags[i]]."""
    n = len(a)
    K = len(lags)
    T = np.zeros((K, n))
    for i, k in enumerate(lags):
        k = int(k)
        # (a*b)[k] = sum_j a[k-j] * b[j] where max(0,k-n+1) <= j <= min(n-1,k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            T[i, j_start:j_end+1] = a[k - j_end : k - j_start : -1]
    return T


def lp_fw_line_search(a_init, time_limit, solver, K=150, refresh_every=10,
                       alphas=None, verbose=True):
    """LP Frank-Wolfe with line search and adaptive active set."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    
    if alphas is None:
        alphas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    y = a.copy()
    
    # Pre-allocate variable
    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    
    iter_count = 0
    n_lp_solved = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute current convolution
        c = fast_conv_self(y)
        M = c.max()
        
        # Find active set
        if iter_count % refresh_every == 0:
            active_idx = np.argsort(c)[::-1][:K]
        
        # Build constraint matrix
        T = build_constraint_matrix(y, active_idx)
        
        # Solve LP
        constraints = [T @ b_var <= M]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_dir = np.asarray(b_var.value, dtype=np.float64)
        if b_dir is None:
            break
        b_dir = np.maximum(b_dir, 0)
        b_sum = b_dir.sum()
        if b_sum < 1e-15:
            break
        b_dir = b_dir / b_sum
        n_lp_solved += 1
        
        # Line search
        best_alpha = 0
        best_trial_val = obj_value(y)
        for alpha in alphas:
            y_trial = (1.0 - alpha) * y + alpha * b_dir
            y_trial = np.maximum(y_trial, 0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            val = obj_value(y_trial)
            if val < best_trial_val - 1e-15:
                best_trial_val = val
                best_alpha = alpha
        
        if best_alpha == 0:
            # Try smaller alphas
            for alpha in [0.005, 0.002, 0.001]:
                y_trial = (1.0 - alpha) * y + alpha * b_dir
                y_trial = np.maximum(y_trial, 0)
                sy = y_trial.sum()
                if sy < 1e-15:
                    continue
                y_trial = y_trial / sy
                val = obj_value(y_trial)
                if val < best_trial_val - 1e-15:
                    best_trial_val = val
                    best_alpha = alpha
        
        if best_alpha == 0:
            # Stuck, try a small random perturbation
            rng = np.random.default_rng(iter_count)
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = y[i] * 0.01
                y[i] -= amt
                y[j] += amt
                y = np.maximum(y, 0)
                y = y / y.sum()
            iter_count += 1
            continue
        
        # Update y
        y = (1.0 - best_alpha) * y + best_alpha * b_dir
        y = np.maximum(y, 0)
        sy = y.sum()
        y = y / sy
        
        # Update best
        val = obj_value(y)
        if val < best_val - 1e-12:
            best_val = val
            best_a = y.copy()
        
        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): obj={best_val:.6f}, alpha={best_alpha:.4f}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    LP-FW done: {iter_count} iters, {n_lp_solved} LPs solved, {elapsed:.0f}s, obj={best_val:.6f}")
    
    return best_a, best_val


def spectral_smooth(a, sigma=2.0):
    """Apply Gaussian smoothing in frequency domain to reduce peak."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    # Gaussian window in frequency (suppresses high freq to smooth in time)
    # Actually, we want to suppress the LOW freq components that create the peak
    # Let's just do a mild time-domain smoothing
    from scipy.ndimage import gaussian_filter1d
    a_smooth = gaussian_filter1d(a, sigma=sigma)
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s > 1e-15:
        a_smooth = a_smooth / s
    return a_smooth


def coordinate_polish(a_init, time_limit, verbose=False):
    """Coordinate descent: move mass from high-peak-contributors to low-impact positions."""
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        c = fast_conv_self(a)
        # Find top peak lags
        top_lags = np.argsort(c)[::-1][:5]
        
        # For each peak lag, find which positions contribute most
        peak_contrib = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_vals = np.arange(j_start, j_end + 1)
            a_vals = a[k_int - j_vals]
            peak_contrib[j_vals] += a[j_vals] * a_vals
        
        # Find best move: source (high contrib) to dest (low impact)
        sources = np.argsort(peak_contrib)[::-1][:20]
        improved = False
        
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Randomly try a few destinations
            n_try = min(n, 100)
            dsts = rng.choice(n, n_try, replace=False)
            
            for dst in dsts:
                if dst == src:
                    continue
                for frac in [0.01, 0.005, 0.002]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    val = obj_value(a_trial)
                    if val < best_val - 1e-12:
                        a = a_trial
                        best_val = val
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
            else:
                stall += 10
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord done: {elapsed:.0f}s, obj={best_val:.6f}")
    
    return best_a, best_val


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_interp = np.interp(x_new, x_old, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
