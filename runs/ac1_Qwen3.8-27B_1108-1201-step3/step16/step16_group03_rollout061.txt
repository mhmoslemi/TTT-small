    """grad[j] = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]"""
    # w has length 2n-1, a has length n
    # grad has length n
    # Use FFT for speed
    # pad a to length 2n-1
    a_pad = np.zeros(2*n-1)
    a_pad[:n] = a
    # We want: grad[j] = sum_i a[i] * w[j+i] for i=0..n-1, j=0..n-1
    # This is the cross-correlation of a and w
    # In numpy: np.correlate(w, a, mode='valid') gives sum_i w[i+j] * a[i] for j=0..(len(w)-len(a))
    # len(w) = 2n-1, len(a) = n, so 'valid' mode gives (2n-1-n+1) = n outputs
    # np.correlate(w, a, 'valid')[j] = sum_i w[i+j] * a[i] for i=0..n-1

def fast_conv(a):

    fft_size = 1 << (size - 1).bit_length()

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def hard_subgradient(y, k_star):
    """Exact subgradient of C[k_star] w.r.t. y."""
    n = len(y)
    k = int(k_star)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * y[j]
    return grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def power_law_init(n, alpha):
    """Generate a_j = (j+1)^{-alpha} normalized to sum 1."""
    j = np.arange(n, dtype=np.float64) + 1.0
    a = j ** (-alpha)
    a = a / a.sum()
    return a

def two_sided_init(n, alpha, offset=1.0):
    """Generate a_j = (min(j, n-1-j) + offset)^{-alpha} normalized."""
    j = np.arange(n, dtype=np.float64)
    d = np.minimum(j, n - 1 - j) + offset
    a = d ** (-alpha)
    a = a / a.sum()
    return a

def compute_gradient_from_weights(a, w):
    """Compute grad[j] = 2 * sum_i w[j+i] * a[i] using correlate."""
    return 2.0 * np.correlate(w, a, mode='valid')

def adam_log_descent(a0, time_limit, beta_start=5.0, beta_end=100.0,
                     lr=0.05, seed=42, verbose=True, anneal_period=15000):
    """Adam optimizer in log-space with log-sum-exp smooth objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    x = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    t0 = time.monotonic()
    eps = 1e-8
    b1, b2 = 0.9, 0.999
    
    while time.monotonic() - t0 < time_limit:
        batch_size = 50
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            t += 1
            
            # Beta annealing
            progress = min(t / float(anneal_period), 1.0)
            beta = beta_start + (beta_end - beta_start) * progress
            
            # Compute convolution
            c = fast_conv(a)
            c_max = c.max()
            
            # Track best (hard max)
            if c_max < best_m - 1e-14:
                best_m = c_max
                best_a = a.copy()

            exp_c = np.exp(beta * (c - c_max))
            w = exp_c / exp_c.sum()
            
            # Gradient in a-space
            grad_a = compute_gradient_from_weights(a, w)
            
            # Gradient in x-space
            grad_x = a * grad_a
            
            # Adam update
            lr_t = lr * (1.0 / (1.0 + 0.01 * t * 1e-4))
            m = b1 * m + (1.0 - b1) * grad_x
            v = b2 * v + (1.0 - b2) * (grad_x * grad_x)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            
            x_new = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            # Convert back to a-space
            x_new = x_new - x_new.max()
            a_new = np.exp(x_new)
            a_new = np.maximum(a_new, 1e-30)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            a = a_new
            x = np.log(a)
        
        if verbose and t % 2000 < batch_size:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AdamLog t={t} ({elapsed:.1f}s): beta={beta:.1f}, J={J:.6f}")

def nesterov_subgrad(a0, time_limit, top_k=3, c0=5.0, seed=42, verbose=True):
    """Nesterov accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 5000
    current_top_k = top_k
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if stall > max_stall // 2:
            current_top_k = min(current_top_k + 1, 10)
        elif stall > max_stall:
            current_top_k = top_k
        
        # Soft subgradient
        k_max = min(current_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 3.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            grad += wi * hard_subgradient(y, k)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            x_restarted = best_a + rng.normal(0, 0.01, n) * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall = 0
    
    return best_a, best_m

def lp_frank_wolfe_cutting_plane(a0, time_limit, solver, verbose=True):
    """LP Frank-Wolfe with cutting plane approach for speed."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    # Start with a small set of active constraints
    ac = fast_conv(y)
    top_k_init = min(20, len(ac))
    active_constraints = set(np.argsort(ac)[::-1][:top_k_init].tolist())
    
    iter_count = 0
    max_active = min(2 * n - 1, 500)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix only for active constraints
        active_list = sorted(active_constraints)
        T = np.zeros((len(active_list), n))
        for idx, k in enumerate(active_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T[idx, i] = y[k_int - i]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Check if new solution violates non-active constraints
        if len(active_constraints) < max_active:
            b_conv = fast_conv(direction)
            for k in range(len(b_conv)):

                    active_constraints.add(k)

                        break

        alphas = [2.0 / (t + 2.0), 0.5, 0.3, 0.2, 0.1, 0.05]
        best_alpha = alphas[0]

            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        iter_count += 1
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-Cut iter={t} ({elapsed:.1f}s): J={J:.6f}, active={len(active_constraints)}")

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Generate power-law initializations and find best alpha
    if best_seq is not None:
        n0 = len(best_seq)
    else:
        n0 = 2400
    
    print(f"\n=== Phase 0: Initialization (n={n0}) ===")
    init_candidates = []
    
    # Power law sweep
    for alpha in [0.35, 0.40, 0.42, 0.44, 0.45, 0.46, 0.47, 0.48, 0.50, 0.52, 0.55]:
        a = power_law_init(n0, alpha)
        sc = evaluate_safe(a)
        if np.isfinite(sc):
            init_candidates.append((sc, a, f"powerlaw_a{alpha}"))
    
    # Two-sided
    for alpha in [0.3, 0.4, 0.5]:
        a = two_sided_init(n0, alpha)
        sc = evaluate_safe(a)
        if np.isfinite(sc):
            init_candidates.append((sc, a, f"twosided_a{alpha}"))
    
    # Previous solution
    if parent is not None:
        a_norm = parent / parent.sum()
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc):
            init_candidates.append((sc, a_norm, "parent"))
    
    if init_candidates:
        init_candidates.sort(key=lambda x: x[0])
        best_init_score, best_init_a, best_init_name = init_candidates[0]
        print(f"  Best init: {best_init_name}, score={best_init_score:.6f}")
        if best_seq is None or best_init_score < best_score:
            best_score = best_init_score
            best_seq = best_init_a
    else:
        best_seq = power_law_init(n0, 0.45)
        best_score = evaluate_safe(best_seq)
        print(f"  Default init score: {best_score:.6f}")
    
    print(f"  Current best: {best_score:.6f}")
    
    # === Phase 1: Adam log-descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam Log-Descent n={len(best_seq)} ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_log_descent(a_in, t1, beta_start=5.0, beta_end=80.0,
                                    lr=0.05, seed=42, verbose=True, anneal_period=12000)
        try_update(a_out, f"adamlog_n{len(a_out)}")
    
    # === Phase 2: Nesterov subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Nesterov Subgrad n={n_cur} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgrad(a_in, t2, top_k=3, c0=5.0, seed=7, verbose=True)
        try_update(a_out, f"nest_n{n_cur}")
    
    # === Phase 3: LP Frank-Wolfe with cutting planes ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Cutting Plane n={n_cur} ===")
    t3 = min(remaining * 0.25, 380.0)
    if t3 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_cutting_plane(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")
    
    # === Phase 4: Adam log-descent round 2 (with higher beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Adam Log round2 n={n_cur} ===")
    t4 = min(remaining * 0.15, 220.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_log_descent(a_in, t4, beta_start=20.0, beta_end=200.0,
                                    lr=0.02, seed=99, verbose=True, anneal_period=5000)
        try_update(a_out, f"adamlog2_n{n_cur}")
    
    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord Descent n={n_cur} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_cur}")
    
    # === Phase 6: Try different n values ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        for delta in [100, -100, 200, -200]:
            n_try = n_best + delta
            if n_try < 800 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 6: n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(remaining * 0.35, 150.0)
            if t6 > 20:
                a_out, _ = adam_log_descent(a_long, t6, beta_start=10.0, beta_end=100.0,
                                            lr=0.03, seed=55 + n_try % 100, verbose=False, anneal_period=5000)
                try_update(a_out, f"adamlog_n{n_try}")
                
                remaining = time_budget - (time.monotonic() - t_global)
                t6b = min(remaining * 0.15, 60.0)
                if t6b > 10:
                    a_out2, _ = coordinate_polish(norm_sum1(best_seq), t6b, verbose=False, seed=77)
                    try_update(a_out2, f"coord_n{n_try}")
    
    # === Phase 7: Final LP if time and n permit ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 3000 and remaining > 60:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_cutting_plane(a_in, t7, solver, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.8, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
