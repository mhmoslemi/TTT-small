"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Improved LP-based Frank-Wolfe with backtracking and adaptive top-K
2. Nesterov mirror descent with adaptive soft subgradient
3. Length sweeping with interpolation

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    """Find the best available LP solver."""
    for s in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_frank_wolfe(a0, time_limit, solver, top_K=300, verbose=True):
    """
    LP-based Frank-Wolfe with backtracking line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Preallocate
    y = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < 1e-16:
            break
        
        # Build constraint matrix (top-K most binding constraints)
        # The constraint is: (y * b)[k] <= M_curr for selected k
        # (y * b)[k] = sum_i y[k-i] * b_i (valid range)
        # So row k of the matrix has entries y[k-i] for valid i
        
        K_sel = min(top_K, 2 * n - 1)
        # Select the K most binding constraints (closest to M_curr)
        margins = M_curr - ac
        selected_k = np.argsort(margins)[:K_sel]
        
        # Build the constraint matrix for selected constraints
        # A[k, i] = y[k-i] if 0 <= k-i < n, else 0
        A_mat = np.zeros((K_sel, n))
        for idx, k in enumerate(selected_k):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                A_mat[idx, i_start:i_end] = y[k_int - np.arange(i_start, i_end)]
        
        # Solve LP: maximize sum(b) subject to A_mat @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A_mat @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking line search
        best_alpha = 2.0 / (t + 2.0)  # Default Nesterov step
        best_obj = fast_autocorr(y).max()
        
        for alpha_try in [2.0/(t+2.0), 4.0/(t+2.0), 8.0/(t+2.0), 1.0/(t+2.0), 1.5/(t+2.0)]:
            if alpha_try >= 1.0:
                alpha_try = 0.9
            y_try = (1.0 - alpha_try) * y + alpha_try * g_dir
            y_try = np.maximum(y_try, 0.0)
            s_try = y_try.sum()
            if s_try < 1e-15:
                continue
            y_try = y_try / s_try
            m_try = fast_autocorr(y_try).max()
            if m_try < best_obj - 1e-15:
                best_obj = m_try
                best_alpha = alpha_try
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        y = y_new
        m_new = fast_autocorr(y).max()
        
        if m_new < best_m:
            best_m = m_new
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, M={M_curr:.8f}")
    
    return best_a, best_m

def softmax_subgradient_descent(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=8000, verbose=True):

            x = x_r

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with high top-K constraints and adaptive step sizes
2. Nesterov-accelerated mirror descent with soft subgradient
3. Length sweeping with interpolation
4. Pair-move coordinate descent for final polish

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    """Find the best available LP solver."""
    for s in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_frank_wolfe(a0, time_limit, solver, top_K=400, verbose=True, step_mode="nesterov"):
    """
    LP-based Frank-Wolfe with adaptive top-K and step size strategies.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    lp_fail_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        ac = fast_autocorr(y)
        M_curr = float(ac.max())
        
        if M_curr < 1e-16:
            break
        
        # Select top-K most binding constraints
        K_sel = min(top_K, 2 * n - 1)
        margins = M_curr - ac
        selected_k = np.argsort(margins)[:K_sel]
        
        # Build constraint matrix
        A_mat = np.zeros((K_sel, n))
        for idx in range(K_sel):
            k_int = int(selected_k[idx])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                A_mat[idx, i_start:i_end] = y[k_int - np.arange(i_start, i_end)]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        rhs = M_curr * np.ones(K_sel)
        constraints = [A_mat @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 20:
                break
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 20:
                break
            continue
        
        lp_fail_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Step size
        if step_mode == "nesterov":
            alpha = 2.0 / (t + 2.0)
        elif step_mode == "fixed":
            alpha = 0.15
        elif step_mode == "aggressive":
            alpha = min(0.4, 4.0 / (t + 2.0))
        else:
            alpha = 2.0 / (t + 2.0)
        
        # FW step
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        y = y_new
        m_new = float(fast_autocorr(y).max())
        
        if m_new < best_m:
            best_m = m_new
            best_a = y.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, M={M_curr:.8f}, K={K_sel}")
    
    return best_a, best_m

    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = float(fast_autocorr(x).max())
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 120)
            cur_beta = max(cur_beta - 0.3, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    
    return best_a, best_m

def pair_move_descent(a0, time_limit, verbose=True, rng_seed=123):
    """
    Pair-move coordinate descent with sensitivity-based destination selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = float(ac.max())
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Score destinations
            scores = np.zeros(n)
            n_sens = min(8, len(top_k))
            for k2 in top_k[:n_sens]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                for d in range(s2, e2):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    d2 = d - k2i
                    if 0 <= d2 < n:
                        scores[d] += a[d2]
            
            scores[src] = np.inf
            dst = int(np.argmin(scores))
            if dst == src or scores[dst] == np.inf:
                dst = int(rng.integers(n))
                if dst == src:
                    continue
            
            for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = float(fast_autocorr(a_trial).max())
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = float(fast_autocorr(a).max())
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall > 0 and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Pair ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Pair final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-strategy LP-FW + MD + length sweep + pair moves.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver()
    print(f"Solver: {solver}, Budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    # === Phase 1: LP-FW with aggressive parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-FW aggressive (n={n_best}) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t1, solver, top_K=500, verbose=True, step_mode="aggressive")
        try_update(a_out, "lp1_agg")
    
    # === Phase 2: LP-FW with Nesterov step ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW nesterov (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver, top_K=400, verbose=True, step_mode="nesterov")
        try_update(a_out, "lp2_nest")
    
    # === Phase 3: MD polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MD polish (n={n_best}) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_out, _ = mirror_descent_optimize(norm_sum1(best_seq), t3, top_k=80, beta=3.5, c0=3.0, restart_every=6000, verbose=True, seed=77)
        try_update(a_out, "md3")
    
    # === Phase 4: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Length sweep (base n={n_best}) ===")
    lengths_to_try = [n_best - 400, n_best - 200, n_best + 200, n_best + 400, n_best + 600]
    lengths_to_try = [l for l in lengths_to_try if 1500 <= l <= 5000 and l != n_best]
    
    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t4a = min(remaining * 0.18, 100.0)
        if t4a > 10:
            a_out, _ = mirror_descent_optimize(a_long, t4a, top_k=50, beta=4.0, c0=4.0, restart_every=5000, verbose=False, seed=new_n % 1000)
            try_update(a_out, f"md_n{new_n}")
    
    # === Phase 5: Pair moves ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair moves (n={n_best}) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        a_out, _ = pair_move_descent(norm_sum1(best_seq), t5, verbose=True, rng_seed=555)
        try_update(a_out, "pair5")
    
    # === Phase 6: Final LP-FW with high K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW high-K (n={n_best}) ===")
    if remaining > 40 and n_best <= 4500:
        t6 = min(remaining * 0.40, 300.0)
        if t6 > 20:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6, solver, top_K=600, verbose=True, step_mode="aggressive")
            try_update(a_out, "lp6_high")
    
    # === Phase 7: Final MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 7: Final MD (n={n_best}) ===")
        t7 = min(remaining * 0.50, 120.0)
        if t7 > 10:
            a_out, _ = mirror_descent_optimize(norm_sum1(best_seq), t7, top_k=100, beta=3.0, c0=2.5, restart_every=5000, verbose=True, seed=2024)
            try_update(a_out, "md7")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
