"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact-subgradient Frank-Wolfe with Nesterov acceleration (no LP needed, O(n*|K|) per step)
2. Adaptive step size with backtracking for faster convergence
3. Hybrid FW + projected gradient for interior movement
4. Multi-restart strategy with diverse initializations

6. Final coordinate polish
"""

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fw_exact_subgradient(a, tol_frac=0.001):
    """Compute optimal FW subgradient: g[i] = 2*min_{k active} a[k-i]."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-15:
        return np.zeros(n), M
    tol = tol_frac * M
    active = np.where(ac >= M - tol)[0]
    g = np.full(n, np.inf)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        vals = 2.0 * a[j]
        g[idx] = np.minimum(g[idx], vals)
    g[~np.isfinite(g)] = 0.0
    return g, M

def compute_smoothed_gradient(a, top_k=200, beta=30.0):
    """Smoothed gradient of max autocorrelation using log-sum-exp approximation."""
    n = len(a)
    ac = fast_autocorr(a)
    num = len(ac)
    k = min(top_k, num)
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        ki_int = int(ki)
        i_start = max(0, ki_int - n + 1)
        i_end = min(n, ki_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = ki_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, ac.max()

def accelerated_fw(a0, time_limit, verbose=True, seed=42):
    """Nesterov-accelerated FW with exact subgradient + hybrid PG steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    y = a.copy()
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t % 50 == 0:
            # Hybrid step: use smoothed gradient for interior movement
            g_smooth, _ = compute_smoothed_gradient(y, top_k=150, beta=20.0)
            g_norm = np.linalg.norm(g_smooth)
            if g_norm > 1e-10:
                eta = 0.5 / (np.sqrt(t) + 1.0)
                x_pg = project_simplex(y - eta * g_smooth)
            else:
                x_pg = y.copy()
        else:
            # FW step with exact subgradient
            g, M_curr = fw_exact_subgradient(y, tol_frac=0.001)
            j = int(np.argmin(g))
            d = np.zeros(n)
            d[j] = 1.0
            alpha = 2.0 / (t + 2.0)
            x_pg = (1.0 - alpha) * x + alpha * d

        x_new = np.maximum(x_pg, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y_new = x_new + beta_t * (x_new - x)
        y_new = project_simplex(y_new)

        x_prev = x
        x = x_new
        y = y_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000 and stall % 500 == 0:
            noise_scale = min(0.03, 0.002 * (stall // 3000))
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            y = x_re
            stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def projected_gradient_polish(a0, time_limit, verbose=True, seed=77):
    """Projected gradient with adaptive beta for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    c = 1.8

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 10000.0)
        beta = 15.0 + (5.0 - 15.0) * frac
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)
        g, _ = compute_smoothed_gradient(y, top_k=200, beta=beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        eta = c / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall % 2000 == 0 and stall > 0:
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            stall = 0
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False
        peak_contrib = np.zeros(n)
        for k in top_k[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_contrib[idx] += a[idx] * a[j_idx]
        sources = np.argsort(peak_contrib)[::-1][:20]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            best_dst = -1
            best_score = 0.0
            n_try = min(n, 300)
            candidates = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)
            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_k[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_score:
                    best_score = score
                    best_dst = dst
            if best_dst < 0:
                continue
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _peak_deflate(base, rng):
    """Reduce peaks in the sequence and boost valleys."""
    n = len(base)
    a = base.copy()
    ac = fast_autocorr(a)
    M = ac.max()
    # Find positions contributing to peaks
    peak_pos = np.zeros(n)
    top_lags = np.argsort(ac)[::-1][:10]
    for k in top_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        peak_pos[idx] += a[idx] * a[j_idx]
    # Deflate high-peak positions
    threshold = np.percentile(peak_pos, 70)
    mask = peak_pos > threshold
    a[mask] *= 0.9
    # Boost low positions
    low_mask = peak_pos < np.percentile(peak_pos, 30)
    a[low_mask] *= 1.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return base.copy()
    return a / s

def _bimodal(n, rng):
    """Generate a bimodal sequence."""
    x = np.linspace(0, 1, n)
    centers = [0.3 + rng.uniform(-0.05, 0.05), 0.7 + rng.uniform(-0.05, 0.05)]
    sigma = rng.uniform(0.08, 0.15)
    f = np.zeros(n)
    for c in centers:
        f += np.exp(-((x - c) ** 2) / (2 * sigma ** 2))
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s
