def subgrad_multi_peak(a, ac, top_k=10, tau=0.1):
    """
    Compute multi-peak subgradient using softmax weighting over top-k peaks.
    Returns (gradient, peak_value).
    """

def interpolate_to_length(f, new_n):
    """Resample sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated mirror descent (log-space) with backtracking line search
2. Softmax multi-peak subgradient with adaptive temperature
3. Periodic spectral power-equalization
4. Nesterov acceleration with adaptive restart
5. Coordinate descent local polish
6. Multi-length exploration
Key improvements: backtracking ensures progress, spectral equalization
provides complementary direction, adaptive temperature sharpens focus
over time, and multi-length search finds optimal scaling.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Full linear autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def multi_peak_subgrad(a, ac, top_k, tau):
    """Softmax-weighted subgradient over top-k autocorrelation peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n), 0.0
    scale = max_val * max(tau, 1e-6)
    expv = np.exp((top_vals - max_val) / scale)
    w = expv / expv.sum()
    g = np.zeros(n)
    for ii in range(k_max):
        k = int(top_idx[ii])
        wi = w[ii]
        if wi < 1e-8:
            continue
        i_s = k - (n - 1)
        if i_s < 0:
            i_s = 0
        i_e = k + 1
        if i_e > n:
            i_e = n
        idx = np.arange(i_s, i_e)
        j = k - idx
        g[idx] += 2.0 * wi * a[j]
    return g, max_val


def mirror_update(a, g, eta):
    """Log-space mirror descent step with renormalization."""
    la = np.log(np.maximum(a, 1e-30)) - eta * g
    la -= la.max()
    a_new = np.exp(la)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_equalize(a, alpha):
    """One step of power spectrum flattening toward uniform."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    pw = np.abs(A) ** 2
    ac_pw = np.maximum(pw[1:], 1e-20)
    target = np.exp(np.mean(np.log(ac_pw)))
    scale = np.ones(len(A))
    scale[1:] = (target / ac_pw) ** alpha
    A2 = A * scale
    a_new = np.fft.irfft(A2, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def run_amd(a0, time_limit, top_k, tau_init, tau_min, c0,
            spectral_every, spectral_alpha, max_stall, rng_seed, verbose=True):
    """
    Accelerated Mirror Descent with backtracking, adaptive temperature,
    spectral equalization, and Nesterov restart.
    Returns (best_a, best_max_autocorr).
    """
    n = len(a0)
    a = a0.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    t_count = 1
    eta = c0 * np.sqrt(n)
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        t_count += 1
        total_iters += 1

        # Nesterov extrapolation
        beta = (t_count - 2.0) / (t_count + 0.5)
        if t_count == 1:
            beta = 0.0
        y = x + beta * (x - x_prev)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = np.maximum(y, 1e-15) / sy

        # Autocorrelation and subgradient
        ac_y = fast_autocorr(y)
        M_y = ac_y.max()

        # Adaptive temperature
        progress = min(1.0, total_iters / max(1, int(time_limit * 15)))
        tau = tau_init + (tau_min - tau_init) * progress

        g, _ = multi_peak_subgrad(y, ac_y, top_k=cur_top_k, tau=tau)

        best_eta = eta * 0.01
        best_M = M_y + 1.0
        for attempt in range(10):
            a_new = mirror_update(y, g, eta)
            M_new = fast_autocorr(a_new).max()
            if M_new < M_y + 1e-12:
                best_eta = eta
                best_M = M_new
                break
            eta *= 0.5

        a_trial = mirror_update(y, g, eta)

        x = a_trial
        M_x = best_M if best_M < M_y + 1.0 else fast_autocorr(x).max()

        # Track best

            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic spectral equalization

            x_eq = spectral_equalize(x, spectral_alpha)
            M_eq = fast_autocorr(x_eq).max()
            if M_eq < best_m - 1e-15:

                stall = 0
            if M_eq < M_x:
                x = x_eq
                x_prev = x_eq

            stall = 0
            noise = rng.normal(0, 0.008, n)
            x_restart = np.maximum(best_a * (1.0 + noise), 1e-15)
            x_restart = x_restart / x_restart.sum()
            x = x_restart
            x_prev = x_restart
            t_count = 1
            eta = c0 * np.sqrt(n)

        # Gentle step size decay

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_m
            print(f"  AMD t={total_iters} ({elapsed:.0f}s): obj={obj:.6f} stall={stall} eta={eta:.3f}")

    return best_a, best_m

    a = a0.copy()

        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M

            stall = 0
        else:
            stall += 1

        # Contributors to peak
        i_s = max(0, k_peak - n + 1)
        i_e = min(n, k_peak + 1)
        if i_s >= i_e:

        idx = np.arange(i_s, i_e)
        j_idx = k_peak - idx

        max_c = contrib.max()
        if max_c < 1e-20:

        pc = int(np.argmax(contrib))
        src = int(idx[pc])
        src2 = int(j_idx[pc])

        # Sensitivity from top peaks
        top_peaks = np.argsort(ac)[::-1][:10]
        sens = np.zeros(n)
        for k in top_peaks:
            ki = int(k)
            s2 = max(0, ki - n + 1)
            e2 = min(n, ki + 1)
            ii = np.arange(s2, e2)
            jj = ki - ii
            sens[ii] += a[jj]
            sens[jj] += a[ii]

        improved = False
        cold = np.argsort(sens)
        n_cold = min(len(cold), n // 3)
        cold_subset = cold[:n_cold].copy()
        rng.shuffle(cold_subset)

        # Try from both src and src2
        for source in [src, src2]:
            if a[source] < 1e-12:
                continue
            for dst in cold_subset:
                if dst == source:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                    amt = a[source] * frac
                    if amt < 1e-18:
                        continue
                    at = a.copy()
                    at[source] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    s = at.sum()
                    if s < 1e-15:
                        continue
                    at = at / s
                    M2 = fast_autocorr(at).max()
                    if M2 < M - 1e-15:
                        a = at
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Escape via random perturbation
            n_p = max(3, n // 200)
            for _ in range(n_p):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Coord ({elapsed:.0f}s): obj={2.0*n*best_m:.6f} stall={stall}")

    return best_a, best_m


def interpolate_to_length(f, new_n):
    """Resample sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss


def safe_evaluate(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multiple phases."""
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = safe_evaluate(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2.0 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = safe_evaluate(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Primary AMD ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: AMD n={n0} (budget {remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_out, _ = run_amd(best_seq, t1, top_k=20, tau_init=0.5, tau_min=0.03,
                           c0=2.5, spectral_every=250, spectral_alpha=0.15,
                           max_stall=4000, rng_seed=1)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [amd1_n{n0}]: {sc:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 2: AMD variant (different hyperparams) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD variant n={n_best} (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = run_amd(best_seq, t2, top_k=30, tau_init=0.7, tau_min=0.02,
                           c0=3.5, spectral_every=200, spectral_alpha=0.25,
                           max_stall=6000, rng_seed=2)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [amd2_n{n_best}]: {sc:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget {remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = coord_descent(best_seq, t3, rng_seed=3)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [coord_n{n_best}]: {sc:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 4: AMD polish (sharper focus) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD polish n={n_best} (budget {remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        a_out, _ = run_amd(best_seq, t4, top_k=10, tau_init=0.2, tau_min=0.01,

                break

    if remaining > 20:

        if t6 > 10:
            a_out, _ = coord_descent(best_seq, t6, rng_seed=6)
            sc = safe_evaluate(a_out)

                best_seq = a_out.copy()
                print(f"  *** NEW BEST [coord_final_n{n_best}]: {sc:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 7: Final AMD sub-polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.5, 120.0)
        if t7 > 10:
            a_out, _ = run_amd(best_seq, t7, top_k=8, tau_init=0.1, tau_min=0.01,
                               c0=1.0, spectral_every=1000, spectral_alpha=0.03,

            sc = safe_evaluate(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [amd_final_n{n_best}]: {sc:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Final validation
    if best_seq is not None:
        fs = safe_evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
