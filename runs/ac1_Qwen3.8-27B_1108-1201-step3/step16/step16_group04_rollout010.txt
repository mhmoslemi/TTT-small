"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed gradient descent in log-space with Nesterov acceleration
2. Adaptive log-sum-exp smoothing (coarse-to-fine peak focus)

4. Peak-targeted coordinate descent for local refinement
5. Multi-length exploration with interpolation
Key insight: gradient of weighted autocorrelation is a convolution, computable in O(n log n).

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_grad_from_weights(w, a):
    """
    Compute grad_j = 2 * sum_m w[j+m] * a[m] for j=0,...,n-1.
    This is 2 * correlate(w, a, 'valid') computed via FFT.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    len_w = len(w)
    fft_size = 1 << (len_w + n - 1 - 1).bit_length()
    W = np.fft.rfft(w.astype(np.float64), fft_size)
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    # correlate(w, a, 'valid')[j] = sum_m w[j+m]*a[m]
    # = ifft(FFT(w) * conj(FFT(a)))[offset]
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    # The valid part: for cross-correlation of sequences of length len_w and n,
    # 'valid' output has length len_w - n + 1 = n (since len_w = 2n-1)
    # The valid part starts at index (n-1)
    return 2.0 * corr[n - 1: n - 1 + n]


def normalize_a(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_precondition(a, n_iter=50, alpha=0.5, blend=0.7):
    """
    Iteratively flatten the power spectrum:
    1. Compute FFT
    2. Modify magnitude: |A_new| = |A| / (|A|^2 + eps)^alpha  (soft flattening)
    3. Inverse FFT, clip negatives, normalize
    4. Blend with previous to maintain stability
    """
    a = normalize_a(a.copy())
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter):
        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        
        # Flatten: reduce dynamic range of magnitude
        mag_max = mag.max()
        mag_min = mag.min() + 1e-30
        # Target: make magnitude more uniform
        # Use power compression: mag_new = mag / (mag^2/max_mag)^alpha
        # Equivalent: mag_new = mag^(1-alpha) * max_mag^alpha
        mag_new = mag / (mag / mag_max) ** alpha
        # Alternative: mag_new = mag * (mag_min / mag) ** alpha = mag^(1-alpha) * mag_min^alpha
        # Let's use: mag_new = mag / ((mag/mag_mean)^2)^beta for compression
        mag_mean = mag.mean() + 1e-30
        ratio = mag / mag_mean
        mag_new = mag / ratio ** (2 * alpha)
        
        A_new = mag_new * phase
        a_new = np.real(np.fft.ifft(A_new, fft_size)[:n])
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        a = blend * a + (1 - blend) * a_new
    
    return a

        sy = y.sum()
        if sy < 1e-30:

        if s_new > 1e-30:
            a_new = a_new / s_new

                b = np.log(a + 1e-30)
                b_prev = b.copy()
            
            # Also try larger perturbation
            if stall_count > n_stall_threshold * 2:
                pert = rng.normal(0, 0.1, n)
                a_trial = best_a * (1.0 + pert)
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m * 1.1:
                    a = a_trial
                    b = np.log(a + 1e-30)
                    b_prev = b.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.1f}s): J={best_J:.6f}, beta={beta:.1f}, stall={stall_count}")
    
    return best_a, best_m


def smoothed_gradient_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    More aggressive variant with:
    - Top-1 peak focusing (hard)
    - Larger step sizes
    - Periodic spectral re-preconditioning
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_J = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    b = np.log(a + 1e-30)
    b_prev = b.copy()
    stall_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        if t > 1:
            momentum = 0.9 * (1.0 - 1.0 / (t + 1.0))
        else:
            momentum = 0.0
        y_b = b + momentum * (b - b_prev)
        y = np.exp(y_b - y_b.max())
        sy = y.sum()
        y = y / sy if sy > 1e-30 else a.copy()
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
            best_J = 2.0 * n * best_m
            stall_count = 0
        else:
            stall_count += 1
        
        # Hard top-3 weighting + soft tail
        K = 3
        top_idx = np.argpartition(ac, -K)[-K:]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        
        # Combine: top peaks get weight 1, rest get small weight
        w_full = np.full(len(ac), 0.01)
        max_top = top_vals.max()
        w_full[top_idx] = 1.0
        # Also include mid-range peaks
        mid_idx = np.where((ac > 0.5 * max_top) & (ac < max_top))[0]
        w_full[mid_idx] = 0.3
        
        grad = fast_grad_from_weights(w_full, y)
        grad = np.maximum(grad, 0.0)
        
        # Step size
        max_ga = np.max(grad * y) + 1e-30
        eta = 1.5 / (max_ga + 1.0)
        
        b_new = b - eta * grad * y
        b_new = np.maximum(b_new, -50.0)
        y_new = np.exp(b_new - b_new.max())
        a_new = y_new / y_new.sum()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-30:
            a_new = a_new / s
        
        b = np.log(a_new + 1e-30)
        b_prev = np.log(a + 1e-30)
        a = a_new
        
        # Periodic spectral re-preconditioning
        if t % 5000 == 0 and stall_count > 100:
            a_spec = spectral_precondition(a, n_iter=10, alpha=0.3, blend=0.5)
            m_spec = fast_autocorr(a_spec).max()
            if m_spec < best_m:
                best_m = m_spec
                best_a = a_spec.copy()
                best_J = 2.0 * n * best_m
                a = a_spec
                b = np.log(a + 1e-30)
                b_prev = b.copy()
                stall_count = 0
        
        # Restart
        if stall_count > 300:
            stall_count = 0
            for scale in [0.05, 0.1, 0.2]:
                pert = rng.normal(0, scale, n)
                a_trial = best_a * (1.0 + pert)
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m * (1.0 + scale):
                    a = a_trial
                    b = np.log(a + 1e-30)
                    b_prev = b.copy()
                    break
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD2 t={t} ({elapsed:.1f}s): J={best_J:.6f}, stall={stall_count}")
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=99):
    """
    Peak-targeted coordinate descent.
    For each top peak, identify the dominant contributing coordinate pair,
    then move mass to reduce the peak.
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_J = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        K = min(10, len(ac))
        top_peaks = np.argpartition(ac, -K)[-K:]
        top_peaks = top_peaks[np.argsort(ac[top_peaks])[::-1]]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find contributing pairs
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            js = np.arange(j_start, j_end + 1)
            pairs = a[js] * a[k_int - js]
            
            if pairs.max() < 1e-20:
                continue
            
            # Find the dominant pair
            dom_idx = js[np.argmax(pairs)]
            partner = k_int - dom_idx
            
            # Strategy: reduce the larger of the pair
            if a[dom_idx] >= a[partner]:
                src = dom_idx
            else:
                src = partner
            
            # Find a good destination: a coordinate that doesn't contribute much to top peaks
            # Score each potential destination by how much it would increase top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            # Prefer destinations not in any peak's support
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            
            # Randomly sample from candidates
            n_sample = min(100, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:n_sample]]
            
            best_dst = -1
            best_penalty = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                # Penalty: contribution to top peaks if we add mass here
                penalty = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    penalty += add * 2.0
                if penalty < best_penalty:
                    best_penalty = penalty
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            
            # Try different move fractions
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    best_J = 2.0 * n * best_m
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Global random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc ({elapsed:.1f}s): J={best_J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / ss


def raised_cosine_init(n):
    """Initialize with raised cosine shape."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def flat_target_init(n):
    """Initialize targeting flat autocorrelation via spectral method."""
    # Start with random, then spectrally precondition
    rng = np.random.default_rng(7)
    a = np.abs(rng.normal(0, 1, n)) + 0.1
    a = a / a.sum()
    a = spectral_precondition(a, n_iter=80, alpha=0.7, blend=0.5)
    return a


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(42)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_a(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Generate initializations
    if best_seq is None:
        n_base = 2400
        best_seq = raised_cosine_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init (raised cosine): n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # Try alternative initializations
    inits = [best_seq.copy()]
    
    # Spectral flat init
    a_flat = flat_target_init(n0)
    inits.append(a_flat)
    
    # Raised cosine
    a_rc = raised_cosine_init(n0)
    inits.append(a_rc)
    
    # Random
    a_rand = np.abs(rng.normal(0, 1, n0)) + 0.1
    a_rand = a_rand / a_rand.sum()
    inits.append(a_rand)
    
    # Evaluate all initializations
    for i, ini in enumerate(inits):
        sc = evaluate_safe(ini)
        label = ["parent", "spectral", "cosine", "random"][i] if i < 4 else f"init{i}"
        print(f"  Init {label}: n={len(ini)}, score={sc:.6f}")
        try_update(ini, label)
    
    # === Phase 1: Spectral preconditioning on best init ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 1: Spectral Preconditioning (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        a_spec = spectral_precondition(best_seq, n_iter=100, alpha=0.6, blend=0.5)
        try_update(a_spec, "spec_prec")
        
        # Also try on the best init
        best_init = min(inits, key=lambda x: evaluate_safe(x))
        a_spec2 = spectral_precondition(best_init, n_iter=100, alpha=0.8, blend=0.4)
        try_update(a_spec2, "spec_prec2")
    
    # === Phase 2: Main SGD optimization ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Smoothed Gradient Descent (n={n_best}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 30:
        a_in = normalize_a(best_seq)
        a_out, _ = smoothed_gradient_descent(
            a_in, t2, top_k_frac=0.03, beta_init=80.0, beta_min=10.0,
            c0=3.0, gamma=0.999, verbose=True, rng_seed=42
        )
        try_update(a_out, f"sgd_n{n_best}")
    
    # === Phase 3: Second SGD variant (hard peak focus) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SGD v2 (hard peaks) n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 20:
        a_in = normalize_a(best_seq)
        a_out, _ = smoothed_gradient_descent_v2(a_in, t3, verbose=True, rng_seed=123)
        try_update(a_out, f"sgd2_n{n_best}")
    
    # === Phase 4: Coordinate descent polish ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate Descent n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = normalize_a(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Try different length ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    n_alt = n_best + 600 if n_best <= 3000 else n_best + 400
    n_alt = max(1000, min(n_alt, 5000))
    
    if n_alt != n_best and remaining > 120:
        print(f"\n=== Phase 5: Alt length n={n_alt}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_long = spectral_precondition(a_long, n_iter=30, alpha=0.5, blend=0.5)
        try_update(a_long, f"interp_n{n_alt}")
        
        t5a = min(remaining * 0.35, 250.0)
        if t5a > 30:
            a_in = normalize_a(a_long)
            a_out, _ = smoothed_gradient_descent(
                a_in, t5a, top_k_frac=0.03, beta_init=60.0, beta_min=8.0,
                c0=2.5, gamma=0.999, verbose=True, rng_seed=200
            )
            try_update(a_out, f"sgd_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.3, 100.0)
            if t5b > 15:
                a_in = normalize_a(best_seq)
                a_out, _ = coordinate_descent(a_in, t5b, verbose=True, rng_seed=555)
                try_update(a_out, f"coord_n{len(best_seq)}_b")
    
    # === Phase 6: Final SGD refinement on best ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Final SGD n={n_best}, budget={remaining:.0f}s ===")
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 20:
            a_in = normalize_a(best_seq)
            a_out, _ = smoothed_gradient_descent(
                a_in, t6, top_k_frac=0.02, beta_init=40.0, beta_min=5.0,
                c0=2.0, gamma=0.9995, verbose=True, rng_seed=777
            )
            try_update(a_out, f"sgd_final_n{n_best}")
    
    # === Phase 7: Final coordinate polish ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final Coord n={n_best} ===")
        t7 = min(remaining * 0.6, 100.0)
        if t7 > 10:
            a_in = normalize_a(best_seq)
            a_out, _ = coordinate_descent(a_in, t7, verbose=False, rng_seed=1234)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 8: Last quick SGD ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 20:
        n_best = len(best_seq)
        t8 = min(remaining * 0.7, 60.0)
        if t8 > 10:
            a_in = normalize_a(best_seq)
            a_out, _ = smoothed_gradient_descent(
                a_in, t8, top_k_frac=0.01, beta_init=30.0, beta_min=3.0,
                c0=1.5, gamma=0.9998, verbose=False, rng_seed=4321
            )
            try_update(a_out, f"sgd_very_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
