import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def fast_conv(a):
    n = len(a)
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    A = rfft(a, N)
    return irfft(A*A, N)[:size]

def lse_grad(a, tau):
    """Return (smooth_max_value, gradient) for log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    
    b = fast_conv(a)
    b_max = b.max()
    
    # Softmax weights (for numerical stability)
    logits = tau * (b - b_max)
    exp_logits = np.exp(logits)
    log_sum = b_max + np.log(exp_logits.sum()) / tau
    w = exp_logits / exp_logits.sum()
    
    # Gradient via frequency domain
    # f = sum_k w[k]*b[k], b[k] = (a*a)[k]
    # In freq: f = (1/N)*sum_k W[k]*A[k]^2
    # grad_freq = (2/N)*W*A
    # grad = IFFT(grad_freq)[:n]
    A = fft(a, N)
    W = fft(w, N)
    grad = np.real(ifft(2.0 * W * A / N))[:n]
    
    return log_sum, grad, b

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (LSE) smoothed gradient descent with Adam optimizer
2. Coordinate descent with golden-section line search
3. Multi-restart with diverse structured perturbations
4. Cross-correlation-based gradient via FFT (avoids LP solvers)
Key insight: score is scale-invariant, so we optimize max(b)/(sum(a))^2 with a>=0
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def lse_value_and_grad(a, tau):
    """
    Compute LSE-smoothed max of b=a*a and its gradient.
    Returns (smooth_value, gradient_array, raw_b).
    """
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    b = fast_conv(a)
    b_max = b.max()
    
    # Softmax weights with numerical stability
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    log_sum = b_max + np.log(exp_shifted.sum() + 1e-300) / tau
    w = exp_shifted / (exp_shifted.sum() + 1e-300)
    
    # Gradient: grad = 2 * cross_corr(w, a)[:n]
    # cross_corr(w,a)[i] = sum_m w[i+m]*a[m]
    # FFT: IFFT(conj(FFT(w_pad)) * FFT(a_pad))
    N_grad = 1 << (size + n - 1 - 1).bit_length()
    W_pad = np.zeros(N_grad)
    W_pad[:size] = w
    A_pad = np.zeros(N_grad)
    A_pad[:n] = a
    
    W_freq = fft(W_pad)
    A_freq = fft(A_pad)
    grad_full = np.real(ifft(np.conj(W_freq) * A_freq))
    grad = 2.0 * grad_full[:n]
    
    return log_sum, grad, b


def project_nonneg_sum(x, target_sum):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    n = len(x)
    if target_sum <= 0:
        return np.zeros(n)
    
    # Water-filling / bisection for lambda
    lo = -x.max() - 1.0
    hi = 0.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.maximum(x - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    result = np.maximum(x - lam, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= target_sum / s
    return result


def lse_adam_optimize(a0, time_limit, seed=0):
    """LSE gradient descent with Adam optimizer and tau annealing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    
    # Adaptive learning rate
    lr = 0.1
    lr_min = 1e-6
    lr_max = 1.0
    no_improve = 0
    max_no_improve = 200
    
    # Tau annealing: start smooth, get sharper
    tau_start = 5.0
    tau_end = max(50.0, n * 0.1)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal tau
        progress = min(1.0, t / 3000.0)
        tau = tau_start + (tau_end - tau_start) * progress
        # Periodically use sharp tau for final classification
        if t % 100 == 0:
            tau_eff = tau_end
        else:
            tau_eff = tau
        
        smooth_val, grad, b = lse_value_and_grad(a, tau_eff)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize to maintain sum = sqrt(2n) (for consistency, though score is scale-invariant)
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Restart this step
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Evaluate
        score = fast_conv(a_new).max()
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, lr_max)
        else:
            no_improve += 1
            lr = max(lr * 0.97, lr_min)
        
        if no_improve > max_no_improve:
            # Restart with small perturbation
            noise = 0.01 * rng.standard_normal(n)
            a = a + noise * best_score / n
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.1
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=0):
    """Coordinate descent with golden-section search per coordinate."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max() / s ** 2
    
    t0 = time.monotonic()
    phi = (1.0 + np.sqrt(5.0)) / 2.0  # golden ratio
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved_any = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old = a[j]
            if old < 1e-10:
                # Try setting to a small positive value
                candidates = [norm / n * 0.5, norm / n, norm / n * 2]
                best_x = old
                best_sc = best_score
                for x in candidates:
                    a[j] = x
                    sc = 2.0 * n * fast_conv(a).max() / a.sum() ** 2
                    a[j] = old  # restore
                    if sc < best_sc:
                        best_sc = sc
                        best_x = x
                if best_x != old:
                    a[j] = best_x
                    s = a.sum()
                    a = a * norm / s
                    sc_final = 2.0 * n * fast_conv(a).max() / norm ** 2
                    if sc_final < best_score:
                        best_score = sc_final
                        best_a = a.copy()
                        improved_any = True
                continue
            
            # Golden section search for optimal a[j]
            # Score as function of x = a[j]:
            # S(x) = 2n * max_k b_k(x) / (sum_others + x)^2
            # b_k(x) = b_k(old) + delta_k(x)
            # where delta follows the formula derived
            
            b_old = fast_conv(a)
            S = a.sum() - old  # sum of others
            
            def score_at_x(x):
                if x < 0:
                    return float('inf')
                # Compute new b efficiently
                b_new = b_old.copy()
                # Affected lags: k in [max(0,j), min(2n-2, j+n-1)]
                lo = max(0, j)
                hi = min(2 * n - 1, j + n)
                if lo < hi:
                    # For k in [lo, hi), delta = 2*(x-old)*a[k-j] (or x^2-old^2 at k=2j)
                    idx_range = np.arange(lo, hi)
                    a_idx = np.minimum(idx_range - j, 2 * j - idx_range)
                    # Actually, a[k-j] where k-j ranges from lo-j to hi-1-j
                    a_lo = max(0, lo - j)
                    a_hi = min(n, hi - j)
                    a_slice = a[a_lo:a_hi]
                    delta = 2.0 * (x - old) * a_slice
                    # Correction at k=2j
                    if 2 * j >= lo and 2 * j < hi:
                        pos = 2 * j - lo
                        delta[pos] = x * x - old * old - 2.0 * (x - old) * old + 2.0 * (x - old) * old
                        # Actually: at k=2j, the change is x^2 - old^2, not 2*(x-old)*a[j]
                        # a[j] = old, so 2*(x-old)*old would be the "formula" value
                        # Correction: x^2 - old^2 - 2*(x-old)*old = (x-old)^2
                        delta[pos] = x * x - old * old
                    b_new[lo:hi] += delta
                
                max_b = b_new.max()
                total_sum = S + x
                if total_sum < 1e-15:
                    return float('inf')
                return 2.0 * n * max_b / (total_sum * total_sum)
            
            # Golden section search on [0, upper_bound]
            x_lo = 0.0
            x_hi = max(old * 4.0, norm * 0.3)
            
            # First check endpoints and current
            sc_lo = score_at_x(x_lo)
            sc_hi = score_at_x(x_hi)
            sc_mid = score_at_x(old)
            
            # Find a good bracket
            if sc_lo < sc_mid and sc_lo < sc_hi:
                x_hi = old
                sc_hi = sc_mid
                sc_mid = score_at_x(x_hi / 2)
                old_sc = sc_lo
                best_x = x_lo
                if sc_mid < sc_lo:
                    best_x = x_hi / 2
                    old_sc = sc_mid
            elif sc_hi < sc_mid and sc_hi < sc_lo:
                x_lo = old
                sc_lo = sc_mid
                sc_mid = score_at_x((x_lo + x_hi) / 2)
                best_x = x_hi
                old_sc = sc_hi
                if sc_mid < sc_hi:
                    best_x = (x_lo + x_hi) / 2
                    old_sc = sc_mid
            else:
                best_x = old
                old_sc = sc_mid
            
            # Golden section
            gr = 1.0 / phi
            c = x_hi - gr * (x_hi - x_lo)
            d = x_lo + gr * (x_hi - x_lo)
            sc_c = score_at_x(c)
            sc_d = score_at_x(d)
            
            for _ in range(12):
                if x_hi - x_lo < 1e-10:
                    break
                if sc_c < sc_d:
                    x_hi = d
                    d = c
                    sc_d = sc_c
                    c = x_hi - gr * (x_hi - x_lo)
                    sc_c = score_at_x(c)
                else:
                    x_lo = c
                    c = d
                    sc_c = sc_d
                    d = x_lo + gr * (x_hi - x_lo)
                    sc_d = score_at_x(d)
                
                best_c = min(sc_c, sc_d)
                if best_c < old_sc:
                    best_x = c if sc_c < sc_d else d

            x_opt = (x_lo + x_hi) / 2

            if sc_opt < old_sc:
                best_x = x_opt
                old_sc = sc_opt
            
            if abs(best_x - old) > 1e-10:
                a[j] = best_x
                s_total = a.sum()
                if s_total > 1e-15:
                    a = a * norm / s_total
                # Verify

                if sc_verify < best_score:
                    best_score = sc_verify
                    best_a = a.copy()
                    improved_any = True
                else:
                    a = best_a.copy()  # revert
        
        if not improved_any and passes % 5 == 0:

                a[j] = max(0, a[j] * (1 + 0.1 * rng.standard_normal()))

                sc = 2.0 * n * fast_conv(a).max() / norm ** 2
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved_any = True
    
    return best_a, best_score


def try_update(best_seq, best_score, candidate, label, t_global):
    """Evaluate candidate and update best if improved."""
    if candidate is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s  # normalize to sum=1 for evaluation
    sc = evaluate_sequence([float(x) for x in a])
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def structured_perturbation(a, seed, mode=0):
    """Create diverse perturbations of a."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a_copy = a.copy()
    
    if mode == 0:
        # Gaussian noise
        noise = 0.05 * rng.standard_normal(n)
        a_copy = a_copy + noise
    elif mode == 1:
        # Low-frequency sinusoidal perturbation
        freq = rng.integers(1, max(2, n // 50))
        phase = 2 * np.pi * rng.random()
        amp = 0.03 * a.max()
        a_copy = a_copy + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 2:
        # Piecewise block perturbation
        n_blocks = rng.integers(3, 10)
        block_size = n // n_blocks
        for b in range(n_blocks):
            scale = 1.0 + 0.1 * rng.standard_normal()
            a_copy[b * block_size:(b + 1) * block_size] *= max(0.1, scale)
    elif mode == 3:
        # Mirror + blend
        a_rev = a_copy[::-1].copy()
        t = 0.1 + 0.2 * rng.random()
        a_copy = (1 - t) * a_copy + t * a_rev
    elif mode == 4:
        # Sparse large perturbation
        n_pert = max(5, n // 20)
        idx = rng.choice(n, size=n_pert, replace=False)
        a_copy[idx] *= (1 + 0.3 * rng.standard_normal(n_pert))
    
    a_copy = np.maximum(a_copy, 0.0)
    s = a_copy.sum()
    if s < 1e-15:
        return a
    return a_copy / s
