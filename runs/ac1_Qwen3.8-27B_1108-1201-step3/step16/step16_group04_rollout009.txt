```python
def build_conv_matrix(y, n):
    """Build the matrix A such that A @ b = conv(y, b)."""
    # A has shape (2n-1, n)
    # A[k, j] = y[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(y[k-j])
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
def solve_lp_top_k(y, M, k_list, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in k_list, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in k_list:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum
```

```python
def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Fast coordinate descent: try moving mass from peak-contributing coords to low-impact coords."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Top peaks
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find good destination: low sensitivity to top peaks
            # Sensitivity of destination d: sum over top peaks of a[peak - d] or a[d - peak]
            sens = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                d1 = k2i - np.arange(n)
                valid1 = (d1 >= 0) & (d1 < n)
                sens[valid1] += a[d1[valid1]]
                d2 = np.arange(n) - k2i
                valid2 = (d2 >= 0) & (d2 < n)
                sens[valid2] += a[d2[valid2]]
            
            # Pick low-sensitivity destinations
            low_sens = np.argsort(sens)[:min(200, n)]
            chosen = low_sens[rng.permutation(len(low_sens))[:min(100, len(low_sens))]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                if sens[dst] < best_score and a[dst] < 0.05 * a.sum()/n * 10:  # not too occupied
                    best_score = sens[dst]
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Perturbation
            n_pert = min(10, n)
            sources = rng.choice(n, n_pert, replace=False)
            dests = rng.choice(n, n_pert, replace=False)
            for src_i, dst_i in zip(sources, dests):
                if a[src_i] > 1e-12:
                    amt = a[src_i] * rng.uniform(0.01, 0.05)
                    a[src_i] -= amt
                    a[dst_i] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
    
    return best_a, best_m
```

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def spectral_equalize(a, n_iter=30, blend=0.5, eps=1e-10):
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - blend) * a + blend * a_new
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_lse_grad(a, tau):
    n = len(a)
    ac = fast_autocorr(a)
    tau_ac = tau * ac
    m = tau_ac.max()
    exp_vals = np.exp(tau_ac - m)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = exp_vals / exp_sum
    lse_val = (m + math.log(max(exp_sum, 1e-300))) / tau
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[:n]
    return grad, lse_val, ac
