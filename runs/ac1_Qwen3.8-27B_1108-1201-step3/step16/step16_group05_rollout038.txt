def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        for x in a_list:
            if x < 0:
                x = 0.0
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_smoothed(a, beta, top_k=None):
    """
    Compute gradient of (1/beta)*log(sum_topk exp(beta*b_k))
    where b = conv(a,a).
    Returns (gradient_vector, max_b).
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    if top_k is None:
        top_k = min(100, len(b))
    
    top_idx = np.argsort(b)[::-1][:top_k]
    top_vals = b[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # grad[m] += 2 * w * a[k-m] if 0 <= k-m < n
        m_start = max(0, k_int - n + 1)
        m_end = min(n, k_int + 1)
        if m_start >= m_end:
            continue
        # For m in [m_start, m_end): a[k-m] is valid
        idx = np.arange(m_start, m_end)
        partner = k_int - idx  # these are the a[partner] values
        grad[idx] += 2.0 * w * a[partner]
    
    return grad, max_b

def compute_gradient_exact(a, top_k=5):
    """
    Compute weighted gradient using top-k exact peaks.
    Returns (gradient_vector, max_b).
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    top_idx = np.argsort(b)[::-1][:top_k]
    
    # Weight by peak value (heavier weight for higher peaks)
    top_vals = b[top_idx]
    weights = top_vals / top_vals.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        m_start = max(0, k_int - n + 1)
        m_end = min(n, k_int + 1)
        if m_start >= m_end:
            continue
        idx = np.arange(m_start, m_end)
        partner = k_int - idx
        grad[idx] += 2.0 * w * a[partner]
    
    return grad, max_b

def spectral_equalization(a, n_freq_bins=50, alpha=0.3, n_iter=50):
    """
    Spectral equalization: try to flatten |A(f)|^2 in the frequency domain.
    Works by reducing high spectral peaks and boosting low ones.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.abs(A) ** 2
    
    # We want to reduce the dynamic range of B
    # Target: make B more uniform (but keep DC fixed since sum(a)=1)
    B_target = np.ones_like(B)
    B_target[0] = B[0]  # Keep DC
    
    # Blend toward flat spectrum
    B_new = (1 - alpha) * B + alpha * B_target * (B.sum() / B_target.sum())
    B_new[0] = B[0]  # Preserve DC
    
    # Reconstruct: A_new = A * sqrt(B_new / B) (phase preserved)
    ratio = np.sqrt(B_new / (B + 1e-30))
    A_new = A * ratio
    
    # Inverse FFT
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    
    # Project to simplex
    a_new = project_simplex(a_new)
    
    return a_new

def peak_shave(a, n_steps=50, verbose=False):
    """
    Peak shaving: iteratively identify the peak of conv(a,a) and move mass
    to reduce it.
    """
    n = len(a)
    a = norm_sum1(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    for step in range(n_steps):
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        
        # Gradient of b[k_star] w.r.t. a
        g = np.zeros(n)
        m_start = max(0, k_star - n + 1)
        m_end = min(n, k_star + 1)
        if m_start < m_end:
            idx = np.arange(m_start, m_end)
            partner = k_star - idx
            g[idx] = 2.0 * a[partner]
        
        # We want to decrease b[k_star], so move in direction -g
        # But we're on the simplex, so use projected gradient
        # Find positions with high g (source) and low g (destination)
        g_sorted = np.argsort(g)
        
        # Move mass from high-g to low-g positions
        improved = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            src = g_sorted[-1]  # highest gradient
            dst = g_sorted[0]   # lowest gradient
            if src == dst:
                break
            if a[src] < 1e-10:
                break
            
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                break
            a_trial = a_trial / s
            
            b_trial = fast_autocorr(a_trial)
            max_b_trial = b_trial.max()
            
            if max_b_trial < max_b - 1e-16:
                a = a_trial
                max_b = max_b_trial
                improved = True
                break
        
        if not improved:
            break
    
    return a, max_b

def multiplicative_update(a, b, n_iter=100, tau=0.1):
    """
    Multiplicative (EM-like) update for reducing peak convolution.
    a_new[i] ∝ a[i] * (1 - tau * g[i]) where g is the gradient.
    """
    n = len(a)
    a = norm_sum1(a)
    
    for _ in range(n_iter):
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        max_b = b[k_star]
        
        # Gradient
        g = np.zeros(n)
        m_start = max(0, k_star - n + 1)
        m_end = min(n, k_star + 1)
        if m_start < m_end:
            idx = np.arange(m_start, m_end)
            partner = k_star - idx
            g[idx] = 2.0 * a[partner]
        
        # Normalize gradient
        g_max = g.max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        
        # Multiplicative update
        a_new = a * (1.0 - tau * g_norm)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        b_new = fast_autocorr(a_new)
        if b_new.max() >= max_b:
            break
        a = a_new
    
    return a, fast_autocorr(a).max()

def multiplicative_refine(a0, time_limit, verbose=False):

    n = len(a0)

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        a_curr = best_a.copy()

                break

            k_star = int(np.argmax(b))
            max_b = b[k_star]

            m_end = min(n, k_star + 1)

                partner = k_star - idx
                g[idx] = 2.0 * a_curr[partner]

            if g_max < 1e-15:

def generate_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    inits = []

    inits.append(np.ones(n) / n)

    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())
    
    # 3. Sin^2 bump
    s2 = np.sin(np.pi * x) ** 2
    s2 = np.maximum(s2, 1e-10)
    inits.append(s2 / s2.sum())

    gauss = np.exp(-(x - 0.5) ** 2 / (2 * 0.15 ** 2))
    inits.append(gauss / gauss.sum())

    bim = np.exp(-(x - 0.3) ** 2 / (2 * 0.1 ** 2)) + np.exp(-(x - 0.7) ** 2 / (2 * 0.1 ** 2))
    inits.append(bim / bim.sum())

    flat = np.ones(n) + rng.normal(0, 0.1, n)
    flat = np.maximum(flat, 0.01)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Now, back to the spectral analysis. The key relation is:
conv(a,a) = IFFT(FFT(a_padded) * FFT(a_padded))

In the frequency domain, the "spectrum" of the convolution is A[f] * A[f] (not |A[f]|^2).

For real a, A[-f] = conj(A[f]), so A[f]*A[f] is NOT conjugate symmetric in general! Wait, but the result must be real (since we're convolving two real signals). Let me check:

(A[f] * A[f]) for -f: A[-f] * A[-f] = conj(A[f]) * conj(A[f]) = conj(A[f] * A[f])

Yes! So A[f]*A[f] IS conjugate symmetric, and its IFFT is real. Good.

Now, what's the "shape" of A[f]*A[f]?
A[0] = sum(a) = 1 (DC)
A[f] for f ≠ 0: depends on the signal

For uniform a: A[f] = (1/n) * D(f) where D is the Dirichlet kernel.
A[f]*A[f] = (1/n^2) * D(f)^2

The IFFT of this gives the triangle (the uniform convolution).

For a smooth signal (concentrated at low frequencies): A[f] decays fast for large f.
A[f]*A[f] decays even faster.
IFFT of a rapidly decaying spectrum → slowly varying b (smooth, spread out) → low peak.

So confirmed: smooth a → flat b → low max(b).

The "smoothest" non-negative a with sum=1 is the one with the most concentrated spectrum at DC. This is related to prolate spheroidal wave functions (PSWFs), which maximize the energy concentration in a band.

For our problem, we want to maximize the energy of a in the low-frequency band [0, W] where W is small. This is exactly the PSWF problem!

The optimal a (approximately) is a scaled PSWF truncated to length n.

This gives us a great initialization: use the PSWF as the starting point!

In practice, the PSWF can be computed as the eigenvector of the prolate matrix:
P[i,j] = sinc((i-j) * W / n) (or similar)

The largest eigenvector of this matrix gives the PSWF.

For our problem, the "band" W should be chosen to balance between:
- Too narrow: a is very smooth but the convolution peak might not be minimized
- Too wide: a has more high-frequency content, leading to a more peaked convolution
