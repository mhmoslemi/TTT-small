6. Zero-pad back to length n, project onto simplex
7. Iterate

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with exact max (top-k weighted)
2. Hinge-penalty smooth optimization (alternating M and a)
3. KKT linear-system polishing (solve active-set stationarity conditions)
4. Coordinate descent with peak-aware mass redistribution
5. Multi-length exploration and diverse initializations

import numpy as np
import time


def fast_convolution(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_exact(a, top_k=3, beta_weight=5.0):
    """
    Compute subgradient of max_k (a*a)_k using weighted top-k active lags.
    d(a*a)_k / d a_j = 2 * a_{k-j} if 0 <= k-j < n, else 0.
    """
    n = len(a)
    c = fast_convolution(a)
    k_eff = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_eff]
    top_vals = c[top_idx]
    if top_vals[0] < 1e-15:
        return np.zeros(n)
    # Weight: exponential emphasis on the maximum
    shifted = top_vals - top_vals[0]
    weights = np.exp(beta_weight * shifted)
    weights /= weights.sum()

    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # g[j] += 2*w*a[k-j] for valid j (0 <= k-j < n means j in [k-n+1, k] intersect [0, n-1])
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        # a[k-j] for j in [j_start, j_end): index k-j goes from k-j_start to k-j_end+1
        idx_in_a = k_int - np.arange(j_start, j_end)
        g[j_start:j_end] += 2.0 * w * a[idx_in_a]
    return g, c.max()


def nesterov_subgradient(a0, time_limit, top_k=3, eta0=0.5, beta_weight=5.0,
                         verbose=True, seed=42):
    """Nesterov-accelerated subgradient method with exact max."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    eta = eta0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        mu = (t - 1.0) / (t + 1.5)
        y = x + mu * (x - x_prev)
        y = project_simplex(y)

        # Subgradient
        g, M_y = subgradient_exact(y, top_k=top_k, beta_weight=beta_weight)

        # Adaptive step size based on gradient norm
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        eta_t = eta / (1.0 + 0.1 * np.sqrt(t))

        # Gradient step
        x_new = project_simplex(y - eta_t * g)

        x_prev = x
        x = x_new

        # Track best
        M_new = fast_convolution(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            # Perturb from best
            noise_scale = 0.003 * (1 + 0.5 * np.log1p(t / 10000))
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-14)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            # Occasionally try different top_k
            if top_k < 8:
                top_k = min(top_k * 2, 8)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      NSG restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={top_k}, eta={eta:.4f}")

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NSG t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def hinge_optimization(a0, time_limit, verbose=True, seed=42):
    """
    Optimize sum_k max(0, c_k - M)^2 + lambda * M
    by alternating between a and M. This creates a smooth surrogate.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    M = fast_convolution(a).max() * 0.95  # Slightly below current peak
    lr_a = 0.01
    lr_M = 0.001

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolution(a)
        excess = np.maximum(0.0, c - M)
        n_excess = np.count_nonzero(excess > 1e-12)

        if n_excess == 0:
            M = max(M * 0.99, c.max() * 0.9)
            continue

        # Gradient w.r.t. a: 2 * sum_k excess_k * dc_k/da
        g = np.zeros(n)
        # Only iterate over active lags (excess > 0) for efficiency
        active = np.where(excess > 1e-12)[0]
        # Use top active for speed
        if len(active) > 50:
            active = active[np.argsort(-excess[active])[:50]]

        for k in active:
            k_int = int(k)
            ek = excess[k_int]
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue
            idx_in_a = k_int - np.arange(j_start, j_end)
            g[j_start:j_end] += 2.0 * ek * a[idx_in_a]

        # Update a
        a = project_simplex(a - lr_a * g / (n_excess + 1))

        # Update M: decrease if too lax, increase if too tight
        c_new = fast_convolution(a)
        M_new = M - lr_M * (c_new.max() - M)
        M = max(M_new, c_new.max() * 0.85)

        # Decay learning rates
        if t % 100 == 0:
            lr_a *= 0.98
            lr_M *= 0.98

        # Track best
        m_curr = c_new.max()
        if m_curr < best_m - 1e-16:
            best_m = m_curr
            best_a = a.copy()

        # Periodic restart
        if t % 5000 == 0:
            noise = rng.normal(0, 0.002, n) * best_a
            a_re = np.maximum(best_a + noise, 1e-14)
            a_re = a_re / a_re.sum()
            if fast_convolution(a_re).max() < best_m:
                a = a_re

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Hinge t={t} ({elapsed:.0f}s): J={J:.6f}, M={M:.2e}, n_act={n_excess}")

    return best_a, best_m


def kkt_polish(a0, max_iter=50, verbose=True):
    """
    Polish using KKT conditions: at optimum, for the active lag(s),
    the subgradient should be constant on the support.
    Solve the linear system: sum_l w_l * a_{k_l - j} = lambda for all j in support.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > 30:
            break

        c = fast_convolution(a)
        max_c = c.max()
        # Find active lags (within tolerance of max)
        tol = max_c * 0.01 + 1e-12
        active_idx = np.where(np.abs(c - max_c) < tol)[0]
        if len(active_idx) == 0:
            active_idx = [int(np.argmax(c))]

        # Use top active lags
        top_act = active_idx[np.argsort(-c[active_idx])[:min(5, len(active_idx))]]
        act_vals = c[top_act]
        weights = act_vals / act_vals.sum()

        # Build the linear system
        # Variables: a_0, ..., a_{n-1}, lambda (total n+1 variables)
        # Equations for j = 0, ..., n-1:
        #   sum_l w_l * a_{k_l - j} - lambda = 0
        # Equation n: sum_i a_i = 1

        # Matrix B is (n+1) x (n+1)
        # Row j: has entries 2*w_l at column (k_l - j) for each l where 0 <= k_l - j < n
        #         and -1 at column n (lambda)
        # Row n: all 1s in columns 0..n-1, 0 in column n

        B = np.zeros((n + 1, n + 1))
        rhs = np.zeros(n + 1)

        for j in range(n):
            B[j, n] = -1.0  # -lambda
            for w_l, k_l in zip(weights, top_act):
                k_int = int(k_l)
                idx = k_int - j
                if 0 <= idx < n:
                    B[j, idx] += 2.0 * w_l
        rhs[j] = 0.0

        # Sum constraint
        B[n, :n] = 1.0
        rhs[n] = 1.0

        # Solve the system
        try:
            sol = np.linalg.solve(B, rhs)
        except np.linalg.LinAlgError:
            # Use least squares as fallback
            try:
                sol = np.linalg.lstsq(B, rhs, rcond=None)[0]
            except Exception:
                break

        a_new = np.maximum(sol[:n], 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        m_new = fast_convolution(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Makes it worse; try a convex combination
            for alpha in [0.1, 0.3, 0.5]:
                a_mix = (1 - alpha) * a + alpha * a_new
                a_mix = np.maximum(a_mix, 0.0)
                sm = a_mix.sum()
                if sm < 1e-10:
                    continue
                a_mix = a_mix / sm
                m_mix = fast_convolution(a_mix).max()
                if m_mix < best_m - 1e-14:
                    best_m = m_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break
            else:
                # No improvement, stop
                break

        if verbose and it % 5 == 0:
            J = 2.0 * n * best_m
            print(f"      KKT iter={it}: J={J:.6f}, max_c={max_c:.2e}, n_act={len(top_act)}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=42):
    """Coordinate descent: move mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_convolution(a)
        max_c = c.max()
        # Top peaks
        top_peaks = np.argsort(c)[::-1][:min(10, len(c))]

        # Compute contribution of each position to the peaks
        contrib = np.zeros(n)
        for k in top_peaks[:6]:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start >= j_end:
                continue
            idx_in_a = k_int - np.arange(j_start, j_end)
            contrib[j_start:j_end] += a[idx_in_a]

        # Find high-contribution sources
        sources = np.argsort(contrib)[::-1][:20]
        improved = False

        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Find best destination: minimize increase in peak
            # Score destination by how much it adds to peak lags
            n_try = min(n, 300)
            if n_try < n:
                candidates = np.random.choice(n, n_try, replace=False)
            else:
                candidates = np.arange(n)

            best_dst = -1
            best_score = 0.0
            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_peaks[:5]:
                    k_int = int(k)
                    # If we add to position dst, it adds a[dst]*a[k-dst] to c_k (if valid)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue

            # Try moving mass
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolution(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_convolution(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
    except Exception:
        pass

    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Nesterov Subgradient (main workhorse, 40% time) ===
    rem = remaining_time()
    print(f"\n=== Phase 1: NSG n={n0} (budget {min(rem*0.40, 500):.0f}s) ===")
    t1 = min(rem * 0.40, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(
            a_in, t1, top_k=3, eta0=0.3, beta_weight=5.0,
            verbose=True, seed=42)
        try_update(a_out, f"nsg_n{n0}")

    # === Phase 2: Hinge optimization (15% time) ===
    rem = remaining_time()
    print(f"\n=== Phase 2: Hinge n={len(best_seq)} (budget {min(rem*0.15, 200):.0f}s) ===")
    t2 = min(rem * 0.15, 200.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = hinge_optimization(a_in, t2, verbose=True, seed=42)
        try_update(a_out, f"hinge_n{len(best_seq)}")

    # === Phase 3: NSG again with different params (15% time) ===
    rem = remaining_time()
    print(f"\n=== Phase 3: NSG v2 n={len(best_seq)} (budget {min(rem*0.15, 200):.0f}s) ===")
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(
            a_in, t3, top_k=5, eta0=0.15, beta_weight=8.0,
            verbose=True, seed=77)
        try_update(a_out, f"nsg2_n{len(best_seq)}")

    # === Phase 4: KKT Polish (10% time) ===
    rem = remaining_time()
    print(f"\n=== Phase 4: KKT n={len(best_seq)} (budget {min(rem*0.10, 100):.0f}s) ===")
    t4 = min(rem * 0.10, 100.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = kkt_polish(a_in, max_iter=30, verbose=True)
        try_update(a_out, f"kkt_n{len(best_seq)}")

    # === Phase 5: Coordinate polish (10% time) ===
    rem = remaining_time()
    print(f"\n=== Phase 5: Coord n={len(best_seq)} (budget {min(rem*0.10, 100):.0f}s) ===")
    t5 = min(rem * 0.10, 100.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Try slightly different length (10% time) ===
    rem = remaining_time()
    n1 = n0 + 200
    if rem > 80:
        print(f"\n=== Phase 6: NSG n={n1} (budget {min(rem*0.10, 100):.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n1)
        t6 = min(rem * 0.10, 100.0)
        if t6 > 20:
            a_out, _ = nesterov_subgradient(
                a_long, t6, top_k=3, eta0=0.3, beta_weight=5.0,
                verbose=True, seed=99)
            try_update(a_out, f"nsg_n{n1}")

    # === Phase 7: Structured initialization + NSG ===
    rem = remaining_time()
    if rem > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Structured inits n={n_best} ===")
        x = np.linspace(0, 1, n_best)

        inits = []
        # Cosine window
        cos_w = np.cos(np.pi * (x - 0.5)) ** 2
        inits.append(np.maximum(cos_w, 0) / np.maximum(cos_w, 0).sum())
        # Power-law edge concentration
        edge = np.abs(2 * x - 1) ** 0.5 + 0.1
        inits.append(edge / edge.sum())
        # Blend with best
        blend = 0.5 * norm_sum1(best_seq) + 0.5 * inits[0]
        inits.append(norm_sum1(blend))

        t_per = min(rem * 0.08, 60.0)
        for idx, init in enumerate(inits[:3]):
            rem = remaining_time()
            if rem < 25:
                break
            t7 = min(t_per, rem - 15)
            if t7 > 15:
                a_out, _ = nesterov_subgradient(
                    init, t7, top_k=3, eta0=0.3, beta_weight=5.0,
                    verbose=False, seed=200 + idx)
                try_update(a_out, f"struct{idx}_nsg")

    # === Phase 8: Final KKT + Coord polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_best} ===")
    if rem > 40:
        t8 = min(rem * 0.3, 60.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = kkt_polish(a_in, max_iter=20, verbose=False)
            try_update(a_out, f"kkt_final")

        rem = remaining_time()
        if rem > 20:
            t8b = min(rem * 0.5, 50.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t8b, verbose=False, seed=314)
                try_update(a_out, f"coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
