Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent with soft subgradient (fast primary method)
2. LP-Frank-Wolfe with constraint generation (precise refinement)
3. Peak-targeted coordinate descent for final polish
4. Multi-length exploration

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=50, beta=3.0):
    """Compute a weighted subgradient of max(a*a)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Weight peaks by exponential of their value
    # Use top-k peaks with exponential weighting
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Exponential weighting (soft-max)
    w = np.exp(beta * top_vals / M)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    # Compute subgradient
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # For peak at position k, the gradient w.r.t. a_i is (a_{k-i} + a_i) for valid i
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        # a_{k-i} for i in idx
        k_minus_i = k_int - idx
        grad[idx] += wi * (a[k_minus_i] + a[idx])
    
    return grad

def nesterov_mirror_descent(a0, time_limit, top_k=50, beta=3.0, gamma=4.0, verbose=True):
    """Nesterov-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(1234)
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 5000
    best_gamma = gamma
    gamma_decay_count = 0
    max_gamma_decays = 15
    
    # Adaptive top_k: start small, increase over time
    # Adaptive beta: start moderate, increase for sharpness
    
    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_nest = (t - 1.0) / (t + 1.5)
        y = x + beta_nest * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation of y
        ac = fast_autocorr(y)
        
        # Adaptive top_k: use more peaks as we converge
        current_top_k = min(top_k, max(10, int(top_k * min(1.0, t / 10000))))
        current_beta = beta + min(5.0, t / 20000.0)
        
        # Compute subgradient
        grad = compute_subgradient(y, ac, top_k=current_top_k, beta=current_beta)
        
        # Mirror descent step
        eta = best_gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart
        if stall_count > max_stall:
            if gamma_decay_count < max_gamma_decays:
                best_gamma *= 0.5
                gamma_decay_count += 1
            # Restart from best with perturbation
            perturbation = rng.normal(0, 0.01, n)
            x_restarted = best_a * (1.0 + perturbation)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall_count = 0
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

    """Build one row of the constraint matrix for peak index k."""
    n = len(y)
    row = np.zeros(n)
    k_int = int(k)
    i_start = max(0, k_int - n + 1)

def lp_fw_constraint_gen(a0, time_limit, solver_name, verbose=True, max_active=200):
    """LP-Frank-Wolfe with constraint generation."""
    import cvxpy as cp
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    # Active constraint set
    active_set = []
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Update active set: peaks within 5% of max
        threshold = M_curr * 0.95
        candidates = np.where(ac >= threshold)[0]
        
        # Limit active set size
        if len(candidates) > max_active:
            # Keep the top max_active
            top_candidates = candidates[np.argsort(ac[candidates])[::-1][:max_active]]
            candidates = top_candidates
        active_set = list(candidates)
        
        # Solve LP: max sum(b) s.t. b >= 0, (y*b)_k <= M_curr for k in active_set
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_set:
            row = build_constraint_row(y, k)
            if row.any():
                constraints.append(row @ b_var <= M_curr)
        
        if not constraints:
            break
            
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # FW update with step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_idx = int(np.argmax(ac))
        
        # Find positions contributing to the peak
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            continue
        
        # Try moving mass from high-contribution positions to low-impact positions
        # Sort by contribution (descending)
        src_order = np.argsort(contrib)[::-1]
        
        improved = False
        for src_rank in src_order[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            
            # Try each possible destination
            # Score destinations by how much they'd reduce the peak
            for frac in [0.01, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                # Try moving to random destinations
                for _ in range(20):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    CD polish: J={J:.6f}")
    return best_a, best_m

def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=50, beta=3.0):
    """Compute weighted subgradient of max(a*a) using top peaks."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    w = np.exp(beta * top_vals / M)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        k_minus_i = k_int - idx
        grad[idx] += wi * (a[k_minus_i] + a[idx])
    return grad

def nesterov_mirror_descent(a0, time_limit, top_k=50, beta=3.0, gamma=4.0, verbose=True):
    """Nesterov-accelerated mirror descent on the simplex with adaptive restarts."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    cur_gamma = gamma
    n_decays = 0
    max_decays = 20
    phase = 0

    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive parameters
        progress = min(1.0, t / 50000.0)
        cur_top_k = max(10, int(top_k * (0.3 + 0.7 * progress)))
        cur_beta = beta + 3.0 * progress

        # Nesterov momentum
        beta_nest = (t - 1.0) / (t + 1.5)
        y = x + beta_nest * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        grad = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Mirror descent step
        eta = cur_gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive restart
        if stall_count > max_stall:
            if n_decays < max_decays:
                cur_gamma *= 0.6
                n_decays += 1
            # Choose restart strategy based on phase
            phase = (phase + 1) % 3
            if phase == 0:

            else:

                rand_dir = rng.dirichlet(np.ones(n))
                mix = rng.uniform(0.1, 0.3)
                x_restarted = (1.0 - mix) * best_a + mix * rand_dir
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            stall_count = 0

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, gamma={cur_gamma:.2f}")

        # Find contributing positions
        i_start = max(0, peak_idx - n + 1)

        if i_start > i_end:
            stall += 1
            continue
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_idx - idx]

        if contrib.max() < 1e-20:
            stall += 1
            continue

        # Strategy 1: Move mass from high-contribution to low-sensitivity
        src_order = np.argsort(contrib)[::-1]

        # Compute sensitivity: how much each position contributes to top peaks
        top_peaks = np.argsort(ac)[::-1][:15]
        sensitivity = np.zeros(n)
        for pk in top_peaks:

            if s_start <= s_end:

                sensitivity[s_idx] += a[pk_int - s_idx] + a[s_idx]

        improved = False
        for src_rank in src_order[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])

            # Try moving to low-sensitivity positions
            dst_candidates = np.argsort(sensitivity)[:max(30, n // 20)]
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:

                    a_trial = a.copy()

                    if M_trial < best_m - 1e-16:

                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_try = 10
            for _ in range(n_try):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.03)

                M_new = fast_autocorr(a).max()

                    stall = 0
                    break
                else:

    if verbose:
        J = 2.0 * n * best_m
        print(f"    CD polish: J={J:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    solver = get_solver()

    # === Phase 1: Nesterov Mirror Descent (fast, ~40% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 1: Mirror Descent n={n0} ===")
        t1 = min(remaining * 0.35, 550.0)

        # Run with default params
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t1,
                                            top_k=50, beta=3.0, gamma=4.0, verbose=True)
        try_update(a_out, "nmd_default")

        # Run with different params if time remains
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t1b = min(remaining * 0.3, 200.0)
            a_out2, _ = nesterov_mirror_descent(norm_sum1(best_seq), t1b,
                                                 top_k=20, beta=6.0, gamma=8.0, verbose=False)
            try_update(a_out2, "nmd_sharp")

    # === Phase 2: LP-FW with constraint generation (~30% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW CG n={n_cur} ===")
        t2 = min(remaining * 0.30, 450.0)
        a_out, _ = lp_fw_constraint_gen(norm_sum1(best_seq), t2, solver,
                                         max_active=250, verbose=True)
        try_update(a_out, f"lp_cg_n{n_cur}")

    # === Phase 3: Coordinate Descent Polish (~12% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord Descent n={n_cur} ===")
        t3 = min(remaining * 0.12, 180.0)
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t3, verbose=True, seed=42)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 4: Second LP-FW round (~15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.15, 250.0)
        a_out, _ = lp_fw_constraint_gen(norm_sum1(best_seq), t4, solver,
                                         max_active=200, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 5: Multi-n exploration (~18% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: Multi-n exploration (base n={n_best}) ===")

        test_ns = []
        for delta in [200, -200, 400, -400, 600, -600]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)

            # Quick mirror descent
            t_md = min(remaining * 0.25, 80.0)
            a_out, _ = nesterov_mirror_descent(a_inter, t_md,
                                                top_k=40, beta=4.0, gamma=5.0, verbose=False)
            try_update(a_out, f"nmd_n{nt}")

            # Quick LP-FW
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t_lp = min(remaining * 0.3, 100.0)
                a_out, _ = lp_fw_constraint_gen(norm_sum1(best_seq), t_lp, solver,
                                                 max_active=200, verbose=False)
                try_update(a_out, f"lp_n{nt}")

    # === Phase 6: Final polish with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final polish n={n_best}, {remaining:.0f}s remaining ===")

        # Coordinate descent with different seed
        t_cd = min(remaining * 0.4, 120.0)
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t_cd, verbose=True, seed=12345)
        try_update(a_out, "cd_final")

        # One more LP if time
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t_final = min(remaining * 0.7, 150.0)
            a_out, _ = lp_fw_constraint_gen(norm_sum1(best_seq), t_final, solver,
                                             max_active=300, verbose=True)
            try_update(a_out, "lp_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
