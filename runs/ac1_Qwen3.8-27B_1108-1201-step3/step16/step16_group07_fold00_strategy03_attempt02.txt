        def random_individual_bounded(n):
            # Create a sequence of n nonnegative floats, each between 0 and 1000, and normalized to sum = sqrt(2*n)
            a = np.random.uniform(0, 1000, n)
            s = np.sum(a)
            if s <  np.finfo(float).eps * n:
                a = np.ones(n) * np.sqrt(2*n) / n
            else:
                a = a * np.sqrt(2*n) / s
            return a
