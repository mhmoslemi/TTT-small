def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1

def spectral_equalize_v2(a, n_iter=40, eps=1e-10, blend=0.3):
    """
    Improved spectral flattening with adaptive epsilon and blending.
    Uses the fact that we want to minimize the peak of IFFT(|A|^2).
    """
    a = normalize(a).copy()
    n = len(a)
    for it in range(n_iter):
        A = np.fft.rfft(a, 2 * n)
        P = np.abs(A) ** 2
        # We want |A|^2 to be as flat as possible

        C = 1.0 / np.sqrt(P + eps)

        A_new = A * C

    return a


def logsumexp_max(ac, beta):
    """Compute smoothed max using log-sum-exp."""
    m = ac.max()
    if m < 1e-30:
        return m, np.ones_like(ac) / len(ac)
    shifted = (ac - m) * beta
    # Clip to prevent overflow
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    # Smoothed max approximation
    smooth_max = m + np.log(exp_shifted.sum()) / beta
    return smooth_max, weights


def projected_gradient_step(a, ac, beta, step_size):
    """
    Take a projected gradient step on the simplex.
    Gradient of sum_k w_k * C_k(a) where C_k(a) = sum_j a_j * a_{k-j}
    dC_k/da_i = a[k-i] + a[i-k] (with bounds)
    """
    n = len(a)
    _, weights = logsumexp_max(ac, beta)
    
    # Compute gradient: grad_i = sum_k weights[k] * (a[k-i] + a[i-k])
    grad = np.zeros(n)
    # Only use significant weights for efficiency
    sig_mask = weights > 1e-10
    sig_idx = np.where(sig_mask)[0]
    
    for k in sig_idx:
        k_int = int(k)
        w = weights[k_int]
        # Contribution to grad[i] from lag k: a[k-i] + a[i-k] (where valid)
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        i1_start = max(0, k_int - n + 1)
        i1_end = min(n - 1, k_int)
        i2_start = max(0, k_int)
        i2_end = min(n - 1, k_int + n - 1)
        
        if i1_end >= i1_start:
            idx1 = np.arange(i1_start, i1_end + 1)
            grad[idx1] += w * a[k_int - idx1]
        
        if i2_end >= i2_start and k_int != 0:
            idx2 = np.arange(i2_start, i2_end + 1)
            # a[i-k] where i-k is in [0, n-1]
            # But we need to avoid double-counting when k=0
            if k_int == 0:
                grad[idx2] += w * a[idx2]  # a[i-0] = a[i]
            else:
                grad[idx2] += w * a[idx2 - k_int]
    
    # Projected gradient step: minimize <grad, a> s.t. a >= 0, sum(a) = 1
    # Optimal solution: a_new = proj_simplex(a - step_size * grad)
    a_trial = a - step_size * grad
    a_new = project_simplex(a_trial)
    return a_new


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    """
    if solver is None:
        return None
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        # Only add constraint if it's meaningful
        if y_coeff.max() < 1e-15:
            continue
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_structured_init(n, seed):
    """Generate a random but structured initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    strategy = seed % 5
    if strategy == 0:
        # Sine squared
        a = np.sin(np.pi * x) ** 2
    elif strategy == 1:
        # Triangle wave
        a = 1 - 2 * np.abs(x - 0.5)
    elif strategy == 2:
        # Gaussian
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2))
    elif strategy == 3:
        # Random with smoothing
        a = rng.random(n)
        # Apply moving average
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    else:
        # Sum of sines
        a = np.ones(n)
        for _ in range(5):
            freq = rng.uniform(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project vector onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_equalize(a, n_iter=25, eps=1e-10):
    """Iterative spectral flattening with clipping."""
    a = normalize(a).copy()
    n = len(a)
    for it in range(n_iter):
        fft_len = 2 * n
        A = np.fft.rfft(a, fft_len)
        P = np.abs(A) ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.fft.irfft(A_new, fft_len)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        blend = 0.4 + 0.3 * (it / max(n_iter, 1))
        a = (1.0 - blend) * a + blend * a_new
    return a


def compute_gradient(a, ac, beta, top_k=100):
    """
    Compute gradient of smoothed max of autocorrelation.
    Uses log-sum-exp smoothing with top-k lags for efficiency.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    num_lags = len(ac)
    k_eff = min(top_k, num_lags)
    
    # Get top-k lags by autocorrelation value
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    
    # Compute softmax weights
    m = top_vals.max()
    if m < 1e-30:
        return np.zeros(n)
    shifted = (top_vals - m) * beta
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    
    # Compute gradient: grad[i] = sum_k w_k * [a[k-i] + a[i-k]] (with bounds)
    grad = np.zeros(n)
    
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        if wi < 1e-12:
            continue
        # Term 1: a[k-i] valid when 0 <= k-i < n => k-n+1 <= i <= k
        i1s = max(0, k_int - n + 1)
        i1e = min(n - 1, k_int)
        if i1e >= i1s:
            idx = np.arange(i1s, i1e + 1)
            grad[idx] += wi * a[k_int - idx]
        # Term 2: a[i-k] valid when 0 <= i-k < n => k <= i <= k+n-1
        i2s = max(0, k_int)
        i2e = min(n - 1, k_int + n - 1)
        if i2e >= i2s and k_int > 0:
            idx = np.arange(i2s, i2e + 1)
            grad[idx] += wi * a[idx - k_int]
        elif k_int == 0:
            # a[i-0] = a[i], already counted in term 1
            # term 1 gives a[0-i] which is only a[0] for i=0
            # term 2 gives a[i-0] = a[i] for all i
            grad[:] += wi * a[:]
    
    return grad


def smooth_gradient_descent(a0, time_limit, beta_init=40.0, beta_end=200.0,
                             c0=1.5, top_k=80, verbose=True):
    """
    Smoothed projected gradient with backtracking and adaptive beta.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    step_size = c0 / n
    max_iter = 300000
    stall = 0
    
    for t in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        progress = min(t / (0.8 * max_iter), 1.0)
        beta = beta_init + (beta_end - beta_init) * progress
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled too long
        if stall > 5000:
            pert = rng.normal(0, 0.003, n)
            a = a * (1.0 + pert)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0
            step_size = c0 / n
        
        # Compute gradient
        grad = compute_gradient(a, ac, beta, top_k=top_k)
        
        # Projected gradient step with backtracking
        improved = False
        for attempt in range(6):
            a_new = project_simplex(a - step_size * grad)
            M_new = fast_autocorr(a_new).max()
            if M_new < M_curr - 1e-15:
                a = a_new
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                improved = True
                break
            step_size *= 0.5
        
        if not improved:
            step_size = max(c0 / n / 4, step_size)
        
        # Periodic perturbation
        if t % 3000 == 0 and t > 0 and stall > 500:
            for _ in range(3):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    M_p = fast_autocorr(a).max()
                    if M_p < best_m:
                        best_m = M_p
                        best_a = a.copy()
                    else:
                        a[i1] += amt
                        a[i2] -= amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                    break
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.0f}, stall={stall}")
    
    return best_a, best_m


def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    if solver is None:
        return None
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        if y_coeff.max() < 1e-15:
            continue
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if len(constraints) == 0:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_frank_wolfe(a0, time_limit, solver, top_k_init=50, top_k_max=200,
                   margin_frac=0.03, verbose=True):
    """
    LP Frank-Wolfe with margin-based constraint selection and line search.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    current_top_k = top_k_init
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            # Try with smaller top_k to escape
            if current_top_k > 20:
                current_top_k = max(20, current_top_k // 2)
            stall = 0
        
        # Margin-based constraint selection
        threshold = M_curr * (1.0 - margin_frac)
        active_mask = ac >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(ac))])
        
        if len(active_idx) > current_top_k:
            active_vals = ac[active_idx]
            order = np.argsort(active_vals)[::-1][:current_top_k]
            active_idx = active_idx[order]
        
        # Solve LP
        direction = solve_lp_direction(a, M_curr, active_idx, solver)
        if direction is None:
            if len(active_idx) > 15:
                direction = solve_lp_direction(a, M_curr, active_idx[:15], solver)
            if direction is None:
                if time.monotonic() - t0 > time_limit * 0.5:
                    break
                continue
        
        # Line search
        best_alpha = 0.0
        best_M_trial = M_curr
        for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            # Tiny step
            a_trial = 0.999 * a + 0.001 * direction
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            if fast_autocorr(a_trial).max() < M_curr:
                a = a_trial
            else:
                if time.monotonic() - t0 > time_limit * 0.3:
                    break
                continue
        
        a = (1.0 - best_alpha) * a + best_alpha * direction
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        iter_count += 1
        
        if iter_count % 15 == 0 and current_top_k < top_k_max:
            current_top_k += 5
        
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, k={current_top_k}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with multiple move sizes."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))
        
        # Find top contributing pairs at the peak
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_lag - idx]
        
        # Also check second and third peaks
        sorted_lags = np.argsort(ac)[::-1]
        top_lags = sorted_lags[:min(5, len(sorted_lags))]
        
        all_srcs = set()
        for lag in top_lags:
            lag_i = int(lag)
            s = max(0, lag_i - n + 1)
            e = min(n - 1, lag_i)
            for j in range(s, e + 1):
                if a[j] * a[lag_i - j] > 0.7 * M_curr / max(je - js + 1, 1):
                    all_srcs.add(j)
                    all_srcs.add(lag_i - j)
        
        src_list = list(all_srcs)[:30]
        if not src_list:
            src_list = [int(np.argmax(a))]
        
        improved = False
        
        for src in src_list:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find good destinations (outside peak regions)
            in_peak = np.zeros(n, dtype=bool)
            for lag in top_lags[:3]:
                lag_i = int(lag)
                s = max(0, lag_i - n + 1)
                e = min(n - 1, lag_i)
                in_peak[s:e + 1] = True
                s2 = max(0, lag_i - (n - 1))
                e2 = min(n - 1, lag_i + (n - 1))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            
            n_try = min(80, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:n_try]]
            
            found = False
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.35]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = int(rng.integers(1, 4))
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-15)
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
        else:
            stall = 0
        
        if verbose and stall % 30 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_init(n, seed):
    """Generate a structured random initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    strat = seed % 6
    if strat == 0:
        a = np.sin(np.pi * x) ** 2
    elif strat == 1:
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif strat == 2:
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif strat == 3:
        a = rng.random(n)
        k = np.ones(15) / 15
        a = np.convolve(a, k, mode='same')
    elif strat == 4:
        a = np.ones(n) * 0.5
        for _ in range(3):
            freq = rng.uniform(2, 15)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0)
    else:
        a = np.sin(np.pi * x) ** 2 * (1 + 0.2 * rng.random(n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def optimize_single_length(n, time_budget, solver, parent_seq=None, verbose=True):
    """
    Full optimization pipeline for a single sequence length.
    Returns the best sequence found.
    """
    if parent_seq is not None and len(parent_seq) == n:
        a = normalize(parent_seq)
    elif parent_seq is not None:
        a = interpolate_sequence(parent_seq, n)
    else:
        a = random_init(n, n)
    
    # Spectral preprocessing
    a = spectral_equalize(a, n_iter=15)
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    def elapsed():
        return time.monotonic() - t0
    
    def rem_time():
        return time_budget - elapsed()
    
    # Stage 1: SGD
    if rem_time() > time_budget * 0.3:
        t_sgd = min(rem_time() * 0.35, time_budget * 0.35)
        a_out, m_out = smooth_gradient_descent(a, t_sgd, beta_init=30, beta_end=150,
                                                c0=1.5, top_k=80, verbose=verbose)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
    
    # Stage 2: LP FW
    if rem_time() > 30:
        t_lp = min(rem_time() * 0.35, time_budget * 0.35)
        a_in = normalize(best_a)
        a_out, m_out = lp_frank_wolfe(a_in, t_lp, solver, top_k_init=50, top_k_max=150,
                                       margin_frac=0.05, verbose=verbose)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
    
    # Stage 3: Coordinate polish
    if rem_time() > 20:
        t_coord = min(rem_time() * 0.4, 60.0)
        a_in = normalize(best_a)
        a_out, m_out = coordinate_polish(a_in, t_coord, verbose=False, rng_seed=n)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    - Load parent, evaluate it
    - Multi-length exploration with full optimization per length
    - Cross-length refinement
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def remaining():
        return time_budget - (time.monotonic() - t_global)
    
    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = arr.sum()
        if s < 0.01:
            return
        arr_norm = arr / s
        sc = evaluate_safe(arr_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={time.monotonic()-t_global:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2400
        best_seq = random_init(n_base, 42)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Quick spectral improvement on parent ===
    if remaining() > 500:
        print(f"\n=== Phase 1: Spectral on parent n={n_parent} ===")
        a_spec = spectral_equalize(best_seq, n_iter=25)
        try_update(a_spec, "spec_parent")
    
    # === Phase 2: Full optimization on parent length ===
    if remaining() > 400:
        t2 = min(remaining() * 0.25, 400.0)
        print(f"\n=== Phase 2: Optimize n={n_parent}, budget={t2:.0f}s ===")
        a_out, m_out = optimize_single_length(n_parent, t2, solver, parent_seq=best_seq, verbose=True)
        try_update(a_out, f"opt_n{n_parent}")
    
    # === Phase 3: Multi-length exploration ===
    rem = remaining()
    if rem > 300:
        # Generate candidate lengths
        n_best = len(best_seq)
        candidates = set()
        # Nearby lengths
        for d in range(-300, 301, 50):
            c = n_best + d
            if 1200 <= c <= 4500:
                candidates.add(c)
        # Some nice lengths
        for nice in [2048, 2200, 2350, 2500, 2650, 2800, 3000, 3200, 3500, 4000]:
            if 1200 <= nice <= 4500:
                candidates.add(nice)
        
        # Remove the parent length (already tried)
        candidates.discard(n_best)
        candidates = sorted(candidates)
        
        # Select top candidates based on remaining time
        max_alts = min(len(candidates), max(2, int(rem / 180)))
        # Prefer lengths close to n_best
        candidates_sorted = sorted(candidates, key=lambda x: abs(x - n_best))
        selected = candidates_sorted[:max_alts]
        
        print(f"\n=== Phase 3: Multi-length exploration ({len(selected)} lengths), rem={rem:.0f}s ===")
        
        per_len_time = max(40, (rem * 0.6) / len(selected))
        
        for alt_n in selected:
            if remaining() < 80:
                break
            print(f"  --- n={alt_n} (budget {per_len_time:.0f}s) ---")
            a_out, m_out = optimize_single_length(alt_n, per_len_time, solver,
                                                   parent_seq=best_seq, verbose=False)
            try_update(a_out, f"alt_n{alt_n}")
    
    # === Phase 4: Refined LP on best ===
    rem = remaining()
    if rem > 150:
        n_best = len(best_seq)
        t4 = min(rem * 0.3, 250.0)
        print(f"\n=== Phase 4: LP refined n={n_best}, budget={t4:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, m_out = lp_frank_wolfe(a_in, t4, solver, top_k_init=80, top_k_max=250,
                                       margin_frac=0.02, verbose=True)
        try_update(a_out, f"lp_refined_n{n_best}")
    
    # === Phase 5: SGD with high beta (fine-tuning) ===
    rem = remaining()
    if rem > 100:
        n_best = len(best_seq)
        t5 = min(rem * 0.25, 180.0)
        print(f"\n=== Phase 5: Fine SGD n={n_best}, budget={t5:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, m_out = smooth_gradient_descent(a_in, t5, beta_init=150, beta_end=400,
                                                c0=0.5, top_k=120, verbose=True)
        try_update(a_out, f"fine_sgd_n{n_best}")
    
    # === Phase 6: Coordinate polish ===
    rem = remaining()
    if rem > 60:
        n_best = len(best_seq)
        t6 = min(rem * 0.35, 150.0)
        print(f"\n=== Phase 6: Coord polish n={n_best}, budget={t6:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, m_out = coordinate_polish(a_in, t6, verbose=True, rng_seed=999)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 7: Final LP ===
    rem = remaining()
    if rem > 60:
        n_best = len(best_seq)
        t7 = min(rem * 0.5, 120.0)
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        a_in = normalize(best_seq)
        a_out, m_out = lp_frank_wolfe(a_in, t7, solver, top_k_init=100, top_k_max=300,
                                       margin_frac=0.01, verbose=True)
        try_update(a_out, f"final_lp_n{n_best}")
    
    # === Phase 8: Last coordinate polish ===
    rem = remaining()
    if rem > 20:
        n_best = len(best_seq)
        t8 = min(rem * 0.7, 60.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final coord n={n_best} ===")
            a_in = normalize(best_seq)
            a_out, m_out = coordinate_polish(a_in, t8, verbose=False, rng_seed=2024)
            try_update(a_out, f"final_coord_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
