    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        peak_val = conv.max()

        # Find top lags
        top_lags = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            # Contributions to conv[k]: a[i] * a[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)

            for i in range(i_start, i_end):
                j = k - i
                prod = a[i] * a[j]
                if prod < 1e-10:
                    continue

                # Try reducing a[i] or a[j] and adding to random positions
                for scale in step_scales:
                    amt = prod * scale
                    if amt > 0.5 * max(a[i], a[j]):
                        amt = 0.3 * max(a[i], a[j])

                    # Try moving from i to a random position
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == i or dst == j:
                            continue
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[dst] += amt
                        if a_trial[i] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        m_trial = fast_conv(a_trial).max()
                        if m_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break

        if improved:
            stall = 0
        else:
            stall += 1
            # Larger random perturbation
            idx = rng.integers(n, size=2)
            i, j = int(idx[0]), int(idx[1])
            if i != j and a[i] > 1e-6:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[j] -= amt
                    a[i] += amt
                    a = a / a.sum()

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False):
    """Fine coordinate descent polish."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try random pairs
        for _ in range(min(200, n // 10)):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue
            for frac in [0.001, 0.003, 0.01, 0.03]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m = fast_conv(a_trial).max()
                if m < best_m - 1e-15:
                    a = a_trial
                    best_m = m
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            # Perturb
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
    return best_a, best_m


def make_initialization(n, kind='cosine', rng=None):
    """Create various parametric initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.arange(n) / n
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cosine':
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = np.maximum(a, 0.0)
    elif kind == 'gaussian':
        center = n / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x * n - center) / sigma) ** 2)
    elif kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    elif kind == 'multimodal':
        a = np.zeros(n)
        for c in [0.2, 0.5, 0.8]:
            a += np.exp(-0.5 * ((x - c) / 0.1) ** 2)
    elif kind == 'random_smooth':
        a = np.ones(n)
        for _ in range(5):
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            a += np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def try_update(a, label, n_override=None):
    """Try to update the global best."""
    global _global_best, _global_best_score
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    try:
        a_list = [float(v) for v in a_norm]
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _global_best_score:
        _global_best_score = sc
        _global_best = a_norm.copy()
        elapsed = time.monotonic() - _global_t0
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    global _global_best, _global_best_score, _global_t0
    _global_t0 = time.monotonic()
    _global_best = None
    _global_best_score = float('inf')
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    # Load parent sequence
    n_base = 2400
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_a = p / p.sum()
            n_base = len(parent_a)
            try_update(parent_a, "parent")
            print(f"Parent loaded: n={n_base}, score={_global_best_score:.8f}")
    except Exception:
        pass

    if parent_a is None:
        parent_a = make_initialization(n_base, 'cosine')
        try_update(parent_a, "default_init")
        print(f"Default init: n={n_base}, score={_global_best_score:.8f}")

    # Define n values to try
    n_values = sorted(set([
        n_base,
        max(500, n_base - 800),
        n_base + 400,
        n_base + 800,
        n_base + 1200,
    ]))
    # Filter to reasonable range
    n_values = [n for n in n_values if 500 <= n <= 5000]

    # For each n, create initializations
    init_kinds = ['cosine', 'gaussian', 'raised_cosine', 'multimodal', 'random_smooth']
    rng = np.random.default_rng(2024)

    # Allocate time: roughly equal per (n, init) pair, but give more to parent-based
    # Total pairs: len(n_values) * len(init_kinds) + 1 (parent)
    total_pairs = len(n_values) * len(init_kinds) + 1
    time_per_pair = time_budget / (total_pairs * 1.5)  # 1.5x factor for overhead

    results = []  # list of (score, a, n, label)

    # Phase 1: L-BFGS with smoothing from diverse starts
    print("\n=== Phase 1: L-BFGS multi-start ===")
    beta_schedules = [
        [1.0, 5.0, 20.0, 50.0, 100.0, 200.0],
        [2.0, 10.0, 30.0, 80.0, 200.0, 500.0],
    ]

    # First: optimize the parent
    if _global_t0 is not None and time.monotonic() - _global_t0 < time_budget:
        n_p = len(parent_a)
        x0 = np.log(parent_a + 1e-30)
        t_remaining = time_budget - (time.monotonic() - _global_t0)
        t_alloc = min(time_per_pair * 3, t_remaining * 0.25)
        if t_alloc > 30:
            print(f"\n  [Parent n={n_p}] L-BFGS (budget={t_alloc:.0f}s)")
            for bs_idx, bs in enumerate(beta_schedules):
                if time.monotonic() - _global_t0 > time_budget - 60:
                    break
                x_best, sc = lbfgs_smoothed(x0, bs, max_iter=300,
                                            time_limit=min(t_alloc / (len(beta_schedules) + 1), 200.0),
                                            verbose=False)
                a_best = softmax_from_x(x_best)
                try_update(a_best, f"parent_lbfgs_bs{bs_idx}")

    # Then: diverse starts at various n
    for ni, n in enumerate(n_values):
        if time.monotonic() - _global_t0 > time_budget - 60:
            break
        for ki, kind in enumerate(init_kinds):
            if time.monotonic() - _global_t0 > time_budget - 60:
                break
            t_elapsed = time.monotonic() - _global_t0
            remaining = time_budget - t_elapsed
            if remaining < 30:
                break

            # Create initialization
            if kind == 'from_parent' and parent_a is not None and len(parent_a) != n:
                # Interpolate parent to new n
                old_n = len(parent_a)
                old_x = np.linspace(0, 1, old_n)
                new_x = np.linspace(0, 1, n)
                a_init = np.interp(new_x, old_x, parent_a)
                a_init = np.maximum(a_init, 0.0)
                a_init = a_init / a_init.sum()
            else:
                a_init = make_initialization(n, kind, rng)

            x0 = np.log(a_init + 1e-30)

            # Choose beta schedule
            bs = beta_schedules[ki % len(beta_schedules)]

            t_alloc = min(time_per_pair * 1.5, remaining * 0.3)
            if t_alloc > 20:
                x_best, sc = lbfgs_smoothed(x0, bs, max_iter=200,
                                            time_limit=min(t_alloc, 150.0),
                                            verbose=False)
                a_best = softmax_from_x(x_best)
                try_update(a_best, f"n{n}_{kind}")

    # Phase 2: Peak reduction on best candidates
    print("\n=== Phase 2: Peak reduction ===")
    if _global_best is not None:
        remaining = time_budget - (time.monotonic() - _global_t0)
        if remaining > 60:
            t_pe = min(remaining * 0.3, 300.0)
            print(f"  Peak reduction on best (n={len(_global_best)}, budget={t_pe:.0f}s)")
            a_out, m = peak_reduction_search(_global_best, t_pe, n_peaks=5, verbose=True)
            try_update(a_out, "peak_red")

    # Phase 3: Coordinate polish
    print("\n=== Phase 3: Coordinate polish ===")
    if _global_best is not None:
        remaining = time_budget - (time.monotonic() - _global_t0)
        if remaining > 40:
            t_cd = min(remaining * 0.25, 200.0)
            print(f"  CD polish (n={len(_global_best)}, budget={t_cd:.0f}s)")
            a_out, m = coordinate_polish(_global_best, t_cd, verbose=True)
            try_update(a_out, "cd_polish")

    # Phase 4: Second round of L-BFGS on the current best with higher beta
    print("\n=== Phase 4: High-beta L-BFGS on best ===")
    if _global_best is not None:
        remaining = time_budget - (time.monotonic() - _global_t0)
        if remaining > 30:
            n_b = len(_global_best)
            x0 = np.log(_global_best + 1e-30)
            high_betas = [50.0, 100.0, 200.0, 500.0, 1000.0]
            t_hb = min(remaining * 0.3, 200.0)
            print(f"  High-beta L-BFGS (n={n_b}, budget={t_hb:.0f}s)")
            x_best, sc = lbfgs_smoothed(x0, high_betas, max_iter=300,
                                        time_limit=min(t_hb, 200.0), verbose=False)
            a_best = softmax_from_x(x_best)
            try_update(a_best, "high_beta")

            # Also try a different beta schedule
            remaining = time_budget - (time.monotonic() - _global_t0)
            if remaining > 20:
                alt_betas = [20.0, 60.0, 150.0, 400.0]
                t_alt = min(remaining * 0.3, 100.0)
                x_best2, sc2 = lbfgs_smoothed(x0, alt_betas, max_iter=200,
                                              time_limit=min(t_alt, 100.0), verbose=False)
                a_best2 = softmax_from_x(x_best2)
                try_update(a_best2, "alt_beta")

    # Phase 5: Final round - try a few more n values with the best shape
    print("\n=== Phase 5: Reshape and retry ===")
    if _global_best is not None:
        remaining = time_budget - (time.monotonic() - _global_t0)
        if remaining > 60:
            best_n = len(_global_best)
            alt_ns = [best_n - 200, best_n + 200, best_n + 400]
            alt_ns = [n for n in alt_ns if 400 <= n <= 5000 and n != best_n]
            for alt_n in alt_ns[:2]:
                if time.monotonic() - _global_t0 > time_budget - 30:
                    break
                # Reshape best to new n
                old_n = len(_global_best)
                old_x = np.linspace(0, 1, old_n)
                new_x = np.linspace(0, 1, alt_n)
                a_res = np.interp(new_x, old_x, _global_best)
                a_res = np.maximum(a_res, 0.0)
                a_res = a_res / a_res.sum()

                x0 = np.log(a_res + 1e-30)
                t_alloc = min(60.0, (time_budget - (time.monotonic() - _global_t0)) * 0.3)
                if t_alloc > 20:
                    x_best, sc = lbfgs_smoothed(x0, [30.0, 100.0, 300.0], max_iter=200,
                                                time_limit=min(t_alloc, 80.0), verbose=False)
                    a_best = softmax_from_x(x_best)
                    try_update(a_best, f"reshape_n{alt_n}")

                    # Quick peak reduction
                    if time.monotonic() - _global_t0 < time_budget - 10:
                        a_out, _ = peak_reduction_search(a_best, 20.0, n_peaks=3)
                        try_update(a_out, f"reshape_peak_n{alt_n}")

    # Final validation
    if _global_best is not None:
        fs = evaluate_sequence([float(x) for x in _global_best])
        if np.isfinite(fs):
            _global_best_score = fs
        print(f"\nFINAL: score={_global_best_score:.8f}, n={len(_global_best)}, "
              f"time={time.monotonic()-_global_t0:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        fallback = make_initialization(1000, 'cosine')
        return [float(x) for x in fallback]

    return [float(x) for x in _global_best]
