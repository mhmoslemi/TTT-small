    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

    """
    One LP-based Frank-Wolfe step.
    Returns the direction b (normalized to sum=1).
    """

    ac = fast_autocorr(a)
    M = ac.max()
    
    # Get active lags
    threshold = (1.0 - tol) * M
    active = np.where(ac >= threshold)[0]
    if len(active) > n_active:
        # Keep the most active
        active_ac = ac[active]
        top_idx = np.argsort(active_ac)[::-1][:n_active]
        active = active[top_idx]

    # conv(a,b)[k] = sum_j a[k-j] * b[j] for valid j
    # Coefficient of b[j] in constraint k is a[k-j] (if 0 <= k-j < n)
    
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = np.zeros((len(active), n))
    b_ub = np.full(len(active), M)
    
    for idx, k in enumerate(active):
        k_int = int(k)
        # b[j] appears with coefficient a[k-j] if 0 <= k-j < n
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            j = np.arange(j_start, j_end)
            A_ub[idx, j] = a[k_int - j]
    
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    b = np.maximum(result.x, 0)
    s = b.sum()
    if s < 1e-15:
        return None

        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_re = best_a + pert
            x_re = np.maximum(x_re, 1e-12)
            x = x_re / x_re.sum()
            y = x.copy()
            c_cur *= 1.3
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SGD restart t={t} ({elapsed:.0f}s): c={c_cur:.2f}, obj={best_val:.6f}")
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): obj={best_val:.6f}")
    return best_a, best_val

def coordinate_descent(a_init, time_limit, verbose=True, seed=123):
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_val = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 80
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lags = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False
        hot_positions = set()
        for k in peak_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot_positions.add(i)
                jj = k_int - i
                if 0 <= jj < n:
                    hot_positions.add(jj)
        hot_list = list(hot_positions)
        if not hot_list:
            break
        cold_positions = [i for i in range(n) if i not in hot_positions]
        if len(cold_positions) > 200:
            cold_positions = list(rng.choice(cold_positions, 200, replace=False))
        rng.shuffle(hot_list)
        n_hot = min(len(hot_list), 50)
        for src in hot_list[:n_hot]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            sorted_cold = sorted(cold_positions, key=lambda j: abs(a[j] - a[src] * 0.5))
            n_cold = min(len(sorted_cold), 40)
            for dst in sorted_cold[:n_cold]:
                if dst == src:
                    continue
                for frac in [0.003, 0.01, 0.03, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_curr - 1e-15:
                        a = a_trial
                        best_val = objective_value(a)
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            n_pert = max(5, n // 80)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < fast_autocorr(best_a).max():
                            best_val = objective_value(a)
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): obj={best_val:.6f}, stall={stall}")
    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        init = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
        init = np.maximum(init, 0)
        init = init / init.sum()
        best_seq = init
        best_score = evaluate_safe(init)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW n={n0} (budget {remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        a_out, _ = accelerated_fw(normalize(best_seq), t1, max_cons=80, tol_frac=0.93,
                                  verbose=True, seed=1)
        try_update(a_out, f"lpfw_n{n0}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW refined n={n_best} ===")
    t2 = min(remaining * 0.25, 440.0)
    if t2 > 30:
        a_out, _ = accelerated_fw(normalize(best_seq), t2, max_cons=120, tol_frac=0.90,
                                  verbose=True, seed=2)
        try_update(a_out, f"lpfw2_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient AMD n={n_best} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = subgradient_amd(normalize(best_seq), t3, c0=2.5, top_k=50,
                                   power=3.5, verbose=True, seed=3)
        try_update(a_out, f"sgrad_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent n={n_best} ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 20:
        a_out, _ = coordinate_descent(normalize(best_seq), t4, verbose=True, seed=4)
        try_update(a_out, f"coord_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW polish n={n_best} ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_out, _ = accelerated_fw(normalize(best_seq), t5, max_cons=150, tol_frac=0.85,
                                  verbose=True, seed=5)
        try_update(a_out, f"lpfw_polish_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length exploration ===")
    if remaining > 120:
        tried_lengths = set([n_best])
        length_candidates = [n_best - 300, n_best - 150, n_best + 150, n_best + 300, n_best + 500]
        length_candidates = [nc for nc in length_candidates if 800 <= nc <= 4000 and nc != n_best]
        rng_ml = np.random.default_rng(99)
        rng_ml.shuffle(length_candidates)
        for nc in length_candidates[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            if nc in tried_lengths:
                continue
            tried_lengths.add(nc)
            print(f"  Trying n={nc} (remaining {remaining:.0f}s)")
            a_interp = interpolate_sequence(best_seq, nc)
            t_ml = min(remaining * 0.30, 200.0)
            if t_ml > 20:
                a_out, _ = accelerated_fw(normalize(a_interp), t_ml, max_cons=80,
                                          tol_frac=0.90, verbose=False, seed=len(nc))
                try_update(a_out, f"lpfw_ml_n{nc}")
                if time.monotonic() - t_global > time_budget - 30:
                    break
                a_out2, _ = coordinate_descent(normalize(best_seq), min(remaining * 0.1, 60.0),
                                               verbose=False, seed=nc % 1000)
                try_update(a_out2, f"coord_ml_n{len(a_out2) if a_out2 is not None else nc}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coordinate polish n={n_best} ===")
    if remaining > 20:
        t7 = min(remaining * 0.35, 150.0)
        a_out, _ = coordinate_descent(normalize(best_seq), t7, verbose=True, seed=10)
        try_update(a_out, f"coord_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP-FW n={n_best} ===")
        t8 = min(remaining * 0.5, 120.0)
        a_out, _ = accelerated_fw(normalize(best_seq), t8, max_cons=100, tol_frac=0.88,
                                  verbose=False, seed=11)
        try_update(a_out, f"lpfw_final_n{n_best}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
