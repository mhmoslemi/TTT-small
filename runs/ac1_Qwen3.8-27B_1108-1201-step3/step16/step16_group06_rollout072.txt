import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_seq(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize_positive(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_init(n, shape):
    x = np.linspace(0.0, 1.0, n)
    if shape == 'ramp':
        a = np.linspace(1.0, 2.0, n)
    elif shape == 'ramp2':
        a = np.linspace(0.1, 2.0, n)
    elif shape == 'sin2':

    elif shape == 'sin4':
        a = np.sin(np.pi * x) ** 4
    elif shape == 'gauss':
        a = np.exp(-3.0 * (2.0 * x - 1.0) ** 2)
    elif shape == 'parab':
        a = 1.0 - (2.0 * x - 1.0) ** 2

    else:

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def softmax_grad(y, C, beta):
    """Compute gradient of soft-max of C with respect to y, using FFT."""
    n = len(y)
    shifted = beta * (C - C.max())
    p = np.exp(shifted)
    ps = p.sum()
    if ps < 1e-30:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = p / ps
    y_flip = y[::-1]
    conv_pw = np.convolve(p, y_flip)
    grad = 2.0 * conv_pw[n - 1:2 * n - 1]
    return grad


def topk_grad(y, top_idx, weights):
    """Compute gradient for top-k weighted peaks (direct method)."""
    n = len(y)
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        ki = int(k)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += 2.0 * wi * y[ki - idx]
    return grad

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    eta = eta0
    stall = 0
    stall_limit = 400
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute autocorrelation
        C = fast_autocorr(y)

        # Compute gradient
        if use_topk:
            k_actual = min(top_k, 2 * n - 1)
            top_idx = np.argsort(C)[::-1][:k_actual]
            top_vals = np.maximum(C[top_idx], 1e-30)
            w = top_vals ** 3.0
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones(k_actual) / k_actual
            else:
                w = w / ws
            grad = topk_grad(y, top_idx, w)
        else:
            grad = softmax_grad(y, C, beta)

        # Mirror descent (multiplicative) update
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        x_prev = x
        x = a_new
        iter_count += 1

        # Decay step size
        eta = eta0 / (1.0 + 0.003 * iter_count)

        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Perturbation to escape local minima
        if stall > stall_limit:
            stall = 0
            scale = 0.003 + 0.005 * min(1.0, iter_count / 10000.0)
            noise = rng.normal(0, scale, n)
            x_pert = best_a * (1.0 + noise)
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            x = x_pert
            x_prev = x_pert
            eta = eta0 * 0.3

        if verbose and iter_count % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={iter_count} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coord_polish(a0, time_limit, rng_seed=99, verbose=False):
    """Peak-targeted coordinate descent: move mass from high-contrib to low-contrib positions."""
    n = len(a0)
    a = normalize_positive(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        C = fast_autocorr(a)
        k_peak = int(np.argmax(C))

        j_lo = max(0, k_peak - n + 1)
        j_hi = min(n - 1, k_peak)
        js = np.arange(j_lo, j_hi + 1)
        if len(js) == 0:
            break
        contrib = a[js] * a[k_peak - js]
        if contrib.max() < 1e-20:
            break

        src_order = np.argsort(contrib)[::-1]
        improved = False

        # Compute contribution of each position to the peak
        peak_contrib_all = np.zeros(n)
        for j in js:
            peak_contrib_all[j] = a[j] * a[k_peak - j]

        for si in range(min(15, len(src_order))):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(js[src_order[si]])
            if a[src] < 1e-12:
                continue

            # Find best destination (low peak contribution)
            candidates = np.argsort(peak_contrib_all)[:50]
            for di in range(min(20, len(candidates))):
                dst = int(candidates[di])
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    M_t = fast_autocorr(a_t).max()
                    if M_t < best_m - 1e-15:
                        a = a_t
                        best_m = M_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Try random small move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()

    if verbose:
        J = 2.0 * n * best_m
        print(f"    CoordPD ({time.monotonic() - t0:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_n = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_n)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_seq(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = eval_seq(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    if best_seq is None:
        best_seq = make_init(2400, 'ramp2')
        best_score = eval_seq(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-start mirror descent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.35
    print(f"\n=== Phase 1: Multi-start MD ({t_phase1:.0f}s) ===")

    shapes = ['ramp2', 'ramp', 'asym_cos', 'sin2', 'gauss', 'parab']
    t_per = max(8.0, t_phase1 / (len(shapes) + 1))

    # Try parent first if available
    if parent is not None:
        t_avail = min(t_per, time_budget - (time.monotonic() - t_global) - 20)
        if t_avail > 5:
            print(f"  MD parent (n={n0}, t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent(normalize_positive(parent), t_avail,
                                      beta=5.0, eta0=0.4, momentum=0.85,
                                      verbose=False, rng_seed=1)
            try_update(a_out, "md_parent")

    for i, shape in enumerate(shapes):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_avail = min(t_per, remaining - 30)
        if t_avail < 5:
            break
        print(f"  MD '{shape}' t={t_avail:.0f}s...")
        a_init = make_init(n0, shape)
        a_out, _ = mirror_descent(a_init, t_avail,
                                  beta=5.0, eta0=0.4, momentum=0.85,
                                  verbose=False, rng_seed=100 + i)
        try_update(a_out, f"md_{shape}")

    # === Phase 2: Focused MD on best (25% time) with increasing beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t_phase2 = remaining * 0.25
        print(f"\n=== Phase 2: Focused MD on best (t={t_phase2:.0f}s) ===")
        n_cur = len(best_seq)

        # Run with beta=3 (smooth)
        t_sub = t_phase2 * 0.3
        a_out, _ = mirror_descent(normalize_positive(best_seq), t_sub,
                                  beta=3.0, eta0=0.5, momentum=0.9,
                                  verbose=False, rng_seed=200)
        try_update(a_out, "md_beta3")

        # Run with beta=8 (sharper)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_sub = min(t_phase2 * 0.4, remaining - 30)
            if t_sub > 10:
                a_out, _ = mirror_descent(normalize_positive(best_seq), t_sub,
                                          beta=8.0, eta0=0.3, momentum=0.9,
                                          verbose=False, rng_seed=300)
                try_update(a_out, "md_beta8")

        # Run with top-k mode
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_sub = min(t_phase2 * 0.3, remaining - 20)
            if t_sub > 10:
                a_out, _ = mirror_descent(normalize_positive(best_seq), t_sub,
                                          beta=5.0, eta0=0.3, momentum=0.85,
                                          verbose=False, rng_seed=400,
                                          use_topk=True, top_k=80)
                try_update(a_out, "md_topk")

    # === Phase 3: Coordinate polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_phase3 = remaining * 0.15
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Coord polish n={n_cur} (t={t_phase3:.0f}s) ===")
        a_out, _ = coord_polish(normalize_positive(best_seq), t_phase3,
                                rng_seed=500, verbose=True)
        try_update(a_out, "coord_p3")

    # === Phase 4: MD round 2 after polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t_phase4 = remaining * 0.15
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: MD round 2 n={n_cur} (t={t_phase4:.0f}s) ===")
        a_out, _ = mirror_descent(normalize_positive(best_seq), t_phase4,
                                  beta=10.0, eta0=0.2, momentum=0.9,
                                  verbose=False, rng_seed=600)
        try_update(a_out, "md_r2")

    # === Phase 5: Try slightly different lengths (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t_phase5 = remaining * 0.10
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: Multi-n exploration ===")
        test_ns = [n_cur + 200, n_cur - 200, n_cur + 400]
        t_per_n = max(10.0, t_phase5 / len(test_ns))

        for nt in test_ns:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            if nt < 800 or nt > 4000:
                continue
            print(f"  Trying n={nt} (t={t_per_n:.0f}s)...")
            a_interp = interpolate_to(best_seq, nt)
            a_out, _ = mirror_descent(normalize_positive(a_interp), t_per_n,
                                      beta=5.0, eta0=0.4, momentum=0.85,
                                      verbose=False, rng_seed=700 + nt)
            try_update(a_out, f"md_n{nt}")

    # === Phase 6: Final coordination round (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_phase6 = remaining * 0.6
        n_cur = len(best_seq)
        print(f"\n=== Phase 6: Final polish n={n_cur} (t={t_phase6:.0f}s) ===")
        a_out, _ = coord_polish(normalize_positive(best_seq), t_phase6,
                                rng_seed=800, verbose=False)
        try_update(a_out, "coord_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_final = remaining * 0.7
            a_out, _ = mirror_descent(normalize_positive(best_seq), t_final,
                                      beta=15.0, eta0=0.15, momentum=0.9,
                                      verbose=False, rng_seed=900)
            try_update(a_out, "md_final")

    # Final validation
    if best_seq is not None:
        fs = eval_seq(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
