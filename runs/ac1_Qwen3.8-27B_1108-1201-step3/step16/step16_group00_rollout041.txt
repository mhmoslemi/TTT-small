def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return norm_sum1(a)


def make_exp_ramp(n, rate=0.5):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(rate * x / n)
    return norm_sum1(a)


def make_symmetric_ramp(n):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.abs(x - center) + 1.0
    return norm_sum1(a)


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200, n_restart=3):
    """LP-based Frank-Wolfe with top-K constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for restart in range(n_restart):
        y = a.copy()
        x = a.copy()
        prev_y = a.copy()

        for t in range(1, 500000):
            if time.monotonic() - t0 > time_limit:
                break

            T_full = np.zeros((2 * n - 1, n))
            for j in range(n):
                T_full[j:j + n, j] = y

            ac = fast_autocorr(y)
            M_curr = ac.max()
            if M_curr < 1e-16:
                break

            constraint_vals = T_full @ y
            margins = M_curr - constraint_vals
            K_sel = min(top_K, 2 * n - 1)
            binding_idx = np.argsort(margins)[:K_sel]

            T_sel = T_full[binding_idx, :]
            M_sel = M_curr * np.ones(K_sel)

            b_var = cp.Variable(n, nonneg=True)
            constraints = [T_sel @ b_var <= M_sel]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                continue
            g_dir = b_val / b_sum

            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy

            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx

            prev_y = y
            y = y_new
            x = x_new

            my = fast_autocorr(y).max()
            if my < best_m:
                best_m = my
                best_a = y.copy()

            mx = fast_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()

            if verbose and t % 20 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LP r{restart} iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")

        # After each LP phase, restart from best
        a = best_a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP r{restart} done ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Careful coordinate descent that only accepts improvements."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find positions with low "sensitivity"
            scores = np.zeros(n)
            for k2 in top_k[:min(10, n_top)]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    d2 = d - k2i
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    if 0 <= d2 < n:
                        scores[d] += a[d2]

            candidates = np.where(a < best_m * 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            candidate_scores = scores[candidates]
            if len(candidate_scores) > 0:
                best_dst_idx = candidates[np.argmin(candidate_scores)]
            else:
                best_dst_idx = int(rng.integers(n))

            dst = int(best_dst_idx)
            if dst == src:
                continue

            for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    stall = 0
                    break

        if not improved_this_round:
            stall += 1
            if stall > 30:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def mixed_ramp_lp(a0, time_limit, solver, verbose=True, top_K=150):
    """
    Mixed approach: alternate between LP-FW steps and direct
    gradient-based steps toward the linear ramp direction.
    This helps escape local minima while maintaining progress.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute linear ramp direction
    linear = make_power_ramp(n, 1.0)

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Every 20 iterations, blend toward linear ramp direction
        if t % 20 == 0:
            blend = 0.1
            y = (1 - blend) * y + blend * linear
            y = np.maximum(y, 0.0)
            y = y / y.sum()

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MixedLP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Directly evaluate analytical candidates (linear ramp is key!)
    2. LP-FW from best candidate
    3. Coordinate descent polish
    4. Length extension if beneficial
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # === Phase 1: Direct evaluation of analytical candidates ===
    print("\n=== Phase 1: Analytical candidates ===")

    # Try different lengths
    lengths_to_try = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
    powers_to_try = [0.9, 1.0, 1.05, 1.1, 1.2, 1.5]

    results = []

    for n in lengths_to_try:
        for p in powers_to_try:
            if time.monotonic() - t_global > time_budget - 100:
                break
            a = make_power_ramp(n, p)
            sc = evaluate_safe(a)
            results.append((sc, n, p, a))
            if np.isfinite(sc):
                print(f"  Power ramp n={n}, p={p:.2f}: score={sc:.6f}")

    # Also try exponential
    for n in [3000, 5000, 8000]:
        for rate in [0.3, 0.5, 1.0, 2.0]:
            a = make_exp_ramp(n, rate)
            sc = evaluate_safe(a)
            results.append((sc, n, rate, a))
            if np.isfinite(sc):
                print(f"  Exp ramp n={n}, rate={rate:.1f}: score={sc:.6f}")

    # Filter valid results and find best
    valid_results = [(sc, n, p, a) for sc, n, p, a in results if np.isfinite(sc)]
    if valid_results:
        valid_results.sort(key=lambda x: x[0])
        best_sc, best_n, best_p, best_a = valid_results[0]
        print(f"\n  Best analytical: score={best_sc:.6f}, n={best_n}, p={best_p}")
        best_seq = best_a.copy()
        best_score = best_sc

    # Try parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_score = evaluate_safe(p)
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = p.copy()
                print(f"  Parent is better: {p_score:.6f} (n={len(p)})")
            else:
                print(f"  Parent score: {p_score:.6f} (n={len(p)}) - not better")
    except Exception:
        pass

    # Compute theoretical score for linear ramp at each length
    print("\n  Theoretical linear ramp scores:")
    for n in [2000, 3000, 4000, 5000, 6000, 8000, 10000]:
        s = 4.0 * (n**3 + 6*n**2 - n - 6) / (3.0 * n * (n+1)**2)
        print(f"    n={n}: theoretical={s:.6f}")

    # === Phase 2: LP-FW from best analytical candidate ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (remaining budget: {remaining:.0f}s) ===")

    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60 and n_best <= 8000:
        print(f"  Running LP-FW on n={n_best} for {t2:.0f}s...")
        a_in = norm_sum1(best_seq)
        top_K = min(200, n_best)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=top_K, n_restart=2)
        try_update(a_out, f"lp_fw_n{n_best}")

    # === Phase 3: Coordinate descent on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate descent (remaining: {remaining:.0f}s) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        print(f"  Running coord descent on n={n_best} for {t3:.0f}s...")
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 4: Mixed LP with linear ramp blending ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Mixed LP (remaining: {remaining:.0f}s) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30 and n_best <= 8000:
        print(f"  Running MixedLP on n={n_best} for {t4:.0f}s...")
        a_in = norm_sum1(best_seq)
        a_out, _ = mixed_ramp_lp(a_in, t4, solver, verbose=True, top_K=150)
        try_update(a_out, f"mixed_n{n_best}")

    # === Phase 5: Try extending the best sequence to longer lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length extension (remaining: {remaining:.0f}s) ===")

    n_best = len(best_seq)
    # Try extending by interpolation
    extension_lengths = []
    for ext in [n_best + 1000, n_best + 2000, n_best + 4000]:
        if ext <= 12000 and ext > n_best:
            extension_lengths.append(ext)

    for new_n in extension_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Extending to n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        sc = evaluate_safe(a_long)
        print(f"    Interpolated score: {sc:.6f}")
        try_update(a_long, f"interp_n{new_n}")

        # Also try fresh linear ramp at new length
        a_fresh = make_power_ramp(new_n, 1.0)
        sc2 = evaluate_safe(a_fresh)
        print(f"    Fresh linear ramp score: {sc2:.6f}")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a_fresh.copy()
            print(f"  *** Fresh linear ramp at n={new_n} is better!")

        # Quick LP on the extended/fresh best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 8000:
            t5 = min(remaining * 0.4, 150.0)
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=False, top_K=150, n_restart=1)
            try_update(a_out, f"lp_ext_n{new_n}")

    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final coord polish (remaining: {remaining:.0f}s) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.4, 200.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: One more LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (remaining: {remaining:.0f}s) ===")
    if remaining > 30 and n_best <= 8000:
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=300, n_restart=1)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback linear ramp.")
        n_fb = 4000
        best_seq = make_power_ramp(n_fb, 1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}, n={n_fb}")

    return [float(x) for x in best_seq]
