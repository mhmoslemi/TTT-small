```python
def smoothed_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    # FFT sizes
    ac_size = 2 * n - 1
    ac_fft_size = 1 << (ac_size - 1).bit_length()
    grad_fft_size = 1 << (3 * n - 3).bit_length()  # need >= (2n-1) + n - 1 = 3n - 2
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    # Precompute
    A_fft = np.fft.rfft(a, ac_fft_size)
    
    # parameters
    eps = 0.01  # smoothing parameter
    eta = 0.1 / n  # learning rate
    momentum = 0.9
    
    v = np.zeros(n)  # momentum velocity
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        
        # Compute weights
        w = np.exp(ac / eps)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(ac_size) / ac_size
        else:
            w = w / w_sum
        
        # Compute gradient via FFT
        W = np.zeros(ac_fft_size)
        W[:ac_size] = w
        A = np.zeros(ac_fft_size)
        A[:n] = a
        
        W_fft = np.fft.rfft(W, grad_fft_size)
        A_fft = np.fft.rfft(A, grad_fft_size)
        cross = np.fft.irfft(W_fft * np.conj(A_fft), grad_fft_size)[:n]
        
        g = (2.0 / eps) * cross
        
        # Gradient step with momentum
        v = momentum * v + g
        a_new = a - eta * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check improvement
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        score_new = 2.0 * n * M_new
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive epsilon
        if t % 1000 == 0:
            eps *= 0.95
            if eps < 1e-4:
                eps = 1e-4
                
        if verbose and t % 10000 == 0:
            print(f"    SGD t={t}: score={best_score:.6f}, eps={eps:.6f}")
    
    return best_a, best_score
```

One more thought: the **exact projection** onto {a >= 0, sum(a) = 1} is:

The function f(tau) = sum(max(v[i] - tau, 0)) is monotonically decreasing in tau.
f(-inf) = inf, f(inf) = 0.

tau = (sum_{i: v[i] > tau} v[i] - 1) / count.

import numpy as np
import time

def fast_conv(a):
    """Compute np.convolve(a, a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def exact_projection(v):
    """Project v onto {a >= 0, sum(a) = 1} (Euclidean projection)."""
    n = len(v)
    # Sort in decreasing order
    order = np.argsort(v)[::-1]
    v_sorted = v[order]
    # Find the right tau
    # For k active elements: tau = (sum(v_sorted[:k]) - 1) / k
    # We need v_sorted[k-1] > tau >= v_sorted[k] (with v_sorted[n] = -inf)
    
    cumulative = np.cumsum(v_sorted)
    ks = np.arange(1, n + 1)
    taus = (cumulative - 1.0) / ks
    # Need: v_sorted[k-1] > taus[k-1]
    # i.e., v_sorted[ks-1] > (cumulative[ks-1] - 1) / ks
    # i.e., ks * v_sorted[ks-1] > cumulative[ks-1] - 1
    # i.e., (ks-1) * v_sorted[ks-1] + v_sorted[ks-1] > cumulative[ks-1] - 1
    # i.e., (ks-1) * (v_sorted[ks-1]) + v_sorted[ks-1] + 1 > cumulative[ks-1]
    # Hmm, let me just use the condition directly:
    # v_sorted[k-1] > tau_k = (cumsum_k - 1) / k
    # means: k * v_sorted[k-1] > cumsum_k - 1
    # means: k * v_sorted[k-1] + 1 > cumsum_k
    
    valid = ks * v_sorted > (cumulative - 1.0)
    # Also need tau >= 0 for the constraint to be active (optional)
    # Find the largest k that is valid
    valid_indices = np.where(valid)[0]
    if len(valid_indices) == 0:
        # All elements are zero? Just return uniform
        return np.ones(n) / n
    k_best = valid_indices[-1] + 1  # 1-indexed
    tau = (cumulative[k_best - 1] - 1.0) / k_best
    a = np.maximum(v - tau, 0.0)
    return a  # Should already sum to ~1

def smoothed_mgd(a0, time_limit, verbose=True, rng_seed=42):
    """
    Smoothed Mirror/Gradient Descent with log-sum-exp.
    Transitions from smooth to exact subgradient as eps -> 0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Precompute FFT sizes
    ac_size = 2 * n - 1
    ac_fft_size = 1 << (ac_size - 1).bit_length()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    rng = np.random.default_rng(rng_seed)
    
    # Parameters
    eps = 1.0 / n  # initial smoothing (gives moderate smoothness)
    eps_min = 1e-5 / n
    eps_decay = 0.97
    momentum = 0.0  # will increase
    momentum_target = 0.9
    eta_base = 5.0 * n  # will be scaled by 1/sqrt(t)
    
    # For gradient computation via FFT
    # cross[j] = sum_{k=0}^{n-1} w[j+k] * a[k]
    # Use: cross = IFFT(FFT(w, N) * conj(FFT(a, N)))[:n] where N >= 3n-2
    grad_fft_size = 1 << (3 * n - 3).bit_length()
    
    v = np.zeros(n)  # momentum buffer
    x = a.copy()  # Nesterov position
    x_mom = a.copy()
    
    for t in range(1, 10000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (1.0 + np.sqrt(1 + 4 * momentum * momentum)) / 2
            # Actually, use standard Nesterov:
            # y = x + (t-1)/(t+1.5) * (x - x_prev)
            beta_nest = (t - 1.0) / (t + 1.5) * momentum
            y = x + beta_nest * (x - x_mom)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            y = y / sy
        else:
            y = a
        
        x_mom = x
        
        # Compute autocorrelation of y
        ac = fast_conv(y)
        
        # Compute weights (softmax with temperature eps)
        ac_shifted = ac - ac.max()  # for numerical stability
        w = np.exp(ac_shifted / eps)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(ac_size) / ac_size
        else:
            w = w / w_sum
        
        # Compute gradient: g[j] = (2/eps) * sum_k w[j+k] * y[k]
        # = (2/eps) * cross_correlation(w, y)[j]
        W = np.zeros(grad_fft_size)
        W[:ac_size] = w
        Y = np.zeros(grad_fft_size)
        Y[:n] = y
        
        W_fft = np.fft.rfft(W, grad_fft_size)
        Y_fft = np.fft.rfft(Y, grad_fft_size)
        cross = np.fft.irfft(W_fft * np.conj(Y_fft), grad_fft_size)[:n]
        
        g = (2.0 / eps) * cross
        
        # Normalize gradient (optional, for consistent step size)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-30:
            g = g / g_norm  # unit norm gradient
        
        # Step size
        eta = eta_base / (np.sqrt(t) + 1.0)
        
        # Update with momentum
        v = momentum * v + g
        a_new = y - eta * v
        a_new = exact_projection(a_new)
        
        x = a_new
        
        # Evaluate
        M_new = fast_conv(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        # Decay epsilon
        if t % 500 == 0:
            eps = max(eps * eps_decay, eps_min)
            # Increase momentum gradually
            momentum = min(momentum + 0.001, momentum_target)
        
        if verbose and t % 20000 == 0:
            print(f"    SMD t={t} ({elapsed:.1f}s): M={best_m:.8f}, score={2*n*best_m:.6f}, eps={eps:.2e}")
    
    return best_a, best_m
