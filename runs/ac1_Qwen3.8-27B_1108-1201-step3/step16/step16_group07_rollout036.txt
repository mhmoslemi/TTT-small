def project_simplex(v, S):
    """Exact projection onto {a >= 0, sum(a) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

    # Build constraint matrix T: T[idx, j] = a_fixed[k - j] if 0 <= k-j < n
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        # b_j appears in (a*b)[k] with coefficient a_{k-j} if 0 <= k-j < n
        # So j ranges from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

        polyak_step = min(polyak_step, 4.0 / np.sqrt(n))

        # Blend with 1/sqrt(t) step
        sqrt_step = base_step / np.sqrt(max(t, 1))
        step = 0.5 * polyak_step + 0.5 * min(sqrt_step, 6.0 / np.sqrt(n))
        step = max(step, 0.1 / np.sqrt(n))

        # Gradient step
        x_new = y_extrap - step * grad
        # Project onto simplex {a >= 0, sum(a) = norm}
        x_new = project_simplex(x_new, norm)

        # Update ergodic average (geometric weighting)
        alpha_avg = 2.0 / (t + 2.0)
        x_avg = (1.0 - alpha_avg) * x_avg + alpha_avg * x_new

        # Nesterov update for y
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * x_new
        y_new = project_simplex(y_new, norm)

        y_prev = y
        y = y_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_avg = fast_autocorr(x_avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = x_avg.copy()
            stall = 0

        # Progressively increase beta
        if t % 100 == 0:
            beta *= 1.15

        # Restart if stalled
        if stall > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            x_avg = best_a.copy()
            stall = 0
            beta = max(5.0, beta * 0.5)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMS iter={t} ({elapsed:.0f}s): best={best_score:.10f}, beta={beta:.1f}, step={step:.6f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with progressive active-set tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 60
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Progressive active set
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            # Start with wider tolerance, narrow over time
            tol = 0.12 * (1.0 - progress) + 0.008 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(5, n // 60)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap for efficiency
            max_active = int(2.0 * n)
            if len(active_set) > max_active:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Retry with full active set
            if t % 5 == 0:
                active_set = np.arange(conv_size)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                # Fallback: use subgradient step
                k_max = np.argmax(conv_y)
                g = subgradient_at_k(y, k_max, n)
                g_norm = np.linalg.norm(g)
                if g_norm > 1e-15:
                    b = y - 0.5 / np.sqrt(n) * g / g_norm * n
                    b = np.maximum(b, 0.0)
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        b = b * norm / b_sum
                    else:
                        continue
                else:
                    continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.90)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Periodic check of damped average
        if t % 40 == 0 and t > 100:
            a_damped = 0.6 * best_a + 0.4 * a
            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:
                a_damped = a_damped * norm / s_d
                score_d = fast_autocorr(a_damped).max()
                if score_d < best_score:
                    best_score = score_d
                    best_a = a_damped.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def dual_guided_polish(a0, time_limit, solver, verbose=False):
    """Use LP dual variables to guide subgradient updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 500
    step = 1.0 / n ** 0.3

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get active set with adaptive tolerance
        tol = 0.06
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 80):]
        # Cap
        if len(active_set) > 300:
            active_set = np.argsort(conv)[-300:]

        # Solve LP to get duals
        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue

        # Use duals or exponential weights
        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-6.0 * (M_curr - conv[active_set]) / max(M_curr * 0.03, 1e-10))
            weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k(a, int(k), n)

        # Project gradient: zero out components where a is already 0
        support_thresh = 0.002 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            break

        # Direction: reduce where grad is high
        grad_mean = np.mean(grad[support])
        direction = np.zeros(n)
        direction[support] = grad[support] - grad_mean

        a_trial = a - step * direction
        a_trial = project_simplex(a_trial, norm)

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 3.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish iter={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False, seed=0):
    """Fast coordinate descent with efficient autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]

            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Candidate values
            typical = norm / n
            candidates = [0.0, aj * 0.2, aj * 0.5, aj * 0.8, aj * 1.2, aj * 1.6, aj * 2.5]
            candidates += [typical * 0.3, typical * 0.7, typical * 1.3, typical * 2.0, typical * 4.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.4)) for c in candidates]))

            best_x = aj
            best_obj = fast_autocorr(a).max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = (norm - aj) + x
                if new_sum < 1e-15:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def sinusoidal_init(n, mode=0, seed=42):
    """Generate a sinusoidal initialization with various frequency structures."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    if mode == 0:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif mode == 1:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 3 * x + 0.5)
    elif mode == 2:
        a = 0.4 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    elif mode == 3:
        a = 0.3 + 0.4 * np.cos(2 * np.pi * x) ** 2 + 0.2 * np.cos(2 * np.pi * 11 * x)
    elif mode == 4:
        # Edge-emphasized
        a = 0.3 + 0.5 * (1 - np.cos(2 * np.pi * x)) ** 0.5
    elif mode == 5:
        # Center-emphasized
        a = 0.3 + 0.5 * np.cos(np.pi * x) ** 2
    else:
        a = np.ones(n) * 0.5 + rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * 0.5
        s = a.sum()
    return a / s


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
