def fast_autocorr(a):

def norm_sum1(a):

def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_l2_flatness_gradient(a):
    """Gradient of sum_k (c[k] - 1/(2n-1))^2 with respect to a, where c = a*a."""
    n = len(a)
    c = fast_autocorr(a)
    c_mean = 1.0 / (2 * n - 1)
    e = c - c_mean
    # gradient = 2 * (e * a_rev) + 2 * (e * a) (both linear convolutions, truncated to n)
    # (e * a)[i] = sum_j e[j] * a[i-j] gives one part
    # (e * a_rev)[i] = sum_j e[j] * a_rev[i-j] = sum_j e[j] * a[n-1-(i-j)] gives the other
    fft_size = 1 << (2 * n - 1).bit_length()
    E = np.fft.rfft(e, fft_size)
    A = np.fft.rfft(a, fft_size)
    Ar = np.fft.rfft(a[::-1], fft_size)
    grad1 = np.fft.irfft(E * A, fft_size)[:n]
    grad2 = np.fft.irfft(E * Ar, fft_size)[:n]
    return 2.0 * (grad1 + grad2)


def compute_smoothed_max_gradient(a, beta, top_k=200):
    """Gradient of (1/beta)*log(sum exp(beta*c[k])) using top-k approximation."""
    n = len(a)
    c = fast_autocorr(a)
    num = len(c)
    k_eff = min(top_k, num)
    top_idx = np.argsort(c)[::-1][:k_eff]
    top_vals = c[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        # (a*a)[k] = sum_i a[i]*a[k-i], d/d a[j] = a[k-j] + a[j-k] (if valid)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += wi * a[j_idx]
        # Also the "reverse" contribution
        i_start2 = max(0, k_int - n + 1)
        i_end2 = min(n, k_int + 1)
        if i_start2 >= i_end2:
            continue
        idx2 = np.arange(i_start2, i_end2)
        j_idx2 = idx2 - k_int  # a[j-k] where j = idx2, so a[idx2 - k]
        valid = (j_idx2 >= 0) & (j_idx2 < n)
        if valid.any():
            grad[idx2[valid]] += wi * a[j_idx2[valid]]
    return grad, c.max()

        y = project_simplex(y)

        # Combined gradient
        g_max, _ = compute_smoothed_max_gradient(y, cur_beta, top_k=current_top_k)
        g_l2 = compute_l2_flatness_gradient(y)
        # Normalize L2 gradient to comparable scale
        gn_max = np.linalg.norm(g_max)
        gn_l2 = np.linalg.norm(g_l2)
        if gn_max < 1e-15:
            gn_max = 1e-15
        if gn_l2 < 1e-15:
            gn_l2 = 1e-15
        g_combined = g_max / gn_max + cur_l2_weight * g_l2 / gn_l2

        # Step size
        if backtracking:
            eta = c0 / (math.sqrt(t) + 1.0)
            # Backtracking: ensure descent
            M_curr = fast_autocorr(y).max()
            for bt in range(8):
                x_new = project_simplex(y - eta * g_combined)
                M_new_candidate = fast_autocorr(x_new).max()
                if M_new_candidate < M_curr + 1e-12:
                    break
                eta *= 0.5
        else:
            eta = c0 / (math.sqrt(t) + 1.0)
            x_new = project_simplex(y - eta * g_combined)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise_scale = 0.003 + 0.005 * (t / max_iter)
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            cur_l2_weight = max(cur_l2_weight, 0.3)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG restart t={t} ({elapsed:.0f}s): J={J:.6f}, l2w={cur_l2_weight:.3f}")

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, l2w={cur_l2_weight:.3f}, tk={current_top_k}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_constraints=None):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if solver is None:
        return best_a, best_m

    y = a.copy()
    x_mom = a.copy()

    # Precompute the structure of the Toeplitz matrix for constraint generation
    # T[k, j] = y[j] if 0 <= k-j < n, else 0
    # This is the matrix such that (T @ b)[k] = (y * b)[k]

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build LP: maximize sum(b) s.t. (y*b)[k] <= M_curr for all k, b >= 0
        # Use only top constraints for speed
        n_total = 2 * n - 1
        if max_constraints is not None:
            top_indices = np.argsort(ac)[::-1][:max_constraints]
        else:
            top_indices = np.arange(n_total)

        b_var = cp.Variable(n, nonneg=True)

        # Build constraint matrix rows only for top constraints
        T_rows = []
        for k in top_indices:
            k_int = int(k)
            row = np.zeros(n)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    row[i] += y[j]
            T_rows.append(row)

        T = np.array(T_rows)
        rhs = np.array([M_curr] * len(top_indices))

        constraints = [T @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx

        y = y_new

        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-14:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent: peak-aware mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        # Identify peak contributors
        peak_contrib = np.zeros(n)
        for k in top_k[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii, jj in zip(idx, j_idx):
                peak_contrib[ii] += a[jj]

        sources = np.argsort(peak_contrib)[::-1][:25]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Find best destination
            best_dst = -1
            best_delta = 0.0
            n_try = min(n, 400)
            if n > n_try:
                candidates = rng.choice(n, n_try, replace=False)
            else:
                candidates = np.arange(n)

            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_k[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_delta:
                    best_delta = score
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def generate_structured_init(n, rng):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    inits = []

    # Cosine bump with various shapes
    for p in [2, 3, 4, 6, 8]:
        c = (1.0 + np.cos(2.0 * np.pi * x)) ** p
        c = np.maximum(c, 0)
        if c.sum() > 1e-10:
            inits.append(c / c.sum())

    # Bimodal
    for sigma in [0.08, 0.12, 0.15]:
        bm = np.exp(-((x - 0.3) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * sigma ** 2))
        inits.append(bm / bm.sum())

    # Quartic center
    q4 = (1.0 - (2 * x - 1) ** 2) ** 4
    q4 = np.maximum(q4, 0)
    if q4.sum() > 1e-10:
        inits.append(q4 / q4.sum())

    # Random binary with various densities
    for density in [0.2, 0.3, 0.4, 0.5]:
        m = max(1, int(n * density))
        idx = rng.choice(n, m, replace=False)
        b = np.zeros(n)
        b[idx] = 1.0
        inits.append(b / b.sum())

    # Smoothed random
    r = rng.random(n)
    # Apply a simple smoothing
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0)
    if r_smooth.sum() > 1e-10:
        inits.append(r_smooth / r_smooth.sum())

    return inits


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Initialization ===
    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
        n0 = len(best_seq)
    else:
        n0 = 2400
        x = np.linspace(0, 1, n0)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    # === Phase 1: Try structured initializations with quick APG ===
    rem = remaining_time()
    print(f"\n=== Phase 1: Structured inits (n={n0}) ===")
    if rem > 120:
        inits = generate_structured_init(n0, rng)
        t_per = min(rem * 0.08, 30.0)
        tested = 0
        for idx, init in enumerate(inits[:8]):
            rem = remaining_time()
            if rem < 30:
                break
            # Quick score check first
            init_score = evaluate_safe(init)
            if init_score < best_score:
                best_score = init_score
                best_seq = init.copy()
                print(f"  Init {idx} is already better: {init_score:.6f}")
            # Short APG from this init
            t_wp = min(t_per, rem - 20)
            if t_wp > 10:
                a_out, _ = accelerated_projected_gradient(
                    init, t_wp, beta_start=30.0, beta_end=10.0,
                    top_k_start=80, top_k_end=150,
                    l2_weight_start=1.0, l2_weight_end=0.2,
                    c0=1.5, verbose=False, seed=1000 + idx,
                    backtracking=True)
                try_update(a_out, f"struct{idx}")
            tested += 1
        print(f"  Tested {tested} structured inits")

    # === Phase 2: Main APG from best so far ===
    rem = remaining_time()
    print(f"\n=== Phase 2: Main APG (n={len(best_seq)}) ===")
    t2 = min(rem * 0.30, 400.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_projected_gradient(
            a_in, t2, beta_start=25.0, beta_end=6.0,
            top_k_start=150, top_k_end=350,
            l2_weight_start=0.8, l2_weight_end=0.0,
            c0=2.0, verbose=True, seed=42,
            backtracking=True)
        try_update(a_out, f"apg_main_n{len(best_seq)}")

    # === Phase 3: LP Frank-Wolfe ===
    rem = remaining_time()
    print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
    t3 = min(rem * 0.18, 250.0)
    if t3 > 30 and len(best_seq) <= 3000 and solver is not None:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                  max_constraints=min(4 * len(best_seq), 3000))
        try_update(a_out, f"lpfw_n{len(best_seq)}")

    # === Phase 4: APG polish after LP ===
    rem = remaining_time()
    print(f"\n=== Phase 4: APG polish ===")
    t4 = min(rem * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_projected_gradient(
            a_in, t4, beta_start=15.0, beta_end=8.0,
            top_k_start=200, top_k_end=400,
            l2_weight_start=0.3, l2_weight_end=0.0,
            c0=1.5, verbose=False, seed=77,
            backtracking=True)
        try_update(a_out, f"apg_polish")

    # === Phase 5: Coordinate polish ===
    rem = remaining_time()
    print(f"\n=== Phase 5: Coord polish ===")
    t5 = min(rem * 0.10, 100.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Multi-length exploration ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length (current n={n_best}) ===")

    # Try longer
    for n_trial in [n_best + 400, n_best + 800, n_best + 1200]:
        rem = remaining_time()
        if rem < 100 or n_trial > 4000:
            break
        a_long = interpolate_sequence(best_seq, n_trial)
        a_in = norm_sum1(a_long)
        t6 = min(rem * 0.12, 120.0)
        if t6 > 20:
            a_out, _ = accelerated_projected_gradient(
                a_in, t6, beta_start=20.0, beta_end=7.0,
                top_k_start=150, top_k_end=300,
                l2_weight_start=0.5, l2_weight_end=0.0,
                c0=2.0, verbose=False, seed=200 + n_trial,
                backtracking=True)
            try_update(a_out, f"long_n{n_trial}")

            # Quick coord polish
            rem = remaining_time()
            if rem > 30:
                t6b = min(rem * 0.05, 40.0)
                if t6b > 10:
                    a_out2, _ = coordinate_polish(a_out, t6b, verbose=False, seed=55)
                    try_update(a_out2, f"coord_long_n{n_trial}")

    # Try shorter (if current is large)
    if n_best > 1500:
        rem = remaining_time()
        if rem > 80:
            n_short = max(1200, n_best - 600)
            a_short = interpolate_sequence(best_seq, n_short)
            a_in = norm_sum1(a_short)
            t6s = min(rem * 0.10, 80.0)
            if t6s > 20:
                a_out, _ = accelerated_projected_gradient(
                    a_in, t6s, beta_start=20.0, beta_end=8.0,
                    top_k_start=150, top_k_end=250,
                    l2_weight_start=0.5, l2_weight_end=0.0,
                    c0=2.0, verbose=False, seed=300,
                    backtracking=True)
                try_update(a_out, f"short_n{n_short}")

    # === Phase 7: Final LP at best length ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 2800 and rem > 50 and solver is not None:
        print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
        t7 = min(rem * 0.20, 120.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True,
                                      max_constraints=min(4 * n_best, 3000))
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    if rem > 30:
        print(f"\n=== Phase 8: Final coord polish (n={n_best}) ===")
        t8 = min(rem * 0.50, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: One more APG with very fine settings ===
    rem = remaining_time()
    if rem > 40:
        print(f"\n=== Phase 9: Fine APG ===")
        t9 = min(rem * 0.50, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = accelerated_projected_gradient(
                a_in, t9, beta_start=10.0, beta_end=5.0,
                top_k_start=250, top_k_end=500,
                l2_weight_start=0.1, l2_weight_end=0.0,
                c0=1.0, verbose=False, seed=999,
                backtracking=True)
            try_update(a_out, f"fine_apg")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
