        # The actual iterate:
        momentum = (t - 1.0) / (t + 2.0)
        momentum = min(momentum, 0.9)
        a_new = y_new + momentum * (y_new - y_prev)
        a_new = simplex_project(a_new, 1.0)
        
        y_prev = y
        y = y_new
        a = a_new
        n_iters += 1
        
        # Track best
        score_y = fast_conv(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            patience = 0
            # Increase step slightly
            step = min(step * 1.01, 1.0)
        else:
            score_a = fast_conv(a).max()
            if score_a < best_score - 1e-12:
                best_score = score_a
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                step *= 0.98
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    GradDesc iter={n_iters} ({elapsed:.1f}s): best={best_score:.8f}, T={T:.4f}, step={step:.4f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=0, verbose=False):
    """
    Aggressive coordinate descent.
    Works on simplex (sum=1, a>=0).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t_start = time.monotonic()
    
    no_improve_passes = 0
    max_no_improve = 3
    n_passes = 0
    
    while time.monotonic() - t_start < time_limit:
        n_passes += 1
        if no_improve_passes >= max_no_improve:
            break
        
        conv = fast_conv(a)
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t_start > time_limit:
                break
            
            aj = a[j]
            
            # Generate candidates
            candidates = np.array([0.0, aj * 0.2, aj * 0.5, aj, aj * 1.5, aj * 2.0])
            # Add gradient-informed candidates
            # The subgradient at j suggests direction
            grad_j = 0.0
            # Find which k in [j, j+n-1] has max conv
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            if k_lo <= k_hi:
                max_k = k_lo + np.argmax(conv[k_lo:k_hi + 1])
                if max_k == 2 * j:
                    grad_j = 2 * aj
                else:
                    grad_j = 2 * a[max_k - j]
            
            # Add candidates in the suggested direction
            if grad_j > 0:
                candidates = np.append(candidates, aj * 0.8)
            candidates = np.append(candidates, 1.0 / n)
            candidates = np.append(candidates, 2.0 / n)
            
            # Filter to valid range
            candidates = np.clip(candidates, 0.0, 1.0)
            candidates = np.unique(candidates)
            
            best_v = aj
            best_obj = best_score  # Use current best as benchmark
            
            for v in candidates:
                if abs(v - aj) < 1e-14:
                    continue
                
                # Compute new conv in affected range
                # conv_new[k] for k in [j, j+n-1]
                delta = v - aj
                new_conv_seg = conv.copy()
                
                # For k in [j, k_hi]:
                if k_lo <= k_hi:
                    new_conv_seg[k_lo:k_hi + 1] += 2.0 * delta * a[k_lo - j:k_hi - j + 1]
                    # Correction for k = 2j
                    if k_lo <= 2 * j <= k_hi:
                        new_conv_seg[2 * j] += delta * delta
                
                # All other positions unchanged
                new_sum = 1.0 - aj + v
                if new_sum < 1e-12:
                    continue
                
                # Score = max(new_conv) / new_sum^2
                new_score = new_conv_seg.max() / (new_sum * new_sum)
                
                if new_score < best_obj:
                    best_obj = new_score
                    best_v = v
            
            if abs(best_v - aj) > 1e-12:
                a[j] = best_v
                a = normalize(a, n)
                conv = fast_conv(a)
                curr_score = conv.max()
                if curr_score < best_score - 1e-13:
                    best_score = curr_score
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve_passes += 1
        else:
            no_improve_passes = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    CD pass={n_passes} ({elapsed:.1f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response_round(a0, M, active_set, n):
    """Solve LP best response."""
    # Build constraint matrix
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a0[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val / b_val.sum()

def lp_optimize(a0, time_limit, verbose=True):
    """LP best-response iteration with progressive active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t_start = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t_start > time_limit:
            break
        if stall > 100:
            break
        
        conv_y = fast_conv(y)
        M_curr = conv_y.max()
        
        # Active set
        tol = 0.05 * (1.0 - min(1.0, t / 2000.0)) + 0.005
        active_set = np.where(conv_y >= M_curr * (1.0 - tol))[0]
        n_top = max(5, n // 50)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))
        
        b = lp_best_response_round(y, M_curr, active_set, n)
        if b is None:
            stall += 1
            continue
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = simplex_project(y_new, 1.0)
        
        momentum = min((t - 1.0) / (t + 2.0), 0.9)
        a_new = y_new + momentum * (y_new - y_prev)
        a_new = simplex_project(a_new, 1.0)
        
        y_prev = y
        y = y_new
        a = a_new
        
        score_y = fast_conv(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            score_a = fast_conv(a).max()
            if score_a < best_score - 1e-12:
                best_score = score_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
    
    if verbose:
        print(f"    LP done: best={best_score:.8f}")
    
    return best_a, best_score
