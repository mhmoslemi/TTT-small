def fast_autocorr(a):
    """Full linear autocorrelation via FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_full_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*ac[k])) w.r.t. a.
    Returns (gradient of length n, autocorrelation of length 2n-1).
    """
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    shifted = ac - max_ac
    exp_vals = np.exp(beta * shifted)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(ac)
        w[np.argmax(ac)] = 1.0
    else:
        w = exp_vals / w_sum

    fft_size = 1 << (3 * n - 1).bit_length()
    Fa = np.fft.rfft(a, fft_size)
    Fw = np.fft.rfft(w, fft_size)
    cross_corr_wa = np.fft.irfft(Fw * np.conj(Fa), fft_size)[:n]
    cross_corr_aw = np.fft.irfft(Fa * np.conj(Fw), fft_size)[:n]
    grad = 2.0 * (cross_corr_wa + cross_corr_aw)
    return grad, ac


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def parametric_candidates(n):
    """Generate diverse parametric initializations of length n."""
    x = np.linspace(0, 1, n)
    candidates = []

    # Power-law back-loaded
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (x + 1e-8) ** p
        if a.sum() > 0:
            candidates.append(a / a.sum())

    # Power-law front-loaded
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (1.0 - x + 1e-8) ** p
        if a.sum() > 0:
            candidates.append(a / a.sum())

    # Sine power
    for p in [1, 2, 3, 4, 6, 10]:
        a = np.sin(np.pi * x) ** p
        if a.sum() > 0:
            candidates.append(a / a.sum())

    # Gaussian
    for mu in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.15, 0.25, 0.4]:
            a = np.exp(-((x - mu) / sigma) ** 2)
            if a.sum() > 0:
                candidates.append(a / a.sum())

    # Exponential
    for alpha in [1, 2, 3, 5, 10]:
        a = np.exp(-alpha * x)
        if a.sum() > 0:
            candidates.append(a / a.sum())
        a = np.exp(-alpha * (1 - x))
        if a.sum() > 0:
            candidates.append(a / a.sum())

    # Bimodal
    for mu1, mu2 in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.7)]:
        a = np.exp(-40 * (x - mu1) ** 2) + np.exp(-40 * (x - mu2) ** 2)
        if a.sum() > 0:
            candidates.append(a / a.sum())

    # Random
    rng = np.random.default_rng(0)
    for _ in range(5):
        a = np.abs(rng.normal(0, 1, n))
        if a.sum() > 0:
            candidates.append(a / a.sum())

    return candidates


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return np.asarray(f, dtype=np.float64).copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                a_out, _ = apg_full(a_in, t, beta_start=10.0, beta_end=3.0,
                                    c0=1.0, verbose=False, seed=500 + round_i,
                                    restart_patience=1000)
                try_update(a_out, f"final_apg{round_i}")
                a_in = normalize_nonneg(best_seq)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
