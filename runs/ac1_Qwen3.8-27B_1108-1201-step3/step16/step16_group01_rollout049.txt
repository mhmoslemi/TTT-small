Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth proximal gradient with log-sum-exp approximation and adaptive beta
2. Euclidean projection onto simplex (exact, O(n log n))
3. Backtracking line search
4. AMD with exact active-set subgradient for final polish
5. Coordinate descent for local refinement
6. Multi-length exploration with diverse initializations
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def proj_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_objective(a, beta):
    """Smooth approximation to max(conv(a,a)) using log-sum-exp."""
    ac = fast_autocorr(a)
    max_ac = ac.max()
    return max_ac + np.log(np.sum(np.exp(beta * (ac - max_ac)))) / beta

def smooth_gradient(a, beta):
    """Gradient of smooth approximation using FFT-based cross-correlation."""
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    exp_ac = np.exp(beta * (ac - max_ac))
    weights = exp_ac / exp_ac.sum()
    
    # grad[j] = sum_k weights[k] * a[k-j] for valid indices
    # This is cross-correlation of weights (len 2n-1) and a (len n)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    return corr[:n]

def proximal_gradient(a0, time_limit, beta_start=5.0, beta_end=200.0, 
                      eta_start=0.1, max_iters=100000, restart_count=0, verbose=True):
    """
    Proximal gradient method for minimizing max(conv(a,a)) over the simplex.
    Uses log-sum-exp smooth approximation with adaptive beta.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    it = 0
    
    # Adaptive beta: start low (smoother), increase over time
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        progress = min(it / (max_iters * 0.8), 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient
        grad = smooth_gradient(a, beta)
        
        # Lipschitz constant estimate: L ~ beta * max(a) * 2 (rough)
        # Use backtracking line search
        # f(a + eta*grad) should decrease
        f_curr = smooth_objective(a, beta)
        
        eta = eta_start / (1.0 + 0.01 * it)
        c = 0.01
        rho = 0.5
        
        for ls in range(30):
            a_new = proj_simplex(a - eta * grad)
            f_new = smooth_objective(a_new, beta)
            if f_new <= f_curr - c * eta * np.dot(grad, a - a_new):
                break
            eta *= rho
        
        a = a_new
        
        # Check actual (hard) objective
        M = fast_autocorr(a).max()
        if M < best_val - 1e-15:
            best_val = M
            best_a = a.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PG it={it} ({elapsed:.1f}s): beta={beta:.1f}, eta={eta:.4f}, M={M:.6f}, best={best_val:.6f}")
    
    return best_a, best_val

def nesterov_smooth(a0, time_limit, beta_start=10.0, beta_end=500.0,
                    c0=3.0, max_iters=200000, verbose=True):
    """
    Nesterov-accelerated mirror descent with smooth gradient.
    """
    n = len(a0)
    a = a0.copy()
    x = a0.copy()  # previous iterate for Nesterov
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (it - 1.0) / (it + 1.5)
        y = a + beta_t * (a - x)
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:

    """Gradient of (1/beta)*log(sum_k exp(beta*conv_k(a)))."""
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    exp_ac = np.exp(beta * (ac - max_ac))
    weights = exp_ac / exp_ac.sum()

        if stall > max_stall:
            stall = 0
            noise_scale = 0.005 + 0.005 * min(stall / max_stall, 1.0)
            noise = rng.normal(0, noise_scale, n) * best_a
            a_restarted = np.maximum(best_a + noise, 1e-15)
            a_restarted = a_restarted / a_restarted.sum()
            a = a_restarted
            x_prev = a_restarted
            c = c * 1.1

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_val
            print(f"      NMD it={it} ({elapsed:.1f}s): beta={smooth_beta:.1f}, obj={obj:.6f}, M={best_val:.8f}")

    return best_a, best_val


def coordinate_descent_peaks(a0, time_limit, verbose=True, rng_seed=123):
    """Peak-aware coordinate descent: systematically reduce peak autocorrelation contributors."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-20:
            break
        peak_lags = np.argsort(ac)[::-1][:min(15, len(ac))]

        improved = False

        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]

            if contrib.max() < 1e-20:
                continue

            max_pos = np.argmax(contrib)
            src_i = idx[max_pos]
            src_j = j_idx[max_pos]
            src = src_i if a[src_i] >= a[src_j] else src_j

            # Build sensitivity map
            sens = np.zeros(n)
            for k2 in peak_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                j2 = k2i - idx2
                sens[idx2] += a[j2]

            # Find low-sensitivity destinations
            low_sens = np.argsort(sens)[:50]
            tried = 0
            for dst in low_sens:
                if time.monotonic() - t0 > time_limit:
                    break
                if dst == src or sens[dst] > sens[src] * 0.5:
                    continue
                tried += 1
                if tried > 20:
                    break
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > max_stall:
                n_pert = max(5, n // 50)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.01, 0.04)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                if stall > max_stall * 3:
                    break
            if verbose and stall % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"      CD ({elapsed:.1f}s): obj={2.0*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def multi_pass_nodecent(a0, time_limit, c_values, beta_pairs, verbose=True):
    """Run Nesterov MD with multiple (c, beta) parameter combinations sequentially."""
    a = a0.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for c_val, (b_start, b_end) in zip(c_values, beta_pairs):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        seg_time = remaining / max(1, len(c_values))
        a_out, m_out = nesterov_smooth_md(
            a, seg_time, beta_start=b_start, beta_end=b_end,
            c=c_val, verbose=verbose, rng_seed=int(c_val * 1000)
        )
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
        a = a_out.copy()

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search: Nesterov smooth mirror descent with adaptive beta annealing,
    followed by coordinate descent polish and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    a_norm = best_seq / best_seq.sum()
    n0 = len(a_norm)

    # === Phase 1: Nesterov smooth MD, main parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov smooth MD n={n0} (c=3, beta 5->300) ===")
    t1 = min(remaining * 0.22, 360.0)
    if t1 > 30:
        a_out, _ = nesterov_smooth_md(a_norm, t1, beta_start=5.0, beta_end=300.0,
                                      c=3.0, verbose=True, rng_seed=1)
        try_update(a_out, f"nmd1_n{n0}")

    # === Phase 2: Nesterov smooth MD, different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Nesterov smooth MD n={n_best} (c=5, beta 10->500) ===")
    t2 = min(remaining * 0.20, 330.0)
    if t2 > 30:
        a_out, _ = nesterov_smooth_md(best_seq / best_seq.sum(), t2, beta_start=10.0, beta_end=500.0,
                                      c=5.0, verbose=True, rng_seed=2, switch_to_hard=0.85)
        try_update(a_out, f"nmd2_n{n_best}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coordinate descent n={n_best} ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 15:
        a_out, _ = coordinate_descent_peaks(best_seq / best_seq.sum(), t3, verbose=True, rng_seed=3)
        try_update(a_out, f"cd1_n{n_best}")

    # === Phase 4: Nesterov smooth MD, aggressive ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Nesterov smooth MD aggressive n={n_best} (c=8, beta 50->800) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        a_out, _ = nesterov_smooth_md(best_seq / best_seq.sum(), t4, beta_start=50.0, beta_end=800.0,
                                      c=8.0, verbose=True, rng_seed=4, switch_to_hard=0.7)
        try_update(a_out, f"nmd4_n{n_best}")

    # === Phase 5: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent 2 n={n_best} ===")
    t5 = min(remaining * 0.10, 160.0)
    if t5 > 15:
        a_out, _ = coordinate_descent_peaks(best_seq / best_seq.sum(), t5, verbose=True, rng_seed=5)
        try_update(a_out, f"cd2_n{n_best}")

    # === Phase 6: Nesterov from best with low beta (explore) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Nesterov explore n={n_best} (c=2, beta 2->100) ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 20:
        a_out, _ = nesterov_smooth_md(best_seq / best_seq.sum(), t6, beta_start=2.0, beta_end=100.0,
                                      c=2.0, verbose=True, rng_seed=6)
        try_update(a_out, f"nmd6_n{n_best}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        print(f"\n=== Phase 7: Multi-length exploration ===")
        lengths_to_try = [n_best + 300, n_best - 300, n_best + 600, n_best - 600]
        lengths_to_try = [l for l in lengths_to_try if 800 <= l <= 3500]
        for n_try in lengths_to_try[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={n_try}")
            a_interp = interpolate_sequence(best_seq, n_try)
            t_seg = min(remaining * 0.25, 120.0)
            if t_seg > 20:
                a_out, _ = nesterov_smooth_md(a_interp, t_seg, beta_start=5.0, beta_end=300.0,
                                              c=3.0, verbose=False, rng_seed=7)
                try_update(a_out, f"nmd_len_n{n_try}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_best} ===")
    if remaining > 30:
        t8 = min(remaining * 0.25, 150.0)
        if t8 > 10:
            a_out, _ = coordinate_descent_peaks(best_seq / best_seq.sum(), t8, verbose=True, rng_seed=8)
            try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 9: Final Nesterov polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final Nesterov polish n={n_best} ===")
        t9 = min(remaining * 0.40, 150.0)
        if t9 > 15:
            a_out, _ = nesterov_smooth_md(best_seq / best_seq.sum(), t9, beta_start=100.0, beta_end=500.0,
                                          c=4.0, verbose=False, rng_seed=9, switch_to_hard=0.5)
            try_update(a_out, f"nmd_polish_n{n_best}")

    # === Phase 10: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.50, 100.0)
        if t10 > 10:
            a_out, _ = coordinate_descent_peaks(best_seq / best_seq.sum(), t10, verbose=False, rng_seed=10)
            try_update(a_out, f"cd_polish_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
