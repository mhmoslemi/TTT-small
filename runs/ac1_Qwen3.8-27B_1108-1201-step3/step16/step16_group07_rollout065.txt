def fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def subgradient_at_k(a, k, n):
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        grad[lo:hi] = 2.0 * a[k - np.arange(lo, hi)]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def mirror_descent_polish(a0, time_limit, verbose=False, seed=42):

    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 300
    rng = np.random.default_rng(seed)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        conv = fast_autocorr(a)
        M_curr = conv.max()
        n_peaks = max(5, min(20, n // 50))
        top_indices = np.argsort(conv)[-n_peaks:]
        weights = np.exp(-10.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k(a, int(k), n)
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            break
        exp_factor = np.exp(-eta * grad / max_grad)
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new * norm / a_new.sum()
        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-14:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            eta = min(eta * 1.01, 5.0 / np.sqrt(n))
        else:
            patience += 1
            eta *= 0.98
            if eta < 1e-6:
                break
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): best={best_score:.10f}")

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 5
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)
            candidates = [0.0, aj * 0.3, aj * 0.6, aj * 1.4, aj * 1.8, norm / n]
            candidates.append(aj + (rng.random() - 0.5) * 0.1 * norm / n)
            candidates = [max(0.0, c) for c in candidates]
            best_x = aj
            best_obj = best_score
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                new_conv = conv.copy()
                delta = x - aj
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                new_max = new_conv.max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

def subgradient_polish(a0, time_limit, verbose=False, seed=0):

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:

            break
        conv = fast_autocorr(a)
        M_curr = conv.max()
        n_peaks = max(3, min(10, n // 80))
        top_indices = np.argsort(conv)[-n_peaks:]
        if t % 15 == 0:
            threshold = M_curr * (0.85 + 0.1 * rng.random())
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices.astype(int), k_rand)
        weights = np.exp(-8.0 * (M_curr - conv[top_indices.astype(int)]) / max(M_curr * 0.02, 1e-10))

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        a_trial = a - step * grad * n ** 0.35
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()

            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def perturb_and_optimize(best_seq, time_limit, solver, rng, seed):

    t2 = min(remaining * 0.10, 150.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Mirror descent n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = mirror_descent_polish(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "mirror", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp2", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Full LP polish n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = full_lp_polish(a_norm, t4, solver, verbose=True)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "fulllp", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cd", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        print(f"\n=== Phase 6: LP Nesterov r3 n={n_cur}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=7777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp3", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t7_total = min(remaining * 0.10, 120.0)
    t7_each = t7_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t7 = min(t7_each, remaining - 15)
        if t7 < 15:
            break
        print(f"\n=== Phase 7r{r}: Perturb n={n_cur}, budget={t7:.0f}s ===")
        a_out = perturb_and_optimize(best_seq, t7, solver, rng, seed=500 + r * 111)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.4, 60.0)
        print(f"\n=== Phase 8: Final CD n={n_cur}, budget={t8:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t8, verbose=False, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cdFinal", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.5, 40.0)
        print(f"\n=== Phase 9: Final subgrad n={n_cur}, budget={t9:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_polish(a_norm, t9, verbose=False, seed=314)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "subgFinal", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
