import time

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def mirror_descent(a0, time_limit, top_k=50, beta=3.0, eta0=1.0, verbose=True, rng_seed=42):
    """Nesterov-accelerated mirror descent on soft-max of autocorrelation."""
    n = len(a0)
    a = normalized(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta = eta0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        mu = 0.9  # momentum
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute autocorrelation
        C = fast_autocorr(y)
        
        # Find top-k peaks
        k_actual = min(top_k, len(C))
        top_idx = np.argsort(C)[::-1][:k_actual]
        top_vals = np.maximum(C[top_idx], 1e-30)
        
        # Soft weights
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_actual) / k_actual
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            # dC[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # First term: j in [max(0, k-n+1), min(k, n-1)], a[k-j]
            j1_start = max(0, k_int - n + 1)
            j1_end = min(n, k_int + 1)
            if j1_start < j1_end:
                j1 = np.arange(j1_start, j1_end)
                grad[j1] += wi * y[k_int - j1]
            
            # Second term: j in [max(0, k_int), min(n, k_int+n)], a[j-k]
            # 0 <= j-k < n => k <= j < k+n
            j2_start = max(0, k_int)
            j2_end = min(n, k_int + n)
            if j2_start < j2_end:
                j2 = np.arange(j2_start, j2_end)
                grad[j2] += wi * y[j2 - k_int]

        # a_new[j] = a[j] * exp(-eta * grad[j])

import time

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_seq(a):

    s = a.sum()
    if s < 1e-15:

def make_initialization(n, shape_type='sin2'):
    x = np.linspace(0.0, 1.0, n)
    if shape_type == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif shape_type == 'sin4':
        a = np.sin(np.pi * x) ** 4
    elif shape_type == 'cos':

        j1_end = min(n, k_int + 1)

            grad[j1] += wi * y[k_int - j1]

        j2_end = min(n, k_int + n)

            grad[j2] += wi * y[j2 - k_int]
    return grad

def mirror_descent_accelerated(a0, time_limit, top_k=30, beta=3.0, eta0=0.5, momentum=0.9, verbose=False, rng_seed=42, adapt_k=True):
    n = len(a0)
    a = normalize_positive(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta = eta0
    stall = 0
    stall_threshold = 300
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute autocorrelation
        C = fast_autocorr(y)
        M_curr = C.max()
        
        # Adaptive top-k
        k_actual = min(top_k, len(C))
        if adapt_k and M_curr < best_m * 1.01:
            k_actual = min(max(k_actual, 20), 80)
        
        top_idx = np.argsort(C)[::-1][:k_actual]
        top_vals = np.maximum(C[top_idx], 1e-30)
        
        # Weights
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_actual) / k_actual
        else:
            w = w / w_sum
        
        # Gradient
        grad = compute_gradient(y, top_idx, w)
        
        # Mirror descent update
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        iter_count += 1
        
        # Decay eta
        eta = eta0 / (1.0 + 0.005 * iter_count)
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Perturbation
        if stall > stall_threshold:
            stall = 0
            noise_scale = 0.005 + 0.01 * min(1.0, stall_threshold / 1000.0)
            noise = rng.normal(0, noise_scale, n)
            x_pert = best_a * (1.0 + noise)
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            x = x_pert
            x_prev = x_pert
            eta = eta0 * 0.5  # Reduce step size after perturbation
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={iter_count} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Targeted coordinate descent on peak-contributing positions."""
    n = len(a0)
    a = normalize_positive(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        C = fast_autocorr(a)
        # Find the single worst peak
        k_peak = int(np.argmax(C))
        
        # Find contributing positions
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        js = np.arange(j_start, j_end + 1)
        contrib = a[js] * a[k_peak - js]
        
        if contrib.max() < 1e-20:
            break
        
        # Try to reduce the peak by moving mass from high-contribution to low-impact positions
        # Sort by contribution (descending)
        src_order = np.argsort(contrib)[::-1]
        
        # Compute "impact" of each position on the peak
        # Moving mass from src to dst changes C[k_peak] by -a[dst]*a[src] + ... (complex)
        # Simpler: try small transfers from top-contributing to random low-contributing positions
        improved = False
        for si in src_order[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(js[si])
            # Find a good destination: low contribution to the peak
            dst_candidates = np.arange(n)
            # Exclude positions near the peak
            dst_candidates = dst_candidates[dst_candidates != src]
            
            # Score destinations by their contribution to the peak (lower is better)
            dst_contrib = np.zeros(n)
            for j in dst_candidates:
                kj = k_peak - j
                if 0 <= kj < n:
                    dst_contrib[j] = a[j] * a[kj]
            
            dst_order = np.argsort(dst_contrib)[:20]
            
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                dst = int(dst_order[0])  # Try the best candidate first
                if dst == src:
                    continue
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    CoordPD ({time.monotonic()-t0:.1f}s): J={J:.6f}")
    return best_a, best_m
