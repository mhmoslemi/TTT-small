Actually, I want to reconsider the LP-FW formulation. The key question is: what's the right LP to solve at each FW step?

The problem: `min_{a ∈ Δ} f(a)` where `f(a) = max_k (a ⊛ a)_k` and `Δ = {a ≥ 0 : sum(a) = 1}`.

At point `a`, the subgradient of `f` is:
`∂f(a) = conv{∇f_k(a) : k ∈ K_active}` where `K_active = {k : f_k(a) = f(a)}`.

The FW step minimizes a linear function over the simplex. The best linear lower bound of `f` at `a` is:
`L(b) = f(a) + g^T (b - a)` for some `g ∈ ∂f(a)`.

Minimizing `L(b)` over `Δ` is equivalent to minimizing `g^T b` over `Δ` (since `f(a) - g^T a` is constant).

The minimum of `g^T b` over the simplex is achieved at a vertex: `b = e_i` where `i = argmin_j g_j`.

So the standard FW step is: find the coordinate with the smallest subgradient component, and move towards that basis vector.

But we have the "max" structure, so `g` is a convex combination of the individual gradients. The best choice of `g` (to minimize `g^T b`) gives the best FW direction.

This is equivalent to solving:
`min_t` s.t. `g = sum_k λ_k ∇f_k(a)`, `λ_k ≥ 0`, `sum λ_k = 1`, ... 

Actually, this gets complicated. The standard approach for "min max" problems in FW is:

At point `a`, solve:
`min_b max_k (a ⊛ b)_k` s.t. `b ∈ Δ`

This is a linear program:
`min_t` s.t. `(a ⊛ b)_k ≤ t` for all k, `sum(b) = 1`, `b ≥ 0`

This gives the FW direction. The previous code's formulation (`max sum(b)` s.t. `(a ⊛ b)_k ≤ M, b ≥ 0`) is equivalent up to normalization.

def fast_autocorr(a):
    """Compute autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_simplex(a):
    """Project/normalize onto the simplex (sum=1, non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_subgradient(a, ac, temperature):
    """Compute subgradient using softmax weighting of all lags."""
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = len(top_idx)
    weights /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad

def topk_subgradient(a, ac, top_k=80, beta=4.0):
    """Compute subgradient from top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def coordinate_search(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with random target selection."""
    n = len(a0)
    a = norm_simplex(a0).copy()

    max_stall = 10
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(100, n // 2)
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    break
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = a / a.sum()
        
        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_simplex(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_subgradient(a, ac, temperature):
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = len(top_idx)
    weights /= w_sum
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad

def topk_subgradient(a, ac, top_k=100, beta=3.5):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

        beta_t = (t - 1.0) / (t + 1.5)

def subgradient_optimize(a0, time_limit, verbose=True, seed=99):
    """Projected subgradient method with Polyak-like step size."""
    n = len(a0)
    a = norm_simplex(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    running_sum = np.zeros(n)
    avg_count = 0
    # Lower bound estimate: use 0.7 * current as a conservative bound
    lower_bound = best_m * 0.7
    step_scale = 1.0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(x)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))
        # Subgradient of the active lag
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_star - idx
        g = np.zeros(n)
        g[idx] = 2.0 * x[j_idx]
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        # Step size: Polyak-like with decay
        gap = max(M_curr - lower_bound, 1e-10)
        eta = step_scale * gap / (g_norm * g_norm * np.sqrt(t + 1))
        eta = min(eta, 0.1)
        x_new = x - eta * g
        x_new = project_simplex(x_new)
        x = x_new
        running_sum += x
        avg_count += 1
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            lower_bound = max(lower_bound, best_m * 0.95)
        # Also check running average
        if avg_count % 200 == 0:
            avg = running_sum / avg_count
            avg = project_simplex(avg)
            M_avg = fast_autocorr(avg).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg.copy()
                lower_bound = max(lower_bound, best_m * 0.95)
        # Decay step scale
        if t % 5000 == 0:
            step_scale *= 0.8
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SG t={t} ({elapsed:.1f}s): J={J:.6f}, M={M_new:.8f}")
    return best_a, best_m
