```
def fast_conv(a):
    a=np.asarray(a,float)

    size=1 << (2*n-2).bit_length()
```

```
def fast_conv(a):
    a=np.asarray(a,dtype=np.float64)
    n=a.size
    if n==0: return np.array([])
    size=2*n-1
    fft_len=1
    while fft_len < size: fft_len <<=1
    A=np.fft.rfft(a, fft_len)
    out=np.fft.irfft(A*A, fft_len)[:size]
    out[out < 0.0] = 0.0
    return out
```

```
def interpolate_sequence(f,new_n):
    f=np.asarray(f,float)
    old_n=f.size
    if old_n==0: return np.ones(new_n)/new_n
    s=f.sum()
    if s<1e-300: return np.ones(new_n)/new_n
    if old_n==new_n: return f.copy()
    x_old=np.linspace(0,1,old_n)
    x_new=np.linspace(0.1,new_n)
    out=np.interp(x_new,x_old,f/s)
    out=np.maximum(out,0.0)
    ss=out.sum()
    if ss<1e-300: out=np.ones(new_n)/new_n
    else: out=out/ss
    return out
```

```python
"""
Search for non-negative step-function heights minimizing the evaluation function.

The implementation combines:
  - analytic / deconvolution-based global initializers,
  - smooth log-sum-exp mirror descent,
  - peak-cycling accelerated mirror descent,
  - trust-region QP refinement via CVXPY,
  - very fast exact local mass-transfer coordinate descent,
  - fixed-point deconvolution polishing,
  - multi-length interpolation.

All helper functions are top-level and there are no closures or lambdas.
"""

import math
import time
import numpy as np

try:
    import cvxpy as cp
    HAS_CVXPY = True
except Exception:
    HAS_CVXPY = False


_SELECTED_SOLVER = None


def get_solver():
    global _SELECTED_SOLVER
    if not HAS_CVXPY:
        return None
    if _SELECTED_SOLVER is not None:
        return _SELECTED_SOLVER

    selected = None
    try:
        for name in ("CLARABEL", "ECOS", "SCS", "PDLP"):
            try:
                solver_name = getattr(cp, name)
                x = cp.Variable(4, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
                prob.solve(solver=solver_name, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    selected = solver_name
                    break
            except Exception:
                pass
    except Exception:
        pass

    if selected is None:
        selected = cp.ECOS

    _SELECTED_SOLVER = selected
    print("  CVXPY solver selected:", selected)
    return selected


def elapsed_since(t0):
    return time.monotonic() - t0


def remaining_budget(t0, budget):
    rem = budget - (time.monotonic() - t0)
    return rem if rem > 0.0 else 0.0


def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_len = next_pow2(size)
    A = np.fft.rfft(a, fft_len)
    out = np.fft.irfft(A * A, fft_len)[:size]
    out[out < 0.0] = 0.0
    return out


def fast_cross(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    L = x.size + y.size - 1
    if L <= 0:
        return np.zeros(0, dtype=np.float64)
    fft_len = next_pow2(L)
    out = np.fft.irfft(np.fft.rfft(x, fft_len) * np.fft.rfft(y, fft_len), fft_len)[:L]
    out[out < 0.0] = 0.0
    return out


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s


def surr_score(a):
    """Fast surrogate for the official objective, assuming scale invariance."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01 or not np.isfinite(s):
        return float("inf")
    an = a / s
    c = fast_conv(an)
    return float(2.0 * a.size * c.max())


def evaluate_safe(a):
    try:
        lst = [float(x) for x in a]
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    if new_n <= 0:
        return np.ones(1, dtype=np.float64)
    if old_n == 0:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    if old_n == new_n:
        return f.copy()

    s = f.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)

    if old_n == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)

    xf = f / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(x_new, x_old, xf)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-300:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    return out / ss


def apply_move_ac(ac, a, i, j, amt):
    """
    Exact O(n) update of the full linear convolution after moving mass amt
    from coordinate i to coordinate j.
    """
    n = a.size
    out = np.array(ac, copy=True)

    # coordinate i loses amt
    out[i:i + n] -= 2.0 * amt * a
    out[2 * i] += amt * amt

    # coordinate j gains amt
    out[j:j + n] += 2.0 * amt * a
    out[2 * j] += amt * amt

    k = i + j
    if k < out.size:
        out[k] -= 2.0 * amt * amt

    out[out < 0.0] = 0.0
    return out


def hard_subgradient(a, k):
    n = a.size
    k = int(k)
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for i in range(lo, hi):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * a[j]
    return g


def lse_gradient(a, ac, beta):
    n = a.size
    if beta <= 0.0:
        k = int(np.argmax(ac))
        return hard_subgradient(a, k)

    amax = float(ac.max())
    scaled = (ac - amax) / beta
    e = np.exp(np.clip(scaled, -50.0, 0.0))
    esum = float(e.sum())
    if esum <= 1e-300:
        w = np.zeros_like(ac)
        w[int(np.argmax(ac))] = 1.0
    else:
        w = e / esum

    wr = w[::-1]
    conv = fast_cross(a, wr)
    g = 2.0 * conv[:2 * n - 1][::-1]
    return g


def mirror_trial(y, g, eta, n):
    ly = np.log(np.maximum(y, 1e-30)) - eta * g
    mx = ly.max()
    a = np.exp(ly - mx)
    a = np.maximum(a, 1e-30)
    s = a.sum()
    if s < 1e-15 or not np.isfinite(s):
        return None, float("inf")
    a = a / s
    return a, float(2.0 * n * fast_conv(a).max())


def lse_mirror_opt(a0, time_limit, seed=0, lr0=1.0, anneal=50.0, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    ac = fast_conv(a)
    best = a.copy()
    best_M = float(2.0 * n * ac.max())

    rng = np.random.default_rng(seed)
    x = a.copy()
    xp = a.copy()
    xs = best_M

    t = 0
    stall = 0
    max_stall = 1500
    t0 = time.monotonic()

    ac_range = float(ac.max() - ac.min())
    beta_start = max(0.2 * ac_range, 1e-4 * float(ac.max()) + 1e-14)
    beta_end = max(beta_start / max(anneal, 1.0), 1e-7 * float(ac.max()) + 1e-16)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-9))
        beta = beta_end + (beta_start - beta_end) * 0.5 * (1.0 + math.cos(math.pi * frac))

        t += 1
        mom = (t - 1.0) / (t + 1.5)
        y = x + mom * (x - xp)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_conv(y)
        yM = float(2.0 * n * ac_y.max())
        g = lse_gradient(y, ac_y, beta)
        gnorm = float(np.linalg.norm(g, np.inf))

        if gnorm < 1e-300:
            x_new = y.copy()
            xs_new = yM
        else:
            g = g / gnorm
            eta = lr0 / (math.sqrt(t) + 1.0)
            accepted = False
            x_new = None
            xs_new = float("inf")

            for te in (eta, eta * 0.4, eta * 0.16, eta * 0.064):
                aa, mm = mirror_trial(y, g, te, n)
                if aa is None:
                    continue
                if mm < yM - 1e-14 or mm < xs - 1e-14:
                    x_new = aa
                    xs_new = mm
                    accepted = True
                    break

            if not accepted:
                if stall > max_stall:
                    x_new = best.copy()
                    noise = rng.normal(0.0, 0.012, n) * x_new
                    x_new = np.maximum(x_new + noise, 1e-12)
                    x_new = x_new / x_new.sum()
                    xs_new = float(2.0 * n * fast_conv(x_new).max())
                    stall = 0
                else:
                    x_new = x.copy()
                    xs_new = xs

        xp = x
        x = x_new
        xs = xs_new

        if xs < best_M - 1e-14:
            best_M = xs
            best = x.copy()
            stall = 0
        else:
            stall += 1

        if verbose and t % 700 == 0:
            print(f"    LSE mirror t={t} beta={beta:.3e} obj={best_M:.10f} ({elapsed:.1f}s)")

    return best

def coordinate_fast_opt(a0, time_limit, seed=0, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    ac = fast_conv(a)
    M = float(2.0 * n * ac.max())
    best = a.copy()
    best_M = M

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    stall = 0
    round_no = 0

    fracs = (0.02, 0.01, 0.005, 0.002)

    def rand_idx(sz):
        sz = max(0, min(int(sz), n))
        if sz == 0:
            return np.zeros(0, dtype=int)
        return rng.integers(0, n, size=sz, replace=False)

    while time.monotonic() - t0 < time_limit:
        round_no += 1

        if stall > 800:
            a = best.copy()
            sc = float(rng.uniform(0.002, 0.012))
            noise = rng.normal(0.0, sc, n) * a
            a = np.maximum(a + noise, 1e-12)
            a = a / a.sum()
            ac = fast_conv(a)
            M = float(2.0 * n * ac.max())
            stall = 0
            if M < best_M:
                best_M = M
                best = a.copy()

        top_peaks = np.argsort(ac)[::-1][:6]
        src = np.zeros(n, dtype=bool)

        for kobj in top_peaks:
            k = int(kobj)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            idx = np.arange(lo, hi, dtype=int)
            if idx.size == 0:
                continue
            contrib = a[idx] * a[k - idx]
            contrib = np.maximum(contrib, 0.0)
            top = idx[np.argsort(contrib)[::-1][:4]]
            src[top] = True

        src[np.argsort(a)[::-1][:40]] = True
        rr = rand_idx(40)
        if rr.size:
            src[rr] = True

        src_idx = np.where(src)[0]
        if src_idx.size > 120:
            perm = rng.permutation(src_idx.size)[:120]
            src_idx = src_idx[perm]

        sinks = np.concatenate((np.argsort(a)[:30], rand_idx(30)))
        if sinks.size == 0:
            sinks = np.arange(n)
        sinks = sinks.astype(int)

        improved = False

        for iobj in src_idx:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(iobj)
            ai = a[i]
            if ai <= 1e-14:
                continue

            best_j = -1
            best_frac = 0.0
            best_Mj = M + 1.0
            best_ac = None

            for frac in fracs:
                amt = ai * frac
                if amt <= 0.0:
                    continue
                for jobj in sinks:
                    j = int(jobj)
                    if j == i:
                        continue
                    ac_new = apply_move_ac(ac, a, i, j, amt)
                    Mj = float(2.0 * n * ac_new.max())
                    if Mj < best_Mj - 1e-12:
                        best_Mj = Mj
                        best_j = j
                        best_frac = frac
                        best_ac = ac_new

            if best_j >= 0:
                amt = ai * best_frac
                a[i] -= amt
                a[best_j] += amt
                ac = best_ac
                M = best_Mj
                improved = True
                if M < best_M - 1e-14:
                    best_M = M
                    best = a.copy()
                    stall = 0

        if not improved:
            stall += 1

        if improved and improved % 200 == 0:
            pass

        if round_no % 200 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-6 or not np.isfinite(s):
                a = a / s
                ac = fast_conv(a)
                M = float(2.0 * n * ac.max())
                if M < best_M:
                    best_M = M
                    best = a.copy()

        if verbose and round_no % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      coord round={round_no} obj={best_M:.10f} stall={stall} ({elapsed:.1f}s)")

    return best


def formal_sqrt_poly(c, n):
    """
    Causal formal square root of a polynomial C(x) = sum c[k] x^k.
    Returns normalized nonnegative coefficients p[0:n] satisfying
    (p*p)[k] ~= c[k] for early k, with clipping at zero.
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    need = 2 * n - 1
    if c.size < need:
        cc = np.zeros(need, dtype=np.float64)
        cc[:c.size] = c
        c = cc
    else:
        c = c[:need].copy()

    c = np.maximum(c, 0.0)
    c0 = float(c[0])
    if c0 <= 1e-300 or not np.isfinite(c0):
        return np.full(n, 1.0 / n, dtype=np.float64)

    p = np.zeros(n, dtype=np.float64)
    p[0] = math.sqrt(c0)

    for m in range(1, n):
        s = 0.0
        if m > 1:
            s = float(np.dot(p[1:m], p[m - 1:0:-1]))
        val = (float(c[m]) - s) / (2.0 * p[0])
        if val < 0.0:
            val = 0.0
        p[m] = val

    ss = p.sum()
    if ss <= 1e-300 or not np.isfinite(ss):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / ss


def analytic_candidates(n):
    out = []
    n = max(int(n), 2)

    x = np.arange(n, dtype=np.float64)

    def add(arr):
        try:
            a = norm_sum1(arr)
            if np.all(np.isfinite(a)):
                out.append(a)
        except Exception:
            pass

    add(np.ones(n))

    # Inverse-square-root recurrence: coefficients of (1-x)^(-1/2)
    p = np.ones(n, dtype=np.float64)
    for i in range(1, n):
        p[i] = p[i - 1] * ((2.0 * i - 1.0) / (2.0 * i))
    add(p)
    add(p + p[::-1])
    add(0.7 * p + 0.3 * p[::-1])

    for alpha in (0.35, 0.5, 0.75, 1.0, 1.5):
        a = (x + 1.0) ** (-alpha)
        add(a)
        add(a + a[::-1])

    for gamma in (0.3, 0.6, 1.0, 2.0):
        a = np.exp(-gamma * x / max(n - 1, 1))
        add(a)
        add(a + a[::-1])

    return out


def deconvolve_candidates(a, factors=(0.75, 0.9)):
    a = norm_sum1(a)
    n = a.size
    ac = fast_conv(a)
    M = float(ac.max())
    out = []
    if M <= 0.0 or not np.isfinite(M):
        return [a]

    base = max(1.0, M)
    for f in factors:
        ct = np.minimum(ac, f * M)
        ct = np.maximum(ct, 1e-12 * base)
        if ct[0] < 1e-4 * base:
            ct[0] = 1e-4 * base
        p = formal_sqrt_poly(ct, n)
        out.append(norm_sum1(p))
        q = 0.5 * (p + p[::-1])
        out.append(norm_sum1(q))

    flat = formal_sqrt_poly(np.ones(2 * n - 1), n)
    out.append(norm_sum1(flat))

    return out


def trust_region_opt(a0, time_limit, solver, max_active=None, r0=0.05, verbose=True):
    if solver is None:
        return norm_sum1(a0)

    a = norm_sum1(a0)
    n = a.size
    if n < 50:
        return a

    if max_active is None:
        max_active = 10 if n > 3200 else 15

    ac = fast_conv(a)
    M_raw = float(ac.max())
    M_obj = float(2.0 * n * M_raw)
    best = a.copy()
    best_M = M_obj

    rng = np.random.default_rng(12345)
    r = r0
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 30:
            a = best.copy()
            sc = float(rng.uniform(0.002, 0.01))
            noise = rng.normal(0.0, sc, n) * a
            a = np.maximum(a + noise, 1e-12)
            a = a / a.sum()
            ac = fast_conv(a)
            M_raw = float(ac.max())
            M_obj = float(2.0 * n * M_raw)
            r = 0.05
            stall = 0

        it += 1
        order = np.argsort(ac)[::-1]
        active = order[:max_active]
        if active.size == 0:
            break

        G = np.zeros((active.size, n), dtype=np.float64)
        for idx, kobj in enumerate(active):
            G[idx] = hard_subgradient(a, int(kobj))
        ac_active = ac[active]

        d = cp.Variable(n)
        tvar = cp.Variable()
        cons = [
            cp.sum(d) == 0,
            cp.norm(d) <= r,
            d >= -0.25 * a
        ]
        cons.append(G @ d + ac_active <= tvar)
        prob = cp.Problem(cp.Minimize(tvar + 1e-3 * cp.sum_squares(d)), cons)

        st0 = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue

        dval = np.asarray(d.value, dtype=np.float64).ravel()
        if dval.size != n or not np.all(np.isfinite(dval)):
            continue

        best_step = 0.0
        best_cand = a.copy()
        best_raw = M_raw

        for step in (0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5):
            cand = np.maximum(a + step * dval, 0.0)
            ss = cand.sum()
            if ss <= 1e-15:
                continue
            cand = cand / ss
            val = float(fast_conv(cand).max())
            if val < best_raw - 1e-12:
                best_raw = val
                best_step = step
                best_cand = cand

        if best_step > 0.0:
            a = best_cand
            M_raw = best_raw
            M_obj = float(2.0 * n * M_raw)
            if M_obj < best_M - 1e-14:
                best_M = M_obj
                best = a.copy()
                stall = 0
            if best_step >= 0.9:
                r = min(r * 1.25, 0.20)
            else:
                r = max(r * 0.75, 0.01)
        else:
            stall += 1
            r = max(r * 0.6, 0.01)

        if verbose and it % 5 == 0:
            print(f"      trust it={it} r={r:.4f} obj={best_M:.10f} ({time.monotonic()-st0:.1f}s/solve)")

    return best


def deconvolve_iteration_opt(a0, time_limit, seed=0, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    ac = fast_conv(a)
    M = float(2.0 * n * ac.max())
    best = a.copy()
    best_M = M

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    shave = 0.98
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        ac = fast_conv(a)
        M_raw = float(ac.max())
        M = float(2.0 * n * M_raw)
        base = max(1.0, M_raw)

        target = np.minimum(ac, shave * M_raw)
        target = np.maximum(target, 1e-10 * base)
        if target[0] < 1e-4 * base:
            target[0] = 1e-4 * base

        p = formal_sqrt_poly(target, n)
        p = norm_sum1(p)

        improved = False
        for mix in (1.0, 0.8, 0.6, 0.4, 0.2):
            cand = mix * p + (1.0 - mix) * a
            cand = norm_sum1(cand)
            val = surr_score(cand)
            if np.isfinite(val) and val < M - 1e-12:
                a = cand
                M = val
                improved = True
                break

        if improved:
            if M < best_M - 1e-14:
                best_M = M
                best = a.copy()
            shave = max(shave * 0.985, 0.55)
        else:
            shave = max(shave - 0.04, 0.50)
            if it % 20 == 0:
                a = best.copy()
                sc = float(rng.uniform(0.001, 0.008))
                noise = rng.normal(0.0, sc, n) * a
                a = np.maximum(a + noise, 1e-12)
                a = a / a.sum()

        if verbose and it % 40 == 0:
            print(f"      deconv it={it} shave={shave:.3f} obj={best_M:.10f} ({time.monotonic()-t0:.1f}s)")

    return best

def update_from_phase(best_seq, best_surr, trial, label, t0):
    if trial is None:
        return best_seq, best_surr
    trial = norm_sum1(trial)
    sc = surr_score(trial)
    if np.isfinite(sc) and sc < best_surr - 1e-12:
        best_surr = sc
        best_seq = trial.copy()
        print(f"  phase {label}: surrogate {sc:.10f} at t={elapsed_since(t0):.1f}s")
    return best_seq, best_surr


def actual_update(best_seq, best_score, trial, label, t0):
    if trial is None:
        return best_seq, best_score
    trial = norm_sum1(trial)
    sc = evaluate_safe(trial)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        best_score = sc
        best_seq = trial.copy()
        print(f"  actual {label}: score {sc:.10f} at t={elapsed_since(t0):.1f}s")
    return best_seq, best_score


def deconvolve_round(base, time_limit, seed=0):
    base = norm_sum1(base)
    best = base.copy()
    best_s = surr_score(base)
    t0 = time.monotonic()

    cands = deconvolve_candidates(base)
    scored = []
    for idx, c in enumerate(cands):
        c = norm_sum1(c)
        sc = surr_score(c)
        if np.isfinite(sc):
            scored.append((sc, c))

    scored.sort(key=tuple_score)

    for idx, item in enumerate(scored[:8]):
        rem = remaining_budget(t0, time_limit)
        if rem < 20.0:
            break
        c = item[1]
        t1 = min(rem * 0.5, 40.0)
        a = coordinate_fast_opt(c, t1, seed=seed + idx, verbose=False)
        s = surr_score(a)
        if np.isfinite(s) and s < best_s - 1e-12:
            best_s = s
            best = a.copy()
            print(f"      deconv candidate {idx}: obj={best_s:.10f}")

    return best


def explore_lengths(base, time_limit, seed=0, verbose=True):
    base = norm_sum1(base)
    best = base.copy()
    best_s = surr_score(base)
    n_base = base.size
    t0 = time.monotonic()

    try_list = []
    ratios = (0.75, 0.9, 1.05, 1.2, 1.5)
    for r in ratios:
        nt = int(round(n_base * r))
        if 800 <= nt <= 5000 and nt not in try_list:
            try_list.append(nt)
    for nt in (3000, 4096):
        if nt not in try_list:
            try_list.append(nt)

    for nt in try_list:
        rem = remaining_budget(t0, time_limit)
        if rem < 80.0:
            break

        a = interpolate_sequence(base, nt)
        t1 = min(rem * 0.55, 120.0)
        a = coordinate_fast_opt(a, t1, seed=seed + nt, verbose=False)

        rem = remaining_budget(t0, time_limit)
        if rem > 25.0:
            t2 = min(rem * 0.35, 60.0)
            a = lse_mirror_opt(a, t2, seed=seed * 10 + nt, verbose=False)

        s = surr_score(a)
        if np.isfinite(s) and s < best_s - 1e-12:
            best_s = s
            best = a.copy()
            if verbose:
                print(f"      length {nt}: obj={best_s:.10f}")

    return best


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    budget = min(float(budget_s) - 120.0, 1700.0)
    if budget < 60.0:
        budget = max(30.0, float(budget_s) - 20.0)

    print(f"Total effective budget: {budget:.1f}s")
    solver = get_solver()

    best_seq = None
    best_surr = float("inf")
    best_score = float("inf")
    parent_seq = None

    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        raw = np.clip(raw, 0.0, 1000.0)
        if raw.size > 0 and np.all(np.isfinite(raw)) and raw.sum() > 1e-10:
            parent_seq = norm_sum1(raw)
            psc = evaluate_safe(parent_seq)
            psur = surr_score(parent_seq)
            if np.isfinite(psc):
                best_score = psc
            if np.isfinite(psur):
                best_surr = psur
            best_seq = parent_seq.copy()
            print(f"Parent loaded: n={parent_seq.size}, surrogate={psur:.10f}, actual={psc:.10f}")
    except Exception as exc:
        print("Parent unavailable or invalid:", exc)

    if best_seq is None:
        n0 = 2400
        best_seq = np.ones(n0, dtype=np.float64) / n0
        best_surr = surr_score(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Using default uniform: n={n0}, surrogate={best_surr:.10f}, actual={best_score:.10f}")

    if best_seq.size > 5000:
        best_seq = interpolate_sequence(best_seq, 4096)
        best_surr = surr_score(best_seq)
        best_score = evaluate_safe(best_seq)
        parent_seq = best_seq.copy()
        print(f"Downsampled initial sequence to n={best_seq.size}")

    n0 = best_seq.size

    # Build a modest pool of global candidates.
    pool = []
    pool.append((surr_score(best_seq), best_seq, "parent"))

    sym = norm_sum1(0.5 * (best_seq + best_seq[::-1]))
    pool.append((surr_score(sym), sym, "sym"))

    for idx, c in enumerate(analytic_candidates(n0)):
        pool.append((surr_score(c), c, f"analytic_{idx}"))

    for idx, c in enumerate(deconvolve_candidates(best_seq)):
        pool.append((surr_score(c), c, f"deconv_{idx}"))

    valid_pool = []
    for sc, c, lb in pool:
        if np.isfinite(sc):
            valid_pool.append((sc, c, lb))
    valid_pool.sort(key=tuple_score)

    a_main = best_seq
    if valid_pool and valid_pool[0][0] < best_surr * 0.999:
        a_main = valid_pool[0][1]
        print(f"Initial main candidate from pool: {valid_pool[0][2]}, obj={valid_pool[0][0]:.10f}")

    # Quick local polish before heavier methods.
    rem = remaining_budget(t0, budget)
    tp = min(60.0, rem * 0.04)
    if tp > 10.0:
        out = coordinate_fast_opt(a_main, tp, seed=101, verbose=False)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "quick_coord", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "quick_actual", t0)
        a_main = best_seq

    # Smooth nonconvex optimization.
    rem = remaining_budget(t0, budget)
    tp = min(300.0, rem * 0.18)
    if tp > 20.0:
        out = lse_mirror_opt(a_main, tp, seed=202, lr0=1.1, anneal=50.0, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "lse", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "lse_actual", t0)
        a_main = best_seq

    # Peak cycling.
    rem = remaining_budget(t0, budget)
    tp = min(200.0, rem * 0.13)
    if tp > 20.0:
        out = peak_amd_opt(a_main, tp, seed=303, lr0=1.2, top_count=4, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "peak_amd", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "peak_actual", t0)
        a_main = best_seq

    # Fixed-point deconvolution.
    rem = remaining_budget(t0, budget)
    tp = min(180.0, rem * 0.12)
    if tp > 15.0:
        out = deconvolve_iteration_opt(a_main, tp, seed=404, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "deconv_iter", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "deconv_actual", t0)
        a_main = best_seq

    # QP trust-region refinement if CVXPY is available.
    rem = remaining_budget(t0, budget)
    tp = min(380.0, rem * 0.22)
    if tp > 80.0 and solver is not None and a_main.size <= 5000:
        out = trust_region_opt(a_main, tp, solver, r0=0.04, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "trust", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "trust_actual", t0)
        a_main = best_seq

    # Long local mass-transfer polish.
    rem = remaining_budget(t0, budget)
    tp = min(450.0, rem * 0.25)
    if tp > 30.0:
        out = coordinate_fast_opt(a_main, tp, seed=505, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "coord_long", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "coord_actual", t0)
        a_main = best_seq

    # Deconvolution-based candidate round from current best.
    rem = remaining_budget(t0, budget)
    tp = min(140.0, rem * 0.12)
    if tp > 25.0:
        out = deconvolve_round(a_main, tp, seed=606)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "deconv_round", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "deconv_round_actual", t0)
        a_main = best_seq

    # Try nearby / larger lengths.
    rem = remaining_budget(t0, budget)
    tp = min(220.0, rem * 0.14)
    if tp > 80.0 and a_main.size <= 5000:
        out = explore_lengths(a_main, tp, seed=707, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "multi_length", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "multi_actual", t0)
        a_main = best_seq

    # Final polish: one more coordinate pass and a short LSE pass.
    rem = remaining_budget(t0, budget)
    tp = min(180.0, rem * 0.5)
    if tp > 10.0:
        out = coordinate_fast_opt(a_main, tp, seed=808, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "final_coord", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "final_actual", t0)

    rem = remaining_budget(t0, budget)
    tp = min(120.0, rem * 0.7)
    if tp > 20.0:
        out = lse_mirror_opt(a_main, tp, seed=909, lr0=0.9, anneal=25.0, verbose=True)
        best_seq, best_surr = update_from_phase(best_seq, best_surr, out, "final_lse", t0)
        best_seq, best_score = actual_update(best_seq, best_score, best_seq, "final_lse_actual", t0)

    # Final validation and safety cleanup.
    best_seq = norm_sum1(best_seq)
    final_score = evaluate_safe(best_seq)
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    if parent_seq is not None:
        parent_seq = norm_sum1(parent_seq)
        pfinal = evaluate_safe(parent_seq)
        if np.isfinite(pfinal) and pfinal < best_score:
            best_seq = parent_seq.copy()
            best_score = pfinal

    final_list = []
    for x in best_seq:
        y = float(x)
        if not np.isfinite(y):
            y = 0.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        final_list.append(y)

    # If somehow all zero, fallback to a valid tiny sequence.
    if not final_list or sum(final_list) < 0.01:
        final_list = [0.02]

    final_score = evaluate_safe(final_list)
    print(f"FINAL returned: n={len(final_list)}, surrogate={surrr if False else surr_score(final_list):.10f}, actual={final_score:.10f}")
    return final_list
```
