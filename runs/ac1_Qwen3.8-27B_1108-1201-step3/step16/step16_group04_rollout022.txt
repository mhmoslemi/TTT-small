def lp_frank_wolfe(a0, time_limit, solver, full_lp_start=50, k_init=40, k_max=150):
    """
    LP Frank-Wolfe with Nesterov acceleration and adaptive constraint count.
    Phase 1: top-K LP for fast progress.
    Phase 2: full LP for refinement.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute structure for full LP
    rows_full, cols_full = build_conv_structure(n)

    x = a.copy()
    x_prev = a.copy()
    current_k = k_init
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Solve LP: use full LP after full_lp_start iterations
        if t >= full_lp_start and n <= 2600:
            direction = solve_lp_full(y, M_curr, rows_full, cols_full, n, solver)
        else:
            active_lags = np.argsort(ac)[::-1][:current_k]
            direction = solve_lp_topk(y, M_curr, active_lags, n, solver)

        if direction is None:
            # Fallback: use subgradient direction
            k_star = int(ac.argmax())
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            grad = np.zeros(n)
            for j in range(j_start, j_end + 1):
                grad[j] = y[k_star - j]
            direction = grad / max(grad.sum(), 1e-15)

        # Adaptive step size with backtracking
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        # Check if step helped; if not, try smaller step
        M_new = fast_autocorr(x_new).max()
        if M_new > M_curr + 1e-12:
            # Try half step
            alpha_half = alpha * 0.5
            x_new2 = (1.0 - alpha_half) * y + alpha_half * direction
            x_new2 = np.maximum(x_new2, 0.0)
            sx2 = x_new2.sum()
            if sx2 > 1e-15:
                x_new2 = x_new2 / sx2
                M_new2 = fast_autocorr(x_new2).max()
                if M_new2 < M_curr:
                    x_new = x_new2
                    M_new = M_new2
                else:
                    x_new = y.copy()
                    M_new = M_curr

        x_prev = y
        x = x_new

        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()

        # Adapt K
        if t % 20 == 0 and current_k < k_max:
            current_k = min(current_k + 5, k_max)

        if t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            mode = "full" if (t >= full_lp_start and n <= 2600) else f"top{current_k}"
            print(f"  LP-FW t={t} ({elapsed:.1f}s): J={J:.6f} [{mode}]")

    return best_a, best_m


def waterfilling_search(a0, time_limit, rng_seed=42):
    """
    Peak-targeted proportional water-filling:
    At each step, identify the dominant lag and redistribute mass
    from contributing pairs to reduce the peak.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    iteration = 0
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_star = int(ac.argmax())

        # Find all pairs contributing to the peak lag
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        pairs = [(i, k_star - i) for i in range(i_start, i_end + 1)]

        # Sort pairs by contribution (descending)
        pairs.sort(key=lambda p: a[p[0]] * a[p[1]], reverse=True)

        improved = False
        for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
            a_trial = a.copy()
            # Move mass from the top contributing pairs to random low-peak positions
            n_move = max(3, n // 50)
            for _ in range(n_move):
                if not pairs:
                    break
                # Pick a random top pair
                pi = int(rng.integers(0, min(len(pairs), 20)))
                i, j = pairs[pi]
                amt = a_trial[min(i, j)] * frac * 0.5
                if amt < 1e-18:
                    continue
                # Move from the larger element
                if a_trial[i] >= a_trial[j]:
                    a_trial[i] -= amt
                    src = i
                else:
                    a_trial[j] -= amt
                    src = j
                if a_trial[src] < -1e-15:
                    a_trial[src] += amt
                    continue
                # Destination: random position that doesn't contribute to top peaks
                top_lags = np.argsort(ac)[::-1][:20]
                in_peak = np.zeros(n, dtype=bool)
                for kl in top_lags:
                    s2 = max(0, int(kl) - n + 1)
                    e2 = min(n - 1, int(kl))
                    in_peak[s2:e2 + 1] = True
                candidates = np.where(~in_peak)[0]
                if len(candidates) < 10:
                    candidates = np.arange(n)
                dst = candidates[int(rng.integers(len(candidates)))]
                a_trial[dst] += amt

            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()

            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            n_pert = max(5, n // 100)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    amt = a[i] * 0.005
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a / s_t
            iteration += 1

        if iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  WF iter={iteration} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=42):
    """Fine-grained coordinate descent with multiple move sizes and peak targeting."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_ks = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_ks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            max_idx = np.argmax(contrib)
            src = idx[max_idx]
            dst_in_pair = ki - src

            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_ks[:10]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)

            # Score candidates by sensitivity
            best_dst = -1
            best_score = float('inf')
            sample = candidates[rng.permutation(len(candidates))[:min(100, len(candidates))]]
            for dst in sample:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_ks[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                old_ai = a[i]
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] = old_ai
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)

        if stall > 0 and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Coord stall={stall} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 0: Spectral equalization ===
    print(f"\n=== Phase 0: Spectral Equalization ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_spec = spectral_equalize(best_seq, n_iter=40, eps=1e-8, blend=0.3)
        try_update(a_spec, "spectral")

    # === Phase 1: LP-FW (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW n={len(best_seq)}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, full_lp_start=30, k_init=50, k_max=120)
        try_update(a_out, f"lpfw1_n{len(a_out)}")

    # === Phase 2: Water-filling ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Water-filling n={len(best_seq)}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = waterfilling_search(a_in, t2, rng_seed=42)
        try_update(a_out, f"wf_n{len(a_out)}")

    # === Phase 3: LP-FW refined ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW refined n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, full_lp_start=10, k_init=80, k_max=200)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        for n_alt in [n_best + 200, n_best - 200, n_best + 400]:
            n_alt = max(1200, min(n_alt, 3200))
            if n_alt == n_best or n_alt <= 0:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 5: Alt length n={n_alt} ===")
            a_alt = interpolate_sequence(best_seq, n_alt)
            a_alt = spectral_equalize(a_alt, n_iter=20, eps=1e-8, blend=0.4)

            t5 = min(remaining * 0.4, 200.0)
            if t5 > 40:
                a_in = normalize(a_alt)
                a_out, _ = lp_frank_wolfe(a_in, t5, solver, full_lp_start=20, k_init=50, k_max=100)
                try_update(a_out, f"alt_lpfw_n{n_alt}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60:
                t5b = min(remaining * 0.3, 100.0)
                if t5b > 20:
                    a_in = normalize(best_seq)
                    a_out, _ = coordinate_polish(a_in, t5b, rng_seed=999)
                    try_update(a_out, f"alt_coord_n{len(a_out)}")

    # === Phase 6: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60 and n_best <= 2800:
        print(f"\n=== Phase 6: Final LP n={n_best} ===")
        t6 = min(remaining * 0.6, 300.0)
        if t6 > 40:
            a_in = normalize(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, full_lp_start=5, k_init=100, k_max=250)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 7: Final coordinate ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final Coord n={n_best} ===")
        t7 = min(remaining * 0.8, 150.0)
        if t7 > 15:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, rng_seed=7777)
            try_update(a_out, f"coord_veryfinal_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
