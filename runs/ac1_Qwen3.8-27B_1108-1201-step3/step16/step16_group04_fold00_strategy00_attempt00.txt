def nesterov_soft_gradient(a_in, time_budget):
    """Nesterov-style soft gradient for posterior distributions."""
    n = len(a_in)
    time_used = 0
    best_seq = a_in.copy()

    # Use a lookahead strategy in the sieve
    for iteration in range(int(time_budget / 5)):
        # Decide if time is over and update accordingly
        if time_used > time_budget:
            break

        # Compute next polynomial step using derivative information
        grad = np.gradient(best_seq)
        grad_size = np.sum(grad * grad)
        grad_at = grad / grad_size if grad_size > 0 else grad

        # Apply telekinesis-like scaling to the gradient
        beta = 0.99 / (iteration+1.0)
        grad_scaled = beta * grad * np.clip(grad_at, 1e-5, None)

        # Combine and enforce non-negativity
        new_seq = best_seq * (1 + grad_scaled)
        new_seq = new_seq / new_seq.sum()

        # Re-evaluate the new sequence
        new_m = np.correlate(new_seq, new_seq, mode="full").max()
        score = 2.0 * n * new_m / (sum(new_seq)**2)

        if score < 1.5028:
            # Found a better distribution, commit and reset
            best_seq = new_seq

        else:
            time_used += 10

    return best_seq, score
