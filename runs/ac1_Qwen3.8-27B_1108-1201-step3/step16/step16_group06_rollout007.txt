            stall = 0
        else:
            mx = fast_autocorr(x).max()
            if mx < best_m - 1e-12:
                best_m = mx
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            y = best_a + pert
            y = normalize(y)
            y_prev = y.copy()
            x = y.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.1f}s): score={2*n*best_m:.6f}")

    return best_a, best_m


def subgradient_mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    gamma = 3.0
    stall = 0
    max_stall = 5000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()
        peak_mask = ac >= M - 1e-10
        peak_idx = np.where(peak_mask)[0]

        sg = np.zeros(n)
        for k in peak_idx:
            k_int = int(k)
            for j in range(n):
                ref = k_int - j
                if 0 <= ref < n:
                    sg[j] += 2.0 * a[ref]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * sg
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            a = best_a + pert
            a = normalize(a)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD t={t} ({elapsed:.1f}s): score={2*n*best_m:.6f}")

    return best_a, best_m


def peak_targeted_cd(a0, time_limit, verbose=True, rng_seed=42, top_peaks=5):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        peak_indices = np.argsort(ac)[::-1][:top_peaks]

        sensitivity = np.zeros(n)
        for pk in peak_indices:
            pk_int = int(pk)
            for pos in range(n):
                ref = pk_int - pos
                if 0 <= ref < n:
                    sensitivity[pos] += a[ref]

        for peak_k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(peak_k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            src_ranking = np.argsort(contrib)[::-1]

            improved = False
            for src_rank in src_ranking[:8]:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[src_rank])
                if a[src] < 1e-15:
                    continue
                candidates = np.argsort(sensitivity)[:max(30, n // 15)]
                for dst in candidates:
                    if dst == src:
                        continue
                    if sensitivity[dst] > sensitivity[src] * 0.5:
                        continue
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-18:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CD ({elapsed:.1f}s): score={2*n*best_m:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetrize(a):
    a = 0.5 * (a + a[::-1])
    return normalize(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.cos(np.pi * (x - 0.5)) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # Symmetrize initial
    sym = symmetrize(best_seq)
    sym_score = evaluate_safe(sym)
    if sym_score < best_score:
        best_seq = sym
        best_score = sym_score
        print(f"  Symmetrized: {best_score:.6f}")

    # === Phase 1: LP-FW main (45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 1: LP-FW n={n_cur} (main) ===")
        t1 = min(remaining * 0.45, 700.0)
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t1, solver, verbose=True)
        try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 2: CD polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: CD polish n={n_cur} ===")
        t2 = min(remaining * 0.15, 200.0)
        a_out, _ = peak_targeted_cd(normalize(best_seq), t2, verbose=True, rng_seed=42, top_peaks=8)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 3: LP-FW round 2 (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP-FW round 2 n={n_cur} ===")
        t3 = min(remaining * 0.20, 350.0)
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t3, solver, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 4: SMD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: SMD n={n_cur} ===")
        t4 = min(remaining * 0.10, 120.0)
        a_out, _ = subgradient_mirror_descent(normalize(best_seq), t4, verbose=True)
        try_update(a_out, f"smd_n{n_cur}")

    # === Phase 5: Multi-n exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        n_best = len(best_seq)
        test_ns = []
        for delta in [200, 400, -200, -400, 600, -600]:
            nt = n_best + delta
            if 800 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.3, 120.0)
            a_out, _ = lp_frank_wolfe(normalize(a_inter), t_per, solver, verbose=False)
            try_update(a_out, f"multin_n{nt}")
            # Symmetrize and check
            sym = symmetrize(a_out)
            try_update(sym, f"multin_sym_n{nt}")

    # === Phase 6: Final CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final CD n={n_best} ===")
        t6 = min(remaining * 0.5, 100.0)
        a_out, _ = peak_targeted_cd(normalize(best_seq), t6, verbose=True, rng_seed=777, top_peaks=10)
        try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(remaining * 0.7, 150.0)
        a_out, _ = lp_frank_wolfe(normalize(best_seq), t7, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final symmetrization check
    if best_seq is not None:
        sym = symmetrize(best_seq)
        sym_sc = evaluate_safe(sym)
        if sym_sc < best_score:
            best_score = sym_sc
            best_seq = sym
            print(f"  Final symmetrization improved: {best_score:.6f}")

        # One more CD pass on the symmetrized version
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = peak_targeted_cd(normalize(best_seq), min(remaining * 0.5, 30.0), verbose=False, rng_seed=999)
            try_update(a_out, "cd_very_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
