These are the SAME term! So a[i] appears in conv[k] exactly once (as a[i]*a[k-i]), unless i = k-i = k/2, in which case it appears as a[i]^2.

So:
- For k where 0 <= k-i < n and k ≠ 2i: conv_new[k] = conv[k] + delta * a[k-i]
- For k = 2i (if 0 <= i < n, always true): conv_new[2i] = conv[2i] + 2*delta*a[i] + delta^2
- For all other k: conv_new[k] = conv[k]

The range of affected k values: k must satisfy 0 <= k-i < n, i.e., i <= k <= i+n-1. Also k must be in [0, 2n-2]. So k in [max(0, i), min(2n-2, i+n-1)].

This is a range of at most n values. So updating the convolution after changing one coordinate is O(n), and computing the new max is O(n) (or O(1) if we maintain a heap).

For coordinate descent, each step is O(n), and we do n steps per full pass, so one pass is O(n^2). For n=2400, that's about 6M operations, which is very fast (maybe 0.1 seconds in Python with NumPy).

Actually, in Python, even with NumPy, O(n) per step means 2400 operations per step, and 2400 steps per pass, so about 6M NumPy operations per pass. Each NumPy operation on a scalar is slow, but if we vectorize the update (update all affected k at once), it's fast.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based autocorrelation
2. Incremental coordinate descent with O(n) per-step updates (no full FFT recompute)
3. Backtracking mirror descent with exact active-set subgradient
4. Multiple structured initializations (exponential, two-bump, parent)
5. Adaptive multi-restart with perturbation scheduling
Key innovation: incremental convolution update for coordinate descent enables
100x more local search iterations than the FFT-recompute approach.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def exact_subgradient(a, ac, top_k=20):
    """Compute subgradient from top-k active lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** 4.0
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_update(y, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def amd_backtrack(a0, time_limit, c0=3.0, verbose=True, rng_seed=42):
    """Accelerated mirror descent with backtracking-like adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    c_current = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_autocorr(y)
        g = exact_subgradient(y, ac_y, top_k=25)

        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n) * 0.5
        a_new = mirror_update(y, g, eta)

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            noise_scale = 0.003 + 0.008 * min(stall_count / max_stall, 3.0)
            perturbation = rng.normal(0, noise_scale, n) * best_a
            x_restarted = np.maximum(best_a + perturbation, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            c_current = c_current * 1.15

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def peak_sensitivity_vectorized(a, ac, top_k=30):
    """Compute peak sensitivity for each coordinate (vectorized)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]

    # Weight by how close to max
    w = (top_vals / max(top_vals.max(), 1e-30)) ** 2

    sens = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        sens[idx] += wi * a[j_idx]
    return sens


def coord_descent_incremental(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent with incremental convolution updates.
    Key: O(n) per step instead of O(n log n) FFT.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50
    full_pass = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        full_pass += 1
        # Recompute conv from scratch every few passes for numerical stability
        if full_pass % 5 == 0:
            conv = fast_autocorr(a)
            M_curr = conv.max()
            if M_curr < best_m - 1e-15:
                best_m = M_curr
                best_a = a.copy()
                best_conv = conv.copy()

        M_curr = conv.max()
        # Compute peak sensitivity
        top_k_sens = min(25, len(conv))
        top_lags = np.argsort(conv)[::-1][:top_k_sens]

        # Build sensitivity vector
        sens = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            w = conv[k_int] / max(conv[top_lags[0]], 1e-30)
            sens[idx] += w * a[j_idx]

        # Sort: hot (high sensitivity) and cold (low sensitivity)
        order = np.argsort(sens)
        hot = order[::-1][:min(60, n // 2)]
        cold = order[:min(60, n // 2)]

        improved = False
        rng.shuffle(hot)
        rng.shuffle(cold)

        for src in hot:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_m
            if a[src] < 1e-12:
                continue

            for dst in cold:
                if dst == src:
                    continue
                if a[dst] > a[src]:
                    continue

                # Try moving mass from src to dst
                for frac in [0.05, 0.15, 0.3, 0.5]:
                    delta = -a[src] * frac
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue

                    # Apply change to src
                    a[src] += delta
                    a[dst] += amt

                    # Update convolution incrementally
                    # Change a[src] by delta:
                    k_start_s = max(0, src)
                    k_end_s = min(2 * n - 1, src + n)
                    affected_s = np.arange(k_start_s, k_end_s)
                    delta_conv_s = delta * a[affected_s - src]
                    mask_2i_s = (affected_s == 2 * src)
                    if mask_2i_s.any():
                        # Need original a[src] before update
                        orig_src = a[src] - delta
                        delta_conv_s[mask_2i_s] = 2 * delta * orig_src + delta * delta
                    conv[k_start_s:k_end_s] += delta_conv_s

                    # Change a[dst] by amt:
                    k_start_d = max(0, dst)
                    k_end_d = min(2 * n - 1, dst + n)
                    affected_d = np.arange(k_start_d, k_end_d)
                    delta_conv_d = amt * a[affected_d - dst]
                    mask_2i_d = (affected_d == 2 * dst)
                    if mask_2i_d.any():
                        orig_dst = a[dst] - amt
                        delta_conv_d[mask_2i_d] = 2 * amt * orig_dst + amt * amt
                    conv[k_start_d:k_end_d] += delta_conv_d

                    # Check improvement
                    M_new = conv.max()
                    if M_new < best_m - 1e-15:
                        best_m = M_new
                        best_a = a.copy()
                        best_conv = conv.copy()
                        improved = True
                        stall = 0
                        break
                    else:
                        # Revert
                        a[src] -= delta
                        a[dst] -= amt
                        conv[k_start_s:k_end_s] -= delta_conv_s
                        conv[k_start_d:k_end_d] -= delta_conv_d

                if improved:
                    break

            if improved:
                # After an improvement, break to recompute sensitivity
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_perturb = max(5, n // 80)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s

                # Update conv incrementally
                # For i:
                k_start_i = max(0, i)
                k_end_i = min(2 * n - 1, i + n)
                affected_i = np.arange(k_start_i, k_end_i)
                delta_i = -amt
                dc_i = delta_i * a[affected_i - i]
                m2i = (affected_i == 2 * i)
                if m2i.any():
                    dc_i[m2i] = 2 * delta_i * (a[i] + amt) + delta_i * delta_i
                conv[k_start_i:k_end_i] += dc_i
                # For j:
                k_start_j = max(0, j)
                k_end_j = min(2 * n - 1, j + n)
                affected_j = np.arange(k_start_j, k_end_j)
                delta_j = amt
                dc_j = delta_j * a[affected_j - j]
                m2j = (affected_j == 2 * j)
                if m2j.any():
                    dc_j[m2j] = 2 * delta_j * (a[j] - amt) + delta_j * delta_j
                conv[k_start_j:k_end_j] += dc_j

                M_new = conv.max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    best_conv = conv.copy()
                    stall = 0
                    improved = True
                else:
                    # Revert
                    conv[k_start_i:k_end_i] -= dc_i
                    conv[k_start_j:k_end_j] -= dc_j
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}, pass={full_pass}")

    # Final check
    final_conv = fast_autocorr(best_a)
    final_m = final_conv.max()
    if final_m < best_m:
        best_m = final_m
    return best_a, best_m


def coord_descent_random(a0, time_limit, verbose=True, rng_seed=777):
    """Randomized coordinate descent with incremental updates (diverse search)."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_m = conv.max()
    t0 = time.monotonic()

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Random pair
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if a[i] < 1e-12:
            stall += 1
            continue

        frac = rng.uniform(0.01, 0.4)
        amt = a[i] * frac
        delta_i = -amt
        delta_j = amt

        a[i] += delta_i
        a[j] += delta_j

        # Update conv for i
        k_start_i = max(0, i)
        k_end_i = min(2 * n - 1, i + n)
        affected_i = np.arange(k_start_i, k_end_i)
        dc_i = delta_i * a[affected_i - i]
        m2i = (affected_i == 2 * i)
        if m2i.any():
            dc_i[m2i] = 2 * delta_i * (a[i] - delta_i) + delta_i * delta_i
        conv[k_start_i:k_end_i] += dc_i

        # Update conv for j
        k_start_j = max(0, j)
        k_end_j = min(2 * n - 1, j + n)
        affected_j = np.arange(k_start_j, k_end_j)

        m2j = (affected_j == 2 * j)
        if m2j.any():
            dc_j[m2j] = 2 * delta_j * (a[j] - delta_j) + delta_j * delta_j
        conv[k_start_j:k_end_j] += dc_j

        M_new = conv.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            # Revert
            conv[k_start_i:k_end_i] -= dc_i
            conv[k_start_j:k_end_j] -= dc_j
            a[i] -= delta_i
            a[j] -= delta_j
            stall += 1

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    RandCD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    final_conv = fast_autocorr(best_a)
    final_m = final_conv.max()
    if final_m < best_m:
        best_m = final_m
    return best_a, best_m


def make_exponential_init(n, alpha=0.5):
    """Create an exponential decay initialization (asymmetric)."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * x / n)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_two_bump_init(n, pos1=0.25, pos2=0.7, width=0.1, ratio=1.0):
    """Create a two-bump initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / width) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_linear_ramp_init(n, slope=2.0):
    """Linear ramp (maximally asymmetric)."""
    x = np.linspace(0, 1, n)
    a = (1.0 + slope * x)
    a = np.maximum(a, 0)
    return a / a.sum()

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = make_exponential_init(n_base, alpha=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Try multiple initializations with quick AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-init AMD (budget={remaining:.0f}s) ===")

    inits = {}
    inits['parent'] = best_seq.copy()
    inits['exponential'] = make_exponential_init(n0, alpha=0.8)
    inits['two_bump'] = make_two_bump_init(n0, pos1=0.2, pos2=0.75, width=0.12, ratio=0.6)
    inits['ramp'] = make_linear_ramp_init(n0, slope=1.5)

    # Quick AMD on each init
    t_per_init = min(remaining * 0.12, 120.0)
    for name, init_a in inits.items():
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t_alloc = min(t_per_init, remaining * 0.5)
        if t_alloc < 10:
            continue
        print(f"  AMD on '{name}' (n={len(init_a)}, t={t_alloc:.0f}s)")
        a_out, _ = amd_backtrack(norm_sum1(init_a), t_alloc, c0=3.0, verbose=False, rng_seed=hash(name) % 10000)
        _try_update(a_out, f"amd_{name}")

    # === Phase 2: Extended AMD on best so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Extended AMD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_out, _ = amd_backtrack(norm_sum1(best_seq), t2, c0=4.0, verbose=True, rng_seed=42)
        _try_update(a_out, "amd_extended")

    # === Phase 3: Coordinate descent with incremental updates (MAIN) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent incremental n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 20:
        a_out, _ = coord_descent_incremental(norm_sum1(best_seq), t3, verbose=True, rng_seed=1)
        _try_update(a_out, "coord_incr")

    # === Phase 4: Randomized coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Random coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 15:
        a_out, _ = coord_descent_random(norm_sum1(best_seq), t4, verbose=True, rng_seed=77)
        _try_update(a_out, "rand_coord")

    # === Phase 5: AMD polish from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: AMD polish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        a_out, _ = amd_backtrack(norm_sum1(best_seq), t5, c0=2.0, verbose=False, rng_seed=88)
        _try_update(a_out, "amd_polish")

    # === Phase 6: More coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 300.0)
    if t6 > 15:
        a_out, _ = coord_descent_incremental(norm_sum1(best_seq), t6, verbose=True, rng_seed=2)
        _try_update(a_out, "coord_r2")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 7: Multi-length (budget={remaining:.0f}s) ===")
        for n_try in [n_best + 400, n_best + 800, n_best - 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 500 or n_try > 5000:
                continue
            a_long = interpolate_sequence(best_seq, n_try)
            t_alloc = min(remaining * 0.25, 120.0)
            if t_alloc < 20:
                continue
            print(f"  Trying n={n_try}")
            # Quick AMD then coordinate descent
            a_tmp, _ = amd_backtrack(a_long, t_alloc * 0.4, c0=3.0, verbose=False, rng_seed=n_try % 1000)
            _try_update(a_tmp, f"amd_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = coord_descent_incremental(norm_sum1(a_tmp), min(remaining * 0.4, 80.0), verbose=False, rng_seed=n_try % 1000)
                _try_update(a_out, f"coord_n{n_try}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 10:
            a_out, _ = coord_descent_incremental(norm_sum1(best_seq), t8, verbose=True, rng_seed=3)
            _try_update(a_out, "final_coord")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9 = min(remaining * 0.5, 60.0)
            if t9 > 10:
                a_out, _ = coord_descent_random(norm_sum1(best_seq), t9, verbose=False, rng_seed=99)
                _try_update(a_out, "final_rand")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
