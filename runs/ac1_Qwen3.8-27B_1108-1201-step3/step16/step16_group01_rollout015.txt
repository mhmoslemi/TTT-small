"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectrally-informed initialization (flat DFT magnitude)
2. AMD with adaptive penalized subgradient (soft-max pressure)
3. Backtracking line search for step size

import numpy as np
import time

def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))

        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def penalized_subgradient(a, ac, top_k=50, temperature=0.1):
    """
    Compute a subgradient using a softmax-weighted combination of top-k lags.
    This provides richer gradient information than just the active set.
    """
    n = len(a)
    if ac.max() < 1e-30:
        return np.zeros(n)
    
    # Use top-k lags with softmax weighting
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    
    # Softmax weighting (temperature controls how "peaky" the weights are)
    exp_vals = np.exp(top_vals / (temperature * ac.max() + 1e-15))
    w = exp_vals / exp_vals.sum()
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    
    return g

def exact_subgradient(a, ac, threshold_frac=0.99):
    """Exact subgradient from active set (lags near max)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    return g

def combined_subgradient(a, ac, exact_frac=0.15, soft_top_k=50, soft_temp=0.05):
    """
    Combined subgradient: exact active set + soft pressure from nearby lags.
    This gives both the hard constraint (reduce peaks) and soft pressure (prevent 
    nearby lags from growing).
    """
    M = ac.max()
    if M < 1e-30:
        return np.zeros(len(a))
    
    g_exact = exact_subgradient(a, ac, threshold_frac=0.99)
    g_soft = penalized_subgradient(a, ac, top_k=soft_top_k, temperature=soft_temp)
    
    # Normalize both
    ne = np.linalg.norm(g_exact)
    ns = np.linalg.norm(g_soft)
    if ne > 1e-15:
        g_exact = g_exact / ne
    if ns > 1e-15:
        g_soft = g_soft / ns
    
    return (1 - exact_frac) * g_exact + exact_frac * g_soft

def mirror_update(y, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

def amd_with_backtracking(a0, time_limit, c0=3.0, threshold_frac=0.99,

        if time.monotonic() - t0 > time_limit:

def spectral_init(n, freq_decay=2.0, n_harmonics=20, rng_seed=0):
    """
    Create an initialization with controlled spectral properties.
    We want the DFT magnitude to be somewhat peaked (concentrated at low freq)
    for flat autocorrelation, but not too concentrated (which would make the
    time signal too spiky).
    """
    rng = np.random.default_rng(rng_seed)
    
    # Create a spectrum with controlled magnitude
    # We want |A[m]| to decay with frequency (smooth signal)
    # but with some random phase for diversity
    m = np.arange(n)
    # Power spectrum: decays as 1/(1+decay*|m|)^alpha
    alpha = freq_decay
    power = 1.0 / (1.0 + alpha * m) ** 2
    # Add some random variation
    power *= (1.0 + 0.3 * rng.random(n))
    
    # Convert to time domain with random phases
    phase = rng.uniform(0, 2 * np.pi, n)
    A = np.sqrt(power) * np.exp(1j * phase)
    A = np.concatenate([A[:1], A[n//2:][::-1], A[1:n//2]])  # Make real
    
    # Actually, let's just do a real FFT approach
    # Create rfft spectrum
    A_r = np.zeros(n // 2 + 1, dtype=complex)
    A_r[0] = np.sqrt(power[0])  # DC
    for i in range(1, n // 2 + 1):
        A_r[i] = np.sqrt(power[i]) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    
    a = np.fft.irfft(A_r, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def harmonic_init(n, base_freq=1, n_harmonics=5, rng_seed=0):
    """Initialize with a multi-harmonic pattern."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n, endpoint=False)
    a = np.zeros(n)
    for k in range(n_harmonics):
        freq = base_freq * (k + 1)
        amp = 1.0 / (k + 1) ** 1.5
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    a = a ** 2  # Squaring makes it more non-negative
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:

def exact_subgradient(a, ac, threshold_frac=0.99):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    return g


def soft_subgradient(a, ac, top_k=50, temperature=0.05):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    exp_vals = np.exp(top_vals / (temperature * M + 1e-15))
    w = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def combined_subgradient(a, ac, exact_frac=0.2, soft_top_k=40, soft_temp=0.03):
    M = ac.max()
    if M < 1e-30:

    g_soft = soft_subgradient(a, ac, top_k=soft_top_k, temperature=soft_temp)
    ne = np.linalg.norm(g_exact)
    ns = np.linalg.norm(g_soft)

def mirror_update(y, g, eta):

    return a_new / s


def spectral_shape_step(a, power_alpha=0.3, n_iters=3):

    a = np.asarray(a, dtype=np.float64).copy()

        while M_new > best_m * 1.02 and bt < 5:
            eta *= 0.5
            a_new = mirror_update(y, g, eta)
            M_new = fast_autocorr(a_new).max()
            bt += 1

        x_prev = x
        x = a_new

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Spectral shaping step (occasional)
        if do_spectral_shape and t > 0 and t % shape_interval == 0:
            x_shaped = spectral_shape_step(x, power_alpha=shape_power, n_iters=2)
            M_shaped = fast_autocorr(x_shaped).max()
            if M_shaped < best_m:
                best_m = M_shaped
                best_a = x_shaped.copy()
                x = x_shaped
                x_prev = x_shaped
                stall_count = 0
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    SpectralShape t={t} ({elapsed:.1f}s): J={2.0*n*best_m:.6f}")

        if stall_count > max_stall:
            stall_count = 0
            restart_count += 1
            strategy = restart_count % 5
            if strategy == 0:
                pert = rng.normal(0, 0.01, n) * best_a
                x_restarted = best_a + pert
            elif strategy == 1:
                x_restarted = best_a[::-1] + rng.normal(0, 0.005, n) * best_a
            elif strategy == 2:
                x_pos = np.arange(n) / n
                freq = rng.integers(2, 12)
                pert = 0.03 * np.sin(2 * np.pi * freq * x_pos)
                x_restarted = best_a * (1.0 + pert)
            elif strategy == 3:
                shift = rng.integers(1, max(2, n // 20))
                x_restarted = np.roll(best_a, int(shift)) + rng.normal(0, 0.003, n)
            else:
                x_restarted = spectral_shape_step(best_a, power_alpha=0.4, n_iters=5)
                x_restarted = x_restarted + rng.normal(0, 0.005, n) * x_restarted

            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            current_c = c0 * (1.0 + 0.15 * min(restart_count, 10))

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart {restart_count} t={t} ({elapsed:.1f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def peak_coord_descent(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 60
    step_scales = [0.001, 0.003, 0.008, 0.015, 0.03, 0.06]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_k = np.argsort(ac)[::-1][:min(20, len(ac))]

        # Build sensitivity map
        peak_sens = np.zeros(n)
        n_top = min(8, len(peak_k))
        for k2 in peak_k[:n_top]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n, k2i + 1)
            idx2 = np.arange(s2, e2)
            j2 = k2i - idx2
            peak_sens[idx2] += a[j2]

        sens_max = peak_sens.max()
        if sens_max < 1e-20:
            break

        hot_thresh = 0.4 * sens_max
        cold_thresh = 0.05 * sens_max
        hot_positions = np.where(peak_sens > hot_thresh)[0]
        cold_positions = np.where(peak_sens < cold_thresh)[0]

        if len(cold_positions) < 5:
            cold_positions = np.where(peak_sens < 0.2 * sens_max)[0]
        if len(hot_positions) == 0:
            hot_positions = np.where(peak_sens == sens_max)[0]
        if len(cold_positions) == 0:
            cold_positions = np.where(peak_sens == peak_sens.min())[0]

        # Sample hot and cold
        n_hot = min(len(hot_positions), 25)
        n_cold = min(len(cold_positions), 25)
        if n_hot < len(hot_positions):
            hot_sample = rng.choice(hot_positions, size=n_hot, replace=False)
        else:
            hot_sample = hot_positions
        if n_cold < len(cold_positions):
            cold_sample = rng.choice(cold_positions, size=n_cold, replace=False)
        else:
            cold_sample = cold_positions

        improved = False
        for src in hot_sample:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in cold_sample:
                if dst == src:
                    continue
                for scale in step_scales:
                    amt = a[src] * scale
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                continue

        if not improved:
            stall += 1
            if stall % 3 == 0:
                n_pert = max(5, n // 80)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0.0)
                                if a.sum() > 1e-15:
                                    a = a / a.sum()

        if verbose and stall % 15 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def spectral_init(n, decay=2.0, rng_seed=0):
    rng = np.random.default_rng(rng_seed)
    m = np.arange(n // 2 + 1)
    power = 1.0 / (1.0 + decay * m) ** 1.5
    power *= (1.0 + 0.2 * rng.random(len(power)))
    A_r = np.zeros(n // 2 + 1, dtype=complex)
    A_r[0] = np.sqrt(power[0])
    for i in range(1, len(power)):
        A_r[i] = np.sqrt(power[i]) * np.exp(1j * rng.uniform(0, 2 * np.pi))
    a = np.fft.irfft(A_r, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def harmonic_init(n, base_freq=2, n_harm=5, rng_seed=0):
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n, endpoint=False)
    a = np.zeros(n)
    for k in range(n_harm):
        freq = base_freq * (k + 1)
        amp = 1.0 / (k + 1) ** 1.2
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    a = a ** 2
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_trial_update(a, best_score_ref, best_seq_ref, label, t0_global):
    if a is None:
        return best_score_ref, best_seq_ref
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_score_ref, best_seq_ref
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score_ref:
        elapsed = time.monotonic() - t0_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return sc, a_norm.copy()
    return best_score_ref, best_seq_ref


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    n0 = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            best_seq = parent.copy()
            best_score = ps
            n0 = len(parent)
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n0)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    # Phase 1: Diverse initialization (45s)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Diverse init (budget {remaining:.0f}s) ===")
    t1 = min(remaining * 0.025, 45.0)
    if t1 > 10:
        inits = [
            ("spectral1", spectral_init(n0, decay=1.5, rng_seed=1)),
            ("spectral2", spectral_init(n0, decay=3.0, rng_seed=2)),
            ("spectral3", spectral_init(n0, decay=0.8, rng_seed=3)),
            ("harm1", harmonic_init(n0, base_freq=2, n_harm=5, rng_seed=4)),
            ("harm2", harmonic_init(n0, base_freq=3, n_harm=4, rng_seed=5)),
            ("gauss", np.exp(-((np.linspace(0, 1, n0) - 0.5) ** 2) / 0.08)),
            ("triangle", np.maximum(1.0 - 2.0 * np.abs(np.linspace(0, 1, n0) - 0.5), 0.0)),
        ]
        for name, init in inits:
            init = norm_sum1(init)
            best_score, best_seq = make_trial_update(init, best_score, best_seq, f"init_{name}", t_global)

    # Phase 2: Primary AMD with spectral shaping (500s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD+spectral n={n_best} (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_out, _ = amd_combined(norm_sum1(best_seq), t2, c0=3.0, exact_frac=0.2,
                                soft_top_k=40, soft_temp=0.03, verbose=True, rng_seed=1,
                                max_stall=3000, do_spectral_shape=True, shape_interval=3000,
                                shape_power=0.15)
        best_score, best_seq = make_trial_update(a_out, best_score, best_seq, f"amd_shape_n{n_best}", t_global)

    # Phase 3: AMD exact (250s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD exact n={n_best} (budget {remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = amd_combined(norm_sum1(best_seq), t3, c0=5.0, exact_frac=1.0,
                                soft_top_k=5, soft_temp=0.01, verbose=True, rng_seed=7,
                                max_stall=2000, do_spectral_shape=False)
        best_score, best_seq = make_trial_update(a_out, best_score, best_seq, f"amd_exact_n{n_best}", t_global)

    # Phase 4: Coordinate descent (200s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: CoordDesc n={n_best} (budget {remaining:.0f}s) ===")
    t4 = min(remaining * 0.13, 200.0)
    if t4 > 15:
        a_out, _ = peak_coord_descent(norm_sum1(best_seq), t4, verbose=True, rng_seed=15)
        best_score, best_seq = make_trial_update(a_out, best_score, best_seq, f"coord_n{n_best}", t_global)

    # Phase 5: AMD polish (200s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: AMD polish n={n_best} (budget {remaining:.0f}s) ===")
    t5 = min(remaining * 0.13, 200.0)
    if t5 > 20:
        a_out, _ = amd_combined(norm_sum1(best_seq), t5, c0=2.5, exact_frac=0.3,
                                soft_top_k=30, soft_temp=0.02, verbose=True, rng_seed=25,
                                max_stall=2000, do_spectral_shape=True, shape_interval=5000,
                                shape_power=0.1)
        best_score, best_seq = make_trial_update(a_out, best_score, best_seq, f"amd_polish_n{n_best}", t_global)

    # Phase 6: Multi-length exploration (200s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length (budget {remaining:.0f}s) ===")
    if remaining > 80:
        for delta in [300, -300, 500, -500, 200, -200]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            n_try = n_best + delta
            if n_try < 400 or n_try > 5000:
                continue
            a_new_len = interpolate_sequence(best_seq, n_try)
            t_ml = min(remaining * 0.10, 70.0)
            if t_ml > 15:
                a_out, _ = amd_combined(norm_sum1(a_new_len), t_ml, c0=3.0, exact_frac=0.2,
                                        soft_top_k=40, soft_temp=0.03, verbose=False,
                                        rng_seed=30 + n_try % 50, max_stall=1500,
                                        do_spectral_shape=True, shape_interval=2000, shape_power=0.15)
                best_score, best_seq = make_trial_update(a_out, best_score, best_seq,
                                                          f"amd_len{n_try}", t_global)

    # Phase 7: Final coordinate descent (remaining)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CoordDesc n={n_best} (budget {remaining:.0f}s) ===")
    if remaining > 30:
        t7 = min(remaining * 0.40, 120.0)
        if t7 > 10:
            a_out, _ = peak_coord_descent(norm_sum1(best_seq), t7, verbose=True, rng_seed=50)
            best_score, best_seq = make_trial_update(a_out, best_score, best_seq,
                                                      f"coord_final_n{n_best}", t_global)

    # Phase 8: Final AMD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final AMD n={n_best} (budget {remaining:.0f}s) ===")
        t8 = min(remaining * 0.50, 80.0)
        if t8 > 10:
            a_out, _ = amd_combined(norm_sum1(best_seq), t8, c0=1.5, exact_frac=0.4,
                                    soft_top_k=20, soft_temp=0.01, verbose=False,
                                    rng_seed=60, max_stall=1000, do_spectral_shape=False)
            best_score, best_seq = make_trial_update(a_out, best_score, best_seq,
                                                      f"amd_final_n{n_best}", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
