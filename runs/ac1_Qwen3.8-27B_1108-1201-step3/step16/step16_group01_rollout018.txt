Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated Mirror Descent (AMD) with tight active-set subgradient and backtracking line search
2. Exact coordinate descent using analytical gradient of each autocorrelation lag
3. Multi-length exploration via interpolation
4. Simulated annealing restarts for local minimum escape
5. Final Polish phase with very small steps
Key improvements: backtracking (not fixed rate), tighter active set (top-5),

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_max_ac(a, ac, top_k=5):
    """
    Subgradient of max_k ac[k] w.r.t. a.
    d(ac[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Get top_k lags
    top_idx = np.argsort(ac)[::-1][:top_k]
    
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        # Term 1: a[k-j] for valid j (0 <= k-j < n => k-n+1 <= j <= k)
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n, k_int + 1)
        if j_lo < j_hi:
            js = np.arange(j_lo, j_hi)
            g[js] += a[k_int - js]
        
        # Term 2: a[j-k] for valid j (0 <= j-k < n => k <= j < k+n)
        j_lo2 = max(0, k_int)
        j_hi2 = min(n, k_int + n)
        if j_lo2 < j_hi2:
            js = np.arange(j_lo2, j_hi2)
            g[js] += a[js - k_int]
    
    return g

def mirror_step(a, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def exact_coord_descent(a0, time_limit, verbose=True, rng_seed=777):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated Mirror Descent (AMD) with tight active-set subgradient and backtracking
2. Exact coordinate descent using analytical gradient of autocorrelation peaks
3. Multi-length exploration via interpolation with AMD+CD on each
4. Diverse initializations (chirp, gaussian, random) to avoid local minima
5. Adaptive restart strategy with annealing
Key: backtracking ensures progress, tight active set (top 3-5) for focus,
and multi-length sweep catches optimal n.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interp_to_length(f, new_n):
    """Interpolate sequence to new length preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def subgradient_max_ac(a, ac, top_k=5):
    """
    Subgradient of max_k ac[k] w.r.t. a.
    d(ac[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    Uses only top_k lags for focus.
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)

    top_idx = np.argsort(ac)[::-1][:top_k]

    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        # Term 1: a[k-j] for j where 0 <= k-j < n
        j_lo = max(0, k_int - n + 1)
        j_hi = min(n, k_int + 1)
        if j_lo < j_hi:
            js = np.arange(j_lo, j_hi)
            g[js] += a[k_int - js]
        # Term 2: a[j-k] for j where 0 <= j-k < n
        j_lo2 = max(0, k_int)
        j_hi2 = min(n, k_int + n)
        if j_lo2 < j_hi2:
            js = np.arange(j_lo2, j_hi2)
            g[js] += a[js - k_int]

    return g


def mirror_step(a, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def amd_backtrack(a0, time_limit, verbose=False, rng_seed=42, top_k=5, max_bt=8):
    """
    AMD with backtracking line search and Nesterov momentum.
    Returns (best_sequence, best_max_ac).
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2500
    eta_scale = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient
        ac_y = fast_autocorr(y)
        g = subgradient_max_ac(y, ac_y, top_k=top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            break
        g = g / gn

        # Backtracking line search (limited)
        M_curr = ac_y.max()
        eta = eta_scale / (1.0 + 0.1 * np.log1p(t))
        a_new = x.copy()
        M_new = M_curr
        for _ in range(max_bt):
            candidate = mirror_step(y, g, eta)
            M_cand = fast_autocorr(candidate).max()
            if M_cand < M_curr - 1e-15:
                a_new = candidate
                M_new = M_cand
                break
            eta *= 0.5
        else:
            # Use the last candidate even if no improvement (minimum step)
            a_new = candidate
            M_new = M_cand

        x_prev = x
        x = a_new

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            eta_scale = min(eta_scale * 1.05, 5.0)
        else:
            stall += 1
            if stall % 1000 == 0:
                eta_scale = max(eta_scale * 0.6, 0.2)

        # Restart if stuck
        if stall > max_stall:
            stall = 0
            noise_level = 0.003 + 0.007 * min(stall / max_stall, 3.0)
            x_new = best_a * (1.0 + rng.normal(0, noise_level, n))
            x_new = normalize(x_new)
            x = x_new
            x_prev = x_new
            eta_scale = 1.0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta_scale:.3f}, stall={stall}")

    return best_a, best_m


def exact_coord_descent(a0, time_limit, verbose=False, rng_seed=777):
    """
    Coordinate descent: move mass from high-peak-sensitivity to low-sensitivity positions.
    Uses exact autocorrelation evaluation for each candidate move.
    Returns (best_sequence, best_max_ac).
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 30
    n_high = min(40, n // 10)
    n_low = min(40, n // 10)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-20:
            break

        # Compute peak lags
        peak_lags = np.argsort(ac)[::-1][:min(15, len(ac))]

        # Compute sensitivity: how much each position contributes to peak lags
        sens = np.zeros(n)
        for k in peak_lags[:8]:
            k_int = int(k)
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n, k_int + 1)
            if j_lo < j_hi:
                js = np.arange(j_lo, j_hi)
                sens[js] += a[k_int - js]
            j_lo2 = max(0, k_int)
            j_hi2 = min(n, k_int + n)
            if j_lo2 < j_hi2:
                js = np.arange(j_lo2, j_hi2)
                sens[js] += a[js - k_int]

        improved = False

        # Get high and low sensitivity positions
        order = np.argsort(sens)
        high_idx = order[::-1][:n_high]
        low_idx = order[:n_low]

        for src in high_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            best_dst = -1
            best_frac = 0.0
            best_new_m = best_m + 1e-14
            best_trial = None

            # Sort low-sensitivity by similarity (prefer similar values)
            low_sorted = low_idx[np.argsort(np.abs(a[low_idx] - a[src]))[:n_low]]

            for dst in low_sorted:
                if dst == src:
                    continue
                if a[dst] > a[src] * 2.0:
                    continue

                for frac in [0.01, 0.03, 0.06, 0.12]:
                    delta = a[src] * frac
                    if delta < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_dst = dst
                        best_frac = frac
                        best_trial = a_trial

            if best_dst >= 0 and best_new_m < best_m - 1e-14:
                a = best_trial
                best_m = best_new_m
                best_a = a.copy()
                improved = True
                stall = 0
                # Update sensitivity incrementally (recompute next iteration)
                break

        if not improved:
            # Try a few random moves
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(rng.integers(n))
                dst = int(rng.integers(n))
                if src == dst or a[src] < 1e-8:
                    continue
                frac = float(rng.uniform(0.01, 0.08))
                delta = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            # Escape: multi-point perturbation
            n_pert = max(3, n // 200)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = float(rng.uniform(0.02, 0.06))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def chirp_init(n, freq=1.0):
    """Chirp-like initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x * x / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def mixed_init(n, rng):
    """Random weighted mixture of structured inits."""
    a1 = chirp_init(n, freq=rng.uniform(0.5, 2.0))
    a2 = gaussian_init(n, sigma_frac=rng.uniform(0.2, 0.4))
    a3 = np.ones(n) / n
    w1 = rng.dirichlet([1, 1, 1])
    a = w1[0] * a1 + w1[1] * a2 + w1[2] * a3
    return a / a.sum()


def run_amd_then_cd(a0, time_amd, time_cd, rng_seed, top_k=5):
    """Run AMD followed by coordinate descent. Returns best sequence."""
    a_amd, _ = amd_backtrack(a0, time_amd, verbose=False, rng_seed=rng_seed, top_k=top_k)
    a_cd, _ = exact_coord_descent(a_amd, time_cd, verbose=False, rng_seed=rng_seed + 1000)
    # AMD on the CD result for a final polish
    remaining_check = time_amd + time_cd
    a_final, _ = amd_backtrack(a_cd, min(30.0, time_cd * 0.3), verbose=False, rng_seed=rng_seed + 2000, top_k=top_k)
    # Pick the best
    sc_amd = fast_autocorr(normalize(a_amd)).max()
    sc_cd = fast_autocorr(normalize(a_cd)).max()
    sc_final = fast_autocorr(normalize(a_final)).max()
    n = len(a0)
    if sc_cd * n < sc_amd * n and sc_cd * n < sc_final * n:
        return a_cd
    elif sc_final * n < sc_amd * n:
        return a_final
    else:
        return a_amd


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase approach:
    1. Quick AMD on parent to establish baseline
    2. Extended AMD with different active set sizes
    3. Coordinate descent polish
    4. Multi-length exploration
    5. Diverse initialization exploration
    6. Final combined AMD+CD polish
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = chirp_init(n_base, freq=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Chirp init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Helper to update best
    def _update_candidate(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: AMD with top-5 on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD top5 n={n0} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = amd_backtrack(best_seq, t1, verbose=True, rng_seed=1, top_k=5)
        _update_candidate(a_out, f"amd5_n{n0}")

    # === Phase 2: AMD with top-3 (tighter focus) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD top3 n={n_best} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = amd_backtrack(best_seq, t2, verbose=True, rng_seed=2, top_k=3)
        _update_candidate(a_out, f"amd3_n{n_best}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord n={n_best} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = exact_coord_descent(best_seq, t3, verbose=True, rng_seed=3)
        _update_candidate(a_out, f"coord_n{n_best}")

    # === Phase 4: AMD polish from improved point ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD polish n={n_best} ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_out, _ = amd_backtrack(best_seq, t4, verbose=True, rng_seed=4, top_k=5)
        _update_candidate(a_out, f"amd_polish_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 5: Multi-length ===")
        # Try lengths around the current best
        candidate_lengths = []
        for dn in [-300, -200, -100, 100, 200, 300, 400, 500]:
            nl = n_best + dn
            if 600 <= nl <= 4000 and nl != n_best:
                candidate_lengths.append(nl)

        for nl in candidate_lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 25:
                break
            print(f"  Trying n={nl} (remaining={remaining:.0f}s)")
            a_interp = interp_to_length(best_seq, nl)
            t_amd = min(remaining * 0.20, 120.0)
            t_cd = min(remaining * 0.08, 50.0)
            if t_amd < 15:
                continue
            a_out, _ = amd_backtrack(a_interp, t_amd, verbose=False, rng_seed=10 + nl % 50, top_k=5)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10 and t_cd > 10:
                a_out2, _ = exact_coord_descent(a_out, t_cd, verbose=False, rng_seed=20 + nl % 50)
                # Pick better
                m1 = fast_autocorr(normalize(a_out)).max()
                m2 = fast_autocorr(normalize(a_out2)).max()
                if m2 * nl < m1 * nl:
                    a_out = a_out2
            _update_candidate(a_out, f"multi_n{nl}")

    # === Phase 6: Diverse initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Diverse init n={n_best} ===")
        inits = [
            ("chirp2", chirp_init(n_best, freq=1.5)),
            ("gauss", gaussian_init(n_best, sigma_frac=0.25)),
            ("mixed", mixed_init(n_best, np.random.default_rng(77))),
        ]
        for label, a_init in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_sub = min(remaining * 0.25, 100.0)
            if t_sub < 15:
                continue
            print(f"  Init: {label}")
            a_out, _ = amd_backtrack(a_init, t_sub, verbose=False, rng_seed=30, top_k=5)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_out2, _ = exact_coord_descent(a_out, min(remaining * 0.3, 40.0), verbose=False, rng_seed=40)
                m1 = fast_autocorr(normalize(a_out)).max()
                m2 = fast_autocorr(normalize(a_out2)).max()
                if m2 * n_best < m1 * n_best:
                    a_out = a_out2
            _update_candidate(a_out, f"div_{label}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish n={n_best} ===")
        t7_amd = min(remaining * 0.35, 80.0)
        t7_cd = min(remaining * 0.25, 60.0)
        if t7_amd > 10:
            a_out, _ = amd_backtrack(best_seq, t7_amd, verbose=False, rng_seed=99, top_k=5)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10 and t7_cd > 10:
                a_out2, _ = exact_coord_descent(a_out, t7_cd, verbose=False, rng_seed=100)
                m1 = fast_autocorr(normalize(a_out)).max()
                m2 = fast_autocorr(normalize(a_out2)).max()
                if m2 * n_best < m1 * n_best:
                    a_out = a_out2
            _update_candidate(a_out, f"final_n{n_best}")

    # === Phase 8: Ultra-fine coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Ultra-fine coord n={n_best} ===")
        t8 = min(remaining * 0.40, 50.0)
        if t8 > 8:
            a_out, _ = exact_coord_descent(best_seq, t8, verbose=False, rng_seed=200)
            _update_candidate(a_out, f"ultra_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
