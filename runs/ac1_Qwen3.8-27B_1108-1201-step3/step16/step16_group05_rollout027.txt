# Global solver
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_max_and_gradient(a, beta):
    """
    Compute smoothed max objective and its full gradient using FFT.
    Returns (smoothed_max_value, gradient).
    """
    n = len(a)
    ac = fast_autocorr(a)  # length 2n-1
    max_ac = ac.max()
    shifted = ac - max_ac
    exp_vals = np.exp(beta * shifted)
    log_sum = np.log(exp_vals.sum())
    smoothed_val = max_ac + log_sum / beta
    w = exp_vals / exp_vals.sum()
    
    # Full gradient via FFT
    # grad[j] = 2 * sum_d w[j+d] * a[d] for valid j+d
    # = 2 * conv(a, w_rev)[2n-2-j]
    w_rev = w[::-1]
    
    # Use FFT for the convolution
    conv_size = n + len(w_rev) - 1  # = n + 2n-1 - 1 = 3n-2
    fft_size = 1 << (conv_size - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    W_fft = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A_fft * W_fft, fft_size)[:conv_size]
    
    # grad[j] = 2 * conv_result[2n-2-j] for j=0,...,n-1
    # Extract: indices from n-1 to 2n-2 of conv_result, reversed
    grad = 2.0 * conv_result[n-1 : 2*n-1][::-1]
    
    return smoothed_val, grad, ac, w


def nesterov_smoothed_gradient(a0, time_limit, beta_start=30.0, beta_end=8.0,
                                 verbose=True, seed=42, c0=2.0):
    """
    Nesterov-accelerated projected gradient with full smoothed-max.
    Uses backtracking line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        frac = min(1.0, t / 30000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)
        
        # Gradient at y
        _, grad_y, ac_y, _ = compute_smoothed_max_and_gradient(y, cur_beta)
        
        # Backtracking line search
        gn = np.linalg.norm(grad_y)
        if gn < 1e-15:
            break
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Try the step
        x_new = project_simplex(y - eta * grad_y)
        
        # Verify with Armijo condition (simplified: just check objective decreased)
        # For efficiency, we skip full Armijo and just accept the step
        # but periodically check
        x_prev = x
        x = x_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.2)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NSG t={t} ({elapsed:.0f}s): obj={best_obj:.6f}, beta={cur_beta:.1f}, stall={stall}")
    
    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx
        
        y = y_new
        
        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-15:
                best_m = mc
                best_a = candidate.copy()
        
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Refined coordinate descent: move mass from high-peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 80
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find peak lags
        num_peaks = min(15, len(ac))
        peak_lags = np.argsort(ac)[::-1][:num_peaks]
        
        # Compute peak contribution per position
        peak_contrib = np.zeros(n)
        for k in peak_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            peak_contrib[idx] += contrib
        
        # Find sources (high contribution) and try to move to low-impact destinations
        sources = np.argsort(peak_contrib)[::-1][:min(50, n)]
        improved = False
        
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Score destinations: prefer positions that don't contribute to peaks
            best_dst = -1
            best_score = float('inf')
            
            # Sample destinations
            if n > 200:
                candidates = rng.choice(n, 200, replace=False)
            else:
                candidates = np.arange(n)
            
            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in peak_lags[:8]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                if impact < best_score:
                    best_score = impact
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            # Try moving
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

"""

2. Backtracking-free step with periodic restart and adaptive beta annealing
3. LP Frank-Wolfe (with constraint reduction) for exact polishing
4. Peak-aware coordinate descent for final refinement
5. Multi-length exploration and diverse initializations

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation (a * a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_max_grad_full(a, beta):
    """
    Compute the gradient of the smoothed max of the autocorrelation.
    Uses FFT for O(n log n) computation.
    Returns (smoothed_value, gradient_array).
    """
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    shifted = ac - max_ac
    exp_vals = np.exp(beta * shifted)
    exp_sum = exp_vals.sum()
    smoothed_val = max_ac + np.log(exp_sum) / beta
    w = exp_vals / exp_sum

    # Full gradient: grad[j] = 2 * sum_{d=0}^{n-1} w[j+d] * a[d] (valid indices only)
    # Computed as: grad = 2 * conv(a, w_rev)[n-1 : 2n-1][::-1]
    w_rev = w[::-1]
    conv_len = n + len(w_rev) - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    A_f = np.fft.rfft(a, fft_sz)
    W_f = np.fft.rfft(w_rev, fft_sz)
    conv_res = np.fft.irfft(A_f * W_f, fft_sz)[:conv_len]
    grad = 2.0 * conv_res[n - 1: 2 * n - 1][::-1]

    return smoothed_val, grad, ac


def nesterov_full_grad(a0, time_limit, beta_start=30.0, beta_end=8.0,
                        c0=2.0, verbose=True, seed=42):
    """
    Nesterov-accelerated projected gradient with full (non-approximated) smoothed-max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_obj = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2500

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta: anneal from sharp to smooth
        frac = min(1.0, t / 25000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        # Compute gradient at y
        _, grad_y, _ = compute_smoothed_max_grad_full(y, cur_beta)

        # Step size: standard for Nesterov on simplex
        eta = c0 / (np.sqrt(t) + 1.0)

        # Projected gradient step
        x_new = project_simplex(y - eta * grad_y)

        x_prev = x
        x = x_new

        # Check true objective
        M_new = fast_autocorr(x).max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation on long stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.15)
            noise_scale = 0.002 + 0.003 * (max_stall / 2500.0)
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NSG t={t} ({elapsed:.0f}s): obj={best_obj:.6f}, beta={cur_beta:.1f}, stall={stall}")

    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, solver, max_constraints=None, verbose=True):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    Optionally uses reduced constraint set for speed.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()
    num_iters = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix
        if max_constraints is not None and max_constraints < (2 * n - 1):
            # Use only top constraints
            top_idx = np.argsort(ac)[::-1][:max_constraints]
            T_full = np.zeros((2 * n - 1, n))
            for j in range(n):
                T_full[j:j + n, j] = y
            T = T_full[top_idx, :]
            M_rhs = ac[top_idx]
        else:
            T = np.zeros((2 * n - 1, n))
            for j in range(n):
                T[j:j + n, j] = y
            M_rhs = ac

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_rhs, cp.sum(b_var) <= 1.0]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx

        y = y_new
        num_iters = t

        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-15:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Peak-aware coordinate descent for final refinement."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 60

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        num_peaks = min(12, len(ac))
        peak_lags = np.argsort(ac)[::-1][:num_peaks]

        # Compute per-position peak contribution
        peak_contrib = np.zeros(n)
        for k in peak_lags[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            peak_contrib[idx] += contrib

        sources = np.argsort(peak_contrib)[::-1][:min(40, n)]
        improved = False

        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Score destinations by their impact on peak lags
            best_dst = -1
            best_impact = float('inf')
            num_sample = min(n, 300)
            if n > num_sample:
                candidates = rng.choice(n, num_sample, replace=False)
            else:
                candidates = np.arange(n)

            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in peak_lags[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                if impact < best_impact:
                    best_impact = impact
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.004)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 20 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Phases:
    1. NSG with moderate beta (broad optimization)
    2. NSG with high beta (sharpening)
    3. LP-FW for exact local optimum
    4. Coordinate polish
    5. Alternative lengths
    6. Alternative initializations
    7. Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load or create initial sequence
    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(p_norm)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: NSG moderate beta ===
    rem = remaining_time()
    n0 = len(best_seq)
    print(f"\n=== Phase 1: NSG n={n0} (moderate beta) ===")
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_full_grad(
            a_in, t1, beta_start=25.0, beta_end=12.0,
            c0=2.0, verbose=True, seed=42)
        try_update(a_out, f"nsg_mod_n{n0}")

    # === Phase 2: NSG high beta (sharpen) ===
    rem = remaining_time()
    print(f"\n=== Phase 2: NSG n={len(best_seq)} (high beta) ===")
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_full_grad(
            a_in, t2, beta_start=50.0, beta_end=20.0,
            c0=1.5, verbose=True, seed=77)
        try_update(a_out, f"nsg_sharp_n{len(best_seq)}")

    # === Phase 3: LP-FW ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP-FW n={n_cur} ===")
    t3 = min(rem * 0.15, 250.0)
    if t3 > 15 and n_cur <= 3000:
        # Use reduced constraints for larger n
        max_c = max(2000, n_cur) if n_cur > 2000 else None
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, max_constraints=max_c, verbose=True)
        try_update(a_out, f"lpfw_n{n_cur}")

    # === Phase 4: Coordinate polish ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord n={n_cur} ===")
    t4 = min(rem * 0.08, 100.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Try different lengths ===
    rem = remaining_time()
    if rem > 120:
        n0_orig = n0
        length_trials = [n0_orig + 400, n0_orig - 400, n0_orig + 200]
        length_trials = [lt for lt in length_trials if 1000 <= lt <= 3200]
        for n_trial in length_trials:
            rem = remaining_time()
            if rem < 80:
                break
            if n_trial == len(best_seq):
                continue
            print(f"\n=== Phase 5: NSG n={n_trial} ===")
            a_interp = interpolate_to_length(best_seq, n_trial)
            t5 = min(rem * 0.18, 180.0)
            a_out, _ = nesterov_full_grad(
                a_interp, t5, beta_start=25.0, beta_end=10.0,
                c0=2.0, verbose=True, seed=99)
            try_update(a_out, f"nsg_n{n_trial}")

            # Quick LP on result
            rem = remaining_time()
            if rem > 30 and n_trial <= 2800:
                t5b = min(rem * 0.25, 60.0)
                a_out2, _ = lp_frank_wolfe(
                    norm_sum1(a_out), t5b, solver,
                    max_constraints=max(1500, n_trial) if n_trial > 2000 else None,
                    verbose=False)
                try_update(a_out2, f"lpfw_n{n_trial}")

    # === Phase 6: Alternative initialization ===
    rem = remaining_time()
    if rem > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Alt init n={n_best} ===")
        x = np.linspace(0, 1, n_best)
        # Raised cosine
        init_alt = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        init_alt = np.maximum(init_alt, 0)
        init_alt = init_alt / init_alt.sum()

        t6 = min(rem * 0.25, 100.0)
        a_out, _ = nesterov_full_grad(
            init_alt, t6, beta_start=20.0, beta_end=10.0,
            c0=2.0, verbose=True, seed=300)
        try_update(a_out, f"alt_raisedcos_nsg")

    # === Phase 7: Final LP + Coord ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 2500 and rem > 30:
        print(f"\n=== Phase 7: Final LP n={n_best} ===")
        t7 = min(rem * 0.35, 80.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")

    rem = remaining_time()
    if rem > 20:
        print(f"\n=== Phase 8: Final Coord n={len(best_seq)} ===")
        t8 = min(rem * 0.6, 60.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
