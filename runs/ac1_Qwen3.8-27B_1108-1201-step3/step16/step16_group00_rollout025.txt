def get_peak_lags(ac, n_peaks=20):

def amd_optimize_v2(a0, time_limit, verbose=True):

                continue
            contrib = a[idx] * a[pk_i - idx]
            if len(contrib) == 0 or contrib.max() < 1e-20:
                continue

            # Find positions involved in ALL peak lags (most "dangerous")
            all_peak_positions = set()
            for pk2 in peaks:
                s2 = max(0, int(pk2) - n + 1)
                e2 = min(n, int(pk2) + 1)
                all_peak_positions.update(range(s2, e2))
            non_peak = np.array([i for i in range(n) if i not in all_peak_positions])

            for src in idx[np.argsort(contrib)[::-1][:3]]:
                if a[src] < 1e-10:
                    continue

                # Find best destination
                if len(non_peak) > 0:
                    # Score destinations: lower is better
                    scores = np.zeros(n)
                    for pk2 in peaks[:8]:
                        pk2_i = int(pk2)
                        for d in range(len(non_peak)):
                            dd = non_peak[d]
                            d1 = pk2_i - dd
                            if 0 <= d1 < n:
                                scores[d] += a[d1]
                    best_d = non_peak[np.argmin(scores)]
                else:
                    best_d = int(rng.integers(n))
                    if best_d == src:
                        best_d = (best_d + 1) % n

                if best_d == src:
                    continue

                # Line search: binary search for best fraction
                best_frac = 0.0
                best_trial_m = best_m
                for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_d] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = frac

                if best_frac > 0:
                    amt = a[src] * best_frac
                    a[src] -= amt
                    a[best_d] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True
                        stall = 0
                    break  # Move on to next peak

        # Strategy 2: 2D move - take from high contributor, add to two destinations
        if not improved and stall > 50:
            ac_cur = fast_autocorr(a)
            peaks_cur = get_peak_lags(ac_cur, n_peaks=5)
            src = int(np.argmax(a))
            if a[src] > 1e-5:
                all_peak_pos = set()
                for pk2 in peaks_cur:
                    s2 = max(0, int(pk2) - n + 1)
                    e2 = min(n, int(pk2) + 1)
                    all_peak_pos.update(range(s2, e2))
                non_peak = np.array([i for i in range(n) if i not in all_peak_pos])
                if len(non_peak) >= 2:
                    d1 = non_peak[rng.integers(len(non_peak))]
                    d2 = non_peak[rng.integers(len(non_peak))]
                    if d1 == d2:
                        d2 = (d2 + 1) % n
                    a_trial = a.copy()
                    amt = a[src] * 0.05
                    a_trial[src] -= amt
                    a_trial[d1] += amt * 0.6
                    a_trial[d2] += amt * 0.4
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0

        if not improved:
            stall += 1
            # Random kick
            if stall % 20 == 0:
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-5:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord2 ({elapsed:.0f}s): J={J:.6f}, stall={stall}, round={round_count}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord2 final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True):
    """
    LP-based Frank-Wolfe with:
    - Adaptive top-K selection
    - Nesterov acceleration
    - Periodic re-evaluation of constraint set
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    top_K = 100

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix for top-K most binding constraints
        # (y*b)[k] = sum_i y[i] * b[k-i] for valid indices
        # We want: sum_i y[k-i] * b[i] <= M for k in top-K
        # This is: T @ b <= M * ones where T[k, i] = y[k-i] if valid

        # Find top-K constraints
        # Constraint k: sum_{i: 0<=k-i<n} y[k-i] * b[i] <= M
        # The "tightness" of constraint k is M - (y*y)[k] = M - ac[k]
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build T matrix (K_sel x n)
        T = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Adaptively increase K
        if t % 100 == 0:
            top_K = min(top_K + 20, 300)

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP2 iter={t} ({elapsed:.0f}s): J={J:.6f}, K={top_K}")

    return best_a, best_m


def parametric_sweep(n, verbose=True):
    """Sweep over parametric families to find good initializations."""
    best_a = None
    best_m = float('inf')

    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
        a = (np.arange(1, n + 1, dtype=np.float64) ** p)
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    # Exponentials
    for p in [-2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0]:
        a = np.exp(p * np.linspace(0, 1, n))
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    # Cosine windows
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** alpha
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    # Power law + cosine taper
    for p in [0.5, 1.0, 1.5]:
        for beta in [0.5, 1.0, 2.0]:
            x = np.linspace(0, 1, n)
            a = (x + 0.01) ** p * np.cos(np.pi * x / 2) ** beta
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                m = fast_autocorr(a).max()
                if m < best_m:
                    best_m = m
                    best_a = a.copy()

    # Truncated power (peak in middle)
    for p in [0.5, 1.0, 1.5, 2.0]:
        x = np.linspace(0, 1, n)
        a = np.minimum(x, 1 - x) ** p + 0.001
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    if verbose:
        J = 2.0 * n * best_m
        print(f"  Parametric sweep: best J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def peak_equalize(a0, time_limit, verbose=True):
    """
    Try to equalize the top few peak lags of the convolution.
    If lags k1, k2, ... are all near the max, try to make them exactly equal
    while minimizing the common value.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()
        peaks = get_peak_lags(ac, n_peaks=8)

        if len(peaks) < 2:
            break

        # Try to reduce the max by adjusting coordinates
        # For each pair of coordinates (i, j), compute the effect on each peak
        # This is too expensive for large n, so we use a greedy approach

        # Find the coordinate that most contributes to the highest peak
        pk_max = int(np.argmax(ac))
        i_start = max(0, pk_max - n + 1)
        i_end = min(n, pk_max + 1)
        idx = np.arange(i_start, i_end)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[pk_max - idx]
        if len(contrib) == 0:
            break

        # Find the coordinate to reduce
        src = int(idx[np.argmax(contrib)])
        if a[src] < 1e-10:
            break

        # Find best destination: minimize the maximum increase across all peaks
        peak_list = [int(pk) for pk in peaks]

        def max_increase(dst):
            """Compute max increase in convolution at peak lags if we add 1 to position dst."""
            max_inc = 0.0
            for pk in peak_list:
                # (a + e_dst * amt)[*] at pk increases by amt * (a[pk-dst] + a[pk-dst] if pk-dst != dst)
                inc = 0.0
                d = pk - dst
                if 0 <= d < n:
                    inc += a[d]
                    if d != dst:
                        inc += a[d]
                d2 = dst - pk
                # Actually, the increase in (a*a)[pk] from adding to a[dst] is:
                # a[dst]*a[pk-dst] + a[pk-dst]*a[dst] = 2*a[pk-dst] (if pk-dst valid and != dst)
                # or a[dst]^2 if pk-dst = dst
                d = pk - dst
                if 0 <= d < n:
                    if d == dst:
                        inc = a[dst]
                    else:
                        inc = 2.0 * a[d]
                    if inc > max_inc:
                        max_inc = inc
            return max_inc

        # Try a subset of candidate destinations
        candidate_starts = list(range(0, n, max(1, n // 200)))
        if len(candidate_starts) < 50:
            candidate_starts = list(range(0, n, max(1, n // 50)))
        # Add some random candidates
        rng = np.random.default_rng(42 + iteration)
        for _ in range(50):
            candidate_starts.append(int(rng.integers(n)))

        best_dst = -1
        best_score = float('inf')
        for d in candidate_starts:
            if d == src:
                continue
            sc = max_increase(d)
            if sc < best_score:
                best_score = sc
                best_dst = d

        if best_dst < 0:
            break

        # Try different fractions
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try a tiny random perturbation
            i = int(np.random.default_rng(iteration).integers(n))
            if a[i] > 1e-5:
                j = int(np.random.default_rng(iteration + 1).integers(n))
                if j != i:
                    amt = a[i] * 0.005
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True

        if not improved and iteration > 20:
            break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakEqualize ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization:
    1. Load parent or create initial
    2. Parametric sweep for good initialization
    3. AMD optimization (exponential family)
    4. LP Frank-Wolfe
    5. Peak equalization
    6. Coordinate descent polish
    7. Length exploration
    8. Final polish rounds
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = (x + 0.01) ** 1.0
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Parametric sweep at current length ===
    print(f"\n=== Phase 1: Parametric sweep (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    p_sweep_a, p_sweep_m = parametric_sweep(n0, verbose=True)
    try_update(p_sweep_a, "parametric")

    # Also try parent-based blends
    if parent is not None:
        parent_norm = norm_sum1(parent)
        # Blend with various power laws
        for p in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            ramp = (np.arange(1, n0 + 1, dtype=np.float64) ** p)
            ramp = norm_sum1(ramp)
            for alpha in [0.3, 0.5, 0.7]:
                blend = (1 - alpha) * parent_norm + alpha * ramp
                blend = norm_sum1(blend)
                m = fast_autocorr(blend).max()
                if m < best_m if 'best_m' in dir() else True:
                    try_update(blend, f"blend_p{p}_a{alpha}")

    # === Phase 2: AMD optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: AMD optimization (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, m_out = amd_optimize_v2(a_in, t2, verbose=True)
        try_update(a_out, f"amd_n{len(a_out)}")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n_best}) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, m_out = lp_frank_wolfe_v2(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 4: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak equalization (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, m_out = peak_equalize(a_in, t4, verbose=True)
        try_update(a_out, f"peak_eq_n{n_best}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, m_out = coordinate_descent_v2(a_in, t5, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: AMD again on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: AMD round 2 (n={n_best}) ===")
    t6 = min(remaining * 0
