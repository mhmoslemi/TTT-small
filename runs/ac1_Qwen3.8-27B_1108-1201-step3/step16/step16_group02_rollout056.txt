Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Multi-lag subgradient mirror descent
4. Pairwise coordinate descent for final polish

import numpy as np
import cvxpy as cp

def fast_conv(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_shape(a, time_limit, verbose=True):
    """Flatten FFT magnitude to reduce peak autocorrelation."""
    a = norm_sum1(a)
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    alpha = 0.7
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(x, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        
        # Relative compression: bring all mags closer to the mean
        mag_mean = mag.mean()
        mag_new = mag ** alpha
        # Also raise the floor
        floor = 0.1 * mag_mean
        mag_new = np.maximum(mag_new, floor)
        
        A_new = mag_new * np.exp(1j * phase)
        x_new = np.fft.irfft(A_new, fft_size)[:n]
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            break
        x_new = x_new / s
        
        m_new = fast_conv(x_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            # Accept with some probability (simulated annealing)
            if np.random.random() < np.exp(-(m_new - best_m) / (0.001 * best_m + 1e-15)):
                x = x_new
            # Reduce alpha (more aggressive flattening)
            alpha = max(0.1, alpha * 0.995)
        
        x = x_new
        
        if verbose and (stall % 100 == 0):
            elapsed = time.monotonic() - t0
            print(f"    Spec ({elapsed:.1f}s): score={2*n*best_m:.6f}, alpha={alpha:.4f}, stall={stall}")
    
    return best_a, best_m
