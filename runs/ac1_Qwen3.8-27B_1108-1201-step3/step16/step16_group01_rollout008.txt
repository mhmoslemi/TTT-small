import numpy as np
import cvxpy as cp
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgradient_active(a, c, active_lags):
    """Compute subgradient at active lags."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        g[idx] += 2.0 * a[j_idx]
    return g

def mirror_step(a, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def solve_bundle_lp(cuts, n, solver):
    """Solve the bundle LP subproblem."""
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [a_var.sum() == 1, M_var >= 0]
    for (g, f_val, a_val) in cuts:
        # M >= f_val + g^T (a - a_val)
        constraints.append(M_var >= f_val + g @ a_var - g @ a_val)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_sol = np.asarray(a_var.value, dtype=np.float64)
    a_sol = np.maximum(a_sol, 0)
    s = a_sol.sum()
    if s < 1e-15:
        return None
    return a_sol / s

import numpy as np
import cvxpy as cp
import time

# Global solver cache
_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.HIGHS, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def fast_autocorr(a):
    """FFT-based linear autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, c, active_lags):
    """
    Compute subgradient of max(c_k) at the active lags.
    g_i = 2 * sum_{k in active} a[k-i] (with valid index handling)
    """
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += 2.0 * a[j_idx]
    return g


def mirror_step(a, g, eta):
    """Mirror descent (exponential) step in log-space."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def solve_bundle_lp(cuts, n, solver, max_time=5.0):
    """
    Solve the bundle LP:
      min M
      s.t. M >= f_m + g_m^T (a - a_m)  for each cut m
           sum(a) = 1, a >= 0
    Returns the optimal a or None.
    """
    if len(cuts) == 0:
        return None
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [a_var.sum() == 1]
    for (g, f_val, a_val) in cuts:
        rhs = f_val - g @ a_val
        constraints.append(M_var >= rhs + g @ a_var)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        start = time.monotonic()
        prob.solve(solver=solver, verbose=False)
        if time.monotonic() - start > max_time:
            return None
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_sol = np.asarray(a_var.value, dtype=np.float64)
    if a_sol is None or not np.all(np.isfinite(a_sol)):
        return None
    a_sol = np.maximum(a_sol, 0.0)
    s = a_sol.sum()
    if s < 1e-15:
        return None
    return a_sol / s


def init_raised_cosine(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return normalize(a)


def init_spectral(n, rng):
    """Spectral initialization: random phases with shaping."""
    # Create a power spectrum with desired shape
    n_freq = n // 2 + 1
    # Low-pass filter to avoid high-frequency content
    freq = np.arange(n_freq) / n_freq
    shape = np.exp(-4 * freq ** 2) + 0.1  # Gaussian low-pass + DC
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    # Build complex spectrum
    spectrum = np.sqrt(shape) * np.exp(1j * phases)
    # Ensure hermitian symmetry
    if n % 2 == 0:
        spectrum = np.concatenate([spectrum[:n//2+1], np.conj(spectrum[:n//2-1:-1])])
    else:
        spectrum = np.concatenate([spectrum[:n//2+1], np.conj(spectrum[:n//2:-1])])
    a = np.real(np.fft.ifft(spectrum))
    return normalize(a)


def init_random(n, rng):
    """Random positive initialization."""
    a = rng.exponential(1.0, n)
    return normalize(a)


def bundle_mirror_descent(a0, time_limit, solver, rng, verbose=True):
    """
    Main optimization: Adaptive Bundle Mirror Descent.
    Combines fast mirror descent steps with periodic LP subproblem solves.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Bundle storage
    cuts = []  # (g, f_val, a_val)
    max_cuts = 80
    lp_interval = 150  # Solve LP every N iterations
    lp_blend = 0.5  # Blend LP solution with current
    
    # Adaptive parameters
    eta = 5.0 / np.sqrt(n)  # Initial step size
    eta_min = 0.01 / np.sqrt(n)
    eta_max = 50.0 / np.sqrt(n)
    eta_decay = 0.9999  # Slow decay
    adaptive_thresh = 0.995  # Active set threshold (fraction of max)
    
    stall_count = 0
    max_stall = 3000
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation
        c = fast_autocorr(a)
        M_curr = c.max()
        
        # Check improvement
        if M_curr < best_M - 1e-15:
            best_M = M_curr
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Determine active set
        # Adaptive: include lags that are within adaptive_thresh of max
        active_thresh = adaptive_thresh * M_curr
        active = np.where(c >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([int(c.argmax())])
        # Cap active set size
        if len(active) > 60:
            # Keep only the top 60
            top_idx = np.argsort(c)[::-1][:60]
            active = top_idx
        
        # Compute subgradient
        g = compute_subgradient(a, c, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-20:
            break
        g = g / g_norm  # Normalize to unit L2
        
        iter_count += 1
        
        # --- Bundle LP (periodic) ---
        if iter_count % lp_interval == 0 and len(cuts) >= 5:
            a_lp = solve_bundle_lp(cuts, n, solver, max_time=3.0)
            if a_lp is not None:
                c_lp = fast_autocorr(a_lp)
                M_lp = c_lp.max()
                # Blend LP solution with current
                a_new = (1 - lp_blend) * a + lp_blend * a_lp
                a_new = normalize(a_new)
                M_new = fast_autocorr(a_new).max()
                if M_new < M_curr:
                    a = a_new
                    # Reduce step size after LP
                    eta = max(eta * 0.5, eta_min)
                    continue
                else:
                    # LP didn't help, just use its subgradient info
                    pass
            # If LP failed, increase interval
            lp_interval = min(lp_interval + 20, 500)
        
        # --- Mirror descent step ---
        a_new = mirror_step(a, g, eta)
        M_new = fast_autocorr(a_new).max()
        
        if M_new < M_curr - 1e-15:
            a = a_new
            # Increase step size slightly
            eta = min(eta * 1.01, eta_max)
        else:
            # Backtracking: reduce step size
            eta = max(eta * 0.7, eta_min)
            # Try a smaller step
            if eta > eta_min:
                a_new2 = mirror_step(a, g, eta)
                M_new2 = fast_autocorr(a_new2).max()
                if M_new2 < M_curr:
                    a = a_new2
        
        # --- Update bundle ---
        if iter_count % 10 == 0:  # Add cut periodically
            cuts.append((g.copy(), M_curr, a.copy()))
            if len(cuts) > max_cuts:
                # Remove the oldest cut that's least "active"
                # (least tight at current point)
                viol = []
                for i, (g_m, f_m, a_m) in enumerate(cuts):
                    # How much does this cut "bind"? (M_curr - (f_m + g_m^T(a - a_m)))
                    val = f_m + g_m @ a - g_m @ a_m
                    viol.append(M_curr - val)
                # Remove the one with the largest violation (least binding)
                remove_idx = np.argmax(viol)
                cuts.pop(remove_idx)
        
        # --- Decay step size ---
        eta *= eta_decay
        
        # --- Restart if stuck ---
        if stall_count > max_stall:
            stall_count = 0
            # Perturb from best
            noise = rng.normal(0, 0.01, n) * best_a
            a = normalize(best_a + noise)
            x_prev = a.copy()
            # Reset some parameters
            eta = 5.0 / np.sqrt(n)
            adaptive_thresh = max(0.98, adaptive_thresh - 0.005)
            # Clear some cuts
            if len(cuts) > 20:
                cuts = cuts[len(cuts)//2:]
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    BMD restart ({elapsed:.1f}s): thresh={adaptive_thresh:.4f}, eta={eta:.6f}")
            
            max_stall = int(max_stall * 1.2)  # Be more patient next time
        
        # Verbose logging
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    BMD iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, "
                  f"active={len(active)}, eta={eta:.6f}, stall={stall_count}")
    
    return best_a, best_M


def local_peak_reduction(a0, time_limit, rng, verbose=True):
    """
    Targeted local search: identify peak lags and perform coordinated
    mass transfers to reduce them.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M_curr = c.max()
        
        if M_curr < best_M - 1e-15:
            best_M = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peak lags
        n_peaks = min(10, len(c))
        peak_lags = np.argsort(c)[::-1][:n_peaks]
        
        # For each peak, find the "worst" pair and try to reduce it
        improved = False
        for k in peak_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            if len(idx) == 0:
                continue
            
            # Contribution of each pair
            contrib = a[idx] * a[j_idx]
            total = contrib.sum()
            if total < 1e-20:
                continue
            
            # Find the pair with maximum contribution
            max_i_local = np.argmax(contrib)
            src = idx[max_i_local]
            dst = j_idx[max_i_local]
            
            # Compute "peak pressure" at each position
            pressure = np.zeros(n)
            for k2 in peak_lags[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                j2 = k2i - idx2
                pressure[idx2] += a[j2]
            
            # Find low-pressure positions (good destinations)
            # Exclude positions near peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peak_lags[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) > 100:
                candidates = rng.choice(candidates, size=100, replace=False)
            
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Sort by pressure (want low pressure)
            candidates = candidates[np.argsort(pressure[candidates])[:20]]
            
            # Try transfers
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            for dst in candidates:
                if dst == src or a[dst] >= a[src]:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = src_val * frac
                    if amt < 1e-18:
                        continue
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0.0)
                    s_t = trial.sum()
                    if s_t < 1e-15:
                        continue
                    trial = trial / s_t
                    M_trial = fast_autocorr(trial).max()
                    if M_trial < best_M - 1e-15:
                        a = trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape
            n_pert = max(5, n // 200)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
            if not improved:
                stall += 1
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Local ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_M


def refine_length(a0, target_n, time_limit, solver, rng, verbose=True):
    """Interpolate to target length and optimize."""
    old_n = len(a0)
    if old_n == target_n:
        return bundle_mirror_descent(a0, time_limit, solver, rng, verbose)
    
    # Interpolate
    s = a0.sum()
    if s < 1e-15:
        a_new = np.ones(target_n) / target_n
    else:
        a_norm = a0 / s
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, target_n)
        a_new = np.interp(new_x, old_x, a_norm)
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = np.ones(target_n) / target_n
        else:
            a_new = a_new / ss
    
    return bundle_mirror_descent(a_new, time_limit, solver, rng, verbose)
