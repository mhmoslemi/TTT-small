```python
    weights has length 2n-1.
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based full gradient computation for smoothed max objective
2. Nesterov-accelerated mirror descent (multiplicative weights)
3. Adaptive beta schedule with backtracking
4. Multiple diverse structured initializations
5. Final coordinate polish with peak-aware moves
"""

import numpy as np
import time

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad_fft(a, weights):
    """
    Compute grad[j] = 2 * sum_m a[m] * weights[m+j] for j=0..n-1
    using FFT-based convolution.
    weights has length 2n-1.
    """
    n = len(a)
    if n == 0:
        return np.array([])
    w_rev = weights[::-1]  # length 2n-1
    # conv(a, w_rev) has length n + (2n-1) - 1 = 3n-2
    L = 3 * n - 2
    fft_size = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:L]
    # g[j] = sum_m a[m] * w[m+j] = conv(a, w_rev)[(2n-2) - j]... 
    # Let me re-derive.
    # conv(a, w_rev)[k] = sum_i a[i] * w_rev[k-i] = sum_i a[i] * w[2n-2-(k-i)]
    #                     = sum_i a[i] * w[2n-2-k+i]
    # We want g[j] = sum_m a[m] * w[m+j]
    # So we need 2n-2-k+m = m+j => 2n-2-k = j => k = 2n-2-j
    # Therefore g[j] = conv(a, w_rev)[2n-2-j]
    # So g = conv[2n-2 : 2n-2-n+1 : -1] = conv[n-2 : 2n-2 : -1]... 
    # Wait, for j=0: g[0] = conv[2n-2]
    # For j=n-1: g[n-1] = conv[2n-2-(n-1)] = conv[n-1]
    # So g[j] = conv[2n-2-j] for j=0..n-1
    # That means g = conv[n-1 : 2n-1][::-1] = conv[n-1:2n-1] reversed
    # Or equivalently: g = conv[2n-2 : n-2 : -1]
    result = conv[n-1:2*n-1][::-1]
    return 2.0 * result

def smooth_max_and_grad(a, beta):
    """Compute smoothed max of autocorrelation and its gradient using full lags."""
    ac = fast_conv(a)
    n = len(a)
    num_lags = len(ac)
    
    # Compute softmax weights
    max_val = ac.max()
    shifted = beta * (ac - max_val)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    
    # Smoothed max value
    smax = max_val + log_sum / beta
    
    # Weights
    weights = exp_vals / exp_vals.sum()
    
    # Gradient using FFT
    grad = compute_grad_fft(a, weights)
    
    return smax, grad, ac

def mirror_descent(a, grad, eta):
    """Multiplicative weight update (mirror descent with KL)."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    # Numerical stability
    log_a_new = log_a_new - log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a
    return a_new / s

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_apg(a0, time_limit, beta_start=30.0, beta_end=10.0,
                  verbose=True, seed=42):
    """Nesterov-accelerated projected gradient on the simplex with full gradient."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta_c = 1.0
    stall = 0
    max_stall = 2000
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        t = iterations + 1
        
        # Adaptive beta
        frac = min(1.0, t / 10000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient at y
        _, grad, _ = smooth_max_and_grad(y, cur_beta)
        
        # Step size
        eta = eta_c / (np.sqrt(t) + 1.0)
        
        # Projected gradient step
        x_new = project_simplex(y - eta * grad)
        
        # Check if we made progress
        x_prev = x
        x = x_new
        iterations += 1
        
        M_new = fast_conv(x).max()
        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                print(f"      APG restart iter={t}: obj={2*n*best_obj:.6f}")
        
        if verbose and iterations % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      APG iter={iterations} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}, beta={cur_beta:.1f}")
    
    return best_a, 2.0 * n * best_obj

def mirror_descent_optimize(a0, time_limit, beta=20.0, verbose=True, seed=42):
    """Mirror descent (multiplicative weights) optimization."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    eta_start = 0.5
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        t = iterations + 1
        eta = eta_start / (np.sqrt(t) + 1.0)
        
        _, grad, _ = smooth_max_and_grad(a, beta)
        
        # Multiplicative update
        a_new = mirror_descent(a, grad, eta)
        
        M_new = fast_conv(a_new).max()
        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller step
            a_new = mirror_descent(a, grad, eta * 0.1)
            M_new = fast_conv(a_new).max()
            if M_new < best_obj:
                best_obj = M_new
                best_a = a_new.copy()
                stall = 0
        
        a = a_new
        iterations += 1
        
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            # Perturb
            mask = rng.random(n) < 0.1
            a[mask] *= rng.uniform(0.5, 1.5, mask.sum())
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
        
        if verbose and iterations % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter={iterations} ({elapsed:.0f}s): obj={2*n*best_obj:.6f}")
    
    return best_a, 2.0 * n * best_obj

def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent with peak-aware moves."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False
        
        # For each top peak, find the most contributing element pair
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            products = a[idx] * a[j_idx]
            
            # Find the max contributing pair
            max_i = idx[np.argmax(products)]
            
            # Try reducing max_i and adding to a random low-impact position
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based full gradient for smoothed max (log-sum-exp) objective - more accurate than top-k
2. Nesterov-accelerated projected gradient with adaptive beta
3. Mirror descent (multiplicative weights) as alternative geometry
4. Smart coordinate polish with gradient-informed swaps
5. Multiple diverse initializations for global exploration
"""

import numpy as np
import time


def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, weights):
    """
    Compute grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[m+j] for j=0..n-1
    using FFT-based convolution. weights has length 2n-1.
    """
    n = len(a)
    if n == 0:
        return np.array([])
    w_rev = weights[::-1]
    L = 3 * n - 2
    fft_size = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:L]
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def smooth_max_and_grad(a, beta):
    """Compute smoothed max of autocorrelation and its full gradient via FFT."""
    ac = fast_conv(a)
    max_val = ac.max()
    shifted = beta * (ac - max_val)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    smax = max_val + log_sum / beta
    weights = exp_vals / exp_vals.sum()
    grad = compute_grad_fft(a, weights)
    return smax, grad, ac, ac.max()


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_apg(a0, time_limit, beta_start=40.0, beta_end=15.0,
                  verbose=True, seed=42, gamma=1.5):
    """Nesterov-accelerated projected gradient with full FFT gradient."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    _, _, _, best_m = smooth_max_and_grad(a, 10.0)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 1500
    iterations = 0
    best_bc = beta_start

    while time.monotonic() - t0 < time_limit:
        t = iterations + 1
        frac = min(1.0, t / 15000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov extrapolation
        bt = (t - 1.0) / (t + 1.5)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Full gradient via FFT
        _, grad, _, _ = smooth_max_and_grad(y, cur_beta)

        # Adaptive step: based on gradient norm
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            gn = 1e-15
