def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * np.conj(A), fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def subgradient_vec(a, ac, top_k=50, beta=3.0):
    n = len(a)
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    max_v = top_vals.max()
    w = np.exp(beta * (top_vals - max_v))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / ws
    grad = np.zeros(n)
    for i in range(k):
        pk = int(top_idx[i])
        c = 2.0 * w[i]
        m_lo = max(0, pk - n + 1)
        m_hi = min(n - 1, pk)
        if m_lo <= m_hi:
            j_lo = pk - m_hi
            j_hi = pk - m_lo
            grad[m_lo:m_hi + 1] += c * a[j_lo:j_hi + 1][::-1]
    return grad

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def find_solver():
    import cvxpy as cp
    candidates = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Fallback solver: ECOS")
    return cp.ECOS


def make_fourier_init(n, freq, amp, phase=0.0):
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * math.pi * freq * x / n + phase)
    a = np.maximum(a, 0.0)
    return normalize(a)


def make_sine_init(n, freq):
    x = np.arange(n, dtype=np.float64)
    a = np.sin(math.pi * x / n) ** 2
    a = a + 0.1 * np.cos(2.0 * math.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    return normalize(a)
