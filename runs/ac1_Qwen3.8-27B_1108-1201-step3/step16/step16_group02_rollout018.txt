```python
        # For cross: conv(y, s)
        ys_conv_full = np.convolve(y, s)  # This is O(n^2) which is slow!
        
        # Better: use FFT
        # conv(y, s) can be computed with FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Y = np.fft.rfft(y, fft_size)
        S = np.fft.rfft(s, fft_size)
        ys_conv = np.fft.irfft(Y * S, fft_size)[:size]
        
        # Now find best t using ternary search on [0, 1]
        # f(t) = max_k [(1-t)^2 * yy[k] + 2t(1-t) * ys[k] + t^2 * ss[k]]
        # This is max of quadratics, hence convex in t
        
        def obj(t_val):
            val = (1-t_val)**2 * yy_conv + 2*t_val*(1-t_val) * ys_conv + t_val**2 * ss_conv
            return val.max()
        
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2
        a_ls, b_ls = 0.0, 1.0
        c_ls = b_ls - gr * (b_ls - a_ls)
        d_ls = a_ls + gr * (b_ls - a_ls)
        fc = obj(c_ls)
        fd = obj(d_ls)
        for _ in range(30):
            if fc < fd:
                b_ls = d_ls
                d_ls = c_ls
                fd = fc
                c_ls = b_ls - gr * (b_ls - a_ls)
                fc = obj(c_ls)
            else:
                a_ls = c_ls
                c_ls = d_ls
                fc = fd
                d_ls = a_ls + gr * (b_ls - a_ls)
                fd = obj(d_ls)
        
        t_best = (a_ls + b_ls) / 2
        y_new = (1 - t_best) * y + t_best * s
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()
        
        # Recompute
        yy_conv = fast_autocorr(y_new)
        M_new = yy_conv.max()
        
        if M_new < best_m:
            best_m = M_new
            best_a = y_new.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FW2 iter={t}: score={2*best_m:.8f}, t={t_best:.4f} ({elapsed:.1f}s)")
    
    return best_a, best_m
```

What about using the fact that at the optimum, the subgradient condition must hold? At the optimum a*, we have:

where N_C is the normal cone of the constraint set C = {a : sum(a) = 1, a >= 0}.

This means there exists λ (Lagrange multiplier for sum constraint) and μ_i >= 0 (for non-negativity) such that:

This is a very useful characterization! It means we can use it for polishing: if the subgradient is not constant on the support, we can improve by transferring mass from high-subgradient positions to low-subgradient positions.

```python
def subgradient_balance(a, n_iters=1000):
    """Balance the subgradient on the support of a."""
    n = len(a)
    a = a.copy()
    
    for it in range(n_iters):
        conv_a = fast_autocorr(a)
        M = conv_a.max()
        peak_lags = np.where(conv_a >= M * 0.999)[0]
        
        # Compute subgradient (average over peak lags)
        g = np.zeros(n)
        for k in peak_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                g[idx] += 2.0 * a[k_int - idx] / len(peak_lags)
        
        # On the support of a, we want g to be constant
        support = np.where(a > 1e-10)[0]
        if len(support) < 2:
            break
        
        g_support = g[support]
        g_mean = g_support.mean()
        g_std = g_support.std()
        
        if g_std < 1e-10:
            break
        
        # Transfer mass from high-g to low-g positions
        threshold = g_mean + 0.5 * g_std
        high = support[g[support] > threshold]
        low = support[g[support] < g_mean - 0.5 * g_std]
        
        if len(high) == 0 or len(low) == 0:
            break
        
        # Move a small amount
        for h in high[:5]:
            for l in low[:5]:
                amt = min(a[h], a[l]) * 0.01
                a[h] -= amt
                a[l] += amt
        
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a
```

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_conv(a, b):
    """Compute cross-convolution a * b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # Start with top lags, expand as we iterate
        n_cons = min(base_constraints + t * 2, max_constraints)
        
        # Primary: top lags by value
        top_lags = np.argsort(yy_conv)[::-1][:n_cons]
        
        # Also ensure we include lag 0 and lag 2n-2 (edges)
        if 0 not in top_lags:
            top_lags = np.append(top_lags, 0)
        if (2*n-2) not in top_lags:
            top_lags = np.append(top_lags, 2*n-2)
        
        # Build constraint matrix
        K = top_lags
        T = np.zeros((len(K), n))
        rhs_offset = np.zeros(len(K))
        
        for row, k in enumerate(K):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row, cols] = 2.0 * y[k_int - cols]
                rhs_offset[row] = yy_conv[k_int]
        
        # Solve LP
        M_var = cp.Variable()
        a_var = cp.Variable(n, nonneg=True)
        cons = [cp.sum(a_var) == 1]
        
        # Constraint: T @ a_var <= M_var + rhs_offset
        cons.append(T @ a_var <= M_var + rhs_offset)
        
        prob = cp.Problem(cp.Minimize(M_var), cons)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        s = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_sum = s.sum()
        if s_sum < 1e-15:
            continue
        s = s / s_sum
        
        # Check if LP solution is the same as current (convergence)
        if np.abs(s - y).max() < 1e-8:
            break
        
        # Golden section line search
        # f(t) = max_k [(1-t)^2 * yy[k] + 2t(1-t) * ys[k] + t^2 * ss[k]]
        ss_conv = fast_autocorr(s)
        ys_conv = cross_conv(y, s)
        
        # Ensure same length
        L = min(len(yy_conv), len(ys_conv), len(ss_conv))
        yy_L = yy_conv[:L]
        ys_L = ys_conv[:L]
        ss_L = ss_conv[:L]
        
        # Precompute coefficients for each lag
        # f(t)[k] = A_k * t^2 + B_k * t + C_k
        # where C_k = yy[k], B_k = 2*(ys[k] - yy[k]), A_k = ss[k] - 2*ys[k] + yy[k]
        A_coeff = ss_L - 2.0 * ys_L + yy_L
        B_coeff = 2.0 * (ys_L - yy_L)
        C_coeff = yy_L
        
        def obj(t_val):
            val = A_coeff * t_val**2 + B_coeff * t_val + C_coeff
            return val.max()
        
        # Golden section search on [0, 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_ls, b_ls = 0.0, 1.0
        c_ls = b_ls - gr * (b_ls - a_ls)
        d_ls = a_ls + gr * (b_ls - a_ls)
        fc = obj(c_ls)
        fd = obj(d_ls)
        
        for _ in range(40):
            if fc < fd:
                b_ls = d_ls
                d_ls = c_ls
                fd = fc
                c_ls = b_ls - gr * (b_ls - a_ls)
                fc = obj(c_ls)
            else:
                a_ls = c_ls
                c_ls = d_ls
                fc = fd
                d_ls = a_ls + gr * (b_ls - a_ls)
                fd = obj(d_ls)
        
        t_best = (a_ls + b_ls) / 2.0
        
        # Update
        y_new = (1.0 - t_best) * y + t_best * s
        y_new = np.maximum(y_new, 0.0)
        y_new_sum = y_new.sum()
        if y_new_sum < 1e-15:
            break
        y_new = y_new / y_new_sum
        
        yy_conv = fast_autocorr(y_new)
        M_new = yy_conv.max()
        
        if M_new < best_m:
            best_m = M_new
            best_a = y_new.copy()
        
        y = y_new
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t}: score={2.*best_m:.8f}, t_step={t_best:.4f}, K={len(K)} ({elapsed:.1f}s)")
    
    return best_a, best_m
