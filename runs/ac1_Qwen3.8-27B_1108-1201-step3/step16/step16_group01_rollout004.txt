def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Resample a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    ratio = ac / M
    weights = np.maximum(ratio - min_frac, 0.0) ** p
    total_w = weights.sum()
    if total_w < 1e-30:
        # Fall back to just the max lag
        k_max = int(ac.argmax())
        g = np.zeros(n)
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_max - idx
        g[idx] += 2.0 * a[j_idx]
        return g

    weights = weights / total_w
    g = np.zeros(n)
    # Only process lags with significant weight (efficiency)
    active_mask = weights > 1e-6
    active_indices = np.where(active_mask)[0]
    for k in active_indices:
        k_int = int(k)
        w_k = weights[k_int]
        if w_k < 1e-8:
            continue
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * w_k * a[j_idx]
    return g


def exact_active_subgradient(a, ac, threshold_frac=0.99):
    """Compute subgradient using only lags near the maximum (active set)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    n_active = len(active)
    g /= n_active  # Average over active constraints
    return g


def mirror_update_log(y, g, eta):
    """Mirror descent update in log-space for non-negativity."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def symmetrize_reverse(a):
    """Average a sequence with its reverse (exploits reversal symmetry)."""
    a_rev = a[::-1].copy()
    a_sym = 0.5 * (a + a_rev)
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s


def amd_peak_pressure(a0, time_limit, c0=3.0, p_power=3.0, min_frac=0.5,
                      verbose=True, rng_seed=42, use_sym=True):
    """
    AMD with peak-pressure weighted subgradient and adaptive threshold.
    Novel: uses weighted subgradient (not binary active set) for smoother optimization.
    """
    n = len(a0)
    a = norm_sum1(a0)
    if use_sym:
        a = symmetrize_reverse(a)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    current_min_frac = min_frac
    c_current = c0
    p_current = p_power

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = peak_weighted_subgradient(y, ac, p=p_current, min_frac=current_min_frac)

        # Step size
        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n) * 0.5

        a_new = mirror_update_log(y, g, eta)

        if use_sym and t % 50 == 0:
            a_new = symmetrize_reverse(a_new)

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive: adjust parameters if stuck
        if stall_count > max_stall:
            stall_count = 0
            # Perturb from best
            noise_scale = 0.003 + 0.005 * (stall_count // max_stall)
            perturbation = rng.normal(0, noise_scale, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            if use_sym:
                x_restarted = symmetrize_reverse(x_restarted)
            x = x_restarted
            x_prev = x_restarted
            # Adapt parameters
            if current_min_frac > 0.3:
                current_min_frac = max(0.3, current_min_frac - 0.03)
            p_current = min(p_current * 0.9, 1.5)
            c_current = c_current * 1.3
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD-PP restart t={t} ({elapsed:.1f}s): "
                      f"J={J:.6f}, min_frac={current_min_frac:.3f}, p={p_current:.2f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD-PP t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m

def coordinate_descent_peaks(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent focusing on peak reduction.
    Strategy: identify positions contributing most to top peaks,
    transfer mass to positions with lowest peak sensitivity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_k = np.argsort(ac)[::-1][:min(20, len(ac))]

        # Compute peak sensitivity for each position
        peak_sens = np.zeros(n)
        for k in peak_k[:10]:
            k_int = int(k)
            s2 = max(0, k_int - n + 1)
            e2 = min(n, k_int + 1)
            idx2 = np.arange(s2, e2)
            j2 = k_int - idx2
            peak_sens[idx2] += a[j2]

        # Find hot positions (high sensitivity) and cold positions (low sensitivity)
        hot_thresh = peak_sens.max() * 0.7
        cold_thresh = peak_sens.max() * 0.3
        hot_positions = np.where(peak_sens > hot_thresh)[0]
        cold_positions = np.where(peak_sens < cold_thresh)[0]

        if len(hot_positions) == 0 or len(cold_positions) == 0:
            stall += 1
            continue

        improved = False
        # Shuffle for diversity
        rng.shuffle(hot_positions)
        rng.shuffle(cold_positions)

        for src in hot_positions[:30]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            for dst in cold_positions[:20]:
                if dst == src:
                    continue
                if a[dst] > a[src]:
                    continue

                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random escape
            n_perturb = max(5, n // 80)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.06)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
                if improved:
                    break

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordPeak ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=40, use_sym=True):
    """LP-based Frank-Wolfe for final polishing."""
    n = len(a0)
    if n > 2500:
        return amd_exact_active(a0, time_limit, c0=3.0, verbose=verbose, rng_seed=99)

    a = norm_sum1(a0)
    if use_sym:
        a = symmetrize_reverse(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        top_lags = np.argsort(ac)[::-1][:min(40, len(ac))]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]

        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[j] += y[ii]
            constraints.append(coeff @ b_var <= M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        for candidate in [y, x]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def cosine_init(n, alpha=0.5):
    """Generate a cosine-windowed initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** (2 * alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def raised_cosine_init(n, beta=2.0):
    """Generate a raised-cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** beta
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def smooth_random_init(n, rng, smoothness=5):
    """Generate a smooth random initialization."""
    coarse = rng.random(n // smoothness + 1)
    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def bimodal_init(n, separation=0.6):
    """Generate a bimodal (two-bump) initialization."""
    x = np.linspace(0, 1, n)
    center1 = 0.5 * (1 - separation)
    center2 = 0.5 * (1 + separation)
    width = 0.15
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multiple phases and diverse strategies.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = cosine_init(n_base, alpha=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick AMD peak-pressu on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD peak-pressure n={n0} ===")
    t1 = min(remaining * 0.20, 250.0)
    if t1 > 20:
        a_out, _ = amd_peak_pressure(norm_sum1(best_seq), t1, c0=3.0, p_power=3.0,
                                     min_frac=0.5, verbose=True, rng_seed=1, use_sym=True)
        try_update(a_out, f"amd_pp_n{n0}")

    # === Phase 2: AMD exact active (different regime) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD exact-active n={n_best} ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_out, _ = amd_exact_active(norm_sum1(best_seq), t2, c0=4.0, threshold_frac=0.995,
                                    verbose=True, rng_seed=2, use_sym=True)
        try_update(a_out, f"amd_act_n{n_best}")

    # === Phase 3: Diverse initializations at different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Diverse inits ===")
    n_best = len(best_seq)

    # Try a few different lengths and initializations
    candidates_to_try = []
    for n_try in [n_best, min(n_best + 400, 3000), max(n_best - 400, 1600)]:
        if n_try != n_best:
            candidates_to_try.append((n_try, "interp"))
        else:
            candidates_to_try.append((n_try, "parent"))

    # Also try different shapes at the best length
    rng2 = np.random.default_rng(77)
    extra_inits = [
        (cosine_init(n_best, alpha=0.3), "cos03"),
        (cosine_init(n_best, alpha=1.0), "cos10"),
        (raised_cosine_init(n_best, beta=3.0), "rcos"),
        (bimodal_init(n_best, separation=0.5), "bimodal"),
        (smooth_random_init(n_best, rng2, smoothness=8), "srand"),
    ]

    n_total_candidates = len(candidates_to_try) + len(extra_inits)
    time_per_candidate = max(remaining * 0.35 / max(n_total_candidates, 1), 20.0)
    print(f"  {n_total_candidates} candidates, ~{time_per_candidate:.0f}s each")

    # Interpolated/grown versions
    for n_try, kind in candidates_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        if kind == "parent":
            a_init = norm_sum1(best_seq)
        else:
            a_init = interpolate_sequence(best_seq, n_try)
        t_cand = min(time_per_candidate, remaining * 0.3)
        if t_cand > 15:
            a_out, _ = amd_peak_pressure(a_init, t_cand, c0=3.0, p_power=2.5,
                                         min_frac=0.4, verbose=False, rng_seed=3, use_sym=True)
            try_update(a_out, f"amd_pp_n{n_try}_{kind}")

    # Extra initializations at best length
    for a_init, label in extra_inits:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_cand = min(time_per_candidate, remaining * 0.25)
        if t_cand > 15:
            a_out, _ = amd_peak_pressure(a_init, t_cand, c0=3.0, p_power=3.0,
                                         min_frac=0.5, verbose=False, rng_seed=5, use_sym=True)
            try_update(a_out, f"amd_pp_{label}_n{len(a_init)}")

    # === Phase 4: Extended AMD on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Extended AMD n={n_best} ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:

                                     min_frac=0.3, verbose=True, rng_seed=6, use_sym=True)
        try_update(a_out, f"amd_pp_ext_n{n_best}")

    # === Phase 5: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW n={n_best} ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20 and n_best <= 2500:
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver, verbose=True,
                                  max_iters=30, use_sym=True)
        try_update(a_out, f"lp_fw_n{n_best}")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent n={n_best} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 15:
        a_out, _ = coordinate_descent_peaks(norm_sum1(best_seq), t6, verbose=True, rng_seed=7)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 7: Final AMD polish with very tight threshold ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: AMD polish n={n_best} ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_out, _ = amd_exact_active(norm_sum1(best_seq), t7, c0=2.0, threshold_frac=0.999,
                                    verbose=True, rng_seed=8, use_sym=True)
        try_update(a_out, f"amd_polish_n{n_best}")

    # === Phase 8: Another LP pass if time allows ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 2500:
        print(f"\n=== Phase 8: LP final n={n_best} ===")
        t8 = min(remaining * 0.30, 120.0)
        if t8 > 20:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t8, solver, verbose=True,
                                      max_iters=25, use_sym=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final coord n={n_best} ===")
        t9 = min(remaining * 0.40, 100.0)
        if t9 > 10:
            a_out, _ = coordinate_descent_peaks(norm_sum1(best_seq), t9, verbose=False, rng_seed=9)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 10: One more AMD on the absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final AMD n={n_best} ===")
        t10 = min(remaining * 0.50, 150.0)
        if t10 > 15:
            a_out, _ = amd_peak_pressure(norm_sum1(best_seq), t10, c0=2.0, p_power=2.0,
                                         min_frac=0.4, verbose=False, rng_seed=10, use_sym=True)
            try_update(a_out, f"amd_ult_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
