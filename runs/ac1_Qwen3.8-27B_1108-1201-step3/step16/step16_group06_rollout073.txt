_global_solver = None


def get_solver():

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_convolve(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_gradient(a, beta):
    n = len(a)
    ac = fft_autocorr(a)
    scaled = beta * ac
    m = scaled.max()
    exp_vals = np.exp(scaled - m)
    p = exp_vals / exp_vals.sum()
    # grad[i] = (2/beta) * sum_j p[i+j] * a[j]
    # = (2/beta) * correlate(p, a, 'full')[i + n - 1]
    corr = np.correlate(p, a, mode='full')
    grad = corr[n - 1: 2 * n - 1] * (2.0 / beta)
    return grad

        if M_curr < 1e-14:
            break

        import scipy.sparse as sp
        rows = []
        cols = []
        vals = []
        for j in range(n):
            if y[j] > 1e-18:
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                if k_end > k_start:
                    rows.extend(range(k_start, k_end))
                    cols.extend([j] * (k_end - k_start))
                    vals.extend([y[j]] * (k_end - k_start))

        T = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.atleast_1d(b_var.value).flatten(), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        syy = y_new.sum()
        if syy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / syy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fft_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fft_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coord_descent(a_init, time_limit, verbose=True, seed=123):
    n = len(a_init)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_m = fft_convolve(a, a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fft_convolve(a, a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        partners = k_star - idx
        contrib = a[idx] * a[partners]

        if contrib.max() < 1e-20:
            stall += 1
            continue

        top_peaks = np.argsort(ac)[::-1][:min(15, 2 * n - 1)]
        impact = np.zeros(n)
        for pk in top_peaks:
            pk = int(pk)
            ps = max(0, pk - n + 1)
            pe = min(n - 1, pk)
            for pos in range(ps, pe + 1):
                d = pk - pos
                impact[pos] += a[d]

        improved = False
        src_order = idx[np.argsort(contrib)[::-1]][:8]

        for src in src_order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-15:
                continue

            safe_positions = np.argsort(impact)[:max(40, n // 8)]
            found = False
            for dst in safe_positions:
                if dst == src:
                    continue
                for frac in [0.002, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    ct = fft_convolve(a_trial, a_trial)
                    mt = ct.max()
                    if mt < M_curr - 1e-14:
                        a = a_trial
                        M_curr = mt
                        if M_curr < best_m:
                            best_m = M_curr
                            best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    new_m = fft_convolve(a, a).max()
                    if new_m < best_m:
                        best_m = new_m
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDescent ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_asymmetric_init(n, tilt=0.4, shape='linear'):
    x = np.linspace(0, 1, n)
    if shape == 'linear':
        a = 1.0 + tilt * (1.0 - 2.0 * x)
    elif shape == 'quadratic':
        a = (1.0 + tilt * (1.0 - 2.0 * x)) ** 2
    elif shape == 'exp':
        a = np.exp(tilt * (1.0 - 2.0 * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(budget_s=1800):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load error: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score(a_norm)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        best_seq = make_asymmetric_init(n_base, tilt=0.3, shape='linear')
        best_score = eval_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: LSE warmup on parent (60s)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 1: LSE warmup n={n0} ===")
        t1 = min(remaining * 0.05, 60.0)
        a_out, _ = lse_nesterov(norm_sum1(best_seq), t1, beta_start=15.0,
                                beta_end=150.0, verbose=True, gamma=6.0)
        try_update(a_out, "lse1")

    # Phase 2: LP-FW main (500s)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 2: LP-FW main n={n0} ===")
        t2 = min(remaining * 0.35, 500.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 3: Coordinate descent polish (80s)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 3: Coord polish n={n0} ===")
        t3 = min(remaining * 0.06, 80.0)
        a_out, _ = coord_descent(norm_sum1(best_seq), t3, verbose=True, seed=42)
        try_update(a_out, "coord1")

    # Phase 4: LP-FW round 2 (300s)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 4: LP-FW round 2 n={n0} ===")
        t4 = min(remaining * 0.2, 300.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t4, solver, verbose=True)
        try_update(a_out, "lp_r2")

    # Phase 5: Try smaller n (reduce n for more headroom in ratio)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        print(f"\n=== Phase 5: Try smaller n ===")
        for delta in [-200, -400, -100]:
            nt = n_best + delta
            if nt < 600:
                continue
            if time.monotonic() - t_global > time_budget - 100:
                break
            print(f"  Expanding to n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_sub = min(remaining * 0.12, 80.0)
            a_out, _ = lse_nesterov(norm_sum1(a_inter), t_sub, beta_start=30.0,
                                    beta_end=200.0, verbose=False, gamma=7.0)
            try_update(a_out, f"lse_n{nt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60:
                t_sub2 = min(remaining * 0.12, 80.0)
                a_out, _ = lp_frank_wolfe(norm_sum1(a_inter), t_sub2, solver, verbose=False)
                try_update(a_out, f"lp_n{nt}")

    # Phase 6: Try larger n
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 6: Try larger n ===")
        for delta in [200, 400]:
            nt = n_best + delta
            if nt > 4000:
                continue
            if time.monotonic() - t_global > time_budget - 80:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_sub = min(remaining * 0.1, 70.0)
            a_out, _ = lse_nesterov(norm_sum1(a_inter), t_sub, beta_start=30.0,
                                    beta_end=250.0, verbose=False, gamma=7.0)
            try_update(a_out, f"lse_n{nt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60:
                t_sub2 = min(remaining * 0.1, 70.0)
                a_out, _ = lp_frank_wolfe(norm_sum1(a_inter), t_sub2, solver, verbose=False)
                try_update(a_out, f"lp_n{nt}")

    # Phase 7: Fresh asymmetric init at best n
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        print(f"\n=== Phase 7: Fresh asymmetric starts ===")
        n_best = len(best_seq)
        for tilt, shape in [(0.6, 'linear'), (0.8, 'exp'), (0.5, 'quadratic')]:
            if time.monotonic() - t_global > time_budget - 100:
                break
            a_init = make_asymmetric_init(n_best, tilt=tilt, shape=shape)
            # Blend with current best
            a_blend = 0.5 * norm_sum1(best_seq) + 0.5 * a_init
            a_blend = norm_sum1(a_blend)
            t_sub = min(remaining * 0.1, 60.0)
            a_out, _ = lse_nesterov(a_blend, t_sub, beta_start=20.0,
                                    beta_end=300.0, verbose=False, gamma=8.0,
                                    seed=99)
            try_update(a_out, f"fresh_{shape}_t{tilt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t_sub2 = min(remaining * 0.08, 50.0)
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t_sub2, solver, verbose=False)
                try_update(a_out, f"lp_fresh_{shape}")

    # Phase 8: Final LSE with high beta
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Final LSE high-beta ===")
        n_best = len(best_seq)
        t8 = min(remaining * 0.2, 120.0)
        a_out, _ = lse_nesterov(norm_sum1(best_seq), t8, beta_start=100.0,
                                beta_end=800.0, verbose=True, gamma=10.0, seed=77)
        try_update(a_out, "lse_final")

    # Phase 9: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 9: Final LP ===")
        n_best = len(best_seq)
        t9 = min(remaining * 0.4, 200.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t9, solver, verbose=True)
        try_update(a_out, "lp_final")

    # Phase 10: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 10: Final coord ===")
        n_best = len(best_seq)
        t10 = min(remaining * 0.5, 100.0)
        a_out, _ = coord_descent(norm_sum1(best_seq), t10, verbose=True, seed=314)
        try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
