def norm_sum1(a):

    if s < 1e-15:
        return np.ones_like(a) / len(a)

def compute_smoothed_weights(c, beta, top_k=300):
    """Compute softmax weights for top-k lags."""
    n_lags = len(c)
    k_eff = min(top_k, n_lags)
    top_idx = np.argsort(c)[::-1][:k_eff]
    top_vals = c[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    return top_idx, weights


def compute_gradient(a, top_idx, weights):
    """Compute gradient of sum_k w_k * c[k] where c = conv(a,a)."""
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):

        if wi < 1e-15:

            grad[0] += 2.0 * wi * a[0]

        else:
            # k_int in [1, n-1]
            grad[:n - k_int] += wi * a[k_int:]
            grad[k_int:] += wi * a[:n - k_int]
    return grad

        # Step size
        eta = gamma / (np.sqrt(t) + 1.0)

        # Exponentiated update (mirror descent on simplex)
        u = np.log(a + 1e-30) - eta * grad
        u = u - u.max()  # numerical stability
        a_new = np.exp(u)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Periodic restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            # Structured perturbation
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            a = x_re
            current_top_k = min(current_top_k + 30, 400)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      EG restart t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, top_k={current_top_k}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      EG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m


def projected_gradient(a0, time_limit, beta_start=25.0, beta_end=5.0,
                       top_k=200, c0=1.5, verbose=True, seed=42):
    """Nesterov-accelerated projected gradient on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    current_top_k = top_k

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 20000.0)
        cur_beta = beta_start + (beta_end - beta_start) * (frac ** 0.6)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_autocorr(y)
        top_idx, weights = compute_smoothed_weights(c, cur_beta, top_k=current_top_k)
        grad = compute_gradient(y, top_idx, weights)

        eta = gamma / (np.sqrt(t) + 1.0)
        x_new = y - eta * grad
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            noise = rng.normal(0, 0.004, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            current_top_k = min(current_top_k + 20, 350)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      PG restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m


def peak_directed_polish(a0, time_limit, verbose=True, seed=42):
    """Directly target the peak lags by redistributing mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M_curr = c.max()
        # Get top lags
        top_lags = np.argsort(c)[::-1][:15]
        improved = False

        # For each peak lag, find the pair contributing most
        for k_int in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_int)
            if k_int >= n:
                k_eff = 2 * n - 2 - k_int
            else:
                k_eff = k_int

            if k_eff == 0:
                continue

            # Find contributing pairs for lag k_eff
            i_start = max(0, 0)
            i_end = min(n - k_eff, n)
            if i_end <= i_start:
                continue

            indices = np.arange(i_start, i_end)
            products = a[indices] * a[indices + k_eff]
            if len(products) == 0:
                continue

            # Top contributing pairs
            top_pairs = np.argsort(products)[::-1][:20]
            for tp in top_pairs[:10]:
                src = indices[tp]
                dst = indices[tp] + k_eff
                if a[src] < 1e-10 or a[dst] < 1e-10:
                    continue

                # Try reducing the product by moving mass
                # Option 1: reduce src, add to a low-impact position
                # Score positions by their impact on peak lags
                peak_lags_set = set(int(x) for x in top_lags[:6])
                best_score = -np.inf
                best_pos = -1
                n_try = min(n, 300)
                candidates = rng.choice(n, n_try, replace=False) if n > n_try else np.arange(n)

                for pos in candidates:
                    if pos == src or pos == dst:
                        continue
                    # Score: how much does adding to pos increase peak lags?
                    impact = 0.0
                    for pl in peak_lags_set:
                        pl_int = int(pl)
                        if pl_int >= n:
                            pl_eff = 2 * n - 2 - pl_int
                        else:
                            pl_eff = pl_int
                        if pl_eff == 0:
                            if pos == 0:
                                impact += a[pos]
                        else:
                            if pos >= pl_eff and pos - pl_eff < n:
                                impact += a[pos - pl_eff]
                            if pos + pl_eff < n:
                                impact += a[pos + pl_eff]
                    score = -impact + 0.1 * a[pos]  # prefer positions with some mass
                    if score > best_score:
                        best_score = score
                        best_pos = pos

                if best_pos < 0:
                    continue

                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_pos] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PeakPolish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      PeakPolish done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_polish(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration (limited iterations)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Build Toeplitz matrix once
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, min(j + n, 2 * n - 1))
        T[rows, j] = a[:len(rows)]

    y = a.copy()
    max_iters = 500

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(y)
        M_curr = c.max()
        if M_curr < 1e-14:
            break

        b_var = cp.Variable(n, nonneg=True)
        # Rebuild T with current y
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            rows = np.arange(j, min(j + n, 2 * n - 1))
            T[rows, j] = y[:len(rows)]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-16:
            best_m = mc
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def structured_init(n, kind=0):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1  # centered in [-1, 1]
    if kind == 0:
        # Sin^2
        f = np.sin(np.pi * x) ** 2
    elif kind == 1:
        # Gaussian center
        f = np.exp(-8 * t ** 2)
    elif kind == 2:
        # Bimodal
        f = np.exp(-25 * (t + 0.3) ** 2) + np.exp(-25 * (t - 0.3) ** 2)
    elif kind == 3:
        # Quartic
        f = np.maximum(1 - t ** 2, 0) ** 2
    elif kind == 4:
        # Asymmetric bump
        f = np.exp(-6 * (t - 0.2) ** 2 - 2 * t)
    elif kind == 5:
        # Triangle
        f = np.maximum(1 - np.abs(t), 0)
    else:
        f = np.ones(n)
    f = np.maximum(f, 0)
    s = f.sum()
    return f / s if s > 1e-15 else np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = get_parent_sequence()
    n0 = 2400
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
        n0 = len(parent)
    else:
        best_seq = structured_init(n0, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            improvement = best_score - sc
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, improve={improvement:.6f}, t={elapsed:.0f}s)")

    # =========================================================
    # Phase 1: Quick multi-n exploration (150s)
    # =========================================================
    print(f"\n=== Phase 1: Multi-n exploration ===")
    t1_total = min(150.0, remaining_time() * 0.1)
    n_candidates = [max(n0 - 400, 1600), n0 - 200, n0, n0 + 200, n0 + 400]
    t_per_n = t1_total / len(n_candidates)

    for nc in n_candidates:
        rem = remaining_time()
        if rem < 20:
            break
        t_alloc = min(t_per_n, rem - 10)
        if t_alloc < 10:
            continue

        if nc == len(best_seq):
            a_init = best_seq.copy()
        else:
            a_init = interpolate_sequence(best_seq, nc)

        a_out, _ = exponentiated_gradient(a_init, t_alloc,
                                           beta_start=25.0, beta_end=8.0,
                                           top_k=150, c0=1.0, verbose=False, seed=nc)
        try_update(a_out, f"explore_n{nc}")

    # =========================================================
    # Phase 2: Main EG optimization at best n (500s)
    # =========================================================
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Main EG n={n_best} ===")
    t2_total = min(500.0, remaining_time() * 0.32)

    # Run 1: Standard EG
    t2a = t2_total * 0.5
    if t2a > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = exponentiated_gradient(a_in, t2a,
                                           beta_start=35.0, beta_end=4.0,
                                           top_k=250, c0=1.2, verbose=True, seed=42)
        try_update(a_out, "eg_main1")

    # Run 2: PG (different dynamics)
    rem = remaining_time()
    t2b = min(t2_total * 0.35, rem * 0.4)
    if t2b > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = projected_gradient(a_in, t2b,
                                       beta_start=30.0, beta_end=5.0,
                                       top_k=250, c0=1.8, verbose=True, seed=99)
        try_update(a_out, "pg_main")

    # Run 3: EG with different params
    rem = remaining_time()
    t2c = min(t2_total * 0.3, rem * 0.3)
    if t2c > 20:
        a_in = norm_sum1(best_seq)
        # Blend with a structured init for diversity
        blend = 0.7 * a_in + 0.3 * structured_init(n_best, 4)
        a_out, _ = exponentiated_gradient(blend, t2c,
                                           beta_start=20.0, beta_end=6.0,
                                           top_k=200, c0=0.8, verbose=False, seed=77)
        try_update(a_out, "eg_blend")

    # =========================================================
    # Phase 3: Peak-directed polish (300s)
    # =========================================================
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak polish n={n_best} ===")
    t3_total = min(300.0, remaining_time() * 0.2)

    t3a = t3_total * 0.6
    if t3a > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_directed_polish(a_in, t3a, verbose=True, seed=42)
        try_update(a_out, "peak1")

    # EG after peak polish
    rem = remaining_time()
    t3b = min(t3_total * 0.4, rem * 0.3)
    if t3b > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = exponentiated_gradient(a_in, t3b,
                                           beta_start=50.0, beta_end=15.0,
                                           top_k=300, c0=0.5, verbose=False, seed=123)
        try_update(a_out, "eg_afterpeak")

    # =========================================================
    # Phase 4: LP polish (200s)
    # =========================================================
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP polish n={n_best} ===")
    if n_best <= 3000:
        t4_total = min(200.0, remaining_time() * 0.15)
        if t4_total > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_polish(a_in, t4_total, solver, verbose=True)
            try_update(a_out, f"lp_n{n_best}")

    # =========================================================
    # Phase 5: Diverse restarts (250s)
    # =========================================================
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Diverse restarts n={n_best} ===")
    t5_total = min(250.0, remaining_time() * 0.2)
    t_per_restart = t5_total / 6

    inits = []
    # Blend parent with structured inits
    for kind in range(6):
        s_init = structured_init(n_best, kind)
        blend = 0.6 * norm_sum1(best_seq) + 0.4 * s_init
        inits.append(norm_sum1(blend))

    # Also try pure structured inits
    for kind in [0, 4]:
        inits.append(structured_init(n_best, kind))

    for idx, init in enumerate(inits[:6]):
        rem = remaining_time()
        if rem < 20:
            break
        t_alloc = min(t_per_restart, rem - 10)
        if t_alloc < 10:
            continue
        a_out, _ = exponentiated_gradient(init, t_alloc,
                                           beta_start=25.0, beta_end=6.0,
                                           top_k=200, c0=1.0, verbose=False, seed=200 + idx)
        try_update(a_out, f"restart{idx}")

    # =========================================================
    # Phase 6: Final multi-scale refinement (200s)
    # =========================================================
    print(f"\n=== Phase 6: Final refinement ===")
    t6_total = min(200.0, remaining_time() * 0.15)

    # Try interpolating to a slightly different length and optimizing
    n_alt = n_best + 100
    if n_alt != n_best and t6_total > 40:
        a_alt = interpolate_sequence(best_seq, n_alt)
        t6a = t6_total * 0.5
        a_out, _ = exponentiated_gradient(a_alt, t6a,
                                           beta_start=20.0, beta_end=5.0,
                                           top_k=200, c0=1.0, verbose=False, seed=500)
        try_update(a_out, f"alt_n{n_alt}")

    # Final peak polish at best
    rem = remaining_time()
    t6b = min(t6_total * 0.5, rem * 0.4)
    if t6b > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_directed_polish(a_in, t6b, verbose=False, seed=999)
        try_update(a_out, "final_peak")

    # =========================================================
    # Phase 7: Ultra-fine EG (remaining time)
    # =========================================================
    rem = remaining_time()
    print(f"\n=== Phase 7: Ultra-fine (remaining {rem:.0f}s) ===")
    if rem > 30:
        t7 = rem * 0.8
        a_in = norm_sum1(best_seq)
        a_out, _ = exponentiated_gradient(a_in, t7,
                                           beta_start=60.0, beta_end=20.0,
                                           top_k=400, c0=0.3, verbose=True, seed=314)
        try_update(a_out, "ultra_fine")

    # =========================================================
    # Final validation
    # =========================================================
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
