    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_full_gradient(a, ac, mu):
    """Full log-sum-exp gradient: 2 * sum_j w[i+j] * a[j] via FFT."""
    n = len(a)
    ac_shifted = (ac - ac.max()) / mu
    w = np.exp(ac_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / w_sum
    a_rev = a[::-1]
    N = 1 << (3 * n - 1).bit_length()
    Fa = np.fft.rfft(a_rev, N)
    Fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(Fa * Fw, N)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_from_ac(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def nesterov_mirror_descent(a0, time_limit, mu_start=0.5, mu_min=0.005,
                            mu_ratio=0.6, c_smooth=5.0, restart_every=4000,
                            verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with full gradient and mu-continuation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_ac_max = fast_autocorr(a).max()
    best_score = 2.0 * n * best_ac_max
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    mu = mu_start
    stall = 0
    total_iter = 0

    while time.monotonic() - t0 < time_limit:
        block_end = min(total_iter + 600, restart_every)
        while total_iter < block_end and time.monotonic() - t0 < time_limit:
            t = total_iter + 1
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            ac = fast_autocorr(y)
            mu_eff = max(mu, ac.max() * 0.01)
            grad = compute_full_gradient(y, ac, mu_eff)

            L = 4.0 / mu_eff + c_smooth
            eta = 1.0 / (np.sqrt(t) * L)
            log_a = np.log(y + 1e-15) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            x_prev = x
            x = a_new
            total_iter += 1

        # Check progress and decide on mu update / restart
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_ac_max - 1e-12:
            best_ac_max = M_x
            best_a = x.copy()
            best_score = 2.0 * n * best_ac_max
            stall = 0
        else:
            stall += 1

        if stall < 8 and mu > mu_min:
            mu *= mu_ratio
        elif stall >= restart_every or stall >= 25:
            # Restart with perturbation
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            total_iter = 0
            stall = 0
            mu = max(mu_start, mu * 2.0)
            if verbose:
                print(f"    NMD restart: score={best_score:.6f}, mu={mu:.4f}")

        if verbose and total_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD iter={total_iter} ({elapsed:.0f}s): score={best_score:.6f}, mu={mu:.4f}, stall={stall}")

    return best_a, best_ac_max


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=250):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr + 1e-8]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            b_val = y.copy()
            b_sum = 1.0
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            best_score = 2.0 * n * best_m

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            best_score = 2.0 * n * best_m

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s, {[f'{x:.4f}' for x in (time.monotonic()-t0, best_score)])}: score={best_score:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with sensitivity-based moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute penalty for each destination
            n_sens = min(15, n_top)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:n_sens]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            scores = np.zeros(n)
            for k2 in top_k[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    d2 = d - k2i
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        scores[d] += a[d2]

            candidate_scores = scores[candidates]
            if len(candidate_scores) > 0:
                best_dst = candidates[np.argmin(candidate_scores)]
            else:
                best_dst = int(rng.integers(n))

            if best_dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.08, 0.15]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    best_score = 2.0 * n * best_m
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        best_score = 2.0 * n * best_m
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            best_score = 2.0 * n * best_m
                            stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): score={best_score:.6f}, stall={stall}")

    if verbose:
        print(f"    Coord final: score={best_score:.6f}")
    return best_a, best_m


def make_linear_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_bimodal(n, center_sep=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.1) ** 2) + np.exp(-50 * (x - 0.9) ** 2) + 0.01
    return norm_sum1(a)


def make_sine_squared(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 6, 7]:
        coeff = rng.normal(0, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_gaussian_bump(n, center=0.5, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-(x - center) ** 2 / (2 * width ** 2))
    return norm_sum1(a)
