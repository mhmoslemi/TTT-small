```python
"""
Minimize score = 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

Approach:
- Keep sequences normalized to sum 1, so scaling/clipping at [0, 1000] is inactive.
- Use a smooth log-sum-exp approximation to the maximum autocorrelation peak.
- Optimize in log-domain via softmax so the simplex constraint and nonnegativity
  are handled implicitly.
- Use L-BFGS on the smooth surrogate with tau annealing, then a final projected
  multiplicative balancing/line-search polish.
- Try the parent sequence at a few nearby resampled lengths and keep the exact best.
"""

import math
import time

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def next_pow2(m):
    m = int(max(1, m))
    return 1 << int((m - 1).bit_length())


def normalize_one(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def autocorr_fft(a, fft_size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if fft_size is None:
        fft_size = next_pow2(2 * n - 1)
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)
    return c[:2 * n - 1]


def score_normalized(a, fft_a=None):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    n = a.size
    if fft_a is None:
        fft_a = next_pow2(2 * n - 1)
    c = autocorr_fft(a, fft_a)
    mb = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mb) or mb < 0.0:
        mb = 0.0
    return float(2.0 * n * mb / (s * s))


def fast_score(a):
    return score_normalized(a)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if float(a.sum()) < 0.01:
        return float("inf")
    try:
        val = float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float("inf")
    if not np.isfinite(val):
        return float("inf")
    return val


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return None
    if not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=np.float64)
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        z = np.zeros(n, dtype=np.float64)
    z = z - zmax
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones(n, dtype=np.float64)
        s = float(n)
    return e / s


def init_z_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-300)
    z = np.log(a)
    z = z - float(np.mean(z))
    return z


def smooth_objective_grad(z, n, fft_a, fft_g, tau):
    z = np.asarray(z, dtype=np.float64)
    if z.size != n:
        z = np.zeros(n, dtype=np.float64)
    a = softmax_from_z(z)
    if a is None:
        return float("inf"), np.zeros(n, dtype=np.float64)

    if tau <= 0.0 or not np.isfinite(tau):
        tau = 1e-18

    c = autocorr_fft(a, fft_a)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M < 0.0:
        M = 0.0

    v = (c - M) / tau
    v = np.clip(v, -745.0, 0.0)
    ev = np.exp(v)
    S = float(np.sum(ev))
    if not np.isfinite(S) or S <= 0.0:
        S = 1.0
        ev = np.zeros_like(c)
        if c.size:
            ev[int(np.argmax(c))] = 1.0

    F = M + tau * math.log(S)
    score = 2.0 * n * F

    w = ev / S
    A = rfft(a, fft_g)
    W = rfft(w, fft_g)
    H = irfft(np.conj(W) * A, fft_g)
    Ga = 2.0 * H[:n]

    dot = float(np.dot(a, Ga))
    Gz = a * (Ga - dot)
    Gz *= 2.0 * n

    if not np.all(np.isfinite(Gz)):
        Gz = np.zeros(n, dtype=np.float64)

    return float(score), Gz


def run_lbgfs(z, n, fft_a, fft_g, tau, maxiter):
    options = {
        "maxiter": maxiter,
        "maxfun": max(1, 3 * maxiter),
        "ftol": 0.0,
        "gtol": 1e-9,
        "maxcor": 10,
        "maxls": 20,
    }
    res = minimize(
        smooth_objective_grad,
        z,
        method="L-BFGS-B",
        jac=True,
        args=(n, fft_a, fft_g, tau),
        options=options,
    )
    if res is None or getattr(res, "x", None) is None:
        return z
    return np.asarray(res.x, dtype=np.float64)


def perturb_sequence(a, sigma, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return normalize_one(a)
    noise = rng.standard_normal(n)
    F = np.fft.rfft(noise)
    f = np.fft.rfftfreq(n)
    cutoff = max(1.0 / n, 8.0 / n)
    F *= np.exp(-((f / cutoff) ** 2))
    F[0] = 0.0
    noise = np.fft.irfft(F, n)
    out = a * np.exp(sigma * noise)
    return normalize_one(out)


def balance_line_search(a, tau, tvals, fft_a, fft_g):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = autocorr_fft(a, fft_a)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return a

    v = (c - M) / tau
    v = np.clip(v, -745.0, 0.0)
    ev = np.exp(v)
    S = float(np.sum(ev))
    if S <= 0.0 or not np.isfinite(S):
        return a
    w = ev / S

    A = rfft(a, fft_g)
    W = rfft(w, fft_g)
    H = irfft(np.conj(W) * A, fft_g)
    Ga = 2.0 * H[:n]

    D = Ga - float(np.mean(Ga))
    scale = float(np.std(D))
    if not np.isfinite(scale) or scale < 1e-30:
        return a

    best = a.copy()
    best_sc = score_normalized(a, fft_a)

    for t in tvals:
        cand = a * np.exp(-float(t) * D / scale)
        cand = normalize_one(cand)
        sc = score_normalized(cand, fft_a)
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best = cand
            best_sc = sc
    return best


def balance_optimize(a0, time_limit, verbose=False):
    a = normalize_one(a0)
    n = a.size
    fft_a = next_pow2(2 * n - 1)
    fft_g = next_pow2(3 * n - 2)
    t0 = time.monotonic()

    best = a.copy()
    best_score = score_normalized(best, fft_a)
    current = a.copy()
    current_score = best_score

    multipliers = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]

    for mult in multipliers:
        if time.monotonic() - t0 > time_limit:
            break
        M = max(best_score / (2.0 * n), 1e-12)
        tau = max(M * mult, 1e-18)

        tvals = [1.0, 0.6, 0.35, 0.2, 0.1, 0.05]
        while time.monotonic() - t0 < max(2.0, time_limit - 1.0):
            cand = balance_line_search(current, tau, tvals, fft_a, fft_g)
            sc = score_normalized(cand, fft_a)
            if np.isfinite(sc) and sc < current_score - 1e-13:
                current = cand
                current_score = sc
                if sc < best_score - 1e-14:
                    best = cand.copy()
                    best_score = sc
                    if verbose:
                        print("      balance n={} score={:.10f} tau_mult={:.3f}".format(n, best_score, mult))
                tvals = [0.5, 0.3, 0.18, 0.1, 0.05, 0.025]
            else:
                break

    return best, best_score


def smooth_optimize_length(a0, time_limit, verbose=True, seed=0):
    a = normalize_one(a0)
    n = a.size
    if n < 5:
        return a, fast_score(a)

    fft_a = next_pow2(2 * n - 1)
    fft_g = next_pow2(3 * n - 2)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    best = a.copy()
    best_score = fast_score(best)

    lb_limit = max(5.0, time_limit * 0.68)

    if n <= 900:
        base_iter = 160
    elif n <= 1600:
        base_iter = 105
    elif n <= 2400:
        base_iter = 70
    else:
        base_iter = 45

    multipliers = [
        2.0,
        1.0,
        0.5,
        0.25,
        0.125,
        0.0625,
        0.03125,
        0.015625,
        0.0078125,
    ]

    for restart in range(4):
        if time.monotonic() - t0 > lb_limit:
            break

        if restart == 0:
            current = best.copy()
            current_score = best_score
        else:
            current = perturb_sequence(best, 0.10 * (restart + 1), rng)
            current_score = fast_score(current)
            if current_score < best_score - 1e-12:
                best = current.copy()
                best_score = current_score
                if verbose:
                    print("      restart perturbation n={} score={:.10f}".format(n, best_score))

        no_improve = 0
        for mult in multipliers:
            if time.monotonic() - t0 > lb_limit:
                break

            M = max(best_score / (2.0 * n), 1e-12)
            tau = max(M * mult, 1e-18)
            z = init_z_from_a(current)

            it = base_iter
            if mult < 0.1:
                it = int(it * 1.4)
            # avoid absurdly long single calls when little lb time remains
            left_lb = lb_limit - (time.monotonic() - t0)
            if left_lb < 60.0:
                it = min(it, max(20, int(left_lb / 2.0)))

            try:
                znew = run_lbgfs(z, n, fft_a, fft_g, tau, max(5, it))
                out = softmax_from_z(znew)
                if out is None:
                    out = current
                out = normalize_one(out)
                sc = fast_score(out)

                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best = out.copy()
                    best_score = sc
                    current = out.copy()
                    current_score = sc
                    no_improve = 0
                    if verbose:
                        print("      lbgfs n={} tau_mult={:.4f} score={:.10f}".format(n, mult, best_score))
                else:
                    no_improve += 1
                    if np.isfinite(sc) and sc < current_score - 1e-12:
                        current = out.copy()
                        current_score = sc

                if no_improve >= 3 and restart < 3:
                    break

            except Exception as exc:
                no_improve += 1
                if verbose:
                    print("      lbgfs failed: {}".format(exc))
                if no_improve >= 3:
                    break

    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30.0:
        bal_seq, bal_score = balance_optimize(best, min(remaining, 350.0), verbose=verbose)
        if np.isfinite(bal_score) and bal_score < best_score - 1e-13:
            best = bal_seq.copy()
            best_score = bal_score
            if verbose:
                print("      balance polish n={} score={:.10f}".format(n, best_score))

    return best, best_score


def resample_sequence(seq, n):
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    if m == n:
        return normalize_one(seq)
    x = np.linspace(0.0, m - 1.0, m)
    y = np.linspace(0.0, n - 1.0, n)
    out = np.interp(y, x, seq)
    out = np.maximum(out, 0.0)
    return normalize_one(out)


def candidate_sort_key(item):
    return item[0]
```
