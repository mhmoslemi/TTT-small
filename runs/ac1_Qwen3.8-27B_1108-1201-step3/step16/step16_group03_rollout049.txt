def smooth_objective_and_grad(a, p):
    """
    Compute the smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Returns (value, grad_a) where grad_a has the same shape as a.
    """
    n = len(a)
    m = 2 * n - 1
    ac = fast_conv(a)
    ac_pos = np.maximum(ac, 1e-30)
    
    # S = (1/m) * sum(ac^p)
    ac_p = ac_pos ** p
    S = (np.sum(ac_p) / m) ** (1.0 / p)  # This is the smooth max value
    S_inner = np.sum(ac_p) / m  # = S^p
    
    # w = ac^(p-1)
    w = ac_pos ** (p - 1)
    
    # corr[i] = sum_j a[j] * w[j+i] (cross-correlation)
    # Using FFT: corr = ifft(FFT(a) * conj(FFT(w)))
    N = 1 << (m + n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(A * np.conj(W), N)[:n]
    
    # grad_a[i] = (2/m) * S_inner^(1/p - 1) * corr[i]
    # Note: S_inner = S^p, so S_inner^(1/p-1) = S^(p*(1/p-1)) = S^(1-p)
    C = (2.0 / m) * S ** (1 - p)
    grad = C * corr
    
    return S, grad


def smooth_objective_and_grad_theta(a, p):
    """
    Compute gradient in the theta (log) parameterization.
    a = softmax(theta), so grad_theta[i] = a[i] * (grad_a[i] - dot(grad_a, a))
    """
    val, grad_a = smooth_objective_and_grad(a, p)
    mean_grad = np.dot(grad_a, a)
    grad_theta = a * (grad_a - mean_grad)
    return val, grad_theta

            grad += wi * hard_subgradient(y, k)
        
        eta = c0 / math.sqrt(t + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            x_r = best_a + rng.normal(0, 0.003, n) * best_a
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            stall = 0
        
        if t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      HardSG t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with sparse constraint matrix."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    if solver is None:
        return best_a, best_m
    
    y = a.copy()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build sparse constraint matrix: (T @ b)[k] = sum_j T[k,j] * b[j]
        # T[k,j] = y[k-j] if 0 <= k-j < n, else 0
        if HAS_SCIPY:
            rows = []
            cols = []
            vals = []
            for j in range(n):
                k_start = j
                k_end = j + n
                valid_ks = np.arange(k_start, min(k_end, 2 * n - 1))
                for k in valid_ks:
                    rows.append(k)
                    cols.append(j)
                    vals.append(y[k - j])
            T = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        else:
            T = np.zeros((2 * n - 1, n))
            for j in range(n):
                for k in range(j, min(j + n, 2 * n - 1)):
                    T[k, j] = y[k - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Line search
        alphas = [2.0 / (t + 2.0), 0.5, 0.3, 0.2, 0.1, 0.05]
        best_alpha = alphas[0]
        best_trial_m = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()
        
        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            tm = fast_conv(trial).max()
            if tm < best_trial_m:
                best_trial_m = tm
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def directed_peak_shaving(a0, time_limit, seed=42):
    """Directly target top peaks by redistributing mass from contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:8]
        improved = False
        
        for k_int in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k_int)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue
            
            # Find the position that contributes most to this peak
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find destination: position that minimizes impact on all top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                # Estimate how much moving mass to dst increases other peaks
                penalty = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    if 0 <= k2i - dst < n:
                        penalty += a[k2i - dst]
                    if 0 <= dst - k2i < n:
                        penalty += a[dst - k2i]
                if penalty < best_score:
                    best_score = penalty
                    best_dst = dst
            
            if best_dst < 0:
                continue
            dst = best_dst
            
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = src_val * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
    
    return best_a, best_m


def generate_parametric_init(n, seed=0):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    inits = []
    
    # U-shaped (cosine)
    c = rng.uniform(0.1, 0.5)
    a = 1.0 - c * np.sin(np.pi * x) ** 2
    inits.append(np.maximum(a, 0.01))
    
    # Edge-concentrated
    e = rng.uniform(0.3, 0.7)
    a = np.exp(-e * (2 * x - 1) ** 2)
    inits.append(a)
    
    # Oscillatory
    f = rng.uniform(2, 8)
    amp = rng.uniform(0.1, 0.4)
    a = 1.0 + amp * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    inits.append(np.maximum(a, 0.01))
    
    # Double peak
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.05, 0.2)
    a = np.exp(-((x - 0.2) / width) ** 2) + np.exp(-((x - 0.8) / width) ** 2)
    inits.append(a)
    
    # Random smooth
    freqs = rng.choice(np.arange(1, 10), size=5, replace=False)
    amps = rng.uniform(0.1, 0.3, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    a = np.ones(n)
    for i, (f, amp, ph) in enumerate(zip(freqs, amps, phases)):
        a += amp * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(a, 0.01))
    
    # Sin^2 (similar to previous best structure)
    a = np.sin(np.pi * x) ** 2
    inits.append(a)
    
    # Truncated exponential
    rate = rng.uniform(1, 5)
    a = np.exp(-rate * np.abs(2 * x - 1))
    inits.append(a)
    
    return [norm_sum1(a) for a in inits]


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    total_time = 1800.0
    time_budget = total_time - 50.0  # leave 50s buffer
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p.copy()
            p_norm = parent / parent.sum()
            p_score = evaluate_safe(p_norm)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = p_norm
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Smooth gradient with progressive p ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth gradient (n={n0}) ===")
    
    p_schedule = [5, 10, 20, 50]
    for p_val in p_schedule:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        alloc = min(remaining * 0.25, 250.0)
        if alloc < 20:
            break
        print(f"  p={p_val}, time={alloc:.0f}s")
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_md(a_in, p_val, alloc, lr_init=3.0, name=f"smooth_p{p_val}")
        try_update(a_out, f"smooth_p{p_val}_n{n0}")
    
    # === Phase 2: Hard subgradient Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Hard subgradient (n={n_cur}) ===")
    
    for seed_val in [42, 7, 123]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        alloc = min(remaining * 0.2, 300.0)
        if alloc < 30:
            break
        print(f"  seed={seed_val}, time={alloc:.0f}s")
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_hard_subgrad(a_in, alloc, c0=5.0, top_k=1, seed=seed_val)
        try_update(a_out, f"hard_n{n_cur}_s{seed_val}")
    
    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n_cur}) ===")
    if n_cur <= 3000 and remaining > 60:
        alloc = min(remaining * 0.15, 200.0)
        if alloc > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, alloc, solver, verbose=True)
            try_update(a_out, f"lp_n{n_cur}")
    
    # === Phase 4: Directed peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Peak shaving (n={n_cur}) ===")
    if remaining > 30:
        alloc = min(remaining * 0.15, 200.0)
        if alloc > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = directed_peak_shaving(a_in, alloc, seed=42)
            try_update(a_out, f"shave_n{n_cur}")
    
    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length exploration ===")
    
    for delta in [-400, -200, 200, 400]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        n_try = n_best + delta
        if n_try < 800 or n_try > 5000:
            continue
        print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
        a_interp = interpolate_sequence(best_seq, n_try)
        
        # Smooth gradient on new length
        alloc = min(remaining * 0.3, 150.0)
        if alloc > 20:
            a_out, _ = nesterov_smooth_md(a_interp, 20, alloc, lr_init=3.0, name=f"n{n_try}_smooth")
            try_update(a_out, f"smooth_n{n_try}")
        
        # Hard subgrad
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            continue
        alloc = min(remaining * 0.2, 100.0)
        if alloc > 15:
            a_in = norm_sum1(best_seq)
            a_long = interpolate_sequence(a_in, n_try)
            a_out, _ = nesterov_hard_subgrad(a_long, alloc, c0=4.0, top_k=1, seed=55 + abs(delta))
            try_update(a_out, f"hard_n{n_try}")
    
    # === Phase 6: Parametric initializations + quick optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Parametric diverse inits (n={n_best}) ===")
    if remaining > 100:
        inits = generate_parametric_init(n_best, seed=99)
        for i, init in enumerate(inits):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            alloc = min(remaining * 0.15, 100.0)
            if alloc < 15:
                break
            a_out, _ = nesterov_smooth_md(init, 10, alloc, lr_init=3.0, name=f"param{i}")
            try_update(a_out, f"param{i}_n{n_best}")
    
    # === Phase 7: Final exact subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 40:
        alloc = min(remaining * 0.5, 300.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_hard_subgrad(a_in, alloc, c0=6.0, top_k=2, seed=314)
        try_update(a_out, f"final_hard_n{n_best}")
    
    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final shaving (n={n_best}) ===")
    if remaining > 30:
        alloc = min(remaining * 0.7, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = directed_peak_shaving(a_in, alloc, seed=777)
        try_update(a_out, f"final_shave_n{n_best}")
    
    # === Phase 9: One more LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 60:
        print(f"\n=== Phase 9: Final LP (n={n_best}) ===")
        alloc = min(remaining * 0.6, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, alloc, solver, verbose=True)
        try_update(a_out, f"final_lp_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
