2. Adaptive epsilon schedule for progressive peak sharpening
3. Multi-start with diverse initializations
4. Multi-length exploration with interpolation
5. Final coordinate descent polishing

import numpy as np
import time
from scipy.optimize import minimize

_EPS = 1e-4  # global epsilon for log-sum-exp smoothing

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Evaluate using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_from_theta(theta):
    """Convert theta to probability simplex via softmax."""
    theta_max = theta.max()
    a = np.exp(theta - theta_max)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return a / s

def _compute_grad_a(a, p):
    """Compute grad_a[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] via FFT."""
    n = len(a)
    flip_a = a[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    F_flip_a = np.fft.rfft(flip_a, L)
    F_p = np.fft.rfft(p, L)
    C = np.fft.irfft(F_flip_a * F_p, L)
    return 2.0 * C[n-1 : 2*n-1]

def _obj_func(theta):
    """Objective: log-sum-exp of convolution at global _EPS."""
    a = softmax_from_theta(theta)
    n = len(a)
    ac = fast_conv(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / _EPS
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    f_val = _EPS * (np.log(exp_s.sum()) + ac_max / _EPS)
    return f_val

def _grad_func(theta):
    """Gradient of log-sum-exp smoothed objective w.r.t. theta."""
    a = softmax_from_theta(theta)
    n = len(a)
    ac = fast_conv(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / _EPS
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    grad_a = _compute_grad_a(a, p)
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)
    return grad_theta

    global _EPS
    _EPS = epsilon
    result = minimize(

        x0=theta0,
        jac=_grad_func,
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 50}
    )
    return result.x, result.fun, result.nit

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with softmax parametrization
2. Adaptive epsilon schedule for progressive peak sharpening
3. Multi-start with diverse initializations (sine, gaussian, random, blend)
4. Multi-length exploration with interpolation
5. Final coordinate descent polishing with mass transfer moves
"""

import numpy as np
import time
from scipy.optimize import minimize

_EPS = 1e-4


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_from_theta(theta):
    """Convert theta to probability simplex via softmax."""
    theta_max = theta.max()
    a = np.exp(theta - theta_max)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return a / s


def _compute_grad_a(a, p):
    """Compute grad_a[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] via FFT."""
    n = len(a)
    flip_a = a[::-1]
    need = 3 * n - 2
    L = 1 << (need - 1).bit_length()
    F_flip_a = np.fft.rfft(flip_a, L)
    F_p = np.fft.rfft(p, L)
    C = np.fft.irfft(F_flip_a * F_p, L)
    return 2.0 * C[n - 1:2 * n - 1]


def _obj_func(theta):
    """Objective: log-sum-exp of convolution at global _EPS."""
    a = softmax_from_theta(theta)
    n = len(a)
    ac = fast_conv(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / _EPS
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    f_val = _EPS * (np.log(np.sum(exp_s)) + ac_max / _EPS)
    return f_val


def _grad_func(theta):
    """Gradient of log-sum-exp smoothed objective w.r.t. theta."""
    a = softmax_from_theta(theta)
    n = len(a)
    ac = fast_conv(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / _EPS
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    grad_a = _compute_grad_a(a, p)
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)
    return grad_theta


def run_lbfgs(theta0, epsilon, max_iter=800, ftol=1e-14):
    """Run L-BFGS optimization for a fixed epsilon."""
    global _EPS
    _EPS = epsilon
    try:
        result = minimize(
            fun=_obj_func,
            x0=theta0,
            jac=_grad_func,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 50}
        )
        return result.x, result.fun, result.nit
    except Exception:
        return theta0, float('inf'), 0


def epsilon_schedule_optimize(a_init, time_budget, verbose=True):
    """Run adaptive epsilon schedule L-BFGS optimization."""
    n = len(a_init)

def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return norm_sum1(a)


def make_init_uniform(n):
    """Uniform initialization."""
    return np.ones(n) / n


def make_init_random(n, seed, smoothness=0.05):
    """Random smooth initialization."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) + rng.normal(0, smoothness, n)
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def _try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence. Returns (new_best_seq, new_best_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a_norm)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score
