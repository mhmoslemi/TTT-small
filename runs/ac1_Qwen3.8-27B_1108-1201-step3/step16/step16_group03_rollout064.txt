def fast_conv(a):

    A = np.fft.rfft(a, fft_size)

def norm_sum1(a):

    if s < 1e-15:
        return np.ones_like(a) / len(a)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1)
    cond = (1.0 - u * ar) > cssv
    if not cond.any():
        s = v.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return v / s
    idx = np.where(np.cumsum(cond) == np.sum(cond))[0][0]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    S = exp_bc.sum()
    lse = (bc_max + np.log(S)) / beta
    w = exp_bc / S
    L = 1 << ((3 * n - 2) - 1).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    grad_a = 2.0 * cross
    obj = 2.0 * n * lse
    return obj, grad_a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
