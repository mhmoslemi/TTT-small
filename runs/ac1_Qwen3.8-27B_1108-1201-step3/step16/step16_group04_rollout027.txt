def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=20, eps=1e-10):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2 + eps
        C = 1.0 / np.sqrt(P)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def compute_weighted_subgradient(y, ac, top_k=100, power=6.0):
    """
    Compute weighted subgradient using softmax-weighted top-K peaks.
    This gives a much better FW direction than using just the single peak lag.
    """
    n = len(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Softmax weights (higher for larger peaks)
    max_v = top_vals.max()
    w = np.exp(power * (top_vals - max_v) / max_v)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    # Compute gradient: for lag k, d(short)/d(a[i]) = 2*a[k-i] (if valid) + 2*a[i-k] (if valid and != k-i)
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j1 = k_int - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * wi * y[j1]
            j2 = i - k_int
            if 0 <= j2 < n and j2 != j1:
                grad[i] += 2.0 * wi * y[j2]
    return grad

def solve_lp_direction(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized direction or None on failure.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

    return best_a, best_peak


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase hybrid strategy.
    Uses fast subgradient FW first, then LP FW for refinement,
    then mirror descent for diversity, and coordinate descent for polish.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        # Initialize with a smooth non-symmetric sequence
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 0: Spectral pre-conditioning ===
    print(f"\n=== Phase 0: Spectral Equalization n={n0} ===")
    a_spec = spectral_equalize(best_seq, n_iter=40)
    try_update(a_spec, "spectral")
    
    # === Phase 1: Fast weighted-subgradient FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: FastFW n={n_cur}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_in = normalize(best_seq)
        a_out, _ = fast_fw(a_in, t1, top_k=120, power=6.0, verbose=True)
        try_update(a_out, f"fastfw_n{n_cur}")
    
    # === Phase 2: LP FW (main refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LPFW n={n_cur}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw(a_in, t2, solver, top_k_init=80, top_k_max=400, verbose=True)
        try_update(a_out, f"lpfw_n{n_cur}")
    
    # === Phase 3: Mirror descent for diversity ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: MirrorDescent n={n_cur}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = normalize(best_seq)
        a_out, _ = mirror_descent(a_in, t3, top_k=120, power=5.0, c0=5.0, verbose=True)
        try_update(a_out, f"md_n{n_cur}")
    
    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord n={n_cur}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")
    
    # === Phase 5: Second round of fast FW with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: FastFW v2 n={n_cur}, budget={remaining:.0f}s ===")
        t5 = min(remaining * 0.30, 300.0)
        if t5 > 30:
            a_in = normalize(best_seq)
            # Different hyperparameters
            a_out, _ = fast_fw(a_in, t5, top_k=200, power=4.0, verbose=True)
            try_update(a_out, f"fastfw2_n{n_cur}")
    
    # === Phase 6: LP FW again with more constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 6: LPFW v2 n={n_cur}, budget={remaining:.0f}s ===")
        t6 = min(remaining * 0.35, 250.0)
        if t6 > 30:
            a_in = normalize(best_seq)
            a_out, _ = lp_fw(a_in, t6, solver, top_k_init=150, top_k_max=600, verbose=True)
            try_update(a_out, f"lpfw2_n{n_cur}")
    
    # === Phase 7: Try alternate length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_alt = n_cur + 300
    if n_alt != n_cur and remaining > 150:
        print(f"\n=== Phase 7: Alt length n={n_alt}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_long = spectral_equalize(a_long, n_iter=20)
        try_update(a_long, f"alt_init_n{n_alt}")
        
        t7 = min(remaining * 0.40, 250.0)
        if t7 > 30:
            a_in = normalize(a_long)
            a_out, _ = fast_fw(a_in, t7, top_k=100, power=6.0, verbose=False)
            try_update(a_out, f"fastfw_alt_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.4, 150.0)
            if t7b > 30:
                a_in = normalize(best_seq)
                a_out, _ = lp_fw(a_in, t7b, solver, top_k_init=100, top_k_max=300, verbose=False)
                try_update(a_out, f"lpfw_alt_n{n_cur}")
    
    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final Coord n={n_cur} ===")
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 15:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_descent(a_in, t8, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_cur}")
    
    # === Phase 9: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 3000 and remaining > 40:
        print(f"\n=== Phase 9: Final LP n={n_cur} ===")
        t9 = min(remaining * 0.8, 100.0)
        if t9 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_fw(a_in, t9, solver, top_k_init=200, top_k_max=500, verbose=True)
            try_update(a_out, f"lpfw_very_final_n{n_cur}")
    
    # === Phase 10: Last spectral touch ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_touched = normalize(best_seq)
        # Very gentle spectral equalization (just 3 iterations)
        for _ in range(3):
            A = np.fft.fft(a_touched)
            P = A.real**2 + A.imag**2 + 1e-12
            scale = 1.0 / np.sqrt(P)
            # Gentle: only 10% update
            A_new = A * (0.9 + 0.1 * scale / scale.max() * scale.max())
            a_new = np.real(np.fft.ifft(A))  # keep phase
            # Actually just blend original with spectral equalized
            a_spec_short = spectral_equalize(a_touched, n_iter=3)
            a_touched = 0.7 * a_touched + 0.3 * a_spec_short
            a_touched = normalize(a_touched)
        try_update(a_touched, "spectral_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
