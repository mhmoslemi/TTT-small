"""
Search for a non-negative step-history sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2

The algorithm uses:
- multiple asymmetric parametric starts and the previous construction if present
- Nesterov mirror descent with adaptive active-peak softmax subgradients and
  line search over logarithmic step sizes
- an exact 2-index mass-transfer coordinate polish using O(n) convolution updates
- adaptive restarts and limited length exploration
"""

import numpy as np
import time
import math


def _fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float("inf")
    B = _fast_conv(a)
    if B.size == 0:
        return float("inf")
    return float(2.0 * n * B.max())


def _evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return float("inf")
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        n = a.size if a.size else 1
        return np.ones(n, dtype=np.float64) / max(1, n)
    return a / s


def _interpolate_sequence(f, new_n):
    f = _normalize(f)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, float(f[0]), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-300:
        g = np.ones(new_n, dtype=np.float64) / new_n
    else:
        g = g / s
    return g


def _update_best(a, best_a, best_score):
    if a is None:
        return best_a, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return best_a, best_score
    a = a / s
    sc = _score_norm(a)
    if math.isfinite(sc) and sc < best_score:
        return a.copy(), sc
    return best_a, best_score


def _make_initial_candidates(n, parent=None):
    cands = []
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    h = 0.5 / max(1, n - 1)

    if parent is not None:
        p = _normalize(parent)
        if p.size > 0:
            cands.append(p)
            cands.append(p[::-1].copy())

    cands.append(np.ones(n, dtype=np.float64) / float(n))

    # Asymmetric beta-type densities.
    for p in (0.0, 0.25, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 9.0):
        for q in (0.0, 0.1, 0.25, 0.5, 1.0, 2.0, 3.0):
            w = (x + h) ** p * (1.0 - x + h) ** q
            w = np.maximum(w, 0.0)
            s = float(w.sum())
            if s > 1e-300:
                cands.append(w / s)

    # Mild Fourier perturbations.
    for amp in (0.1, 0.25, 0.4, 0.6):
        for sh in (0.0, 0.05, 0.1, 0.2, 0.35, 0.5):
            w = 1.0 + amp * np.cos(2.0 * np.pi * (x - sh))
            w = np.maximum(w, 1e-12)
            s = float(w.sum())
            if s > 1e-300:
                cands.append(w / s)

    # Gaussian bumps, including asymmetric pairs.
    for c in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75):
        for w in (0.04, 0.08, 0.14, 0.22):
            g = np.exp(-0.5 * ((x - c) / w) ** 2)
            s = float(g.sum())
            if s > 1e-300:
                cands.append(g / s)

    for w1 in (0.35, 0.45, 0.55, 0.65):
        for w2 in (0.2, 0.3, 0.4):
            g = w1 * np.exp(-0.5 * ((x - 0.34) / 0.09) ** 2)
            g += (1.0 - w1) * np.exp(-0.5 * ((x - 0.66) / 0.11) ** 2)
            s = float(g.sum())
            if s > 1e-300:
                cands.append(g / s)

    rng = np.random.default_rng(20240611)
    for alpha in (0.5, 1.0, 2.0):
        try:
            r = rng.dirichlet(np.full(n, alpha, dtype=np.float64))
            cands.append(r)
        except Exception:
            pass

    return cands


def _rank_candidates(cands, max_keep=10):
    scored = []
    idx = 0
    for a in cands:
        a = _normalize(a)
        if a.size == 0 or not np.all(np.isfinite(a)):
            continue
        sc = _score_norm(a)
        if math.isfinite(sc):
            scored.append((float(sc), idx, a.copy()))
            idx += 1
    scored.sort()
    out = []
    for sc, _, a in scored[:max_keep]:
        out.append((sc, a))
    return out


def _active_weights(B, topk, eps, beta):
    L = len(B)
    if L == 0:
        return np.array([], dtype=np.float64)
    M = float(B.max())
    w = np.zeros(L, dtype=np.float64)
    mask = B >= (M - max(1e-18, eps * max(1.0, M)))
    cnt = int(mask.sum())
    take = int(max(1, min(L, topk)))
    cap = int(max(take, 40))

    if cnt == 0:
        order = np.argsort(B)[::-1][:take]
        w[order] = 1.0
    elif cnt < take:
        order = np.argsort(B)[::-1][:take]
        w[order] = 1.0
    elif cnt > cap:
        order = np.argsort(B)[::-1][:cap]
        w[order] = 1.0
    else:
        w[mask] = 1.0

    s = float(w.sum())
    if s <= 1e-300:
        w[int(np.argmax(B))] = 1.0
        s = 1.0

    w *= np.exp(beta * (B - M))
    w[w < 1e-300] = 0.0
    s2 = float(w.sum())
    if s2 <= 1e-300:
        w[int(np.argmax(B))] = 1.0
    else:
        w /= s2
    return w


def _grad_from_weights(a, w):
    n = len(a)
    L = 2 * n - 1
    if n <= 1:
        return np.zeros_like(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    if len(w) < L:
        tmp = np.zeros(L, dtype=np.float64)
        tmp[:len(w)] = w[:L]
        w = tmp
    else:
        w = w[:L].copy()

    ar = a[::-1]
    need = 3 * n - 2
    fft_size = 1 << (need - 1).bit_length()
    AR = np.fft.rfft(ar, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(AR * W, fft_size)[:need]
    start = n - 1
    return 2.0 * corr[start:start + n]


def _mirror_step(y, g, eta):
    y = np.maximum(y, 1e-300)
    logy = np.log(y)
    z = logy - eta * g
    z -= float(z.max())
    a = np.exp(z)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return y / float(y.sum())
    return a / s


def _nesterov_descent(a0, duration, seed=1, verbose=True,
                       beta0=6.0, beta1=32.0,
                       eps0=0.025, eps1=0.006,
                       topk0=2, topk1=28,
                       scale0=1.0):
    t0 = time.monotonic()
    a = _normalize(a0)
    best = a.copy()
    bestM = _score_norm(best)
    x = a.copy()
    x_prev = a.copy()
    xM = bestM
    scale = float(scale0)
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    max_stall = 2200
    deltas = (0.20, 0.09, 0.04, 0.018, 0.007)

    while time.monotonic() - t0 < duration:
        if it % 1500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    nesterov it={it} t={elapsed:.1f}s val={2.0*len(x)*bestM:.8f} stall={stall}")

        prog = min(1.0, float(it) / 6000.0)
        beta = beta0 + (beta1 - beta0) * prog
        eps = eps0 + (eps1 - eps0) * prog
        topk = int(max(1, min(80, topk0 + it // 1500)))

        if stall > 1200:
            eps = min(0.08, eps * 1.8)
            topk = min(100, topk + 12)
            beta = max(3.0, beta * 0.85)

        base_mom = 0.0
        if it > 0:
            base_mom = min(0.88, float(it) / (it + 1.6))
        mom = base_mom * max(0.25, 1.0 - 0.45 * min(1.0, float(stall) / max(1, max_stall)))
        if it == 0:
            y = x.copy()
        else:
            y = x + mom * (x - x_prev)
            y = np.maximum(y, 1e-20)
            sy = float(y.sum())
            if sy <= 1e-300:
                y = x.copy()
            else:
                y = y / sy

        B = _fast_conv(y)
        M_y = float(B.max())
        w = _active_weights(B, topk, eps, beta)
        g = _grad_from_weights(y, w)
        gnorm = float(np.max(np.abs(g)))

        if not math.isfinite(gnorm) or gnorm <= 1e-18:
            p = best + rng.normal(0.0, 0.004, len(best)) * best
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            x = p
            x_prev = p.copy()
            xM = _score_norm(x)
            if xM < bestM:
                bestM = xM
                best = x.copy()
            it += 1
            stall += 1
            continue

        best_trial = None
        best_trialM = xM
        for d in deltas:
            eta = min(1e8, (scale * d) / gnorm)
            a_t = _mirror_step(y, g, eta)
            M_t = _score_norm(a_t)
            if M_t < best_trialM:
                best_trialM = M_t
                best_trial = a_t

        if best_trial is not None and best_trialM < xM - 1e-15:
            x_prev = x.copy()
            x = best_trial
            xM = best_trialM
            scale = min(8.0, scale * 1.035)
            stall = 0
        else:
            x_prev = x.copy()
            scale = max(1e-4, scale * 0.78)
            stall += 1

        if xM < bestM:
            bestM = xM
            best = x.copy()

        if stall > max_stall:
            pert_scale = 0.003 + 0.004 * (stall % 3)
            p = best + rng.normal(0.0, pert_scale, len(best)) * best
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            x = p
            x_prev = p.copy()
            xM = _score_norm(x)
            scale = max(0.4, scale * 0.5)
            stall = 0
            if xM < bestM:
                bestM = xM
                best = x.copy()

        it += 1

    return _normalize(best)


def _eval_move(d, B, c, tmp, i, j, L):
    scale = 2.0 * d
    np.multiply(c, scale, out=tmp)
    tmp += B
    d2 = d * d
    pos = 2 * j
    if 0 <= pos < L:
        tmp[pos] += d2
    pos = i + j
    if 0 <= pos < L:
        tmp[pos] -= 2.0 * d2
    pos = 2 * i
    if 0 <= pos < L:
        tmp[pos] += d2
    return float(tmp.max())


def _refine_move(d_best, a_i, B, c, tmp, i, j, L):
    if d_best <= 0.0 or a_i <= 0.0:
        return 0.0
    span = max(1e-14, 0.8 * d_best, 1e-4 * a_i)
    lo = max(0.0, d_best - span)
    hi = min(a_i, d_best + span)
    if hi - lo <= 1e-15:
        return d_best

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _eval_move(c1, B, c, tmp, i, j, L)
    f2 = _eval_move(c2, B, c, tmp, i, j, L)

    for _ in range(14):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _eval_move(c1, B, c, tmp, i, j, L)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _eval_move(c2, B, c, tmp, i, j, L)

    return 0.5 * (lo + hi)


def _coordinate_polish(a0, duration, seed=11, verbose=True):
    t0 = time.monotonic()
    a = _normalize(a0)
    n = len(a)
    L = 2 * n - 1
    B = _fast_conv(a)
    xM = float(B.max())
    best = a.copy()
    bestM = xM
    rng = np.random.default_rng(seed)

    c = np.empty(L, dtype=np.float64)
    tmp = np.empty(L, dtype=np.float64)
    fracs = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1)

    stall = 0
    max_stall = 350
    outer = 0

    while time.monotonic() - t0 < duration and stall < max_stall:
        outer += 1
        if verbose and outer % 25 == 0:
            print(f"    coord outer={outer} val={2.0*n*bestM:.8f} stall={stall}")

        # Active-peak gradient for move direction.
        w = _active_weights(B, 10, 0.008, 12.0)
        g = _grad_from_weights(a, w)

        # Contributions to the most harmful peaks.
        top_peaks = np.argsort(B)[::-1][:6]
        peak_contrib = np.zeros(n, dtype=np.float64)
        for kval in top_peaks:
            k = int(kval)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            idx = np.arange(i0, i1, dtype=np.int64)
            peak_contrib[idx] += a[idx] * a[k - idx]

        order_g = np.argsort(g)
        src_peak = np.argsort(peak_contrib)[::-1][:80]
        src_g = order_g[::-1][:80]
        dst_g = order_g[:220]
        dst_low = np.argsort(a)[:160]

        pairs = []
        seen = set()

        for s_i in list(src_peak) + list(src_g):
            s_i = int(s_i)
            if a[s_i] < 1e-10:
                continue
            for d_i in list(dst_g) + list(dst_low):
                d_i = int(d_i)
                if s_i == d_i:
                    continue
                key = (s_i, d_i)
                if key in seen:
                    continue
                seen.add(key)
                pairs.append(key)
                if len(pairs) >= 1600:
                    break
            if len(pairs) >= 1600:
                break

        if len(pairs) > 900:
            choice = rng.choice(len(pairs), size=900, replace=False)
            pairs = [pairs[int(zz)] for zz in choice]

        improved = False
        for (i, j) in pairs:
            if time.monotonic() - t0 > duration - 2.0:
                break
            if i == j or a[i] <= 1e-12:
                continue

            c.fill(0.0)
            c[j:j + n] += a
            c[i:i + n] -= a

            ai = float(a[i])
            best_val = xM
            best_d = 0.0

            for fr in fracs:
                d = ai * fr
                if d <= 0.0:
                    continue
                val = _eval_move(d, B, c, tmp, i, j, L)
                if val < best_val:
                    best_val = val
                    best_d = d

            if best_d > 0.0 and best_val < xM - 1e-15:
                d_ref = _refine_move(best_d, ai, B, c, tmp, i, j, L)
                val_ref = _eval_move(d_ref, B, c, tmp, i, j, L)
                if d_ref > 0.0 and val_ref < best_val:
                    best_d = d_ref
                    best_val = val_ref

                a[i] -= best_d
                a[j] += best_d
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s <= 1e-300:
                    a = best.copy()
                else:
                    a = a / s
                B = _fast_conv(a)
                xM = float(B.max())
                if xM < bestM:
                    bestM = xM
                    best = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 120:
                pert_scale = 0.006 + 0.006 * (stall - 120) // 60
                p = a * np.exp(rng.normal(0.0, min(0.12, pert_scale), n))
                p = np.maximum(p, 1e-15)
                s = float(p.sum())
                if s > 1e-300:
                    a = p / s
                    B = _fast_conv(a)
                    xM = float(B.max())
                    stall = 0
                    if xM < bestM:
                        bestM = xM
                        best = a.copy()
        else:
            stall = 0

    if verbose:
        print(f"    coord done t={time.monotonic()-t0:.1f}s best={2.0*n*bestM:.8f}")
    return _normalize(best)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    usable = max(60.0, min(float(budget_s), 1800.0) - 120.0)
    deadline = t_start + usable

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 1e-12:
            parent = p
    except Exception:
        parent = None

    base_n = 2400
    if parent is not None:
        if 800 <= len(parent) <= 6144:
            base_n = int(len(parent))
        else:
            parent = _interpolate_sequence(parent, 2400)

    length_set = [base_n]
    for cand in (3072, 4096, int(base_n * 1.18), int(base_n * 0.86)):
        cand = int(cand)
        if 800 <= cand <= 6144 and cand not in length_set:
            length_set.append(cand)
    length_set = length_set[:3]

    best_a = None
    best_score = float("inf")
    start_rank = []
    sid = 0

    for n in length_set:
        p_n = _interpolate_sequence(parent, n) if parent is not None else None
        cands = _make_initial_candidates(n, p_n)
        ranked = _rank_candidates(cands, max_keep=5)
        local_best = ranked[0][0] if ranked else float("inf")
        for sc, a in ranked:
            best_a, best_score = _update_best(a, best_a, best_score)
            if sc < local_best + 1.5:
                start_rank.append((float(sc), sid, a.copy()))
                sid += 1
        if best_a is None:
            a = np.ones(n, dtype=np.float64) / float(n)
            best_a, best_score = _update_best(a, best_a, best_score)
            start_rank.append((float(best_score), sid, a.copy()))
            sid += 1

    start_rank.sort()

    if best_a is None:
        best_a = np.ones(2400, dtype=np.float64) / 2400.0
        best_score = _score_norm(best_a)
        start_rank.append((float(best_score), 1000, best_a.copy()))

    print(f"Initial best: score={best_score:.8f}, n={len(best_a)}")

    # Short exploratory descents from a few distinct starts.
    for rank in start_rank[:3]:
        sc, _, a = rank
        rem = deadline - time.monotonic()
        if rem < 120.0:
            break
        dur = min(75.0, 0.18 * rem)
        if dur <= 10.0:
            break
        try:
            a2 = _nesterov_descent(a, dur, seed=101 + int(_), verbose=False)
        except Exception:
            a2 = a
        best_a, best_score = _update_best(a2, best_a, best_score)
        print(f"  quick-run score={best_score:.8f}")

    # Main descent from current best.
    rem = deadline - time.monotonic()
    if rem > 220.0:
        dur = min(670.0, 0.40 * rem)
        try:
            a2 = _nesterov_descent(best_a, dur, seed=7, verbose=True)
        except Exception:
            a2 = best_a
        best_a, best_score = _update_best(a2, best_a, best_score)
        print(f"  main-descent score={best_score:.8f}, n={len(best_a)}")

    # Secondary descent from another start if available.
    rem = deadline - time.monotonic()
    if rem > 260.0 and len(start_rank) >= 2:
        # Prefer a different length if possible.
        second_a = None
        for rank in start_rank[1:]:
            _, _, a = rank
            if len(a) != len(best_a):
                second_a = a
                break
        if second_a is None:
            second_a = start_rank[1][2]
        dur = min(250.0, 0.20 * rem)
        try:
            a2 = _nesterov_descent(second_a, dur, seed=211, verbose=False)
        except Exception:
            a2 = second_a
        best_a, best_score = _update_best(a2, best_a, best_score)
        print(f"  second-descent score={best_score:.8f}, n={len(best_a)}")

    # Coordinate polishing main.
    rem = deadline - time.monotonic()
    if rem > 150.0:
        dur = min(520.0, 0.32 * rem)
        try:
            a2 = _coordinate_polish(best_a, dur, seed=423, verbose=True)
        except Exception:
            a2 = best_a
        best_a, best_score = _update_best(a2, best_a, best_score)
        print(f"  coord-main score={best_score:.8f}, n={len(best_a)}")

    # Length move if time remains.
    rem = deadline - time.monotonic()
    if rem > 220.0:
        bn = len(best_a)
        if bn < 3600:
            alt_n = 4096
        elif bn > 4600:
            alt_n = 3072
        else:
            alt_n = 5120
        if 800 <= alt_n <= 6144 and alt_n != bn:
            a_alt = _interpolate_sequence(best_a, alt_n)
            dur1 = min(150.0, 0.30 * rem)
            dur2 = min(90.0, 0.18 * rem)
            try:
                a1 = _nesterov_descent(a_alt, max(20.0, dur1), seed=901, verbose=False)
                a2 = _coordinate_polish(a1, max(10.0, dur2), seed=902, verbose=False)
            except Exception:
                a2 = a_alt
            best_a, best_score = _update_best(a2, best_a, best_score)
            print(f"  length-move alt_n={alt_n} score={best_score:.8f}, n={len(best_a)}")

    # Final coordinate polish.
    rem = deadline - time.monotonic()
    if rem > 45.0:
        dur = min(160.0, 0.55 * rem)
        try:
            a2 = _coordinate_polish(best_a, max(5.0, dur), seed=999, verbose=True)
        except Exception:
            a2 = best_a
        best_a, best_score = _update_best(a2, best_a, best_score)
        print(f"  final-polish score={best_score:.8f}, n={len(best_a)}")

    # Validation and return.
    best_a = _normalize(best_a)
    fs = _evaluate_safe(best_a)
    if not math.isfinite(fs):
        best_a = np.ones(2400, dtype=np.float64) / 2400.0
        best_a = _normalize(best_a)

    print(f"\nFINAL score={best_score:.8f}, n={len(best_a)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in best_a]
