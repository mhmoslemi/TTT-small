    ac = fast_autocorr(a)

    for offset in [-2, -1, 1, 2]:
        idx = peak_idx + offset

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_conv_row(y, k, n):
    """Build the row of the convolution matrix for lag k.
    Returns (coeffs, indices) such that sum(coeffs[i] * b[indices[i]]) = (y*b)_k
    """
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start > j_end:
        return None, None
    b_indices = np.arange(k - j_end, k - j_start + 1)
    y_vals = y[j_start:j_end + 1]
    return y_vals, b_indices

def smoothed_lbgfs(a0, time_limit, epsilon=0.01, verbose=True, max_iters=5000):
    """L-BFGS on the log-sum-exp smoothed objective with softmax parametrization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Softmax parametrization: a_i = softmax(z)_i
    # Work with log(a_i) as the variable, with a small floor
    z = np.log(np.maximum(a, 1e-10)).astype(np.float64)

    def objective_and_grad(z_vec):
        # Compute softmax
        z_shifted = z_vec - z_vec.max()
        exp_z = np.exp(z_shifted)
        a_vec = exp_z / exp_z.sum()
        
        # Compute convolution
        c = np.convolve(a_vec, a_vec)
        # Log-sum-exp smoothing
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / epsilon)
        lse = c.max() + epsilon * np.log(exp_c.sum())
        obj = 2.0 * n * lse
        
        # Gradient: d(obj)/d(z_i)
        # d(obj)/d(a_i) * d(a_i)/d(z_i)
        # d(a_i)/d(z_i) = a_i * (delta_{ij} - a_j) for softmax
        # d(obj)/d(a_i) = 2n * sum_k p_k * 2*a_{k-i} * 1[0<=k-i<n] / (sum a) ... 
        # Wait, since we're using softmax, a is determined by z.
        # Let's compute grad w.r.t. a first, then chain rule.
        
        p = exp_c / exp_c.sum()  # probability over k
        # grad_a[i] = 2n * sum_k p_k * 2 * a_{k-i} * 1[0<=k-i<n]
        # But we need to be careful: the softmax constraint means we should
        # project the gradient onto the tangent space of the simplex.
        
        grad_a = np.zeros(n)
        for i in range(n):
            # grad_a[i] = 2 * 2n * sum_{k: 0<=k-i<n} p_k * a_{k-i}
            # = 4n * sum_{k: 0<=k-i<n} p_k * a_{k-i}
            # Let j = k-i, so k = i+j, j in [0, n-1], k in [i, i+n-1]
            # k must be in [0, 2n-2]
            k_start = max(0, i)
            k_end = min(2*n-2, i+n-1)
            if k_start <= k_end:
                # grad_a[i] = 4n * sum_{k=k_start}^{k_end} p_k * a[k-i]
                # a[k-i] where k-i ranges from k_start-i to k_end-i
                j_vals = np.arange(k_start - i, k_end - i + 1)
                grad_a[i] = 4.0 * n * np.sum(p[k_start:k_end+1] * a_vec[j_vals])
        
        # Chain rule: dz -> da
        # a_i = softmax(z)_i, so da_i/dz_j = a_i*(delta_ij - a_j)
        # d(obj)/dz_j = sum_i d(obj)/da_i * da_i/dz_j
        # = sum_i grad_a[i] * a_i * (delta_ij - a_j)
        # = grad_a[j]*a_j - a_j * sum_i grad_a[i]*a_i
        
        sum_grad_a = np.dot(grad_a, a_vec)
        grad_z = a_vec * (grad_a - sum_grad_a)
        
        # Project out the all-ones direction (since softmax is invariant to adding constant)
        # Actually, the softmax parametrization already handles this:
        # if we add c to all z, a doesn't change. So the gradient should be 
        # orthogonal to 1. Let's verify: sum_j grad_z[j] = sum_j a_j*(grad_a[j] - sum_grad_a)
        # = sum_j a_j*grad_a[j] - sum_grad_a * sum_j a_j = sum_grad_a - sum_grad_a = 0. Good!
        
        return obj, grad_z

    # L-BFGS via scipy
    from scipy.optimize import minimize
    
    # Initial objective
    obj0, grad0 = objective_and_grad(z)
    
    best_obj = obj0
    best_z = z.copy()
    
    # Use scipy L-BFGS-B with maximum iterations controlled by time
    result = minimize(
        objective_and_grad,
        z,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iters, 'maxfun': max_iters, 'disp': False}
    )
    
    if result.fun < best_obj:
        best_obj = result.fun
        best_z = result.x.copy()
    
    # Convert back
    z_shifted = best_z - best_z.max()
    a_result = np.exp(z_shifted) / np.exp(z_shifted).sum()
    a_result = a_result / a_result.sum()
    
    # Verify
    score = 2.0 * n * fast_autocorr(a_result).max()
    if score < best_score:
        best_a = a_result
        best_score = score
    
    # Also check the original
    score_orig = 2.0 * n * fast_autocorr(a).max()
    if score_orig < best_score:
        best_a = a.copy()
        best_score = score_orig
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a, best_score


def mirror_descent(a0, time_limit, top_k=30, verbose=False, max_iters=1000000):

        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.02, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

    return best_a, best_m
