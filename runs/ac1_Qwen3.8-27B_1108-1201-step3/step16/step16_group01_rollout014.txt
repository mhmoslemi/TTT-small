def fast_conv_full(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def soft_max_gradient(a, beta, top_k=80):
    """

    nk = len(conv_a)

def mirror_update(a, g, eta):
    """Multiplicative (mirror) update: a * exp(-eta*g), then normalize."""
    log_shift = -eta * g
    max_ls = log_shift.max()
    a_new = a * np.exp(log_shift - max_ls)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

        # Get contributions for the peak
        idx_list = np.arange(i_start, i_end)
        j_list = peak_lag - idx_list
        contribs = a[idx_list] * a[j_list]
        
        # Sort by contribution (descending)
        order = np.argsort(contribs)[::-1]
        
        improved = False
        for oi in order[:min(40, len(order))]:
            if time.monotonic() - t0 > time_limit:
                break
            i_pos = int(idx_list[oi])
            j_pos = int(j_list[oi])
            
            if a[i_pos] < 1e-12:
                continue
            
            # Try moving mass from (i_pos, j_pos) pair to a random other position
            for _attempt in range(5):
                dst = int(rng.integers(n))
                if dst == i_pos or dst == j_pos:
                    continue
                # Move from the larger of a[i_pos], a[j_pos]
                if a[i_pos] >= a[j_pos]:
                    src = i_pos
                else:
                    src = j_pos
                
                for frac in [0.003, 0.008, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0.0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    try_val = fast_conv_full(a_try).max()
                    if try_val < best_val - 1e-14:
                        a = a_try
                        best_val = try_val
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Larger perturbation
                n_pert = max(10, n // 30)
                sources = rng.choice(n, size=n_pert, replace=False)
                for i in sources:
                    if a[i] > 1e-8:
                        j = int(rng.integers(n))
                        if j != i:
                            frac = rng.uniform(0.02, 0.1)
                            amt = a[i] * frac
                            a[i] -= amt
                            a[j] += amt
                            a = np.maximum(a, 0.0)
                            s = a.sum()
                            if s > 1e-15:
                                a = a / s
                                try_val = fast_conv_full(a).max()
                                if try_val < best_val:
                                    best_val = try_val
                                    best_a = a.copy()
                                    stall = 0
                                else:
                                    a[i] += amt
                                    a[j] -= amt
                                    a = np.maximum(a, 0.0)
                                    if a.sum() > 1e-15:
                                        a = a / a.sum()
        
        if verbose and stall % 60 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PSD ({elapsed:.1f}s): J={2*n*best_val:.6f}, stall={stall}")
    
    return best_a, best_val * 2 * n


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, style='sin_cos', rng_seed=0):
    """Create diverse initializations."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0, 1, n)
    
    if style == 'uniform':
        a = np.ones(n)
    elif style == 'sin':
        a = np.sin(np.pi * x) ** 2 + 1e-6
    elif style == 'sin_cos':
        a = np.sin(np.pi * x) ** 2 + 0.2 * np.cos(2 * np.pi * x) ** 2 + 1e-6
    elif style == 'cosine_bell':
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 2 + 1e-6
    elif style == 'u_shape':
        a = (x - 0.5) ** 2 + 0.05 + 1e-6
    elif style == 'w_shape':
        a = np.cos(4 * np.pi * x) ** 2 + 0.3 + 1e-6
    elif style == 'random_smooth':
        a = np.abs(np.sin(3 * np.pi * x + rng.uniform(0, 2*np.pi))) + \
            0.5 * np.abs(np.sin(5 * np.pi * x + rng.uniform(0, 2*np.pi))) + 1e-6
    elif style == 'bimodal':
        a = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2) + 1e-6
    elif style == 'trapezoid':
        a = np.clip(1.0 - 2.0 * np.abs(x - 0.5), 0.1, 1.0)
    elif style == 'power':
        a = x ** 0.5 + (1 - x) ** 0.5 + 1e-6
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def lfbfgs_subgradient(a0, time_limit, top_k=40, verbose=True, rng_seed=77):
    """
    L-BFGS-like optimization using subgradient approximation.
    Uses a simple 2-momentum scheme with adaptive step.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = fast_conv_full(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Simple momentum-based method
    v = np.zeros(n)
    beta_m = 0.9
    lr = 0.1 / np.sqrt(n)
    stall = 0
    max_stall = 600
    
    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = soft_max_gradient(a, beta=20.0, top_k=top_k)
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-30:
            g = g / g_norm
        
        v = beta_m * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        cur_val = fast_conv_full(a_new).max()
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 0.5 / np.sqrt(n))
        else:
            stall += 1
            lr *= 0.95
            v *= 0.5
            # Don't update a
        
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise = rng.normal(0, 0.01, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 0.1 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      LBF restart it={it} ({elapsed:.1f}s): J={2*n*best_val:.6f}")
        
        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LBF it={it} ({elapsed:.1f}s): J={2*n*best_val:.6f}, lr={lr:.4f}, stall={stall}")
    
    return best_a, best_val * 2 * n


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2400
        best_seq = make_initialization(n_base, 'sin_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")
    
    # === Phase 1: Soft-max gradient descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Soft-max descent n={n0} (t={t1:.0f}s) ===")
        a_out, _ = soft_max_descent(normalize(best_seq), t1, beta0=15.0, beta_min=5.0,
                                     top_k=80, eta0=0.3, verbose=True, rng_seed=1)
        try_update(a_out, f"smd_n{n0}")
    
    # === Phase 2: Peak-shaving coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Peak shaving CD n={n_best} (t={t2:.0f}s) ===")
        a_out, _ = peak_shaving_cd(normalize(best_seq), t2, verbose=True, rng_seed=2)
        try_update(a_out, f"pcd_n{n_best}")
    
    # === Phase 3: Pair swap descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Pair swap n={n_best} (t={t3:.0f}s) ===")
        a_out, _ = pair_swap_descent(normalize(best_seq), t3, verbose=True, rng_seed=3)
        try_update(a_out, f"psd_n{n_best}")
    
    # === Phase 4: Second round of soft-max with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        print(f"\n=== Phase 4: SMD round 2 n={n_best} (t={t4:.0f}s) ===")
        a_out, _ = soft_max_descent(normalize(best_seq), t4, beta0=25.0, beta_min=8.0,
                                     top_k=40, eta0=0.2, verbose=True, rng_seed=4)
        try_update(a_out, f"smd2_n{n_best}")
    
    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length exploration ===")
        lengths_to_try = []
        for dl in [-400, -200, 200, 400, 600]:
            nl = n_best + dl
            if 800 <= nl <= 4000:
                lengths_to_try.append(nl)
        
        for nl in lengths_to_try[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"  Trying n={nl}")
            a_interp = interpolate_to_length(best_seq, nl)
            t_try = min(remaining * 0.25, 150.0)
            if t_try > 20:
                a_out, _ = soft_max_descent(normalize(a_interp), t_try, beta0=15.0,
                                             beta_min=5.0, top_k=60, eta0=0.3,
                                             verbose=False, rng_seed=10)
                try_update(a_out, f"sml_n{nl}")
                a_out2, _ = peak_shaving_cd(normalize(a_out), min(remaining*0.1, 50.0),
                                            verbose=False, rng_seed=11)
                try_update(a_out2, f"pcm_n{nl}")
    
    # === Phase 6: L-BFGS style refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 15:
        print(f"\n=== Phase 6: LBF n={n_best} (t={t6:.0f}s) ===")
        a_out, _ = lfbfgs_subgradient(normalize(best_seq), t6, top_k=30,
                                       verbose=True, rng_seed=5)
        try_update(a_out, f"lbf_n{n_best}")
    
    # === Phase 7: Final polish with very fine steps ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish n={n_best} ===")
        t7 = min(remaining * 0.30, 100.0)
        if t7 > 10:
            a_out, _ = soft_max_descent(normalize(best_seq), t7, beta0=30.0, beta_min=10.0,
                                         top_k=20, eta0=0.1, verbose=False, rng_seed=6)
            try_update(a_out, f"fin_n{n_best}")
            a_out2, _ = peak_shaving_cd(normalize(best_seq), min(remaining*0.2, 60.0),
                                        verbose=False, rng_seed=7)
            try_update(a_out2, f"fin_cd_n{n_best}")
    
    # === Phase 8: Diverse restart exploration (if time remains) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 8: Diverse restarts ===")
        styles = ['u_shape', 'w_shape', 'bimodal', 'random_smooth', 'trapezoid']
        for style in styles[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            n_try = len(best_seq)
            a_init = make_initialization(n_try, style, rng_seed=hash(style) % 1000)
            # Mix with current best
            a_mixed = 0.5 * normalize(a_init) + 0.5 * normalize(best_seq)
            a_mixed = normalize(a_mixed)
            t_try = min(remaining * 0.35, 80.0)
            if t_try > 15:
                a_out, _ = soft_max_descent(a_mixed, t_try, beta0=10.0, beta_min=5.0,
                                             top_k=60, eta0=0.3, verbose=False,
                                             rng_seed=20)
                try_update(a_out, f"div_{style}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
