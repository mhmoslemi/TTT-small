"""

2. Backtracking line search for adaptive step sizing

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_sw_weights(ac, beta):
    """Compute softmax weights over autocorrelation with temperature beta."""
    max_ac = ac.max()
    shifted = beta * (ac - max_ac)
    max_shifted = shifted.max()
    log_w = shifted - max_shifted
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = np.argmax(ac)
        w = np.zeros_like(ac)
        w[idx] = 1.0
    else:
        w = w / w_sum
    return w

def compute_grad_fft(a, w):
    """Compute gradient of LSE objective via FFT cross-correlation.
    grad[j] = 2 * sum_m a[m] * w[m+j]
    """
    n = len(a)
    N = 1 << (3 * n).bit_length()
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    corr = np.fft.ifft(np.conj(A) * W).real[:n]
    return 2.0 * corr

def compute_grad_direct(a, w, top_k=200):
    """Compute gradient directly for top-k weights (faster for sparse w)."""
    n = len(a)
    grad = np.zeros(n)
    if len(w) <= top_k:
        active = np.where(w > 1e-15)[0]
    else:
        active = np.argsort(w)[::-1][:top_k]
        active = active[w[active] > 1e-15]
    for k in active:
        wk = w[k]
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start >= j_end:
            continue
        indices = np.arange(j_start, j_end)
        a_idx = k - indices
        grad[indices] += 2.0 * wk * a[a_idx]
    return grad

def lse_objective(a, beta):
    """Compute LSE of autocorrelation (smoothed max)."""
    ac = fast_autocorr(a)
    max_ac = ac.max()
    shifted = beta * (ac - max_ac)
    max_shifted = shifted.max()
    return max_ac + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / beta

def mirror_descent_bt(a_init, beta, time_limit, c_init=5.0, verbose=False):
    """Mirror descent with backtracking line search and Nesterov momentum."""
    n = len(a_init)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_obj = lse_objective(a, beta)
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    eta = c_init
    momentum = 0.0
    best_momentum = 0.0
    stall = 0
    max_stall = 500

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            m = beta / (2.0 + np.sqrt(4.0 + beta * beta))
            y = x + momentum * (x - x_prev)
            y = np.maximum(y, 1e-18)
            sy = y.sum()
            y = y / sy
        else:
            y = x.copy()
            m = 0.0

        ac = fast_autocorr(y)
        w = compute_sw_weights(ac, beta)
        g = compute_grad_fft(y, w)

        # Backtracking line search in log-space
        # New point: a_new = exp(log(y) - eta * g / y) (KL projection step)
        # Actually mirror descent step: a_new = y * exp(-eta * g) / sum(y * exp(-eta * g))
        obj_new = None
        for bt in range(20):
            scale = eta * (0.5 ** bt)
            log_ratio = -scale * g
            max_lr = log_ratio.max()
            a_new = y * np.exp(log_ratio - max_lr)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            obj_new = lse_objective(a_new, beta)
            if obj_new <= lse_objective(y, beta) + 1e-10:
                break

        if obj_new is None or not np.isfinite(obj_new) or obj_new > lse_objective(y, beta) * 1.1:
            eta = eta * 0.5
            continue

        # Check if momentum should be restarted
        if obj_new > lse_objective(x, beta) and momentum > 0.05:
            momentum = 0.0
            x_prev = x

        # Update momentum (Nesterov)
        target_m = beta / (2.0 + np.sqrt(4.0 + beta * beta))
        momentum = 0.9 * momentum + 0.1 * target_m
        momentum = min(momentum, 0.5)

        x_prev = x
        x = a_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.002, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            momentum = 0.0
            eta = c_init
            stall = 0
            max_stall = max(200, max_stall // 2)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * fast_autocorr(best_a).max()
            print(f"      MD-BT beta={beta:.1f} t={t} ({elapsed:.1f}s): score={score:.6f}, eta={eta:.4f}")

    return best_a

def targeted_peak_reduction(a_init, time_limit, verbose=False):
    """Directly reduce peak lags by redistributing mass."""
    n = len(a_init)
    a = norm_sum1(a_init.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        max_val = ac.max()
        
        # Find lags near the max
        threshold = max_val * 0.995
        peak_lags = np.where(ac >= threshold)[0]

        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(ac)])

        improved = False
        for k in peak_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # For this lag, find which pairs (i, k-i) contribute most
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            indices = np.arange(j_start, j_end)
            a_vals = a[indices]
            a_opp = a[k_int - indices]
            products = a_vals * a_opp
            total = products.sum()
            if total < 1e-15:
                continue

            # Identify the "heavy" contributors
            top_contrib = np.argsort(products)[::-1][: min(5, len(products))]
            for tc in top_contrib:
                i = indices[tc]
                j = k_int - i
                if i == j or j < 0 or j >= n:
                    continue
                # Try moving mass from the larger of a[i], a[j] to a "quiet" position
                src = i if a[i] >= a[j] else j
                if a[src] < 1e-8:
                    continue
                # Find a "quiet" destination (low autocorrelation contribution)
                n_try = min(5, n)
                for _ in range(n_try):
                    dst = int(rng.integers(n))
                    if dst == src or dst == j:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Small random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"      TPR it={it} ({elapsed:.1f}s): score={score:.6f}")

    return best_a

def coordinate_descent_polish(a_init, time_limit, verbose=False, rng_seed=42):

    n = len(a_init)
    a = norm_sum1(a_init.copy())

    stall = 0
    max_stall = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        sweep_improved = False
        order = list(range(n))
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            a_i = a[i]
            if a_i < 1e-14:
                continue

            best_trial_m = best_m
            best_move = None
            # Try moving to several random destinations
            n_try = min(6, n)
            for _ in range(n_try):
                dst = int(rng.integers(n))
                if dst == i:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                    amt = a_i * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[dst] += amt
                    if a_trial[i] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m - 1e-15:
                        best_trial_m = m_trial
                        best_move = (dst, amt)

            if best_move is not None:
                dst, amt = best_move
                a[i] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                best_m = best_trial_m
                best_a = a.copy()
                sweep_improved = True

        if sweep_improved:
            stall = 0
        else:
            stall += 1
            # Escape: random perturbation
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt

                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:

                        stall = 0

                    else:
                        a[i] += amt

            elapsed = time.monotonic() - t0

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # Alpha function shape: high at edges, low in center
        init2 = (np.abs(x - 0.5) ** 0.5 + 0.1)
        init2 = np.maximum(init2, 0.0)
        init2 = init2 / init2.sum()
        a_out, _ = full_optimization(init2, t2 * 0.6, verbose=True)
        try_update(a_out, f"Ushaped_n{n0}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            # Try cos-shaped initialization
            init3 = np.cos(np.pi * x) ** 2 + 0.05
            init3 = np.maximum(init3, 0.0)
            init3 = init3 / init3.sum()
            a_out, _ = full_optimization(init3, remaining * 0.5, verbose=True)
            try_update(a_out, f"Cos_n{n0}")

    # === Phase 3: Alternative length (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_alt = n0 + 400
        print(f"\n=== Phase 3: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        t3 = min(remaining * 0.5, 300.0)
        a_out, _ = full_optimization(a_long, t3, verbose=True)
        try_update(a_out, f"AltLen_n{n_alt}")

    # === Phase 4: Final polish at best length (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 4: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        # High-beta mirror descent for final precision
        a_in = norm_sum1(best_seq)
        t4a = min(remaining * 0.4, 100.0)
        if t4a > 20:
            a_out = mirror_descent_bt(a_in, 500.0, t4a, c_init=0.5, verbose=False)
            m_out = fast_autocorr(a_out).max()
            score_out = 2.0 * n_best * m_out
            if score_out < best_score:
                try_update(a_out, "FinalMD_hibeta")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t4b = min(remaining * 0.5, 80.0)
            a_out = coordinate_descent_polish(best_seq, t4b, verbose=False, rng_seed=1234)
            try_update(a_out, "FinalCD")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t4c = min(remaining * 0.5, 50.0)
            a_out = targeted_peak_reduction(best_seq, t4c, verbose=False)
            try_update(a_out, "FinalTPR")

    # Final validation
    if best_seq is not None:
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
