                # We wish to force the low autocorrelation sequence with this method
                a_coord, j_coord, _ = my_coordinate_descent(current_seq, remaining_time)
                if j_coord < best_J * 0.995:
                    current_seq = a_coord
                    current_J = j_coord
                    best_seq = current_seq
                    print(f"Coordination descent complete: J={best_J:.6f}")
                # Update time and check for next phase
                iteration_count += 1
            except Exception:
                print(f"Coordination descent disrupted at iteration {iteration_count}")
                continue
                
            # End phase logic here, but we have streamlined
            remaining_time -= (time.time() - start_time)
            
        # Post-optimization handling with fallback to known parameters
        try:
            return best_seq.tolist(), best_J
        except Exception:
            # Return a meaningful fallback
            seq_zeros = np.ones(min(max_valid_length, 500)) / min(max_valid_length, 500)
            fall_J = evaluate_my_convolution(seq_zeros)
            return seq_zeros.tolist(), fall_J

    except Exception as e:
        # Absolute fallback
        fallback_seq = np.ones(240) / 240.0
        fallback_J = evaluate_my_convolution(fallback_seq)
        print(f"Error: {e}")
        return fallback_seq.tolist(), fallback_J
