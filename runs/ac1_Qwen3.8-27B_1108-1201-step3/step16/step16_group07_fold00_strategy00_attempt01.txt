    # Compute the full autocorrelation (convolution) of the sequence
    conv = np.convolve(seq, seq, mode="full")
    max_conv = np.max(conv)

    # Normalize the sequence by its sum (if default, clip to enforce positive）
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return np.inf

    # Compute the score
    # Note: This evaluates the normalized sequence directly, but the objective is defined for any sequence.
    score = float(2 * n * max_conv / (sum_val ** 2))
    return score
