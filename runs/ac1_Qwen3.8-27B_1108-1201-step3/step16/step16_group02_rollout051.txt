            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): score={score:.8f}, alpha={best_alpha:.3f}")
    
    return best_a, best_m

def spectral_shaping(a0, time_limit, verbose=True):
    """Use frequency domain to find directions that flatten the convolution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = obj_value(a)
    t0 = time.monotonic()
    
    # Zero-pad to length >= 2n-1
    L = 1 << (2 * n - 1).bit_length()
    
    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, L)
        C = np.fft.irfft(np.abs(A)**2, L)[:2*n-1]
        M = C.max()
        
        # Target: flatten C to mean value
        target = 1.0 / (2 * n - 1)
        
        # Reduce peaks: scale down high values, scale up low values
        # Use soft thresholding
        C_mod = C.copy()
        C_mod[C_mod > target * 1.5] = target * 1.5
        C_mod[C_mod < target * 0.3] = target * 0.3
        
        # Pad back to full length
        C_full = np.zeros(L)
        C_full[:2*n-1] = C_mod
        
        # New |A|^2
        C_new = np.fft.rfft(C_full)
        # C_new should be real and non-negative for valid |A|^2
        # Take the positive part
        C_new = np.maximum(C_new, 0)
        
        # Update A: keep phase of A, use new magnitude
        phase_A = np.angle(A)
        mag_new = np.sqrt(C_new)
        
        # Ensure DC is 1 (sum of a = 1)
        mag_new[0] = 1.0
        
        A_new = mag_new * np.exp(1j * phase_A)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = obj_value(a_new)
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        # Blend with previous
        a = 0.7 * a + 0.3 * a_new
        a = normalize(a)
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    SpecShape iter={it} ({elapsed:.1f}s): score={score:.8f}")
    
    return best_a, best_m

def nesterov_md(a0, time_limit, init_temp=0.01, final_temp=0.002, c0=3.0, verbose=True):
    """Nesterov accelerated mirror descent with full softmax subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, t / 3000.0)
        temp = init_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-5)
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        # Softmax subgradient over top lags
        top_k = min(100, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        log_w = np.log(top_vals) - np.log(top_vals).max()
        w = np.exp(log_w / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = obj_value(x)
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.5)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def local_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive local search via random pair swaps with line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_tries = min(200, n // 5)
        
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            src, dst = i, j
            if a[src] < a[dst]:
                src, dst = j, i
            
            # Try multiple fractions via binary search
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = obj_value(a_trial)
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            idx = rng.choice(n, size=min(10, n//4), replace=False)
            pert = np.zeros(n)
            pert[idx] = rng.normal(0, 0.005, len(idx))
            a_pert = a * (1 + pert)
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
                m_p = obj_value(a_pert)
                if m_p < best_m:
                    best_m = m_p
                    best_a = a_pert.copy()
                    a = a_pert
                    stall = 0
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Polish ({elapsed:.1f}s): J={J:.8f}, stall={stall}")
    
    return best_a, best_m

def select_best_cand(a_list, M_list, n):
    """Select the best candidate by score."""
    best = 0
    for i, (a, M) in enumerate(zip(a_list, M_list)):
        sc = 2.0 * n * M
        if sc < 2.0 * n * M_list[best]:
            best = i
    return best

def lp_fwk_variant(a0, time_limit, solver, variant='fre', verbose=True):
    """
    LP-FW variant with acceleration.
    variant='std': standard FW
    variant='fre': freedom from the minimum (FreM)
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = obj_value(a)
    t0 = time.monotonic()
    
    from scipy.sparse import csr_matrix
    
    if variant == 'fre':
        y = a.copy()  # freedom point
        x = a.copy()  # actual iterate
    else:
        y = a.copy()
    
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Build full constraint matrix
        n_lags = 2 * n - 1
        rows = []
        cols = []
        vals = []
        
        for k in range(n_lags):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                aik = a[k - j]
                if aik > 1e-15:
                    rows.append(k)
                    cols.append(j)
                    vals.append(aik)
        
        if len(rows) == 0:
            break
        
        T_sparse = csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), 
                                                  np.array(cols, dtype=np.int32))), 
                              shape=(n_lags, n))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sparse @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        g = b_val / b_sum
        
        if variant == 'fre':
            # FreM update
            x_new = (1.0 - 1.0/t) * x + (1.0/t) * g
            x_new = normalize(x_new)
            y_new = (1.0 - 2.0/(t+2.0)) * y + (2.0/(t+2.0)) * (x_new - y)
            y_new = normalize(y_new)
            x = x_new
            y = y_new
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y = normalize(y_new)
        
        # Check both y and (if FreM) x
        my = obj_value(y)
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if variant == 'fre':
            mx = obj_value(x)
            if mx < best_m:
                best_m = mx
                best_a = x.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    FWk({variant}) iter={t} ({elapsed:.1f}s): score={score:.8f}")
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    a_in = normalize(best_seq)
    
    # === Phase 1: Full-constraint LP-FW (55% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.55, 900.0)
    print(f"\n=== Phase 1: LP-FW full n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, m_out = lp_frank_wolve_full(a_in, t1, solver, verbose=True)
        try_update(a_out, f"FW_full_n{n0}")
    
    # === Phase 2: FreM variant (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 2: FreM n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_in2 = normalize(best_seq)
        a_out, m_out = lp_fwk_variant(a_in2, t2, solver, variant='fre', verbose=True)
        try_update(a_out, f"FreM_n{len(best_seq)}")
    
    # === Phase 3: Spectral shaping + LP-FW restart (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 3: Spectral+FW n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 40:
        a_in3 = normalize(best_seq)
        # First do spectral shaping to get a new starting point
        t3a = t3 * 0.3
        t3b = t3 * 0.7
        a_spectral, _ = spectral_shaping(a_in3, t3a, verbose=False)
        try_update(a_spectral, f"Spectral_n{len(best_seq)}")
        # Then LP-FW from the spectral point
        a_out3, _ = lp_frank_wolve_full(normalize(best_seq), t3b, solver, verbose=False)
        try_update(a_out3, f"FW2_n{len(best_seq)}")
    
    # === Phase 4: NMD for diversity (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 4: NMD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        # Start from a slightly perturbed version
        pert = rng.normal(0, 0.01, len(best_seq))
        a_in4 = normalize(best_seq * (1 + pert))
        a_out4, _ = nesterov_md(a_in4, t4, init_temp=0.01, final_temp=0.002,
                                 c0=3.0, verbose=False)
        try_update(a_out4, f"NMD_n{len(best_seq)}")
    
    # === Phase 5: Local polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 5: Polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out5, _ = local_polish(normalize(best_seq), t5, verbose=True, rng_seed=789)
        try_update(a_out5, f"Polish_n{len(best_seq)}")
    
    # === Phase 6: Try slightly different n (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n0 + 200
        print(f"\n=== Phase 6: Alt n={n_alt} (budget={remaining:.0f}s) ===")
        # Interpolate
        a_long = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, n0), 
                           normalize(best_seq))
        a_long = normalize(a_long)
        t6 = min(remaining * 0.8, 120.0)
        if t6 > 30:
            a_out6, _ = lp_frank_wolve_full(a_long, t6, solver, verbose=False)
            try_update(a_out6, f"FW_n{n_alt}")
    
    # === Phase 7: Final LP-FW polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final FW n={n_best} (budget={remaining:.0f}s) ===")
        a_out7, _ = lp_frank_wolve_full(normalize(best_seq), remaining, solver, verbose=True)
        try_update(a_out7, f"FW_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
