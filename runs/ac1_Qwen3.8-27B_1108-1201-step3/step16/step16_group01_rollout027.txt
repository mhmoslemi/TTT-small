import numpy as np
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def autocorr_grad(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] wrt a.
    Uses: d conv(a,a)[k]/da[i] = 2*a[k-i] if 0 <= k-i < n else 0
    So grad[i] = 2*sum_k w[k]*a[k-i] (valid k)
    = 2 * (w_cross a)[i] where cross is cross-correlation
    """
    n = len(a)
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]  (w has length 2n-1)
    # This is the cross-correlation of w and a at shift i
    # = 2 * (flip(a) * w)[i + n - 1]  (full convolution)
    a_flip = a[::-1]
    # conv(a_flip, w) has length n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (3*n - 3).bit_length()
    AF = np.fft.rfft(a_flip, fft_size)
    WF = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(AF * WF, fft_size)
    return 2.0 * conv[n-1:2*n-1]

def power_mean_obj_grad(z, p):
    """Objective and gradient for log-parameterized power mean."""
    # Softmax
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    
    # Autocorrelation
    ac = fast_autocorr(a)
    ac = np.maximum(ac, 1e-30)
    
    # Power mean
    ac_p = ac ** p
    sum_ac_p = ac_p.sum()
    obj = 2.0 * n * sum_ac_p ** (1.0/p)
    
    # Gradient wrt a: grad_a[i] = 2 * sum_j w[i+j] * a[j]
    # where w[k] = p * ac[k]^{p-1} / sum_ac_p^{(p-1)/p} * (1/p) * (1/p)
    # d/dam [sum_ac_p]^{1/p} = (1/p) * (sum_ac_p)^{1/p - 1} * sum_k p * ac[k]^{p-1} * dam/da
    # = (sum_ac_p)^{1/p - 1} * sum_k ac[k]^{p-1} * dam/da
    # where dam/da is the gradient of ac[k] wrt a
    
    # w[k] = ac[k]^{p-1} / sum_ac_p^{(p-1)/p}
    # This is the "soft max" weight
    w = ac ** (p-1) / (sum_ac_p ** ((p-1.0)/p))
    
    # grad_a = autocorr_grad(a, w)  [this gives the gradient of power_mean wrt a]
    grad_a = autocorr_grad(a, w)
    
    # Chain rule: grad_z = a * (grad_a - a @ grad_a)
    # (standard softmax gradient)
    mean_grad = float(a @ grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with power-mean continuation (smoothed objective -> true max)
2. LP-based Frank-Wolfe polishing via cvxpy
3. Aggressive coordinate descent for final refinement
4. Multi-start with diverse initializations
"""
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_autocorr_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] with respect to a.
    weights has length 2n-1 (same as autocorrelation).
    Uses: d conv(a,a)[k]/da[i] = 2*a[k-i] if 0 <= k-i < n else 0
    So grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    Computed via FFT: 2 * (full_conv(a[::-1], weights))[n-1 : 2n-1]
    """
    n = len(a)
    a_flip = a[::-1]
    # full convolution of a_flip (length n) and weights (length 2n-1)
    # result length: n + (2n-1) - 1 = 3n-2
    conv_size = n + 2 * n - 1  # = 3n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    AF = np.fft.rfft(a_flip, fft_size)
    WF = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(AF * WF, fft_size)
    return 2.0 * conv[n-1 : 2*n-1]


def power_mean_objective_and_grad(z, p):
    """
    Compute objective and gradient for log-parameterized sequence.
    a = softmax(z), objective = 2*n*power_mean(autocorr(a), p)
    """
    n = len(z)
    # Softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Autocorrelation
    ac = fast_autocorr(a)
    ac = np.maximum(ac, 1e-30)
    
    # Power mean computation in log-space for numerical stability
    log_ac = np.log(ac)
    log_sum = p * log_ac.max() + np.log(np.sum(np.exp(p * log_ac - p * log_ac.max())))
    sum_ac_p = np.exp(log_sum)  # = sum ac^p
    
    obj_val = 2.0 * n * np.exp(log_sum / p)
    
    # Weights: w[k] = ac[k]^{p-1} / (sum ac^p)^{(p-1)/p}
    # In log-space: log_w[k] = (p-1)*log(ac[k]) - (p-1)/p * log_sum
    log_w = (p - 1.0) * log_ac - (p - 1.0) / p * log_sum
    w = np.exp(log_w)
    
    # Gradient of power mean wrt a
    grad_a = compute_autocorr_gradient(a, w)
    
    # Full gradient of 2n * power_mean
    full_grad_a = 2.0 * n * grad_a
    
    # Chain rule for softmax: grad_z = a * (grad_a - a @ grad_a)
    mean_g = float(a @ full_grad_a)
    grad_z = a * (full_grad_a - mean_g)
    
    return obj_val, grad_z


def norm_sum1(a):
    """Normalize to sum 1, ensure non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=100):
    """LP-based Frank-Wolfe direction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = evaluate_safe(best_a)
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < 1e-14:
            break
        
        # Build LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for top lags
        top_lags = np.argsort(ac)[::-1][:min(100, len(ac))]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[j] += a[ii]
            constraints.append(coeff @ b_var <= M)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        b_dir = b_val / b_sum
        
        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b_dir
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        score = evaluate_safe(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): score={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Aggressive coordinate descent with peak-aware mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = evaluate_safe(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        peak_lags = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        improved = False
        
        # Find hot positions
        hot = set()
        for k in peak_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot.add(i)
                j = k_int - i
                if 0 <= j < n:
                    hot.add(j)
        
        hot_list = list(hot)
        cold_list = [i for i in range(n) if i not in hot]
        
        if not hot_list or not cold_list:
            break
        
        # Try transfers
        n_check = min(len(hot_list), 20)
        rng.shuffle(hot_list)
        n_cold = min(len(cold_list), 50)
        rng.shuffle(cold_list)
        
        for src in hot_list[:n_check]:
            if a[src] < 1e-10:
                continue
            for dst in cold_list[:n_cold]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M - 1e-15:
                        a = a_trial
                        score = evaluate_safe(a)
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(max(3, n // 100)):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        M_new = fast_autocorr(a).max()
                        if M_new < M:
                            score = evaluate_safe(a)
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.1f}s): score={best_score:.6f}, stall={stall}")
    
    return best_a, best_score


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
