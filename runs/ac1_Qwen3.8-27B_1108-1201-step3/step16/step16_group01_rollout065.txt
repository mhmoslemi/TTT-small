"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
  1. Adam optimizer in log-space whose gradient is a softmax-smoothed
     autocorrelation "heat" map, computed via a single FFT cross-correlation
     per step:  g[i] = sum_k p[k] * a[k-i],  p = softmax(tau * ac).
  2. Adaptive temperature tau (ramps from smooth ~blurry to sharp ~subgradient).
  3. Periodic stochastic restarts from the incumbent to escape local minima.
  4. Peak-aware coordinate descent (move mass from hot -> cold positions)
     for cheap local polish.
  5. Multi-length exploration (n +/- 200) since 2*n*M changes with length.

All helpers are top-level, no closures, no lambdas. No file/network IO.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Linear autocorrelation conv(a,a) via zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad_fft(a, p):
    """
    Smoothed gradient direction g[i] = sum_k p[k] * a[k-i]  (i = 0..n-1),
    where p is the (non-negative) weight over all 2n-1 lags.
    Computed as the linear cross-correlation of p with a via
        IFFT( FFT(p) * conj(FFT(a)) )
    which (from the correlation theorem) equals sum_k p[k] * a[k-i].
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = len(a)
    p_len = len(p)
    size = 1 << (n + p_len - 1).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    g = np.fft.irfft(P * np.conj(A), size)[:n]
    return g


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_of(a):
    """Safely call the provided evaluate_sequence on a normalized list."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a = a / s
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Piecewise-linear resample of a normalized sequence to new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def cosine_init(n, rng):
    """A smooth, spread-out non-negative starter (raised cosine + floor)."""
    x = np.linspace(0.0, 1.0, n)
    v = (0.5 * (1.0 + np.cos(2.0 * np.pi * x))) ** 1.5 + 0.05
    v = v + rng.normal(0.0, 0.01, n)
    v = np.maximum(v, 0.0)
    v = v / v.sum()
    return v


def adam_logspace(a0, time_limit, lr=0.05, beta1=0.9, beta2=0.999,
                  tau0=8.0, tau_end=120.0, eps=1e-8,
                  restart_patience=3000, verbose=True, seed=42):
    """
    Adam in log-space minimizing the softmax-smoothed peak autocorrelation.
    a is kept normalized to sum 1; only best_m (peak of conv) is tracked,
    since score = 2*n*best_m  (with sum==1).
    """
    n = len(a0)
    a = norm_sum1(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0

    print_freq = 20000
    while True:
        if time.monotonic() - t0 > time_limit:
            break
        t += 1

        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-9))
        # ramp temperature: smooth early (explore), sharp late (exploit)
        tau = tau0 + (tau_end - tau0) * (frac ** 1.3)

        ac = fast_autocorr(a)
        M = float(ac.max())
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # softmax weights over lags (numerically stable)
        p = np.exp(tau * (ac - M))
        p_sum = float(p.sum())
        if p_sum < 1e-30:
            p = np.zeros_like(ac)
            p[int(ac.argmax())] = 1.0
        else:
            p = p / p_sum

        g = compute_grad_fft(a, p)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = lr / (1.0 + 0.0008 * t)

        log_a = np.log(np.maximum(a, 1e-15))
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - log_a.max()
        log_a = np.clip(log_a, -28.0, 0.0)
        a = np.exp(log_a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # stochastic restart from incumbent when stuck
        if stall > restart_patience:
            stall = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            noise = rng.normal(0.0, 0.02, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            if verbose:
                el = time.monotonic() - t0
                print(f"    Adam restart ({el:.1f}s): J={2.0*n*best_m:.6f}")

        if verbose and t % print_freq == 0:
            el = time.monotonic() - t0
            print(f"    Adam t={t} ({el:.1f}s): J={2.0*n*best_m:.6f} tau={tau:.1f} stall={stall}")

    return best_a, best_m


def coord_descent(a0, time_limit, verbose=True, seed=123):
    """Peak-aware coordinate descent: move small mass hot -> cold if it lowers the peak."""
    n = len(a0)
    a = norm_sum1(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 60
    fracs = (0.004, 0.01, 0.02, 0.04)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        if ac.max() < 1e-20:
            break

        # top peak lags
        order = np.argsort(ac)[::-1]
        top = order[:min(8, len(order))]

        # hot positions = those paired into the top peaks
        hot = np.zeros(n, dtype=bool)
        for k in top[:5]:
            k_int = int(k)
            i0 = max(0, k_int - (n - 1))
            i1 = min(n, k_int + 1)
            if i1 > i0:
                hot[i0:i1] = True

        cold = np.where(~hot)[0]
        if len(cold) == 0:
            cold = np.arange(n)

        hot_pos = np.where(hot)[0]
        improved = False

        if len(hot_pos) > 300:
            hot_pos = rng.choice(hot_pos, size=300, replace=False)

        for src in hot_pos:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            n_try = min(len(cold), 24)
            if n_try <= 0:
                break
            dsts = rng.choice(cold, size=n_try, replace=False)
            for dst in dsts:
                if dst == src:
                    continue
                if a[dst] > a[src]:
                    continue
                for f in fracs:
                    amt = a[src] * f
                    if amt < 1e-16:
                        continue
                    at = a.copy()
                    at[src] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    Mt = float(fast_autocorr(at).max())
                    if Mt < best_m - 1e-15:
                        a = at
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # escape: a few accepted-only random transfers
            n_pert = max(5, n // 40)
            for _ in range(n_pert):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                amt = a[i] * rng.uniform(0.02, 0.08)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s <= 1e-15:
                    continue
                a = a / s
                Mn = float(fast_autocorr(a).max())
                if Mn < best_m:
                    best_m = Mn
                    best_a = a.copy()
                    stall = 0
                    improved = True
                else:
                    a[j] -= amt
                    a[i] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()

        if verbose and stall % 20 == 0:
            el = time.monotonic() - t0
            print(f"    Coord ({el:.1f}s): J={2.0*n*best_m:.6f} stall={stall}")

    return best_a, best_m

    best_seq = norm_sum1(start)
