def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_convolution_matrix(a):
    """Build T where T[k,j] = a[j] if j <= k <= j+n-1, else 0."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j+n, j] = a[j]
    return T


def mirror_descent_nesterov(a0, time_limit, top_k=50, beta=5.0, verbose=True, seed=42):
    """Nesterov-accelerated mirror descent with adaptive peak weighting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0
    max_stall = 5000
    best_gamma = gamma
    
    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Compute gradient vectorized
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        # Multiplicative update
        log_ratio = log_ratio = -eta * grad
        # Clip to avoid overflow
        log_ratio = np.clip(log_ratio, -50, 50)
        a_new = y * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 500 == 0:
                gamma *= 0.95
                if gamma < 0.5:
                    # Restart from best
                    x = best_a.copy()
                    x_prev = best_a.copy()
                    gamma = best_gamma
                    stall_count = 0

        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.05, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            gamma = 4.0

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, gamma={gamma:.2f}")

    return best_a, best_m


def lp_frank_wolfe_fast(a0, time_limit, solver_name, verbose=True, max_iters=100000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = build_convolution_matrix(y)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m

def peak_targeted_coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent targeting peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 1000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        top_peaks = np.argsort(ac)[::-1][:5]
        
        improved = False
        for pk in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(pk)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Try moving mass from high-contrib to low-impact positions
            src = int(idx[np.argmax(contrib)])
            
            # Find best destination
            # Compute sensitivity
            sensitivity = np.zeros(n)
            for p in top_peaks[:3]:
                p_int = int(p)
                i_s = max(0, p_int - n + 1)
                i_e = min(n, p_int + 1)
                ii = np.arange(i_s, i_e)
                jj = p_int - ii
                sensitivity[ii] += a[jj]
            
            # Try a few random low-sensitivity destinations
            low_sens = np.argsort(sensitivity)[:min(100, n // 5)]
            rng_perm = rng.permutation(len(low_sens))
            for _ in range(min(20, len(low_sens))):
                dst = int(low_sens[rng_perm[_]])
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_convolution_matrix(a):
    """Build T where T[k,j] = a[j] if j <= k <= j+n-1, else 0."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j + n, j] = a[j]
    return T


def compute_gradient_peaks(ac, a, top_k=50, beta=5.0):
    """Compute weighted subgradient from top peaks of autocorrelation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_nesterov(a0, time_limit, top_k=50, beta=5.0, verbose=True, seed=42):
    """Nesterov-accelerated mirror descent with soft peak weighting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0
    max_stall = 5000

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        grad = compute_gradient_peaks(y, a, top_k, beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_ratio = -eta * grad
        log_ratio = np.clip(log_ratio, -50.0, 50.0)
        a_new = y * np.exp(log_ratio)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count % 1000 == 0:
                gamma *= 0.92
                if gamma < 0.3:
                    x = best_a.copy()
                    x_prev = best_a.copy()
                    gamma = 4.0
                    stall_count = 0

        if stall_count > max_stall:
            stall_count = 0
            sigma = 0.05 / (1 + 0.1 * stall_count / max_stall)
            perturbation = rng.normal(0, sigma, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            gamma = 4.0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, gamma={gamma:.3f}")

    return best_a, best_m

    n = len(a)

    return np.fft.irfft(A * A, fft_size)[:size]

def build_convolution_matrix(a):
    """Build T where T[k,j] = a[j] if j <= k <= j+n-1, else 0."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j + n, j] = a[j]

def compute_gradient_peaks(ac, a, top_k=50, beta=5.0):
    """Compute weighted subgradient from top peaks of autocorrelation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad
