    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=40, eps=1e-10, blend=0.6):
    """Flatten power spectrum via iterative DFT correction with adaptive blending."""
    a = normalize(a).copy()
    n = len(a)
    for i in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Adaptive blend: more aggressive early, more conservative late
        b = blend * (1.0 - 0.5 * i / max(n_iter, 1))
        a = (1.0 - b) * a + b * a_new
    return a


def multi_init(n, rng):
    """Try multiple initialization strategies and return the best."""
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Raised cosine
    x = np.arange(n)
    candidates.append((np.sin(np.pi * x / (n - 1)) ** 2 + 1e-15))
    
    # Hann window
    candidates.append(0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) + 1e-15)
    
    # Gaussian centered
    sigma = n / 4.0
    g = np.exp(-((x - (n - 1) / 2.0) ** 2) / (2 * sigma ** 2))
    candidates.append(g + 1e-15)
    
    # Two-peak (anti-symmetric energy)
    half = n // 2
    p1 = np.exp(-((x - half * 0.3) ** 2) / (2 * (n * 0.15) ** 2))
    p2 = np.exp(-((x - half * 1.7) ** 2) / (2 * (n * 0.15) ** 2))
    candidates.append(p1 + p2 + 1e-15)
    
    # Random
    for _ in range(3):
        candidates.append(rng.random(n) + 1e-8)
    
    best_a = None
    best_score = float('inf')
    for c in candidates:
        c = normalize(c)
        c = spectral_equalize(c, n_iter=15)
        sc = fast_autocorr(c).max() * 2 * n
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def solve_lp_direction(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized direction b/sum(b) or None on failure.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw_optimize(a0, time_limit, solver, top_k_init=40, top_k_max=120, verbose=True):
    """
    Accelerated LP Frank-Wolfe with adaptive constraint selection.
    Uses hierarchical constraint management: start tight, expand as needed.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    
    # Track constraint violations for adaptive expansion
    violation_count = 0
    last_k_expand = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        # Hierarchical constraint selection
        # Start with top-K, then check for violated constraints outside
        top_idx = np.argsort(ac)[::-1]
        active_lags = top_idx[:current_top_k]
        
        # Check constraints just outside the active set for violations
        if current_top_k < len(ac) - 1:
            next_idx = top_idx[current_top_k:current_top_k + 20]
            for kx in next_idx:
                if ac[kx] > M_curr * 0.95:
                    active_lags = np.append(active_lags, kx)
                    break
        
        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            direction = x.copy()  # fallback: no movement
        
        # Step size with Lswatch-Lloyd improvement
        alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        # Check improvement
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            violation_count = 0
        else:
            violation_count += 1
        
        # Adaptive K: expand when stalled, contract when improving
        if t % 30 == 0:
            if violation_count > 50 and current_top_k < top_k_max:
                current_top_k = min(current_top_k + 10, top_k_max)
                violation_count = 0
            elif violation_count == 0 and current_top_k > top_k_init:
                current_top_k = max(current_top_k - 5, top_k_init)
        
        # Early termination if converged
        if t > 100 and (M_curr - best_m) < 1e-10:
            break
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, K={current_top_k}")
    
    return best_a, best_m


def log_nesterov_optimize(a0, time_limit, top_k=50, beta=4.0, c0=3.0, verbose=True, rng_seed=123):
    """
    Nesterov-accelerated mirror descent in log-domain with adaptive smoothing.
    Uses softmax parameterization: a[i] = exp(x[i]) / sum(exp(x))
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Log-domain parameterization
    x = np.log(a + 1e-30)
    x_prev = x.copy()
    stall_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation in log-domain
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y_log = x + beta_t * (x - x_prev)
        else:
            y_log = x.copy()
        
        # Convert to probability
        y_log_shifted = y_log - y_log.max()
        y = np.exp(y_log_shifted)
        y = y / y.sum()
        y = np.maximum(y, 1e-30)
        
        # Compute gradient in probability space
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step in log-domain
        # Gradient of J w.r.t. x: grad_x[i] = grad_prob[i] * y[i] (by chain rule)
        # Actually, for softmax, the natural gradient is simpler
        eta = c0 / (np.sqrt(t) + 1.0)
        x_new = y_log - eta * grad / (y + 1e-30)
        
        # Normalize
        x_new_shifted = x_new - x_new.max()
        a_new = np.exp(x_new_shifted)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = y_log
        x = x_new
        a = a_new
        
        # Check
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count > 2000:
            stall_count = 0
            pert = rng.normal(0, 0.003, n)
            x_new_log = np.log(best_a + 1e-30) + pert
            x = x_new_log - x_new_log.max()
            x += np.exp(x - np.exp(x).max()).sum()
            x_prev = x.copy()
            a = np.exp(x - x.max())
            a = a / a.sum()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LogNSG t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination using sensitivity analysis
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(300, len(candidates))]]
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.10]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and not improved:
                pass
        
        if not improved:
            stall += 1
            # Global perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def simulated_annealing(a0, time_limit, rng_seed=999, verbose=True):
    """
    Simulated annealing in the sequence space.
    Proposes random perturbations and accepts based on temperature schedule.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    T = 0.001  # initial temperature
    T_min = 1e-6
    cool_rate = 0.999
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            break
        
        # Propose perturbation: move mass from one position to another
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = rng.uniform(0.001, 0.05)
        amt = a[i] * frac
        if amt < 1e-18:
            continue
        
        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        
        M_trial = fast_autocorr(a_trial).max()
        delta = M_trial - a.sum() * 0  # we compare max values directly
        
        # Accept if better or with probability exp(-delta/T)
        if M_trial < best_m:
            a = a_trial
            best_m = M_trial
            best_a = a_trial.copy()
        elif M_trial <= fast_autocorr(a).max() or rng.random() < np.exp(-(M_trial - fast_autocorr(a).max()) / (T + 1e-10)):
            a = a_trial
        
        T *= cool_rate
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA t={t} ({elapsed:.1f}s): J={J:.6f}, T={T:.2e}")
    
    return best_a

    # Try slightly longer

    if n_alt1 != n_best and remaining > 100:
        print(f"\n=== Phase 7a: Alt length n={n_alt1} ===")
        a_long = interpolate_sequence(best_seq, n_alt1)
        a_long = spectral_equalize(a_long, n_iter=10)
        t7a = min(remaining * 0.25, 150.0)
        if t7a > 20:
            a_in = normalize(a_long)
            a_out, _ = lp_fw_optimize(a_in, t7a, solver, top_k_init=40, top_k_max=80, verbose=False)
            try_update(a_out, f"lpfw_n{n_alt1}")
    
    # Try slightly shorter
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt2 = n_best - 200
    if n_alt2 >= 800 and n_alt2 != n_best and remaining > 80:
        print(f"\n=== Phase 7b: Alt length n={n_alt2} ===")
        a_short = interpolate_sequence(best_seq, n_alt2)
        a_short = spectral_equalize(a_short, n_iter=10)
        t7b = min(remaining * 0.25, 120.0)
        if t7b > 20:
            a_in = normalize(a_short)
            a_out, _ = lp_fw_optimize(a_in, t7b, solver, top_k_init=40, top_k_max=80, verbose=False)
            try_update(a_out, f"lpfw_n{n_alt2}")
    
    # === Phase 8: Intensive final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW n={n_best}, budget={remaining:.0f}s ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 30:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t8, solver, top_k_init=100, top_k_max=180, verbose=True)
        try_update(a_out, f"lpfw_final_n{n_best}")
    
    # === Phase 9: Final coordinate polish (intensive) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Coord n={n_best}, budget={remaining:.0f}s ===")
    t9 = min(remaining * 0.30, 100.0)
    if t9 > 10:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t9, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 10: Last LP squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 10: Last LP n={n_best} ===")
        t10 = min(remaining * 0.70, 80.0)
        if t10 > 15:
            a_in = normalize(best_seq)
            a_out, _ = lp_fw_optimize(a_in, t10, solver, top_k_init=120, top_k_max=200, verbose=False)
            try_update(a_out, f"lpfw_last_n{n_best}")
    
    # === Phase 11: Ultra-fine coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t11 = min(remaining * 0.80, 50.0)
        if t11 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t11, verbose=False, rng_seed=1234)
            try_update(a_out, f"coord_ultra_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
