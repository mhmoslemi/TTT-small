def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            stall_count = 0

    K_growth_threshold = 30

    b_var = cp.Variable(n, nonneg=True)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive constraint selection: include all lags above a threshold
        sorted_ac = np.sort(ac)[::-1]
        # Use a dynamic threshold: include lags that are within (1 - margin) of the peak
        margin = max(0.05, 0.2 * (1.0 - 0.95 ** (t / 200.0)))
        threshold = M_curr * (1.0 - margin)
        active_mask = ac >= threshold
        n_active = int(active_mask.sum())

        # Blend between top-K and threshold-based selection
        K_use = min(max(cur_top_K, n_active), top_K_max, 2 * n - 1)

        # Get the indices of the most binding constraints
        if K_use >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_use]

        # Build constraint matrix
        T_sel = np.zeros((len(binding_idx), n))
        for row_idx, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            T_sel[row_idx, i_start:i_end] = y[i_start:i_end]

        M_sel = np.array([ac[k] for k in binding_idx])
        # Use a slightly tight bound to ensure feasibility
        M_sel = M_sel + 1e-8

        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if nesterov:
            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
        else:
            x_new = y_new

        prev_y = y
        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            no_improve = 0

        # Adaptive K growth
        if no_improve > K_growth_threshold and cur_top_K < top_K_max:
            cur_top_K = min(cur_top_K + 50, top_K_max)
            no_improve = 0

        if verbose and (t % 5 == 0 or t <= 3):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={cur_top_K}, n_active={n_active}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with sensitivity-based 2-point moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved_this_round = False

        for k_idx in range(len(top_k)):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_k[k_idx])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute cost for each destination
            n_sens = min(15, n_top)
            scores = np.zeros(n)
            for k2 in top_k[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    scores[d] += add

            # Find best destination (low score, not same as source)
            scores[src] = np.inf
            dst = int(np.argmin(scores))

            # Try multiple fractions
            found = False
            for frac in [0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.15]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization focused on LP-FW."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_init_halfsine(n_base, power=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Quick multi-init exploration ===
    print(f"\n=== Phase 1: Multi-init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.12, 180.0)

    if t_phase1 > 30:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_init_halfsine(n0, alpha=0.0, power=1.0))
        labels.append("halfsine")

        inits.append(make_init_halfsine(n0, alpha=2.0, power=1.5))
        labels.append("halfsine_p15")

        inits.append(make_init_parabola(n0, edge=0.01))
        labels.append("parabola")

        inits.append(make_init_ramp_asym(n0, skew=0.4))
        labels.append("asym_ramp")

        inits.append(make_init_ramp_asym(n0, skew=-0.3))
        labels.append("asym_ramp2")

        # Blend parent with different shapes
        if parent is not None:
            for bl in [0.3, 0.5, 0.7]:
                blend = bl * norm_sum1(parent) + (1 - bl) * make_init_halfsine(n0, power=1.2)
                inits.append(norm_sum1(blend))
                labels.append(f"blend_{bl}")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  SGD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = smooth_gradient_optimize(init, t_avail, top_k=40, beta=4.0, c0=4.0,
                                                 restart_every=3000, verbose=False)
            try_update(a_out, f"sgd_{lbl}")

    # === Phase 2: Main LP-FW on best sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K_init=150, top_K_max=600, nesterov=True)
        try_update(a_out, f"lp_main_n{len(a_in)}")

    # === Phase 3: Smooth gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: SGD polish (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_gradient_optimize(a_in, t3, top_k=80, beta=3.5, c0=3.0,
                                             restart_every=8000, verbose=False)
        try_update(a_out, f"sgd_polish_n{len(a_in)}")

    # === Phase 4: Second LP-FW round ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW second round (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K_init=300, top_K_max=1000, nesterov=True)
        try_update(a_out, f"lp_r2_n{len(a_in)}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord descent (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length sweep (base n={n_best}) ===")

    candidate_lengths = []
    for delta in [-200, -100, 100, 200, 400, 800]:
        nl = n_best + delta
        if 1500 <= nl <= 5000:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths))

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        t_lp = min(remaining * 0.25, 120.0)
        if t_lp > 15:
            a_out, _ = lp_frank_wolfe(a_long, t_lp, solver, verbose=False, top_K_init=200, top_K_max=500, nesterov=True)
            try_update(a_out, f"lp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_sgd = min(remaining * 0.10, 60.0)
            if t_sgd > 10:
                a_in2 = norm_sum1(best_seq)
                a_interpolated = interpolate_sequence(a_in2, new_n)
                a_out, _ = smooth_gradient_optimize(a_interpolated, t_sgd, top_k=50, beta=4.0, c0=3.0,
                                                     restart_every=5000, verbose=False)
                try_update(a_out, f"sgd_n{new_n}")

    # === Phase 7: Final LP-FW on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K_init=400, top_K_max=1200, nesterov=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
    if remaining > 15:
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_init_halfsine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
