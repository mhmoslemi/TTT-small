    return pnorm_val, grad

def pnorm_gradient_descent(a0, time_limit, p_init=4.0, p_max=32.0, 
                            top_k_init=30, top_k_max=80,
                            lr=0.01, nesterov=True, 
                            restart_every=20000, verbose=False):
    """Adaptive p-norm smoothed gradient descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    
    # Adaptive parameters
    p_curr = p_init
    top_k_curr = top_k_init
    stall = 0
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Schedule: increase p and top_k over time
        progress = min(1.0, t / (restart_every * 2))
        p_sched = p_init + (p_max - p_init) * progress
        top_k_sched = int(top_k_init + (top_k_max - top_k_init) * progress)
        
        # Nesterov extrapolation
        if nesterov and t > 1:
            beta = (t - 1.0) / (t + 1.5)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()
            sy = 1.0
        
        # Compute gradient
        ac = fast_autocorr(y)
        pnorm_val, grad = pnorm_smooth_max(y, ac, p=p_sched, top_k=top_k_sched)
        
        # Gradient step with adaptive learning rate
        eta = lr / (np.sqrt(t) + 1.0)
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        a_new = y - eta * grad
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with perturbation
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            p_curr = max(p_init, p_sched - p_init * 0.5)
            top_k_curr = max(top_k_init, top_k_sched - 10)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      PGD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PGD t={t} ({elapsed:.0f}s): J={J:.6f}, p={p_sched:.1f}, top_k={top_k_sched}")
    
    return best_a, best_m

def peak_chasing_cd(a0, time_limit, verbose=False, rng_seed=42):
    """Peak-targeted coordinate descent with sensitivity analysis."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    n_top_lags = 15
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top contributing lags
        top_lags = np.argsort(ac)[::-1][:n_top_lags]
        
        improved = False
        
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            
            if i_start >= i_end:
                continue
            
            # Find the pair with maximum contribution at this lag
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib_idx = np.argmax(contrib)
            src = int(idx[max_contrib_idx])
            dst_partner = k_int - src
            src_val = a[src]
            
            if src_val < 1e-12:
                continue
            
            # Compute sensitivity: how much does moving mass from src to position d
            # increase the peak?
            # For each candidate destination d, compute the increase in top lags
            n_cand = min(30, n)
            # Sample candidate positions
            if n <= 30:
                candidates = np.arange(n)
            else:
                candidates = rng.choice(n, size=n_cand, replace=False)
            
            best_dst = -1
            best_score = float('inf')
            
            for d in candidates:
                d_int = int(d)
                if d_int == src:
                    continue
                # Compute how much this move would increase the top lags
                penalty = 0.0
                for k2 in top_lags[:5]:
                    k2_int = int(k2)
                    # Moving mass from src to d changes conv at lag k2 by:
                    # -a[src]*a[k2-src] + a[d]*a[k2-d] (approximately, for small moves)
                    # The change is proportional to a[k2-d] - a[k2-src] (times the moved mass)
                    delta = 0.0
                    if 0 <= k2_int - src < n:
                        delta -= a[k2_int - src]
                    if 0 <= k2_int - d_int < n:
                        delta += a[k2_int - d_int]
                    penalty += abs(max(delta, 0))
                
                if penalty < best_score:
                    best_score = penalty
                    best_dst = d_int
            
            if best_dst < 0:
                continue
            
            # Try moving a small fraction
            found = False
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found:
                # Try random small perturbation
                if stall > 100:
                    j = int(rng.integers(n))
                    if j != src and a[src] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def spectral_equalization(a0, time_limit, verbose=False):
    """Iteratively equalize the autocorrelation using spectral methods."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
        
        # Find the target level (slightly above mean)
        target = ac.mean() * 1.5
        
        # Find lags above target
        above = np.where(ac > target)[0]
        if len(above) == 0:
            break
        
        # For each above-target lag, compute the "pressure" gradient
        grad = np.zeros(n)
        for k in above:
            k_int = int(k)
            excess = ac[k_int] - target
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            for i in range(i_start, i_end + 1):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += excess * a[j]
        
        # Apply small negative gradient step
        lr = 0.01
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < M_curr:
            a = a_new
        else:
            # Try smaller step
            lr = lr * 0.5
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            M_new = fast_autocorr(a_new).max()
            if M_new < M_curr:
                a = a_new
            else:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.005
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    if verbose and it % 500 == 0:
                        print(f"      SpecEq it={it}: J={2.0*n*best_m:.6f}")
    
    return best_a, best_m

For asymmetric sequences, the center lag is sum a[i]*a[n-1-i], which by Cauchy-Schwarz is at most (sum a[i]^2)^{1/2} * (sum a[n-1-i]^2)^{1/2} = sum a[i]^2. And sum a[i]^2 >= 1/n.

This is exactly what the optimization is trying to find! A sequence that is spread out enough to keep lag 0 small, but asymmetric enough to keep the center lag small.

"""

3. Peak-targeted coordinate descent with sensitivity-based move selection

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_chirp_window(n, asym=0.3, spectral_amp=0.05, seed=0):
    """Create a chirp-windowed sequence with spectral perturbation."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    # Asymmetric linear ramp (creates anti-correlation with reverse)
    ramp = 1.0 + asym * (2.0 * x / (n - 1) - 1.0)
    # Ensure positivity
    ramp = np.maximum(ramp, 0.1)
    # Add spectral perturbation for flat power spectrum
    n_terms = min(30, n // 8)
    perturb = np.zeros(n)
    for _ in range(n_terms):
        freq = int(rng.integers(1, max(2, n // 3)))
        amp = rng.uniform(spectral_amp * 0.5, spectral_amp * 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        perturb += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = ramp * (1.0 + perturb)
    a = np.maximum(a, 1e-6)
    return norm_sum1(a)


def make_sinusoidal(n, freq=1.0, floor=0.05):
    """Sinusoidal initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * freq * x) ** 2 + floor
    return norm_sum1(a)


def make_flat_spectrum(n, seed=0):
    """Sequence designed for flat power spectrum via random cosine sum."""
    rng = np.random.default_rng(seed)
    a = np.ones(n, dtype=np.float64)
    n_terms = min(40, n // 6)
    for _ in range(n_terms):
        freq = int(rng.integers(1, max(2, n // 2)))
        amp = rng.uniform(0.02, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def compute_pnorm_gradient(a, w, top_lags):
    """Compute weighted gradient of autocorrelation peak."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_lags):
        if wi < 1e-15:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def annealed_pgd(a0, time_limit, p_start=4.0, p_end=64.0,
                 top_k_start=30, top_k_end=80,
                 lr_base=0.015, nesterov=True,
                 restart_every=15000, verbose=False, rng_seed=42):
    """Annealed p-norm gradient descent with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    total_iters = restart_every * 4

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Annealing schedule
        progress = min(1.0, float(t) / float(total_iters))
        p_curr = p_start + (p_end - p_start) * progress ** 0.7
        top_k_curr = int(top_k_start + (top_k_end - top_k_start) * progress)

        # Nesterov extrapolation
        if nesterov and t > 1:
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        top_idx = np.argsort(ac)[::-1][:top_k_curr]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** p_curr
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k_curr) / top_k_curr
        else:
            w = w / w_sum

        grad = compute_pnorm_gradient(y, w, top_idx)

        # Adaptive learning rate
        eta = lr_base / (np.sqrt(t) + 1.0) * (1.0 + 0.3 * (1.0 - progress))
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        a_new = y - eta * grad
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with random perturbation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.01 * (1.0 - progress)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J_val = 2.0 * n * best_m
                print(f"        PGD restart ({elapsed:.0f}s): J={J_val:.6f}, p={p_curr:.1f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J_val = 2.0 * n * best_m
            print(f"        PGD t={t} ({elapsed:.0f}s): J={J_val:.6f}, p={p_curr:.1f}, top_k={top_k_curr}")

    return best_a, best_m


def peak_cd(a0, time_limit, verbose=False, rng_seed=42):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    max_stall = 800
    n_top = min(20, 2 * n - 1)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            max_ci = np.argmax(contrib)
            src = int(idx[max_ci])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute sensitivity for candidate destinations
            n_cand = min(50, n)
            if n <= 50:
                candidates = np.arange(n)
            else:
                candidates = rng.choice(n, size=n_cand, replace=False)

            best_dst = -1
            best_penalty = float('inf')

            for d in candidates:
                d_int = int(d)
                if d_int == src:
                    continue
                penalty = 0.0
                for k2 in top_lags[:6]:
                    k2_int = int(k2)
                    delta = 0.0
                    if 0 <= k2_int - src < n:
                        delta -= a[k2_int - src]
                    if 0 <= k2_int - d_int < n:
                        delta += a[k2_int - d_int]
                    if delta > 0:
                        penalty += delta * delta
                if penalty < best_penalty:
                    best_penalty = penalty
                    best_dst = d_int

            if best_dst < 0:
                continue

            # Try multiple move fractions
            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.25, 0.5]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Escape: random pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:

                    best_a = a.copy()

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J_val = 2.0 * n * best_m
            print(f"        CD ({elapsed:.0f}s): J={J_val:.6f}, stall={stall}")

    return best_a, best_m


def spectral_pressure(a0, time_limit, verbose=False, rng_seed=77):
    """Spectral pressure equalization: push mass from hot lags to cold positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        # Target: slightly above mean
        target = ac.mean() * 1.3

        # Find lags above target
        above_mask = ac > target
        if not np.any(above_mask):
            # Try random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.003
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            continue

        # Compute pressure gradient from above-target lags
        above_lags = np.where(above_mask)[0]
        # Only use top few to limit computation
        top_above = above_lags[np.argsort(ac[above_lags])[::-1]][:5]

        grad = np.zeros(n, dtype=np.float64)
        for k in top_above:
            k_int = int(k)
            excess = ac[k_int] - target
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            grad[idx] += excess * a[j_idx]

        # Apply step
        lr = 0.008
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < M_curr:
            a = a_new
        else:
            # Halve step
            a_new = a - lr * 0.5 * grad
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            M_new = fast_autocorr(a_new).max()
            if M_new < M_curr:
                a = a_new
            else:
                # Small random nudge
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.002
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J_val = 2.0 * n * best_m
            print(f"        SpecPress it={it} ({elapsed:.0f}s): J={J_val:.6f}")

    return best_a, best_m

    best_name = 'current_best'
    init_ranks.insert(0, (2.0 * n0 * fast_autocorr(state['best_seq']).max(), best_name))

    t_per = max(40.0, t_available_p2 / 4.0)
    runs = 0
    for j_val, name in init_ranks[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        t_avail = min(t_per, remaining - 80)
        if t_avail < 30:
            break
        runs += 1
        if name == best_name:
            init_arr = norm_sum1(state['best_seq'])
        else:
            init_arr = inits[name]
        n_init = len(init_arr)
        print(f"  PGD run {runs}: '{name}' n={n_init} (t={t_avail:.0f}s)")
        a_out, _ = annealed_pgd(init_arr, t_avail,
                                  p_start=4.0, p_end=64.0,
                                  top_k_start=30, top_k_end=80,
                                  lr_base=0.015, verbose=False,
                                  rng_seed=42 + runs)
        try_update(a_out, f"pgd_{name}")

    # === Phase 3: Peak-chasing CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.22, 280.0)
    if t3 > 30:
        n_best = len(state['best_seq'])
        print(f"\n=== Phase 3: Peak CD (n={n_best}, t={t3:.0f}s) ===")
        a_out, _ = peak_cd(norm_sum1(state['best_seq']), t3, verbose=False, rng_seed=42)
        try_update(a_out, "peak_cd")

    # === Phase 4: Spectral pressure ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        n_best = len(state['best_seq'])
        print(f"\n=== Phase 4: Spectral Pressure (n={n_best}, t={t4:.0f}s) ===")
        a_out, _ = spectral_pressure(norm_sum1(state['best_seq']), t4, verbose=False, rng_seed=77)
        try_update(a_out, "spec_press")

    # === Phase 5: PGD polish with high p ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        n_best = len(state['best_seq'])
        print(f"\n=== Phase 5: PGD Polish (n={n_best}, t={t5:.0f}s) ===")
        a_out, _ = annealed_pgd(norm_sum1(state['best_seq']), t5,
                                  p_start=16.0, p_end=128.0,
                                  top_k_start=50, top_k_end=100,
                                  lr_base=0.005, verbose=False,
                                  rng_seed=314)
        try_update(a_out, "pgd_polish")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['best_seq'])
    print(f"\n=== Phase 6: Length Exploration (base n={n_best}) ===")

    lengths_to_try = []
    for delta in [400, -400, 800, -800, 1200, -1200]:
        new_n = n_best + delta
        if 1600 <= new_n <= 5000 and new_n != n_best:
            lengths_to_try.append(new_n)

    for new_n in lengths_to_try[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_seq(state['best_seq'], new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t_sub = min(100.0, remaining - 40)
        if t_sub > 30:
            a_out, _ = annealed_pgd(a_interp, t_sub,
                                      p_start=4.0, p_end=48.0,
                                      top_k_start=30, top_k_end=70,
                                      lr_base=0.01, verbose=False,
                                      rng_seed=new_n)
            try_update(a_out, f"pgd_n{new_n}")

    # === Phase 7: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.25, 300.0)
    if t7 > 25:
        n_best = len(state['best_seq'])
        print(f"\n=== Phase 7: Final CD (n={n_best}, t={t7:.0f}s) ===")
        a_out, _ = peak_cd(norm_sum1(state['best_seq']), t7, verbose=False, rng_seed=999)
        try_update(a_out, "final_cd")

    # === Phase 8: Final PGD squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.35, 200.0)
    if t8 > 25:
        n_best = len(state['best_seq'])
        print(f"\n=== Phase 8: Final PGD Squeeze (n={n_best}, t={t8:.0f}s) ===")
        a_out, _ = annealed_pgd(norm_sum1(state['best_seq']), t8,
                                  p_start=32.0, p_end=256.0,
                                  top_k_start=60, top_k_end=120,
                                  lr_base=0.003, verbose=False,
                                  rng_seed=777)
        try_update(a_out, "final_pgd")

    # === Final validation ===
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = fs
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3200
        state['best_seq'] = make_sinusoidal(n_fb, freq=1.0)
        state['best_score'] = evaluate_safe(state['best_seq'])
        print(f"Fallback: score={state['best_score']:.6f}")

    return [float(x) for x in state['best_seq']]
