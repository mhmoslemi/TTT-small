def fast_autocorr(a):
    """Compute full autocorrelation using FFT. Returns array of length 2n-1."""

    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # We need conv of a_rev with w, then take elements [n-1 : 2n-1]
    a_rev = a[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    return conv[n-1:2*n-1]


def soft_max_weights(ac, t):
    """Compute softmax weights for the autocorrelation peaks."""
    ac_shifted = t * (ac - ac.max())
    exp_ac = np.exp(ac_shifted)
    w = exp_ac / exp_ac.sum()
    return w


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() == 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_max_gradient_descent(a0, time_limit, t_init=20.0, t_final=500.0, 
                                verbose=True, rng_seed=42):
    """
    Nesterov-accelerated projected gradient descent with soft-max smoothing.
    Uses exact gradient computed via FFT.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()
    
    # Step size (inverse of Lipschitz constant approximation)
    # Lipschitz constant of the gradient is roughly 2*max(a) * (2n-1) / (sum a)^2
    # But we'll use adaptive step size
    gamma = 2.0  # base step size parameter
    stall_count = 0
    max_stall = 5000
    
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        t_temp = t_init * (t_final / t_init) ** progress
        
        # Nesterov extrapolation
        if total_iters == 0:
            y = x.copy()
        else:
            beta = (total_iters - 0.5) / (total_iters + 1.5)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        w = soft_max_weights(ac, t_temp)
        grad = compute_autocorr_gradient(y, w)
        
        # Adaptive step size (backtracking)
        eta = gamma / (total_iters + 1.0)
        
        # Gradient step
        x_new = y - eta * grad
        x_new = project_simplex(x_new)
        
        # Check improvement
        M_new = fast_autocorr(x_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        x_prev = x
        x = x_new
        total_iters += 1
        
        # Restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            # Perturb from best
            pert = rng.normal(0, 0.01, n)
            a_restart = best_a * (1 + pert)
            a_restart = np.maximum(a_restart, 1e-10)
            a_restart = a_restart / a_restart.sum()
            x = a_restart
            x_prev = a_restart
        
        if verbose and total_iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD iter={total_iters} ({elapsed:.1f}s): J={J:.6f}, t={t_temp:.1f}")
    
    return best_a, best_m


def multiplicative_update(a0, time_limit, alpha=0.5, verbose=True, rng_seed=42):
    """
    Multiplicative weight update for flattening the effective gradient.
    a_j <- a_j * h_j^(-alpha) where h_j = sum_k w_k * a[k-j]
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    iters = 0
    t_temp = 50.0
    
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        w = soft_max_weights(ac, t_temp)
        h = compute_autocorr_gradient(a, w)
        h = np.maximum(h, 1e-20)
        
        # Multiplicative update
        a_new = a * h ** (-alpha)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        iters += 1
        
        # Slowly increase temperature
        t_temp = min(t_temp * 1.001, 500.0)
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MWU iter={iters} ({elapsed:.1f}s): J={J:.6f}, t={t_temp:.1f}")
    
    return best_a, best_m


def build_convolution_matrix(a):
    """Build the matrix T where T[k,j] = a[j] if j <= k <= j+n-1, else 0."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j+n, j] = a[j]
    return T


def lp_frank_wolfe_fast(a0, time_limit, solver_name, verbose=True, max_iters=10000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    iter_count = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix
        T = build_convolution_matrix(y)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated update
        alpha_step = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_step) * y + alpha_step * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        return float('inf')

def fast_autocorr(a):
    """Compute full autocorrelation using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_autocorr_gradient(a, w):
    """
    Compute grad[m] = 2 * sum_{i=0}^{n-1} a[i] * w[m+i] for m=0,...,n-1.
    This is d(sum_k w[k]*(a*a)[k])/da[m].
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    a_rev = a[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    # grad[m] = 2 * conv[m + n - 1]
    return 2.0 * conv[n - 1:2 * n - 1]


def soft_max_weights(ac, t):
    """Compute softmax weights for the autocorrelation peaks."""
    ac_shifted = t * (ac - ac.max())
    exp_ac = np.exp(ac_shifted)
    w = exp_ac / exp_ac.sum()
    return w

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def norm_sum1(a):
    """Normalize to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def nesterov_projected_gradient(a0, time_limit, t_init=10.0, t_final=500.0,
                                 gamma=1.0, verbose=True, rng_seed=42):
    """
    Nesterov-accelerated projected gradient descent with soft-max smoothing.
    Uses exact gradient computed via FFT. O(n log n) per iteration.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    total_iters = 0
    best_eta = gamma

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        t_temp = t_init * (t_final / t_init) ** progress

        # Nesterov extrapolation
        if total_iters == 0:
            y = x.copy()
        else:
            beta = (total_iters - 0.5) / (total_iters + 1.5)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute gradient
        ac = fast_autocorr(y)
        w = soft_max_weights(ac, t_temp)
        grad = compute_autocorr_gradient(y, w)

        # Adaptive step size
        eta = gamma / (np.sqrt(total_iters + 1.0) + 1.0)

        # Gradient step with projection
        x_new = y - eta * grad
        x_new = project_simplex(x_new)

        # Evaluate
        M_new = fast_autocorr(x_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Reduce step size if stuck
        if stall_count > max_stall:
            stall_count = 0
            gamma *= 0.8
            # Random restart from best
            if rng.random() < 0.3:
                scale = rng.uniform(0.5, 1.5)
                a_restart = best_a * scale
                a_restart = np.maximum(a_restart, 1e-10)
                a_restart = a_restart / a_restart.sum()
                x = a_restart
                x_prev = a_restart
            else:
                pert = rng.normal(0, 0.02, n)
                a_restart = best_a * (1.0 + pert)
                a_restart = np.maximum(a_restart, 1e-10)
                a_restart = a_restart / a_restart.sum()
                x = a_restart
                x_prev = a_restart

        x_prev = x
        x = x_new
        total_iters += 1

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NPG iter={total_iters} ({elapsed:.1f}s): J={J:.6f}, t={t_temp:.1f}, eta={eta:.4f}")

    return best_a, best_m


def mirror_descent_softmax(a0, time_limit, t_init=50.0, t_final=1000.0,
                           eta0=0.1, verbose=True, rng_seed=42):
    """
    Mirror descent (multiplicative weights) with soft-max smoothing.
    a[j] <- a[j] * exp(-eta * g[j]), then normalize.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    total_iters = 0
    stall_count = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))
        t_temp = t_init * (t_final / t_init) ** progress

        # Compute gradient
        ac = fast_autocorr(a)
        w = soft_max_weights(ac, t_temp)
        grad = compute_autocorr_gradient(a, w)

        # Adaptive step size
        eta = eta0 / (np.sqrt(total_iters + 1.0) + 1.0)

        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            eta0 *= 0.7
            # Restart
            if rng.random() < 0.5:
                a = best_a.copy()
            else:
                pert = rng.normal(0, 0.03, n)
                a = best_a * (1.0 + pert)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()

        a = a_new
        total_iters += 1

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={total_iters} ({elapsed:.1f}s): J={J:.6f}, t={t_temp:.1f}")

    return best_a, best_m


def build_convolution_matrix(a):
    """Build the (2n-1) x n convolution matrix."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n))
    for j in range(n):
        T[j:j + n, j] = a[j]
    return T


def lp_frank_wolfe_fast(a0, time_limit, solver_name, verbose=True, max_iters=5000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    iter_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix
        T = build_convolution_matrix(y)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated update
        alpha_step = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_step) * y + alpha_step * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def exact_subgradient_step(a, top_k=20):
    """
    Compute exact subgradient (uniform over top-k peaks) and return the
    steepest descent direction on the simplex (i.e., the LP-FW vertex without solving LP).
    """
    n = len(a)
    ac = fast_autocorr(a)
    # Use top-k peaks with weighting
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w_full = np.zeros(len(ac))
    w_full[top_idx] = top_vals
    w_full = w_full / w_full.sum()
    grad = compute_autocorr_gradient(a, w_full)
    return grad


def subgradient_polish(a0, time_limit, top_k=30, verbose=True, rng_seed=42):
    """
    Subgradient method with averaging for the exact max.
    Uses Polyak step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Running average
    avg_a = a.copy()
    total_weight = 0.0
    x = a.copy()
    x_prev = a.copy()
    total_iters = 0
    gamma = 1.0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        if total_iters > 0:
            beta = (total_iters - 0.5) / (total_iters + 1.5)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        grad = exact_subgradient_step(y, top_k)
        eta = gamma / (np.sqrt(total_iters + 1.0) + 1.0)

        x_new = y - eta * grad
        x_new = project_simplex(x_new)

        # Update running average
        w_t = 2.0 / (total_iters + 2.0)
        avg_a = (total_weight * avg_a + w_t * x_new) / (total_weight + w_t)
        total_weight += w_t

        M_new = fast_autocorr(x_new).max()
        M_avg = fast_autocorr(avg_a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x_new.copy()
            stall = 0
        elif M_avg < best_m - 1e-15:
            best_m = M_avg
            best_a = avg_a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            stall = 0
            gamma *= 0.9
            if rng.random() < 0.3:
                x = best_a.copy()
                x_prev = best_a.copy()
            avg_a = best_a.copy()
            total_weight = 0.0

        x_prev = x
        x = x_new
        total_iters += 1

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad iter={total_iters} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies:
    1. Nesterov projected gradient with soft-max (fast, O(n log n) per iter)
    2. Mirror descent with soft-max (multiplicative weights)
    3. Subgradient polish with averaging
    4. LP-Frank-Wolfe for vertex exploration
    5. Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    # Track best
    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Helper to check and update best (inlined to avoid closure)
    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Nesterov projected gradient (low temp) ~250s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print(f"\n=== Phase 1: NPG (t=5->50) n={n0} ===")
        t1 = min(remaining * 0.15, 250.0)
        a_out, _ = nesterov_projected_gradient(norm_sum1(best_seq), t1,
                                                t_init=5.0, t_final=50.0,
                                                gamma=1.5, verbose=True, rng_seed=42)
        check_update(a_out, "npg_p1")

    # === Phase 2: Mirror descent (medium temp) ~150s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 2: MD (t=20->100) n={len(best_seq)} ===")
        t2 = min(remaining * 0.08, 150.0)
        a_out, _ = mirror_descent_softmax(norm_sum1(best_seq), t2,
                                          t_init=20.0, t_final=100.0,
                                          eta0=0.05, verbose=True, rng_seed=17)
        check_update(a_out, "md_p2")

    # === Phase 3: NPG (medium-high temp) ~250s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print(f"\n=== Phase 3: NPG (t=30->200) n={len(best_seq)} ===")
        t3 = min(remaining * 0.15, 250.0)
        a_out, _ = nesterov_projected_gradient(norm_sum1(best_seq), t3,
                                                t_init=30.0, t_final=200.0,
                                                gamma=1.0, verbose=True, rng_seed=123)
        check_update(a_out, "npg_p3")

    # === Phase 4: LP-FW main ~400s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW n={n_cur} ===")
        t4 = min(remaining * 0.22, 400.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t4, solver, verbose=True)
        check_update(a_out, f"lp_p4_n{n_cur}")

    # === Phase 5: NPG (high temp) ~200s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print(f"\n=== Phase 5: NPG (t=100->500) n={len(best_seq)} ===")
        t5 = min(remaining * 0.12, 200.0)
        a_out, _ = nesterov_projected_gradient(norm_sum1(best_seq), t5,
                                                t_init=100.0, t_final=500.0,
                                                gamma=0.5, verbose=True, rng_seed=456)
        check_update(a_out, "npg_p5")

    # === Phase 6: Subgradient polish ~150s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 6: SubGrad polish n={len(best_seq)} ===")
        t6 = min(remaining * 0.08, 150.0)
        a_out, _ = subgradient_polish(norm_sum1(best_seq), t6, top_k=30,
                                      verbose=True, rng_seed=789)
        check_update(a_out, "subg_p6")

    # === Phase 7: LP-FW round 2 ~200s ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 7: LP-FW r2 n={n_cur} ===")
        t7 = min(remaining * 0.15, 200.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t7, solver, verbose=True)
        check_update(a_out, f"lp_p7_n{n_cur}")

    # === Phase 8: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Multi-n exploration ===")
        test_ns = []
        for delta in [200, -200, 400, -400, 600]:
            nt = n_best + delta
            if 500 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)

        for nt in test_ns[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            # Quick NPG at new length
            t_per = min(remaining * 0.12, 100.0)
            a_out, _ = nesterov_projected_gradient(norm_sum1(a_inter), t_per,
                                                    t_init=50.0, t_final=300.0,
                                                    gamma=1.0, verbose=False, rng_seed=nt)
            check_update(a_out, f"npg_n{nt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t_lp = min(remaining * 0.15, 120.0)
                a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t_lp, solver, verbose=False)
                check_update(a_out, f"lp_n{nt}")

    # === Phase 9: Final NPG polish (very high temp) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final NPG polish n={n_best} ===")
        t9 = min(remaining * 0.35, 200.0)
        a_out, _ = nesterov_projected_gradient(norm_sum1(best_seq), t9,
                                                t_init=300.0, t_final=2000.0,
                                                gamma=0.3, verbose=True, rng_seed=999)
        check_update(a_out, f"npg_final_n{n_best}")

    # === Phase 10: Final LP if time remaining ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final LP n={n_best} ===")
        t10 = min(remaining * 0.7, 150.0)
        a_out, _ = lp_frank_wolfe_fast(norm_sum1(best_seq), t10, solver, verbose=True)
        check_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
