Let me also think about the Nesterov acceleration for LP-FW. The standard FW update is:
x_{t+1} = (1 - α_t) * x_t + α_t * g_t
where α_t = 2/(t+2) for the standard rates.

The Nesterov-accelerated FW (also called "Accele FW" or "FACA") uses:
y_t = x_t + (t-1)/(t+2) * (x_t - x_{t-1})  (extrapolation)
g_t = argmin_{g in C} <∇f(y_t), g>  (linear minimization)
x_{t+1} = (1 - α_t) * y_t + α_t * g_t
where α_t = 2/(t+2)

This gives O(1/t^2) convergence for the function value.

In our case, the "linear minimization" is the LP:
max_b <∇f(y_t), b> subject to b in {nonneg, sum(b)=1}
≈ max_b sum(b) subject to T(y) @ b <= M(y), b >= 0

(We're approximating the linear objective with the LP, which is valid because both define the "steepest descent direction" on the simplex under the autocorrelation constraint.)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-initialization with structured windows (Kaiser, raised cosine, sinc, Dirichlet)
2. FFT-based full-gradient accelerated projected gradient (APG)
3. LP Frank-Wolfe with Nesterov acceleration (main heavy lifter, given most time)
4. Multi-restart perturbation search from best solution
5. Final LP-FW polish and coordinate descent
Strategy: Give LP-FW the majority of time budget (it's the most effective method),
use APG primarily for initialization and restarts.

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s

        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(a, weights):
    """Compute grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m] using FFT."""
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]

def smoothed_gradient_full(a, beta):
    """Full gradient of smoothed max using FFT-based cross-correlation."""
    n = len(a)
    ac = fast_autocorr(a)
    shifted = ac - ac.max()
    max_shift = np.max(shifted)
    if max_shift > 500:
        shifted = shifted / max_shift * 500
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        weights = np.zeros_like(ac)
        weights[np.argmax(ac)] = 1.0
    else:
        weights = exp_vals / total
    grad = compute_gradient_fft(a, weights)
    return grad, ac.max()

def apg_optimize(a0, time_limit, beta_start=30.0, beta_end=100.0, c0=2.0, seed=42, verbose=False):
    """Nesterov-accelerated projected gradient on simplex with full FFT gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        t = iter_count
        frac = min(1.0, t / float(max(1, int(time_limit * 50))))
        cur_beta = beta_start + (beta_end - beta_start) * frac

        beta_t = (t - 1.0) / (t + 1.5) if t > 1 else 0.0
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g, _ = smoothed_gradient_full(y, cur_beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        eta = c0 / (np.sqrt(t + 1) * gn)

        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new
        iter_count += 1

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.2)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"        APG iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m

def build_toeplitz(a):
    """Build (2n-1) x n Toeplitz matrix T where T[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_lp_iters=50000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, max_lp_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = build_toeplitz(y)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if t % 100 == 0 and verbose:
                print(f"        LP-FW iter={t}: LP failed ({prob.status})")
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            if verbose and t % 50 == 0:
                print(f"        LP-FW iter={t}: b_sum too small, stopping")
            break
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-14:
            best_m = mc
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"        LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, M={M_curr:.6f}")

        if best_m < 1e-10:
            break

    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        peak_contrib = np.zeros(n)
        for k in top_k[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            peak_contrib[idx] += contrib

        sources = np.argsort(peak_contrib)[::-1][:20]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            best_dst = -1
            best_delta = 0.0
            n_try = min(n, 300)
            candidates = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)

            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_k[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_delta:
                    best_delta = score
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"        CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    inits = []

    # Raised cosine (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(hann / hann.sum())

    # Gaussian (various widths)
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Sinc-squared
    sinc_arg = 3.0 * (x - 0.5)
    sinc2 = np.sinc(sinc_arg) ** 2
    sinc2 = np.maximum(sinc2, 0)
    if sinc2.sum() > 0:
        inits.append(sinc2 / sinc2.sum())

    # Dirichlet (random)
    d = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(d)

    # Bimodal
    bim = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    inits.append(bim / bim.sum())

    # Quartic center
    q4 = np.maximum(1.0 - (2 * x - 1) ** 2, 0) ** 2
    if q4.sum() > 0:
        inits.append(q4 / q4.sum())

    return inits
