```python
def compute_gradient(a, w, n):
    """Compute gradient of log-sum-exp of autocorrelation w.r.t. a.
    w: softmax weights of length 2n-1
    a: sequence of length n
    Returns: gradient of length n
    """
    # Need FFT size >= (2n-1) + n - 1 = 3n - 2
    L = 1
    while L < 3*n - 2:
        L *= 2
    
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    
    # g1 = cross-correlation of w and a: (w star a)[j] = sum_m w[m+j] a[m]
    # In FFT: ifft(FFT(w) * conj(FFT(a)))
    g1 = np.fft.irfft(W * np.conj(A), L)[:n]
    
    # g2 = convolution of w and a: (w * a)[j] = sum_m w[j-m] a[m]
    # In FFT: ifft(FFT(w) * FFT(a))
    g2 = np.fft.irfft(W * A, L)[:n]
    
    return g1 + g2
```
