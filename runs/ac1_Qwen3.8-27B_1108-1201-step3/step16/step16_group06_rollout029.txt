def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_fft_size(n):
    return 1 << (2 * n).bit_length()

def compute_softmax_weights(ac, beta):
    shifted = ac - ac.max()
    p = np.exp(beta * shifted)
    ps = p.sum()
    if ps < 1e-30:
        return np.ones_like(ac) / len(ac)
    return p / ps

def compute_grad_fft(a, p, fft_size):
    n = len(a)
    A_fft = np.fft.rfft(a, fft_size)
    P_fft = np.fft.rfft(p, fft_size)
    q = np.fft.irfft(np.conj(A_fft) * P_fft, fft_size)[:n]
    idx = np.arange(n)
    grad = 2.0 * q - p[2 * idx] * a
    return grad

def _lp_step_fast(a, beta, iter_num, solver, max_lp_time=5.0):
    """Single LP Frank-Wolfe step."""
    try:
        import cvxpy as cp
        n = len(a)
        y = norm_sum1(a)
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            return None

        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y[j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        t_start = time.monotonic()
        prob.solve(solver=solver, verbose=False)
        if time.monotonic() - t_start > max_lp_time:
            return None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            return None
        g_dir = b_val / b_sum

        alpha = 2.0 / (iter_num + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        return norm_sum1(y_new)
    except Exception:
        return None

                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall >= max_stall:
                break
            if stall % 500 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-15:
                        a = a / s_a
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

        if verbose and iteration % 5000 == 0 and iteration > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD iter={iteration} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS
