import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT. a must be a numpy array."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_shape(a, n_iter=25, blend=0.5):
    """Iteratively flatten the power spectrum, projecting onto non-negative simplex."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        P = np.maximum(P, 1e-20)
        C = 1.0 / np.sqrt(P)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - blend) * a + blend * a_new
    return a


def mirror_descent_nesterov(a0, time_limit, eta_base=0.8, verbose=True):
    """Nesterov-accelerated entropic mirror descent with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 8000
    rng = np.random.default_rng(123)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Autocorrelation
        ac = fast_autocorr(y)
        M = ac.max()

        if M < best_m:
            best_m = M
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Subgradient at the peak lag
        k_max = int(np.argmax(ac))
        g = np.zeros(n)
        j_start = max(0, k_max - n + 1)
        j_end = min(n, k_max + 1)
        j_idx = np.arange(j_start, j_end)
        g[j_idx] = 2.0 * y[k_max - j_idx]

        # Adaptive step size based on gradient magnitude
        g_max = g.max()
        if g_max < 1e-15:
            g_max = 1.0
        eta = eta_base / g_max

        # Backtracking: try full step first, then halve
        for trial in range(8):
            log_a = np.log(y) - eta * g
            log_a = log_a - log_a.max()
            a_new = np.exp(log_a)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            M_trial = fast_autocorr(a_new).max()
            if M_trial <= M + 1e-12:
                break
            eta *= 0.5

        x_prev = y
        x = a_new

        # Periodic restart if stalled
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, eta={eta:.4f}")

    return best_a, best_m


def fista_logsumexp(a0, time_limit, beta0=3.0, beta_max=50.0, verbose=True):
    """FISTA with log-sum-exp smoothed max, FFT-based gradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    z = a.copy()
    k_fista = 1.0
    beta = beta0

    L = 2 * n - 1
    fft_size = 1 << (L + n - 2).bit_length()

    stall = 0
    max_stall = 5000
    rng = np.random.default_rng(456)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation at z
        ac = fast_autocorr(z)
        M = ac.max()
        if M < best_m:
            best_m = M
            best_a = z.copy()
            stall = 0
        else:
            stall += 1

        # Log-sum-exp smoothing weights
        max_ac = ac.max()
        logits = beta * (ac - max_ac)
        exp_logits = np.exp(logits)
        Z_eps = exp_logits.sum()
        w = exp_logits / Z_eps

        # Gradient via FFT: grad[j] = 2 * sum_m w[j+m] * z[m]
        u_rev = w[::-1]
        U = np.fft.rfft(u_rev, fft_size)
        Zf = np.fft.rfft(z, fft_size)
        conv_full = np.fft.irfft(U * Zf, fft_size)[:L + n - 1]
        grad = 2.0 * conv_full[L - n:L][::-1]

        # FISTA update
        new_k = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * k_fista ** 2))
        momentum = (k_fista - 1.0) / new_k
        z_new = x + momentum * (x - z)
        z_new = np.maximum(z_new, 1e-30)
        z_new = z_new / z_new.sum()

        # Gradient step with simplex projection
        lr = 2.0 / (t + 2.0) * (1.0 + 0.5 / (beta * 0.1 + 1.0))
        x_new = z - lr * grad
        x_new = np.maximum(x_new, 0.0)
        s_x = x_new.sum()
        if s_x < 1e-15:
            x_new = z_new.copy()
        else:
            x_new = x_new / s_x

        x = x_new
        z = z_new
        k_fista = new_k

        # Grow beta periodically
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * 1.08, beta_max)

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.002, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            z = x_new.copy()
            k_fista = 1.0

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")

    return best_a, best_m


def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)

    constraints = [cp.sum(b_var) <= 10.0]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw_nesterov(a0, time_limit, solver, top_k_init=80, top_k_max=300, verbose=True):
    """Nesterov-accelerated LP Frank-Wolfe with backtracking and adaptive K."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    current_k = top_k_init
    stall = 0
    max_stall = 2000
    rng = np.random.default_rng(789)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-20)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Select active constraints: top-K + near-max
        sorted_ac = np.argsort(ac)[::-1]
        top_k_idx = sorted_ac[:current_k]
        near_max_thresh = M_curr * 0.95
        near_max_idx = np.where(ac >= near_max_thresh)[0]
        active_set = np.unique(np.concatenate([top_k_idx, near_max_idx]))
        active_set = active_set[:min(len(active_set), top_k_max)]

        # Solve LP
        direction = solve_lp_direction(y, M_curr * 1.001, active_set, solver)
        if direction is None:
            continue

        # Backtracking line search
        best_alpha = 2.0 / (t + 2.0)
        best_M_alpha = M_curr
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 2.0 / (t + 2.0), 0.1]:
            x_trial = (1.0 - alpha) * y + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            st = x_trial.sum()
            if st < 1e-15:
                continue
            x_trial = x_trial / st
            M_trial = fast_autocorr(x_trial).max()
            if M_trial < best_M_alpha:
                best_M_alpha = M_trial
                best_alpha = alpha
            if M_trial <= M_curr * 0.999:
                break

        x_new = (1.0 - best_alpha) * y + best_alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        M_x = fast_autocorr(x).max()
        if M_x < best_m:
            best_m = M_x
            best_a = x.copy()
            stall = 0

        # Adapt K
        if t % 100 == 0 and current_k < top_k_max:
            current_k = min(current_k + 10, top_k_max)

        # Restart
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.002, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, K={current_k}, alpha={best_alpha:.3f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with multiple move sizes."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
