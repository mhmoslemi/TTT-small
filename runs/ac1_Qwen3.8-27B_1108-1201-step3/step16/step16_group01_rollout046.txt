def get_fft_size(n):
    """Get next power of 2 >= 3n for linear cross-correlation."""
    return 1 << (3 * n).bit_length()

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lse_gradient(a, weights, fft_size):
    """
    Compute gradient of log-sum-exp objective.
    grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m]
    = 2 * (weights cross-correlated with a)[j]
    """
    n = len(a)
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    return 2.0 * cross[:n]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lse_adam(a0, time_limit, T_init=0.1, T_decay=0.99995, T_min=0.001, 
             lr_init=0.01, verbose=True, rng_seed=42):
    """
    Log-sum-exp smoothed gradient descent with Adam optimizer.
    Temperature anneals from T_init to T_min.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = get_fft_size(n)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr_init
    
    # Pre-allocate
    ac = np.zeros(2*n-1)
    weights = np.zeros(2*n-1)
    
    stall_count = 0
    max_stall = 5000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = max(T_min, T_init * (T_decay ** t))
        
        # Compute autocorrelation
        A_fft = np.fft.rfft(a, fft_size)
        ac = np.fft.irfft(A_fft * A_fft, fft_size)[:2*n-1]
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            # Increase temperature temporarily
            T_init = max(T_init * 1.5, T)
            lr = lr_init * 1.5
            # Small perturbation
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.01, n) * a
            a = np.maximum(a + pert, 0)
            a = a / a.sum()
            continue
        
        # Compute softmax weights
        ac_shifted = ac - ac.max()
        weights[:2*n-1] = np.exp(ac_shifted / T)
        w_sum = weights[:2*n-1].sum()
        if w_sum < 1e-30:
            weights[:] = 0
            weights[n-1] = 1.0  # center
        else:
            weights[:2*n-1] /= w_sum
        
        # Compute gradient via FFT cross-correlation
        W = np.fft.rfft(weights, fft_size)
        cross = np.fft.irfft(W * np.conj(A_fft), fft_size)
        grad = 2.0 * cross[:n]
        
        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
        m_hat = m_adam / (1 - beta1 ** t)
        v_hat = v_adam / (1 - beta2 ** t)
        
        # Learning rate decay
        lr_eff = lr_init / (1 + t * 1e-6)
        
        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        
        a = a_new
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE t={t} ({elapsed:.1f}s): M={best_m:.10f}, J={J:.6f}, T={T:.6f}")
    
    return best_a, best_m

def exact_subgradient(a, ac, threshold_frac=0.99):
    """Compute subgradient using lags near the maximum."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    return g, len(active)

def amd_refine(a0, time_limit, c0=3.0, threshold_frac=0.99, verbose=True, rng_seed=42):
    """AMD with exact subgradient for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g, _ = exact_subgradient(y, ac, threshold_frac=threshold_frac)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn
        
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.005, n) * best_a

            x = x / x.sum()
            x_prev = x.copy()

            elapsed = time.monotonic() - t0

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent: move mass from peak-contributing positions to others.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_k = np.argsort(ac)[::-1][:10]
        improved = False
        
        # For each peak, find contributing pairs and try to reduce them
        for k in peak_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with max contribution
            max_pair = np.argmax(contrib)
            src = idx[max_pair] if a[idx[max_pair]] >= a[j_idx[max_pair]] else j_idx[max_pair]
            
            if a[src] < 1e-10:
                continue
            
            # Try moving to random low-value positions
            for _ in range(20):
                dst = int(rng.integers(n))
                if dst == src or a[dst] >= a[src]:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                if a.sum() > 1e-15:
                    a /= a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0)
                        if a.sum() > 1e-15:
                            a /= a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
