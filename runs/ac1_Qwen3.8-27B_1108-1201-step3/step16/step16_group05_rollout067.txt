"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Frank-Wolfe with Nesterov acceleration and adaptive top-K constraints
2. Smoothed-max accelerated projected gradient
3. Peak-targeted coordinate descent with asymmetric moves
4. Multi-length exploration and asymmetric initialization
"""

import numpy as np
import time
import cvxpy as cp

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _solver


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, beta, top_k=200):
    n = len(a)
    ca = fast_conv(a)
    k_eff = min(top_k, len(ca))
    top_idx = np.argsort(ca)[::-1][:k_eff]
    top_vals = ca[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(np.clip(beta * shifted, -500, 500))
    wsum = exp_vals.sum()
    if wsum < 1e-300:
        wsum = 1.0
    weights = exp_vals / wsum
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, ca.max()


def apg_smoothed(a0, time_limit, beta_start=30.0, beta_end=5.0, top_k=200,
                 c0=2.0, verbose=True, seed=42):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 2000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 15000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g, _ = compute_smoothed_grad(y, cur_beta, top_k=top_k)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        eta = gamma / (np.sqrt(float(t)) + 1.0)

        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            top_k = min(top_k + 20, 400)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, top_k=300, verbose=True, seed=42):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()
    stall = 0
    max_stall = 50

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ca = fast_conv(y)
        M_curr = ca.max()
        if M_curr < 1e-14:
            break

        K_use = min(top_k, 2 * n - 1)
        if t < 5:
            K_use = min(80, K_use)
        elif t < 20:
            K_use = min(150, K_use)
        elif t < 50:
            K_use = min(250, K_use)

        # Build top-K constraint rows
        top_indices = np.argsort(ca)[::-1][:K_use]
        T = np.zeros((K_use, n), dtype=np.float64)
        for row_idx, k in enumerate(top_indices):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            cols = np.arange(i_start, i_end)
            T[row_idx, cols] = y[k_int - cols]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y_mom = y_new + beta_t * (y_new - y)
        y_mom = np.maximum(y_mom, 0.0)
        sm = y_mom.sum()
        if sm < 1e-15:
            y_mom = y_new.copy()
        else:
            y_mom = y_mom / sm

        y = y_new

        improved = False
        for candidate in [y, y_mom]:
            mc = fast_conv(candidate).max()
            if mc < best_m - 1e-15:
                best_m = mc
                best_a = candidate.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                stall = 0
                noise = rng.normal(0, 0.002, n) * best_a
                y_re = np.maximum(best_a + noise, 1e-12)
                y = y_re / y_re.sum()
                max_stall = int(max_stall * 1.2)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coord_polish(a0, time_limit, verbose=True, seed=123):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ca = fast_conv(a)
        M_curr = ca.max()
        top_peaks = np.argsort(ca)[::-1][:min(30, len(ca))]
        improved = False

        # Compute peak contribution at each position
        peak_contrib = np.zeros(n, dtype=np.float64)
        for k in top_peaks[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            peak_contrib[idx] += contrib

        sources = np.argsort(peak_contrib)[::-1][:40]
        for src in sources[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Find best destination: low impact on peaks
            n_try = min(n, 800)
            if n > n_try:
                dst_candidates = rng.choice(n, n_try, replace=False)
            else:
                dst_candidates = np.arange(n)

            best_dst = -1
            best_score = -np.inf
            for dst in dst_candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_peaks[:10]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local minimum
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall > 0 and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def make_asymmetric_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Asymmetric bump: shifted peak
    center = rng.uniform(0.35, 0.65)
    width = rng.uniform(0.15, 0.35)
    base = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    # Add asymmetric tail
    tail = np.exp(-(x - center) / max(0.1, width * 2)) * 0.5
    a = base + tail
    a = np.maximum(a, 0.0)
    return a / a.sum()


def propose_candidate(budget_s=1800.0):

    time_budget = min(float(budget_s) - 60.0, 1700.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = get_parent()
    if parent is not None:
        p_norm = normalize(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        best_seq = make_asymmetric_init(n_base, seed=7)

        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # Phase 1: APG at parent length
    rem = remaining()
    print(f"\n=== Phase 1: APG n={n0} ===")
    t1 = min(rem * 0.25, 350.0)
    if t1 > 30:
        a_in = normalize(best_seq)
        a_out, _ = apg_smoothed(a_in, t1, beta_start=30.0, beta_end=5.0,

    # Phase 3: APG polish

    # Phase 4: Coordinate polish

    # Phase 6: Try shorter length (sometimes works better)

    # Phase 9: Final coordinate polish
    rem = remaining()
    n_best = len(best_seq)
    if rem > 30:

        t9 = min(rem * 0.6, 80.0)
