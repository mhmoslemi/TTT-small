   ```python
   def smooth_gradient(a, t):
       """Compute gradient of log-sum-exp autocorrelation."""
       n = len(a)
       conv = fast_autocorr(a)
       max_conv = conv.max()
       shifted = t * (conv - max_conv)
       exp_shifted = np.exp(shifted)
       sum_exp = exp_shifted.sum()
       weights = t * exp_shifted / sum_exp  # w_k
       
       # grad[j] = 2 * sum_k w_k * a[k-j] for 0 <= k-j < n
       # Use cross-correlation via FFT
       # result[j] = sum_k w_k * a[k-j]
       # = (w * flip(a))[n-1+j] where * is linear convolution
       
       b = a[::-1]  # flip
       total_len = len(weights) + n - 1  # = (2n-1) + n - 1 = 3n-2
       fft_size = 1 << (total_len - 1).bit_length()
       W = rfft(weights, fft_size)
       B = rfft(b, fft_size)
       conv_result = irfft(W * B, fft_size)[:total_len]
       # result[j] = conv_result[n-1+j] for j = 0, ..., n-1
       result = conv_result[n-1:n-1+n]
       return 2.0 * result
   ```
