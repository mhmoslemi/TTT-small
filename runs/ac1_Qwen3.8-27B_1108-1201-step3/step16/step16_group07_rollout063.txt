If we solve the LP for the active set S at iteration t, the solution b_t can be used as a warm start for the LP at iteration t+1 (with a slightly different active set). Most LP solvers support this.

But cvxpy doesn't directly support warm starts. So we'd need to use a solver interface that does.

Alternatively, we can approximate the LP solution using a few iterations of the Frank-Wolfe (conditional gradient) method, which is much cheaper per iteration.

**Frank-Wolfe for the LP:**
max c^T b s.t. T b <= M, b >= 0

The FW step: given current b, solve the linear subproblem:
max c^T s s.t. T s <= M, s >= 0, ||s||_1 = 1 (or just s >= 0, max s_j)

The subproblem is: max s_j s.t. sum_k T[k,j] * s_j <= M, s_j >= 0.
So s_j <= M / T[k,j] for all k where T[k,j] > 0.
The optimal s is a single coordinate: s = e_j * min(M / max_k T[k,j], 1) (or just e_j if we normalize later).

Actually the FW subproblem for `max c^T s s.t. T s <= M, s >= 0` is:
s* = argmax c_j * s_j s.t. T[:,j] * s_j <= M, s_j >= 0
    = e_j * min(M / max_k T[k,j], infinity)

So s* is a unit vector times a scalar. The FW update is:
b_new = (1 - gamma) * b + gamma * s*

where gamma = 2/(t+2) (for the 2/(t+2) step size).

This is much cheaper than solving the full LP! Each FW iteration is O(|S| * n) for computing the subproblem (finding the best j) and O(n) for the update.

But FW converges slowly (O(1/t) rate vs O(1/t^2) for LSWD or acceleration). With Nesterov acceleration for FW, we get O(1/t^2).

Let me implement an accelerated FW as a replacement for the LP in the inner loop, and use the full LP only for occasional "corrections."

def subgradient_fft(a, w):
    """
    Compute subgradient of max_k conv[a,a][k] using FFT cross-correlation.
    a: sequence of length n
    w: weight vector of length 2n-1 (nonzero at active peaks)
    Returns: gradient vector of length n

    W = rfft(w, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return 2.0 * corr[:n]

def active_set_weights(conv, M, K, beta=50.0):
    """Compute weights for active set using exponential soft-max."""

    size = len(conv)

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def subgradient_fft(a, w):
    """
    Compute subgradient of max_k conv[a,a][k] using FFT cross-correlation.
    g[j] = 2 * sum_m a[m] * w[m+j] for m=0..n-1
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(A * np.conj(W), fft_size)
    return 2.0 * corr[:n]


def subgradient_direct(a, active_idx, weights):
    """Direct subgradient computation for small active sets."""
    n = len(a)
    g = np.zeros(n)
    for k, w in zip(active_idx, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            g[lo:hi] += 2.0 * w * a[k - np.arange(lo, hi)]
    return g


def normalize_to(a, n, target_sum=None):

        target_sum = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    return a * target_sum / s

def get_weights(conv, M, K, beta):
    """Exponential soft-max weights on top K peaks."""
    size = len(conv)
    K_eff = min(K, size)
    top_idx = np.argpartition(conv, -K_eff)[-K_eff:]
    diff = M - conv[top_idx]
    w = np.exp(-beta * diff / max(M * 0.01, 1e-10))
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(K_eff) / K_eff
    else:
        w = w / w_sum
    return top_idx, w

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    step = 0.5
    n_iters = 0

    stall_patience = 300
    conv_size = 2 * n - 1

        K = int(K_init + (K_max - K_init) * progress)
        K = max(K, 5)

        M_curr = conv_y.max()

        if M_curr < 1e-12:

        top_idx, w_top = get_weights(conv_y, M_curr, K, beta)

        s_new = a_new.sum()
        if s_new < 1e-15:

        else:

            y_new = a_new.copy()

            y_new = a_new + momentum * (a_new - y_prev)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:

            else:

        a = a_new
        n_iters += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            step = min(step * 1.005, 4.0)
        else:
            stall += 1
            if stall > stall_patience:
                y = best_a.copy()
                y_prev = best_a.copy()
                a = best_a.copy()
                stall = 0
                step = max(step * 0.6, 0.05)

        if t % 100 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall = 0

        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, K={K}, step={step:.3f}")

    return best_a, best_score


def lp_polish(a0, time_limit, top_K=25, verbose=False, seed=0, momentum_cap=0.92):
    """LP-based best-response with Nesterov acceleration."""
    import cvxpy as cp
    solver = get_solver()
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        conv_size = 2 * n - 1

        K = min(top_K, conv_size)
        active_set = np.argpartition(conv_y, -K)[-K:]
        active_set = np.sort(active_set)

        T = np.zeros((K, n))
        for idx, k in enumerate(active_set):
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n, k_int + 1)
            if lo < hi:
                T[idx, lo:hi] = y[k_int - np.arange(lo, hi)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if n_iters > 50:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 1
            continue

        g = b_val * norm / b_sum
        n_iters += 1
        stall = 0

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), momentum_cap)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPPolish t={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Fast coordinate descent with FFT-based autocorrelation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        if no_improve > 6:
            break

        conv = fast_autocorr(a)
        cur_max = conv.max()

        # Focus on coordinates with largest gradient contribution
        grad_magnitude = np.zeros(n)
        for k in range(0, size, max(1, size // 100)):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad_magnitude[lo:hi] += abs(a[k - np.arange(lo, hi)])

        # Try top coordinates
        n_try = min(n, 200)
        top_coords = np.argsort(grad_magnitude)[-n_try:]
        rng.shuffle(top_coords)

        improved = False
        for j in top_coords:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm - aj
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            candidates = [0.0, aj * 0.3, aj * 0.7, aj * 1.3, aj * 1.7]
            typical = norm / n
            candidates += [typical * 0.5, typical * 1.5]
            candidates = [max(0.0, min(c, norm * 0.3)) for c in candidates]

            best_x = aj
            best_obj = cur_max

            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
                        cur_max = conv.max()

        if improved:
            no_improve = 0

        if verbose and no_improve == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def perturb_sequence(a, scale, rng):
    """Add multiplicative noise to explore nearby solutions."""
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.3)
    perturbed = a * noise
    s = perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return perturbed / s


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
