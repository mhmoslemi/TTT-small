def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

def norm_sum1(a):
    """Normalize so sum = 1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective_value(a):
    """Compute J = 2*n*max(ac)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def smooth_objective(a, top_k=50, temperature=0.01):
    """Smoothed objective using log-sum-exp of top-K autocorrelation values."""
    ac = fast_autocorr(a)
    s = a.sum()
    ac_norm = ac / (s * s)  # Normalize by (sum a)^2
    top_idx = np.argsort(ac_norm)[::-1][:top_k]
    vals = ac_norm[top_idx] / temperature
    max_val = vals.max()
    # Log-sum-exp for numerical stability
    lse = max_val + np.log(np.sum(np.exp(vals - max_val)))
    return 2.0 * len(a) * lse * temperature / top_k

def bb_step(a, grad, a_prev, grad_prev, n):
    """Barzilai-Borwein step size."""
    s_vec = a - a_prev
    y_vec = grad - grad_prev
    s_norm = np.dot(s_vec, s_vec)
    y_norm = np.dot(y_vec, y_vec)
    sy = np.dot(s_vec, y_vec)
    if sy < 1e-15:
        return 0.01
    gamma_bb = s_norm / sy
    gamma_bb = np.clip(gamma_bb, 1e-6, 10.0)
    return gamma_bb

def gradient_optimize(a0, time_limit, top_k=50, temperature=0.01, verbose=True):
    """Smoothed gradient descent with BB step size and restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    a_prev = a.copy()
    grad_prev = smooth_gradient(a, top_k, temperature).copy()
    stall = 0
    max_stall = 500
    cur_temp = temperature
    cur_top_k = top_k
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad = smooth_gradient(a, cur_top_k, cur_temp)
        
        # BB step size
        step = bb_step(a, grad, a_prev, grad_prev, n)
        
        # Gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        if t % 50 == 0 or t < 10:
            obj = objective_value(a_new)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        
        # Adaptive temperature: decrease over time
        cur_temp = max(temperature * 0.5, temperature / (1 + t * 0.001))
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            cur_temp = temperature  # Reset temperature
            cur_top_k = min(cur_top_k + 10, 150)
            # Perturbation
            pert = rng.normal(0, 0.01, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            a_prev = a.copy()
            grad_prev = smooth_gradient(a, cur_top_k, cur_temp).copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Grad restart t={t} ({elapsed:.0f}s): J={best_obj:.6f}, top_k={cur_top_k}, T={cur_temp:.4f}")
            max_stall = min(max_stall + 100, 2000)
        
        a_prev = a
        a = a_new
        grad_prev = grad
    
    # Final exact evaluation
    final_obj = objective_value(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj

            best_obj = obj_y
            best_a = y.copy()
        
        obj_x = objective_value(x)
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): J={best_obj:.6f}")
    
    return best_a, best_obj

def coordinate_2d(a0, time_limit, verbose=True, rng_seed=42):
    """2D coordinate descent: pick two positions, find optimal transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:20]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find highest contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find best destination: position with lowest "cost"
            # Cost of putting mass at position d = sum over top lags k2 of a[d relative to k2]
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    d2 = d - k2i
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    if 0 <= d2 < n:
                        scores[d] += a[d2]
            
            # Destination should have low score and be different from src
            candidates = np.argsort(scores)
            moved = False
            for dst in candidates[:20]:
                if dst == src:
                    continue
                # Try transfer
                amt = a[src] * 0.1
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                obj_trial = objective_value(a_trial)
                if obj_trial < best_obj - 1e-12:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    moved = True
                    stall = 0
                    break
            
            if not moved:
                # Try random destination
                if a[src] > 1e-10:
                    dst = int(rng.integers(n))
                    if dst != src:
                        for frac in [0.005, 0.01, 0.02, 0.05]:
                            amt = a[src] * frac
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            a_trial = a_trial / a_trial.sum()
                            obj_trial = objective_value(a_trial)
                            if obj_trial < best_obj - 1e-12:
                                a = a_trial
                                best_obj = obj_trial
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.03)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = objective_value(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord2D ({elapsed:.0f}s): J={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_ramp(n, power=1.0, direction='left'):
    """Create a ramp sequence."""
    if direction == 'left':
        x = np.arange(n, 0, -1, dtype=np.float64)
    else:
        x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)

def make_exp_decay(n, rate=0.5, direction='left'):
    """Create an exponential decay sequence."""
    if direction == 'left':
        x = np.arange(n - 1, -1, -1, dtype=np.float64)
    else:
        x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * x / n)
    return norm_sum1(a)

def make_bimodal(n, center1=0.2, center2=0.8, width=0.1):
    """Create a bimodal sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-(x - center1)**2 / (2*width**2)) + np.exp(-(x - center2)**2 / (2*width**2))
    return norm_sum1(a)
