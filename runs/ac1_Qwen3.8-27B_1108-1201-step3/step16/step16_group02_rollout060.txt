    return a / s

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = (u * idx + 1.0 > cssv)
    rho = np.sum(cond)
    if rho == 0:
        rho = 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def smooth_max_subgrad(a, ac, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*ac_k)) w.r.t. a."""
    n = len(a)
    max_ac = ac.max()
    log_w = beta * (ac - max_ac)
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        thresh = max_ac - 1e-8
        w = (ac >= thresh).astype(np.float64)
        w_sum = w.sum()
        if w_sum < 1e-10:
            return np.ones(n) / n
        w /= w_sum
    else:
        w /= w_sum

    grad = np.zeros(n)
    for k in range(len(ac)):
        wk = w[k]
        if wk < 1e-14:
            continue
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        m = ki - idx
        coeffs = 2.0 * a[m]
        if ki % 2 == 0:
            ih = ki // 2
            if i_start <= ih < i_end:
                coeffs[ih - i_start] = 2.0 * a[ih]
        grad[idx] += wk * coeffs
    return grad

def fista_smooth(a0, time_limit, beta_start=30.0, beta_end=300.0,
                 verbose=True, rng_seed=42):
    """FISTA with adaptive log-sum-exp smoothing on the simplex."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    y = a.copy()
    t_iter = 0
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        frac = min(1.0, t_iter / 500.0)
        beta = beta_start + (beta_end - beta_start) * frac

        ac = fast_conv(y)
        g = smooth_max_subgrad(y, ac, beta)

        eta = 2.0 / (np.sqrt(t_iter) + 1.0)
        x_new = project_simplex(y - eta * g)

        t_plus = t_iter + 1
        beta_nest = (np.sqrt(t_plus + 1) - 1) / 2.0
        z = (t_iter / (t_iter + 1)) * x_new + (1 - t_iter / (t_iter + 1)) * x
        y_new = x_new + beta_nest * (x_new - x)
        y = y_new
        x_prev = x
        x = x_new

        mx = fast_conv(x).max()
        my = fast_conv(y).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            stall = 0
        elif my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = max(50, max_stall // 2)
            # Structured perturbation
            mode = rng.integers(1, 6)
            if mode == 1:
                phase = rng.uniform(0, 2 * np.pi)
                freq = rng.uniform(1, 3)
                pert = 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            elif mode == 2:
                center = rng.uniform(0.2, 0.8)
                width = rng.uniform(0.1, 0.3)
                dist = np.abs(np.arange(n) / n - center)
                pert = -0.03 * np.exp(-0.5 * (dist / width) ** 2)
            elif mode == 3:
                pert = rng.normal(0, 0.01, n)
            elif mode == 4:
                # Shift modulation
                shift = rng.uniform(-0.1, 0.1)
                xx = np.linspace(0, 1, n)
                pert = 0.02 * np.cos(2 * np.pi * (xx + shift))
            else:
                # Local bump
                center = int(rng.integers(n // 4, 3 * n // 4))
                radius = int(n * 0.05)
                pert = np.zeros(n)
                ll = max(0, center - radius)
                rr = min(n, center + radius)
                xx = np.arange(ll, rr)
                pert[ll:rr] = 0.02 * np.exp(-0.5 * ((xx - center) / (radius / 2)) ** 2)

            x_r = best_a * (1.0 + pert)
            x_r = project_simplex(x_r)
            x = x_r
            x_prev = x_r
            y = x_r.copy()

        if verbose and t_iter % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t_iter} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_m

def subgradient_nesterov(a0, time_limit, verbose=True, rng_seed=77):
    """Nesterov accelerated subgradient using exact active set (top lags)."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    y = a.copy()
    stall = 0
    max_stall = 300

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M = ac.max()

        # Active lags: within 0.5% of max
        thresh = M * 0.9995
        active = np.where(ac >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        if len(active) > 50:
            top_idx = np.argsort(ac)[::-1][:50]
            active = top_idx

        # Compute subgradient: average of gradients of active lags
        g = np.zeros(n)
        for k in active:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            m = ki - idx
            coeffs = 2.0 * a[m]
            if ki % 2 == 0:
                ih = ki // 2
                if i_start <= ih < i_end:
                    coeffs[ih - i_start] = 2.0 * a[ih]
            g += coeffs / len(active)

        # Normalize gradient (scale invariance means we should use relative step)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        eta = 2.0 * best_m / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)

        t_plus = t + 1
        beta_nest = (np.sqrt(t_plus + 1) - 1) / 2.0
        y_new = x_new + beta_nest * (x_new - x)
        y_new = project_simplex(y_new)
        y = y_new
        x_prev = x
        x = x_new

        mx = fast_conv(x).max()
        my = fast_conv(y).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            stall = 0
        elif my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = max(100, max_stall // 2)
            pert = rng.normal(0, 0.005, n)
            x_r = project_simplex(best_a + pert)
            x = x_r
            y = x_r.copy()
            x_prev = x_r

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.1f}s): J={J:.8f}, |active|={len(active)}")

    return best_a, best_m

def coordinate_optimize(a0, time_limit, verbose=True, rng_seed=99):
    """Exact coordinate-wise optimization with efficient convolution updates."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 3

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        order = list(range(n))
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit - 5:
                break

            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Compute base convolution (a[i] = 0) and slopes
            a_save = a[i]
            a[i] = 0.0
            base_conv = fast_conv(a)
            a[i] = a_save

            # Slope for each k: d conv[k]/d a[i]
            # conv[k] = sum_j a[j]*a[k-j]
            # Terms with a[i]: a[i]*a[k-i] (if 0<=k-i<n) + a[k-i]*a[i] (if 0<=k-i<n, k-i != i)
            # If k-i == i (k=2i): a[i]^2, slope = 2*a[i] but we're evaluating at variable x so slope=2x
            # If k-i != i: slope = 2*a[k-i]
            # If k-i out of range: slope = 0

            slope = np.zeros(len(base_conv))
            k_min = max(0, i - (n - 1))
            k_max = min(2 * n - 2, i + (n - 1))
            for k in range(k_min, k_max + 1):
                m = k - i
                if 0 <= m < n:
                    if k == 2 * i:
                        slope[k] = 0.0  # quadratic term handled separately
                    else:
                        slope[k] = 2.0 * a[m]

            # Find max of base_conv (the "floor")
            base_max = base_conv.max()
            # Find the k that achieves base_max (excluding k=2i if quadratic dominates)
            
            # Try a grid of values for a[i]
            mean_val = 1.0 / n
            candidates = np.concatenate([
                np.array([0.0]),
                np.array([old_val * f for f in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]]),
                np.array([mean_val * f for f in [0.5, 1.0, 2.0, 3.0]]),
            ])
            candidates = np.clip(candidates, 0.0, 10.0)
            candidates = np.unique(candidates)

            best_x = old_val
            best_trial_m = best_m

            for x_val in candidates:
                # new_conv[k] = base_conv[k] + x_val * slope[k] + (x_val^2 if k==2i else 0)
                new_conv_vals = base_conv + x_val * slope
                if 2 * i < len(base_conv):
                    new_conv_vals[2 * i] += x_val * x_val
                trial_m = new_conv_vals.max()
                if trial_m < best_trial_m:
                    best_trial_m = trial_m
                    best_x = x_val

            # Refine around best_x with golden section
            if best_x < old_val * 0.5 or best_x > old_val * 1.5:
                lo = max(0.0, best_x * 0.8)
                hi = best_x * 1.2
                for _ in range(10):
                    gm1 = hi - (hi - lo) * 0.618
                    gm2 = lo + (hi - lo) * 0.618
                    for x_val in [gm1, gm2]:
                        new_conv_vals = base_conv + x_val * slope
                        if 2 * i < len(base_conv):
                            new_conv_vals[2 * i] += x_val * x_val
                        trial_m = new_conv_vals.max()
                        if trial_m < best_trial_m:
                            best_trial_m = trial_m
                            best_x = x_val
                    # Narrow interval
                    m1 = base_conv + gm1 * slope
                    if 2 * i < len(base_conv):
                        m1[2 * i] += gm1 * gm1
                    m2 = base_conv + gm2 * slope
                    if 2 * i < len(base_conv):
                        m2[2 * i] += gm2 * gm2
                    if m1.max() > m2.max():
                        lo = gm1
                    else:
                        hi = gm2

            if best_x != old_val and best_trial_m < best_m - 1e-15:
                a[i] = best_x
                a = norm_sum1(a)
                best_m = fast_conv(a).max()
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        if verbose and (stall >= 1 or stall % 2 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordOpt pass={stall} ({elapsed:.1f}s): J={J:.8f}")

    return best_a, best_m

def multi_restart_search(a0, time_limit, n_restarts=5, verbose=True, rng_seed=200):
    """Try multiple structured perturbations and short FISTA from each."""
    n = len(a0)
    best_a = norm_sum1(a0).copy()
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    per_restart_time = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 10:
            break

        # Generate perturbation
        mode = r % 7
        if mode == 0:
            # Frequency modulation
            freq = rng.uniform(2, 5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.03
            pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif mode == 1:
            # Asymmetric shift
            center = rng.uniform(0.3, 0.7)
            width = rng.uniform(0.15, 0.35)
            xx = np.arange(n) / n
            pert = -0.04 * np.exp(-0.5 * ((xx - center) / width) ** 2)
        elif mode == 2:
            # Two bumps
            c1 = rng.uniform(0.2, 0.4)
            c2 = rng.uniform(0.6, 0.8)
            xx = np.arange(n) / n
            pert = 0.03 * (np.exp(-0.5 * ((xx - c1) / 0.1) ** 2) - np.exp(-0.5 * ((xx - c2) / 0.1) ** 2))
        elif mode == 3:
            # Random smooth perturbation
            coarse = rng.normal(0, 1, size=max(4, n // 200))
            pert = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse)
            pert = 0.02 * (pert - pert.mean())
        elif mode == 4:
            # Linear tilt
            slope = rng.uniform(-0.04, 0.04)
            pert = slope * (np.arange(n) / n - 0.5)
        elif mode == 5:
            # Cosine window modification
            xx = np.arange(n) / n
            window = 0.5 + 0.5 * np.cos(2 * np.pi * xx)
            pert = 0.03 * (window - 0.5)
        else:
            # Pure random
            pert = rng.normal(0, 0.015, n)

        a_pert = best_a * (1.0 + pert)
        a_pert = project_simplex(a_pert)

        # Short FISTA
        t_rem = min(per_restart_time, time_limit - (time.monotonic() - t0) - 5)
        if t_rem < 5:
            continue
        a_out, _ = fista_smooth(a_pert, t_rem, beta_start=50.0, beta_end=200.0,
                                verbose=False, rng_seed=rng_seed + r)
        m_out = fast_conv(a_out).max()
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MultiRestart r={r} ({elapsed:.1f}s): J={J:.8f} [improved]")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n = 2400
        x = np.linspace(0, 1, n)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: FISTA smooth max (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: FISTA n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, _ = fista_smooth(best_seq, t1, beta_start=30.0, beta_end=300.0,
                                verbose=True, rng_seed=42)
        try_update(a_out, f"FISTA_n{n0}")

    # === Phase 2: Subgradient Nesterov (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: SubGrad n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = subgradient_nesterov(best_seq, t2, verbose=True, rng_seed=77)
        try_update(a_out, f"SubGrad_n{len(best_seq)}")

    # === Phase 3: Coordinate optimization (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: CoordOpt n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = coordinate_optimize(best_seq, t3, verbose=True, rng_seed=99)
        try_update(a_out, f"CoordOpt_n{len(best_seq)}")

    # === Phase 4: FISTA polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: FISTA polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = fista_smooth(best_seq, t4, beta_start=100.0, beta_end=500.0,
                                verbose=False, rng_seed=555)
        try_update(a_out, f"FISTA_polish_n{len(best_seq)}")

    # === Phase 5: Multi-restart (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: MultiRestart n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = multi_restart_search(best_seq, t5, n_restarts=4, verbose=True, rng_seed=200)
        try_update(a_out, f"MultiRestart_n{len(best_seq)}")

    # === Phase 6: Coordinate optimization after all (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = remaining * 0.10
    print(f"\n=== Phase 6: CoordOpt final n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = coordinate_optimize(best_seq, t6, verbose=True, rng_seed=333)
        try_update(a_out, f"CoordOpt_final_n{len(best_seq)}")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n0 + 600
        print(f"\n=== Phase 7: Longer n={n_long} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_long)
        t7a = min(remaining * 0.5, 300.0)
        if t7a > 60:
            a_out, _ = fista_smooth(a_long, t7a, beta_start=30.0, beta_end=200.0,
                                    verbose=True, rng_seed=777)
            try_update(a_out, f"FISTA_long_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.6, 200.0)
            if t7b > 30:
                a_out, _ = coordinate_optimize(best_seq, t7b, verbose=False, rng_seed=444)
                try_update(a_out, f"CoordOpt_long_n{len(best_seq)}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
