def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize array to sum to 1, clipping negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.ones(n) * s / n
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_ac(a, top_k=1, beta=0.0):
    """
    Compute a subgradient of f(a) = max_k ac_k(a) at point a.
    Uses top_k peaks with softmax weighting controlled by beta.
    Returns gradient vector of length n.
    """
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    
    if beta == 0:
        # Uniform over top_k peaks
        top_idx = np.argsort(ac)[::-1][:k_max]
        w = np.ones(k_max) / k_max
    else:
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for j in range(n):
            # d(ac_k)/d(a_j) = a[k-j] if 0 <= k-j < n, plus a[j-k] if 0 <= j-k < n
            val = 0.0
            if 0 <= k_int - j < n:
                val += a[k_int - j]
            if 0 <= j - k_int < n and j != k_int - j:
                val += a[j - k_int]
            # Note: if k_int - j == j (i.e., 2j == k_int), the term is 2*a[j]
            # which is already handled since both conditions give a[j]
            grad[j] += wi * val
    
    return grad

def lse_smooth_objective(a, tau):
    """Compute the log-sum-exp smoothing of max autocorrelation."""
    ac = fast_autocorr(a)
    ac_max = ac.max()
    # For numerical stability
    shifted = (ac - ac_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return ac_max + tau * log_sum

def lse_smooth_gradient(a, tau):

    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_lse_gradient(a, ac, tau):
    """Compute gradient of LSE-smoothed max autocorrelation.
    
    g[m] = 2 * sum_k p_k * a[k-m] where p_k = softmax(ac/tau)
    """
    n = len(a)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / tau, -50.0, 50.0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(ac)) / len(ac)
    else:
        p = p / p_sum
    
    g = np.zeros(n, dtype=np.float64)
    # Only process significant p values
    threshold = 1e-8
    significant = np.where(p > threshold)[0]
    for k in significant:
        k_int = int(k)
        m_start = max(0, k_int - n + 1)
        m_end = min(n - 1, k_int)
        if m_start <= m_end:
            indices = np.arange(m_start, m_end + 1)
            a_indices = k_int - indices
            g[indices] += 2.0 * p[k_int] * a[a_indices]
    
    return g


def lse_refine(a0, time_limit, verbose=True):
    """Refine using LSE-smoothed L-BFGS-B with tau annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    M_curr = best_M
    tau = M_curr * 0.1  # initial smoothing
    tau_min = M_curr * 0.001
    
    iteration = 0
    while tau > tau_min and time.monotonic() - t0 < time_limit * 0.7:
        iteration += 1
        tau = max(tau * 0.7, tau_min)
        
        # Capture current tau for the callback
        current_tau = tau
        
        def fun_and_grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-15:
                return 1e10, np.zeros_like(x)
            x_norm = x / s
            ac = fast_autocorr(x_norm)
            ac_max = ac.max()
            shifted = np.clip((ac - ac_max) / current_tau, -50.0, 50.0)
            exp_shifted = np.exp(shifted)
            val = ac_max + current_tau * np.log(exp_shifted.sum())
            grad = compute_lse_gradient(x_norm, ac, current_tau)
            return val, grad
        
        result = minimize(fun_and_grad, a, method='L-BFGS-B',
                         bounds=[(0.0, None)] * n,
                         options={'maxiter': min(500, max(50, int(time_limit / 20))),
                                  'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 2000})
        
        if result.x is not None:
            a_new = np.maximum(result.x, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                M_new = fast_autocorr(a_new).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a_new.copy()
                    a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LSE iter={iteration} tau={tau:.2e} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M


def build_convolution_matrix(y):
    n = len(y)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n), dtype=np.float64)
    for j in range(n):
        T[j:j+n, j] = y[j]
    return T


def lp_frank_wolfe(a0, time_limit, solver_name, verbose=True, max_iters=50000):
    """LP-based Frank-Wolfe with warm-starting."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    prev_b_sol = None
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix (linearized around y)
        T = build_convolution_matrix(y)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        
        # Warm start
        if prev_b_sol is not None:
            b_var.value = prev_b_sol
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        prev_b_sol = b_val.copy()
        
        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Intensive coordinate descent targeting the peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find contributing indices to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        if contrib.max() < 1e-20:
            stall += 100
            continue
        
        # Compute sensitivity for top peaks
        top_peaks = np.argsort(ac)[::-1][:min(5, len(ac))]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                if 0 <= pk_int - pos < n:
                    sensitivity[pos] += a[pk_int - pos]
        
        improved = False
        # Try moving from high-contribution to low-sensitivity positions
        src_order = np.argsort(contrib)[::-1][:min(20, len(contrib))]
        
        for si in src_order:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[si])
            if a[src] < 1e-15:
                continue
            
            # Find best destinations (low sensitivity)
            candidates = np.argsort(sensitivity)[:min(100, n // 5)]
            
            for dst in candidates:
                if dst == src:
                    continue
                if a[dst] > 0.5:  # Don't move to already-large positions
                    continue
                
                # Try several fraction sizes
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m + 0.01 * best_m:  # Allow slight increase for exploration
                        pass  # Keep the perturbation
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDescent ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
