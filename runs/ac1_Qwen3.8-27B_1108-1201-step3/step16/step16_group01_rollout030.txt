def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad_fft(a, p):
    """Compute g[i] = sum_k p[k] * a[k-i] via FFT cross-correlation."""
    n = len(a)
    p_len = len(p)
    size = 1 << (n + p_len - 1).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    g = np.fft.irfft(P * np.conj(A), size)[:n]
    return g

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t0 = time.monotonic()

    v = np.zeros(n)
    t = 0
    stall = 0
    total_iters = max(1, int(time_limit * 3000))  # estimate

        if elapsed > time_limit:
            break
        t += 1

        tau = tau0 + (tau_end - tau0) * frac

        M = ac.max()

            best_a = a.copy()
            stall = 0

            stall += 1

        p_sum = p.sum()
        if p_sum < 1e-30:

        g = compute_grad_fft(a, p)
