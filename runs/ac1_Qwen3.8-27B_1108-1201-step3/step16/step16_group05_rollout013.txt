def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. Nesterov APG with adaptive smoothed-max (broad optimization)
2. LP Frank-Wolfe with Nesterov acceleration (direction finding)
3. Peak-directed coordinate moves (local refinement)
4. Multi-length exploration (structural search)
5. Spectral perturbation (harmonic refinement)

import numpy as np
import cvxpy as cp
import time

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    print("  LP solver fallback: ECOS")
    return _lp_solver


def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fz = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fz)
    return np.fft.irfft(A * A, fz)[:size]


def proj_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def norm1(a):
    """Clip to non-negative and normalize to unit sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_max_grad(a, beta, top_k):
    """
    Gradient of (1/beta)*log(sum(exp(beta*c_k))) where c=conv(a,a).
    Returns (gradient_vector, max_c_value).
    """
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    ke = min(top_k, len(c))
    top = np.argsort(c)[::-1][:ke]
    tv = c[top]
    sh = tv - tv[0]
    ev = np.exp(beta * sh)
    w = ev / ev.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top):
        ki = int(k)
        kmj = ki - np.arange(n)
        mask = (kmj >= 0) & (kmj < n)
        if mask.any():
            grad[mask] += 2.0 * wi * a[kmj[mask]]
    return grad, M


def run_apg(a0, time_limit, beta_start, beta_end, top_k_start, top_k_end,
            c_accel, verbose, seed):
    """Nesterov accelerated projected gradient with adaptive smoothed max."""
    n = len(a0)
    x = norm1(a0)
    best_x = x.copy()
    best_M = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x_prev = x.copy()
    stall = 0
    max_stall = 2000
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / 30000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        tk = int(top_k_start + (top_k_end - top_k_start) * frac)
        
        # Nesterov extrapolation
        bt = (it - 1.0) / (it + 1.5)
        y = x + bt * (x - x_prev)
        y = proj_simplex(y)
        
        # Gradient
        g, _ = smooth_max_grad(y, beta, tk)
        
        # Step size
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            gn = 1e-15
        eta = c_accel / (np.sqrt(it) + 1.0)
        
        # Projected step
        x_new = proj_simplex(y - eta * g)
        x_prev = x
        x = x_new
        
        # Check improvement
        M_new = fast_conv(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.25)
            noise = rng.normal(0, 0.003 * (1 + stall * 0.1), n) * best_x
            x_re = np.maximum(best_x + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose and it % 10000 == 0:
                el = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    APG restart it={it} ({el:.0f}s): J={J:.6f}")
        
        if verbose and it % 20000 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    APG it={it} ({el:.0f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_x, best_M


def run_lp_fw(a0, time_limit, solver, verbose, seed):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    y = norm1(a0)
    best_x = y.copy()
    best_M = fast_conv(y).max()
    t0 = time.monotonic()
    x_mom = y.copy()
    it = 0
    rng = np.random.default_rng(seed)
    
    # Precompute the Toeplitz structure
    # T[i,j] = 1 if i >= j and i-j < n (for the convolution structure)
    # Actually, conv(y,b)[k] = sum_j y[k-j]*b[j] for valid indices
    # We need: for each k in [0, 2n-2], sum_j y[k-j]*b[j] <= M
    # This is T @ b <= M where T[k,j] = y[k-j] if 0<=k-j<n else 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_conv(y)
        M_curr = c.max()
        if M_curr < 1e-14:
            break
        
        # Build LP
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraint matrix T (2n-1 x n)
        # T[k, j] = y[k-j] if 0 <= k-j < n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if time.monotonic() - t0 > time_limit * 0.9:
                break
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov FW update
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * x_mom + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        bt = (it - 1.0) / (it + 2.0)
        x_mom = y_new + bt * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx
        
        y = y_new
        
        # Check improvement
        for cand in [y, x_mom]:
            mc = fast_conv(cand).max()
            if mc < best_M - 1e-14:
                best_M = mc
                best_x = cand.copy()
        
        if verbose and it % 5 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW it={it} ({el:.0f}s): J={J:.6f}")
        
        # Periodic restart
        if it % 100 == 0:
            noise = rng.normal(0, 0.001, n) * best_x
            x_re = np.maximum(best_x + noise, 1e-12)
            x_re = x_re / x_re.sum()
            y = x_re
            x_mom = x_re
    
    return best_x, best_M


def peak_directed_moves(a0, time_limit, verbose, seed):
    """Peak-directed coordinate descent: move mass away from peak contributors."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Top peak lags
        top_peaks = np.argsort(c)[::-1][:10]
        
        # Compute contribution of each position to peak lags
        contrib = np.zeros(n)
        for k in top_peaks:
            ki = int(k)
            kmj = ki - np.arange(n)
            mask = (kmj >= 0) & (kmj < n)
            contrib[mask] += a[kmj[mask]] * (c[ki] / M_curr)
        
        # Sources: positions with high contribution
        sources = np.argsort(contrib)[::-1][:50]
        
        improved = False
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Find best destination: low contribution
            # Score each position by its contribution (lower = better destination)
            # Use random subset for efficiency
            n_try = min(n, 300)
            dsts = rng.choice(n, n_try, replace=False) if n > n_try else np.arange(n)
            
            best_dst = -1
            best_score = 0.0
            for dst in dsts:
                if dst == src:
                    continue
                if contrib[dst] > 0.5 * contrib[src]:
                    continue
                score = contrib[src] - contrib[dst]
                if score > best_score:
                    best_score = score
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            # Try moving mass
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[best_dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-15:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if n > 2:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.003, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        M_new = fast_conv(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0) / a.sum()
    
    if verbose:
        el = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Peak-directed done ({el:.0f}s): J={J:.6f}")
    return best_a, best_M


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def spectral_perturb(a0, time_limit, verbose, seed):
    """Try adding small sinusoidal perturbations to reduce max convolution."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Try various frequencies and amplitudes
    freqs = np.arange(1, max(2, n // 100) + 1)
    amplitudes = [0.001, 0.003, 0.005, 0.01, 0.02]
    
    for freq in freqs:
        if time.monotonic() - t0 > time_limit:
            break
        for amp in amplitudes:
            if time.monotonic() - t0 > time_limit:
                break
            for phase in [0.0, np.pi/3, 2*np.pi/3, np.pi/2]:
                x = np.arange(n)
                pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
                a_t = a + pert
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 0.01:
                    continue
                a_t = a_t / s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-15:
                    best_M = M_t
                    best_a = a_t.copy()
                    # Try to increase amplitude
                    for amp2 in [amp * 2, amp * 3, amp * 5]:
                        a_t2 = a + amp2 * np.sin(2 * np.pi * freq * x / n + phase)
                        a_t2 = np.maximum(a_t2, 0.0)
                        s2 = a_t2.sum()
                        if s2 < 0.01:
                            continue
                        a_t2 = a_t2 / s2
