The max autocorrelation is a piecewise-linear function of the block (since each conv[k] is linear in each coordinate in the block, except when k = 2j for j in the block, where it's quadratic).

For a single coordinate j:
conv[k](a[j]=x) = base[k] + 2*x*a[k-j] for k != 2j (and k-j in [0,n))
conv[2j](a[j]=x) = base[2j] - a[j]^2 + x^2

So max_k conv[k](x) is the maximum of several linear functions (plus one quadratic). The minimum of this maximum can be found by bisection or by solving a small LP.

Actually, for a single coordinate, the problem is:
minimize max( max_{k != 2j} (base[k] + 2*x*a[k-j]), base[2j] - a[j]^2 + x^2 )
subject to x >= 0

This is a 1D problem that can be solved by ternary search or bisection.

But we also need to maintain the sum constraint. If we change a[j] from c to x, the sum changes by (x-c). To maintain sum = sqrt(2n), we need to redistribute the difference to other coordinates.

This makes it more complex. Let me simplify: fix the sum by adjusting all coordinates proportionally (or just the non-peak coordinates).

Actually, a simpler approach: work in the "fractional" domain. Let a[i] = s * p[i] where s = sqrt(2n) and p[i] >= 0, sum(p) = 1. Then we're optimizing over the probability simplex, and the objective is:
max_k (s^2 * sum_i p[i]*p[k-i]) = s^2 * max_k conv_p[k]

So we minimize max_k conv_p[k] where p is a probability vector. This is cleaner.

In this formulation, changing p[j] to p[j]+delta (and subtracting delta/n from all other components to maintain sum=1) gives:
conv_p[k] changes by: 2*delta*p[k-j] - delta/n * (2*sum_i p[i]) ... this gets messy.

Let me just work with the sum-constrained formulation and use a simple approach:
1. Change a[j] by delta
2. Rescale all components to restore sum = sqrt(2n)
3. Check if the max autocorrelation decreased

This is what the previous coordinate descent does. Let me make it more efficient.

**Efficient coordinate descent:**
For each coordinate j:
1. Compute the current conv.
2. For candidate values x of a[j]:
   a. Update conv: conv[k] += 2*(x - a[j])*a[k-j] for k in [j, j+n-1] (with correction at k=2j)

   c. Check if max(new_conv) < current max
3. Accept the best x

The key efficiency gain: instead of computing the full FFT each time, we update conv incrementally.

For n=1600, the conv has 3199 elements. Updating it for one coordinate takes O(n) time. Trying 10 candidates takes O(10n) = O(16000). With n=1600 coordinates per sweep, one sweep takes O(1600 * 16000) = O(25.6M) operations. That's fast enough to do many sweeps.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based optimization,
Polyak subgradient, and exact coordinate descent.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def build_constraint_matrix(a, active_set):
    """Build the constraint matrix T where T[k, j] = a[k-j] if valid."""
    n = len(a)
    k_arr = np.asarray(active_set)
    size = len(k_arr)
    T = np.zeros((size, n), dtype=np.float64)
    for idx, k in enumerate(k_arr):
        k = int(k)
        # T[idx, j] = a[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)  # exclusive
        if lo < hi:
            # j = lo, ..., hi-1; k-j = k-lo, ..., k-(hi-1)
            T[idx, lo:hi] = a[k - np.arange(lo, hi)]
    return T

def lp_best_response(a, M, active_set, solver):
    """Solve: max sum(b) s.t. T @ b <= M, b >= 0."""
    n = len(a)
    k_arr = np.asarray(active_set)
    size = len(k_arr)
    T = build_constraint_matrix(a, k_arr)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def lp_min_max_crosscorrelation(a, s_target, active_set, solver):
    """Solve: min t s.t. (a*b)[k] <= t for k in active_set, sum(b) = s_target, b >= 0."""
    n = len(a)
    k_arr = np.asarray(active_set)
    size = len(k_arr)
    T = build_constraint_matrix(a, k_arr)
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [
        T @ b_var <= t_var,
        cp.sum(b_var) == s_target
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response optimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1
    
    y = a.copy()
    y_prev = a.copy()
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Active set: top peaks + threshold
        progress = min(1.0, t / 500.0)
        tol = 0.1 * (1 - progress) + 0.01 * progress
        threshold = M_curr * (1.0 - tol)
        active = np.where(conv_y >= threshold)[0]
        n_top = max(20, n // 40)
        top_k = np.argsort(conv_y)[-n_top:]
        active = np.unique(np.concatenate([active, top_k]))
        if len(active) > 2 * n:
            # Keep top 2n
            vals = conv_y[active]
            keep = np.argsort(vals)[-2*n:]
            active = active[keep]

        # Solve LP
        b, duals = lp_best_response(y, M_curr, active, solver)
        if b is None:
            # Retry with full active set (but capped)
            active = np.argsort(conv_y)[-n*2:]
            b, duals = lp_best_response(y, M_curr, active, solver)
            if b is None:
                continue

        g = normalize_to(b, n)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, n)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, n)
        
        y_prev = y
        y = y_new
        a = a_new

        # Check improvement
        for candidate in [y, a]:
            sc = fast_autocorr(candidate).max()
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = candidate.copy()
                stall = 0
            else:
                stall += 1

        # Restart if stalled
        if stall > 100:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall = 0

        if verbose and t % 100 == 0:
            print(f"    LP iter={t}: best={best_score:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_a, best_score

def polyak_subgradient(a0, time_limit, seed=42):
    """Subgradient method with Polyak step size for min max(autocorr)."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    z_star = best_score  # Lower bound estimate
    
    t0 = time.monotonic()
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Subgradient: pick the argmax peak (or top few)
        # For simplicity, use the single max peak
        k_star = np.argmax(conv)
        
        # Subgradient of conv[k_star] w.r.t. a
        grad = np.zeros(n)
        for j in range(n):
            # d conv[k_star] / d a[j] = a[k_star - j] + a[k_star - j] (if both valid)
            # Actually: conv[k] = sum_i a[i]*a[k-i]
            # d/d a[j]: a[k-j] (from i=j term) + a[k-j] (from k-i=j term)
            # = 2*a[k-j] if 0 <= k-j < n and j != k-j
            # = 2*a[j] if k = 2j (since i=j appears only once)
            # Wait, let me be more careful.
            # conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
            # d/d a[j]: for each term where i=j or k-i=j
            # If i=j: term is a[j]*a[k-j], derivative w.r.t. a[j] is a[k-j]
            # If k-i=j, i=k-j: term is a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j]
            # If j = k-j (i.e., k=2j): same term counted once, derivative is 2*a[j]
            # So: d conv[k]/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] if 0<=k-j<n and k!=2j
            #                                   = 2*a[j] if k=2j
            #                                   = a[k-j] if only one of i=j or k-i=j is valid
            #                                   = 0 if neither is valid
            
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                if k_star == 2 * j:
                    grad[j] = 2.0 * a[j]
                else:
                    grad[j] = 2.0 * a[k_minus_j]
            # Also need to check if k-j is valid but j is... wait, j is always in [0,n)
            # The other case: k-i = j means i = k-j, which is valid if 0 <= k-j < n
            # That's the same condition. So we're good.
        
        # Project gradient: subtract mean on support to maintain sum constraint
        # Actually, we want to move in direction -grad while maintaining sum(a) = norm
        # and a >= 0. Use projected subgradient.
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Polyak step: step = (M_curr - z_star) / ||grad||^2
        polyak_step = (M_curr - z_star) / (g_norm * g_norm)
        # Cap the step
        max_step = 2.0 / (g_norm * n ** 0.3)
        step = min(polyak_step, max_step)
        
        # Projected update
        a_trial = a - step * grad
        # Project onto {x >= 0, sum(x) = norm}
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = np.ones(n) * norm / n
        else:
            a_trial = a_trial * norm / s_trial
        
        # Check improvement
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            z_star = score_trial * 0.999  # Slightly optimistic lower bound
            patience = 0
        else:
            patience += 1
            if patience > 500:
                break
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=0, verbose=False):
    """Coordinate descent with incremental autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        conv = fast_autocorr(a)
        current_max = conv.max()
        
        # Process coordinates in order of sensitivity (high first)
        # Sensitivity of max conv to a[j] is related to how much a[j] contributes to peaks
        order = np.arange(n)
        rng.shuffle(order)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            if aj < 1e-10:
                # Try setting to a small positive value
                candidates = [norm / n * 0.5, norm / n]
            else:
                typical = norm / n
                candidates = [
                    0.0,
                    aj * 0.5,
                    aj * 0.75,
                    aj * 1.25,
                    aj * 1.5,
                    typical * 0.5,
                    typical,
                    typical * 1.5,
                    typical * 2.0,
                ]
                # Add candidates based on local sensitivity
                # The peak at k=2j has value related to aj^2
                candidates.append(aj * 0.8)
                candidates.append(aj * 1.1)
            
            # Remove duplicates and out-of-range
            candidates = list(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))
            candidates.sort()
            
            best_x = aj
            best_obj = current_max
            
            s_others = norm - aj  # Wait, this isn't right after normalization
            # Actually, let's work with the raw values
            s = a.sum()
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                
                # Compute new autocorrelation
                # Delta = x - aj
                delta = x - aj
                new_conv = conv.copy()
                
                # Update: conv[k] += 2*delta*a[k-j] for valid k
                # k ranges from j to j+n-1
                lo = max(0, j)
                hi = min(size, j + n)
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                
                # Correction at k = 2j
                if 2 * j < size:
                    # We added 2*delta*a[j] but should have added (x^2 - aj^2) = delta*(x+aj)
                    # Difference: delta*(x+aj) - 2*delta*a[j] = delta*(x+aj-2*aj) = delta*(x-aj) = delta^2
                    new_conv[2 * j] += delta * delta
                
                # Rescale to maintain sum = norm
                new_sum = s - aj + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale
                
                if new_max < best_obj - 1e-14:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-12:
                # Apply update
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                
                # Verify and update conv
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    conv = fast_autocorr(a)
                    current_max = conv.max()
                    improved = True
                else:
                    # Revert
                    a = best_a.copy()
                    conv = fast_autocorr(a)
                    current_max = conv.max()
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweep % 2 == 0:
            print(f"    CD sweep={sweep}: best={best_score:.10f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score

def subgradient_balanced(a0, time_limit, seed=7):
    """Multi-peak subgradient with balanced weighting and adaptive step."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, n)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 5.0 / (n ** 0.5)
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Select top peaks with diversity
        n_peaks = max(5, min(20, n // 50))
        top_idx = np.argsort(conv)[-n_peaks:]
        
        # Weight by peak height
        peak_vals = conv[top_idx]
        weights = np.exp(-10.0 * (M_curr - peak_vals) / max(M_curr * 0.05, 1e-10))
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for j in range(n):
                kmj = k - j
                if 0 <= kmj < n:
                    if k == 2 * j:
                        grad[j] += w * 2.0 * a[j]
                    else:
                        grad[j] += w * 2.0 * a[kmj]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Make gradient orthogonal to all-ones (to maintain sum)
        grad = grad - np.mean(grad)
        
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = np.ones(n) * norm / n
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 10.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-12 or patience > 1000:
                break
    
    return best_a, best_score

def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s  # Normalize to sum=1 for evaluation
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

def compute_subgradient_single(a, k, n):
    """Compute subgradient of conv[k] w.r.t. a."""
    grad = np.zeros(n)
    for j in range(n):
        kmj = k - j
        if 0 <= kmj < n:
            if k == 2 * j:
                grad[j] = 2.0 * a[j]
            else:
                grad[j] = 2.0 * a[kmj]
    return grad
