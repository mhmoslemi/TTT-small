    Iterative spectral shaping: minimize peak autocorrelation by
    iteratively flattening the magnitude spectrum.
    
    Algorithm:
    1. Compute DFT of a
    2. Compute autocorrelation (IFFT of |A|^2)
    3. Find the target: we want autocorrelation to be flat
    4. In frequency domain, this means |A|^2 should have a specific shape

        P = A.real ** 2 + A.imag ** 2

def compute_convolution(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def normalize_simplex(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Uses FFT for O(n log n) complexity.
    Returns gradient vector of length n.
    """
    n = len(a)
    if b is None:
        b = compute_convolution(a)
    # Softmax weights
    b_max = b.max()
    exp_mu_b = np.exp(mu * (b - b_max))
    w_sum = exp_mu_b.sum()
    if w_sum < 1e-30:
        w = np.ones_like(b) / len(b)
    else:
        w = exp_mu_b / w_sum

    # Part A: cross-correlation of w and a
    # (w ∘ a)[j] = sum_m w[m+j] * a[m] for m = 0..n-1
    fft_size = 1 << (3 * n - 2).bit_length()
    W_fft = np.fft.rfft(w, fft_size)
    A_fft = np.fft.rfft(a, fft_size)
    part_a = np.fft.irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    # Part B: a[j] * sum_{k=j}^{j+n-1} w[k]
    P = np.zeros(2 * n)
    P[1:] = np.cumsum(w)
    window_sums = P[n:] - P[:n]  # length n: window_sums[j] = P[n+j] - P[j]
    part_b = a * window_sums

    return part_a + part_b


def fista_logsumexp(a0, time_limit, mu_schedule=None, verbose=True):
    """
    FISTA with log-sum-exp smoothing.
    mu_schedule: list of (target_mu, n_iters) pairs, or None for default.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = compute_convolution(a).max()
    t0 = time.monotonic()

    if mu_schedule is None:
        mu_schedule = [(20.0, 2000), (50.0, 2000), (100.0, 2000),
                       (200.0, 2000), (400.0, 2000), (800.0, 2000),
                       (1500.0, 2000), (3000.0, 5000)]

    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    total_iter = 0

    for mu, max_iters in mu_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        # Step size: conservative based on mu
        L = 2.0 + 0.1 * mu  # approximate Lipschitz constant
        eta = 1.0 / L

        for _ in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            total_iter += 1

            # Nesterov extrapolation
            if total_iter <= 1:
                y = x.copy()
            else:
                beta = (t_k - 1.0) / (t_k + 1.5)
                y = x + beta * (x - x_prev)
                y = project_simplex(y, 1.0)

            # Compute gradient
            b = compute_convolution(y)
            grad = compute_logsumexp_gradient(y, mu, b)

            # Gradient step + projection
            x_new = y - eta * grad
            x_new = project_simplex(x_new, 1.0)

            # Update FISTA parameters
            t_k_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            t_k = t_k_new

            x_prev = x
            x = x_new

            # Track best
            M_curr = compute_convolution(x).max()
            if M_curr < best_m - 1e-15:
                best_m = M_curr
                best_a = x.copy()

            if verbose and total_iter % 200 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FISTA iter={total_iter} ({elapsed:.1f}s): J={J:.6f}, mu={mu:.0f}")

    return best_a, best_m


def subgradient_fw(a0, time_limit, verbose=True):
    """
    Subgradient Frank-Wolfe for min max_k (a*a)[k] over simplex.
    Very fast (O(n) per iteration) but slower convergence.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = compute_convolution(a).max()
    t0 = time.monotonic()

    alpha_t = 1.0
    episode = 1

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Every 20000 iters, restart with fresh alpha
        if t % 20000 == 0:
            episode += 1
            alpha_t = 1.0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SG-FW episode={episode} t={t} ({elapsed:.1f}s): J={J:.6f}")

        # Compute convolution and find argmax
        b = compute_convolution(a)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        # Subgradient: g[j] = a[k_star - j] + a[j] for valid indices
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        g = np.zeros(n)
        j_idx = np.arange(j_start, j_end + 1)
        g[j_idx] = a[k_star - j_idx] + a[j_idx]

        # Find min gradient coordinate (FW vertex oracle)
        j_min = int(np.argmin(g))

        # Step size
        alpha_t = 2.0 / (t % 20000 + 2.0)

        # Update
        a_new = (1.0 - alpha_t) * a + alpha_t * np.zeros(n)
        a_new[j_min] = 1.0  # This is wrong, let me fix
        # Actually: a_new = (1-alpha)*a + alpha*e_{j_min}
        a_new = (1.0 - alpha_t) * a.copy()
        a_new[j_min] += alpha_t
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        a = a_new

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, top_k_init=60, top_k_max=200, verbose=True):
    """
    LP Frank-Wolfe with adaptive constraint selection.
    Solves: max sum(b) s.t. (a*b)[k] <= M for top-k active lags, b >= 0.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = compute_convolution(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute convolution
        ac = compute_convolution(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]

        # Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [b_var >= 0]
        M_val = float(M_curr)

        for k in active_lags:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_idx = np.arange(j_start, j_end + 1)
            y_coeff = y[k_int - j_idx]
            if y_coeff.max() < 1e-15:
                continue
            constraints.append(y_coeff @ b_var[j_idx] <= M_val)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Step size
        alpha = 2.0 / (t + 2.0)

        # Update
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        # Check improvement
        mx = compute_convolution(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()

        # Adapt top_k
        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m


def spectral_shape_iterative(a, n_iter=50, relax=0.3, verbose=False):
    """
    Iterative spectral shaping: flatten the power spectrum to reduce convolution peak.
    Uses Gerchberg-Saxton style iterations.
    """
    a = normalize_simplex(a).copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)

        # Power spectrum
        P = mag ** 2
        # Target: flat power spectrum
        mean_P = P.mean()
        # Relaxation: move towards flat
        new_mag = np.sqrt((1.0 - relax) * P + relax * mean_P)

        A_new = new_mag * phase
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

        if verbose and it % 10 == 0:
            M = compute_convolution(a).max()
            print(f"      Spectral iter {it}: M={M:.6f}")

    return a


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with mass transfer."""
    n = len(a0)
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_m = compute_convolution(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    fracs = [0.003, 0.007, 0.015, 0.03, 0.06, 0.1]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = compute_convolution(a)
        M_curr = b.max()
        k_peak = int(np.argmax(b))
        top_k = np.argsort(b)[::-1][:min(15, len(b))]
        improved = False

        # Focus on the peak lag
        k_target = k_peak
        j_start = max(0, k_target - n + 1)
        j_end = min(n - 1, k_target)
        idx = np.arange(j_start, j_end + 1)
        contrib = a[idx] * a[k_target - idx]

        if contrib.max() < 1e-20:
            stall += 1
            continue

        # Find source: element contributing most to peak
        src_local = int(np.argmax(contrib))
        src = idx[src_local]

        # Find destination: minimize impact on other peaks
        # Compute sensitivity: for each candidate dst, how much would it increase other peaks?
        other_peaks = top_k[1:8]  # exclude the main peak
        candidates = np.where(a < a.max() * 0.5)[0]
        if len(candidates) < 100:
            candidates = np.arange(n)
        chosen = candidates[rng.permutation(len(candidates))[:min(300, len(candidates))]]

        best_dst = -1
        best_score = float('inf')

        for dst in chosen:
            if dst == src:
                continue
            score = 0.0
            for k2 in other_peaks:
                k2i = int(k2)
                d1 = k2i - dst
                d2 = dst - k2i
                add = 0.0
                if 0 <= d1 < n:
                    add += a[d1]
                if 0 <= d2 < n and d2 != d1:
                    add += a[d2]
                score += add
            if score < best_score:
                best_score = score
                best_dst = dst

        if best_dst < 0:
            stall += 1
            continue

        dst = best_dst
        found = False
        for frac in fracs:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = compute_convolution(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                found = True
                break

        if not improved:
            stall += 1
            # Random perturbation
            if a.max() > 1e-10:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = compute_convolution(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / max(a.sum(), 1e-15)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def make_window(n, kind='hann'):
    """Create various window functions as initializations."""
    x = np.linspace(0, 1, n)
    if kind == 'hann':
        return 0.5 * (1 - np.cos(2 * np.pi * x))
    elif kind == 'blackman':
        return 0.54 - 0.46 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif kind == 'triangle':
        return 1.0 - 2.0 * np.abs(x - 0.5)
    elif kind == 'cosine':
        return np.sin(np.pi * x)
    elif kind == 'flat':
        return np.ones(n)
    elif kind == 'power1':
        return x ** 0.5
    elif kind == 'power2':
        return (1 - x) ** 0.5
    else:
        return np.ones(n)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize_simplex(parent)
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        best_seq = make_window(n_base, 'hann')
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 0: Spectral pre-conditioning ===
    print(f"\n=== Phase 0: Spectral shaping ===")
    a_spec = spectral_shape_iterative(best_seq, n_iter=30, relax=0.3)
    try_update(a_spec, "spectral")

    # === Phase 1: FISTA with log-sum-exp (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA n={n0}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        a_in = normalize_simplex(best_seq)
        mu_sched = [(10.0, 1500), (30.0, 1500), (80.0, 1500),
                    (200.0, 1500), (500.0, 2000), (1000.0, 2000),
                    (2000.0, 3000), (5000.0, 5000)]
        a_out, _ = fista_logsumexp(a_in, t1, mu_schedule=mu_sched, verbose=True)
        try_update(a_out, f"fista_n{n0}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW n={len(best_seq)}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = normalize_simplex(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, top_k_init=60, top_k_max=180, verbose=True)
        try_update(a_out, f"lpfw_n{len(best_seq)}")

    # === Phase 3: FISTA again (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FISTA refined n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = normalize_simplex(best_seq)
        mu_sched2 = [(100.0, 2000), (300.0, 2000), (800.0, 2000),
                     (2000.0, 3000), (5000.0, 5000), (10000.0, 5000)]
        a_out, _ = fista_logsumexp(a_in, t3, mu_schedule=mu_sched2, verbose=True)
        try_update(a_out, f"fista2_n{n_best}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = normalize_simplex(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try different initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Alt inits, budget={remaining:.0f}s ===")
    if remaining > 150:
        alt_n = 2400
        for init_kind in ['blackman', 'cosine', 'triangle']:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            a_init = make_window(alt_n, init_kind)
            a_init = a_init / a_init.sum()
            # Quick spectral shaping
            a_init = spectral_shape_iterative(a_init, n_iter=15, relax=0.4)
            sc = evaluate_safe(a_init)
            print(f"  Init {init_kind}: score={sc:.6f}")
            # Quick FISTA
            t_alt = min(remaining * 0.2, 100.0)
            if t_alt > 20:
                mu_sched_alt = [(50.0, 1000), (200.0, 1000), (500.0, 2000),
                                (1000.0, 2000), (3000.0, 3000)]
                a_out, _ = fista_logsumexp(a_init, t_alt, mu_schedule=mu_sched_alt, verbose=False)
                try_update(a_out, f"fista_{init_kind}")

    # === Phase 6: Subgradient FW (fast, good for fine tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SG-FW n={n_best}, budget={remaining:.0f}s ===")
    if remaining > 60:
        t6 = min(remaining * 0.15, 200.0)
        if t6 > 20:
            a_in = normalize_simplex(best_seq)
            a_out, _ = subgradient_fw(a_in, t6, verbose=True)
            try_update(a_out, f"sgfw_n{n_best}")

    # === Phase 7: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 3000 and remaining > 80
