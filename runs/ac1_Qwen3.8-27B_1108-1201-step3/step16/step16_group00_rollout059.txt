def peak_targeting_step(a, top_L=5):
    """
    Find a direction d that reduces the peak of conv(a,a).
    Returns (d, t_opt) where d is the direction and t_opt is the LP optimal value.
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Find top L peaks
    top_idx = np.argsort(ac)[::-1][:top_L]
    top_idx = sorted(top_idx.tolist())  # sort for consistency
    
    # Build the LP
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [d >= -a]  # a + d >= 0
    constraints.append(cp.sum(d) == 0)  # maintain sum
    
    # For each top peak k*, the constraint is:
    # conv(a, d)[k*] = sum_{i: 0<=k*-i<n} a[k*-i] * d[i] <= t
    for k_star in top_idx:
        k_int = int(k_star)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        coeff = a[k_int - idx]
        constraints.append(coeff @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CLARABEL, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, float('inf')
    
    d_val = np.maximum(np.asarray(d.value, dtype=np.float64), -a + 1e-15)
    return d_val, float(t.value)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-targeting LP descent
with line search, coordinate polish, and multi-length exploration.

import numpy as np
import time
import cvxpy as cp

# Global solver
_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(cp.square(x))), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return _solver
        except:
            continue
    _solver = cp.SCS
    return _solver

def fast_conv(a, b=None):
    if b is None:
        b = a
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss

Now, for the coordinate descent polish, let me think of a more efficient approach:

Instead of random destinations, I'll use a "gradient" approach:
1. Compute the gradient of the peak with respect to each a[i].
2. Move mass from high-gradient positions to low-gradient positions.

The gradient of conv(a,a)[k*] with respect to a[i] is:
d/d a[i] conv(a,a)[k*] = a[k*-i] + a[i-k*] (if valid)

For the max of multiple peaks, the subgradient is the max over the active peaks.

So the "cost" of increasing a[i] is: cost[i] = max_k a[k-i] (over valid k in top peaks).
And the "cost" of decreasing a[i] is negative of this (saving).

The best move is: decrease a[i] at the position with highest cost, increase a[j] at the position with lowest cost.
