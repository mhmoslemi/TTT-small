- If k = i+j: t^2 term = (-t)*(t) + t*(-t) = -2t^2
- If k = 2j: t^2 term = t*(t) = t^2
- Otherwise: 0

So the exact formula is:
conv(a')_k = conv(a)_k + 2t*(a_{k-j} - a_{k-i}) + t^2 * (delta_{k,2i} - 2*delta_{k,i+j} + delta_{k,2j})

For the 1-D minimax problem, we want to find t in [0, a_i] that minimizes:
max_k (conv(a)_k + 2t*(a_{k-j} - a_{k-i}) + t^2 * q_k)

where q_k = delta_{k,2i} - 2*delta_{k,i+j} + delta_{k,2j}

This is a minimax of quadratics, which can be solved numerically (e.g., by bisection on t if we can check the constraint).

Actually, for practical purposes, the t^2 term is usually small (since t is small), so the linear approximation is often good enough. We can use the linear approximation to find a candidate t, then verify with the exact computation.

def fast_conv(a):

    s = a.sum()

        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

        samples = samples[samples != i]

                pass

            # For each top peak k: new_c[k] ≈ c[k] + 2t*(a[k'-j] - a[k'-i])

                val_j = a[k_int - j] if 0 <= k_int - j < n else 0.0
                slopes.append(2.0 * (val_j - val_i))
                base_vals.append(c[k_int])

            base_vals = np.array(base_vals)

            # Minimum is where the "upper envelope" changes slope from negative to positive
            
            # Simple approach: binary search on t
            t_lo = 0.0

            # The optimal t is where the two "tightest" constraints are equal
            # For simplicity, try log-spaced values
            
            best_t_trial = 0
            best_max_trial = base_vals.max()
            
            for t_trial in [t_hi * f for f in [0.01, 0.03, 0.1, 0.3, 0.5, 0.7, 1.0]]:
                new_vals = base_vals + slopes * t_trial
                new_max = new_vals.max()
                if new_max < best_max_trial:
                    best_max_trial = new_max
                    best_t_trial = t_trial
            
            if best_t_trial > 0 and best_max_trial < best_new_max:
                # Verify with actual computation
                t_val = best_t_trial
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                actual_max = fast_conv(a_trial).max()
                if actual_max < best_new_max:
                    best_new_max = actual_max
                    best_pair = (i, j)
                    best_t = t_val
    
    if best_pair is not None:
        i, j = best_pair
        a_new = a.copy()
        a_new[i] -= best_t
        a_new[j] += best_t
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        return a_new, True
    return a, False

And we want to find t that minimizes max_k (conv[k] + t*s[k] + t^2*q[k]).

This is a 1-D problem that can be solved by bisection or by finding the intersection of the two "tightest" lines.

```python
def exact_1d_solve(c, s, q, max_t, n_bisect=20):
    """
    Find t in [0, max_t] that minimizes max_k (c[k] + t*s[k] + t^2*q[k]).
    Uses bisection on the subgradient.
    """
    # The function f(t) = max_k (c[k] + t*s[k] + t^2*q[k]) is convex in t
    # (max of convex functions)
    # So we can use bisection on the derivative (subgradient)
    
    def f(t):
        vals = c + t * s + t * t * q
        return vals.max()
    
    def subgrad(t):
        vals = c + t * s + t * t * q
        idx = np.argmax(vals)
        return s[idx] + 2 * t * q[idx]
    
    # Bisection
    lo, hi = 0.0, max_t
    for _ in range(n_bisect):
        mid = (lo + hi) / 2
        sg = subgrad(mid)
        if sg < 0:
            lo = mid
        else:
            hi = mid
    
    t_star = (lo + hi) / 2
    return t_star, f(t_star)
```

This is much more efficient! We compute the slope vector s once, then do bisection which only requires O(2n) operations per evaluation (just computing max of c + t*s + t^2*q).

    step_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M = c.max()
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        
        for pk in top_peaks[:3]:
            if improved:
                break
            k_int = int(pk)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            
            # Find top contributors
            contribs = []
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    val = a[i] * a[j]
                    if val > 1e-15:
                        contribs.append((val, i))
            contribs.sort(reverse=True)
            
            for _, src in contribs[:max_src_per_peak]:
                if a[src] < 1e-10:
                    continue
                
                # Compute slope vector for this src
                # s[k] = -2 * a[k-src] (decreasing a[src] by t)
                # We need a[k-src] for k = 0..2n-2
                # a[k-src] is non-zero when 0 <= k-src < n, i.e., src <= k < src+n
                slope = np.zeros(2 * n - 1)
                slope[src:src+n] = -2.0 * a
                
                # Now try different destinations
                # Score destinations by how much they help (negative slope at peaks)
                # For dst: additional slope is +2 * a[k-dst]
                # Net slope at peak k: -2*a[k-src] + 2*a[k-dst]
                
                # Sample destinations
                n_samples = min(50, n - 1)
                samples = rng.choice(n, size=n_samples, replace=False)
                samples = samples[samples != src]
                
                # Score: we want dst that makes slope negative at the most critical peaks
                # Critical peaks: those with c[k] > 0.9 * M
                critical = [int(pk2) for pk2 in top_peaks if c[int(pk2)] > 0.85 * M]
                
                dst_scores = []
                for dst in samples:
                    # Net slope at each critical peak
                    score = 0
                    for k2 in critical:
                        aj = a[k2 - dst] if 0 <= k2 - dst < n else 0.0
                        ai = a[k2 - src] if 0 <= k2 - src < n else 0.0
                        if c[k2] > 0.9 * M:
                            score += 2.0 * (aj - ai) * c[k2]  # Weight by peak height
                    dst_scores.append((score, dst))

                        continue

                    net_slope = slope.copy()
                    net_slope[dst:dst+n] += 2.0 * a

def mirror_descent_lse(a0, time_limit, beta_start=5.0, beta_end=50.0, verbose=False, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    step = 0
    eta = 0.1
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fast_conv(a)
        M = c.max()
        
        # Adaptive temperature
        progress = min(1.0, step / 5000)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Log-sum-exp gradient
        bc = beta * c
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        p = exp_bc / exp_bc.sum()
        
        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
        # Use FFT for efficiency
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        A = np.fft.rfft(a, fft_size)
        # We need: grad[i] = 2 * sum_j p[i+j] * a[j]
        # This is the cross-correlation: (p corr a)[i] = sum_j p[i+j] * a[j]
        # = conv(p_rev, a)[len(p)-1-i] where p_rev = p[::-1]
        p_rev = p[::-1]
        P_rev = np.fft.rfft(p_rev, fft_size)
        corr = np.fft.irfft(P_rev * A, fft_size)
        # corr[m] = sum_j p_rev[m-j] * a[j] = sum_j p[2n-2-m+j] * a[j]
        # We want sum_j p[i+j] * a[j], set 2n-2-m+j = i+j => m = 2n-2-i
        grad = np.zeros(n)
        for i in range(n):
            idx = 2 * n - 2 - i
            if 0 <= idx < fft_size:
                grad[i] = 2.0 * corr[idx]
        
        # Mirror descent step on the simplex
        # a_new[i] ∝ a[i] * exp(-eta * grad[i])
        log_step = -eta * grad / a  # This is the gradient of log(a)
        # Actually, for mirror descent with entropy barrier:
        # a_new[i] = a[i] * exp(-eta * grad[i]) / sum_j a[j] * exp(-eta * grad[j])
        
        exp_grad = np.exp(-eta * grad)
        a_new = a * exp_grad
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        m_new = fast_conv(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive step size
        if step % 100 == 0:
            if m_new < M:
                eta = min(eta * 1.1, 1.0)
            else:
                eta = eta * 0.7
    
    return best_a, best_m
