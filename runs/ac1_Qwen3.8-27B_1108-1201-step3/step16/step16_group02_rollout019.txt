            break

        frac = min(1.0, t / 5000.0)
        T = 0.05 * (1 - frac) + 0.005 * frac

        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # f = log(sum(exp(ac/T)))/T
        # df/da[j] = (1/T) * sum_k (exp(ac[k]/T) / sum(exp(ac[k]/T))) * d(ac[k])/da[j]
        # d(ac[k])/da[j] = 2*a[k-j] (for valid k-j)
        
        log_ac = ac / T
        log_ac -= log_ac.max()
        weights = np.exp(log_ac)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(len(ac)):
            w = weights[k]
            if w < 1e-10:
                continue
            k_int = k
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * w * y[j_idx]
        
        # Normalize gradient (it's a subgradient of the sum-normalized objective)
        # Project: grad - mean(grad) to stay on simplex
        grad = grad - grad.mean()
        
        step = 0.1 / (np.sqrt(t) + 1.0)
        a_new = y - step * grad
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.1f}s): score={2*n*best_m:.8f}")
    
    return best_a, best_m

                # Total: conv[2j] - 2v^2 + v^2 = conv[2j] - v^2...

def nesterov_smoothed(a_init, time_limit, verbose=True):
    """Nesterov accelerated descent on smoothed objective log-sum-exp(T*conv)/T."""
    n = len(a_init)
    a = norm_sum1(a_init).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, t / 10000.0)
        T = 0.03 * (1 - frac) + 0.003 * frac
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_autocorr(y)
        
        # Log-sum-exp weights
        scaled = ac / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()
        
        # Gradient: sum_k w_k * 2*y[k-j] for each j
        grad = np.zeros(n)
        # Efficient computation using convolution
        # grad[j] = 2 * sum_k w_k * y[(k-j) mod n] (with appropriate bounds)
        # = 2 * (w * y)[j] (cross-correlation)
        # Actually: grad[j] = 2 * sum_{k: 0<=k-j<n} w_k * y[k-j]
        # Let m = k-j, then k = j+m, sum_m w_{j+m} * y[m]
        # This is a cross-correlation of w and y
        
        # Use FFT for efficiency
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = np.fft.rfft(weights, fft_size)
        Y = np.fft.rfft(y, fft_size)
        # Cross-correlation: sum_k w[k] * y[k-j] = (w ⋆ y)[j]
        # In FFT: ifft(fft(w) * conj(fft(y))) gives correlation
        # Actually, let me just do it directly for the top weights
        
        # For efficiency, use only top-k weights
        top_k = min(100, len(weights))
        top_idx = np.argsort(weights)[::-1][:top_k]
        top_w = weights[top_idx]
        top_w = top_w / top_w.sum()
        
        for wi, k in zip(top_w, top_idx):
            if wi < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()

            best_a = x.copy()
