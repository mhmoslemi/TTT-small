"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation L-BFGS on log-sum-exp smoothed objective (novel second-order approach)
2. Nesterov-accelerated mirror descent with full weighted subgradient
3. Multi-strategy initialization (window functions, parent, blends)
4. Adaptive length exploration
5. Aggressive coordinate descent for final polish

import numpy as np
import time
import scipy.optimize as spo


def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def softmax(x):
    """Numerically stable softmax."""
    xm = x - x.max()
    ex = np.exp(xm)
    s = ex.sum()
    if s < 1e-30:
        return np.ones_like(x) / len(x)
    return ex / s


def lse_obj_grad(x, beta):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. x (softmax parameterization)."""
    n = len(x)
    a = softmax(x)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    shifted = beta * (ac - max_ac)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    f = 2.0 * n * (max_ac + np.log(Z) / beta)
    w = exp_s / Z
    # Gradient of J = (1/beta)*log(sum exp(beta*ac)) w.r.t. a
    # G[i] = (1/beta) * (Q[i] + P[i])
    # Q[i] = conv(a, w)[i], P[i] = conv(a, w_rev)[2n-2-i]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    Q_full = np.fft.irfft(A * W, fft_size)
    Q = Q_full[:n]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, fft_size)
    P_full = np.fft.irfft(A * W_rev, fft_size)
    P = P_full[n - 1:2 * n - 1][::-1]
    G = (Q + P) / beta
    Ga = np.dot(G, a)
    grad_x = 2.0 * n * a * (G - Ga)
    return f, grad_x

    n = len(a0)
    if beta_schedule is None:

    # Initialize x from a0: x = log(a0) gives softmax(x) proportional to a0

    t_start = time.monotonic()

                               maxls=50, maxcor=30)
            a_res = softmax(res.x)

            if f_actual < best_f:

        shifted = beta_lse * (ac - max_ac)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        # Compute gradient G using FFT
        fft_size = 1 << (3 * n - 3).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        Q_full = np.fft.irfft(A * W, fft_size)[:n]
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, fft_size)
        P_full = np.fft.irfft(A * W_rev, fft_size)
        P = P_full[n - 1:2 * n - 1][::-1]
        G = (Q_full + P) / beta_lse

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * G
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 5000:
            stall_count = 0
            # Restart with perturbation
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta_lse = max(beta_lse * 0.7, 50.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta_lse:.0f}")

        # Adapt beta_lse
        if t % 10000 == 0:
            beta_lse = min(beta_lse * 1.2, 5000.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Aggressive coordinate descent with 2-point moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        # Try moving mass from high-contribution positions to low-contribution ones
        for _ in range(min(50, n // 20)):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick a source from a peak-contributing position
            k = top_k[int(rng.integers(len(top_k)))]
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try random destinations
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if time.monotonic() - t0 > time_limit:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 25 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_raised_cosine(n):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 0.5 * (1.0 + np.cos(x))
    return norm_sum1(a)


def make_hamming(n):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 0.54 - 0.46 * np.cos(x)
    return norm_sum1(a)


def make_blackman(n):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 0.42 - 0.5 * np.cos(x) + 0.08 * np.cos(2 * x)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_triangular(n):
    x = np.linspace(-1, 1, n)
    a = np.maximum(1.0 - np.abs(x), 0.0)
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator using continuation L-BFGS + AMD + coordinate descent."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Generate initializations ===
    print(f"=== Phase 0: Initialization ===")
    print(f"Total time budget: {time_budget:.0f}s")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    n0 = len(best_seq) if best_seq is not None else 3200

    if best_seq is None:
        best_seq = make_raised_cosine(3200)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
        n0 = 3200

    # Generate candidate initializations at n0
    inits = []
    labels = []
    if parent is not None:
        inits.append(norm_sum1(parent))
        labels.append("parent")
    inits.append(make_raised_cosine(n0))
    labels.append("raised_cos")
    inits.append(make_hamming(n0))
    labels.append("hamming")
    inits.append(make_blackman(n0))
    labels.append("blackman")
    inits.append(make_triangular(n0))
    labels.append("triangular")
    inits.append(make_smooth_random(n0, seed=7))
    labels.append("rand7")
    inits.append(make_smooth_random(n0, seed=42))
    labels.append("rand42")
    # Blend parent with raised cosine
    if parent is not None:
        blend = 0.5 * norm_sum1(parent) + 0.5 * make_raised_cosine(n0)
        inits.append(norm_sum1(blend))
        labels.append("blend")

    # Evaluate all inits
    init_scores = []
    for init, lbl in zip(inits, labels):
        sc = 2.0 * len(init) * fast_autocorr(init).max()
        init_scores.append((sc, lbl, init))
    init_scores.sort(key=lambda x: x[0])
    print(f"Init scores: {[(l, f'{s:.6f}') for s, l, _ in init_scores[:5]]}")

    # === Phase 1: Continuation L-BFGS on top initializations ===
    print(f"\n=== Phase 1: Continuation L-BFGS ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.40, 600.0)

    if t_phase1 > 30:
        # Select top 3 unique initializations
        selected = init_scores[:3]
        t_per = t_phase1 / len(selected)
        for sc, lbl, init in selected:
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  L-BFGS on '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = run_lbfqgs_continuation(init, t_avail, verbose=True)
            try_update(a_out, f"lbfqgs_{lbl}")

    # === Phase 2: AMD on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: AMD (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t2, verbose=True)
        try_update(a_out, f"amd_n{len(a_in)}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate descent (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 4: Try L-BFGS again on best (with higher beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BFGS refinement (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        # Use higher beta schedule for refinement
        a_out, _ = run_lbfqgs_continuation(a_in, t4,
                                            beta_schedule=[500, 1000, 2000, 4000],
                                            verbose=True)
        try_update(a_out, f"lbfqgs_refine_n{len(a_in)}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")
    candidate_ns = []
    for delta in [-400, -200, 200, 400, 800]:
        nn = n_best + delta
        if 1500 <= nn <= 5000:
            candidate_ns.append(nn)
    # Remove duplicates and sort
    candidate_ns = sorted(set(candidate_ns))

    for new_n in candidate_ns:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t5 = min(remaining * 0.3, 120.0)
        if t5 < 20:
            break
        a_out, _ = run_lbfqgs_continuation(a_long, t5,
                                            beta_schedule=[200, 500, 1000, 2000],
                                            verbose=False)
        try_update(a_out, f"lbfqgs_n{new_n}")

    # === Phase 6: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final AMD (n={len(best_seq)}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.3, 200.0)
        if t6 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t6, verbose=False)
            try_update(a_out, f"amd_final_n{len(a_in)}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final coordinate (n={len(best_seq)}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{len(a_in)}")

    # === Phase 8: One more L-BFGS with very high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 8: High-beta L-BFGS (n={len(best_seq)}) ===")
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = run_lbfqgs_continuation(a_in, t8,
                                                beta_schedule=[2000, 5000, 10000],
                                                verbose=False)
            try_update(a_out, f"lbfqgs_highbeta_n{len(a_in)}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_raised_cosine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
