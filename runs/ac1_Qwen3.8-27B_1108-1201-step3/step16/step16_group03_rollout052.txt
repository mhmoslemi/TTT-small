def norm_sum1(a):
    """Normalize so that sum = 1 and all entries >= 0."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_damage(a, peak_indices, n):
    """Compute damage vector: for each coordinate j, how much it contributes to top peaks."""
    damage = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        for j in range(n):
            jj = k - j
            if 0 <= jj < n:
                damage[j] += a[j] * a[jj]
    return damage


def nesterov_projected_gradient(a0, time_limit, verbose=True, seed=42):
    """
    Nesterov accelerated projected gradient descent on log-sum-exp smooth max.
    Uses Euclidean projection (clip + renormalize) instead of mirror descent.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    c0 = 3.0
    beta_smooth = 50.0  # log-sum-exp temperature (smaller = more like true max)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute convolution and gradient of log-sum-exp
        ac = fast_conv(y)
        # Anneal the smoothing: start hot, end cold
        progress = min(t / 2000.0, 1.0)
        beta_curr = beta_smooth * (1.0 - progress) + 5.0 * progress

        # Log-sum-exp gradient: softmax-weighted average of peak gradients
        ac_shifted = ac - ac.max()
        weights = np.exp(beta_curr * ac_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros_like(ac)
            weights[ac.argmax()] = 1.0
        else:
            weights = weights / w_sum

        # Gradient: weighted sum of hard subgradients at each peak position
        grad = np.zeros(n)
        # Only use top peaks for efficiency
        top_k = min(20, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_w = weights[top_idx]
        top_w_sum = top_w.sum()
        if top_w_sum < 1e-30:
            top_w = np.ones(top_k) / top_k
        else:
            top_w = top_w / top_w_sum

        for wi, k in zip(top_w, top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            js = np.arange(i_start, i_end)
            grad[js] += wi * 2.0 * y[k - js]

        # Step with backtracking line search
        eta = c0 / (np.sqrt(t) + 1.0)
        best_x = x.copy()
        best_val = ac.max()

        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.1]:
            x_new = y - trial_eta * grad
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                continue
            x_new = x_new / sx
            val = fast_conv(x_new).max()
            if val < best_val:
                best_val = val
                best_x = x_new.copy()
                break

        x_prev = x
        x = best_x

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NPG t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def peak_targeted_sa(a0, time_limit, verbose=True, seed=42, t0_init=None):
    """
    Simulated annealing with peak-targeted moves and O(n) incremental convolution.
    Each iteration:
      1. Find max peak position
      2. Identify source coordinate (largest contributor to max peak)
      3. Identify destination (smallest damage to top peaks)
      4. Move small amount of mass, update convolution incrementally
      5. Accept/reject based on SA criterion
    """
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_conv(a)
    best_a = a.copy()
    best_m = conv.max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)

    # SA parameters
    T_init = max((best_m - conv[np.argsort(conv)[::-1][1]]) * 0.1, best_m * 0.01)
    T_min = best_m * 0.0001
    cool_rate = 0.9997  # very slow cooling

    T = T_init if t0_init is None else t0_init
    n_iter = 0
    n_accept = 0
    conv_recount = 0

    # Precompute valid index ranges for each possible peak position
    # For peak at position k: valid j range is [max(0,k-n+1), min(n,k+1))

    while time.monotonic() - t_start < time_limit:
        # Find max peak
        k_max = int(np.argmax(conv))
        M_curr = conv[k_max]

        # Find source: coordinate with largest contribution to peak k_max
        j_start = max(0, k_max - n + 1)
        j_end = min(n, k_max + 1)
        js = np.arange(j_start, j_end)
        contribs = a[js] * a[k_max - js]
        src_local = int(np.argmax(contribs))
        src = js[src_local]

        if a[src] < 1e-12:
            # Fallback: random move
            src = int(rng.integers(n))
            if a[src] < 1e-12:
                T *= cool_rate
                continue

        # Find top-5 peaks for damage computation
        top5 = np.argsort(conv)[::-1][:5]
        top5 = top5[top5 != k_max]  # exclude the current max (or keep it)

        # Compute damage: sum of contributions to top peaks
        dmg = np.zeros(n)
        for k2 in top5[:4]:
            k2 = int(k2)
            for j in range(n):
                jj = k2 - j
                if 0 <= jj < n:
                    dmg[j] += a[j] * a[jj]

        # Destination: minimize damage, avoid neighbors of src
        # Add small random perturbation to damage for exploration
        dmg_perturbed = dmg + rng.uniform(0, dmg.max() * 0.01 + 1e-10, n)
        # Exclude src and its immediate neighbors
        exclude = set([src])
        for offset in [-2, -1, 0, 1, 2]:
            nn = src + offset
            if 0 <= nn < n:
                exclude.add(nn)
        for ex in exclude:
            if ex < n:
                dmg_perturbed[ex] = np.inf

        dst = int(np.argmin(dmg_perturbed))

        # Choose move amount
        delta = a[src] * rng.uniform(0.005, 0.05)
        if delta < 1e-14:
            T *= cool_rate
            continue

        # Predict new convolution (linear approximation)
        # Delta conv[k] = 2*delta*(a[k-dst] - a[k-src]) for all k
        # Need to be careful about boundaries
        delta_conv = np.zeros(2 * n - 1)

        # Contribution from a[dst] increasing by delta
        for k in range(n):
            kk = k + dst  # position in conv where a[dst] pairs with a[k]
            if 0 <= kk < 2 * n - 1:
                delta_conv[kk] += 2.0 * delta * a[k]

        # Also: a[dst] pairs with a[k-dst] for k in valid range
        # Actually, conv[k] = sum_i a[i]*a[k-i]
        # Increasing a[dst] by delta: adds delta*a[k-dst] to conv[k] for valid k-dst
        # And a[dst]*delta to conv[k] for valid dst as the other operand
        # So total addition to conv[k] from a[dst] += delta:
        #   delta * a[k-dst] (when dst is the "second" index, k-dst is "first")
        #   + delta * a[k-dst] (when dst is the "first" index, k-dst is "second")
        # Wait, conv[k] = sum_i a[i] * a[k-i]. If a[dst] increases by delta:
        #   term where i=dst: delta * a[k-dst] (valid if 0 <= k-dst < n)
        #   term where k-i=dst, i.e., i=k-dst: a[k-dst] * delta (valid if 0 <= k-dst < n)
        #   Total: 2*delta*a[k-dst] if k-dst is valid and dst != k-dst
        #   If k-dst == dst (i.e., k = 2*dst): delta*a[dst] + a[dst]*delta = 2*delta*a[dst]
        #   So it's always 2*delta*a[k-dst] for valid k-dst, counting the diagonal once
        #   Actually no: conv[k] = sum_i a[i]*a[k-i]. The term with i=dst is a[dst]*a[k-dst].
        #   The term with k-i=dst is a[k-dst]*a[dst]. These are the SAME term when i=dst.
        #   So the contribution to conv[k] from a[dst]+=delta is: delta*a[k-dst] (one term, not two!)
        #   UNLESS k-dst = dst, in which case it's delta*a[dst] (still one term, just a_i = a[k-i])
        #   Wait, I'm confusing myself. Let me be very explicit.
        #
        # conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
        # If we add delta to a[dst], the change in conv[k] is:
        #   - If dst is in the valid range for i (0<=dst<n, always true) AND k-dst is valid (0<=k-dst<n):
        #     The term a[dst]*a[k-dst] increases by delta*a[k-dst]
        #   - Additionally, if k-dst is in the valid range AND dst is valid as k-i:
        #     The term a[k-dst]*a[dst] increases by a[k-dst]*delta
        #   - But these are the SAME term! i=dst gives a[dst]*a[k-dst], and this is the only
        #     term that changes. So the total change is delta*a[k-dst] (just once).
        #
        # WAIT. That's only if we think of it as "which i values make a[i] or a[k-i] equal to a[dst]".
        # a[i] = a[dst] when i = dst.
        # a[k-i] = a[dst] when k-i = dst, i.e., i = k-dst.
        # These give two different i values (unless dst = k-dst, i.e., k = 2*dst).
        # So:
        # - Change from i=dst: delta * a[k-dst] (if 0 <= k-dst < n)
        # - Change from i=k-dst: a[dst] * delta (if 0 <= k-dst < n, i.e., 0 <= i < n means 0 <= k-dst < n)
        #   But wait, a[k-dst] * delta only if a[k-dst] is the "fixed" part and a[dst] changes.
        #   The term is a[i]*a[k-i] with i=k-dst: a[k-dst]*a[dst]. If a[dst] increases by delta,
        #   this term increases by a[k-dst]*delta.
        #
        # So total change to conv[k] from a[dst] += delta:
        #   delta * a[k-dst] (from i=dst) + a[k-dst] * delta (from i=k-dst)
        #   = 2 * delta * a[k-dst] (if k != 2*dst, i.e., the two i values are different)
        #   = delta * a[dst] + a[dst] * delta = 2*delta*a[dst] (if k = 2*dst, same i value)
        #
        # In both cases, it's 2*delta*a[k-dst] when 0 <= k-dst < n!
        # (When k=2*dst, a[k-dst] = a[dst], so 2*delta*a[dst] = 2*delta*a[k-dst].)
        #
        # Great, so the formula 2*delta*a[k-dst] is correct for all k where 0 <= k-dst < n.

        # Let me redo this properly with vectorized operations
        delta_conv = np.zeros(2 * n - 1)

        # a[dst] += delta: contributes 2*delta*a[k-dst] to conv[k] for k in [dst, dst+n-1]
        k_range_dst = np.arange(dst, dst + n)
        delta_conv[k_range_dst] += 2.0 * delta * a

        # a[src] -= delta: contributes -2*delta*a[k-src] to conv[k] for k in [src, src+n-1]
        k_range_src = np.arange(src, src + n)
        delta_conv[k_range_src] -= 2.0 * delta * a

        # Second-order correction (usually negligible)
        # At k = src+dst: cross term -2*delta^2
        # At k = 2*src: +delta^2
        # At k = 2*dst: +delta^2
        if 0 <= src + dst < 2 * n - 1:
            delta_conv[src + dst] -= 2.0 * delta * delta
        if 0 <= 2 * src < 2 * n - 1:
            delta_conv[2 * src] += delta * delta
        if 0 <= 2 * dst < 2 * n - 1:
            delta_conv[2 * dst] += delta * delta

        new_conv = conv + delta_conv
        new_max = new_conv.max()

        # SA acceptance
        delta_obj = new_max - M_curr
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-15)):
            # Accept
            a[src] -= delta
            a[dst] += delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                conv = fast_conv(a)
            else:
                # Renormalize (this slightly changes conv, but for small delta it's negligible)
                # Actually, let's NOT renormalize here to keep the incremental update valid
                # We'll renormalize periodically
                conv = new_conv
            n_accept += 1

            if new_max < best_m - 1e-15:
                best_m = new_max
                best_a = a.copy()
        else:
            # Reject: do nothing to a or conv
            pass

        n_iter += 1

        # Periodic renormalization and convolution recount (to fix accumulation errors)
        if n_iter % 5000 == 0:
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_conv(a)
            best_m_check = conv.max()
            if best_m_check < best_m:
                best_m = best_m_check
                best_a = a.copy()
            conv_recount += 1

        # Cooling
        T *= cool_rate

        if verbose and n_iter % 100000 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            rate = n_iter / elapsed if elapsed > 0 else 0
            print(f"    SA iter={n_iter} ({elapsed:.1f}s, {rate:.0f}/s): J={J:.6f}, T={T:.2e}, acc={n_accept/n_iter:.2%}")

    return best_a, best_m


def greedy_peak_shave(a0, time_limit, verbose=True, seed=42):
    """
    Deterministic greedy peak shaving: always make the move that most reduces the max peak.
    No randomness - pure descent.
    """
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_conv(a)
    best_a = a.copy()
    best_m = conv.max()
    t_start = time.monotonic()
    stall = 0
    max_stall = 2000

    while time.monotonic() - t_start < time_limit and stall < max_stall:
        k_max = int(np.argmax(conv))
        M_curr = conv[k_max]

        # Find top peaks
        top_peaks = np.argsort(conv)[::-1][:min(10, len(conv))]

        # For the max peak, find best (src, dst, delta) move
        j_start = max(0, k_max - n + 1)
        j_end = min(n, k_max + 1)
        js = np.arange(j_start, j_end)
        contribs = a[js] * a[k_max - js]

        best_improvement = 0.0
        best_move = None

        # Try each source coordinate
        for src_idx in np.argsort(contribs)[::-1][:10]:  # top 10 contributors
            src = int(js[src_idx])
            if a[src] < 1e-12:
                continue

            # Compute damage for destination selection
            dmg = np.zeros(n)
            for k2 in top_peaks[:5]:
                k2 = int(k2)
                for j in range(n):
                    jj = k2 - j
                    if 0 <= jj < n:
                        dmg[j] += a[j] * a[jj]

            # Try several destinations
            candidates = np.argsort(dmg)[:20]
            for dst in candidates:
                dst = int(dst)
                if dst == src or abs(dst - src) <= 3:
                    continue
                if a[dst] > 0.5:  # don't pour into already-large coordinates
                    continue

                # Try several delta values
                for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = a[src] * frac
                    if delta < 1e-14:
                        continue

                    # Predict new max
                    # Quick estimate: just check the max peak
                    new_peak_val = M_curr - 2.0 * delta * a[k_max - src] if 0 <= k_max - src < n else M_curr
                    # Also check if dst creates a new peak
                    # For k' where a[k'-dst] is large and we're adding 2*delta*a[k'-dst]:
                    # The most dangerous k' is where a is largest near k'-dst
                    # Approximate: check k' = dst + argmax(a)
                    # This is crude but fast

                    if new_peak_val >= best_m - 1e-12:
                        continue

                    # Full check
                    delta_conv_quick = np.zeros(2 * n - 1)
                    k_range_dst = np.arange(dst, dst + n)
                    delta_conv_quick[k_range_dst] += 2.0 * delta * a
                    k_range_src = np.arange(src, src + n)
                    delta_conv_quick[k_range_src] -= 2.0 * delta * a
                    new_conv = conv + delta_conv_quick
                    new_max = new_conv.max()

                    if new_max < best_m - best_improvement:
                        best_improvement = best_m - new_max
                        best_move = (src, dst, delta)

        if best_move is not None:
            src, dst, delta = best_move
            a[src] -= delta
            a[dst] += delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_conv(a)
            best_m = conv.max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print(f"    Shave stall={stall} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diverse_initialization(n, seed=0):
    """Generate a diverse initialization."""
    rng = np.random.default_rng(seed)
    methods = [
        # Smooth noise
        lambda: np.maximum(np.sin(np.pi * np.linspace(0, 1, n)) + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.linspace(0, 1, n)), 0),
        # Two bumps
        lambda: np.exp(-((np.linspace(0, 1, n) - 0.3) ** 2) / 0.02) + np.exp(-((np.linspace(0, 1, n) - 0.7) ** 2) / 0.02),
        # Cosine window
        lambda: 0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(1, 3) * np.linspace(0, 1, n)),
        # Flat with perturbation
        lambda: np.ones(n) + 0.3 * rng.normal(0, 1, n),
    ]
    func = methods[seed % len(methods)]
    a = func()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_n = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            parent_n = len(parent)
    except Exception as e:
        print(f"Could not load parent: {e}")
        best_seq = diverse_initialization(2400, seed=0)
        best_score = evaluate_safe(best_seq)
        print(f"Generated init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick Nesterov projected gradient on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 1: NPG n={n0} ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        a_out, _ = nesterov_projected_gradient(norm_sum1(best_seq), t1, verbose=True, seed=42)
        try_update(a_out, f"npg_n{n0}")

    # === Phase 2: Peak-targeted SA (main workhorse, long run) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 2: SA n={n0} ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_out, _ = peak_targeted_sa(norm_sum1(best_seq), t2, verbose=True, seed=123)
        try_update(a_out, f"sa_n{n0}")

    # === Phase 3: Greedy peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 3: Shave n={n0} ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = greedy_peak_shave(norm_sum1(best_seq), t3, verbose=True, seed=7)
        try_update(a_out, f"shave_n{n0}")

    # === Phase 4: Second SA round (different seed, continue from best) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print(f"\n=== Phase 4: SA round 2 n={n0} ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_out, _ = peak_targeted_sa(norm_sum1(best_seq), t4, verbose=True, seed=999)
        try_update(a_out, f"sa2_n{n0}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        for delta_n in [-200, 200, -100, 100, 400]:
            n_try = n_best + delta_n
            if n_try < 500 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"\n=== Phase 5: Length n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t5 = min(remaining * 0.3, 150.0)
            if t5 > 20:
                a_out, _ = peak_targeted_sa(norm_sum1(a_long), t5, verbose=False, seed=55 + n_try % 100)
                try_update(a_out, f"sa_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = greedy_peak_shave(norm_sum1(best_seq), min(remaining * 0.3, 50.0), verbose=False, seed=77)
                try_update(a_out2, f"shave_n{n_try}")

    # === Phase 6: Final polish with greedy shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final shave n={n_best} ===")
    if remaining > 20:
        t6 = min(remaining * 0.7, 100.0)
        if t6 > 10:
            a_out, _ = greedy_peak_shave(norm_sum1(best_seq), t6, verbose=True, seed=314)
            try_update(a_out, f"final_shave_n{n_best}")

    # === Phase 7: One more SA if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final SA n={n_best} ===")
        a_out, _ = peak_targeted_sa(norm_sum1(best_seq), min(remaining * 0.8, time_budget - (time.monotonic() - t_global) - 10), verbose=True, seed=7777)
        try_update(a_out, f"final_sa_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
