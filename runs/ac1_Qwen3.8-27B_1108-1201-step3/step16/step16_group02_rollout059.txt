```python
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        
        # Active set: lags within 2% of max
        threshold = M_curr * 0.98
        active = np.where(ac >= threshold)[0]
        
        # Uniform weights on active lags
        w = np.zeros(len(ac))
        w[active] = 1.0 / len(active)
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * w[k] * x[j_idx]
        
        # Nesterov step
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        eta = gamma / (np.sqrt(t) + 1.0)
        a_new = y - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.01, n)
            x = best_a * (1 + pert)
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            gamma = max(1.0, gamma * 0.7)
            stall = 0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}, active={len(active)}")
    
    return best_a, best_m

def lp_frank_wolfe_tight(a0, time_limit, solver, verbose=True, use_tight=True):
    """LP Frank-Wolfe with tight constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if use_tight:
            # Use current convolution values as constraints
            # This is tighter than using M for all
            rhs = ac.copy()
            # Add small slack to avoid numerical issues
            rhs = rhs * 1.001 + 1e-15
        else:
            rhs = np.full(2*n-1, M_curr * 1.001 + 1e-15)
        
        # Select a subset of constraints for speed (top K)
        K = min(2*n-1, max(200, int(np.sqrt(2*n) * 30)))
        # Always include the peak lags
        sorted_idx = np.argsort(ac)[::-1]
        selected = sorted_idx[:K]
        
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraint matrix for selected lags
        # (a*b)[k] = sum_i a[i] * b[k-i] for valid i
        # For each selected k, the constraint involves b[k-i] for i where 0<=i<n and 0<=k-i<n
        # i.e., b[j] for j = max(0,k-n+1) to min(n-1, k), with coefficient a[k-j]
        
        # For efficiency, build the constraint matrix
        T = np.zeros((len(selected), n))
        for row_idx, k in enumerate(selected):
            k_int = int(k)
            # b[j] appears in (a*b)[k] with coefficient a[k-j] where j = k-i, i = k-j
            # 0 <= i < n means 0 <= k-j < n means k-n < j <= k
            # 0 <= j < n
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            if j_start < j_end:
                cols = np.arange(j_start, j_end)
                T[row_idx, cols] = a[k_int - cols]
        
        constraints = [T @ b_var <= rhs[selected]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if t % 10 == 0:
                print("    LP solve failed")
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        
        g_dir = b_val / b_sum
        
        # Frank-Wolfe step with diminishing step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={2*n*best_m:.6f}, M={M_curr:.8f}")
    
    return best_a, best_m

def coord_descent_2d(a0, time_limit, verbose=True, rng_seed=42):
    """2D coordinate descent: optimize pairs of coordinates exactly."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 10
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        improved = False
        
        # Try several random pairs
        n_pairs = min(200, n)
        pairs = rng.integers(0, n, size=(n_pairs, 2))
        
        for i, j in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            i, j = int(i), int(j)
            if i == j or a[i] < 1e-15 or a[j] < 1e-15:
                continue
            
            a_i, a_j = a[i], a[j]
            # Transfer mass from i to j: a[i] -> a_i - t, a[j] -> a_j + t
            # t in [0, a_i] (or negative for reverse)
            # The convolution change is quadratic in t, so we can find the optimum analytically
            # (a*a)[k] changes by: -t*(a[k-i] + a[k-i+n]) ... this gets complex
            # Simpler: try a few candidate t values
            
            best_t = 0.0
            best_m_trial = best_m
            
            for t_frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                t_val = t_frac * a_i
                a_trial = a.copy()
                a_trial[i] = a_i - t_val
                a_trial[j] = a_j + t_val
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m_trial:
                    best_m_trial = m_trial
                    best_t = t_val
            
            if best_t > 0 and best_m_trial < best_m:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_m_trial
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def softmax_subgradient(a, ac, temperature):
    """Compute soft-max weighted subgradient."""
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(50, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = len(top_idx)
    weights /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(ac)):
        w_k = weights[k]
        if w_k < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w_k * a[j_idx]
    return grad

def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.001,
                            c0=3.0, verbose=True):
    """Nesterov accelerated mirror descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, t / 5000.0)
        temp = initial_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-5)
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        if temp > 0.005:
            g = softmax_subgradient(y, ac, temp)
        else:
            # Top-k with power weighting
            k_max = min(80, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** 4.0
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
            
            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * y[j_idx]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            stall = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.005, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.5)
        
        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall}")
    
    return best_a, best_m
```

The tight constraint LP is the key innovation. By using c[k] (the actual convolution values) as the right-hand side, the LP is forced to find a b that doesn't increase ANY convolution value, not just the peak. This should lead to a flatter convolution overall.

Let me also add another innovation: **iterative M reduction**. After the LP-FW phase, we can try to reduce M incrementally:
1. Set M_target = current_max * (1 - epsilon) where epsilon is small.
2. Check if the LP max sum(b) s.t. (a*b)[k] <= M_target, b >= 0 is feasible.
3. If feasible, update a towards the solution.
4. If infeasible, decrease epsilon.

This is like a sequential quadratic programming approach applied to our problem.

```python
def min_norm_subgradient(a, ac, active_lags):
    """Compute minimum norm subgradient over active lags."""
    n = len(a)
    # Build subgradient matrix: each row is the subgradient for one active lag
    m = len(active_lags)
    G = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        cols = np.arange(i_start, i_end)
        G[idx, cols] = 2.0 * a[k_int - cols]
    
    # Solve: min ||G^T w||^2 s.t. sum w = 1, w >= 0
    # This is a QP. Use cvxpy.
    w_var = cp.Variable(m)
    obj = cp.Minimize(cp.sum_squares(G.T @ w_var))
    constraints = [cp.sum(w_var) == 1, w_var >= 0]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        w_val = np.ones(m) / m
    else:
        w_val = np.maximum(np.asarray(w_var.value, dtype=np.float64), 0)
        ws = w_val.sum()
        if ws < 1e-15:
            w_val = np.ones(m) / m
        else:
            w_val = w_val / ws
    
    return G.T @ w_val
```
