"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov mirror descent with FFT-based soft subgradient
2. LP Frank-Wolfe with active set method and line search
3. Coordinate descent with bisection for final polish
Key: exploits the game-theoretic structure (minimax) for constraint selection.
"""

import numpy as np
import cvxpy as cp
import time

# Global solver
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_convolution(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad_topk(a, conv, top_k=80, beta=3.0):
    """Compute weighted subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wm, m in zip(w, top_idx):
        m_int = int(m)
        i_start = max(0, m_int - n + 1)
        i_end = min(n, m_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            g[idx] += 2.0 * wm * a[m_int - idx]
    return g

def compute_subgrad_softmax(a, conv, temperature):
    """Compute subgradient using softmax over all lags."""
    n = len(a)
    log_w = (conv - conv.max()) / temperature
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback to top-50
        return compute_subgrad_topk(a, conv, top_k=50, beta=3.0)
    w = w / w_sum
    
    g = np.zeros(n)
    for m in range(len(conv)):
        wm = w[m]
        if wm < 1e-12:
            continue
        m_int = m
        i_start = max(0, m_int - n + 1)
        i_end = min(n, m_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            g[idx] += 2.0 * wm * a[m_int - idx]
    return g

def nmd_optimize(a0, time_limit, verbose=True):
    """Nesterov mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    c_lr = 3.0  # learning rate constant
    stall = 0
    max_stall = 2000
    running_avg = x.copy()
    avg_count = 1
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Adaptive temperature: exponential decay
        frac = min(1.0, t / 8000.0)
        temp = 0.05 * (1 - frac) + 0.001 * frac
        temp = max(temp, 1e-4)
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        ac = fast_convolution(y)
        if temp > 0.01:
            g = compute_subgrad_softmax(y, ac, temp)
        else:
            g = compute_subgrad_topk(y, ac, top_k=100, beta=4.0)
        
        # Mirror descent step
        eta = c_lr / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Update best
        M = fast_convolution(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Ergodic average
        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            avg = avg / avg.sum()
            M_avg = fast_convolution(avg).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg.copy()
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c_lr = max(1.0, c_lr * 0.6)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, temp={temp:.5f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP Frank-Wolfe with active set and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        ac = fast_convolution(y)
        M_curr = ac.max()
        
        # Select active constraints: lags within 2% of max
        threshold = M_curr * 0.98
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > 200:
            active_lags = np.argsort(ac)[::-1][:200]
        
        if len(active_lags) < 5:
            active_lags = np.argsort(ac)[::-1][:50]
        
        # Build LP
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)  # default
        best_obj = fast_convolution((1 - best_alpha) * y + best_alpha * g_dir).max()
        
        try_alphas = [best_alpha, 0.5, 0.3, 0.2, 0.1, 0.7, 0.8]
        for alpha_try in try_alphas:
            y_try = (1 - alpha_try) * y + alpha_try * g_dir
            y_try = np.maximum(y_try, 0)
            sy = y_try.sum()
            if sy < 1e-15:
                continue
            y_try = y_try / sy
            obj_try = fast_convolution(y_try).max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_alpha = alpha_try
        
        y_new = (1 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()
        
        # Nesterov update
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np
