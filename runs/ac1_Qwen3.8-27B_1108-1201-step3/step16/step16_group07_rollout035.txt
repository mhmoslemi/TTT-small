def build_toeplitz(a, n_constraints=None):
    """Build the Toeplitz constraint matrix T where T[k,j] = a[k-j] if valid."""
    n = len(a)
    if n_constraints is None:
        n_constraints = 2 * n - 1
    T = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[k, lo:hi] = a[k - np.arange(lo, hi)]
    return T

        lo = max(0, k - n + 1)

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def spectral_pocs(a, n_iters=50, beta=0.5):
    """
    Alternating projections:
    1. In frequency domain: reduce peak of |F(a)|^2 (equivalent to reducing autocorr peak)
    2. In time domain: project onto non-negative cone
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * norm / s
    
    fft_size = 1 << (2 * n - 1).bit_length() - 1
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        # Compute current autocorr
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Frequency domain: try to flatten the spectrum
        # The autocorr peak is the max of ifft(|F(a)|^2)
        # To reduce it, we can reduce the high-magnitude frequency components
        F = rfft(a, fft_size)
        mag = np.abs(F)
        
        # Identify frequencies that contribute most to the peak
        # Simple approach: threshold the magnitude
        threshold = beta * mag.max()
        F_th = F * (mag > threshold)
        
        # Try the thresholded version
        a_trial = irfft(F_th, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < M:
            a = a_trial
    
    return a

def lp_nesterov_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """LP best-response with Nesterov acceleration and adaptive mixing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 40
    full_constraint_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Progressive active set
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            # Start with loose threshold, tighten over time
            tol = 0.15 * (1.0 - progress) + 0.01 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(10, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap for efficiency
            max_constraints = int(2.0 * n)
            if len(active_set) > max_constraints:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_constraints:]
                active_set = active_set[keep]
            
            full_constraint_count += 1
            if verbose and full_constraint_count % 10 == 0:
                print(f"    Active set size: {len(active_set)}")
        
        # Solve LP
        b = solve_lp(y, M_curr, active_set, solver)
        
        if b is None:
            # Fallback: use all constraints
            if t % 3 == 0:
                b = solve_lp(y, M_curr, np.arange(2 * n - 1), solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Try multiple mixing parameters and pick the best
        best_alpha = 2.0 / (t + 2.0)  # Nesterov default
        best_trial_score = float('inf')
        best_trial = None
        
        alphas = [best_alpha, min(1.0, best_alpha * 2), 0.3, 0.5, 0.1]
        for alpha in alphas:
            y_trial = (1.0 - alpha) * y + alpha * g
            y_trial = np.maximum(y_trial, 0.0)
            s_y = y_trial.sum()
            if s_y < 1e-15:
                continue
            y_trial = y_trial * norm / s_y
            score = fast_autocorr(y_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_trial = y_trial
                best_alpha = alpha
        
        if best_trial is not None:
            y_new = best_trial
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
        
        # Nesterov update for x
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with FFT-based evaluation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        
        # Process a random subset of coordinates
        subset_size = min(n, max(200, n // 2))
        indices = rng.choice(n, size=subset_size, replace=False)
        
        for j in indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            # Try several candidate values
            typical = norm / n
            candidates = [0.0, aj * 0.5, aj * 1.5, typical, typical * 0.5, typical * 2.0]
            candidates = sorted(set([max(0.0, min(c, norm / 4)) for c in candidates]))
            
            best_x = aj
            best_obj = best_score
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[j] = x
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_trial = a_trial * norm / s_trial
                score = fast_autocorr(a_trial).max()
                if score < best_obj:
                    best_obj = score
                    best_x = x
            
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_a, best_score

def subgradient_optimize(a0, time_limit, verbose=False, seed=42):
    """Smoothed subgradient method with softmax peak weighting."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.5 / n**0.3
    patience = 0
    max_patience = 500
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Softmax weighting over peaks
        beta = 20.0 + 10.0 * min(1.0, t / 1000.0)  # Increase concentration over time
        # Use a shifted version to avoid overflow
        shifted = conv - M_curr
        weights = np.exp(beta * shifted / max(M_curr * 0.1, 1e-10))
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = weights / w_sum
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        # Only use top contributing lags
        n_active = max(5, min(50, int(0.05 * (2*n-1))))
        active = np.argsort(weights)[-n_active:]
        
        for k in active:
            k = int(k)
            w = weights[k]
            if w < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * a[k - np.arange(lo, hi)]
            # Second term: a[i-k] for valid i
            lo2 = k
            hi2 = k + n
            if lo2 < n:
                hi2 = min(n, hi2)
                grad[lo2:hi2] += w * a[np.arange(lo2, hi2) - k]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        
        a_trial = a - step * grad * n**0.3
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 2.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-12:
                break
    
    return best_a, best_score

Let me think about what's really limiting the performance:

1. The LP best-response is a very powerful method, but it requires the LP to be solved accurately. With n=1600 and up to 3199 constraints, the LP should be solvable in 1-3 seconds with CLARABEL.

2. The Nesterov acceleration might not be optimal. The "best response" method (solving the LP and mixing) is a form of the multiplicative weight update / KTOS algorithm, which has a convergence rate of O(log(n)/T) for the dual gap. Nesterov acceleration can improve this to O(log(n)/T^2) in theory, but the constants matter.

3. One issue: the normalization step. After each LP solve, we normalize b to have sum = sqrt(2n). But the LP maximizes sum(b) subject to (a*b)[k] <= M. The optimal b has a specific sum, and normalizing changes the effective M. This might be suboptimal.

Let me think about a better formulation.

Actually, the key insight is: the score is scale-invariant. So we can work in any normalization. The LP formulation is:

Given a (with sum = S), solve:
max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0

where M = max_k (a*a)[k].

The optimal b tells us the "best direction" to move. After normalizing b to have the same sum as a, we get a new point on the simplex (well, on the ray from the origin through the simplex).

The issue is that the LP constraint uses M from the current a, not from the new point. This is fine for a single step, but over multiple steps, M might decrease (good) or increase (bad).

A better approach might be to use a **fixed M** for multiple iterations: solve the LP with a fixed M, iterate to convergence, then recompute M. This is like a block method.

Actually, let me think about this differently. The problem is:

minimize max_k (a*a)[k] / (sum a)^2  (times 2n)

subject to a >= 0

This is a fractional programming problem. By Dinkelbach's method, we can convert it to:

For a given lambda, solve: minimize max_k (a*a)[k] - lambda * (sum a)^2 subject to a >= 0, sum a = 1 (or some normalization).

But (sum a)^2 is constant under the normalization, so it reduces to: minimize max_k (a*a)[k] subject to a >= 0, sum a = S.

Which is exactly what we're doing. The LP best-response is a way to find a feasible direction that decreases the max.

Let me try yet another angle: **What if we use a different norm?**

Instead of minimizing the max of the autocorrelation, what if we minimize a smoothed version, like the L_p norm of the autocorrelation for large p? As p -> infinity, L_p -> max. But for finite p, the problem is smoother and might be easier to optimize.

minimize (sum_k ((a*a)[k] / S^2)^p)^(1/p)

This is a smooth convex function (for p >= 1). We can use standard gradient descent with Nesterov acceleration.

The gradient is:
∂/∂a[i] ||a*a||_p^p / S^(2p) = (p/S^(2p)) * (a*a)^{p-1} * ∂(a*a)/∂a[i]

For p large, this concentrates on the peaks.

def fast_autocorr(a):
    """Compute np.convolve(a, a) using FFT (linear convolution)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def solve_lp_full(a, M, solver, constraint_indices=None):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for selected k, b >= 0.
    Uses the Toeplitz structure of the constraint matrix.
    """
    n = len(a)
    if constraint_indices is None:
        constraint_indices = np.arange(2 * n - 1)
    
    size = len(constraint_indices)
    T = np.zeros((size, n))
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def softmax_subgradient(a0, time_limit, verbose=False, seed=42):
    """
    Subgradient method with softmax peak weighting and adaptive step.
    Uses Nesterov acceleration for faster convergence.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    step_base = 0.8 / n**0.25
    patience = 0
    max_patience = 400
    it = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        it += 1
        
        conv = fast_autocorr(y)
        M_curr = conv.max()
        
        # Softmax weighting (concentrate on peaks more as we progress)
        beta = 15.0 + 20.0 * min(1.0, t / 2000.0)
        shifted = conv - M_curr
        # Avoid overflow
        max_shift = shifted.max()
        if max_shift > 0:
            shifted = shifted - max_shift
        weights = np.exp(beta * shifted / max(M_curr * 0.15, 1e-10))
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = weights / w_sum
        
        # Compute weighted subgradient efficiently
        # Only use top-weighted lags
        n_active = max(5, min(80, int(0.05 * (2 * n - 1))))
        active_idx = np.argsort(weights)[-n_active:]
        
        grad = np.zeros(n)
        for k_ind in active_idx:
            k = int(k_ind)
            w = weights[k]
            if w < 1e-12:
                continue
            # d(a*a)[k] / da[i] = a[k-i] + a[i-k] (valid indices only)
            # First term: a[k-i] for i where 0 <= k-i < n => k-n+1 <= i <= k
            lo1 = max(0, k - n + 1)
            hi1 = min(n, k + 1)
            if lo1 < hi1:
                grad[lo1:hi1] += w * a[k - np.arange(lo1, hi1)]
            # Second term: a[i-k] for i where 0 <= i-k < n => k <= i <= k+n-1
            lo2 = max(0, k)
            hi2 = min(n, k + n)
            if lo2 < hi2:
                grad[lo2:hi2] += w * a[np.arange(lo2, hi2) - k]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = y - step_base * alpha * grad * n**0.25
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        a_new = y_new
        if t > 1:
            momentum = min((t - 1.0) / (t + 2.0), 0.85)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
        
        y_prev = y
        y = y_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
            step_base = min(step_base * 1.002, 3.0)
        else:
            patience += 1
            step_base *= 0.995
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def spectral_pocs(a0, n_iters=200):
    """
    Alternating projections between:
    1. Flat-spectrum constraint (reduce |F(a)|^2 peaks)
    2. Non-negativity constraint
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * norm / s
    
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for it in range(n_iters):
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Frequency domain step: reduce peak contributions
        F = rfft(a, fft_size)
        mag2 = np.abs(F)**2
        
        # Identify which frequencies contribute most to the time-domain peak
        # The time-domain peak is ifft(mag2). Reduce its max by shaping mag2.
        # Simple approach: cap the magnitude
        cap = np.sqrt(1.1 * M / n)
        F_new = F * (np.abs(F) <= cap)
        
        a_trial = irfft(F_new, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial * norm / s_trial
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
            a = a_trial
    
    return best_a

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """
    Randomized coordinate descent with FFT evaluation.
    Focuses on coordinates that contribute most to the peak.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)

    s0 = a.sum()

        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Compute sensitivity: which coordinates most affect the peak
    conv = fast_autocorr(a)
    peak_idx = np.argmax(conv)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        peak_idx = np.argmax(conv)
        M_curr = conv[peak_idx]
        
        # Select coordinates that contribute to the peak
        # For peak at index k, the contributing coordinates are those j where a[j]*a[k-j] is large
        lo = max(0, peak_idx - n + 1)
        hi = min(n, peak_idx + 1)
        peak_coords = set(range(lo, hi))
        
        # Add some random coordinates for exploration
        n_rand = max(50, n // 10)
        rand_coords = set(rng.choice(n, size=n_rand, replace=False).tolist())
        
        candidate_coords = list(peak_coords | rand_coords)
        rng.shuffle(candidate_coords)
        
        for j in candidate_coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            typical = norm / n
            
            # Candidate values
            candidates = [0.0, aj * 0.3, aj * 0.7, aj * 1.3, typical * 0.5, typical, typical * 2.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))
            
            best_x = aj
            best_obj = M_curr
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[j] = x
                s_trial = a_trial.sum()
                if s_trial < 1e-15:

        a_out, _ = softmax_subgradient(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)
    
    # === Phase 3: LP best-response round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Best-Response r2 [{t3:.0f}s] ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = lp_best_response_optimize(a_norm, t3, solver, verbose=True, seed=777,
                                              use_full_constraints=False, max_constraints=int(2.0 * n_cur))
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)
    
    # === Phase 4: Spectral POCS ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t4 = min(remaining * 0.06, 60.0)
        print(f"\n=== Phase 4: Spectral POCS [{t4:.0f}s] ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out = spectral_pocs(a_norm, n_iters=500)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"pocs_n{n_cur}", t_global)
    
    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.14, 180.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Coordinate Descent [{t5:.0f}s] ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True, seed=5)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)
    
    # === Phase 6: LP best-response round 3 (final push) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.18, 250.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP Best-Response r3 [{t6:.0f}s] ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = lp_best_response_optimize(a_norm, t6, solver, verbose=True, seed=2024,
                                              use_full_constraints=True, max_constraints=None)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"lp3_n{n_cur}", t_global)
    
    # === Phase 7: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 2
    t7_total = min(remaining * 0.12, 120.0)
    t7_each = t7_total / max(1, n_restarts)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t7 = min(t7_each, remaining - 15)
        if t7 < 25:
            break
        print(f"\n=== Phase 7r{r}: Perturb+Optimize [{t7:.0f}s] ===")
        scale = 0.015 + 0.01 * rng.random()
        noise = 1.0 + scale * rng.standard_normal(n_cur)
        noise = np.maximum(noise, 0.3)
        perturbed = best_seq * noise
        s_p = perturbed.sum()
        perturbed = perturbed / s_p
        a_norm = normalize_to(perturbed, n_cur)
        a_out, _ = lp_best_response_optimize(a_norm, t7, solver, verbose=False, seed=5000 + r * 137,
                                              use_full_constraints=False, max_constraints=int(2.0 * n_cur))
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)
    
    # === Phase 8: Final subgradient micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 25:
        t8 = min(remaining * 0.5, 50.0)
        print(f"\n=== Phase 8: Final Subgradient [{t8:.0f}s] ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = softmax_subgradient(a_norm, t8, verbose=False, seed=314)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f"final_n{n_cur}", t_global)
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
