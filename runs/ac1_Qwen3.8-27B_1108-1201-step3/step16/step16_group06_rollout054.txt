    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_sparse_conv_matrix(a, M):
    """
    Build sparse constraint matrix for: T @ b <= M
    where T[k, j] = a[j] if j <= k <= j+n-1 (i.e., k and j are within n-1 of each other)
    
    Actually, (convolve(a, b))[k] = sum_j a[k-j] * b[j] for valid k-j.
    So T[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    This means for row k, column j is nonzero when k-n+1 <= j <= k.
    """
    n = len(a)
    num_rows = 2 * n - 1
    # Each row k has nonzeros at columns j where max(0, k-n+1) <= j <= min(n-1, k)
    # T[k, j] = a[k-j]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(a[k - js].tolist())
    
    T = sp.csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    return T

import time

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

def build_sparse_T(y):
    """Build sparse matrix T where T[k,j] = y[k-j] if 0<=k-j<n else 0."""
    n = len(y)
    num_rows = 2 * n - 1
    # For row k, nonzero columns are j in [max(0, k-n+1), min(n-1, k)]
    # T[k, j] = y[k-j]
    
    # Use a more efficient construction
    n_nonzero_per_row = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        n_nonzero_per_row.append(j_end - j_start + 1)
    
    total_nnz = sum(n_nonzero_per_row)
    
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    
    pos = 0
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[pos:pos+count] = k
        cols[pos:pos+count] = np.arange(j_start, j_end+1)
        vals[pos:pos+count] = y[k - np.arange(j_start, j_end+1)]
        pos += count
    
    T = sp.csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    return T
