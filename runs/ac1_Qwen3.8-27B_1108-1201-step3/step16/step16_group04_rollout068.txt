def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] where c = conv(a,a).
    g[i] = sum_d a[d]*w[i+d] + sum_d a[d]*w[i-d] (w defined on [0,2n-2]).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    a_rev = a[::-1]
    conv_len = 2 * n - 1
    L = 1 << (3 * n - 1).bit_length()
    if L < 3 * n:
        L = 1 << (3 * n).bit_length()

    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a_rev, L)

    # T2[i] = (a * w)[i] for i = 0..2n-2
    conv_aw = np.fft.irfft(A * W, L)[:conv_len]

    # T1[i] = (w * a_rev)[i+n-1] for i = 0..2n-2
    conv_wa = np.fft.irfft(W * Ar, L)[n - 1:n - 1 + conv_len]

    return conv_aw + conv_wa


def power_sum_gd(a0, time_limit, p_start=2.0, p_end=64.0, verbose=True, rng_seed=42):
    """
    Gradient descent with power-sum approximation of max.
    Uses continuation (p increases over time) and backtracking step size.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = 2.0 / n
    eta_min = 1e-12
    eta_max = 10.0 / n
    stall = 0
    max_stall = 5000

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit * 0.85, 1.0))
        p = p_start * (p_end / p_start) ** frac

        c = fast_conv(a)
        max_c = c.max()
        if max_c < 1e-20:
            break

        # Weights: w[k] proportional to c[k]^(p-1)
        c_clipped = np.maximum(c, 1e-30)
        log_w = (p - 1.0) * np.log(c_clipped)
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(c) / len(c)
        else:
            w = w / w_sum

        g = compute_grad(a, w)

        # Normalize gradient for step size
        g_norm = np.sqrt(np.sum(g * g))
        if g_norm < 1e-20:
            break

        a_new = a - eta * g
        a_new = project_simplex(a_new)

        c_new = fast_conv(a_new)
        M_new = c_new.max()

        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, eta_max)
            stall = 0
        else:
            eta = max(eta * 0.7, eta_min)
            stall += 1

        iter_count += 1

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.01, n)
            a_new = best_a * (1.0 + pert)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
            else:
                a = best_a.copy()
            eta = 2.0 / n

        if verbose and iter_count % 2000 == 0:
            J = 2.0 * n * best_M
            print(f"    PSGD iter={iter_count} ({elapsed:.1f}s): p={p:.1f}, J={J:.6f}, eta={eta:.3e}")

    return best_a, best_M

def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [b_var >= 0]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw_refine(a0, time_limit, solver, top_k_start=30, top_k_end=100, verbose=True):
    """LP Frank-Wolfe with adaptive active set."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    top_k = top_k_start
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0

        # Nesterov extrapolation
        if iter_count == 0:
            y = x.copy()
        else:
            beta_t = (iter_count + 0.0) / (iter_count + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_M:
            best_M = M_curr
            best_a = y.copy()

        # Adaptive top_k
        top_k_current = int(top_k_start + (top_k_end - top_k_start) * min(1.0, elapsed / (time_limit * 0.7)))
        active_lags = np.argsort(ac)[::-1][:top_k_current]

        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            x_prev = y
            x = x.copy()
            iter_count += 1
            continue

        alpha = 2.0 / (iter_count + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        mx = fast_conv(x).max()
        if mx < best_M:
            best_M = mx
            best_a = x.copy()

        iter_count += 1

        if verbose and iter_count % 20 == 0:
            J = 2.0 * n * best_M
            print(f"    LP-FW iter={iter_count} ({elapsed:.1f}s): J={J:.6f}, top_k={top_k_current}")

    return best_a, best_M

    max_stall = 500

        c = fast_conv(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:min(15, len(c))]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find quiet destinations
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(100, len(candidates))]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.07]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and not improved:
                pass

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M


def make_initialization(n, kind, rng=None):
    """Create various asymmetric initializations."""
    if rng is None:
        rng = np.random.default_rng(0)

    if kind == 'exp_decay':
        rate = rng.uniform(0.02, 0.06)
        a = np.exp(-rate * np.arange(n))
    elif kind == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a ** rng.uniform(0.5, 2.0)
    elif kind == 'asym_gauss':
        center = n * rng.uniform(0.2, 0.4)
        sigma = n * rng.uniform(0.1, 0.25)
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif kind == 'bump':
        # Two asymmetric bumps
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.6, 0.85)
        s1 = n * rng.uniform(0.05, 0.15)
        s2 = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((np.arange(n) - c1) / s1) ** 2) + np.exp(-((np.arange(n) - c2) / s2) ** 2)
    elif kind == 'sin_asym':
        # Asymmetric sine
        phase = rng.uniform(0.0, 0.3)
        a = np.sin(np.pi * (np.arange(n) + n * phase) / n)
        a = np.maximum(a, 0.0)
    elif kind == 'random_smooth':
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)

    return normalize(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-start gradient descent + LP-FW + coordinate refinement.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        best_seq = make_initialization(n_base, 'exp_decay', np.random.default_rng(42))
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-start gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_starts = 5
    t_per_start = min(remaining * 0.08 / n_starts, 30.0)
    print(f"\n=== Phase 1: Multi-start GD ({n_starts} starts, {t_per_start:.0f}s each) ===")

    kinds = ['exp_decay', 'ramp', 'asym_gauss', 'bump', 'sin_asym']
    for i in range(min(n_starts, len(kinds))):
        if time.monotonic() - t_global > time_budget - 200:
            break
        a_init = make_initialization(n0, kinds[i], np.random.default_rng(i * 100 + 7))
        t_start = time.monotonic()
        a_out, _ = power_sum_gd(a_init, t_per_start, p_start=2.0, p_end=16.0, verbose=False, rng_seed=i)
        try_update(a_out, f"ms_{kinds[i]}")

    # Also try parent with gradient
    if time.monotonic() - t_global < time_budget - 200:
        t_start = time.monotonic()
        a_out, _ = power_sum_gd(normalize(best_seq), t_per_start, p_start=2.0, p_end=16.0, verbose=False, rng_seed=99)
        try_update(a_out, "ms_parent")

    # === Phase 2: Main power-sum gradient descent with continuation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: PSGD continuation n={len(best_seq)}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = power_sum_gd(a_in, t2, p_start=2.0, p_end=128.0, verbose=True, rng_seed=42)
        try_update(a_out, f"psgd_n{len(best_seq)}")

    # === Phase 3: LP Frank-Wolfe refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP-FW n={len(best_seq)}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and len(best_seq) <= 3000:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_refine(a_in, t3, solver, top_k_start=25, top_k_end=80, verbose=True)
        try_update(a_out, f"lpfw_n{len(best_seq)}")

    # === Phase 4: Log-sum-exp gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE-GD n={len(best_seq)}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_in = normalize(best_seq)
        a_out, _ = lse_gd(a_in, t4, beta_start=5.0, beta_end=150.0, verbose=True, rng_seed=77)
        try_update(a_out, f"lse_n{len(best_seq)}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord n={len(best_seq)}, budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_refine(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Another LP pass if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP-FW final n={len(best_seq)}, budget={remaining:.0f}s ===")
    t6 = min(remaining * 0.3, 200.0)
    if t6 > 30 and len(best_seq) <= 3000:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_refine(a_in, t6, solver, top_k_start=40, top_k_end=120, verbose=True)
        try_update(a_out, f"lpfw2_n{len(best_seq)}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final coord, budget={remaining:.0f}s ===")
    t7 = min(remaining * 0.5, 100.0)
    if t7 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_refine(a_in, t7, verbose=False, rng_seed=1234)
        try_update(a_out, f"coord_final_n{len(best_seq)}")

    # === Phase 8: One more LSE pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LSE, budget={remaining:.0f}s ===")
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 15:
            a_in = normalize(best_seq)
            a_out, _ = lse_gd(a_in, t8, beta_start=20.0, beta_end=300.0, verbose=False, rng_seed=555)
            try_update(a_out, f"lse_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
