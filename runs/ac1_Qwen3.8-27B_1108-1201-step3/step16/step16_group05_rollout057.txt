def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-15:

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, beta, top_k=200):

    top_vals = ac[top_idx]
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n), 0.0
    shifted = top_vals - max_val
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, ac.max()


def apg_backtracking(a0, time_limit, beta_start=30.0, beta_end=5.0,
                     top_k_start=100, top_k_end=400, max_iters=200000, verbose=True, seed=42):
    """APG with backtracking line search and continuation on beta."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    c0 = 1.8
    cur_beta = beta_start
    cur_top_k = top_k_start
    phase_duration = 4000

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / (4.0 * phase_duration))
        cur_beta = beta_start + (beta_end - beta_start) * frac
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g, M_curr = compute_smoothed_gradient(y, cur_beta, top_k=cur_top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            gn = 1e-18

        # Backtracking line search
        eta = c0 / np.sqrt(t + 1.0)
        f_curr = M_curr
        for ls_iter in range(20):
            x_new = project_simplex(y - eta * g)
            M_new = fast_autocorr(x_new).max()
            if M_new <= f_curr - 1e-4 * eta * gn * gn:
                break
            eta *= 0.5
        else:
            x_new = x.copy()
            M_new = M_curr

        x_prev = x
        x = x_new

        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise_scale = max(0.001, 0.01 * (1.0 - frac))
            noise = rng.normal(0, noise_scale, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m


def lp_frank_wolfe_sparse(a0, time_limit, solver, max_constraints=100, verbose=True):
    """LP Frank-Wolfe using only the top constraints for speed."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Only use top constraints
        n_cons = min(max_constraints, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_cons]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]

        M_tight = M_curr * 1.001  # Slight relaxation for numerical feasibility

        for lag in top_lags:
            k_int = int(lag)
            # c[k] = sum_i a[i] * a[k-i]
            # As a function of b: sum_i y[i] * b[k-i] (for fixed y)
            # This is a linear function of b: T[k,:] @ b
            # T[k, j] = y[j] if 0 <= k-j < n, else 0
            # So constraint: sum_j y[j] * b[k-j] <= M_tight
            # which is: sum_j y[j] * b[k-j] <= M_tight
            # Let's build the row: for each j, coefficient is y[j] if (k-j) is valid index for b
            # b is length n, index (k-j) must be in [0, n-1]
            row = np.zeros(n)
            j_start = max(0, k_int - (n - 1))
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                if 0 <= k_int - j < n:
                    row[j] = y[j]
            if np.any(row != 0):
                constraints.append(row @ b_var <= M_tight)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx

        y = y_new

        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-14:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent with peak-aware moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k_idx = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        peak_contrib = np.zeros(n)
        for k in top_k_idx[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            peak_contrib[idx] += contrib

        sources = np.argsort(peak_contrib)[::-1][:20]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            best_dst = -1
            best_delta = 0.0
            n_try = min(n, 300)
            candidates = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)

            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_k_idx[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_delta:
                    best_delta = score
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def spectral_shaping(a, n_harmonics=5, n_iters=50, verbose=False):
    """Try to flatten the power spectrum while maintaining non-negativity."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    for it in range(n_iters):
        ac = fast_autocorr(a)
        M = ac.max()
        # In frequency domain, try to reduce peaks
        A = np.fft.rfft(a, 2 * n)
        # Target: make |A|^2 more flat
        target = np.full(len(A), 1.0 / len(A))
        # Blend
        A_new = 0.7 * A + 0.3 * np.sqrt(target) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            if verbose:
                print(f"        Spectral iter {it}: J={2*n*best_m:.6f}")
        else:
            break

    return best_a


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    rng = np.random.default_rng(2025)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update_local(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Fast APG at parent length ===
    rem = remaining_time()
    print(f"\n=== Phase 1: APG n={n0} ===")
    t1 = min(rem * 0.25, 350.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = apg_backtracking(
            a_in, t1, beta_start=30.0, beta_end=5.0,
            top_k_start=100, top_k_end=500, verbose=True, seed=42)
        try_update_local(a_out, f"apg1_n{n0}")

    # === Phase 2: Spectral shaping ===
    rem = remaining_time()
    print(f"\n=== Phase 2: Spectral shaping n={len(best_seq)} ===")
    t2_spec = 30.0
    if rem > t2_spec + 10:
        a_in = norm_sum1(best_seq)
        a_out = spectral_shaping(a_in, n_harmonics=5, n_iters=100, verbose=True)
        try_update_local(a_out, f"spectral_n{len(best_seq)}")

    # === Phase 3: LP Frank-Wolfe (sparse) ===
    rem = remaining_time()
    print(f"\n=== Phase 3: LP-FW sparse n={len(best_seq)} ===")
    t3 = min(rem * 0.18, 250.0)
    if t3 > 30 and len(best_seq) <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_sparse(a_in, t3, solver, max_constraints=80, verbose=True)
        try_update_local(a_out, f"lpfw_n{len(best_seq)}")

    # === Phase 4: APG polish after LP ===
    rem = remaining_time()
    print(f"\n=== Phase 4: APG polish n={len(best_seq)} ===")
    t4 = min(rem * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = apg_backtracking(
            a_in, t4, beta_start=20.0, beta_end=4.0,
            top_k_start=150, top_k_end=600, verbose=False, seed=77)
        try_update_local(a_out, f"apg_polish_n{len(best_seq)}")

    # === Phase 5: Coordinate polish ===
    rem = remaining_time()
    print(f"\n=== Phase 5: Coord n={len(best_seq)} ===")
    t5 = min(rem * 0.08, 90.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=42)
        try_update_local(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Multi-restart APG with perturbations ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-restart n={n_best} ===")
    if rem > 120:
        n_restarts = 4
        t_per = min(rem * 0.6 / n_restarts, 80.0)
        for r in range(n_restarts):
            rem = remaining_time()
            if rem < 30:
                break
            t_r = min(t_per, rem - 20)
            if t_r < 15:
                break
            # Different perturbation for each restart
            a_base = norm_sum1(best_seq)
            if r == 0:
                # Small Gaussian noise
                noise = rng.normal(0, 0.005, n_best) * a_base
                a_init = np.maximum(a_base + noise, 1e-10)
                a_init = a_init / a_init.sum()
            elif r == 1:
                # Local swap perturbation
                a_init = a_base.copy()
                for _ in range(10):
                    i = int(rng.integers(n_best))
                    j = int(rng.integers(n_best))
                    if i != j and a_init[i] > 1e-6:
                        amt = a_init[i] * rng.uniform(0.01, 0.05)
                        a_init[i] -= amt
                        a_init[j] += amt
                a_init = np.maximum(a_init, 0)
                a_init = a_init / a_init.sum()
            elif r == 2:
                # Spectral perturbation
                noise = rng.normal(0, 0.01, n_best)
                a_init = a_base + noise * a_base
                a_init = np.maximum(a_init, 1e-10)
                a_init = a_init / a_init.sum()
            else:
                # Blend with different structured init
                x = np.linspace(0, 1, n_best)
                alt = np.exp(-20 * (x - 0.35) ** 2) + 0.5 * np.exp(-20 * (x - 0.65) ** 2)
                alt = alt / alt.sum()
                a_init = 0.7 * a_base + 0.3 * alt
                a_init = a_init / a_init.sum()

            a_out, _ = apg_backtracking(
                a_init, t_r, beta_start=25.0, beta_end=5.0,
                top_k_start=100, top_k_end=400, verbose=False, seed=200 + r)
            try_update_local(a_out, f"restart{r}_n{n_best}")

    # === Phase 7: Try longer length ===
    rem = remaining_time()
    n1 = min(len(best_seq) + 500, 3200)
    if n1 > len(best_seq) and rem > 100:
        print(f"\n=== Phase 7: Scale to n={n1} ===")
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        t7 = min(rem * 0.2, 150.0)
        if t7 > 25:
            a_out, _ = apg_backtracking(
                a_in, t7, beta_start=25.0, beta_end=5.0,
                top_k_start=100, top_k_end=400, verbose=True, seed=99)
            try_update_local(a_out, f"apg_n{n1}")
            rem = remaining_time()
            if rem > 30:
                t7b = min(rem * 0.3, 50.0)
                if t7b > 10:
                    a_out2, _ = coordinate_polish(a_out, t7b, verbose=False, seed=55)
                    try_update_local(a_out2, f"coord_n{n1}")

    # === Phase 8: Try shorter length (sometimes helps) ===
    rem = remaining_time()
    n2 = max(len(best_seq) - 400, 1600)
    if n2 < len(best_seq) and rem > 80:
        print(f"\n=== Phase 8: Scale to n={n2} ===")
        a_short = interpolate_sequence(best_seq, n2)
        a_in = norm_sum1(a_short)
        t8 = min(rem * 0.15, 100.0)
        if t8 > 20:
            a_out, _ = apg_backtracking(
                a_in, t8, beta_start=25.0, beta_end=5.0,
                top_k_start=100, top_k_end=300, verbose=False, seed=111)
            try_update_local(a_out, f"apg_n{n2}")

    # === Phase 9: Final LP at best length ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 2800 and rem > 40:
        print(f"\n=== Phase 9: Final LP n={n_best} ===")
        t9 = min(rem * 0.3, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_sparse(a_in, t9, solver, max_constraints=60, verbose=True)
            try_update_local(a_out, f"lp_final_n{n_best}")

    # === Phase 10: Final coordinate polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 10: Final coord n={n_best} ===")
    if rem > 20:
        t10 = min(rem * 0.5, 80.0)
        if t10 > 8:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t10, verbose=True, seed=314)
            try_update_local(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
