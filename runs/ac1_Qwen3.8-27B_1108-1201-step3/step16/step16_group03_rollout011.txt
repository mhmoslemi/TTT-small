def adam_log(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, seed=42):
    """Adam optimizer in log-domain for the non-negativity constraint."""
    n = len(a0)
    a = norm_sum1(a0)
    x = np.log(a + 1e-30)  # log-domain parameter
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = np.exp(x)
        a = a / a.sum()
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_m:
            best_m = M
            best_a = a.copy()
        
        # Subgradient
        k_star = int(np.argmax(ac))
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        for i in range(i_start, i_end):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
        # Gradient of log(a) = 1/a, so grad_log = grad / a
        grad_log = grad / (a + 1e-30)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * grad_log**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    
    return best_a, best_m

import time

def fast_conv(a):

def evaluate_safe(a):

def norm_sum1(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def bathtub_init(n, sharpness=1.0, middle_weight=0.1, seed=42):
    """Create a U-shaped (bathtub) initialization: large at edges, small in middle."""
    j = np.arange(n, dtype=np.float64)
    edge_left = 1.0 / (np.sqrt(j + 1.0) + 1e-10)
    edge_right = 1.0 / (np.sqrt(n - j) + 1e-10)
    a = edge_left + edge_right
    # Add small middle component to avoid zeros
    a = a + middle_weight * np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    rng = np.random.default_rng(seed)
    # Add small noise to break symmetry
    a = a * (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def random_structured_init(n, seed=42):
    """Random initialization with smooth structure."""
    rng = np.random.default_rng(seed)
    # Random smooth function
    n_freq = 3
    freqs = rng.uniform(0, 3, n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    amps = rng.uniform(0.1, 1.0, n_freq)

    a = np.ones(n)

        a = a * (1.0 + amp * np.sin(2 * np.pi * f * x + p))

def gs_spectral_shape(a0, n_iter=2000, relax=0.5, peak_factor=2.0, verbose=False, seed=42):
    """
    Gerchberg-Saxton style spectral shaping with iterative peak reduction.
    Alternates between:
    - Frequency domain: shape the power spectrum to reduce autocorrelation peaks
    - Time domain: enforce non-negativity and normalization
    """
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    for it in range(n_iter):
        # Compute current state
        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2
        ac = np.fft.ifft(power).real[:2 * n - 1]
        ac = np.maximum(ac, 0)
        M_curr = ac.max()

        # Create target: reduce peaks above peak_factor * median
        target = ac.copy()
        med = np.median(target)
        threshold = peak_factor * med
        if threshold < M_curr:
            peaks = target > threshold
            if peaks.any():
                excess = (target[peaks] - threshold).sum()
                target[peaks] = threshold
                non_peaks = ~peaks
                if non_peaks.sum() > 0:
                    target[non_peaks] += excess / non_peaks.sum()
                target = np.maximum(target, 0)
                target = target / target.sum()

        # Frequency domain update
        target_pad = np.zeros(N)
        target_pad[:2 * n - 1] = target
        desired_power = np.fft.fft(target_pad)

        # desired_power should be approximately real; take real part and clip
        desired_mag = np.sqrt(np.maximum(desired_power.real, 0.0))

        curr_mag = np.maximum(np.abs(A), 1e-30)
        # Blended update
        new_mag = (1.0 - relax) * curr_mag + relax * desired_mag
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_new = np.fft.ifft(A_new).real[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            # Restart
            a = norm_sum1(a0)
            continue
        a_new = a_new / s
        a = a_new

        # Track best
        m_new = fast_conv(a).max()
        if m_new < best_m - 1e-12:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive relaxation: increase when stalled
        if stall > max_stall:
            relax = min(relax + 0.05, 1.0)
            peak_factor = max(peak_factor - 0.05, 1.2)
            stall = 0

        if verbose and it % 200 == 0:
            print(f"    GS iter={it}: max_ac={m_new:.8f}, score={2*n*m_new:.6f}, relax={relax:.2f}, peak_f={peak_factor:.2f}")

    return best_a, best_m

            a_restart = best_a * (1.0 + 0.01 * rng.standard_normal(n))
            a_restart = np.maximum(a_restart, 1e-10)
            a_restart = a_restart / a_restart.sum()
            x = np.log(a_restart + 1e-30)
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            stall = 0
            continue

        # Soft subgradient: weighted average of top-k peak gradients
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
            grad *= 1.0  # just accumulate

        # Actually redo properly
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * a[j]

        # Gradient in log-domain: d(log a_i)/da_i = 1/a_i
        grad_log = grad / (a + 1e-30)

        # Adam update
        b1, b2 = 0.9, 0.999
        m_vec = b1 * m_vec + (1 - b1) * grad_log
        v_vec = b2 * v_vec + (1 - b2) * grad_log ** 2
        m_hat = m_vec / (1 - b1 ** t)
        v_hat = v_vec / (1 - b2 ** t)
        lr_t = lr * np.sqrt(1 - b2 ** t) / (1 - b1 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m

def nesterov_amd(a0, time_limit, top_k_init=1, c0=5.0, seed=42, verbose=False):
    """Nesterov accelerated mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    current_top_k = top_k_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        ac = fast_conv(y)
        M_curr = ac.max()

        # Adaptive top_k
        if stall > max_stall // 2:
            current_top_k = min(current_top_k + 1, 6)
        elif stall > max_stall:
            current_top_k = top_k_init

        # Soft subgradient
        k_max = min(current_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 3.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * y[j]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            x_re = best_a + rng.normal(0, 0.005, n) * best_a
            x_re = np.maximum(x_re, 1e-10)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            stall = 0

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD t={t} ({elapsed:.0f}s): score={2*n*best_m:.6f}, top_k={current_top_k}")

    return best_a, best_m

def coordinate_polish(a0, time_limit, seed=123, verbose=False):
    """Coordinate descent with targeted mass transfer from peak contributors."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(5, len(ac))]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            if a[src] < 1e-15:
                continue

            # Find low-impact destination
            # Compute sensitivity: how much would moving mass to dst affect top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:3]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 100)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            # Score candidates by how little they contribute to any top peak
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                score = 0.0
                for k2 in top_peaks[:3]:
                    k2_int = int(k2)
                    j = k2_int - dst
                    if 0 <= j < n:
                        score += a[j]
                    j2 = dst - k2_int + 2 * n  # for other direction (not applicable for linear)
                if score < best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue
            dst = best_dst

            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def try_gs_and_refine(a0, time_budget, label, verbose=True):
    """Run GS then refine with Adam and coordinate descent."""
    t0 = time.monotonic()
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_conv(best_a).max()

    # Phase A: GS spectral shaping
    t_gs = min(time_budget * 0.4, 300.0)
    if t_gs > 10:
        a_gs, m_gs = gs_spectral_shape(best_a, n_iter=3000, relax=0.6, peak_factor=2.5, verbose=verbose, seed=42)
        best_a = a_gs
        best_m = m_gs
        if verbose:
            print(f"  [{label}] After GS: score={2*n*best_m:.6f}, t={time.monotonic()-t0:.0f}s")

    # Phase B: Multi-target GS
    remaining = time_budget - (time.monotonic() - t0)
    t_gs2 = min(remaining * 0.2, 150.0)
    if t_gs2 > 10:
        a_gs2, m_gs2 = gs_multi_target(best_a, n_iter=1500, verbose=verbose, seed=7)
        if m_gs2 < best_m:
            best_a = a_gs2
            best_m = m_gs2
            if verbose:
                print(f"  [{label}] After GS2: score={2*n*best_m:.6f}, t={time.monotonic()-t0:.0f}s")

    # Phase C: Adam refinement
    remaining = time_budget - (time.monotonic() - t0)
    t_adam = min(remaining * 0.25, 300.0)
    if t_adam > 20:
        a_adam, m_adam = adam_log(best_a, t_adam, lr=0.003, top_k=2, beta=2.0, verbose=verbose, seed=99)
        if m_adam < best_m:
            best_a = a_adam
            best_m = m_adam
            if verbose:
                print(f"  [{label}] After Adam: score={2*n*best_m:.6f}, t={time.monotonic()-t0:.0f}s")

    # Phase D: Coordinate polish
    remaining = time_budget - (time.monotonic() - t0)
    t_coord = min(remaining * 0.3, 200.0)
    if t_coord > 10:
        a_coord, m_coord = coordinate_polish(best_a, t_coord, verbose=verbose, seed=314)
        if m_coord < best_m:
            best_a = a_coord
            best_m = m_coord
            if verbose:
                print(f"  [{label}] After Coord: score={2*n*best_m:.6f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-start strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        best_seq = bathtub_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Strategy 1: Refine parent with new methods ===
    remaining = time_budget - (time.monotonic() - t_global)
    if parent is not None and remaining > 200:
        print(f"\n=== Strategy 1: Refine parent (n={n0}) ===")
        t1 = min(remaining * 0.25, 400.0)
        a_out, _ = try_gs_and_refine(parent, t1, "parent_gadam", verbose=True)
        try_update(a_out, f"parent_refined_n{n0}")

    # === Strategy 2: Bathtub initialization (new basin) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200:
        print(f"\n=== Strategy 2: Bathtub init (n={n0}) ===")
        t2 = min(remaining * 0.25, 400.0)
        a_bathtub = bathtub_init(n0, sharpness=1.0, middle_weight=0.05, seed=42)
        print(f"  Bathtub init score: {evaluate_safe(a_bathtub):.6f}")
        a_out, _ = try_gs_and_refine(a_bathtub, t2, "bathtub", verbose=True)
        try_update(a_out, f"bathtub_n{n0}")

    # === Strategy 3: Bathtub with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        print(f"\n=== Strategy 3: Bathtub v2 (n={n0}) ===")
        t3 = min(remaining * 0.15, 250.0)
        a_bat2 = bathtub_init(n0, sharpness=0.8, middle_weight=0.15, seed=77)
        a_out, _ = try_gs_and_refine(a_bat2, t3, "bathtub2", verbose=True)
        try_update(a_out, f"bathtub2_n{n0}")

    # === Strategy 4: Try different length (larger) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200:
        n_try = n0 + 400
        print(f"\n=== Strategy 4: Larger n={n_try} ===")
        t4 = min(remaining * 0.20, 350.0)
        a_long = interpolate_sequence(best_seq, n_try)
        # Blend with bathtub
        a_bat_long = bathtub_init(n_try, seed=55)
        a_blend = 0.7 * a_long + 0.3 * a_bat_long
        a_blend = a_blend / a_blend.sum()
        a_out, _ = try_gs_and_refine(a_blend, t4, f"larger_n{n_try}", verbose=True)
        try_update(a_out, f"larger_n{n_try}")

    # === Strategy 5: Try different length (smaller) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        n_try = max(n0 - 400, 1000)
        print(f"\n=== Strategy 5: Smaller n={n_try} ===")
        t5 = min(remaining * 0.15, 250.0)
        a_short = interpolate_sequence(best_seq, n_try)
        a_bat_short = bathtub_init(n_try, seed=88)
        a_blend = 0.6 * a_short + 0.4 * a_bat_short
        a_blend = a_blend / a_blend.sum()
        a_out, _ = try_gs_and_refine(a_blend, t5, f"smaller_n{n_try}", verbose=True)
        try_update(a_out, f"smaller_n{n_try}")

    # === Strategy 6: Random structured init ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        print(f"\n=== Strategy 6: Random structured (n={len(best_seq)}) ===")
        t6 = min(remaining * 0.15, 200.0)
        n_cur = len(best_seq)
        a_rand = random_structured_init(n_cur, seed=123)
        a_out, _ = try_gs_and_refine(a_rand, t6, "random_struct", verbose=True)
        try_update(a_out, f"random_struct_n{n_cur}")

    # === Strategy 7: Final intensive coordinate polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Strategy 7: Final coord polish (n={len(best_seq)}) ===")
        t7 = min(remaining * 0.4, 300.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t7, verbose=True, seed=999)
        try_update(a_out, f"coord_final_n{len(best_seq)}")

    # === Strategy 8: Final AMD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Strategy 8: Final AMD (n={len(best_seq)}) ===")
        t8 = min(remaining * 0.4, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_amd(a_in, t8, top_k_init=1, c0=6.0, seed=555, verbose=True)
        try_update(a_out, f"amd_final_n{len(best_seq)}")

    # === Strategy 9: One more GS pass on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Strategy 9: Final GS (n={len(best_seq)}) ===")
        t9 = min(remaining * 0.5, 150.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = gs_spectral_shape(a_in, n_iter=3000, relax=0.7, peak_factor=2.0, verbose=True, seed=777)
        try_update(a_out, f"gs_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
