def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=30, eps=1e-10):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def spectral_peak_shaving(a0, time_limit, n_top=20, eta0=0.3, decay=0.999, verbose=True):
    """
    Frequency-domain peak shaving.
    
    Iteratively adjusts the power spectrum to reduce autocorrelation peaks.
    For each peak at lag k, reduce spectral power at frequencies where
    cos(2*pi*f*k/N) > 0 (constructive interference).
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # FFT size for linear autocorrelation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    eta = eta0
    stall = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            break
        
        # Find top peaks
        top_idx = np.argsort(ac)[::-1][:n_top]
        top_vals = np.maximum(ac[top_idx], 1e-20)
        
        # Weight peaks by their value (softmax-like)
        w = top_vals / top_vals.sum()
        
        # Compute the "spectral gradient" for each peak
        # For peak at lag k, the gradient of ac[k] w.r.t. |A[f]|^2 is cos(2*pi*f*k/fft_size)
        f_idx = np.arange(fft_size // 2 + 1)
        
        joint_grad = np.zeros(len(f_idx))
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            joint_grad += wi * np.cos(2.0 * np.pi * f_idx * k_int / fft_size)
        
        # Update: reduce power where joint_grad > 0
        # Use multiplicative update on power spectrum
        A = np.fft.rfft(a, fft_size)
        P = A.real ** 2 + A.imag ** 2  # power spectrum
        
        # Multiplicative update: P_new = P * (1 - eta * joint_grad)
        # Clip to avoid negative
        factor = 1.0 - eta * np.maximum(joint_grad, 0)
        factor = np.maximum(factor, 0.01)  # don't reduce too much
        P_new = P * factor
        
        # Reconstruct A with updated magnitude, keep phase
        P_new = np.maximum(P_new, 1e-30)
        A_new = A / np.sqrt(P + 1e-30) * np.sqrt(P_new)
        
        # Inverse FFT and extract
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend for stability
        a = 0.7 * a + 0.3 * a_new
        
        # Decay eta
        eta *= decay
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpectralPS it={it} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")
    
    return best_a


def multiplicative_peak_shaving(a0, time_limit, n_top=30, eta0=0.1, decay=0.9995, verbose=True):
    """
    Time-domain multiplicative peak shaving.
    
    For each peak at lag k, the gradient w.r.t. a[i] is a[k-i] + a[i-k].
    Use multiplicative update: a[i] <- a[i] * (1 - eta * grad[i])
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = eta0
    stall = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            break
        
        # Find top peaks
        top_idx = np.argsort(ac)[::-1][:n_top]
        top_vals = np.maximum(ac[top_idx], 1e-20)
        
        # Weight peaks
        w = top_vals ** 2
        w = w / w.sum()
        
        # Compute joint gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            # d(ac[k])/da[i] = a[k-i] + a[i-k]
            for i in range(n):
                g = 0.0
                if 0 <= k_int - i < n:
                    g += a[k_int - i]
                if 0 <= i - k_int < n:
                    g += a[i - k_int]
                grad[i] += wi * g
        
        # Multiplicative update
        factor = 1.0 - eta * grad / max(grad.max(), 1e-15)
        factor = np.maximum(factor, 0.01)
        a_new = a * factor
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend
        a = 0.8 * a + 0.2 * a_new
        
        eta *= decay
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultPS it={it} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a

        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def accelerated_lp_fw(a0, time_limit, solver, top_k_init=50, top_k_max=200, verbose=True):
    """Accelerated LP Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]
        
        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            continue
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if t % 50 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, best_m


def soft_subgradient_step(a, ac, top_k=50, beta=5.0):
    """Compute soft subgradient using top-K peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # Only add a[j_idx] where j_idx is valid
        valid = (j_idx >= 0) & (j_idx < n)
        grad[idx[valid]] += 2.0 * wi * a[j_idx[valid]]
    return grad


def nesterov_soft_gradient(a0, time_limit, top_k=50, beta=5.0, c0=3.0, verbose=True):
    """Nesterov-accelerated soft subgradient with mirror descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = soft_subgradient_step(y, ac, top_k=top_k, beta=beta)
        
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = y
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 3000:
            stall_count = 0
            pert = rng.normal(0, 0.005, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSG t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with multiple move sizes."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find destination that minimizes impact on peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(100, len(candidates))]]
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and not improved:
                pass
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
