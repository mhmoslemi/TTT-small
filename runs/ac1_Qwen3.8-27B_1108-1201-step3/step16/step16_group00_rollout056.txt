def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to nonnegative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_full_gradient(a, beta):
    """
    Compute the full softmax gradient of (1/beta)*log(sum exp(beta*c_k))
    using FFT for efficiency.
    
    Returns (gradient, max_c) where gradient = (1/beta) * cross_corr(softmax_weights, a)
    """
    n = len(a)
    c = fast_autocorr(a)
    # Numerically stable softmax
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    
    # Compute cross-correlation: g[i] = sum_j w[i+j] * a[j] for j=0..n-1
    # Using: g = (flip(a) * w_padded)[n-1 : 2n-1]
    b = a[::-1]  # flip a
    # Zero-pad both to sufficient length
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    S = 1 << (conv_len - 1).bit_length()
    B = np.fft.rfft(b, S)
    W = np.fft.rfft(w, S)
    G = np.fft.irfft(B * W, S)
    g = G[n - 1:2 * n - 1]  # length n
    
    # Scale by 1/beta (the log-sum-exp derivative has 1/beta factor, but we
    # actually want the gradient of the log-sum-exp itself for the subgradient)
    # Actually: d/d a_i [(1/beta) log sum exp(beta c_k)] = sum_k w_k * dc_k/da_i
    # where w_k is the softmax. So the gradient is cross_corr(w, a), no 1/beta factor
    # for the subgradient direction. The 1/beta only scales the objective value.
    # For the descent direction, we just need the direction, so we return cross_corr(w, a)
    
    return g, c_max


def nesterov_mirror_descent(a0, time_limit, beta_schedule=None, c0=5.0,
                             restart_stall=3000, verbose=True, seed=42):
    """
    Nesterov-accelerated mirror descent on the simplex using full softmax gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    if beta_schedule is None:
        beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    
    for beta in beta_schedule:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        time_for_beta = min(60.0, (time_limit - elapsed) * 0.25)
        if time_for_beta < 5:
            time_for_beta = (time_limit - elapsed) * 0.5
        
        # Reset Nesterov state for new beta
        x = a.copy()
        x_prev = a.copy()
        stall = 0
        
        for t in range(1, 2000000):
            if time.monotonic() - (t0) > time_limit:
                break
            if time.monotonic() - (t0 + elapsed) > time_for_beta + 30:
                break
            
            # Nesterov lookahead
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
            
            # Compute gradient at y
            g, M_curr = compute_full_gradient(y, beta)
            
            # Step size
            eta = c0 / (np.sqrt(t) + 1.0)
            
            # Mirror descent update
            log_y = np.log(y) - eta * g
            log_y -= log_y.max()
            a_new = np.exp(log_y)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            
            x_prev = x
            x = a_new
            
            # Evaluate
            M_new = fast_autocorr(x).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > restart_stall:
                # Perturb and restart
                pert = rng.normal(0, 0.03, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r.copy()
                stall = 0
                if verbose:
                    elapsed2 = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"      MD restart beta={beta} t={t} ({elapsed2:.0f}s): J={J:.6f}")
        
        # Update current best for next beta
        a = best_a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD beta={beta} done ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, verbose=True, top_K=200):
    """LP-based Frank-Wolfe with constraint selection."""
    import cvxpy as cp
    solver = _get_solver()
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Build constraint matrix for top-K most binding constraints
        # conv(y, b)[k] = sum_j y[k-j] * b[j] (valid indices)
        # We want: for each selected k, sum_j T[k,j] * b[j] <= M_curr
        # T[k,j] = y[k-j] if 0 <= k-j < n, else 0
        
        # Compute margins to find binding constraints
        # conv(y,b)[k] = sum_j y[max(0,k-n+1):min(n,k+1)] * b[j] ... complex
        # Simpler: for each k, the "potential" conv value at b=y is ac[k]
        # Most binding = largest ac[k] relative to M_curr (which IS M_curr)
        # So just pick top-K by ac value
        K_sel = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix (sparse would be better but let's use dense for simplicity)
        # For n up to ~3500 and K up to 200, matrix is 200 x n = 700K entries, fine

        for ii, k in enumerate(top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                T_sel[ii, j] = y[k_int - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_mom = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_mom = y_new + beta_t * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """
    Coordinate descent: for each coordinate, find optimal value by grid search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 50:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try different values for a[i]
            # Current value
            old_val = a[i]
            a_rest_sum = 1.0 - old_val
            
            # Try: keep same, increase, decrease
            # The convolution values that depend on a[i]:
            # c[k] includes a[i]*a[k-i] for valid k-i
            # So c[k] = base_k + a[i]*a[k-i] where base_k doesn't include a[i]
            
            best_local_m = best_m
            best_local_val = old_val
            
            # Grid search around current value
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]:
                new_val = old_val * frac if frac != 0.0 else 0.0
                # Renormalize: a'[i] = new_val, others scaled
                if new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / a_rest_sum if a_rest_sum > 1e-15 else 0
                a_trial = a.copy()
                a_trial[i] = new_val
                if a_rest_sum > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    a_trial[mask] = a[mask] * scale
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_local_m - 1e-12:
                    best_local_m = M_trial
                    best_local_val = new_val
                    # Apply
                    a[i] = new_val
                    if a_rest_sum > 1e-15:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        a[mask] = a_rest_vals * scale if False else a[mask] * scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    improved = True
            
            if M_trial < best_m:
                best_m = M_trial
                best_a = a.copy()
                stall = 0
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = a[i] * 0.05
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a /= a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init_ramp(n, power=1.0, direction='up'):
    """Monotone ramp initialization."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    if direction == 'down':
        a = a[::-1]
    return norm_sum1(a)


def make_init_bimodal(n, sigma=0.15):
    """Two-bump initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / sigma) ** 2) + np.exp(-((x - 0.8) / sigma) ** 2) + 0.01
    return norm_sum1(a)


def make_init_sinusoidal(n, freq=2.0):
    """Sinusoidal perturbation of uniform."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x)
    return norm_sum1(a)


def make_init_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for freq in [1, 2, 3, 4, 5, 6]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        best_seq = make_init_sinusoidal(n_base, freq=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")
    
    # === Phase 1: Nesterov Mirror Descent with full softmax gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: NMD-Softmax (n={n0}) ===")
    t1 = min(remaining * 0.40, 600.0)
    
    if t1 > 60:
        # Multiple initializations
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
            # Slightly perturbed parent
            rng_p = np.random.default_rng(1)
            pert = parent + rng_p.normal(0, 0.02, len(parent)) * parent
            inits.append(norm_sum1(pert))
            labels.append("parent_pert")
        
        inits.append(make_init_sinusoidal(n0, freq=2.0))
        labels.append("sin2")
        inits.append(make_init_sinusoidal(n0, freq=3.0))
        labels.append("sin3")
        inits.append(make_init_smooth_random(n0, seed=7))
        labels.append("rnd_sm")
        
        # Decrypt duplicates
        unique_inits = []
        unique_labels = []
        seen = set()
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0//50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        # Allocate time
        t_per = max(20.0, t1 / len(unique_inits))
        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  NMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = nesterov_mirror_descent(
                init, t_avail,
                beta_schedule=[20, 50, 100, 200, 500, 1000, 2000, 5000],
                c0=5.0, restart_stall=2000, verbose=False, seed=42+idx
            )
            try_update(a_out, f"nmd_{lbl}")
    
    # === Phase 2: LP Frank-Wolfe on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30 and n_best <= 4000:
        t2 = min(t2, remaining - 120)  # leave time for other phases
        if t2 > 20:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, verbose=True, top_K=150)
            try_update(a_out, f"lp_n{n_best}")
    
    # === Phase 3: Second round NMD with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: NMD high-beta polish (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = nesterov_mirror_descent(
            norm_sum1(best_seq), t3,
            beta_schedule=[1000, 3000, 5000, 10000, 20000],
            c0=3.0, restart_stall=3000, verbose=True, seed=99
        )
        try_update(a_out, f"nmd_polish_n{n_best}")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")
    longer_lengths = []
    for delta in [400, 800, 1200, 2000]:
        ln = n_best + delta
        if ln <= n_best + 2000 and ln > n_best:
            longer_lengths.append(ln)
    
    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t5a = min(remaining * 0.35, 150.0)
        if t5a > 15:
            t5a = min(t5a, remaining - 60)
            a_out, _ = nesterov_mirror_descent(
                a_long, t5a,
                beta_schedule=[50, 200, 500, 2000, 5000],
                c0=4.0, restart_stall=3000, verbose=False, seed=7
            )
            try_update(a_out, f"nmd_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3500:
            t5b = min(remaining * 0.3, 80.0)
            if t5b > 15:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5b, verbose=False, top_K=100)
                try_update(a_out, f"lp_n{new_n}")
    
    # === Phase 6: Try shorter sequences (sometimes helps) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60 and n_best > 2500:
        short_n = max(2400, n_best - 400)
        print(f"\n  Trying shorter n={short_n}...")
        a_short = interpolate_sequence(best_seq, short_n)
        t6 = min(remaining * 0.2, 80.0)
        if t6 > 10:
            a_out, _ = nesterov_mirror_descent(
                a_short, t6,
                beta_schedule=[100, 500, 2000, 5000],
                c0=4.0, restart_stall=2000, verbose=False, seed=13
            )
            try_update(a_out, f"nmd_short_n{short_n}")
    
    # === Phase 7: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 3500:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 15:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7, verbose=True, top_K=300)
            try_update(a_out, f"lp_final_n{n_best}")
    
    # === Phase 8: Final NMD squeeze ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final NMD (n={n_best}) ===")
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 10:
            a_out, _ = nesterov_mirror_descent(
                norm_sum1(best_seq), t8,
                beta_schedule=[5000, 10000, 20000, 50000],
                c0=2.0, restart_stall=2000, verbose=True, seed=200
            )
            try_update(a_out, f"nmd_final_n{n_best}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3200
        best_seq = make_init_sinusoidal(n_fb, freq=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
