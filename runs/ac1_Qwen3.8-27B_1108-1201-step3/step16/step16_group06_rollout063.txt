def subgradient_descent(a0, time_limit, verbose=True, rng_seed=0):
    """Nesterov-accelerated subgradient descent with proper subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        ac = fast_autocorr(y)
        k_star = int(np.argmax(ac))
        
        # Subgradient: s[j] = 2*y[k_star - j] if 0 <= k_star - j < n, else 0
        s = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                s[j] = 2.0 * y[idx]
        
        # For proper FW, we want the linear minimizer over the simplex
        # min_b <s, b> = s[argmin s]
        # So the FW direction is e_{argmin(s)}
        j_min = int(np.argmin(s))
        
        # But pure FW (going to vertex) is too aggressive. Use a mix.
        alpha = 2.0 / (t + 2.0)
        direction = np.zeros(n)
        direction[j_min] = 1.0
        
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    y = a.copy()
    x = a.copy()
    active_vertices = [a.copy()]
    active_weights = [1.0]

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < 1e-14:
            break

        C = build_conv_matrix(y)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [C @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate', 'inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue

        s_dir = b_val / b_sum

        # Away-step FW
        alpha = 2.0 / (t + 2.0)

        # Try away step from least useful active vertex
        if len(active_vertices) > 1:
            # Find vertex with worst contribution: highest <s_dir, V_i>
            worst_idx = -1
            worst_val = -np.inf
            for i, v in enumerate(active_vertices):
                dot = np.dot(s_dir, v)
                if dot > worst_val:
                    worst_val = dot
                    worst_idx = i

            if worst_val > 0.3:  # Only step away if it's significantly disagreeing
                gamma = alpha * 0.5
                y_away = y - gamma * active_vertices[worst_idx]
                y_away = y_away + gamma * s_dir
                y_away = np.maximum(y_away, 0.0)
                sa = y_away.sum()
                if sa > 1e-15:
                    y_away = y_away / sa
                    # Remove worst vertex
                    active_vertices.pop(worst_idx)
                y_new = y_away
            else:
                y_new = (1.0 - alpha) * y + alpha * s_dir
                y_new = np.maximum(y_new, 0.0)
                sy = y_new.sum()
                if sy > 1e-15:
                    y_new = y_new / sy
            x_new = y_new
        else:
            y_new = (1.0 - alpha) * y + alpha * s_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new / sy
            x_new = y_new

        # Add new vertex if new
        if np.max(np.abs(y_new - np.array([np.max((1-len(active_vertices))*v + v*0 for v in active_vertices)] if False else np.zeros(n)))) > 1e-8 or len(active_vertices) == 0:
            # Simplified: just track a few recent directions
            if len(active_vertices) < 5:
                active_vertices.append(s_dir.copy())
            else:
                active_vertices[0] = s_dir.copy()

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW-away iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def peak_targeted_cd(a0, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent that specifically targets peak autocorrelation values."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top peaks (exclude lag 0 and last since they're single products)
        peak_indices = np.argsort(ac)[::-1][:8]

        # Compute sensitivity: total contribution of each position to top peaks
        sensitivity = np.zeros(n)
        for pk in peak_indices:
            pk_int = int(pk)
            js = np.arange(max(0, pk_int - n + 1), min(n, pk_int + 1))
            sensitivity[js] += a[js] * a[pk_int - js]

        improved = False
        for pk in peak_indices:
            pk_int = int(pk)
            js = np.arange(max(0, pk_int - n + 1), min(n, pk_int + 1))
            contrib = a[js] * a[pk_int - js]

            if contrib.max() < 1e-20:
                continue

            # Sort by contribution, try to reduce highest
            order = np.argsort(contrib)[::-1]

            for rank in order[:8]:
                src = int(js[rank])
                src_other = pk_int - src

                # Find destination: low sensitivity, avoid peak regions
                candidates = np.argsort(sensitivity)
                for dst in candidates[:max(30, n // 15)]:
                    dst = int(dst)
                    if dst == src:
                        continue

                    # Check feasibility: would moving to dst increase any major peak?
                    feasible = True
                    for p2 in peak_indices[:4]:
                        p2_int = int(p2)
                        other2 = p2_int - dst
                        if 0 <= other2 < n and ac[p2_int] > 0.85 * M_curr:
                            # Moving mass to dst would increase peak p2
                            if a[other2] > 1e-10:
                                feasible = False
                                break
                    if not feasible:
                        continue

                    for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                        amt = max(a[src], a[src_other]) * frac
                        if amt < 1e-18:
                            continue
                        a_trial = a.copy()
                        # Mass from the larger contributor
                        if a[src] >= a[src_other]:
                            a_trial[src] -= amt
                        else:
                            a_trial[src_other] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Perturbation to escape local min
            if stall % 10 == 0:
                n_pert = max(10, n // 50)
                idx = rng.choice(n, size=n_pert, replace=False)
                a_pert = a.copy()
                a_pert[idx] *= (1.0 + rng.normal(0, 0.3, n_pert))
                a_pert = np.maximum(a_pert, 0.0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a_pert = a_pert / sp
                    M_new = fast_autocorr(a_pert).max()
                    if M_new < best_m * 1.1:  # Accept if not too bad
                        a = a_pert
                    else:
                        a = best_a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def subgradient_fw(a0, time_limit, verbose=True, rng_seed=0, top_k_peaks=5, beta=3.0):
    """Nesterov-accelerated method with soft subgradient (weighted peaks)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft subgradient from top peaks
        ac = fast_autocorr(y)
        k_max = min(top_k_peaks, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Gradient of the soft max approximation
        grad = np.zeros(n)
        for wi, pk in zip(w, top_idx):
            pk_int = int(pk)
            js = np.arange(max(0, pk_int - n + 1), min(n, pk_int + 1))
            grad[js] += 2.0 * wi * y[pk_int - js]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 2000:
            stall_count = 0
            perturbation = rng.normal(0, 0.03, n)
            x_restarted = best_a * (1.0 + perturbation)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftSGD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length (preserves sum=1)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    1. Load parent, evaluate
    2. LP-FW (corrected matrix) as primary optimizer
    3. Soft subgradient descent for diversification
    4. Peak-targeted coordinate descent for polishing
    5. Multi-n exploration
    6. Final refinement
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load failed: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        # Raised cosine initialization
        x = np.linspace(0, 1, n_base)
        best_seq = 0.5 * (1 + np.cos(np.pi * x)) ** 2
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Get solver
    solver_name = get_solver()

    # === Phase 1: Soft subgradient warmup (short) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 1: SoftSGD warmup n={n0} ===")
        t1 = min(remaining * 0.05, 60.0)
        a_out, _ = subgradient_fw(norm_sum1(best_seq), t1, verbose=False, rng_seed=1, top_k_peaks=5, beta=3.0)
        try_update(a_out, "softsgd_warmup")

    # === Phase 2: LP-FW (main workhorse, ~35% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80 and solver_name is not None:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW n={n_cur} (main) ===")
        t2 = min(remaining * 0.35, 500.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver_name, verbose=True)
        try_update(a_out, f"lp_main_n{n_cur}")

    # === Phase 3: CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: CD polish n={n_cur} ===")
        t3 = min(remaining * 0.10, 120.0)
        a_out, _ = peak_targeted_cd(norm_sum1(best_seq), t3, verbose=True, rng_seed=42)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 4: LP-FW round 2 (from CD result) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80 and solver_name is not None:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 350.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t4, solver_name, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 5: SoftSGD diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: SoftSGD n={n_cur} ===")
        t5 = min(remaining * 0.08, 80.0)
        a_out, _ = subgradient_fw(norm_sum1(best_seq), t5, verbose=False, rng_seed=77, top_k_peaks=8, beta=5.0)
        try_update(a_out, f"softsgd_n{n_cur}")

    # === Phase 6: CD after diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 6: CD n={n_cur} ===")
        t6 = min(remaining * 0.08, 80.0)
        a_out, _ = peak_targeted_cd(norm_sum1(best_seq), t6, verbose=True, rng_seed=123)
        try_update(a_out, f"cd2_n{n_cur}")

    # === Phase 7: Try larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_target = n_cur + 400
    if n_target > n_cur and remaining > 100 and solver_name is not None:
        print(f"\n=== Phase 7: Expand to n={n_target} ===")
        a_long = interpolate_sequence(best_seq, n_target)
        t7 = min(remaining * 0.15, 150.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(a_long), t7, solver_name, verbose=True)
        try_update(a_out, f"lp_n{n_target}")

    # === Phase 8: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and solver_name is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final LP-FW n={n_best} ===")
        t8 = min(remaining * 0.30, 200.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t8, solver_name, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final CD n={n_best} ===")
        t9 = min(remaining * 0.7, 100.0)
        a_out, _ = peak_targeted_cd(norm_sum1(best_seq), t9, verbose=False, rng_seed=999)
        try_update(a_out, f"cd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
