Minimizes 2*n*max(conv(a,a)) / sum(a)^2  (scale-invariant; normalized to sum=1
so we minimize 2*n*max(self-convolution)).

Key insight: the peak of np.convolve(a,a) for a NON-NEGATIVE sequence is NOT pinned
to sum(a_j^2); the center equals sum_j a[j]*a[n-1-j]. Hence an ASYMMETRIC,
edge-concentrated sequence can push the self-convolution peak well below the
Cauchy-Schrow bound 2, approaching ~1 for large n.

Strategy (different from previous LP-only workhorse):
  1. Multi-strategy initialization (arc-sine / bimodal / Fourier / sparse / parent).
  2. Nesterov-accelerated mirror descent (multiplicative weights) on the simplex
     with a log-sum-exp smoothed max and an O(n log n) FFT gradient
     (cross-correlation of the softmax of the convolution with the sequence).
  3. LP Frank-Wolfe with adaptive active-set as a complementary refiner.

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s

        except Exception:

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    """Linear self-convolution a*a (length 2n-1) via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fs = next_pow2(size)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def xcorr_shift(p, a):
    """
    Return g, length len(a), with g[j] = sum_i a[i] * p[i+j]  (out of range p=0).
    This is the forward-shifted cross-correlation of p with a.
    """
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    L = p.size
    n = a.size
    fs = next_pow2(n + L - 1)
    Pa = np.fft.rfft(p[::-1], fs)
    Aa = np.fft.rfft(a, fs)
    conv_fa = np.fft.irfft(Aa * Pa, fs)[:n + L - 1]
    base = L - 1
    return conv_fa[base - (np.arange(n))][::-1]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s


def compute_obj_normalized(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = fast_conv(a)
    return 2.0 * n * c.max()

def init_arc_sine(n, temperature=1.0):
    j = (np.arange(n) + 0.5) / float(n)
    x = np.clip(j, 0.02, 0.98)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.power(a, temperature)
    return normalize(a)


def init_bimodal(n, alpha=2.0):
    j = np.arange(n, dtype=np.float64)
    x = j / max(1, n - 1)
    a = np.exp(-alpha * ((x - 0.25) ** 2 + (x - 0.75) ** 2))
    a[0] += 0.15
    a[-1] += 0.15
    return normalize(a)


def init_fourier(n, n_harm=24, rng=None):
    if rng is None:

        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * j / n + phase)
    return normalize(a)

def mirror_descent_smoothed(a0, time_limit, beta_init=20.0, beta_max=400.0,
                            c0=2.0, restart_every=1500, verbose=True,
                            rng_seed=42):
    n = a0.size
    a = normalize(a0)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    beta = beta_init
    best_iter = 0
    iters_since_best = 0

    t = 1
    while time.monotonic() - t0 < time_limit:
        if t > 1:
            mom = (t - 1.0) / (t + 1.5)
            y = x + mom * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        max_c = c.max()
        obj = 2.0 * n * max_c
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
            best_iter = t
            iters_since_best = 0
        else:
            iters_since_best += 1

        # softmax(beta * c) -> peak weights
        z = beta * c - (beta * c).max()
        e = np.exp(z)
        es = e.sum()
        if es < 1e-300:
            p = np.ones_like(c) / c.size
        else:
            p = e / es

        # gradient = 2 * sum_k p[k] y[k-j]
        g = 2.0 * xcorr_shift(p, y)
        g = np.maximum(g, 0.0)

        lr = c0 / (np.sqrt(t) + 1.0)
        log_new = np.log(y) - lr * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        x_prev = y
        x = a_new

        # grow the sharpness of the max-smoothing
        if t % 100 == 0:
            beta = min(beta * 1.03, beta_max)

        # periodic restarts from the best point to escape plateaus
        if iters_since_best > restart_every:
            base = best_a
            for _ in range(3):
                mode = rng.integers(3)
                if mode == 0:
                    pert = base * (1.0 + rng.normal(0.0, 0.02, n))
                elif mode == 1:
                    pert = base.copy()
                    srt = rng.permutation(n)[:max(4, n // 40)]
                    for ii in srt:
                        pert[ii] *= rng.uniform(0.5, 1.8)
                else:
                    fresh = [init_arc_sine, init_bimodal, init_fourier][rng.integers(3)]
                    pert = 0.5 * base + 0.5 * fresh(n)
                pert = np.maximum(pert, 1e-12)
                pert = pert / pert.sum()
                po = compute_obj_normalized(pert)
                if po < best_obj:
                    best_obj = po
                    best_a = pert.copy()
                    x = pert.copy()
                    x_prev = pert.copy()
                    iters_since_best = 0
                    break
            else:
                x = base * (1.0 + rng.normal(0.0, 0.01, n))
                x = np.maximum(x, 1e-12)
                x = x / x.sum()
                x_prev = x.copy()
                iters_since_best = 0

        if verbose and t % 1500 == 0:
            print("    MD t=%d (%.1fs): obj=%.6f beta=%.1f"
                  % (t, time.monotonic() - t0, best_obj, beta))
        t += 1

    return best_a, best_obj

def lp_frank_wolfe(a0, time_limit, solver, top_k_init=60, top_k_max=220,
                   verbose=True):
    n = a0.size
    a = normalize(a0)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    cur_k = top_k_init

    t = 1
    while time.monotonic() - t0 < time_limit:
        if t > 1:
            mom = (t - 1.0) / (t + 1.5)
            y = x + mom * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        c = fast_conv(y)
        M = c.max()
        obj = 2.0 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()

        active = np.argsort(c)[::-1][:cur_k]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]
        for k in active:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            ji = np.arange(js, je + 1)
            constraints.append(y[ki - ji] @ b_var[ji] <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            t += 1
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            t += 1
            continue

        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        bs = b.sum()
        if bs < 1e-15:
            t += 1
            continue
        b = b / bs

        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * b
        x_new = np.maximum(x_new, 0.0)
        x_new = x_new / x_new.sum()
        x_prev = y
        x = x_new

        mx = fast_conv(x).max()
        if 2.0 * n * mx < best_obj:
            best_obj = 2.0 * n * mx
            best_a = x.copy()

        if t % 120 == 0 and cur_k < top_k_max:
            cur_k = min(cur_k + 6, top_k_max)

        if verbose and t % 120 == 0:
            print("    LP-FW t=%d (%.1fs): obj=%.6f topk=%d"
                  % (t, time.monotonic() - t0, best_obj, cur_k))
        t += 1

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    if old_n == new_n:
        return normalize(f)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        init_arc_sine(n_base, 1.0),
        init_arc_sine(n_base, 2.0),
        init_bimodal(n_base, 2.0),
        init_bimodal(n_base, 5.0),
        init_fourier(n_base, 24, rng),
        init_random_sparse(n_base, 0.35, rng),
        init_uniform(n_base),
    ]
    candidates.extend(extra)

    # pick the best seed
    for cand in candidates:
        cand = normalize(cand)
        o = compute_obj_normalized(cand)
        if o < best_score or best_seq is None:
            best_score = o
            best_seq = cand.copy()
    print("Best seed n=%d obj=%.6f" % (best_seq.size, best_score))

    remaining = budget - (time.monotonic() - t_global)
    if not np.isfinite(remaining) or remaining <= 0:
        remaining = 60.0

    # ---- Phase 1: mirror descent (fast workhorse) ------------------------
    print("=== Phase 1: Mirror descent n=%d ===" % best_seq.size)
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 20:
        a, _ = mirror_descent_smoothed(best_seq, t1, beta_init=20.0,
                                       beta_max=400.0, c0=2.0,
                                       restart_every=1200, verbose=True,
                                       rng_seed=42)
        o = compute_obj_normalized(a)
        if o < best_score:
            best_score, best_seq = o, a.copy()
            print("  *** MD1 %.6f n=%d" % (o, a.size))
        else:
            o2 = obj_of(a)
            if o2 < best_score:
                best_score, best_seq = o2, a.copy()

    # ---- Phase 2: LP Frank-Wolfe ----------------------------------------
    remaining = budget - (time.monotonic() - t_global)
    print("=== Phase 2: LP-FW n=%d ===" % best_seq.size)
    t2 = min(remaining * 0.22, 320.0)
    if t2 > 25:
        a, _ = lp_frank_wolfe(best_seq, t2, solver,
                              top_k_init=60, top_k_max=220, verbose=True)
        o = compute_obj_normalized(a)
        if o < best_score:
            best_score, best_seq = o, a.copy()
            print("  *** LP-FW %.6f n=%d" % (o, a.size))

    # ---- Phase 3: mirror descent again (refined) -------------------------
    remaining = budget - (time.monotonic() - t_global)
    print("=== Phase 3: Mirror descent refined n=%d ===" % best_seq.size)
    t3 = min(remaining * 0.22, 320.0)
    if t3 > 25:
        a, _ = mirror_descent_smoothed(best_seq, t3, beta_init=40.0,
                                       beta_max=600.0, c0=2.5,
                                       restart_every=1000, verbose=True,
                                       rng_seed=1337)
        o = compute_obj_normalized(a)
        if o < best_score:
            best_score, best_seq = o, a.copy()
            print("  *** MD3 %.6f n=%d" % (o, a.size))

    # ---- Phase 4: coordinate polish --------------------------------------
    remaining = budget - (time.monotonic() - t_global)
    print("=== Phase 4: Coord n=%d ===" % best_seq.size)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a, _ = coordinate_polish(best_seq, t4, verbose=True, rng_seed=42)
        o = compute_obj_normalized(a)
        if o < best_score:
            best_score, best_seq = o, a.copy()
            print("  *** Coord %.6f n=%d" % (o, a.size))

    # ---- Phase 5: length exploration -------------------------------------
    remaining = budget - (time.monotonic() - t_global)
    n0 = best_seq.size
    print("=== Phase 5: Length exploration (base n=%d) ===" % n0)
    for delta in (+400, -400, +800, -800):
        if remaining < 60:
            break
        n_alt = n0 + delta
        n_alt = int(max(800, min(n_alt, 3400)))
        if n_alt == n0:
            continue
        a_alt = interpolate_sequence(best_seq, n_alt)
        tt = min(remaining * 0.20, 120.0)
        a, _ = mirror_descent_smoothed(a_alt, tt, beta_init=30.0,
                                       beta_max=500.0, c0=2.2,
                                       restart_every=900, verbose=False,
                                       rng_seed=99 + n_alt)
        o = compute_obj_normalized(a)
        if o < best_score:
            best_score, best_seq = o, a.copy()
            print("  *** Len n=%d %.6f" % (n_alt, o))
        remaining = budget - (time.monotonic() - t_global)

    # ---- Phase 6: final LP + coord on the winner ------------------------
    remaining = budget - (time.monotonic() - t_global)
    if remaining > 50 and best_seq.size <= 3000:
        t6 = min(remaining * 0.5, 120.0)
        if t6 > 20:
            a, _ = lp_frank_wolfe(best_seq, t6, solver,
                                  top_k_init=80, top_k_max=240, verbose=False)
            o = compute_obj_normalized(a)
            if o < best_score:
                best_score, best_seq = o, a.copy()
                print("  *** Final-LP %.6f" % o)
            remaining = budget - (time.monotonic() - t_global)

    remaining = budget - (time.monotonic() - t_global)
    if remaining > 25:
        t7 = min(remaining * 0.7, 90.0)
        if t7 > 10:
            a, _ = coordinate_polish(best_seq, t7, verbose=False, rng_seed=1234)
            o = compute_obj_normalized(a)
            if o < best_score:
                best_score, best_seq = o, a.copy()
                print("  *** Final-Coord %.6f" % o)

    # ---- final validation ------------------------------------------------
    if best_seq is None:
        print("ERROR: no valid sequence")
        return [1.0]
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("FINAL score=%.6f n=%d elapsed=%.0fs"
          % (best_score, best_seq.size, time.monotonic() - t_global))
    return [float(x) for x in best_seq]
