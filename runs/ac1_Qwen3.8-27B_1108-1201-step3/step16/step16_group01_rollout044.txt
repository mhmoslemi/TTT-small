def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective_normalized(a):
    """Compute 2*n*max_conv for normalized (sum=1) sequence."""
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient_active(a, ac, threshold_frac):
    """Compute subgradient from active set of lags near max."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    return g, len(active)


def compute_subgradient_soft(a, ac, top_k, beta):
    """Compute soft subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def amd_with_backtracking(a0, time_limit, c0=3.0, threshold_frac=0.99,

    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    current_threshold = threshold_frac
    c_current = c0
    eta_current = c0 * np.sqrt(n)

    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        if use_soft:
            g = compute_subgradient_soft(y, ac, top_k=top_k, beta=soft_beta)
            n_active = top_k
        else:
            g, n_active = compute_subgradient_active(y, ac, threshold_frac=current_threshold)

        gn = np.linalg.norm(g)
        if gn < 1e-30:
            break
        g = g / gn

        # Backtracking line search
        eta = eta_current
        best_trial_m = fast_autocorr(y).max()
        accepted = False

        for bt in range(15):
            log_a = np.log(np.maximum(y, 1e-15)) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            M_new = fast_autocorr(a_new).max()

                accepted = True
                break
            eta *= 0.7

            log_a = np.log(np.maximum(y, 1e-15)) - eta * g

            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            M_new = fast_autocorr(a_new).max()

        x = a_new

            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            # Increase eta slightly on success

        else:
            stall_count += 1
            eta_current = max(eta_current * 0.95, 0.1)

            stall_count = 0

        ac = fast_autocorr(a)
        M_curr = ac.max()
        k_peak = int(ac.argmax())

        # Get all lags within 1% of max
        near_peak = np.where(ac >= 0.99 * M_curr)[0]

        # Compute peak sensitivity for each position
        peak_sens = np.zeros(n)
        for k in near_peak:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]

        # Find the position with highest peak sensitivity (most "dangerous")
        # and positions with low sensitivity (safe to add mass)
        hot = np.argsort(peak_sens)[::-1]
        cold = np.argsort(peak_sens)[:min(100, n // 4)]

            if time.monotonic() - t0 > time_limit:

                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial

                if improved:
                    break

            if improved:

                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if not improved and stall > 50:

                            a[i] -= amt

                                M_new = fast_autocorr(a).max()

                                    best_a = a.copy()
                                    stall = 0
                                else:
                                    a[i] += amt
                                    a[j] -= amt

        if verbose and stall % 50 == 0:

    while time.monotonic() - t0 < time_limit:
        it += 1
        M_curr = fast_autocorr(a).max()

        # Generate a random perturbation
        n_pert = rng.integers(1, max(2, n // 100))
        sources = rng.choice(n, size=n_pert, replace=False)

        a_trial = a.copy()
        for i in sources:
            j = int(rng.integers(n))
            if j != i:
                frac = rng.uniform(0.001, 0.02)
                amt = a_trial[i] * frac
                if amt > 1e-15:
                    a_trial[i] -= amt
                    a_trial[j] += amt

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        M_trial = fast_autocorr(a_trial).max()
        delta = M_trial - M_curr

        # Metropolis criterion
        if delta < 0 or (T > T_min and np.exp(-delta / T) > rng.random()):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a.copy()

        T *= cooling_rate

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA it={it} ({elapsed:.1f}s): J={J:.6f}, T={T:.2e}")

    sc = evaluate_safe(best_a)
    if np.isfinite(sc) and sc < _global_best_score:
        _global_best_score = sc
        _global_best = best_a.copy()
        print(f"  *** SA improved global: {sc:.6f} (n={n})")

    return best_a, best_m


def diversified_restart_optimize(a0, time_limit, verbose=True, rng_seed=42):
    """
    Run multiple AMD instances with different parameter settings.
    Track the best across all runs.
    """
    global _global_best, _global_best_score, _global_t0
    n = len(a0)
    t0 = time.monotonic()

    # Different parameter combinations to try
    configs = [
        dict(c0=3.0, threshold_frac=0.99, use_soft=False, top_k=40, soft_beta=4.0),
        dict(c0=4.0, threshold_frac=0.995, use_soft=False, top_k=40, soft_beta=4.0),
        dict(c0=2.0, threshold_frac=0.98, use_soft=False, top_k=40, soft_beta=4.0),
        dict(c0=3.0, threshold_frac=0.99, use_soft=True, top_k=30, soft_beta=5.0),
        dict(c0=3.5, threshold_frac=0.99, use_soft=True, top_k=60, soft_beta=3.0),
        dict(c0=5.0, threshold_frac=0.999, use_soft=False, top_k=40, soft_beta=4.0),
    ]

    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()

    for ci, cfg in enumerate(configs):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        alloc = min(remaining * 0.2, 180.0)
        if alloc < 15:
            break

        rng_i = np.random.default_rng(rng_seed + ci * 100)
        # Slightly perturb the starting point for diversity
        noise = rng_i.normal(0, 0.005, n) * best_a
        a_start = np.maximum(best_a + noise, 1e-12)
        a_start = a_start / a_start.sum()

        if verbose:
            print(f"  Config {ci}: c={cfg['c0']}, thresh={cfg['threshold_frac']}, soft={cfg['use_soft']}, alloc={alloc:.0f}s")

        a_out, m_out = amd_with_backtracking(
            a_start, alloc, c0=cfg['c0'], threshold_frac=cfg['threshold_frac'],
            use_soft=cfg['use_soft'], top_k=cfg['top_k'], soft_beta=cfg['soft_beta'],
            verbose=False, rng_seed=rng_seed + ci * 100
        )

        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            sc = evaluate_safe(best_a)
            if np.isfinite(sc) and sc < _global_best_score:
                _global_best_score = sc
                _global_best = best_a.copy()
                print(f"  *** Diversified improved: {sc:.6f} (n={n})")

    return best_a, best_m


def multi_length_search(a0, time_limit, verbose=True):
    """
    Try different sequence lengths by interpolation.
    """
    global _global_best, _global_best_score, _global_t0
    n0 = len(a0)
    t0 = time.monotonic()

    # Candidate lengths around n0
    candidates = [n0 - 400, n0 - 200, n0, n0 + 200, n0 + 400, n0 + 600]
    candidates = [c for c in candidates if 500 <= c <= 4000]
    # Remove duplicates (n0 might appear twice)
    seen = set()
    unique_candidates = []
    for c in candidates:
        if c not in seen:
            seen.add(c)
            unique_candidates.append(c)

    best_a = a0.copy()
    best_m = fast_autocorr(a0).max()

    for n_try in unique_candidates:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 60:
            break
        alloc = min(remaining * 0.3, 200.0)
        if alloc < 30:
            continue

        print(f"  Multi-length: n={n_try} (alloc={alloc:.0f}s)")
        a_interp = interpolate_sequence(a0, n_try)

        # Quick AMD
        a_out, m_out = amd_with_backtracking(
            a_interp, alloc, c0=3.0, threshold_frac=0.99,
            verbose=False, rng_seed=n_try % 1000
        )

        if m_out < best_m or (n_try != n0 and m_out < fast_autocorr(a0).max()):
            # This length is competitive, do more work
            if verbose:
                print(f"    n={n_try} competitive: J={2*n_try*m_out:.6f}, doing more")
            remaining2 = time_limit - (time.monotonic() - t0)
            alloc2 = min(remaining2 * 0.2, 150.0)
            if alloc2 > 30:
                a_out2, m_out2 = amd_with_backtracking(
                    a_out, alloc2, c0=3.5, threshold_frac=0.995,
                    verbose=False, rng_seed=(n_try + 1) % 1000
                )
                if m_out2 < m_out:
                    a_out = a_out2
                    m_out = m_out2

            if 2.0 * n_try * m_out < 2.0 * len(best_a) * fast_autocorr(best_a).max():
                best_a = a_out.copy()
                best_m = m_out
                sc = evaluate_safe(best_a)
                if np.isfinite(sc) and sc < _global_best_score:
                    _global_best_score = sc
                    _global_best = best_a.copy()
                    print(f"  *** Multi-length improved: {sc:.6f} (n={n_try})")

    return best_a


def pair_exchange_descent(a0, time_limit, verbose=True, rng_seed=555):
    """
    More systematic pair exchange: for each pair of positions involved in
    the peak, try exchanging mass with positions NOT involved in the peak.
    """
    global _global_best, _global_best_score, _global_t0
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Identify peak lags
        peak_threshold = 0.995 * M_curr
        peak_lags = np.where(ac >= peak_threshold)[0]

        # Build set of "hot" positions
        hot_set = set()
        for k in peak_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot_set.add(i)
                j = k_int - i
                if 0 <= j < n:
                    hot_set.add(j)

        cold_positions = [i for i in range(n) if i not in hot_set]
        hot_list = list(hot_set)

        if len(cold_positions) > 100:
            cold_positions = list(rng.choice(cold_positions, size=100, replace=False))

        improved = False
        rng.shuffle(hot_list)

        for src in hot_list[:50]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            # Sort cold by similarity in value
            cold_sorted = sorted(cold_positions, key=lambda j: abs(a[j] - a[src]))
            for dst in cold_sorted[:15]:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Escape move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.03, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairEx ({elapsed:.1f}s): J={J:.6f}, stall={stall}, hot={len(hot_set)}")

    sc = evaluate_safe(best_a)
    if np.isfinite(sc) and sc < _global_best_score:
        _global_best_score = sc
        _global_best = best_a.copy()
        print(f"  *** PairEx improved global: {sc:.6f} (n={n})")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with diversified strategies.
    """
    global _global_best, _global_best_score, _global_t0

    _global_best = None
    _global_best_score = float('inf')
    _global_t0 = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    print(f"Total time budget: {time_budget:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _global_best = parent.copy()
            _global_best_score = p_score
    except Exception:
        print("No parent available, using default init.")

    if _global_best is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        _global_best = best_seq.copy()
        _global_best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={_global_best_score:.6f}")

    n0 = len(_global_best)

    # === Phase 1: Diversified AMD (multiple configs) ===
    remaining = time_budget - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 1: Diversified AMD (n={n0}) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out = diversified_restart_optimize(_global_best, t1, verbose=True, rng_seed=42)
        if a_out is not None:
            a_out = norm_sum1(a_out)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < _global_best_score:
                _global_best_score = sc
                _global_best = a_out.copy()
                print(f"  Phase 1 final: {sc:.6f}")

    if t2 > 30:
        a_out, _ = peak_surgical_descent(_global_best, t2, verbose=True, rng_seed=123)
        if a_out is not None:
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < _global_best_score:
                _global_best_score = sc
                _global_best = a_out.copy()
                print(f"  Phase 2 final: {sc:.6f}")

    # === Phase 3: More AMD with tighter threshold ===
    remaining = time_budget - (time.monotonic() - _global_t0)
    n_best = len(_global_best)
    print(f"\n=== Phase 3: AMD tight (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, _ = amd_with_backtracking(

                print(f"  Phase 3 final: {sc:.6f}")

    # === Phase 4: Pair exchange ===

                print(f"  Phase 4 final: {sc:.6f}")
