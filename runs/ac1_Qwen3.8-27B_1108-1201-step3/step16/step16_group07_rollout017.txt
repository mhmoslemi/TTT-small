    size = 2 * n - 1

        prob.solve(solver=cp.CLARABEL)
        if lam.value is None:
            prob.solve(solver=cp.SCS)
        if lam.value is None:
            prob.solve(solver=cp.ECOS)
        lam_val = np.maximum(np.asarray(lam.value), 0)
        lam_sum = lam_val.sum()
        if lam_sum < 1e-15:
            lam_val = np.ones(nA) / nA
        else:
            lam_val = lam_val / lam_sum

        scores = conv[A]
        lam_val = np.exp(-5.0 * (M - scores) / max(M * 0.01, 1e-10))
        lam_val = lam_val / lam_val.sum()

    direction = G @ lam_val

        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            G[lo:hi, idx] = 2.0 * a[k - np.arange(lo, hi)]

    # Solve QP: min lambda^T H lambda s.t. lambda >= 0, sum(lambda) = 1
    # where H = G^T G - (G^T 1)(1^T G)/n (variance Hessian)
    G1 = G.sum(axis=0)
    H = G.T @ G - np.outer(G1, G1) / n

    # Ensure H is symmetric
    H = 0.5 * (H + H.T)

    # Try to solve QP
    lam_val = None
    try:
        lam = cp.Variable(nA)
        obj = cp.quad_form(lam, cp.psd_wrap(H))
        constraints = [lam >= 0, cp.sum(lam) == 1]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if lam.value is not None:
            lam_val = np.maximum(np.asarray(lam.value), 0.0)
    except Exception:
        pass

    if lam_val is None or lam_val.sum() < 1e-15:
        try:
            lam = cp.Variable(nA)
            obj = cp.quad_form(lam, cp.psd_wrap(H + 1e-8 * np.eye(nA)))
            constraints = [lam >= 0, cp.sum(lam) == 1]
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            if lam.value is not None:
                lam_val = np.maximum(np.asarray(lam.value), 0.0)
        except Exception:
            pass

    if lam_val is None or lam_val.sum() < 1e-15:
        # Fallback: softmax weighting based on peak heights
        scores = conv[A]
        scale = max(M * 0.02, 1e-10)
        lam_val = np.exp(-3.0 * (M - scores) / scale)
        lam_sum = lam_val.sum()
        if lam_sum < 1e-15:
            lam_val = np.ones(nA) / nA
        else:
            lam_val = lam_val / lam_sum

    lam_val = lam_val / lam_val.sum()
    direction = G @ lam_val
    return direction, active, M


def kkt_optimize(a0, time_limit, verbose=False, seed=0, n_peaks_init=None, tol_init=0.05):
    """KKT-balanced optimization with adaptive step size and progressive tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    n_peaks = n_peaks_init if n_peaks_init else max(20, n // 60)
    tol = tol_init
    step = 0.3 / n**0.3
    patience = 0
    max_patience = 400

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        direction, active, M = kkt_direction(a, n_peaks=n_peaks, tol=tol)
        if len(active) == 0:
            break

        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break

        # Try the step with backtracking
        alpha = step
        improved = False
        for _bt in range(8):
            a_trial = a - alpha * direction / dir_norm * n**0.3
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                break
            a_trial = a_trial * norm / s_trial
            score_trial = fast_conv(a_trial).max()
            if score_trial < best_score - 1e-15:
                a = a_trial
                best_score = score_trial
                best_a = a_trial.copy()
                patience = 0
                improved = True
                break
            alpha *= 0.7

        if not improved:
            patience += 1
            step *= 0.97
            if step < 1e-12:
                # Restart from best
                a = best_a.copy()
                step = 0.3 / n**0.3
                patience = 100
                # Increase peaks and decrease tol for progress
                n_peaks = min(n_peaks + max(5, n // 200), n // 3)
                tol = max(tol * 0.95, 0.01)
        else:
            step = min(step * 1.01, 10.0)
            # Gradually tighten
            if t % 100 == 0:
                tol = max(tol * 0.98, 0.01)
                n_peaks = min(n_peaks + 3, n // 3)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT t={t} ({elapsed:.0f}s): score={best_score:.10f}, peaks={len(active)}, tol={tol:.4f}")

    return best_a, best_score


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    solver = get_lp_solver()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0

    y = a.copy()
    y_prev = a.copy()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Update active set periodically
        if active_set is None or t % 60 == 0:
            progress = min(1.0, t / 800.0)
            tol = 0.1 * (1.0 - progress) + 0.01 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 100)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(1.2 * n):
                conv_at = conv_y[active_set]
                keep = np.argsort(conv_at)[-int(1.2 * n):]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            active_set = np.argsort(conv_y)[-max(10, n // 50):]
            b, duals = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.90)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        for candidate in [y, a]:
            score = fast_conv(candidate).max()
            if score < best_score:
                best_score = score
                best_a = candidate.copy()
                stall_count = 0
            else:
                stall_count += 1
                break

        # Ergodic average check
        if t % 80 == 0:
            a_avg = 0.6 * best_a + 0.4 * a
            a_avg = np.maximum(a_avg, 0.0)
            s_avg = a_avg.sum()
            if s_avg > 1e-15:
                a_avg = a_avg * norm / s_avg
                score_avg = fast_conv(a_avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = a_avg.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with efficient updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_conv(a)

        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            # Try several candidate values
            typical = norm / n
            candidates = [0.0, aj * 0.2, aj * 0.5, aj * 0.8, aj * 1.2, aj * 1.5, aj * 2.0]
            candidates += [typical * 0.3, typical, typical * 2.0, typical * 4.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))

            best_obj = best_score
            best_x = aj

            for x in candidates:
                if abs(x - aj) < 1e-14:
                    continue
                old_j = a[j]
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = old_j
                    continue
                a_scaled = a * norm / s_new
                score = fast_conv(a_scaled).max()
                if score < best_obj:
                    best_obj = score
                    best_x = x
                a[j] = old_j

            if abs(best_x - aj) > 1e-14:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    if best_obj < best_score:
                        best_score = best_obj
                        best_a = a.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
