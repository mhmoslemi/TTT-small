2. Backtracking line search for stable convergence

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_objective_and_grad(a, beta):
    """
    Compute the log-sum-exp smooth approximation of max(conv(a,a))
    and its gradient.
    Returns (value, gradient, peak_lag)
    """
    n = len(a)
    ac = fast_autocorr(a)
    # Log-sum-exp trick for numerical stability
    max_ac = ac.max()
    if max_ac < 1e-30:
        return 0.0, np.zeros(n), 0
    shifted = beta * (ac - max_ac)
    log_sum = np.log(np.sum(np.exp(shifted))) + beta * max_ac
    value = log_sum / beta
    
    # Softmax weights
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient: sum_k weights[k] * 2 * a[k-i] for each i
    g = np.zeros(n)
    # For efficiency, only use lags with significant weight
    significant = weights > 1e-10
    active_lags = np.where(significant)[0]
    
    for k in active_lags:
        k_int = int(k)
        w = weights[k_int]
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * w * a[j_idx]
    
    peak_lag = int(ac.argmax())
    return value, g, peak_lag

def mirror_step(a, g, eta):
    """Multiplicative update: a_i *= exp(-eta * g_i), then renormalize."""
    log_a = np.log(np.maximum(a, 1e-300)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

    """

    t0 = time.monotonic()

    beta_start = 5.0
    beta_end = 500.0

    # In 600s, we can do about 6 * 10^6 iterations (optimistic)

    elapsed_frac = 0.0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Coarse-to-fine: optimize at n/2 then upsample
2. Smooth mirror descent with log-sum-exp and backtracking
3. Exact subgradient AMD with Nesterov and adaptive threshold
4. Coordinate-level peak reduction
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_resample(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

def mirror_step(a, g, eta):
    log_a = np.log(np.maximum(a, 1e-300)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def smooth_phase(a0, time_limit, beta_start=5.0, beta_end=500.0, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = 1.0
    stall = 0
    max_stall = 100
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, t / 50000.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        ac = fast_autocorr(a)
        max_ac = ac.max()
        if max_ac < 1e-30:
            break
        
        shifted = beta * (ac - max_ac)
        max_shifted = shifted.max()
        exp_shifted = np.exp(shifted - max_shifted)
        weights = exp_shifted / exp_shifted.sum()
        
        g = np.zeros(n)
        # Only use significant weights
        sig = weights > 1e-8
        active = np.where(sig)[0]
        
        for k in active:
            k_i = int(k)
            w = weights[k_i]
            i_s = max(0, k_i - n + 1)
            i_e = min(n, k_i + 1)
            if i_s >= i_e:
                continue
            idx = np.arange(i_s, i_e)
            j_idx = k_i - idx
            g[idx] += 2.0 * w * a[j_idx]
        
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn
        
        # Backtracking
        improved = False
        for tried in range(6):

            a_new = mirror_step(a, g, eta_try)
            ac_new = fast_autocorr(a_new)
            if ac_new.max() < max_ac - 1e-14:
                a = a_new
                eta = eta_try
                improved = True
                break
        
        if not improved:

            a_new = mirror_step(a, g, eta * 0.05)
            a = a_new
            eta = max(eta * 0.8, 0.01)
        
        m = fast_autocorr(a).max()
        if m < best_m - 1e-15:
            best_m = m
            best_a = a.copy()

        else:

            stall = 0

            a = np.maximum(best_a + noise, 1e-10)
            a = a / a.sum()
            eta = 0.5
        
        if verbose and t % 3000 == 0:

            J = 2.0 * n * best_m
            print(f"    Smooth t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m
