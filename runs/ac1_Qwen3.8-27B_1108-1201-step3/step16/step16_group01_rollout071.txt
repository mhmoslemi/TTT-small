def _soft(z):
    z = np.asarray(z, dtype=np.float64)
    # To avoid overflow
    zm = z.max()
    if not np.isfinite(zm):

    s = w.sum()
    if not np.isfinite(s) or s <= 0.0:

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

Strategy:
- Maintain sum(a) = 1; then the score is 2*n*max(conv(a,a)).
- Use softmax/log-space multiplicative ("mirror") descent with annealed power
  weights on the convolution peaks.
- Polish with very fast exact pairwise mass transfers using incremental
  convolution updates (no FFT per trial).
- Optionally use smoothed L_p objectives and L-BFGS-B in a log-simplex
  parametrization for basin jumping.
- Try small length variations and diverse structured/random starts.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

# Globals for optional L-BFGS objective caching / deadline.
_LB_P = 2.0
_LB_REF = 1.0
_LB_CACHE_Y = None
_LB_CACHE_F = None
_LB_CACHE_G = None
_LBFGS_DEADLINE = 0.0


def _norm_pos(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def _soft(z):
    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=20.0, neginf=-20.0)
    if z.size == 0:
        return np.array([1.0], dtype=np.float64)
    zm = float(z.max())
    if not np.isfinite(zm):
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    w = np.exp(np.clip(z - zm, -745.0, 709.0))
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return w / s


def _resample(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    old_n = int(f.size)
    if old_n == int(new_n):
        return _norm_pos(f)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(int(new_n), dtype=np.float64) / float(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    out = np.interp(new_x, old_x, f_norm)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if not np.isfinite(ss) or ss <= 0.0:
        return np.ones(int(new_n), dtype=np.float64) / float(new_n)
    return out / ss


def _fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)[:size]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _internal_score(a):
    a = _norm_pos(a)
    c = _fast_conv(a)
    m = float(c.max())
    if not np.isfinite(m):
        return float("inf")
    return 2.0 * float(a.size) * m


def _grad_from_weights(a, w, max_loop=150):
    """
    Given weights w_k on lags k=0..2n-2, return g_i = 2*sum_k w_k a_{k-i}.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    size = 2 * n - 1
    if w.size != size:
        tmp = np.zeros(size, dtype=np.float64)
        m = min(w.size, size)
        if m > 0:
            tmp[:m] = w[:m]
        w = tmp

    idx = np.nonzero(w > 1e-18)[0]
    if idx.size == 0:
        return np.zeros(n, dtype=np.float64)

    if idx.size <= max_loop:
        g = np.zeros(n, dtype=np.float64)
        for kk in idx:
            ki = int(kk)
            wk = float(w[ki])
            if wk <= 0.0 or not np.isfinite(wk):
                continue
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s < e:
                j = ki - np.arange(s, e, dtype=np.int64)
                g[s:e] += 2.0 * wk * a[j]
        return g

    fsize = 1 << (size - 1).bit_length()
    Ar = np.fft.rfft(a[::-1], fsize)
    W = np.fft.rfft(w, fsize)
    h = np.fft.irfft(Ar * W, fsize)[:size]
    g = 2.0 * np.asarray(h[n - 1:2 * n - 1], dtype=np.float64)
    return g


def _power_weights(ac, p):
    """
    Stable weights proportional to (ac / max(ac))^p.
    """
    ac = np.asarray(ac, dtype=np.float64)
    if ac.size == 0:
        return np.array([1.0], dtype=np.float64)
    M = float(ac.max())
    if not np.isfinite(M) or M <= 0.0:
        w = np.zeros_like(ac, dtype=np.float64)
        w[0] = 1.0
        return w

    p = float(max(1.0, p))
    crel = ac / M
    np.maximum(crel, 0.0, out=crel)
    crel = np.minimum(crel, 1.0)

    pos = crel > 1e-300
    if not np.any(pos):
        w = np.zeros_like(ac, dtype=np.float64)
        w[int(np.argmax(ac))] = 1.0
        return w

    logw = np.full_like(ac, -np.inf, dtype=np.float64)
    logw[pos] = p * np.log(crel[pos])
    m = float(np.max(logw[pos]))
    if not np.isfinite(m):
        w = np.zeros_like(ac, dtype=np.float64)
        w[int(np.argmax(ac))] = 1.0
        return w

    ew = np.zeros_like(ac, dtype=np.float64)
    ew[pos] = np.exp(logw[pos] - m)
    ww = float(ew.sum())
    if not np.isfinite(ww) or ww <= 0.0:
        w = np.zeros_like(ac, dtype=np.float64)
        w[int(np.argmax(ac))] = 1.0
        return w
    return ew / ww


def _mirror_search(a0, time_limit, p_start=1.2, p_end=40.0,
                   eta0=1.2, momentum0=0.5, verbose=True, seed=1):
    """
    Log-space accelerated mirror descent with power-weighted subgradients.
    """
    a = _norm_pos(a0)
    n = int(a.size)
    ac = _fast_conv(a)
    best_m = float(ac.max())
    best = a.copy()

    if time_limit <= 0 or n == 0:
        return best, best_m

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(z.mean())
    z_old = z.copy()
    eta = float(eta0)
    t = 0
    stall = 0
    max_stall = 80
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        # Reach high p after roughly 30% of the phase, but not faster than 30s.
        p_time = max(30.0, 0.30 * time_limit)
        progress = min(1.0, elapsed / max(1e-9, p_time))
        p = p_start + (p_end - p_start) * progress

        a = _soft(z)
        ac = _fast_conv(a)
        M = float(ac.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        w = _power_weights(ac, p)
        g = _grad_from_weights(a, w, max_loop=120)
        g = np.asarray(g, dtype=np.float64)
        g -= float(g.mean())
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn < 1e-30:
            # No useful direction: perturb from best.
            noise = rng.normal(0.0, 0.02, n) * best
            ap = np.maximum(best + noise, 1e-12)
            ap = ap / ap.sum()
            a = ap
            z = np.log(np.maximum(a, 1e-300))
            z = z - float(z.mean())
            z_old = z.copy()
            eta = 1.0
            stall += 1
            if stall > max_stall * 3:
                break
            continue

        g = g / gn
        mu = float(momentum0 + (0.85 - momentum0) * progress)

        z_new = z.copy()
        a_new = a.copy()
        M_new = M
        accepted = False

        for tried in range(5):
            etr = eta * (0.5 ** tried)
            z_try = z + mu * (z - z_old) - etr * g
            z_try = z_try - float(z_try.mean())
            if not np.all(np.isfinite(z_try)):
                continue
            a_try = _soft(z_try)
            c_try = _fast_conv(a_try)
            M_try = float(c_try.max())
            if not np.isfinite(M_try):
                continue
            if M_try < M - 1e-15:
                z_new = z_try
                a_new = a_try
                M_new = M_try
                accepted = True
                if tried == 0 and M_new < M - 1e-14:
                    eta = min(eta * 1.05, 3.0)
                break

        if not accepted:
            # Try a small step without momentum.
            z_try = z - 0.02 * g
            z_try = z_try - float(z_try.mean())
            if np.all(np.isfinite(z_try)):
                a_try = _soft(z_try)
                c_try = _fast_conv(a_try)
                M_try = float(c_try.max())
                if np.isfinite(M_try) and M_try < M - 1e-15:
                    z_new = z_try
                    a_new = a_try
                    M_new = M_try
                    accepted = True
                else:
                    eta = max(0.05, eta * 0.5)
            else:
                eta = max(0.05, eta * 0.5)

        z_old = z
        z = z_new - float(z_new.mean())
        a = a_new

        if M_new < best_m - 1e-15:
            best_m = M_new
            best = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            noise = rng.normal(0.0, 0.025, n) * best
            ap = np.maximum(best + noise, 1e-12)
            ap = ap / ap.sum()
            a = ap
            z = np.log(np.maximum(a, 1e-300))
            z = z - float(z.mean())
            z_old = z.copy()
            eta = 1.0
            stall = 0

        if verbose and (t % 1500 == 0):
            print(
                f"    mirror n={n} t={t} elapsed={elapsed:.1f}s p={p:.2f} "
                f"cur={2.0*n*M_new:.6f} best={2.0*n*best_m:.6f} stall={stall}"
            )

    return best, best_m


def _trial_transfer_ac(ac, a, src, dst, delta):
    """
    Exact first-line convolution update for moving mass delta from src to dst.
    """
    ac = np.asarray(ac, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    size = 2 * n - 1
    c = ac.copy()
    scale = 2.0 * delta
    c[src:src + n] -= scale * a
    c[dst:dst + n] += scale * a

    d = 2 * src
    if d < size:
        c[d] += delta * delta
    d = 2 * dst
    if d < size:
        c[d] += delta * delta
    d = src + dst
    if d < size:
        c[d] -= 2.0 * delta * delta

    c = np.where(c > 0.0, c, 0.0)
    return c, float(c.max())


def _transfer_polish(a0, time_limit, verbose=True, seed=2):
    """
    Fast local search by exact pairwise transfers.
    """
    a = _norm_pos(a0)
    n = int(a.size)
    ac = _fast_conv(a)
    best = a.copy()
    best_m = float(ac.max())

    if time_limit <= 0:
        return best, best_m

    size = 2 * n - 1
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 350
    accepts = 0
    trials = 0
    frac_list = (0.01, 0.02, 0.04, 0.08, 0.15, 0.30)

    while time.monotonic() - t0 < time_limit:
        if accepts > 80 or trials > 5000:
            ac = _fast_conv(a)
            accepts = 0
            trials = 0

        M = float(ac.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        # Gradient-like pressure from high lags.
        w = _power_weights(ac, 18.0)
        g = _grad_from_weights(a, w, max_loop=80)

        if not np.all(np.isfinite(g)):
            stall += 1
            if stall > max_stall:
                break
            continue

        hi_idx = np.argsort(g)[::-1][:30]
        lo_idx = np.argsort(g)[:30]

        hi_list = []
        seen = set()
        for ii in hi_idx:
            ii = int(ii)
            if ii not in seen and a[ii] > 1e-13:
                hi_list.append(ii)
                seen.add(ii)

        lo_list = []
        seen = set()
        for jj in lo_idx:
            jj = int(jj)
            if jj not in seen:
                lo_list.append(jj)
                seen.add(jj)

        improved = False
        pass_trials = 0

        for src in hi_list:
            if improved:
                break
            for dst in lo_list:
                if src == dst:
                    continue
                pass_trials += 1
                trials += 1
                if pass_trials > 1600:
                    break

                base_delta = float(a[src])
                if base_delta <= 0.0:
                    continue

                for frac in frac_list:
                    delta = base_delta * float(frac)
                    if delta <= 0.0 or delta > 1e-300:
                        continue
                    c_try, M_try = _trial_transfer_ac(ac, a, src, dst, delta)
                    if M_try < best_m - 1e-15:
                        a[src] -= delta
                        a[dst] += delta
                        if a[src] < 0.0:
                            a[src] = 0.0
                        ss = float(a.sum())
                        if abs(ss - 1.0) > 1e-12:
                            a = _norm_pos(a)
                            ac = _fast_conv(a)
                            M = float(ac.max())
                            if M < best_m:
                                best_m = M
                                best = a.copy()
                        else:
                            ac = c_try
                            M = M_try
                            best_m = M_try
                            best = a.copy()
                        accepts += 1
                        improved = True
                        stall = 0
                        break
                if improved or pass_trials > 1600:
                    break

        if not improved:
            stall += 1

            # Sparse random exploration to escape active-set local minima.
            if stall < max_stall and len(hi_list) > 0:
                src = int(rng.choice(np.asarray(hi_list, dtype=np.int64)))
                if a[src] > 1e-13:
                    for frac in (0.005, 0.02, 0.06):
                        delta = float(a[src]) * float(frac)
                        if delta <= 0.0:
                            continue
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        c_try, M_try = _trial_transfer_ac(ac, a, src, dst, delta)
                        if M_try < best_m - 1e-15:
                            a[src] -= delta
                            a[dst] += delta
                            if a[src] < 0.0:
                                a[src] = 0.0
                            ss = float(a.sum())
                            if abs(ss - 1.0) > 1e-12:
                                a = _norm_pos(a)
                                ac = _fast_conv(a)
                                M = float(ac.max())
                            else:
                                ac = c_try
                                M = M_try
                            if M < best_m:
                                best_m = M
                                best = a.copy()
                            accepts += 1
                            improved = True
                            stall = 0
                            break
                        elif M_try < M + 1e-7 * max(1e-12, best_m) and rng.random() < 0.15:
                            # Hill-climb exploration: accept slightly worse current,
                            # but never update global best unless it improves.
                            a[src] -= delta
                            a[dst] += delta
                            if a[src] < 0.0:
                                a[src] = 0.0
                            ss = float(a.sum())
                            if abs(ss - 1.0) > 1e-12:
                                a = _norm_pos(a)
                            else:
                                a = a / max(ss, 1e-300)
                            ac = c_try if abs(ss - 1.0) <= 1e-12 else _fast_conv(a)
                            M = M_try
                            accepts += 1
                            improved = True
                            stall = 0
                            break

            if not improved and stall > max_stall:
                # Larger perturbation from best.
                for _ in range(4):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j:
                        amt = float(a[i]) * float(rng.uniform(0.01, 0.06))
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                ss = float(a.sum())
                if ss <= 0:
                    a = best.copy()
                else:
                    a = a / ss
                ac = _fast_conv(a)
                M = float(ac.max())
                stall = 0

        if verbose and (stall % 150 == 0 or (improved and accepts % 50 == 0)):
            elapsed = time.monotonic() - t0
            print(
                f"    polish n={n} elapsed={elapsed:.1f}s cur={2.0*n*M:.6f} "
                f"best={2.0*n*best_m:.6f} stall={stall}"
            )

    ac = _fast_conv(best)
    best_m = float(ac.max())
    return best, best_m


def _lbfgs_set_params(p, ref):
    global _LB_P, _LB_REF
    _LB_P = float(p)
    _LB_REF = float(max(ref, 1e-300))


def _lbfgs_reset_cache():
    global _LB_CACHE_Y, _LB_CACHE_F, _LB_CACHE_G
    _LB_CACHE_Y = None
    _LB_CACHE_F = None
    _LB_CACHE_G = None


def _lbfgs_compute(y):
    global _LB_CACHE_Y, _LB_CACHE_F, _LB_CACHE_G
    if _LBFGS_DEADLINE > 0.0 and time.monotonic() > _LBFGS_DEADLINE:
        raise TimeoutError("L-BFGS time limit reached")

    y = np.asarray(y, dtype=np.float64)
    if _LB_CACHE_Y is not None and np.array_equal(y, _LB_CACHE_Y):
        return _LB_CACHE_F, _LB_CACHE_G

    a = _soft(y)
    c = _fast_conv(a)
    ref = float(_LB_REF)
    rel = c / ref
    np.maximum(rel, 0.0, out=rel)
    p = float(_LB_P)

    if p <= 1.0:
        F = float(rel.sum())
        q = np.full_like(rel, 1.0 / ref, dtype=np.float64)
    else:
        relp = np.power(rel, p)
        F = float(relp.sum())
        if p == 2.0:
            q = 2.0 * rel / ref
        else:
            q = p * np.power(rel, p - 1.0) / ref

    if not np.isfinite(F):
        F = 0.0
        q = np.zeros_like(q)

    ga = _grad_from_weights(a, q, max_loop=200)
    S = float(np.dot(a, ga))
    gy = a * (ga - S)
    if not np.all(np.isfinite(gy)):
        gy = np.zeros_like(gy)

    _LB_CACHE_Y = y.copy()
    _LB_CACHE_F = F
    _LB_CACHE_G = gy
    return F, gy


def _lbfgs_fun(y):
    F, _ = _lbfgs_compute(y)
    return F


def _lbfgs_grad(y):
    _, G = _lbfgs_compute(y)
    return G


def _lbfgs_phase(a0, time_limit, p_list=(4.0, 8.0), verbose=True):
    if not _HAVE_SCIPY or time_limit <= 5.0:
        return _norm_pos(a0)

    global _LBFGS_DEADLINE

    a = _norm_pos(a0)
    n = int(a.size)
    best = a.copy()
    best_m = float(_fast_conv(a).max())

    t0 = time.monotonic()
    _LBFGS_DEADLINE = t0 + max(0.0, time_limit - 5.0)
    rng = np.random.default_rng(17)

    # Measure one evaluation budget roughly.
    _lbfgs_set_params(2.0, max(best_m, 1e-300))
    _lbfgs_reset_cache()
    y_try = np.zeros(n, dtype=np.float64) + rng.normal(0.0, 1e-9, n)
    mt0 = time.monotonic()
    for _ in range(3):
        try:
            _lbfgs_compute(y_try)
        except Exception:
            pass
    eval_time = max(1e-4, (time.monotonic() - mt0) / 3.0)

    for p in p_list:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        _lbfgs_set_params(float(p), max(best_m, 1e-300))
        _lbfgs_reset_cache()
        y = np.log(np.maximum(best, 1e-300))
        y = y - float(y.mean())

        rem = time_limit - (time.monotonic() - t0)
        if rem <= 3.0:
            break
        # Conservative estimated eval budget; 2x for fun/grad pair.
        max_total = int(rem / eval_time * 0.35)
        maxfun = max(40, min(1200, max_total // 2))
        maxiter = max(15, min(180, maxfun // 12))

        try:
            res = minimize(
                _lbfgs_fun,
                y,
                jac=_lbfgs_grad,
                method="L-BFGS-B",
                options={
                    "maxiter": int(maxiter),
                    "maxfun": int(maxfun),
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 20,
                    "disp": False,
                },
            )
            if res is not None and hasattr(res, "x"):
                a_try = _norm_pos(res.x)
                m_try = float(_fast_conv(a_try).max())
                if np.isfinite(m_try) and m_try < best_m - 1e-15:
                    best_m = m_try
                    best = a_try.copy()
                    if verbose:
                        print(
                            f"    LBFGS p={p:.1f} elapsed={time.monotonic()-t0:.1f}s "
                            f"obj={2.0*n*best_m:.6f}"
                        )
        except Exception:
            pass

    return best


def _make_init_candidates(parent, n, rng, limit=8):
    cands = []

    if parent is not None:
        par = _resample(parent, n)
        cands.append(par)

    x = np.linspace(0.0, 1.0, n)
    two_pi = 2.0 * np.pi

    # Random trigonometric / asymmetric seeds.
    for _ in range(4):
        a = np.ones(n, dtype=np.float64)
        ncomp = int(rng.integers(2, 5))
        for _cc in range(ncomp):
            f = int(rng.integers(1, 6))
            ph = float(rng.uniform(0.0, two_pi))
            amp = float(rng.uniform(0.25, 0.85))
            a += amp * np.cos(two_pi * f * x + ph)
        a = np.maximum(a, 0.02)
        a = a / a.sum()
        cands.append(a)

    # Shifted squared-sine family.
    for alpha in (0.75, 1.0):
        ph = float(rng.uniform(0.0, two_pi))
        a = (1.0 + alpha * np.sin(two_pi * x + ph)) ** 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        cands.append(a)

    # Smooth positive random.
    v = rng.normal(size=n)
    V = np.fft.rfft(v)
    freq = np.fft.rfftfreq(n)
    V = V * np.exp(-max(1e-9, (freq * n * 0.03) ** 2))
    v = np.fft.irfft(V, n)
    a = np.exp(np.clip(v, -5.0, 5.0))
    a += 0.01
    a = a / a.sum()
    cands.append(a)

    # Parent jitter.
    if parent is not None:
        par = _resample(parent, n)
        for scale in (0.05, 0.12):
            a = par * (1.0 + float(scale) * rng.normal(size=n))
            a = np.maximum(a, 1e-9)
            a = a / a.sum()
            cands.append(a)

    scored = []
    for a in cands:
        a = _norm_pos(a)
        s = _internal_score(a)
        if np.isfinite(s):
            scored.append((s, len(scored), a))
    scored.sort()

    out = []
    for s, idx, a in scored[:max(1, int(limit))]:
        out.append(a)
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out


def _safe_eval(seq):
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return float("inf")


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total_budget = min(float(budget_s) - 60.0, 1740.0)
    if total_budget < 90.0:
        total_budget = max(90.0, 0.9 * float(budget_s))

    parent = None
    base_n = 2400

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 100 and np.all(np.isfinite(p)) and float(p.sum()) > 1e-10:
            p = np.clip(p, 0.0, 1000.0)
            parent = p
            ln = int(parent.size)
            if ln < 900:
                parent = _resample(parent, 1400)
                base_n = 1400
            elif ln > 3000:
                parent = _resample(parent, 3000)
                base_n = 3000
            else:
                base_n = ln
    except Exception:
        parent = None
        base_n = 2400

    rng = np.random.default_rng(20240001)
    cands = _make_init_candidates(parent, base_n, rng, limit=8)
    best = cands[0]
    best_score = _internal_score(best)
    print(f"initial base_n={base_n} best_initial={best_score:.6f}")

    def remaining():
        return max(0.0, total_budget - (time.monotonic() - t_start))

    # Optional L-BFGS basin jumping.
    rem = remaining()
    if _HAVE_SCIPY and rem > 260.0:
        t_lb = min(rem * 0.15, 180.0)
        a_try = _lbfgs_phase(best, t_lb, p_list=(4.0, 8.0), verbose=True)
        sc = _internal_score(a_try)
        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            print(f"LBFGS improved: {best_score:.6f}")

    # Main mirror descent.
    rem = remaining()
    if rem > 90.0:
        t_mir = min(rem * 0.45, 600.0)
        a_try, m_try = _mirror_search(
            best, t_mir,
            p_start=1.2, p_end=40.0,
            eta0=1.2, momentum0=0.5,
            verbose=True, seed=101

        sc = _internal_score(a_try)
        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            print(f"mirror improved: {best_score:.6f}")

    # Fast local polishing.
    rem = remaining()
    if rem > 70.0:
        t_pol = min(rem * 0.35, 420.0)
        a_try, m_try = _transfer_polish(best, t_pol, verbose=True, seed=202)
        sc = _internal_score(a_try)
        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            print(f"polish improved: {best_score:.6f}")

    # Second, more peak-focused mirror pass.
    rem = remaining()
    if rem > 100.0:
        t_mir2 = min(rem * 0.25, 260.0)
        a_try, m_try = _mirror_search(
            best, t_mir2,
            p_start=4.0, p_end=120.0,
            eta0=0.8, momentum0=0.6,
            verbose=False, seed=303
        )
        sc = _internal_score(a_try)
        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            print(f"mirror2 improved: {best_score:.6f}")

    # Length exploration.
    rem = remaining()
    if rem > 180.0:
        n_best = int(best.size)
        deltas = (300, -300, 600)

        for delta in deltas:

            if nn < 900 or nn > 3200 or nn in tried:

                    a_try, t1,

            if rem > 25.0:

                sc = _internal_score(a_try)
                if np.isfinite(sc) and sc < best_score:
                    best = a_try
                    best_score = sc
                    print(f"length {nn} improved: {best_score:.6f}")
            rem = remaining()

    # Final local polish with whatever time remains.
    rem = remaining()

        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            print(f"final polish improved: {best_score:.6f}")

        )

        if np.isfinite(sc) and sc < best_score:
            best = a_try
            best_score = sc
            print(f"final mirror improved: {best_score:.6f}")

    # Prepare final sequence. Keep both untrimmed and lightly trimmed variants.
    best_arr = _norm_pos(best)
    seq_untrimmed = [float(x) for x in best_arr]
    score_untrimmed = _safe_eval(seq_untrimmed)

    trimmed = best_arr.copy()
    trimmed[trimmed < 1e-16] = 0.0
    s = float(trimmed.sum())
    if s > 0.0:
        trimmed = trimmed / s
    seq_trimmed = [float(x) for x in trimmed]
    score_trimmed = _safe_eval(seq_trimmed)

    if not np.isfinite(score_trimmed) or score_untrimmed < score_trimmed:
        seq_out = seq_untrimmed
        final_score = score_untrimmed
    else:
        seq_out = seq_trimmed
        final_score = score_trimmed

    if not np.isfinite(final_score):
        # Last resort: return a safe uniform sequence.
        n_safe = int(best.size) if best.size > 0 else 2400
        seq_out = [1.0 / float(n_safe)] * n_safe
        final_score = _safe_eval(seq_out)

    print(f"FINAL internal/call score={final_score:.6f} n={len(seq_out)} time={time.monotonic()-t_start:.1f}s")
    return seq_out
```
