def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def build_constraint_matrix(a, lags):
    """Build sparse constraint matrix for (a*b)_k <= M for k in lags."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        cols.extend(range(i_start, i_end))
        rows.extend([row_idx] * (i_end - i_start))
        vals.extend(a[k_int - i_start:k_int - i_start + (i_end - i_start)].tolist())
    return rows, cols, vals

def mirror_descent_subgradient(a0, time_limit, verbose=True):

            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Active set: lags within 1% of max
        active = np.where(ac >= M * 0.99)[0]
        
        # Compute subgradient: average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * a[j_idx]

        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a
        
        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m:

            best_a = a.copy()

            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.8f}")
    
    return best_a, best_m
