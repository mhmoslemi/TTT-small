def apg_optimize(a0, time_limit, beta_start=30.0, beta_end=5.0,

                 verbose=True, seed=42, use_backtrack=True):

    n = len(a0)

    best_m = C_best.max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    cur_top_k = top_k_init
    gamma = c0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 15000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov extrapolation
        momentum = (t - 1.0) / (t + 1.5)
        y = x + momentum * (x - x_prev)
        y = project_simplex(y)

        C_y = fast_conv(y)
        g, _ = smooth_grad(y, C_y, cur_beta, top_k=cur_top_k)

        # Step size with backtracking
        if use_backtrack:
            eta = gamma / (np.sqrt(t) + 1.0)
            # Backtracking: try to ensure decrease in smoothed objective
            f_y, _ = smooth_grad(y, C_y, cur_beta, top_k=cur_top_k)
            f_y_val = _
            for _bt in range(10):
                x_trial = project_simplex(y - eta * g)
                C_trial = fast_conv(x_trial)
                f_trial, f_trial_val = smooth_grad(x_trial, C_trial, cur_beta, top_k=cur_top_k)
                if f_trial_val <= f_y_val - 0.5 * eta * np.dot(g, x_trial - y):
                    break
                eta *= 0.5
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            x_trial = project_simplex(y - eta * g)

        x_prev = x
        x = x_trial

        C_new = fast_conv(x)
        M_new = C_new.max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            C_best = C_new
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            cur_top_k = min(cur_top_k + 30, top_k_max)
            # Diverse restart
            strategy = rng.integers(3)
            if strategy == 0:
                noise = rng.normal(0, 0.01, n) * best_a
                x_re = np.maximum(best_a + noise, 1e-12)
            elif strategy == 1:

                shift = int(rng.integers(1, max(2, n // 50)))
                x_re = np.roll(best_a, shift)
                x_re = np.maximum(x_re, 1e-12)
            else:
                # Blend with random
                rand = rng.dirichlet(np.ones(n) * 0.1)
                x_re = 0.7 * best_a + 0.3 * rand
            x_re = norm1(x_re)
            x = x_re
            x_prev = x_re.copy()
            if verbose:
                elapsed = time.monotonic() - t0

def lp_fw_optimize(a0, time_limit, solver, verbose=True, m_tighten=0.0):
    """LP Frank-Wolfe with adaptive M tightening and Nesterov momentum."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    C_best = fast_conv(a)
    best_m = C_best.max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        C_y = fast_conv(y)
        M_curr = C_y.max()

        # Adaptive tightening
        tighten = m_tighten * min(1.0, t / 50.0)
        M_eff = M_curr * (1.0 - tighten)
        if M_eff < 1e-15:
            break

        # Build Toeplitz matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_eff, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(y * b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        sb = b_val.sum()
        if sb < 1e-12:
            continue
        b_dir = b_val / sb

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = norm1(y_new)

        beta_t = (t - 0.5) / (t + 1.5)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = norm1(x_mom)

        y = y_new

        for candidate in [y, x_mom]:
            C_c = fast_conv(candidate)
            mc = C_c.max()
            if mc < best_m - 1e-14:
                best_m = mc
                best_a = candidate.copy()
                C_best = C_c

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, tighten={tighten:.3f}")

    return best_a, best_m, C_best

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = norm1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return norm1(f_interp)

def make_inits(n, rng):

    # Cosine concentrated center

    inits.append(norm1(np.maximum(c, 1e-10)))
    # Bimodal

    inits.append(norm1(b))
    # Random smoothed

    inits.append(norm1(chirp ** 2 + 1e-6))
    return inits

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1700.0)
    rng = np.random.default_rng(2025)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_a = None
    best_score = float('inf')
    best_C = None

    def try_update(a, label):
        nonlocal best_a, best_score, best_C
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        C = fast_conv(a_norm)
        score = 2.0 * len(a_norm) * C.max() / (s * s)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_norm.copy()
            best_C = C.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # Initialize
    parent = get_parent()
    if parent is not None:
        best_a = norm1(parent)
        best_C = fast_conv(best_a)
        best_score = 2.0 * len(best_a) * best_C.max()
        print(f"Parent: n={len(best_a)}, score={best_score:.6f}")
        n0 = len(best_a)
    else:
        n0 = 2400
        x = np.linspace(0, 1, n0)
        best_a = norm1(np.sin(np.pi * x) ** 2)
        best_C = fast_conv(best_a)
        best_score = 2.0 * n0 * best_C.max()
        print(f"Default init: n={n0}, score={best_score:.6f}")

    # === Phase 1: APG main optimization (45% of budget) ===
    rem = remaining()
    print(f"\n=== Phase 1: APG n={len(best_a)} (budget={rem*0.45:.0f}s) ===")
    t1 = rem * 0.35
    if t1 > 60:
        a_out, m_out, C_out = apg_optimize(
            best_a, t1, beta_start=30.0, beta_end=5.0,
            top_k_init=120, top_k_max=350, c0=2.5,
            verbose=True, seed=42, use_backtrack=True)
        try_update(a_out, f"apg1_n{len(a_out)}")

    # Second APG with different params
    rem = remaining()
    t1b = rem * 0.12
    if t1b > 40 and best_a is not None:
        a_out, m_out, C_out = apg_optimize(
            best_a, t1b, beta_start=15.0, beta_end=8.0,
            top_k_init=200, top_k_max=400, c0=1.5,
            verbose=False, seed=123, use_backtrack=True)
        try_update(a_out, f"apg1b_n{len(a_out)}")

    # === Phase 2: LP-FW polishing (25% of budget) ===
    rem = remaining()
    print(f"\n=== Phase 2: LP-FW n={len(best_a)} (budget={rem*0.25:.0f}s) ===")
    t2 = rem * 0.25
    if t2 > 60 and len(best_a) <= 3000:
        a_out, m_out, C_out = lp_fw_optimize(
            best_a, t2, solver, verbose=True, m_tighten=0.05)
        try_update(a_out, f"lpfw1_n{len(a_out)}")

    # === Phase 3: APG + Coord refinement (15%) ===
    rem = remaining()
    print(f"\n=== Phase 3: Refine n={len(best_a)} (budget={rem*0.15:.0f}s) ===")
    t3 = rem * 0.08
    if t3 > 30:
        a_out, m_out, C_out = apg_optimize(
            best_a, t3, beta_start=20.0, beta_end=7.0,
            top_k_init=200, top_k_max=400, c0=2.0,
            verbose=False, seed=77, use_backtrack=True)
        try_update(a_out, f"apg2_n{len(a_out)}")

    rem = remaining()
    t3b = rem * 0.08
    if t3b > 20:
        a_out, m_out, C_out = peak_coord_descent(
            best_a, t3b, verbose=True, seed=99)
        try_update(a_out, f"coord1_n{len(a_out)}")

    # === Phase 4: Multi-length exploration (20%) ===
    rem = remaining()
    print(f"\n=== Phase 4: Multi-length (budget={rem*0.25:.0f}s) ===")
    current_n = len(best_a)
    lengths_to_try = []
    # Try neighbors
    for delta in [200, 400, -200, 600]:
        nl = current_n + delta
        if 1600 <= nl <= 3500 and nl != current_n:
            lengths_to_try.append(nl)
    # Also try some standard lengths
    for nl in [2000, 2500, 3000, 3200]:
        if 1600 <= nl <= 3500 and nl not in lengths_to_try and nl != current_n:
            lengths_to_try.append(nl)

    lengths_to_try = sorted(set(lengths_to_try))[:5]
    t_per_length = (rem * 0.25) / max(1, len(lengths_to_try))

    for nl in lengths_to_try:
        rem = remaining()
        if rem < 60:
            break
        if t_per_length < 30:
            continue
        print(f"  Trying n={nl}...")
        a_long = interpolate_seq(best_a, nl)
        # Quick APG
        t_al = min(t_per_length * 0.6, rem * 0.4)
        if t_al > 20:
            a_out, m_out, C_out = apg_optimize(
                a_long, t_al, beta_start=25.0, beta_end=8.0,
                top_k_init=120, top_k_max=300, c0=2.0,
                verbose=False, seed=55 + nl % 100, use_backtrack=True)
            try_update(a_out, f"multil_n{nl}")

        rem = remaining()
        if rem > 30 and best_a is not None and len(best_a) == nl:
            # Quick coord
            t_cl = min(t_per_length * 0.3, rem * 0.3)
            if t_cl > 15:
                a_out, m_out, C_out = peak_coord_descent(
                    best_a, t_cl, verbose=False, seed=nl % 1000)
                try_update(a_out, f"coord_ml_n{nl}")

    # === Phase 5: LP-FW at best length (15%) ===
    rem = remaining()
    print(f"\n=== Phase 5: Final LP n={len(best_a)} (budget={rem*0.20:.0f}s) ===")
    t5 = rem * 0.20
    if t5 > 60 and len(best_a) <= 3000:
        a_out, m_out, C_out = lp_fw_optimize(
            best_a, t5, solver, verbose=True, m_tighten=0.08)
        try_update(a_out, f"lpfw2_n{len(a_out)}")

    # === Phase 6: Diverse initializations + APG (10%) ===
    rem = remaining()
    print(f"\n=== Phase 6: Diverse inits (budget={rem*0.12:.0f}s) ===")
    if rem > 120:
        n_best = len(best_a)
        inits = make_inits(n_best, rng)
        t_per_init = (rem * 0.12) / len(inits)
        for idx, init in enumerate(inits[:4]):
            rem = remaining()
            if rem < 40:
                break
            t_ii = min(t_per_init, rem - 20)
            if t_ii < 15:
                continue
            # Blend with current best for warm start
            blend = 0.5 * norm1(best_a) + 0.5 * init
            blend = norm1(blend)
            a_out, m_out, C_out = apg_optimize(
                blend, t_ii, beta_start=20.0, beta_end=10.0,
                top_k_init=150, top_k_max=300, c0=2.0,
                verbose=False, seed=200 + idx, use_backtrack=True)
            try_update(a_out, f"diverse{idx}_n{len(a_out)}")

    # === Phase 7: Final intensive coordinate descent (remaining) ===
    rem = remaining()
    print(f"\n=== Phase 7: Final coord n={len(best_a)} (budget={rem*0.5:.0f}s) ===")
    t7 = rem * 0.5
    if t7 > 20:
        a_out, m_out, C_out = peak_coord_descent(
            best_a, t7, verbose=True, seed=314)
        try_update(a_out, f"coord_final_n{len(a_out)}")

    # === Phase 8: Final APG micro-polish ===
    rem = remaining()
    if rem > 20:
        t8 = rem * 0.5
        if t8 > 10:
            a_out, m_out, C_out = apg_optimize(
                best_a, t8, beta_start=10.0, beta_end=10.0,
                top_k_init=300, top_k_max=500, c0=1.0,
                verbose=False, seed=999, use_backtrack=True)
            try_update(a_out, f"apg_final_n{len(a_out)}")

    # Final validation
    if best_a is not None:
        C_final = fast_conv(best_a)
        final_score = 2.0 * len(best_a) * C_final.max()
        print(f"\nFINAL: score={final_score:.6f}, n={len(best_a)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_a]
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
