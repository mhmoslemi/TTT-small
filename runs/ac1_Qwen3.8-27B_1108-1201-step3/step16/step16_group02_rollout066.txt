"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 (with sum normalized to 1 this is 2*n*max_b).

Approach (a departure from the previous LP-subset / NMD blend):
  1. Spectral "flat-autocorrelation" initialization (Gerchberg-Saxton style

  2. FULL-constraint LP Frank-Wolfe: instead of a hand-picked subset of lags,
     the linear oracle uses ALL (2n-1) autocorrelation bound constraints, giving
     markedly better step directions. A per-solve time guard automatically falls
     back to a top-lag subset if a single LP becomes too expensive.
  3. Nesterov mirror descent with an active-set (top-k, power-weighted)

  4. Coordinate-descent / mass-transfer polish.
  5. Multi-length orchestration keeping the globally best evaluated sequence.

import numpy as np

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print("  LP solver: %s" % s)
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def default_init(n):
    x = np.linspace(0.0, 1.0, n)
    a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = a + 1e-3
    return a / a.sum()


def spectral_init(n, n_iters=400, seed=42):
    """Constant-magnitude DFT matching (Gerchberg-Saxton) to flatten autocorr."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    for it in range(n_iters):
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        mag = np.abs(A)
        mean_pow = (mag * mag).mean()
        target = np.sqrt(mean_pow)
        if target < 1e-9:
            break
        phases = np.angle(A)
        A_new = target * np.exp(1j * phases)
        A_new[0] = np.abs(A_new[0])
        a_full = np.fft.irfft(A_new, fft_size)
        a_new = np.maximum(a_full[:n], 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        if it % 8 == 0:
            m = fast_autocorr(a_new).max()
            if m < best_m:
                best_m = m
                best_a = a_new.copy()
        a = a_new
    return best_a


def build_constraint_matrix(y, lags):
    n = len(y)
    m = len(lags)
    rows = []
    cols = []
    vals = []
    for r, k in enumerate(lags):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 >= i1:
            continue
        ii = np.arange(i0, i1)
        rows.append(np.full(ii.shape, r, dtype=np.int32))
        cols.append(ii.astype(np.int32))
        vals.append(np.asarray(y[k - ii], dtype=np.float64))
    if not rows:
        return sp.csr_matrix((m, n))
    rr = np.concatenate(rows)
    cc = np.concatenate(cols)
    vv = np.concatenate(vals)
    return sp.csr_matrix((vv, (rr, cc)), shape=(m, n))


def solve_lp_dir(y, M, n, solver, max_lags, full, lp_time_cap):
    """Return sum=1 nonneg direction b from the LIN oracle, or None."""
    order = np.argsort(fast_autocorr(y))[::-1]
    if max_lags and max_lags > 0:
        lags = order[:max_lags]
    else:
        lags = np.arange(2 * n - 1)
    T = build_constraint_matrix(y, lags)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    t0 = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, 0.0
    dt = time.monotonic() - t0
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, dt
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, dt
    return b_val / s, dt


def fp_clean(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s


def lp_frank_wolfe(a0, time_limit, solver, full, max_lags, lp_time_cap, verbose=True):

        if b is None:
            use_full = False

                break
            continue
        if use_full and dt > lp_time_cap:
            use_full = False

def compute_topk_subgradient(a, ac, top_k=80, beta=4.0):
    n = len(a)
    kk = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:kk]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(kk) / kk
    else:
        w = w / ws
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i0 = max(0, ki - n + 1)
        i1 = min(n, ki + 1)
        if i0 >= i1:
            continue
        idx = np.arange(i0, i1)
        grad[idx] += 2.0 * wi * a[ki - idx]
    return grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        a_in = norm_sum1(interpolate_sequence(best_seq, n_full))
        print("\n=== Phase 1: full-constraint LP-FW n=%d (budget=%.0fs) ===" % (n_full, t1))
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, full=True, max_lags=250,
                                  lp_time_cap=10.0, verbose=True)
        try_update(a_out, "LPfull_n%d" % n_full)

    # Phase 2: NMD refiner on current best
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 360.0)
    if t2 > 60 and best_seq is not None:
        print("\n=== Phase 2: NMD n=%d (budget=%.0fs) ===" % (len(best_seq), t2))
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t2,
                                           initial_temp=0.008, final_temp=0.001,
                                           c0=3.0, top_k=90, verbose=True)
        try_update(a_out, "NMD_n%d" % len(best_seq))

    # Phase 3: subset LP-FW on the best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60 and best_seq is not None:
        print("\n=== Phase 3: subset LP-FW n=%d (budget=%.0fs) ===" % (len(best_seq), t3))
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver, full=False,
                                  max_lags=300, lp_time_cap=12.0, verbose=True)
        try_update(a_out, "LPsub_n%d" % len(best_seq))

    # Phase 4: coordinate-descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30 and best_seq is not None:
        print("\n=== Phase 4: CD polish n=%d (budget=%.0fs) ===" % (len(best_seq), t4))
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t4,
                                             rng_seed=777, verbose=True)
        try_update(a_out, "CD_n%d" % len(best_seq))

    # Phase 5: final short NMD + CD swap on the global best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        t5 = min(remaining * 0.5, 120.0)
        print("\n=== Phase 5: final NMD n=%d (budget=%.0fs) ===" % (len(best_seq), t5))
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t5,
                                           initial_temp=0.004, final_temp=0.0008,
                                           c0=2.0, top_k=60, verbose=False)
        try_update(a_out, "NMDfinal_n%d" % len(best_seq))
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.6, 90.0)
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t5b,
                                                 rng_seed=555, verbose=False)
            try_update(a_out, "CDfinal_n%d" % len(best_seq))

    if best_seq is None:
        return [1.0]
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("\nFINAL: score=%.8f, n=%d, time=%.0fs" %
          (best_score, len(best_seq), time.monotonic() - t_global))
    return [float(x) for x in best_seq]
