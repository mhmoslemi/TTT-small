def fast_autocorr(a):
    """Fast autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def soft_subgradient(a, ac, top_k=50, beta=8.0):
    """
    Soft subgradient: weighted combination of gradients from top-k lags.
    Uses power-law weighting for smooth focusing.
    """
    n = len(a)
    n_lags = min(top_k, len(ac))
    # Get top lags
    top_idx = np.argsort(ac)[::-1][:n_lags]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(n_lags) / n_lags
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += wi * a[j_idx]
    
    # The gradient of conv(a,a)[k] w.r.t. a[i] is a[k-i] (for valid k-i)
    # But we need factor of 2 for i ≠ k-i, so the subgradient of max is 2*B_k*a
    # For the soft version, we use the weighted sum
    return 2.0 * g

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, total_iters / 20000)  #Annealing progress
        
        # Nesterov extrapolation
        beta_t = (total_iters) / (total_iters + 1.5) if total_iters > 0 else 0.0
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()

    for step in range(min(n_steps, n)):
        i = idx_order[step % n]
        if a[i] < 1e-12:

        # Terms involving a[i]: a[i]*a[k-i] (j=i, valid if 0<=k-i<n) + a[k-i]*a[i] (j=k-i, valid if 0<=k-i<n and k-i≠i)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s

2. Compute `ac = conv(a,a)`
3. The "target" autocorrelation is `ac_target[k] = 1/(2n-1)` for all k (uniform)
4. Update: `a_new[i] = a[i] * (ac_target / ac)[i]`... this doesn't directly work because the relationship isn't multiplicative.

Let me think differently. The subgradient approach is actually the right one. Let me focus on making it more effective.

**Key improvement idea**: Instead of using a fixed subgradient, use an **adaptive** one that:
- Starts with a "soft" subgradient (all lags contribute, weighted by their value)
- Gradually focuses on the maximum lag(s)
- Uses a "gap" criterion: if the top few lags are close to each other, use all of them; if one dominates, focus on it

This is essentially what the previous code did with `threshold_frac`, but let me make it more continuous.

**Another key idea**: Use **dual variables** (or "Lagrange multipliers") for the active constraints. This is the basis of the LP Frank-Wolfe approach from the previous code, but we can make it more efficient.

Let me combine these ideas into a new algorithm:

**Algorithm: Adaptive Subgradient with Spectral Preconditioning**

1. Maintain `a` (the current iterate)
2. At each step:
   a. Compute `ac = conv(a,a)`
   b. Compute weights `w_k` for each lag k (based on how close `ac[k]` is to `max(ac)`)
   c. Compute subgradient `g = sum_k w_k * grad_k(a)` where `grad_k(a)[i] = a[k-i] + a[k-i]` (appropriate for the quadratic)
   d. Use a **preconditioned** step: instead of `a <- a - eta * g`, use `a <- a * exp(-eta * g / a)` (log-space/mirror step)
   e. Adapt `eta` using backtracking
3. Periodically do coordinate descent for local improvements

The key innovation is the **preconditioning** step (e) and the **adaptive weights** (b).

For the weights, I'll use: `w_k = exp(ac[k] / tau) / sum_j exp(ac[j] / tau)` where `tau` is a temperature parameter that decreases over time.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive FISTA with softmax-weighted subgradient (temperature annealing)
2. Efficient 1D analytic coordinate descent (O(n) per coordinate via conv update formula)
3. Spectral flattening (iterative power spectrum equalization)
4. Multi-phase orchestration with diverse initializations and length exploration
Key innovations: adaptive softmax subgradient, O(n) coordinate descent, spectral method

import numpy as np
import time


def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def softmax_subgradient(a, ac, top_k=50, tau=0.05):
    """
    Softmax-weighted subgradient. tau controls focus:
    small tau = focus on max, large tau = spread across lags.
    """
    n = len(a)
    n_use = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:n_use]
    top_vals = ac[top_idx]
    M = top_vals[0]
    if M < 1e-30:
        return np.zeros(n)
    rel = top_vals / M
    exponents = rel / tau - (rel / tau).max()
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(n_use) / n_use
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += wi * a[j_idx]
    return 2.0 * g

def coord_descent_efficient(a0, time_limit, verbose=True, rng_seed=100):
    """
    Efficient coordinate descent using O(n) formula for conv update.
    For coordinate i: conv_new[k] = conv_old[k] + 2*delta*a[k-i] (for valid k-i != i)
                                              + 2*val*delta + delta^2 (for k=2i)
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 15
    n_sweeps = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        improved = False
        n_sweeps += 1

        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            val = a[i]
            if val < 1e-14:
                continue

            # Affected range: k in [i, i+n-1] (where 0 <= k-i < n)
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > k_hi:
                continue
            a_slice = a[0:k_hi - k_lo + 1]

            # Grid search for optimal delta
            # delta range: [-val, val] (so new value in [0, 2*val])
            # Use adaptive grid: finer near 0 if current value is small
            n_grid = 25
            delta_grid = np.linspace(-val, val * 0.8, n_grid)

            best_delta = 0.0
            best_trial_M = M_curr

            for dd in delta_grid:
                # conv_trial[k] = ac[k] + 2*dd*a_slice[k-k_lo] for k in [k_lo, k_hi]
                # Plus at k=2i: additional dd^2
                # We need max over all k
                # Max outside affected range:
                if k_lo > 0:
                    max_before = ac[0:k_lo]
                    max_out = max(max_before.max() if len(max_before) > 0 else 0.0,
                                  ac[k_hi+1:].max() if k_hi + 1 < len(ac) else 0.0)
                else:
                    max_out = ac[k_hi+1:].max() if k_hi + 1 < len(ac) else 0.0

                # Max inside affected range (linear part)
                inside_vals = ac[k_lo:k_hi+1] + 2.0 * dd * a_slice
                max_inside = inside_vals.max()

                # Handle k=2i quadratic term
                k_2i = 2 * i
                if k_lo <= k_2i <= k_hi:
                    idx_in_range = k_2i - k_lo
                    inside_vals[idx_in_range] += dd * dd  # Add quadratic
                    max_inside = max(max_inside, inside_vals[idx_in_range])

                M_trial = max(max_out, max_inside)
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_delta = dd

            # Refine with finer grid around best
            if abs(best_delta) > 1e-13:
                span = val * 0.1
                lo = max(best_delta - span, -val)
                hi = best_delta + span
                delta_fine = np.linspace(lo, hi, 15)
                for dd in delta_fine:
                    if dd < -val:
                        continue
                    if k_lo > 0:
                        max_before = ac[0:k_lo]
                        max_out = max(max_before.max() if len(max_before) > 0 else 0.0,
                                      ac[k_hi+1:].max() if k_hi + 1 < len(ac) else 0.0)
                    else:
                        max_out = ac[k_hi+1:].max() if k_hi + 1 < len(ac) else 0.0
                    inside_vals = ac[k_lo:k_hi+1] + 2.0 * dd * a_slice
                    max_inside = inside_vals.max()
                    k_2i = 2 * i
                    if k_lo <= k_2i <= k_hi:
                        idx_in_range = k_2i - k_lo
                        inside_vals[idx_in_range] += dd * dd
                        max_inside = max(max_inside, inside_vals[idx_in_range])
                    M_trial = max(max_out, max_inside)
                    if M_trial < best_trial_M:
                        best_trial_M = M_trial
                        best_delta = dd

            if abs(best_delta) > val * 1e-3:
                new_val = val + best_delta
                if new_val < 0:
                    new_val = 0.0
                a[i] = new_val
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = val
                    continue
                a = a / s
                ac = fast_autocorr(a)
                M_new = ac.max()
                if M_new < best_m - 1e-14:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            # Escape: random mass redistribution
            n_pert = max(5, n // 100)
            for _ in range(n_pert):

                q = int(rng.integers(n))
                if p != q and a[p] > 1e-10:
                    amt = a[p] * rng.uniform(0.03, 0.12)
                    a[p] -= amt
                    a[q] += amt
                    a = np.maximum(a, 0.0)

                        a[q] -= amt
        else:
            stall = 0

        if verbose and (stall == 0 or stall % 3 == 0):

def spectral_flatten(a0, time_limit, verbose=True, rng_seed=77):
    """
    Iterative spectral flattening to reduce autocorrelation peaks.
    Makes |FFT(a)|^2 more uniform, which spreads out the autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    alpha = 0.4
    n_iter = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 2000:
            break

        A = np.fft.rfft(a, 2 * n - 1)
        P = np.abs(A) ** 2

        # Target: flatten the spectrum
        P_mean = P.sum() / len(P)
        ratio = P_mean / np.maximum(P, 1e-30)

        # Soft clipping of the ratio
        ratio_clipped = np.minimum(ratio, 5.0)
        ratio_soft = ratio_clipped ** alpha

        A_new = A * ratio_soft
        # Preserve DC component (keeps sum(a) constant)
        A_new[0] = A[0]

        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Blend
        a = (1.0 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_iter += 1

        # Periodic noise injection to escape plateaus
        if n_iter % 500 == 0 and no_improve > 200:
            noise = rng.normal(0, 0.008, n) * a
            a = np.maximum(a + noise, 0)
            a = a / a.sum()

        if verbose and n_iter % 200 == 0:

            J = 2.0 * n * best_m

def fista_with_coord(a0, time_limit, verbose=True, rng_seed=42):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    tau = 0.1
    c0 = 2.5
    n_total = 0
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit:
        # Inner loop: FISTA steps
        n_fista = 500
        for _ in range(n_fista):
            if time.monotonic() - t0 > time_limit:
                break
            progress = min(1.0, n_total / 20000.0)
            tau = 0.15 * (0.008 / 0.15) ** progress
            top_k = max(10, int(60 - 40 * progress))

            beta_t = n_total / (n_total + 1.5) if n_total > 0 else 0.0
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            ac = fast_autocorr(y)
            M = ac.max()
            if M < 1e-30:
                break

            g = softmax_subgradient(y, ac, top_k=top_k, tau=tau)
            eta = c0 / (np.sqrt(n_total + 1) + 1) * np.sqrt(n)

            log_a = np.log(np.maximum(y, 1e-15)) - eta * g
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s

            x_prev = x
            x = a_new
            n_total += 1

        # Check improvement and do coordinate descent
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()

        # Coordinate descent polish (short)
        remaining_cd = min(2.0, (time_limit - (time.monotonic() - t0)) * 0.1)
        if remaining_cd > 0.5:
            x_cd, best_a_cd, best_m_cd = coord_descent_quick(x, best_a, best_m, n_steps=50, rng=rng)
            if best_m_cd < best_m:
                best_m = best_m_cd
                x = norm_sum1(x_cd)
                stall = 0
            else:
                stall += 1000  # Count as partial stall

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.004, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c0 = min(c0 * 1.2, 12.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    FC restart ({elapsed:.1f}s): J={2*n*best_m:.6f}, c0={c0:.2f}")

        if verbose and n_total % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    F+Coord n={n_total} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def coord_descent_quick(a, best_a, best_m, n_steps=100, rng=None):
    """Quick coordinate descent with simple grid search."""
    n = len(a)
    a = norm_sum1(a)
    ac = fast_autocorr(a)
    if rng is None:
        rng = np.random.default_rng()
    idx_order = rng.permutation(n)

    for step in range(min(n_steps, n)):
        i = idx_order[step % n]
        val = a[i]
        if val < 1e-13:
            continue

        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        if k_lo > k_hi:
            continue
        a_slice = a[0:k_hi - k_lo + 1]

        best_delta = 0.0
        best_M = ac.max()

        for dd in np.linspace(-val * 0.7, val * 0.5, 15):
            if k_lo > 0:
                max_before = ac[0:k_lo].max() if k_lo > 0 else 0.0
            else:
                max_before = 0.0
            max_after = ac[k_hi+1:].max() if k_hi + 1 < len(ac) else 0.0
            max_out = max(max_before, max_after)
            inside = ac[k_lo:k_hi+1] + 2.0 * dd * a_slice
            max_inside = inside.max()
            k_2i = 2 * i
            if k_lo <= k_2i <= k_hi:
                inside[k_2i - k_lo] += dd * dd
                max_inside = max(max_inside, inside[k_2i - k_lo])
            M_trial = max(max_out, max_inside)
            if M_trial < best_M:
                best_M = M_trial
                best_delta = dd

        if abs(best_delta) > val * 0.01:
            a[i] = max(0.0, val + best_delta)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
            else:
                a[i] = val

    return a, best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining adaptive FISTA, efficient coordinate descent,
    and spectral flattening in a multi-phase strategy.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.05 * np.cos(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Adaptive FISTA (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adaptive FISTA n={n0} ===")
    t1 = min(remaining * 0.20, 280.0)
    if t1 > 30:
        a_out, _ = adaptive_fista(norm_sum1(best_seq), t1, verbose=True, rng_seed=1)
        try_update(a_out, f"afista1_n{n0}")

    # === Phase 2: Efficient coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coord Desc n={n_best} ===")
    t2 = min(remaining * 0.18, 200.0)
    if t2 > 20:
        a_out, _ = coord_descent_efficient(norm_sum1(best_seq), t2, verbose=True, rng_seed=2)
        try_update(a_out, f"coord1_n{n_best}")

    # === Phase 3: FISTA + Coord combined ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: FISTA+Coord n={n_best} ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 30:
        a_out, _ = fista_with_coord(norm_sum1(best_seq), t3, verbose=True, rng_seed=3)
        try_update(a_out, f"fista_coord_n{n_best}")

    # === Phase 4: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Spectral n={n_best} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = spectral_flatten(norm_sum1(best_seq), t4, verbose=True, rng_seed=4)
        try_update(a_out, f"spectral_n{n_best}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Polish n={n_best} ===")
    t5 = min(remaining * 0.12, 160.0)
    if t5 > 20:
        a_out, _ = coord_descent_efficient(norm_sum1(best_seq), t5, verbose=True, rng_seed=5)
        try_update(a_out, f"coord2_n{n_best}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 6: Multi-length ===")
        for dn in [300, -300, 500, -500]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            nl = n_best + dn
            if 800 <= nl <= 4000:
                print(f"  Trying n={nl}")
                a_interp = interpolate_sequence(best_seq, nl)
                t_sub = min(remaining * 0.30, 120.0)
                if t_sub > 20:
                    a_out, _ = adaptive_fista(norm_sum1(a_interp), t_sub, verbose=False, rng_seed=nl % 1000)
                    try_update(a_out, f"ml_n{nl}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish n={n_best} ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 20:
        a_out, _ = fista_with_coord(norm_sum1(best_seq), t7, verbose=True, rng_seed=7)
        try_update(a_out, f"final_fc_n{n_best}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.35, 150.0)
    if t8 > 15:
        print(f"\n=== Phase 8: Final Coord n={n_best} ===")
        a_out, _ = coord_descent_efficient(norm_sum1(best_seq), t8, verbose=True, rng_seed=8)
        try_update(a_out, f"final_coord_n{n_best}")

    # === Phase 9: Last FISTA sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.40, 100.0)
    if t9 > 15:
        print(f"\n=== Phase 9: Last FISTA n={n_best} ===")
        a_out, _ = adaptive_fista(norm_sum1(best_seq), t9, verbose=False, rng_seed=9)
        try_update(a_out, f"last_fista_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure output is valid
    final = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    s = final.sum()
    if s < 0.01:
        return [1.0]
    final = final / s
    return [float(x) for x in final]
