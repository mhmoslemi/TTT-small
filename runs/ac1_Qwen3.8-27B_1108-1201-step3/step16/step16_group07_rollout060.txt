"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based weighted subgradient with Nesterov acceleration
2. Spectral smoothing (low-pass filtering in frequency domain)
3. LP best-response polish
4. Coordinate descent micro-optimization
Key improvement: O(n log n) subgradient computation via FFT cross-correlation

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    """Compute full linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient_fast(a, w_seq, active_set):
    """
    Compute subgradient of sum_k w_k * autocorr[k] w.r.t. a.
    sg[j] = 2 * sum_k w_k * a[k-j] (valid range)
    
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    
    # Build W: W[k] = w_k for k in active_set
    W = np.zeros(fft_size)
    for idx, k in enumerate(active_set):
        W[k] = w_seq[idx]
    
    # Zero-pad a
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    
    # Cross-correlation: T[j] = sum_m W[m+j] * a[m]
    # FFT(T) = conj(FFT(W)) * FFT(a)
    FW = rfft(W, fft_size)
    FA = rfft(a_pad, fft_size)
    
    sg = 2.0 * irfft(np.conj(FW) * FA, fft_size)[:n]
    return sg

def normalize_to_sum(a, target_sum):
    """Normalize a to have the given sum, maintaining non-negativity."""
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def spectral_smooth(a, alpha):
    """Apply low-pass smoothing in frequency domain. alpha in [0,1], 0=no smoothing, 1=full.
    Actually let's define it as: a_smooth = (1-alpha)*a + alpha*lowpass(a)"""
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    
    # Simple low-pass: attenuate high frequencies
    freqs = np.arange(len(A))
    # Keep low frequencies, attenuate high ones
    # Use a gentle roll-off
    cutoff = len(A) * 0.3
    attenuation = np.ones(len(A))
    mask = freqs > cutoff
    attenuation[mask] = np.exp(-2.0 * (freqs[mask] - cutoff) / cutoff)
    
    A_smooth = A * attenuation
    a_smooth = irfft(A_smooth, fft_size)[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    
    return (1 - alpha) * a + alpha * a_smooth

def nesterov_subgradient(a0, time_limit, verbose=False, seed=0):
    """Core optimization: Nesterov-accelerated subgradient with adaptive peak weighting."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, S)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t_start = time.monotonic()
    
    # Nesterov state
    y = a.copy()
    y_prev = a.copy()
    
    # Step size
    step_base = 2.0 / np.sqrt(n)
    step = step_base
    
    # Smoothing schedule
    smooth_alpha = 0.0  # Start with no smoothing
    
    patience = 0
    max_patience = 1000
    iter_count = 0
    
    while time.monotonic() - t_start < time_limit:
        iter_count += 1
        
        # Compute autocorrelation of y
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-10:
            break
        
        # Adaptive peak weighting: softmax with temperature
        # Temperature decreases over time (more focused on top peaks)
        progress = min(1.0, iter_count / 500.0)
        temp = 0.1 * (1.0 - progress) + 0.01 * progress
        
        # Weight peaks: w_k = exp((conv[k] - M) / (M * temp))
        # This gives more weight to peaks near the maximum
        weights = np.exp((conv_y - M) / (M * temp))
        
        # Only use peaks above a threshold to keep active set small
        threshold = M * (1.0 - 5.0 * temp)
        active_mask = conv_y >= threshold
        if active_mask.sum() < 3:
            # Always include top few
            n_top = max(3, min(n // 20, 50))
            top_idx = np.argsort(conv_y)[-n_top:]
            active_mask[top_idx] = True
        
        # Cap active set size for efficiency
        n_active = int(active_mask.sum())
        if n_active > 200:
            # Keep only top 200
            active_idx = np.where(active_mask)[0]
            conv_at_active = conv_y[active_idx]
            top_in_active = np.argsort(conv_at_active)[-200:]
            active_set = active_idx[top_in_active]
            w_seq = weights[active_set]
        else:
            active_set = np.where(active_mask)[0]
            w_seq = weights[active_set]
        
        # Normalize weights
        w_sum = w_seq.sum()
        if w_sum < 1e-15:
            w_seq = np.ones(len(w_seq)) / len(w_seq)
        else:
            w_seq = w_seq / w_sum
        
        # Compute subgradient
        sg = compute_subgradient_fast(y, w_seq, active_set)
        
        # Nesterov update
        t_iter = float(iter_count)
        alpha = 2.0 / (t_iter + 2.0)
        
        # Gradient step on y
        y_step = y - step * sg / n
        y_step = np.maximum(y_step, 0.0)
        y_step = normalize_to_sum(y_step, S)
        
        # Momentum
        if iter_count == 1:
            a_new = y_step
        else:
            momentum = min((iter_count - 1.0) / (iter_count + 2.0), 0.9)
            a_new = y_step + momentum * (y_step - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, S)
        
        y_prev = y
        y = y_step
        
        # Periodic spectral smoothing
        if iter_count % 100 == 0:
            smooth_alpha = min(0.3, 0.1 * progress)
            y = spectral_smooth(y, smooth_alpha)
            y = normalize_to_sum(y, S)
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
            step = min(step * 1.01, 2.0 * step_base)
        else:
            patience += 1
            if patience % 50 == 0:
                step *= 0.95
            
            # Check a_new too
            score_a = fast_autocorr(a_new).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a_new.copy()
                patience = 0
        
        # Restart if stalled
        if patience > max_patience:
            y = best_a.copy()
            y_prev = best_a.copy()
            a_new = best_a.copy()
            patience = 0
            step = step_base * (0.5 + 0.5 * rng.random())
            
            # Add small perturbation
            perturbation = 0.01 * rng.standardnormal(n)
            y = y + perturbation
            y = np.maximum(y, 0.0)
            y = normalize_to_sum(y, S)
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Subgrad iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.4f}")
    
    return best_a, best_score

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    
    # Build constraint matrix T where T[i, j] = a_fixed[k_i - j] if valid
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=False, seed=0):
    """LP best-response with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, S)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    
    t_start = time.monotonic()
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t_start < time_limit:
        iter_count += 1
        
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        
        # Active set: peaks above threshold
        tol = 0.03 + 0.02 * (iter_count % 10) / 10.0
        active_set = np.where(conv_y >= M * (1.0 - tol))[0]
        n_top = max(5, n // 50)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))
        
        if len(active_set) > 400:
            conv_at_active = conv_y[active_set]
            keep = np.argsort(conv_at_active)[-400:]
            active_set = active_set[keep]
        
        b = lp_best_response(y, M, active_set, solver)
        if b is None:
            patience += 1
            if patience > 100:
                break
            continue
        
        g = normalize_to_sum(b, S)
        
        # Nesterov update
        t_iter = float(iter_count)
        alpha = 2.0 / (t_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to_sum(y_new, S)
        
        if iter_count == 1:
            a_new = g.copy()
        else:
            momentum = min((iter_count - 1.0) / (iter_count + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, S)
        
        y_prev = y
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0
        
        if patience > 200:
            y = best_a.copy()
            y_prev = best_a.copy()
            patience = 0
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent for final polishing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, S)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t_start = time.monotonic()
    no_improve = 0
    max_no_improve = 5
    n_passes = 0
    
    while time.monotonic() - t_start < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t_start > time_limit:
                break
            
            aj = a[j]
            
            # Try several candidate values
            typical = S / n
            candidates = [0.0, aj * 0.5, aj * 1.5, typical * 0.5, typical, typical * 2.0]
            candidates = sorted(set([max(0.0, c) for c in candidates]))
            
            best_x = aj
            best_obj = conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                
                s_others = S - aj
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                
                # Update autocorrelation efficiently
                # autocorr[k] changes by 2*(x-aj)*a[k-j] for k in [j, j+n-1]
                # Plus correction at k=2j: add (x-aj)^2
                delta = x - aj
                new_conv = conv.copy()
                
                lo = j
                hi = min(2 * n - 1, j + n)
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < 2 * n - 1:
                    new_conv[2 * j] += delta * delta
                
                # Scale to maintain sum = S
                scale = S / new_sum
                new_max = new_conv.max() * scale * scale
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                a = normalize_to_sum(a, S)
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a.copy()
                    conv = new_conv * (S / (S - aj + best_x)) ** 2
                    conv = fast_autocorr(a)  # Recompute to be safe
                    improved = True
                else:
                    a = best_a.copy()
                    conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def fast_autocorr(a):
    """Compute full linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient_fft(a, W, fft_size):
    """
    Compute subgradient: sg[j] = 2 * sum_k W[k] * a[k-j] (valid range).
    W is a pre-built weight sequence of length fft_size (zero-padded).
    Uses FFT cross-correlation: T = W cross_corr a, FFT(T) = conj(FFT(W)) * FFT(a).
    """
    n = len(a)
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    FW = rfft(W, fft_size)
    FA = rfft(a_pad, fft_size)
    sg = 2.0 * irfft(np.conj(FW) * FA, fft_size)[:n]
    return sg


def normalize_sum(a, target):
    """Normalize a to have the target sum, maintaining non-negativity."""
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / s)


def nesterov_logsumexp(a0, time_limit, verbose=False, seed=0):
    """
    Core optimization: minimize log-sum-exp(beta * autocorr) via Nesterov-accelerated
    subgradient descent. beta is the inverse temperature (controls peak focus).

    n = len(a0)
    S = np.sqrt(2.0 * n)
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, S)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    # Nesterov state
    y = a.copy()
    y_prev = a.copy()

    # Ergodic average
    ergodic = a.copy()
    ergodic_count = 0

    t_start = time.monotonic()
    iter_count = 0
    patience = 0
    max_patience = 800

    # Temperature schedule: start broad (low beta), focus narrow (high beta)
    # beta = 1/temp
    temp_start = 0.15
    temp_end = 0.005
    warmup_frac = 0.3

    # Step size
    step_base = 1.5 / np.sqrt(n)
    step = step_base

    while time.monotonic() - t_start < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t_start

        # Adaptive temperature
        progress = min(1.0, elapsed / time_limit)
        if progress < warmup_frac:
            # Broad phase: explore all peaks
            temp = temp_start - (temp_start - temp_end * 3) * (progress / warmup_frac)
        else:
            # Focused phase: concentrate on top peaks
            frac = (progress - warmup_frac) / (1.0 - warmup_frac)
            temp = max(temp_end, 0.045 - 0.04 * frac)

        beta = 1.0 / temp

        # Compute autocorrelation
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-10:
            break

        # Compute softmax weights: w_k = exp(beta * (conv[k] - M)) / sum
        # This is the subgradient direction for log-sum-exp
        logits = beta * (conv_y - M)
        # Clip for numerical stability
        logits = np.maximum(logits, -50.0)
        exp_logits = np.exp(logits)
        w_total = exp_logits.sum()
        if w_total < 1e-15:
            w_total = 1.0
        weights = exp_logits / w_total

        # Build W sequence for FFT subgradient computation
        W = np.zeros(fft_size)
        W[:conv_size] = weights
        FW = rfft(W, fft_size)

        # Compute subgradient
        a_pad = np.zeros(fft_size)
        a_pad[:n] = y
        FA = rfft(a_pad, fft_size)
        sg = 2.0 * irfft(np.conj(FW) * FA, fft_size)[:n]

        # Nesterov update
        t_iter = float(iter_count)
        momentum = min((t_iter - 1.0) / (t_iter + 2.0), 0.92) if t_iter > 1 else 0.0

        y_new = y - step * sg
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, S)

        if iter_count <= 2:
            a_new = y_new
        else:
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, S)

        y_prev = y
        y = y_new

        # Update ergodic average (polyak averaging)
        ergodic_count += 1
        alpha_erg = 2.0 / (ergodic_count + 2.0)
        ergodic = (1.0 - alpha_erg) * ergodic + alpha_erg * a_new

        # Track best (check y, a_new, and ergodic)
        score_y = conv_y.max()  # Already computed
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
            step = min(step * 1.005, 3.0 * step_base)
        else:
            patience += 1

        score_a = fast_autocorr(a_new).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            patience = 0

        # Check ergodic every 20 iterations
        ergodic = np.maximum(ergodic, 0.0)

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=False, seed=0):
    """LP best-response with Nesterov acceleration and progressive active-set tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    y_prev = a.copy()
    ergodic = a.copy()
    ergodic_count = 0

    t_start = time.monotonic()
    patience = 0
    iter_count = 0

    while time.monotonic() - t_start < time_limit:
        iter_count += 1

        conv_y = fast_autocorr(y)
        M = conv_y.max()

        # Progressive active set
        progress = min(1.0, iter_count / 500.0)
        tol = 0.06 * (1.0 - progress) + 0.008 * progress
        active_set = np.where(conv_y >= M * (1.0 - tol))[0]
        n_top = max(5, n // 60)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))

        if len(active_set) > 300:
            conv_at_active = conv_y[active_set]
            keep = np.argsort(conv_at_active)[-300:]
            active_set = active_set[keep]

        b = lp_best_response(y, M, active_set, solver)
        if b is None:

                b = lp_best_response(y, M, active_set, solver)
            if b is None:
                patience += 1
                if patience > 50:
                    break
                continue

        g = normalize_sum(b, S)
        iter_count_lp = iter_count

        # Nesterov
        t_iter = float(iter_count_lp)
        alpha = 2.0 / (t_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_sum(y_new, S)

        if iter_count_lp <= 2:
            a_new = g.copy()
        else:

            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_sum(a_new, S)

        y_prev = y
        y = y_new

        # Ergodic
        ergodic_count += 1
        alpha_erg = 2.0 / (ergodic_count + 2.0)
        ergodic = (1.0 - alpha_erg) * ergodic + alpha_erg * a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        score_a = fast_autocorr(a_new).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            patience = 0

        if iter_count_lp % 30 == 0:
            ergodic = np.maximum(ergodic, 0.0)
            ergodic = normalize_sum(ergodic, S)
            score_erg = fast_autocorr(ergodic).max()
            if score_erg < best_score:
                best_score = score_erg
                best_a = ergodic.copy()
                patience = 0

        if patience > 300:
            y = best_a.copy()
            y_prev = best_a.copy()
            patience = 0
            active_set = None

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}")

    n_passes = 0
    conv_size = 2 * n - 1

    while time.monotonic() - t_start < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t_start > time_limit:
                break

            aj = a[j]
            s_others = S - aj

            # Generate candidate values
            typical = S / n
            candidates = [0.0, aj * 0.3, aj * 0.7, aj * 1.3, aj * 1.7,
                          typical * 0.3, typical * 0.7, typical * 1.3, typical * 2.0]
            if aj > 0:
                candidates.append(aj * (1.0 + 0.5 * rng.standard_normal()))
            candidates = sorted(set([max(0.0, min(c, S * 0.3)) for c in candidates]))
            candidates = [c for c in candidates if abs(c - aj) > 1e-10]

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                delta = x - aj
                new_conv = conv.copy()

                lo = j
                hi = min(conv_size, j + n)
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < conv_size:
                    new_conv[2 * j] += delta * delta

                scale = S / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                a = normalize_sum(a, S)
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    conv = fast_autocorr(a)
                    improved = True
                else:
                    a = best_a.copy()
                    conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def dual_guided_polish(a0, time_limit, solver, verbose=False):
    """Use LP dual variables to guide subgradient updates."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_sum(a, S)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t_start = time.monotonic()
    patience = 0
    max_patience = 200
    step = 0.3 / n ** 0.3

    for t in range(1, 2000000):
        if time.monotonic() - t_start > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        # Active set
        tol = 0.04
        active_set = np.where(conv >= M * (1.0 - tol))[0]
        if len(active_set) < 5:
            active_set = np.argsort(conv)[-max(5, n // 50):]

        # Solve LP to get duals
        b, duals = None, None
        try:
            b = lp_best_response(a, M, active_set, solver)
        except Exception:
            pass

        if b is None:
            patience += 1
            continue

        # Use duals for weighting, or fallback to exp weighting
        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(np.asarray(duals, dtype=np.float64), 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-10.0 * (M - conv[active_set]) / max(M * 0.02, 1e-10))
            weights = weights / weights.sum()

        # Compute weighted subgradient via FFT
        W = np.zeros(fft_size)
        for idx, k in enumerate(active_set):
            W[k] = weights[idx]
        FW = rfft(W, fft_size)
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        FA = rfft(a_pad, fft_size)
        grad = 2.0 * irfft(np.conj(FW) * FA, fft_size)[:n]

        # Project gradient: subtract mean on support to maintain sum constraint
        support = np.where(a > 0.001 * a.max())[0]
        if len(support) >= 2:
            grad_mean = np.mean(grad[support])
            grad[support] -= grad_mean

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = normalize_sum(a_trial, S)

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 2.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-12:
                break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Dual t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate():

    Main search: log-sum-exp subgradient + LP polish + dual-guided + coordinate descent.
    Uses FFT-based O(n log n) subgradient computation for speed.
    """

    time_budget = 1770.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load failed: {e}")

    if not np.isfinite(best_score):
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x + 0.7)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Log-sum-exp Nesterov (main, ~40% time) ===
