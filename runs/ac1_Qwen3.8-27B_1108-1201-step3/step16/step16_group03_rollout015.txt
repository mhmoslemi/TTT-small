import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))

        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def norm_diff(v):
    """Project v to have zero sum (for direction vectors on the simplex)."""
    return v - v.mean()

def subgradient_at_peak(a, k):
    """Subgradient of (a*a)[k] w.r.t. a, projected to zero-sum."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return norm_diff(grad)

def momentum_subgradient(a, k):
    """Subgradient of (a*a)[k] w.r.t. a, projected to zero-sum."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return norm_diff(grad)

def simplex_project(v, lower=None):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if lower is None:
        lower = 0.0
    # Standard simplex projection
    mu = np.sort(v)[::-1]
   .cumsum_mu = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n+1) > (.cumsum_mu - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (.cumsum_mu[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, lower)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s

def subgradient_descent(a0, time_limit, verbose=True, top_k=3, lr_init=0.01, momentum=0.9, seed=42):

    lr = lr_init
    stall = 0
    max_stall = 500
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M_curr = ac.max()
        
        # Find top peaks
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        
        # Average subgradient of top peaks
        g = np.zeros(n)
        for k in top_idx:
            g += subgradient_at_peak(a, int(k))

        # Momentum update
        v = momentum * v + g

        if M_new < best_m - 1e-14:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

def zero_sum(v):

def peak_subgradient(a, k):
    """Subgradient of (a*a)[k] w.r.t. a (non-projected)."""
    n = len(a)

    for i in range(i_start, i_end):

def simplex_project(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find rho: largest j such that v_sorted[j] - (cumsum[j]-1)/j > 0
    idx = np.arange(1, n + 1)
    cond = v_sorted * idx > (cumsum - 1.0)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cumsum[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s

        g = np.maximum(g, 0.0)  # For mirror descent, use positive part

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart
        if stall > max_stall:
            pert = rng.normal(0, 0.005, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            stall = 0
            current_top_k = min(current_top_k + 1, 10)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}, stall={stall}")

    return best_a, best_m


def peak_cd_optimize(a0, time_limit, verbose=True, seed=42, max_stall=500):
    """
    Peak-equalization coordinate descent:
    Targets the highest peak and transfers mass from contributing
    positions to safe (non-peak) positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))

        # Contributing pairs for the peak
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_star - idx]

        if contrib.max() < 1e-18:
            stall += 1
            continue

        # Build safe position mask (not in top-5 peaks)
        top_peaks = np.argsort(ac)[::-1][:5]
        in_peak = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            s2 = max(0, int(k2) - n + 1)
            e2 = min(n, int(k2) + 1)
            in_peak[s2:e2] = True
        safe_pos = np.where(~in_peak)[0]
        if len(safe_pos) < 5:
            safe_pos = np.arange(n)

        improved = False
        # Sort contributing positions by contribution (try largest first)
        contrib_order = np.argsort(contrib)[::-1]
        n_try_src = min(len(idx), 15)
        n_try_dst = min(len(safe_pos), 30)

        for ci in range(n_try_src):
            src = int(idx[contrib_order[ci]])
            if a[src] < 1e-12:
                continue
            for di in range(n_try_dst):
                dst = int(safe_pos[di])
                if dst == src:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.4, 0.7]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random escape
            if stall > 50:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.3)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t < 1e-15:
                        break
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Peak CD ({elapsed:.1f}s): J={J:.6f}, stalls={stall}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Resample sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, kind, seed=0):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if kind == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'gaussian':
        a = np.exp(-20 * (x - 0.5) ** 2)
    elif kind == 'bimodal':
        a = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2)
    elif kind == 'triangle':
        a = 1.0 - np.abs(2 * x - 1)
    elif kind == 'random':
        a = rng.random(n)
    elif kind == 'random_smooth':
        a = rng.random(n)
        # Smooth with a box filter
        k = max(3, n // 200)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
    elif kind == 'flat':
        a = np.ones(n)
    else:
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
