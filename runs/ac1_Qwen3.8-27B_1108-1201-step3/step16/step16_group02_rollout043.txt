    return best_a, best_maxD

def lp_frank_wolfe_focused(a0, time_limit, solver, max_constraints=300, verbose=True):
    """LP Frank-Wolfe with focused constraint set."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Select active constraints
        threshold = M_curr * 0.97
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]
        
        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = new_sx = x_new / sx
        
        y = y_new
        x_prev = x_new
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_conv(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, active={len(active_lags)}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Fast coordinate descent polish using pairs of moves."""
    n = len(a0)
    a = normalize_nonneg(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 20
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(100, n)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            for direction in [1, -1]:
                src, dst = (i, j) if direction == 1 else (j, i)
                if a[src] < 1e-12:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.02
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_conv(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        
        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def chirp_init(n, freq=1.0):
    """Generate a chirp-based initialization with good autocorrelation properties."""
    i = np.arange(n, dtype=np.float64)
    # Quadratic phase chirp - creates spreading in frequency domain
    phase = np.pi * freq * i * i / n
    a = 1.0 + 0.4 * np.cos(phase) + 0.2 * np.cos(2.0 * np.pi * 3.0 * i / n)
    a = np.maximum(a, 0.05)
    return a / a.sum()

def random_flat_init(n, rng_seed=123):
    """Generate random initialization with approximately flat autocorrelation via spectral method."""
    rng = np.random.default_rng(rng_seed)
    a = rng.random(n)
    a = a / a.sum()
    
    # Iteratively flatten autocorrelation in spectral domain
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(50):
        A = np.fft.rfft(a, fft_size)
        Ac = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        # We want Ac to be flat; adjust |A| in frequency domain
        # Normalize: target = 1/(2n-1)
        target = 1.0 / (2 * n - 1)
        # In frequency domain, |A|^2 relates to the DFT of Ac
        # Adjust: scale |A| to reduce peaks in Ac
        Ac_shifted = Ac - target
        Ac_fd = np.fft.rfft(Ac_shifted, fft_size)
        # Gradient descent step in frequency domain
        A -= 0.1 * Ac_fd / np.abs(Ac_fd).max()
        a_new = np.fft.irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    
    return a

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, x)) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Try to load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = chirp_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default chirp init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Adam Nesterov with p=4 (main workhorse, 40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 700.0)
    print(f"\n=== Phase 1: AdamND p=4 n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = adam_nesterov_nd(a_in, t1, p_start=4.0, p_end=16.0,
                                     lr=0.02, p_transition_iters=20000, verbose=True)
        try_update(a_out, f"AdamND_p4_n{n0}")
    
    # === Phase 2: Adam Nesterov with p=8->32 (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 2: AdamND p=8->32 n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = adam_nesterov_nd(a_in, t2, p_start=8.0, p_end=64.0,
                                     lr=0.01, p_transition_iters=15000, verbose=True)
        try_update(a_out, f"AdamND_p8_64_n{len(best_seq)}")
    
    # === Phase 3: Try chirp initialization (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Chirp init + AdamND (budget={t3:.0f}s) ===")

    if best_seq is not None:

        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
