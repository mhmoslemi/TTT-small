import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def top_p_gradient(a, r, p):
    """Compute weighted average gradient of top-p autocorrelation peaks.
    
    Returns gradient vector g such that moving in -g direction reduces the top peaks.
    """
    n = len(a)
    top_idx = np.argsort(r)[::-1][:p]
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        L = max(0, k_int - n + 1)
        R = min(k_int, n - 1)
        if L > R:
            continue
        j = np.arange(L, R + 1)
        g[j] += 2.0 * a[k_int - j]
    g /= p
    return g


def weight_top_p_gradient(a, r, p, beta):
    """Compute softmax-weighted gradient of top-p autocorrelation peaks.
    
    Weights are proportional to exp(beta * r[k]) for the top-p lags.
    """
    n = len(a)
    top_idx = np.argsort(r)[::-1][:p]
    top_vals = r[top_idx]
    shifted = top_vals - top_vals[0]
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        L = max(0, k_int - n + 1)
        R = min(k_int, n - 1)
        if L > R:
            continue
        j = np.arange(L, R + 1)
        g[j] += 2.0 * w * a[k_int - j]
    return g
