def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        for c2 in [0.7, 0.8]:
            for sigma in [0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1)**2) / (2*sigma**2)) + np.exp(-((x - c2)**2) / (2*sigma**2))
                try_seq(a, f"bimodal_{c1}_{c2}_{sigma}")
    
    # 6. Sigmoid
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            a = 1.0 / (1.0 + np.exp(-s * (x - c)))
            try_seq(a, f"sigmoid_{c}_{s}")
    
    # 7. Sinusoidal
    for freq in [1, 2, 3]:
        for phase in [0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + phase * np.pi) ** 2, 0.01)
            try_seq(a, f"sin_{freq}_{phase}")
    
    # 8. Triangular
    for peak in [0.3, 0.5, 0.7]:
        a = 1.0 - np.abs(x - peak) / max(peak, 1-peak)
        a = np.maximum(a, 0.01)
        try_seq(a, f"tri_{peak}")
    
    # 9. Exponential
    for rate in [1, 2, 5, 10]:
        a = np.exp(-rate * x)
        try_seq(a, f"exp_{rate}")
        a = np.exp(rate * x)
        try_seq(a, f"exp_neg_{rate}")
    
    # 10. Blend with parent if available
    if best_init is not None and best_score_init is not None and best_score_init < 2.0:
        for w in [0.3, 0.5, 0.7]:
            # Blend with a power law
            a_power = (np.arange(1, n+1, dtype=float)) ** 1.0
            a_blend = w * best_init / best_init.sum() + (1-w) * a_power / a_power.sum()
            try_seq(a_blend, f"blend_w{w}")
    
    elapsed = time.monotonic() - t0
    print(f"  Parametric search done ({elapsed:.0f}s): best J={best_val:.6f}")
    return best_a, best_val


def lp_frank_wolfe_cutting(a0, time_limit, solver, verbose=True, initial_K=200, max_K=3000, add_per_iter=20):
    """LP-based Frank-Wolfe with cutting-plane constraint management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    
    # Initial constraint set: top-K by value
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    
    # Select initial constraints: those with highest values (most likely binding)
    all_indices = np.arange(len(ac_y))
    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update y (Nesterov)
        if t > 1:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
            
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
            y = y_new
            x = x_new
        else:
            # First iteration, just evaluate
            pass
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
        
        # Build constraint matrix for selected constraints
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            # Keep only the most binding ones
            margins = [M_curr - ac[k] for k in constraint_list]
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        T_sel = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                T_sel[jj, idx] = y[k_int - idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            constraint_list = constraint_list[:initial_K]  # Reduce constraints on failure
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Check for violated constraints and add them (cutting plane)
        # Compute full convolution y * g_dir
        conv_yc = fast_autocorr(y)  # This is y*y, not y*g_dir
        # Actually we need y * g_dir, not y * y
        # Let me compute it properly
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Y = np.fft.rfft(y, fft_size)
        G = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Y * G, fft_size)[:size]
        
        # Find constraints that are violated (conv_ys[k] > M_curr)
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            # Add the most violated constraints
            viol_margins = conv_ys[violations] - M_curr
            top_viol = violations[np.argsort(viol_margins)[::-1][:add_per_iter]]
            for k in top_viol:
                constraint_set.add(int(k))
        
        iter_count += 1
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW final ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent with adaptive restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent: move mass from high-contribution positions to low-impact ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        improved = False
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: minimize impact on peak
            # Score each position by how much it contributes to top lags
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            
            # Avoid positions already in peak regions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]
            
            if dst == src:
                continue
            
            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
