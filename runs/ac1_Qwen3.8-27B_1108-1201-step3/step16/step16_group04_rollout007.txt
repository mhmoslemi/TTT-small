import numpy as np
import time

def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def logsumexp(x, beta):
    """Compute log(sum(exp(beta*x)))/beta as a smooth approximation of max(x)."""
    m = x.max()
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def compute_gradient(a, weights):
    """
    Compute the gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    Result has length len(a).
    Uses FFT for efficiency.
    """
    n = len(a)
    # grad_i = 2 * sum_k w_k * a[k-i] (where k-i is valid)
    # This is 2 * cross_correlate(weights, a)[:n]
    size = 1 << (len(weights) + n - 1). bit_length()
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    xc = np.fft.irfft(np.conj(W) * A, size)
    return 2.0 * xc[:n]

             lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
             restart_every=5000, verbose=True):

    Adam optimizer with log-sum-exp smoothing for minimizing peak convolution.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    # Ensure sum = 1
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    n_iter = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        n_iter += 1
        
        # Adaptive beta: increase over time
        frac = min(1.0, t / (2 * restart_every))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        curr_val = ac.max()
        
        if curr_val < best_val:
            best_val = curr_val
            best_a = a.copy()

        ac_shifted = ac - ac.max()
        w = np.exp(beta * ac_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(ac)
            w[ac.argmax()] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = compute_gradient(a, w)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Step

        a_new = simplex_project(a_new)
        s = a_new.sum()
        if s < 1e-15:

        else:

            a_new = best_a * (1 + pert)
            a_new = simplex_project(a_new)
            s = a_new.sum()
            if s > 1e-15:

            t = 0  # Reset timestep for Adam bias correction

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fourier-based fast gradient with log-sum-exp smoothing
2. Adam optimizer with Nesterov momentum on the probability simplex
3. Adaptive beta annealing for progressive sharpening
4. Multi-restart with random perturbations
5. Spectral equalization initialization
6. Multi-length exploration
"""

import numpy as np
import time

def fast_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, w):
    """
    Compute gradient of sum_k w_k * conv(a,a)_k w.r.t. a.
    Result has length len(a).
    """
    n = len(a)
    total_len = len(w) + n - 1
    size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    xc = np.fft.irfft(np.conj(W) * A, size)
    return 2.0 * xc[:n]

def spectral_init(n, n_iter=30):
    """Spectral equalization to get a good starting point."""
    a = np.ones(n)
    a = a / a.sum()
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        P = P + 1e-15
        C = 1.0 / np.sqrt(P)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.7 * a + 0.3 * a_new
    return a

def cosine_window(n):
    """Raised cosine window, normalized to sum=1."""
    if n == 1:
        return np.array([1.0])
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    # This goes from 1 to 1... not what we want
    # Use Hann window: zero at endpoints
    w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    # Actually for n points, Hann is: 0.5*(1-cos(2*pi*i/(n-1))) for i=0..n-1
    # This is 0 at i=0 and i=n-1, peaks at i=(n-1)/2
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def adam_optimize(a0, time_limit, beta_start=5.0, beta_end=500.0,
                  lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                  nesterov=True, restart_interval=3000,
                  verbose=True, seed=42):
    """
    Adam (+ optional Nesterov) optimizer for minimizing peak of conv(a,a)
    on the probability simplex, using log-sum-exp smoothing.
    """
    n = len(a0)
    a = simplex_project(a0.astype(np.float64).copy())
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_val = fast_conv(a).max()
    J_best = 2.0 * n * best_val
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    it = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    best_beta = beta_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t += 1
        
        # Adaptive beta
        progress = min(1.0, it / (5.0 * restart_interval))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute autocorrelation
        ac = fast_conv(a)
        curr_peak = ac.max()
        J_curr = 2.0 * n * curr_peak
        if J_curr < J_best:
            J_best = J_curr
            best_val = curr_peak
            best_a = a.copy()
        
        # Compute smoothed weights
        ac_max = ac.max()
        # For numerical stability
        shifted = beta * (ac - ac_max)
        # Clip to prevent overflow
        shifted = np.clip(shifted, -100, 100)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(ac)
            w[np.argmax(ac)] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = compute_grad(a, w)
        
        # Nesterov lookahead (optional)
        if nesterov and t > 1:
            # y = a + momentum * (a - a_prev) ... but we don't track a_prev easily with Adam
            # Skip Nesterov for simplicity, Adam already has momentum
            pass
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        # Periodic restart

            # Try different perturbation strengths
            strength = rng.choice([0.005, 0.01, 0.02, 0.05])
            pert = rng.normal(0, strength, n)

            a_cand = simplex_project(a_cand)
            sc = a_cand.sum()
            if sc > 1e-15:
                a_cand = a_cand / sc
            J_cand = 2.0 * n * fast_conv(a_cand).max()
            if J_cand < J_best * 1.05:  # Accept if not too much worse
                a = a_cand
            # Reset moments
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam it={it} ({elapsed:.1f}s): J={J_best:.6f}, beta={beta:.1f}, peak={curr_peak:.8f}")
    
    return best_a, best_val

def evaluate_safe(a):
    """Safely evaluate a sequence."""

def normalize_nonneg(a):
