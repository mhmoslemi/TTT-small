            y = y / sy

        # Compute gradient at y
        c_y, grad_y, M_y = compute_conv_and_grad(y, beta, top_k)

        # Mirror descent update
        log_a = np.log(y) - eta * grad_y
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        M_new = fast_conv(a_new).max()

        # Adaptive step size: backtrack if worsening
        if M_new > M_y * 1.001 + 1e-15:
            eta = max(eta * 0.6, 0.01)
            a_new = y.copy()
            M_new = M_y
        elif stall > 500 and eta < 5.0:
            eta = min(eta * 1.05, 5.0)

        x_prev = x
        x = a_new

        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with perturbation
        if stall > max_stall:
            scale = 0.005 + 0.02 * (stall / max_stall)
            x = best_a + rng.normal(0, scale, n) * best_a
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
            eta = max(eta_init, eta * 0.8)

        # Periodic larger perturbation
        if t % 20000 == 0:
            x_pert = best_a + rng.normal(0, 0.03, n) * best_a
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            mp = fast_conv(x_pert).max()
            if mp < best_m:
                best_m = mp
                best_a = x_pert.copy()
                x = x_pert.copy()
                x_prev = x_pert.copy()
                stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    {label} t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, tk={top_k}, eta={eta:.3f}, stall={stall}")

    return best_a


def coord_descent(a0, time_limit, seed=42, verbose=True, label="CD"):
    """Coordinate descent: transfer mass from peak-contributing pairs to low-impact positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500
    iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iters += 1
        c = fast_conv(a)
        M = c.max()
        n_peaks = min(8, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        improved = False

        for k_idx in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k_idx)
            # Find contributing indices for this peak
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            partners = k - idx
            contrib = a[idx] * a[partners]
            if contrib.max() < 1e-20:
                continue

            # Source: index with max contribution to this peak
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Score all potential destinations
            # A destination j reduces peak k by 2*delta*a[k-j] and affects other peaks
            # We want j where a[k-j] is small (reduces this peak) AND j is in "quiet" region
            dest_scores = np.zeros(n)
            for j in range(n):
                # How much does a[j] increase reduce peak k?
                reduce_k = a[k - j] if 0 <= k - j < n else 0.0
                # Penalty: how much does increasing a[j] hurt other top peaks?
                penalty = 0.0
                for k2 in top_peaks[:4]:
                    k2i = int(k2)
                    if 0 <= k2i - j < n:
                        penalty += a[k2i - j]
                dest_scores[j] = reduce_k - 0.5 * penalty / max(n_peaks, 1)

            # Exclude source and positions already in peak regions
            dest_scores[src] = -np.inf
            valid_dests = np.argsort(dest_scores)[::-1][:min(30, n)]

            for dst in valid_dests:
                if time.monotonic() - t0 > time_limit:
                    break
                dst = int(dst)
                if dst == src:
                    continue

                # Try several transfer fractions
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if time.monotonic() - t0 > time_limit:
                break

        if not improved:
            stall += 1
            # Random kick to escape
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    a[i] -= a[i] * frac
                    a[j] += a[i] * frac + a[i] * frac  # rough
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = best_a.copy()
                        stall = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    {label} iter={iters} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, label="LP"):
    """LP-based Frank-Wolfe with line search."""
    if solver is None:
        return a0
    n = len(a0)
    if n > 2000:
        # Too large for dense LP matrix
        return a0

    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    y = a.copy()
    iters = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        iters += 1

        c = fast_conv(y)
        M_curr = c.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        candidates = [best_alpha, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.7, 0.6]
        best_trial_m = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()

        for alpha in candidates:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            tm = fast_conv(trial).max()
            if tm < best_trial_m:
                best_trial_m = tm
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy if sy > 1e-15 else y.copy()
        y = y_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    {label} iter={iters} ({elapsed:.0f}s): J={J:.6f}")

    return best_a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
