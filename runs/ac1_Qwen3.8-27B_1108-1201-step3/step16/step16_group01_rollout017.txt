"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smooth_subgradient(a, ac, top_k=50, temperature=0.05):
    """
    Compute subgradient using softmax weighting over top-k lags.
    This gives a smoother gradient than hard active set.
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weighting (temperature controls sharpness)
    scaled = top_vals / (top_vals.max() + 1e-30) / temperature
    w = np.exp(scaled - scaled.max())
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    
    return g

def exact_subgradient(a, ac, threshold_frac=0.99):
    """Compute exact subgradient from active set."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    return g

def combined_subgradient(a, ac, top_k=50, temperature=0.05, threshold=0.99, alpha=0.5):
    """Combine smooth and exact subgradients."""
    g_smooth = smooth_subgradient(a, ac, top_k=top_k, temperature=temperature)
    g_exact = exact_subgradient(a, ac, threshold_frac=threshold)
    
    # Normalize both to unit norm
    ns = np.linalg.norm(g_smooth)
    ne = np.linalg.norm(g_exact)
    if ns > 1e-30:
        g_smooth = g_smooth / ns
    if ne > 1e-30:
        g_exact = g_exact / ne
    
    g = alpha * g_smooth + (1.0 - alpha) * g_exact
    gn = np.linalg.norm(g)
    if gn > 1e-30:
        g = g / gn
    return g

def mirror_step(a, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g / (a + 1e-15)
    # Actually, for the simplex constraint, the mirror map is exp
    # The Bregman divergence induced by sum(x log x) gives multiplicative updates
    # But for simplicity, let's use the log-barrier approach
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def backtracking_mirrorsolve(a0, time_limit, c0=3.0, top_k=50, temperature=0.05, 
                              threshold=0.99, alpha=0.5, verbose=True, rng_seed=42,
                              max_iter=100000):
    """
    Accelerated mirror descent with backtracking line search.
    Uses combined (smooth + exact) subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_M = ac.max()
    score = 2.0 * n * best_M
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta0 = c0 * np.sqrt(n) / 10.0
    stall = 0
    max_stall = 2000
    
    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac_y = fast_autocorr(y)
        g = combined_subgradient(y, ac_y, top_k=top_k, temperature=temperature,
                                 threshold=threshold, alpha=alpha)
        
        # Adaptive step size with backtracking
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        # Try the step
        a_candidate = mirror_step(y, g, eta)
        M cand = fast_autocorr(a_candidate).max()
        
        # If it didn't improve, try smaller step
        M_y = ac_y.max()
        if M_cand > M_y:
            # Backtrack
            for _ in range(5):
                eta *= 0.5
                a_candidate = mirror_step(y, g, eta)
                M_cand = fast_autocorr(a_candidate).max()
                if M_cand <= M_y:
                    break
        
        x_prev = x
        x = a_candidate
        
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            score = 2.0 * n * best_M
            stall = 0
        else:
            stall += 1
        
        # Restart if stuck
        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.005, n) * best_a
            x_restarted = best_a + noise
            x_restarted = np.maximum(x_restarted, 1e-10)
            x = x_restarted / x_restarted.sum()
            x_prev = x.copy()
            max_stall = int(max_stall * 1.3)
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    BTS restart t={t} ({elapsed:.1f}s): score={score:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BTS t={t} ({elapsed:.1f}s): score={score:.6f}, M={best_M:.8f}")
    
    return best_a, 2.0 * n * best_M

def spectral_initializer(n, rng):
    """
    Initialize using spectral methods:
    Try to create a sequence with approximately flat power spectrum.
    """

    A_flat = np.ones(n) * (n ** 0.5)  # Target flat magnitude

        a = np.ones(n) / n
    return a

def random_initializer(n, rng):
    """Random positive initialization with various distributions."""
    method = rng.integers(4)
    if method == 0:

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""

        a_list = [float(x) for x in a]

        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smooth_max_grad(a, T, top_k=100):
    """
    Compute smooth approximation of max autocorrelation (log-sum-exp)
    and its gradient. As T increases, approximation becomes sharper.
    """
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]

    max_val = top_vals.max()
    shifted = T * (top_vals - max_val)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    log_sum_exp = max_val + np.log(np.sum(exp_shifted)) / T

    # Softmax weights
    w = exp_shifted / exp_shifted.sum()

    # Gradient of smooth max w.r.t. a
    g = np.zeros(n)

        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-10
    step = 0
    stall = 0
    max_stall = 3000

    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()

    for step in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal T
        frac = min(1.0, step / float(anneal_steps))
        T = T_start + (T_end - T_start) * frac * frac  # Quadratic annealing

        # Nesterov extrapolation
        beta_t = (step - 1.0) / (step + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute smooth max gradient at y
        obj, g = smooth_max_grad(y, T=T, top_k=top_k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Learning rate with decay
        lr = lr_scale * np.sqrt(n) / (np.sqrt(step) + 1.0)

        # Mirror descent step (multiplicative weights)
        # a_new[i] ∝ a[i] * exp(-lr * m_hat[i] / sqrt(v_hat[i]))
        update = m_hat / (np.sqrt(v_hat) + eps)
        log_y = np.log(np.maximum(y, 1e-15)) - lr * update
        max_log = log_y.max()
        a_new = np.exp(log_y - max_log)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stuck
        if stall > max_stall:
            stall = 0
            scale = 0.003 + 0.005 * (step / max_iter)
            noise = rng.normal(0, scale, n) * best_a
            x_restarted = best_a + noise
            x_restarted = np.maximum(x_restarted, 1e-12)
            x = x_restarted / x_restarted.sum()
            x_prev = x.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            max_stall = int(max_stall * 1.5)

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    ANNEALED restart step={step} ({elapsed:.1f}s): "
                      f"score={2*n*best_M:.6f}, T={T:.1f}")

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ANNEALED step={step} ({elapsed:.1f}s): "
                  f"score={2*n*best_M:.6f}, T={T:.1f}")

    return best_a, 2.0 * n * best_M


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    Aggressive coordinate descent using peak involvement scoring.
    Identifies positions that contribute most to peak lags and moves
    their mass to low-involvement positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100
    K = min(40, len(ac))

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-20:
            break
        top_k = np.argsort(ac)[::-1][:K]

        # Compute peak involvement for each position
        involvement = np.zeros(n)
        total_peak = ac[top_k].sum()
        if total_peak < 1e-30:
            break
        for k in top_k:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            weight = ac[k] / total_peak
            for i in range(len(idx)):
                involvement[idx[i]] += weight * contrib[i]
                if j_idx[i] != idx[i]:
                    involvement[j_idx[i]] += weight * contrib[i]

        # Identify high and low involvement positions
        if involvement.max() < 1e-30:
            stall += 1
            break
        p75 = np.percentile(involvement[involvement > 0], 75) if (involvement > 0).any() else 0
        p25 = np.percentile(involvement[involvement > 0], 25) if (involvement > 0).any() else 0

        high_pos = np.where(involvement > p75)[0]
        low_pos = np.where(involvement < p25)[0]

        if len(high_pos) == 0 or len(low_pos) == 0:
            stall += 1
            break

        improved = False
        rng.shuffle(high_pos)
        rng.shuffle(low_pos)

        n_high = min(len(high_pos), 25)
        n_low = min(len(low_pos), 60)

        for src in high_pos[:n_high]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            for dst in low_pos[:n_low]:
                if dst == src:
                    continue
                if a[dst] > a[src] * 1.5:
                    continue

                found = False
                for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Escape: batch random transfers
            n_esc = max(5, n // 80)
            for _ in range(n_esc):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD2 ({elapsed:.1f}s): score={2*n*best_M:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_M


def pairwise_refine(a0, time_limit, verbose=True, rng_seed=555):
    """
    Fine-grained pairwise refinement: for each peak-contributing pair,
    try all reasonable transfer fractions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 30
    K = min(15, len(ac))

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        if ac.max() < 1e-20:
            break
        top_k = np.argsort(ac)[::-1][:K]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]

            if contrib.max() < 1e-20:
                continue

            # Sort pairs by contribution (descending)
            pair_order = np.argsort(contrib)[::-1]
            n_pairs = min(len(pair_order), 20)

            for pi in range(n_pairs):
                if time.monotonic() - t0 > time_limit:
                    break
                pos = pair_order[pi]
                i = idx[pos]
                j = j_idx[pos]

                if i == j:
                    continue

                # Try moving from the larger to a random cold position
                if a[i] >= a[j]:
                    src = i
                else:
                    src = j

                if a[src] < 1e-10:
                    continue

                # Try several target positions
                targets = rng.choice(n, size=min(30, n), replace=False)
                for dst in targets:
                    if dst == src or a[dst] > a[src]:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        if amt < 1e-20:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_M - 1e-15:
                            a = a_trial
                            best_M = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.03, 0.1)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pairwise ({elapsed:.1f}s): score={2*n*best_M:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_M


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Primary: Annealed smooth-max mirror descent with Adam.
    Secondary: Multi-scale coordinate descent and pairwise refinement.
    Global: Multi-length exploration and diverse restarts.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.2 * np.cos(2 * np.pi * x) ** 2 + 0.05
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Annealed MD (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Annealed MD n={n0} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 50:
        a_out, sc = annealed_md(
            norm_sum1(best_seq), t1,
            T_start=3.0, T_end=100.0, anneal_steps=8000,
            lr_scale=0.04, top_k=100, verbose=True, rng_seed=1)
        try_update(a_out, f"annealed1_n{n0}")

    # === Phase 2: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coord descent n={n_best} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, sc = coordinate_descent_v2(norm_sum1(best_seq), t2, verbose=True, rng_seed=2)
        try_update(a_out, f"cd_n{n_best}")

    # === Phase 3: Annealed MD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Annealed MD (alt params) n={n_best} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_out, sc = annealed_md(
            norm_sum1(best_seq), t3,
            T_start=5.0, T_end=150.0, anneal_steps=5000,
            lr_scale=0.03, top_k=80, verbose=True, rng_seed=3)
        try_update(a_out, f"annealed3_n{n_best}")

    # === Phase 4: Pairwise refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Pairwise refine n={n_best} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 30:
        a_out, sc = pairwise_refine(norm_sum1(best_seq), t4, verbose=True, rng_seed=4)
        try_update(a_out, f"pairwise_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        print(f"\n=== Phase 5: Multi-length exploration ===")
        lengths_to_try = []
        for delta in [300, -300, 600, -600]:
            n_try = n_best + delta
            if 500 <= n_try <= 3500 and n_try != n_best:
                lengths_to_try.append(n_try)

        for n_try in lengths_to_try[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={n_try} (remaining={remaining:.0f}s)")
            a_interp = interpolate_sequence(best_seq, n_try)
            t5 = min(remaining * 0.30, 200.0)
            if t5 > 30:
                a_out, sc = annealed_md(
                    norm_sum1(a_interp), t5,
                    T_start=3.0, T_end=80.0, anneal_steps=6000,
                    lr_scale=0.04, top_k=100, verbose=False, rng_seed=5)
                try_update(a_out, f"annealed_n{n_try}")
                # Quick coord polish
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 20:
                    a_out2, sc2 = coordinate_descent_v2(
                        norm_sum1(a_out), min(remaining * 0.3, 60.0),
                        verbose=False, rng_seed=6)
                    try_update(a_out2, f"cd_n{n_try}")

    # === Phase 6: Final coordinate refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 6: Final coord n={n_best} ===")
        t6 = min(remaining * 0.30, 150.0)
        if t6 > 20:
            a_out, sc = coordinate_descent_v2(norm_sum1(best_seq), t6, verbose=True, rng_seed=7)
            try_update(a_out, f"cd_final_n{n_best}")

    # === Phase 7: Final annealed sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: MD sub-polish n={n_best} ===")
        t7 = min(remaining * 0.40, 120.0)
        if t7 > 15:
            a_out, sc = annealed_md(
                norm_sum1(best_seq), t7,
                T_start=50.0, T_end=200.0, anneal_steps=3000,
                lr_scale=0.02, top_k=50, verbose=False, rng_seed=8)
            try_update(a_out, f"md_polish_n{n_best}")

    # === Phase 8: Final pairwise ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.50, 80.0)
        if t8 > 10:
            a_out, sc = pairwise_refine(norm_sum1(best_seq), t8, verbose=False, rng_seed=9)
            try_update(a_out, f"pairwise_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
