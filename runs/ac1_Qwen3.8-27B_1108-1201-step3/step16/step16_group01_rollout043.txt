"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed Mirror Descent with adaptive log-sum-exp temperature
2. Cross-correlation via FFT for efficient gradient computation
3. Nesterov acceleration with backtracking step size
4. Targeted coordinate descent polishing
5. Multi-length exploration with interpolation

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def smooth_objective(a, T):
    """Log-sum-exp smoothed version of max(conv)."""
    conv = fast_autocorr(a)
    max_c = conv.max()
    return (1.0/T) * (max_c + np.log(np.sum(np.exp(T*(conv - max_c)))))

def smooth_gradient_fft(a, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*conv[k])) w.r.t. a.
    Result: g[m] = (2/T) * sum_k w[k]*a[k-m] (valid k-m)
    where w[k] = exp(T*conv[k]) / sum exp(T*conv[k])
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_c = np.exp(T * (conv - max_c))
    w = exp_c / exp_c.sum()
    
    # Cross-correlation: g[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # Use FFT for efficiency
    fft_size = 1 << (2 * (2*n - 1) - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # cross[m] = sum_j w[m+j] * a[j]
    # This is ifft(fft(w) * conj(fft(a))) with appropriate indexing
    # Actually: ifft(fft(w)*conj(fft(a)))[m] = sum_j w[j+m]*a[j] (circular, with padding)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    g = 2.0 * cross[:n] / T
    return g

def exact_subgradient_full(a, conv, threshold_frac=0.99):
    """Subgradient using lags within threshold of max."""
    n = len(a)
    M = conv.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(conv >= threshold_frac * M)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            g[j] += 2.0 * a[k_int - j]
    return g

def mirror_update_log(a, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smoothed_mirror_descent(a0, time_limit, T_start=5.0, T_end=500.0, 
                              c0=2.0, verbose=True, rng_seed=42,
                              max_iters=200000, backtracking=True):
    """
    Main optimizer: Smoothed Mirror Descent with adaptive temperature
    and Nesterov acceleration.
    """
    n = len(a0)
    a = norm_sum1(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_obj = smooth_objective(a, T_start)
    best_exact = 2.0 * n * fast_autocorr(a).max()
    
    # Temperature schedule
    total_iters = max_iters
    stall_count = 0
    max_stall = 3000
    eta = c0 * np.sqrt(n)
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(t / total_iters, 1.0)
        T_curr = T_start * (T_end / T_start) ** progress
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient
        if T_curr > 100:
            # Use exact subgradient for high T
            conv_y = fast_autocorr(y)
            g = exact_subgradient_full(y, conv_y, threshold_frac=0.99)
        else:
            g = smooth_gradient_fft(y, T_curr)
        
        # Step size (adaptive)
        eta_t = c0 * np.sqrt(n) / (np.sqrt(t) + 1.0)
        
        # Mirror update
        a_new = mirror_update_log(y, g, eta_t)
        
        x_prev = x
        x = a_new
        
        # Evaluate
        obj_val = smooth_objective(x, T_curr)
        if t % 50 == 0 or obj_val < best_obj - 1e-10:
            exact_val = 2.0 * n * fast_autocorr(x).max()
            if exact_val < best_exact - 1e-12:
                best_exact = exact_val
                best_a = x.copy()
                stall_count = 0
            obj_val = smooth_objective(best_a, T_curr)
        else:
            if obj_val < best_obj - 1e-10:
                stall_count = 0
        
        best_obj = min(best_obj, obj_val)
        stall_count += 1
        
        # Restart if stuck

            stall_count = 0
            noise = rng.normal(0, 0.01, n) * best_a

1. Smoothed Mirror Descent with adaptive log-sum-exp temperature (O(n log n) gradients via FFT)
2. Exact subgradient for high-temperature regime
3. Nesterov acceleration with adaptive step size
4. Targeted coordinate descent with importance-based mass transfer
5. Multi-length exploration with interpolation
Key innovation: FFT-based O(n log n) gradient for the smoothed objective, enabling many more
iterations and better exploration of the non-convex landscape.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smooth_objective_value(a, T):
    """Compute log-sum-exp smoothed max of autocorrelation."""
    conv = fast_autocorr(a)
    max_c = conv.max()
    return (1.0 / T) * (max_c + np.log(np.sum(np.exp(T * (conv - max_c))))), conv


def smooth_gradient_fft(a, T):
    """
    O(n log n) gradient of (1/T)*log(sum_k exp(T*conv[k])) w.r.t. a.
    Uses FFT to compute the weighted cross-correlation.
    g[m] = (2/T) * sum_j w[m+j] * a[j]
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_c = np.exp(T * (conv - max_c))
    w = exp_c / exp_c.sum()

    # g[m] = 2 * sum_j w[m+j]*a[j] / T
    # Using: w_rev[k] = w[2n-2-k], then g[m] = c[2n-2-m] where c = conv(w_rev, a)
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w[::-1], N)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(W * A, N)
    g = 2.0 * c[n - 1:2 * n - 1][::-1] / T
    return g


def exact_subgradient(a, conv, threshold_frac=0.99):
    """Compute subgradient from lags within threshold_frac of max."""
    n = len(a)
    M = conv.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(conv >= threshold_frac * M)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        m_start = max(0, k_int - n + 1)
        m_end = min(n, k_int + 1)
        if m_start >= m_end:
            continue
        g[m_start:m_end] += 2.0 * a[k_int - m_end + 1: k_int - m_start + 1][::-1]
    return g


def mirror_update_log(a, g, eta):
    """Mirror descent step in log-space (multiplicative weight update)."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smoothed_mirror_descent(a0, time_limit, T_start=5.0, T_end=500.0,
                            c0=2.0, verbose=True, rng_seed=42,
                            max_iters=300000, use_backtracking=True):
    """
    Main optimizer: Smoothed Mirror Descent with adaptive temperature,
    Nesterov acceleration, and optional backtracking.
    """
    n = len(a0)
    a = norm_sum1(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_exact = 2.0 * n * fast_autocorr(a).max()

    stall_count = 0
    max_stall = 2000
    best_smooth = np.inf

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature (exponential schedule)
        progress = min(float(t) / float(max_iters), 1.0)
        T_curr = T_start * (T_end / T_start) ** progress

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient (choose method based on temperature)
        if T_curr > 300:
            conv_y = fast_autocorr(y)
            g = exact_subgradient(y, conv_y, threshold_frac=0.99)
        else:
            g = smooth_gradient_fft(y, T_curr)

        # Step size
        eta_t = c0 * np.sqrt(n) / (np.sqrt(t) + 1.0)

        # Optional backtracking
        if use_backtracking and t > 100:
            obj_before, _ = smooth_objective_value(x, T_curr)
            obj_before = obj_before

        # Mirror update
        a_new = mirror_update_log(y, g, eta_t)

        # Backtracking check
        if use_backtracking and t > 100:
            obj_after, _ = smooth_objective_value(a_new, T_curr)
            retries = 0
            while obj_after > obj_before + 1e-10 and retries < 5:
                eta_t *= 0.5
                a_new = mirror_update_log(y, g, eta_t)
                obj_after, _ = smooth_objective_value(a_new, T_curr)
                retries += 1

        x_prev = x
        x = a_new

        # Periodic evaluation of exact objective
        if t % 20 == 0:
            exact_val = 2.0 * n * fast_autocorr(x).max()
            if exact_val < best_exact - 1e-12:
                best_exact = exact_val
                best_a = x.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Smooth objective tracking
        if t % 5 == 0:
            sobj, _ = smooth_objective_value(x, T_curr)
            if sobj < best_smooth - 1e-12:
                best_smooth = sobj

        # Restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            noise_scale = 0.005 + 0.02 * min(stall_count / max_stall, 1.0)
            noise = rng.normal(0, noise_scale, n) * best_a
            x_restart = np.maximum(best_a + noise, 1e-10)
            x_restart = x_restart / x_restart.sum()
            x = x_restart
            x_prev = x_restart
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SMD restart t={t} ({elapsed:.1f}s): T={T_curr:.1f}, exact={best_exact:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            exact_now = 2.0 * n * fast_autocorr(x).max()
            print(f"    SMD t={t} ({elapsed:.1f}s): T={T_curr:.1f}, cur={exact_now:.6f}, best={best_exact:.6f}, stall={stall_count}")

    return best_a, best_exact


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent: move mass from high-importance (peak-contributing) positions
    to low-importance positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_max = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_autocorr(a)
        M = conv.max()

        # Find peak lags (within 98% of max)
        peak_lags = np.where(conv >= 0.98 * M)[0]

        # Compute importance of each position
        importance = np.zeros(n)
        for k in peak_lags:
            k_int = int(k)
            m_start = max(0, k_int - n + 1)
            m_end = min(n, k_int + 1)
            if m_start < m_end:
                importance[m_start:m_end] += a[k_int - m_end + 1: k_int - m_start + 1][::-1]

        # Sort positions by importance
        hot = np.argsort(importance)[::-1][:min(60, n)]
        cold = np.argsort(importance)[:min(100, n)]

        improved = False
        rng.shuffle(hot)
        rng.shuffle(cold)

        for src in hot[:40]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            for dst in cold[:40]:
                if dst == src or a[dst] > a[src] * 1.5:
                    continue

                found = False
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_max - 1e-14:
                        a = a_trial
                        best_max = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random escape: perturb multiple positions
            n_pert = max(3, n // 100)
            idx = rng.choice(n, size=min(n_pert * 2, n), replace=False)
            for i in range(0, len(idx) - 1, 2):
                si, di = int(idx[i]), int(idx[i + 1])
                if a[si] > 1e-8:
                    amt = a[si] * rng.uniform(0.01, 0.06)
                    a[si] -= amt
                    a[di] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_max:
                            best_max = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[si] += amt
                            a[di] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.1f}s): J={2 * n * best_max:.6f}, stall={stall}")

    return best_a, best_max


def softmax_mirror_descent(a0, time_limit, c0=2.5, verbose=True, rng_seed=100,
                           max_iters=200000, top_k=50, beta_power=3.0):
    """
    Alternative: Mirror descent with top-k soft weighting (geometric weighting of top lags).
    Different landscape view than log-sum-exp.
    """
    n = len(a0)
    a = norm_sum1(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_exact = 2.0 * n * fast_autocorr(a).max()
    stall_count = 0
    max_stall = 3000

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute top-k weighted subgradient
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-30:
            g = np.zeros(n)
        else:
            k_max = min(top_k, len(conv_y))
            top_idx = np.argsort(conv_y)[::-1][:k_max]
            top_vals = np.maximum(conv_y[top_idx], 1e-30)
            # Geometric weighting
            rel = top_vals / top_vals[0]
            w = rel ** beta_power
            w = w / w.sum()

            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                m_start = max(0, k_int - n + 1)
                m_end = min(n, k_int + 1)
                if m_start < m_end:
                    g[m_start:m_end] += 2.0 * wi * y[k_int - m_end + 1: k_int - m_start + 1][::-1]

        eta_t = c0 * np.sqrt(n) / (np.sqrt(t) + 1.0)
        a_new = mirror_update_log(y, g, eta_t)

        x_prev = x
        x = a_new

        if t % 30 == 0:
            exact_val = 2.0 * n * fast_autocorr(x).max()
            if exact_val < best_exact - 1e-12:
                best_exact = exact_val
                best_a = x.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            noise = rng.normal(0, 0.01, n) * best_a
            x_restart = np.maximum(best_a + noise, 1e-10)
            x_restart = x_restart / x_restart.sum()
            x = x_restart
            x_prev = x_restart
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SMD2 restart t={t} ({elapsed:.1f}s): exact={best_exact:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            exact_now = 2.0 * n * fast_autocorr(x).max()
            print(f"    SMD2 t={t} ({elapsed:.1f}s): cur={exact_now:.6f}, best={best_exact:.6f}")

    return best_a, best_exact


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Load parent / initialize
    2. Smoothed Mirror Descent (low-to-mid T) for exploration
    3. Top-k soft gradient SMD (different landscape view)
    4. High-T SMD (near-exact) for convergence
    5. Coordinate descent polishing
    6. Multi-length exploration
    7. Final micro-polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x_mesh = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x_mesh) ** 2 + 0.03
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: SMD low-to-mid temperature (exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: SMD explore (T: 3->80) n={n0} ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = smoothed_mirror_descent(norm_sum1(best_seq), t1,
                                           T_start=3.0, T_end=80.0,
                                           c0=2.5, verbose=True, rng_seed=1,
                                           max_iters=200000, use_backtracking=True)
        try_update(a_out, f"smd_explore_n{n0}")

    # Phase 2: Top-k soft gradient SMD (different view)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: SMD-soft (top-k) n={n_best} ===")
    t2 = min(remaining * 0.20, 280.0)
    if t2 > 20:
        a_out, _ = softmax_mirror_descent(norm_sum1(best_seq), t2,
                                          c0=2.0, verbose=True, rng_seed=2,
                                          max_iters=150000, top_k=40, beta_power=4.0)
        try_update(a_out, f"smd_soft_n{n_best}")

    # Phase 3: High-T SMD (near-exact convergence)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SMD refine (T: 50->500) n={n_best} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_out, _ = smoothed_mirror_descent(norm_sum1(best_seq), t3,
                                           T_start=50.0, T_end=500.0,
                                           c0=1.5, verbose=True, rng_seed=3,
                                           max_iters=200000, use_backtracking=True)
        try_update(a_out, f"smd_refine_n{n_best}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t4, verbose=True, rng_seed=4)
        try_update(a_out, f"coord_n{n_best}")

    # Phase 5: SMD again from improved point (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: SMD polish (T: 10->200) n={n_best} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        a_out, _ = smoothed_mirror_descent(norm_sum1(best_seq), t5,
                                           T_start=10.0, T_end=200.0,
                                           c0=1.0, verbose=True, rng_seed=5,
                                           max_iters=150000, use_backtracking=False)
        try_update(a_out, f"smd_polish_n{n_best}")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 6: Multi-length ===")
        deltas = [300, -300, 600, -600, 1000]
        for di, delta in enumerate(deltas):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            n_try = n_best + delta
            if n_try < 400 or n_try > 5000:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_trial = min(remaining * 0.18, 120.0)
            if t_trial < 20:
                continue
            print(f"  Trying n={n_try} (budget {t_trial:.0f}s)")
            a_out, _ = smoothed_mirror_descent(norm_sum1(a_interp), t_trial,
                                               T_start=5.0, T_end=200.0,
                                               c0=2.0, verbose=False, rng_seed=10 + di,
                                               max_iters=100000, use_backtracking=True)
            try_update(a_out, f"multi_n{n_try}")

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final CD n={n_best} ===")
        t7 = min(remaining * 0.30, 100.0)
        if t7 > 10:
            a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t7, verbose=True, rng_seed=7)
            try_update(a_out, f"final_cd_n{n_best}")

    # Phase 8: Final SMD micro-polish (very high T, small steps)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.50, 80.0)
        if t8 > 10:
            a_out, _ = smoothed_mirror_descent(norm_sum1(best_seq), t8,
                                               T_start=200.0, T_end=500.0,
                                               c0=0.5, verbose=False, rng_seed=8,
                                               max_iters=100000, use_backtracking=False)
            try_update(a_out, f"final_smd_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
