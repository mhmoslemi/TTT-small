def norm_sum1(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_topk_subgradient(a, ac, top_k=40, sharpness=8.0):
    """Compute subgradient using top-K lags with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    # Exponential weighting based on how close to max
    gap = np.maximum(top_vals[0] - top_vals, 0.0)
    scale = np.maximum(gap.max(), 1e-10)
    w = np.exp(-sharpness * gap / scale)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        if wi < 1e-10:
            continue
        # subgrad[j] = 2 * a[k-j] if 0 <= k-j < n
        j_range = np.arange(max(0, k_int - n + 1), min(n, k_int + 1))
        a_indices = k_int - j_range
        valid = (a_indices >= 0) & (a_indices < n)
        j_range = j_range[valid]
        a_indices = a_indices[valid]
        if len(j_range) > 0:
            grad[j_range] += 2.0 * wi * a[a_indices]
    return grad

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):

def dual_descent_polish(a0, time_limit, verbose=True):
    """Polish using the KKT-inspired approach: at optimum, a[k*-j] should be constant
    for all j in support. We iteratively adjust to enforce this."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(1, 500):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        k_star = int(np.argmax(ac))
        M_curr = ac[k_star]

        # At optimum: for all j with a[j] > 0, a[k_star - j] should be ~ constant
        # Compute the "violation" and adjust
        j_range = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        a_vals = a[k_star - j_range]
        valid = j_range[a_vals > 1e-12]
        if len(valid) < 2:
            break

        # Target: make a[k_star - j] more uniform for j in support
        target_val = np.mean(a_vals[a_vals > 1e-12])
        lr = 0.05 / (it + 1)

        # Adjust: increase values below target, decrease above
        for j in valid:
            idx_a = k_star - j
            if 0 <= idx_a < n and a[idx_a] > 1e-14:
                diff = target_val - a[idx_a]
                if abs(diff) > 1e-14:
                    a[idx_a] += lr * diff

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    KD iter={it} ({elapsed:.1f}s): J={J:.6f}, k*={k_star}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation and multi-method approach."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: NMD v2 (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.38, 650.0)
    print(f"\n=== Phase 1: NMD2 n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent_v2(a_in, t1, initial_temp=0.015, final_temp=0.001,
                                              c0=3.5, top_k=35, sharpness=7.0, verbose=True)
        try_update(a_out, f"NMD2_n{n0}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.22, 370.0)
    print(f"\n=== Phase 2: LP-FW2 n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t2, solver, max_constraints=200, verbose=True)
        try_update(a_out, f"LP2_n{len(best_seq)}")

    # === Phase 3: NMD polish with tighter parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.14, 240.0)
    print(f"\n=== Phase 3: NMD2 polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent_v2(a_in, t3, initial_temp=0.005, final_temp=0.0005,
                                              c0=2.0, top_k=20, sharpness=10.0, verbose=True)
        try_update(a_out, f"NMD2_polish_{len(best_seq)}")

    # === Phase 4: KKT-inspired polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 140.0)
    print(f"\n=== Phase 4: KKT polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = dual_descent_polish(a_in, t4, verbose=True)
        try_update(a_out, f"KKT_{len(best_seq)}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 150:
        print(f"\n=== Phase 5: Multi-length exploration (budget={remaining:.0f}s) ===")
        alt_lengths = [n0 - 200, n0 + 200, n0 + 400]
        for n_alt in alt_lengths:
            if n_alt < 500:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_long = interpolate_sequence(best_seq, n_alt)
            a_in = norm_sum1(a_long)
            t5 = min(remaining * 0.3, 150.0)
            if t5 > 30:
                a_out, _ = nesterov_mirror_descent_v2(a_in, t5, initial_temp=0.01,
                                                      final_temp=0.001, c0=3.0,
                                                      top_k=30, sharpness=6.0, verbose=False)
                try_update(a_out, f"NMD2_n{n_alt}")

                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 60 and len(best_seq) <= 3500:
                    t5b = min(remaining * 0.4, 120.0)
                    if t5b > 30:
                        a_out, _ = lp_frank_wolfe_v2(a_in, t5b, solver, max_constraints=150,
                                                     verbose=False)
                        try_update(a_out, f"LP2_n{n_alt}")

    # === Phase 6: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 6: Final CD n={n_best} (budget={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        t6 = min(remaining * 0.5, 150.0)
        if t6 > 20:
            a_out, _ = coordinate_descent_v2(a_in, t6, verbose=True, rng_seed=777)
            try_update(a_out, f"CD2_{n_best}")

    # === Phase 7: Final NMD micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final NMD micro n={n_best} (budget={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        t7 = min(remaining * 0.6, 100.0)
        if t7 > 15:
            a_out, _ = nesterov_mirror_descent_v2(a_in, t7, initial_temp=0.003,
                                                  final_temp=0.0003, c0=1.5,
                                                  top_k=15, sharpness=12.0, verbose=False)
            try_update(a_out, f"NMD2_micro_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
