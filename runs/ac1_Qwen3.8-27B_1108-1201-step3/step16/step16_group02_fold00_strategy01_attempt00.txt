import numpy as np

def evaluate_sequence_function(sequence: List[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    n = len(sequence)
    # Reject non-integer/non-float types even if converted
    if not isinstance(sequence, list) or any(not isinstance(x, (float, int, np.number)) for x in sequence):
        return np.inf
    # Reject booleans
    if any(isinstance(x, bool) for x in sequence):
        return np.inf
    # Reject NaN, Infinity
    if any(np.isnan(x) or np.isinf(x) for x in sequence):
        return np.inf
    
    # Convert all elements to float and normalize for consistency
    try:
        a = [float(x) for x in sequence]
    except Exception:
        return np.inf
        
    # Check for negative numbers and set to zero if any
    a = [max(0, x) for x in a]
    
    # Clamp to 1000 if >1000, but note: the normalization might make these measures irrelevant
    a = [min(1000.0, x) for x in a]
    
    n_val = len(a)
    if n_val != n:
        # Pad or truncate if necessary
        if n_val < n:
            a = a + [0.0]*(n-n_val)
        else:
            a = a[:n]
    
    # Check precision of sum
    s = np.sum(a)
    if s < 0.01:
        # Scale up to get meaningful denominator
        a = [x/s * np.sqrt(2*n_val) for x in a]
        s = np.sum(a)
    
    # Compute full self-convolution
    conv_result = np.convolve(a, a, mode='full')
    # conv_result should have length 2*n - 1, but we only need the absolute maximum of all elements
    max_conv = np.max(conv_result)
    
    # Compute denominator with the sum of the potentially scaled sequence
    denom = s * s
    
    # Objective value
    score = (2 * n_val * max_conv) / denom if denom != 0 else np.inf
    
    return score
