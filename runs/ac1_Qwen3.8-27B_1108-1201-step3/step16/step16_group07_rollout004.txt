    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Adaptive T: start low, increase over time
    n_expected = 10000
    step = 1.0 / (n ** 0.5)
    
    for t in range(n_expected):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Animate T
        progress = min(1.0, t / (0.8 * n_expected))
        T = T_start + (T_end - T_start) * progress * progress
        
        conv = fast_conv(a)
        M = conv.max()
        
        # Select active set: top_k peaks
        top_idx = np.argsort(conv)[-top_k:]
        
        # Compute weights
        vals = T * conv[top_idx]
        vals -= vals.max()
        w = np.exp(vals)
        w /= w.sum()
        
        # Create full weight vector (sparse)
        active_set = top_idx.astype(int)
        weights = np.zeros(len(conv))
        weights[active_set] = w
        
        # Compute gradient
        grad = compute_grad(a, weights, active_set)
        
        # Gradient step with momentum
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, n)
        
        # Check improvement
        score_new = fast_conv(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            # Increase step slightly
            step = min(step * 1.002, 5.0)
        else:
            # Revert and decrease step
            step = step * 0.98
            if step < 1e-8:
                break
        
        a = a_new
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): best={best_score:.10f}, T={T:.1f}, step={step:.6f}")
    
    return best_a, best_score


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    
    T_mat = np.zeros((size, n))
    for idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if lo < hi:
            T_mat[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
