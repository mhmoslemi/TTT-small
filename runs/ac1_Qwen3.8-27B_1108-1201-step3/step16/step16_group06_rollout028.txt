import numpy as np

def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s

        top_vals = ac[top_idx]
        
        # Softmax weights
        T_temp = 5.0 / (top_vals[0] + 1e-15)
        exp_vals = np.exp(T_temp * (top_vals - top_vals[0]))
        w_sum = exp_vals.sum()
        weights = exp_vals / w_sum
        
        # Build w_full and compute gradient via FFT
        w_full[:] = 0.0
        for wi, k in zip(weights, top_idx):
            w_full[int(k)] = wi
        
        L = 1 << (3 * n - 1).bit_length()
        Wf = np.fft.rfft(w_full, L)
        Af = np.fft.rfft(y, L)
        cross = np.fft.irfft(Wf * np.conj(Af), L)
        grad = 2.0 * cross[:n]
        
        # Mirror descent step (exponential family / multiplicative weights)
        eta_t = eta / (np.sqrt(t + 1.0) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.05, n)
            x_restarted = best_a * (1.0 + perturbation)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            eta = 4.0 / np.sqrt(n)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
