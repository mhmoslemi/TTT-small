        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    return T


def lp_best_response(a_fixed, M, active_set, solver, n):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    T = build_constraint_matrix(a_fixed, active_set, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def lp_best_response_small(a_fixed, M, active_set, n, max_constraints=200):
    """LP with capped constraint count for speed."""
    if len(active_set) > max_constraints:
        conv_at_active = fast_autocorr(a_fixed)[active_set]
        keep = np.argsort(conv_at_active)[-max_constraints:]
        active_set = active_set[keep]
    return lp_best_response(a_fixed, M, active_set, get_solver(), n)


def nesterov_lp_optimize(a0, time_limit, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with progressive tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 80
    n_iters = 0
    conv_size = 2 * n - 1
    solver = get_solver()

    y = a.copy()
    y_prev = a.copy()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 50 == 0:
            progress = min(1.0, t / 1500.0)
            tol = 0.10 * (1.0 - progress) + 0.003 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 100)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            max_active = int(2.0 * n)
            if len(active_set) > max_active:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver, n)

        if b is None:
            if t % 10 == 0:
                top_only = np.argsort(conv_y)[-max(10, n // 50):]
                b, duals = lp_best_response(y, M_curr, top_only, solver, n)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * target_sum / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * target_sum / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * target_sum / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Occasional ergodic blend
        if t % 100 == 0:
            blend = 0.6 * best_a + 0.4 * a
            blend = np.maximum(blend, 0.0)
            s_b = blend.sum()
            if s_b > 1e-15:
                blend = blend * target_sum / s_b
                score_b = fast_autocorr(blend).max()
                if score_b < best_score - 1e-12:
                    best_score = score_b
                    best_a = blend.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            # Slight perturbation to escape
            tiny_noise = 0.001 * rng.standard_normal(n)
            y = np.maximum(y + tiny_noise, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * target_sum / s_y

        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Coordinate descent exploiting the structure:
    (a*a)[k] change when a[j] changes by d: d*(a[k-j]+a[j]) + d^2*[k=2j]
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        conv = fast_autocorr(a)
        S = a.sum()
        current_score = 2.0 * n * conv.max() / (S * S)
        improved = False

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-10:
                continue

            # b_k = a[k-j] (if valid) + a[j], for k=0..2n-2
            # e_k = 1 if k = 2j else 0
            b_vec = np.zeros(2 * n - 1)
            lo = j  # k-j >= 0 => k >= j
            hi = j + n  # k-j < n => k < j+n
            hi = min(hi, 2 * n - 1)
            if lo < hi:
                b_vec[lo:hi] = a[0:hi - lo]
            b_vec += aj  # add a[j] to all

            e_vec = np.zeros(2 * n - 1)
            if 2 * j < 2 * n - 1:
                e_vec[2 * j] = 1.0

            # Try candidate d values
            typ = target_sum / n
            candidates_d = [0.0,
                            0.3 * (typ - aj),
                            0.7 * (typ - aj),
                            (typ - aj),
                            -0.5 * aj,
                            -aj,  # zero out
                            0.5 * aj,
                            1.5 * aj - aj]  # = 0.5*aj
            # Also try to equalize top two peaks
            top2 = np.argsort(conv)[-3:]
            for k1 in top2:
                for k2 in top2:
                    if k1 != k2:
                        diff_b = b_vec[k1] - b_vec[k2]
                        if abs(diff_b) > 1e-12:
                            d_eq = -(conv[k1] - conv[k2]) / diff_b
                            if abs(d_eq) < 2 * aj + typ:
                                candidates_d.append(d_eq)

            best_d = 0.0
            best_obj = current_score

            for d in candidates_d:
                new_aj = aj + d
                if new_aj < 0:
                    continue
                S_new = S + d
                if S_new < 0.01:
                    continue
                new_conv = conv + d * b_vec + d * d * e_vec
                scale = (2.0 * n) / (S_new * S_new)
                obj = scale * new_conv.max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_d = d

            if abs(best_d) > 1e-12:
                a[j] = a[j] + best_d
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * target_sum / s_new
                    new_conv = fast_autocorr(a)
                    new_score = 2.0 * n * new_conv.max() / (target_sum * target_sum)
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = new_conv
                        S = target_sum
                        current_score = best_score
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
                        S = a.sum()
                        current_score = 2.0 * n * conv.max() / (S * S)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def spectral_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Frequency-domain optimization using exponentiated gradient.
    In frequency domain, autocorrelation peak = IDFT(|A|^2) peak.
    We distribute |A|^2 more evenly to reduce time-domain peaks.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    patience = 0
    max_patience = 300
    step_size = 0.1

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Find peaks and create a "correction" signal
        n_peaks = max(5, n // 40)
        peak_indices = np.argsort(conv)[-n_peaks:]

        # Create correction: want to reduce peaks in time domain
        # In frequency domain, subtract a "delta" at each peak
        correction = np.zeros(fft_size, dtype=np.complex128)
        for k in peak_indices:
            weight = 0.5 + 0.5 * (conv[int(k)] / M_curr)
            correction += weight * np.exp(-2j * np.pi * k * np.arange(fft_size) / fft_size)

        # Modify power spectrum: new_power = power - eta * Re(correction * conj(A))
        # This is the gradient of the max in frequency domain
        grad_freq = np.real(correction * np.conj(A))
        new_power = power - step_size * grad_freq
        new_power = np.maximum(new_power, 1e-10)

        # Reconstruct: keep phase of A, use new amplitude
        phase = np.angle(A)
        new_A = np.sqrt(new_power) * np.exp(1j * phase)
        new_a = irfft(A[..., 0:1] * 0 + new_A * 0 + new_A * 0 + new_A, fft_size)[:n]
        # Simpler: new_a = irfft(new_A, fft_size)[:n]
        new_a = irfft(new_A, fft_size)[:n]
        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = new_a * target_sum / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-13:
            a = a_new.copy()
            best_score = score_new
            best_a = a.copy()
            patience = 0
            step_size = min(step_size * 1.02, 2.0)
        else:
            patience += 1
            step_size *= 0.97
            if step_size < 1e-8:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def multi_peak_subgradient(a0, time_limit, verbose=False, seed=0):
    """Multi-peak subgradient with adaptive step and the shifted-reverse gradient structure."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.8 / n ** 0.3
    patience = 0
    max_patience = 500

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Select peaks with diversity
        n_peaks = max(3, min(15, n // 60))
        top_indices = np.argsort(conv)[-n_peaks:]

        # Add random near-peak
        if t % 10 == 0:
            threshold = M_curr * (0.8 + 0.15 * rng.random())
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices.astype(int), k_rand)

        # Compute weighted subgradient using shifted-reverse structure
        # grad_k[m] = 2*a[k-m] if 0<=k-m<n else 0
        weights = np.exp(-6.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.03, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * 2.0 * a[k - np.arange(lo, hi)]

        # Project gradient onto tangent space (sum constraint)
        grad = grad - np.mean(grad)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad * n ** 0.3
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * target_sum / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial.copy()
            best_score = score_trial
            best_a = a.copy()
            patience = 0
            step = min(step * 1.006, 3.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-10:
                break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_and_reoptimize(best_seq, time_limit, rng, seed):
    """Perturb best and do quick LP optimization."""
    n = len(best_seq)
    scale = 0.005 + 0.01 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(perturbed, time_limit, verbose=False, seed=seed)
    return a_out


def try_update(best_seq, best_score, candidate, label, t_global):
    """Try to update best with candidate."""
    if candidate is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def create_structured_init(n, seed=0):
    """Create structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    # Raised cosine
    a1 = 0.5 * (1 + np.cos(2 * np.pi * x))
    a1 = np.maximum(a1, 1e-6)

    # Gaussian bump
    a2 = np.exp(-4.0 * (x - 0.5) ** 2)
    a2 = np.maximum(a2, 1e-6)

    # Multi-sinusoid
    a3 = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.5)
    a3 = np.maximum(a3, 1e-6)

    # Block structure (Frank-like)
    n_blocks = 8
    block_size = n // n_blocks
    a4 = np.zeros(n)
    for b in range(n_blocks):
        amp = 0.5 + 0.5 * rng.random()
        a4[b * block_size:(b + 1) * block_size] = amp
    a4 = np.maximum(a4, 1e-6)

    # Random smooth
    a5 = 1.0 + 0.3 * rng.standard_normal()
    from scipy.ndimage import gaussian_filter1d
    a5 = gaussian_filter1d(a5, sigma=n // 20)
    a5 = np.maximum(a5, 1e-6)

    candidates = [a1, a2, a3, a4, a5]
    best = None
    best_sc = float('inf')
    for c in candidates:
        c = c / c.sum()
        sc = evaluate_safe(c)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = c.copy()
    return best, best_sc


def propose_candidate():
    """
    Main search: multi-phase optimization combining LP best-response,
    coordinate descent, spectral methods, and perturbation restarts.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1780.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq, best_score = create_structured_init(n_base, seed=7)
        print(f"Structured init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 650.0)

        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, verbose=True, seed=0)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd1_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 (different seed) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)

        print(f"\n=== Phase 3: LP r2 n={n_cur}, budget={t3:.0f}s ===")

        a_out, _ = nesterov_lp_optimize(a_norm, t3, verbose=True, seed=777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Multi-peak subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd2_n{n_cur}", t_global)

    # === Phase 6: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t6_total = min(remaining * 0.30, 300.0)
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t6 = min(t6_each, remaining - 30)
        if t6 < 30:
            break
        print(f"\n=== Phase 6r{r}: Perturbation n={n_cur}, budget={t6:.0f}s ===")
        a_out = perturb_and_reoptimize(best_seq, t6, rng, seed=1000 + r)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final LP n={n_cur} ===")
        t7 = min(remaining * 0.5, 120.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t7, verbose=False, seed=2024)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lpFinal_n{n_cur}", t_global)

    # === Phase 8: Final CD micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_cur} ===")
        t8 = min(remaining * 0.7, 100.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_optimize(a_norm, t8, verbose=False, seed=99)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        best_seq = np.ones(1600) / 1600.0
        best_score = evaluate_safe(best_seq)
        print(f"  Fallback score: {best_score:.8f}")

    return [float(x) for x in best_seq]
