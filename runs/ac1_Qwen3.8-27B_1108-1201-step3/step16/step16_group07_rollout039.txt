def norm_sqrt2n(n):
    return np.sqrt(2.0 * n)


def normalize_seq(a, n):
    """Normalize so sum(a) = sqrt(2n)."""
    target = norm_sqrt2n(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target / n
    return a * target / s


def fast_conv(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def subgrad_one(a, k, n):
    """Subgradient of (a*a)[k] w.r.t. a. O(n) time."""
    g = np.zeros(n)
    s = max(0, k - n + 1)
    e = min(n - 1, k)
    if s <= e:
        g[s:e + 1] += a[k - np.arange(s, e + 1)]
    s2 = max(0, k - e)
    e2 = min(n - 1, k - s)
    if s2 <= e2:
        g[s2:e2 + 1] += a[k - np.arange(s2, e2 + 1)]
    return g


def weighted_subgrad(a, peaks, weights, n):
    """Weighted sum of subgradients at given peak indices."""
    g = np.zeros(n)
    for w, k in zip(weights, peaks):
        if w > 1e-12:
            g += w * subgrad_one(a, int(k), n)
    return g


def subgradient_descent(a0, time_limit, verbose=False, seed=0):
    """
    Subgradient descent with Nesterov momentum and adaptive step size.
    Minimizes max_k (p*p)[k] subject to sum(p)=1, p>=0.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    target = norm_sqrt2n(n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    # Nesterov state
    y = a.copy()
    y_prev = a.copy()
    t_count = 0
    step_base = 1.0 / (n ** 0.3)
    step = step_base
    patience = 0
    max_patience = 800

    iters = 0
    while time.monotonic() - t0 < time_limit and patience < max_patience:
        t_count += 1
        conv = fast_conv(y)
        M = conv.max()

        if M < 1e-15:
            break

        # Find active peaks: top indices with exponential weighting
        n_peaks = max(5, min(30, n // 50))
        sorted_idx = np.argsort(conv)[::-1]
        top_n = sorted_idx[:n_peaks]

        # Exponential weighting based on how close to max
        scale = max(M * 0.01, 1e-10)
        weights = np.exp(-4.0 * (M - conv[top_n]) / scale)
        weights = weights / weights.sum()

        # Compute subgradient
        g = weighted_subgrad(y, top_n, weights, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Nesterov step
        alpha = 2.0 / (t_count + 2.0)
        step_eff = step

        # Projected gradient step on y
        y_new = y - step_eff * g / gn
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov extrapolation for a
        if t_count == 1:
            a_new = y_new.copy()
        else:
            momentum = (t_count - 1.0) / (t_count + 2.0)
            momentum = min(momentum, 0.90)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa

        y_prev = y
        y = y_new
        a = a_new

        iters += 1

        # Track best at y and a
        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            patience = 0
            step = min(step * 1.01, 3.0 * step_base)
        else:
            patience += 1
            if patience % 50 == 0:
                step *= 0.98

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            patience = 0

        # Periodic restart
        if patience > max_patience // 2 and patience % 500 == 0:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            t_count = 0
            patience = 0
            step = step_base

        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={iters} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def lp_solve_best_response(a_fixed, M, active_set, solver, warm_start=None):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    if size == 0:
        return None

    # Build constraint matrix T
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0, max_active=300):
    """
    Nesterov-accelerated LP best-response.
    The game: row player picks p (seq), column player picks q (best response).
    Payoff: max_k (p*q)[k]. We minimize this.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    target = norm_sqrt2n(n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    t = 0
    active_set = None
    stall = 0
    warm_b = None

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_y = fast_conv(y)
        M = conv_y.max()
        if M < 1e-15:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            # Start with wider tolerance, tighten over time
            tol = 0.12 * (1.0 - progress) + 0.02 * progress
            threshold = M * (1.0 - tol)
            candidates = np.where(conv_y >= threshold)[0]
            # Always include top peaks
            n_top = max(5, n // 100)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([candidates, top_k]))
            # Cap size
            if len(active_set) > max_active:
                # Keep the largest ones
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        # Solve LP
        b = lp_solve_best_response(y, M, active_set, solver, warm_start=warm_b)
        if b is None:
            # Fallback: use only top peaks
            active_set = np.argsort(conv_y)[-min(50, n // 20):]
            b = lp_solve_best_response(y, M, active_set, solver)
            if b is None:
                if t % 10 == 0:
                    active_set = None
                continue

        warm_b = b.copy()
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # Normalize to sum=1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.92)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Ergodic average every 100 iters
        if t % 100 == 0:
            a_avg = 0.5 * best_a + 0.5 * a
            a_avg = np.maximum(a_avg, 0.0)
            sa_avg = a_avg.sum()
            if sa_avg > 1e-15:
                a_avg = a_avg / sa_avg
                score_avg = fast_conv(a_avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = a_avg.copy()
                    stall = 0

        # Restart if stalled
        if stall > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            warm_b = None
            stall = 0
            t = max(1, t // 3)  # Reduce t to reset momentum

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): score={best_score:.10f}, |active|={len(active_set) if active_set is not None else 0}")

    return best_a, best_score

def try_update_global(best_seq, best_score, a, label, t_global):

    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s

    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()

        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []

    inits.append(np.ones(n) / n)

    x = np.arange(n) / n
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(6 * np.pi * x), 0.01))
    
    # Endpoint-heavy (to reduce center peak)
    decay = 0.98
    w = np.array([decay ** abs(i - n // 2) for i in range(n)])
    inits.append(w / w.sum())
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(alt / alt.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n)))
    
    # Multi-frequency
    sig = 0.3
    f1 = rng.integers(2, 10)
    f2 = rng.integers(10, 30)

    sig_arr = np.maximum(sig_arr, 0.01)
    inits.append(sig_arr / sig_arr.sum())
    
    return inits


def perturb_seq(best, scale, rng):
    """Perturb a sequence with multiplicative noise."""
    n = len(best)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    pert = best * noise
    s = pert.sum()
    if s < 1e-15:
        return best.copy()
    return pert / s

    # === Phase 0: Try different initializations at same length (20s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t0 = min(remaining * 0.03, 20.0)
        if t0 > 5:
            print(f"\n=== Phase 0: Diverse init scan, budget={t0:.0f}s ===")
            per_init = t0 / 8
            for i in range(8):
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining < 10:
                    break
                inits = make_initializations(n_base, np.random.default_rng(100 + i))
                for init in inits[:2]:
                    remaining = time_budget - (time.monotonic() - t_global)
                    if remaining < 5:
                        break
                    a_out, _ = subgradient_descent(init, min(per_init / 2, 3.0), verbose=False, seed=i)
                    best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"init{i}", t_global)
            # Also try a slightly different length
            for n_alt in [n_base + 100, n_base + 200]:
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining < 15:
                    break
                x = np.linspace(0, 1, n_alt)
                init_alt = np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * x), 0.01)
                init_alt = init_alt / init_alt.sum()
                a_out, _ = subgradient_descent(init_alt, min(5.0, remaining - 10), verbose=False, seed=200)
                best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"init_n{n_alt}", t_global)

    n_cur = len(best_seq)
    print(f"After Phase 0: n={n_cur}, score={best_score:.8f}")

    # === Phase 1: Subgradient descent (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Subgradient n={n_cur}, budget={t1:.0f}s ===")
        a_norm = best_seq.copy()
        a_out, _ = subgradient_descent(a_norm, t1, verbose=True, seed=0)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"subg1_n{n_cur}", t_global)

    n_cur = len(best_seq)

    # === Phase 2: LP best-response (main, 500s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 50:
        print(f"\n=== Phase 2: LP Nesterov n={n_cur}, budget={t2:.0f}s ===")
        a_out, _ = nesterov_lp(best_seq, t2, solver, verbose=True, seed=0, max_active=250)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"lp1_n{n_cur}", t_global)

    n_cur = len(best_seq)

    # === Phase 3: Coordinate descent (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: CD n={n_cur}, budget={t3:.0f}s ===")
        a_out, _ = coord_descent(best_seq, t3, verbose=True, seed=5)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"cd1_n{n_cur}", t_global)

    n_cur = len(best_seq)

    # === Phase 4: LP round 2 with different seed (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP r2 n={n_cur}, budget={t4:.0f}s ===")
        a_out, _ = nesterov_lp(best_seq, t4, solver, verbose=True, seed=777, max_active=200)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    n_cur = len(best_seq)

    # === Phase 5: Multi-restart (400s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_restarts = 4
    t5_total = min(remaining * 0.22, 400.0)
    t5_each = t5_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t5 = min(t5_each, remaining - 20)
        if t5 < 30:
            break
        print(f"\n=== Phase 5r{r}: Restart {r+1}/{n_restarts}, n={len(best_seq)}, budget={t5:.0f}s ===")
        
        # Perturb
        scale = 0.02 + 0.02 * r * 0.5
        perturbed = perturb_seq(best_seq, scale, rng)
        
        # Short LP optimization
        t_lp = t5 * 0.7
        t_cd = t5 * 0.3
        a_out, _ = nesterov_lp(perturbed, t_lp, solver, verbose=False, seed=500 + r * 100, max_active=200)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"restart{r}_lp", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = coord_descent(best_seq, t_cd, verbose=False, seed=600 + r * 50)
            best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"restart{r}_cd", t_global)

    n_cur = len(best_seq)

    # === Phase 6: Subgradient polish (100s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Subgrad polish n={n_cur}, budget={t6:.0f}s ===")
        a_out, _ = subgradient_descent(best_seq, t6, verbose=False, seed=314)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"subg_polish_n{n_cur}", t_global)

    # === Phase 7: Final CD (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.7, 120.0)
        print(f"\n=== Phase 7: Final CD n={n_cur}, budget={t7:.0f}s ===")
        a_out, _ = coord_descent(best_seq, t7, verbose=False, seed=999)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"cd_final_n{n_cur}", t_global)

    # Final subgradient micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.6, 40.0)
        a_out, _ = subgradient_descent(best_seq, t8, verbose=False, seed=271)
        best_seq, best_score, _ = try_update_global(best_seq, best_score, a_out, f"subg_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
