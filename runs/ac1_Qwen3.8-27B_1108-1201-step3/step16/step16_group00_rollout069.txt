def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_u_shaped(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (x ** sharpness) + ((1.0 - x) ** sharpness) + 0.01
    return norm_sum1(a)

def make_inverse_u_shaped(n, sharpness=2.0):

def compute_soft_subgradient(a, ac, top_k=80, beta=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            break

        ac = fast_autocorr(a)
        g = compute_exact_subgrad(a, ac, n_peaks=3)

        step = eta
        accepted = False
        for _ in range(8):
            a_trial = a - step * momentum_vec
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()

            if M_trial < M_curr - 1e-14:

                break

        if not accepted:
            stall += 1
            if stall > 300:
                pert = rng.normal(0, 0.01, n) * a
                a = a + pert
                a = np.maximum(a, 1e-10)

                momentum_vec = np.zeros(n)

                stall = 0
            continue

            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_u_shaped(n_base, sharpness=1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, best_score={best_score:.6f}")

    # Phase 1: Multi-init AMD
    print(f"\n=== Phase 1: Multi-init AMD ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 300.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shaped(n0, sharpness=1.5))
        labels.append("u15")
        inits.append(make_u_shaped(n0, sharpness=3.0))
        labels.append("u30")
        inits.append(make_edge_concentrated(n0, decay=0.05))
        labels.append("edge")
        inits.append(make_smooth_random(n0, seed=7))
        labels.append("sr7")
        inits.append(make_two_spiral(n0, seed=123))
        labels.append("spiral")
        inits.append(make_inverse_u_shaped(n0, sharpness=2.0))
        labels.append("invu")

        if parent is not None:
            noise = rng.normal(0, 0.1, n0)
            blend = 0.8 * norm_sum1(parent) + 0.2 * norm_sum1(noise)
            inits.append(norm_sum1(blend))
            labels.append("pnoisy")

        # Deduplicate
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = np.round(init[::max(1, n0 // 50)], 3)
            is_dup = False
            for existing in unique_inits:
                if np.allclose(sig, np.round(existing[::max(1, n0 // 50)], 3), atol=0.02):
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        t_per = max(10.0, t_phase1 / len(unique_inits))
        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  AMD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=4.0, c0=3.5,
                                     restart_every=5000, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        top_K = min(400, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=top_K)
        try_update(a_out, f"lp_n{n_best}")

    # Phase 3: Subgradient method
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t3, verbose=False, rng_seed=777)
        try_update(a_out, f"subgrad_n{n_best}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42, n_peak_lags=20)
        try_update(a_out, f"coord_n{n_best}")

    # Phase 5: Two-point CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Two-point CD (n={n_best}) ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = two_point_cd(a_in, t5, verbose=False, rng_seed=314)
        try_update(a_out, f"2pt_n{n_best}")

    # Phase 6: Length tuning
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length tuning ===")
    candidate_lengths = []
    if n_best > 1000:
        candidate_lengths.append(n_best - 200)
    candidate_lengths.extend([n_best + 200, n_best + 400])
    candidate_lengths = [l for l in candidate_lengths if 500 <= l <= 5000]

    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.25, 120.0)
        if t6a > 15:
            a_out, _ = amd_optimize(a_long, t6a, top_k=60, beta=4.0, c0=3.0,
                                     restart_every=5000, verbose=False, rng_seed=999)
            try_update(a_out, f"amd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 3500:
            t6b = min(remaining * 0.15, 80.0)
            if t6b > 10:
                a_in = norm_sum1(interpolate_sequence(best_seq, new_n))
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False, top_K=min(300, 2 * new_n - 1))
                try_update(a_out, f"lp_n{new_n}")

    # Phase 7: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t7 = min(remaining * 0.30, 250.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            top_K = min(600, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=top_K)
            try_update(a_out, f"lp_fin_n{n_best}")

    # Phase 8: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord (n={n_best}) ===")
    if remaining > 15:
        t8 = min(remaining * 0.5, 200.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=4242, n_peak_lags=25)
            try_update(a_out, f"coord_fin_n{n_best}")

    # Phase 9: Extra LP if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 3500:
        print(f"\n=== Phase 9: Extra LP (n={n_best}) ===")
        t9 = min(remaining * 0.6, 120.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t9, solver, verbose=False, top_K=min(800, 2 * n_best - 1))
            try_update(a_out, f"lp_extra_n{n_best}")

    # Phase 10: Final two-point CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final 2pt CD (n={n_best}) ===")
        t10 = min(remaining * 0.8, 100.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = two_point_cd(a_in, t10, verbose=False, rng_seed=9999)
            try_update(a_out, f"2pt_fin_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 2400
        best_seq = make_u_shaped(n_fb, sharpness=1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
