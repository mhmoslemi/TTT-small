def cross_corr(a, q, n):
    """Compute (a ⋆ q)[k] = sum_i a[i]*q[i+k] for k=0,...,n-1."""
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()
    A = rfft(a, fft_size)
    Q = rfft(q, fft_size)
    corr = irfft(A * Q, fft_size)
    # corr[k] should give us the circular cross-correlation
    # For linear cross-correlation, we need to extract the right portion
    # (a ⋆ q)[k] = sum_i a[i] * q[i+k]
    # This equals the linear cross-correlation at lag k
    # In FFT terms: IFFT(FFT(a) * FFT(q)) gives circular cross-correlation
    # For our purposes, since we've zero-padded enough, the linear cross-correlation
    # is embedded in the result.
    return corr[:n]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with adaptive epsilon (primary method, O(n log n) per step)
2. LP best-response with Nesterov acceleration (refinement)
3. Coordinate descent (final polish)
4. Multi-restart with diverse initializations

Key innovation: Using the log-sum-exp smoothing of the max function to get a
smooth, differentiable objective that can be optimized with fast gradient methods.
The gradient is computed via FFT-based cross-correlation in O(n log n).

import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_gradient_fft(a, p, n, fft_size):
    """
    Compute g[j] = 2 * sum_{i=0}^{n-1} a[i] * p[j+i] for j=0,...,n-1.
    p has length >= 2n-1 (padded with zeros beyond 2n-2).
    Uses FFT-based cross-correlation.
    """
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(A * P, fft_size)
    return 2.0 * corr[:n]


def smooth_objective(a, eps, conv):
    """Compute F_eps(a) = eps * log(sum exp(conv/eps)) using log-sum-exp trick."""
    max_c = conv.max()
    if max_c < -1e10:
        return 0.0
    shifted = (conv - max_c) / eps
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    return eps * (max_c / eps + np.log(np.sum(np.exp(shifted))))

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)

    if s < 1e-15:

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    if not HAS_CVXPY:
        return None, None
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    if size == 0:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

    t_iter = 0
    step = 1.0
    patience = 0
    max_patience = 200

    # We'll adapt epsilon based on iteration count

            v = np.zeros(n)
            step = 1.0

        eps = max(eps, 1e-10)

        else:

        if len(active) == 0:

        elif len(active) > n_peaks * 3:

            active = active[top]

        weights = weights / weights.sum()

        # Compute weighted subgradient
        g = np.zeros(n)
        for w, k in zip(weights, active):
            k = int(k)
            # g[j] += w * 2 * a[k-j] (if 0 <= k-j < n)

            if lo < hi:
                g[lo:hi] += w * 2.0 * a[k - np.arange(lo, hi)]

        g = g - g_mean

        gn = np.linalg.norm(g)
        if gn < 1e-15:

        g = g / gn

        a_trial = a - step * g * n ** 0.3
        a_trial = np.maximum(a_trial, 0.0)

            score_h = fast_autocorr(h_norm).max()
            if score_h < best_score:
                best_score = score_h
                best_a = h_norm.copy()

        if verbose and t_iter % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t_iter} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    y_prev = a.copy()
    active_set = None
    stall_count = 0
    conv_size = 2 * n - 1
    n_iters = 0
    h = a.copy()

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 25 == 0:
            progress = min(1.0, t / 600.0)
            tol = 0.05 * (1.0 - progress) + 0.002 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(8, n // 60)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            max_constraints = int(2.0 * n)
            if len(active_set) > max_constraints:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_constraints:]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            if t % 10 == 0:
                active_set = np.arange(conv_size)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                stall_count += 1
                if stall_count > 20:
                    active_set = None
                    stall_count = 0
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.94)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        h = h * (n_iters / (n_iters + 1)) + a * (1.0 / (n_iters + 1))

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if n_iters % 200 == 0:
            h_norm = normalize_to(h, n)
            score_h = fast_autocorr(h_norm).max()
            if score_h < best_score:
                best_score = score_h
                best_a = h_norm.copy()
                stall_count = 0

        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            typical = norm / n
            candidates = [0.0, aj * 0.4, aj * 0.7, aj * 1.3, aj * 1.7, typical * 0.6, typical * 1.2, typical * 2.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.25)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                new_max = new_conv.max()
                if new_max < best_obj - 1e-14:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = best_a.copy()

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def generate_initialization(n, method, rng):
    if method == 0:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif method == 1:
        x = np.linspace(0.0, 1.0, n)
        freq = rng.uniform(1.5, 3.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.35 * np.cos(2 * np.pi * freq * x + phase)
    elif method == 2:
        a = 0.5 + 0.25 * rng.standard_normal(n)
    elif method == 3:
        period = rng.integers(n // 8, n // 3)
        x = np.arange(n)
        a = 0.4 + 0.5 * np.cos(2 * np.pi * x / period)
    elif method == 4:
        a = 0.4 + 0.3 * ((-1) ** np.arange(n)) + 0.15 * rng.standard_normal(n)
    elif method == 5:
        x = np.arange(n) / n
        a = 0.3 + 0.6 * np.abs(np.sin(np.pi * x * rng.uniform(1, 4)))
    elif method == 6:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.4 * np.cos(2 * np.pi * 2.5 * x) + 0.2 * np.cos(2 * np.pi * 7.3 * x + 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()
