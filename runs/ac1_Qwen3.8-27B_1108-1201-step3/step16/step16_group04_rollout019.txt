def project_simplex(x):
    """Project onto simplex {a >= 0, sum(a) = 1}."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def solve_lp_full(y, M, solver, use_all=False, top_k=300, constraint_margin=1.0):
    """Solve LP for FW direction with adaptive constraints."""
    n = len(y)
    ac = fast_autocorr(y)
    b_var = cp.Variable(n, nonneg=True)
    
    if use_all:
        lags = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(ac)[::-1][:top_k]
        lags = top_idx
    
    constraints = []
    for k in lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        M_k = M * constraint_margin
        constraints.append(y_coeff @ b_var[j_idx] <= M_k)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        # Restart if severely stalled
        if stall_count > 5000:
            stall_count = 0
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            pert = rng.normal(0, 0.01, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            use_full = False
            top_k = 200
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={top_k}, full={use_full}")
    
    return best_a, best_m


def smoothed_nesterov(a0, time_limit, top_k=80, beta_init=10.0, verbose=True):
    """Nesterov accelerated mirror descent with adaptive log-sum-exp smoothing."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    x = a.copy()
    x_prev = a.copy()
    c0 = 5.0
    beta = beta_init
    stall_count = 0
    
    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute gradient via top-K peaks with adaptive weighting
        k_use = min(top_k, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:k_use]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        
        # Adaptive beta: increase when stalling (sharper peaks)
        if stall_count > 200:
            beta = min(beta * 1.05, 100.0)
        elif stall_count < 50:
            beta = max(beta * 0.95, 5.0)
        
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_use) / k_use
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = y
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        
        # Restart
        if stall_count > 4000:
            stall_count = 0
            pert = rng.normal(0, 0.008, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            beta = beta_init
            c0 = 5.0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSG t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m


def stochastic_coord_descent(a0, time_limit, rng_seed=42, verbose=True):
    """Stochastic coordinate descent with simulated annealing acceptance."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    temp = 0.01
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        # Select source (weighted by value) and destination (uniform)
        src = int(rng.choice(n, p=a / a.sum() + 1e-15))
        dst = int(rng.integers(n))
        if src == dst:
            continue
        
        # Move a small fraction
        frac = rng.uniform(0.001, 0.05)
        amt = a[src] * frac
        if amt < 1e-18:
            continue
        
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        
        m_trial = fast_autocorr(a_trial).max()
        delta = m_trial - best_m
        
        # Acceptance criterion (simulated annealing)
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / (temp + 1e-15))):
            a = a_trial
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Cool down
        if no_improve > 100:
            temp *= 0.95
        if no_improve > max_no_improve:
            # Restart from best with perturbation
            pert = rng.normal(0, 0.01, n)
            a = best_a * (1.0 + pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            temp = 0.01
            no_improve = 0
        
        if verbose and no_improve % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SCD ({elapsed:.1f}s): J={J:.6f}, temp={temp:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with exact move selection."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            candidates = np.arange(n)
            chosen = candidates[rng.permutation(n)[:min(300, n)]]
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.12, 0.18]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if found:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 0: Parametric envelope optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 0: Parametric envelope, n={n0} ===")
    if remaining > 40:
        t0 = min(remaining * 0.03, 30.0)
        a_env, _ = parametric_envelope(n0, time_limit=t0)
        try_update(a_env, "envelope")
    
    # === Phase 1: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)

    if remaining > 30:

        a_spec = spectral_shape(best_seq, n_iter=60, alpha=0.55, damping=0.5)

        try_update(a_spec2, "spectral2")
    
    # === Phase 2: Enhanced LP-FW (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Enhanced LP-FW, n={n_best}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = enhanced_lp_fw(a_in, t2, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n_best}")
    
    # === Phase 3: Smoothed Nesterov mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed NSG, n={n_best}, budget={remaining:.0f}s ===")

    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = smoothed_nesterov(a_in, t3, top_k=80, beta_init=8.0, verbose=True)
        try_update(a_out, f"nsg_n{n_best}")
    
    # === Phase 4: LP-FW again from improved point ===

    else:

        return [1.0]
    
    return [float(x) for x in best_seq]
