"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 (scale-invariant, so work with sum(a)=1, a>=0).

Methods combined:
  1. Nesterov-accelerated top-K subgradient descent. Gradient of weighted sum of the
     top-K autocorrelation peaks is computed via one FFT convolution in O(n log n):
     d(ac[k])/d(a[j]) = 2*a[k-j]  (0<=k-j<n),  else 0.
  2. Sparse-active-set LP Frank-Wolfe: per iteration keep only the binding constraints
     (peaks near the current max) so the LP is small and fast; Nesterov steering.
  3. Peak-targeted coordinate descent with line search on the max peak.
  4. Multi-start from diverse non-negative templates + multi-n exploration via
     linear interpolation of a normalized parent.
"""

import numpy as np
import time
import math

import scipy.sparse as sp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def peak_of(a):
    return float(fast_autocorr(a).max())


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def make_template(kind, n):
    x = np.linspace(0.0, 1.0, n)
    if kind == 0:
        f = np.ones(n)
    elif kind == 1:
        f = np.sin(np.pi * x) ** 2
    elif kind == 2:
        c = n / 2.0
        sig = n / 5.0
        f = np.exp(-0.5 * ((np.arange(n) - c) / sig) ** 2)
    elif kind == 3:
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    else:
        tri = 1.0 - np.abs(2.0 * x - 1.0)
        f = tri ** 0.5
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def compute_gradient_topk(a, w_full):
    """grad[j] = 2 * sum_k w_full[k] * a[k-j]  =  2*(w_full * flip(a))[j+n-1]."""
    n = len(a)
    a_flip = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_full = np.fft.irfft(W * AF, fft_size)
    return 2.0 * conv_full[n - 1:2 * n - 1]


def nesterov_subgradient(a0, time_limit, top_k=40, beta=4.0, gamma=6.0,
                          verbose=False, max_stall=4000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = peak_of(a)
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    iters = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        t = iters + 1
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        order = np.argsort(ac)[::-1][:k_max]
        tv = np.maximum(ac[order], 1e-30)
        w = tv ** beta
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / ws

        w_full = np.zeros(2 * n - 1)
        w_full[order] = w
        grad = compute_gradient_topk(y, w_full)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        mlog = log_a.max()
        a_new = np.exp(log_a - mlog)
        a_new = np.maximum(a_new, 1e-30)
        snew = a_new.sum()
        if snew < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / snew

        x_prev = x
        x = a_new
        iters += 1

        mn = peak_of(x)
        if mn < best_m - 1e-14:
            best_m = mn
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            if iters > 200:
                pert = rng.normal(0, 0.015, n) * best_a
                xr = norm_sum1(best_a + pert)
                x = xr
                x_prev = xr
                iters = 0

        if verbose and iters % 20000 == 0:
            el = time.monotonic() - t0
            print("    NSGD it=%d (%.1fs) J=%.6f" % (iters, el, 2.0 * n * best_m))

    return best_a, best_m


def build_sparse_rows(a, S):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in S:
        kint = int(k)
        js = max(0, kint - n + 1)
        je = min(n - 1, kint)
        for j in range(js, je + 1):
            rows.append(kint)
            cols.append(j)
            vals.append(a[kint - j])
    if not rows:
        return None
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_frank_wolfe_sparse(a0, time_limit, solver, active_frac=0.05,
                          max_active=400, verbose=False):
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = peak_of(a)
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        t = iters + 1
        beta_t = (t - 1.0) / (t + 1.5)
        yy = y + beta_t * (y - y_prev) if iters > 1 else y.copy()
        yy = np.maximum(yy, 1e-15)
        SY = yy.sum()
        if SY < 1e-15:
            yy = y.copy()
        else:
            yy = yy / SY

        ac = fast_autocorr(yy)
        M = float(ac.max())
        if M < 1e-14:
            break

        thr = M * (1.0 - active_frac)
        cand = np.where(ac >= thr)[0]
        if len(cand) < 4:
            idx = np.argsort(ac)[::-1]
            cand = idx[:max(4, min(max_active, len(idx)))]
        else:
            order = ac[cand]
            perm = np.argsort(order)[::-1][:max_active]
            cand = cand[perm]

        T = build_sparse_rows(yy, list(cand))
        if T is None:
            break

        b = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        if b.value is None:
            continue

        bv = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
        bs = bv.sum()
        if bs < 1e-15:
            continue
        g = bv / bs

        alpha = 2.0 / (t + 2.0)
        y_prev = y
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sY = y_new.sum()
        if sY < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sY
        y = y_new

        if t == 1:
            x_new = g.copy()
        else:
            x_new = y_new + beta_t * (y_new - y_prev)
            x_new = np.maximum(x_new, 0.0)
            SX = x_new.sum()
            if SX < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / SX
        x = x_new
        iters += 1

        for candvec, lab in ((y, 'y'), (x, 'x')):
            mv = peak_of(candvec)
            if mv < best_m:
                best_m = mv
                best_a = candvec.copy()

        if verbose and iters % 15 == 0:
            el = time.monotonic() - t0
            print("    LPFW it=%d (%.1fs) J=%.6f act=%d" %
                  (iters, el, 2.0 * n * best_m, len(cand)))

    return best_a, best_m


def coord_descent(a0, time_limit, verbose=False, rng_seed=7, max_stall=800):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = peak_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        kmax_idx = int(np.argmax(ac))
        js = max(0, kmax_idx - n + 1)
        je = min(n - 1, kmax_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[kmax_idx - idx]
        hi = int(idx[int(np.argmax(contrib))])

        # destination: low autocorrelation sensitivity, far from peak contributors
        sens = np.zeros(n)
        for _ in range(n):
            pass
        # approximate sensitivity by a itself weighted
        sens = a.copy()

        candidate_src = hi
        tried = -1
        while tried < 20:
            tried += 1
            dst = int(np.argmin(sens))
            if dst == candidate_src:
                cand_dst = int(np.argmin(np.where(sens == sens.min(),
                                                  np.inf, sens)))
            else:
                cand_dst = dst
            if cand_dst == candidate_src:
                break
            improved = False
            for frac in (0.02, 0.05, 0.1, 0.2, 0.5):
                amt = a[candidate_src] * frac
                if amt < 1e-18:
                    continue
                at = a.copy()
                at[candidate_src] -= amt
                at[cand_dst] += amt
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                mt = peak_of(at)
                if mt < best_m - 1e-16:
                    a = at
                    best_m = mt
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                break
            else:
                # move to next least sensitive destination not equal src
                sens[cand_dst] = np.inf

        if stall == 0:
            continue

        # random walk to escape
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and a[i] > 1e-10:
            frac = float(rng.uniform(0.01, 0.05))
            amt = a[i] * frac
            at = a.copy()
            at[i] -= amt
            at[j] += amt
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st > 1e-15:
                at = at / st
                mt = peak_of(at)
                if mt < best_m:
                    a = at
                    best_m = mt
                    best_a = a.copy()
                    stall = 0
        stall += 1

    if verbose:
        print("    Coord (%.1fs) J=%.6f" % (time.monotonic() - t0, 2.0 * n * best_m))
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print("Time budget: %.0fs  solver: %s" % (time_budget, solver))

    best_seq = None
    best_score = float('inf')

    # load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            pn = p / p.sum()
            ps = evaluate_safe(pn)
            print("Parent: n=%d score=%.6f" % (len(pn), ps))
            best_seq = pn.copy()
            best_score = ps
    except Exception:
        pass

    if best_seq is None:
        n0 = 2048
        best_seq = make_template(1, n0)
        best_score = evaluate_safe(best_seq)
        print("Default init n=%d score=%.6f" % (n0, best_score))

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        an = a / s
        sc = evaluate_safe(an)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = an.copy()
            print("  *** NEW BEST [%s] %.6f  n=%d  t=%.0fs" %
                  (label, sc, len(an), time.monotonic() - t_global))

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # -------- Phase A: NSGD warmup on parent --------
    if remaining() > 40:
        ncur = len(best_seq)
        tA = min(remaining() * 0.10, 90.0)
        out, _ = nesterov_subgradient(norm_sum1(best_seq), tA, top_k=40,
                                      beta=4.0, gamma=6.0, verbose=False)
        try_update(out, "NSGD0_n%d" % ncur)

    # -------- Phase B: sparse LP-FW (main) --------
    if remaining() > 60:
        ncur = len(best_seq)
        tB = min(remaining() * 0.40, 650.0)
        out, _ = lp_frank_wolfe_sparse(norm_sum1(best_seq), tB, solver,
                                       active_frac=0.05, max_active=350,
                                       verbose=True)
        try_update(out, "LPFW_n%d" % ncur)

    # -------- Phase C: coordinate polish --------
    if remaining() > 40:
        ncur = len(best_seq)
        tC = min(remaining() * 0.08, 90.0)
        out, _ = coord_descent(norm_sum1(best_seq), tC, verbose=True, rng_seed=7)
        try_update(out, "coord_n%d" % ncur)

    # -------- Phase D: second LP-FW round --------
    if remaining() > 60:
        ncur = len(best_seq)
        tD = min(remaining() * 0.25, 400.0)
        out, _ = lp_frank_wolfe_sparse(norm_sum1(best_seq), tD, solver,
                                       active_frac=0.03, max_active=250,
                                       verbose=True)
        try_update(out, "LPFW2_n%d" % ncur)

    # -------- Phase E: multi-n exploration from best --------
    if remaining() > 120:
        nbest = len(best_seq)
        deltas = [300, 600, -300, 1200]
        done = 0
        for d in deltas:
            nt = nbest + d
            if nt < 512 or nt > 4096 or nt == nbest:
                continue
            if remaining() < 70:
                break
            ai = interpolate_sequence(best_seq, nt)
            tE = min(remaining() * 0.22, 150.0)
            out, _ = nesterov_subgradient(ai, tE * 0.45, top_k=40, beta=4.0,
                                          gamma=6.0, verbose=False)
            try_update(out, "NSGD_n%d" % nt)
            if remaining() < 60:
                break
            tE2 = min(remaining() * 0.30, 180.0)
            out, _ = lp_frank_wolfe_sparse(norm_sum1(best_seq), tE2, solver,
                                           active_frac=0.04, max_active=300,
                                           verbose=False)
            try_update(out, "LPFW_n%d" % nt)
            done += 1

    # -------- Phase F: final polish at best length --------
    if remaining() > 40:
        nb = len(best_seq)
        tF = min(remaining() * 0.30, 150.0)
        out, _ = coord_descent(norm_sum1(best_seq), tF, verbose=True, rng_seed=2024)
        try_update(out, "coordF_n%d" % nb)

    if remaining() > 40:
        nb = len(best_seq)
        tG = min(remaining() * 0.55, 250.0)
        out, _ = lp_frank_wolfe_sparse(norm_sum1(best_seq), tG, solver,
                                       active_frac=0.02, max_active=200,
                                       verbose=True)
        try_update(out, "LPFWF_n%d" % nb)

    # final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("FINAL score=%.6f n=%d time=%.0fs" %
              (best_score, len(best_seq), time.monotonic() - t_global))
    else:
        print("ERROR no valid sequence")
        return [1.0]

    out_list = [float(max(0.0, v)) for v in best_seq]
    return out_list
