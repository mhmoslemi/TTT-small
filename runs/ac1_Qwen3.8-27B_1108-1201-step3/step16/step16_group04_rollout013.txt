    """Flatten power spectrum via iterative DFT correction."""

def fast_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=30, eps=1e-10):
    a = normalize(a).copy()
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = (A.real ** 2 + A.imag ** 2)
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.4 * a + 0.6 * a_new
    return a


def lp_smoothed_gradient(a, c, p, top_k=200):

    Compute approximate gradient of (sum_k c_k^p)^(1/p) w.r.t. a.

    size = 2 * n - 1
    
    # Select top-k lags
    if len(c) > top_k:
        idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        idx = np.arange(len(c))
    
    # Compute weights: w_k = c_k^(p-1)
    w = np.zeros(size)
    c_top = np.maximum(c[idx], 1e-30)
    w[idx] = c_top ** (p - 1)
    
    # Normalize weights (doesn't affect direction, just scale)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)

    T1 = T1_full[size-1-n+1:size]  # indices size-1-(n-1) to size-1, i.e., size-n to size-1

def mirror_descent_lpsmooth(a0, time_limit, p_start=4.0, p_end=200.0, n_phases=5,
                             c0=2.0, top_k=200, verbose=True):
    """
    L_p smoothed mirror descent with adaptive p.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Schedule p values
    p_values = np.geomspace(p_start, p_end, n_phases)
    
    for phase, p in enumerate(p_values):
        phase_start = time.monotonic()
        phase_budget = (time_budget - (time.monotonic() - t0)) / (n_phases - phase) if time_budget > 0 else 60
        
        x = a.copy()
        x_prev = a.copy()
        stall = 0
        
        for t in range(1, 1000000):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - phase_start > phase_budget:
                break
            
            # Nesterov
            if t > 1:
                beta_t = (t - 1.0) / (t + 1.5)
                y = x + beta_t * (x - x_prev)
            else:
                y = x.copy()
            y = np.maximum(y, 1e-18)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
            
            c = fast_conv(y)
            g = lp_smoothed_gradient(y, c, p, top_k=top_k)
            
            # Mirror descent step
            eta = c0 / (np.sqrt(t) + 1.0)
            log_a = np.log(np.maximum(y, 1e-30)) - eta * g
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            
            x_prev = y
            x = a_new
            
            M_new = fast_conv(x).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > 5000:
                pert = rng.normal(0, 0.003, n)
                x_new = best_a * (1.0 + pert)
                x_new = np.maximum(x_new, 1e-10)
                x_new = x_new / x_new.sum()
                x = x_new
                x_prev = x_new
                stall = 0
            
            if verbose and t % 5000 == 0:
                J = 2.0 * n * best_m
                print(f"      Lp-md phase={phase} p={p:.0f} t={t}: J={J:.6f}")
        
        a = best_a.copy()
    
    return best_a, best_m

    Solve LP: max sum(b) s.t. (a0*b)[k] <= M_target for k in active set, b >= 0.
    Use cutting plane: start with top-K, add violated constraints.
    """
    n = len(a0)
    t0 = time.monotonic()
    
    # Start with top-K constraints
    c = fast_conv(a0)
    top_k_init = 50

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_idx = np.arange(j_start, j_end + 1)
            y_coeff = a0[k_int - j_idx]
            constraints.append(y_coeff @ b_var[j_idx] <= M_target)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            return None
        b_val = b_val / b_sum
        
        # Check for violated constraints
        conv_ab = fast_conv(b_val)  # Wait, this is b*b, not a*b
        # We need (a0 * b_val)[k] for all k
        # a0 * b_val is the cross-convolution

        A0 = np.fft.rfft(a0, fft_size)
        B = np.fft.rfft(b_val, fft_size)
        cross = np.fft.irfft(A0 * B, fft_size)[:2*n-1]

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_conv(a, b):
    """Compute linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=30, eps=1e-10):
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.4 * a + 0.6 * a_new
    return a


def lp_smoothed_gradient(a, c, p, top_k=200):
    """Compute approximate gradient of (sum c^p)^(1/p) w.r.t. a using top-k lags."""
    n = len(a)
    size = 2 * n - 1
    if len(c) > top_k:
        idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        idx = np.arange(len(c))
    w = np.zeros(size)
    c_top = np.maximum(c[idx], 1e-30)
    w[idx] = c_top ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    T2_full = np.fft.irfft(W * A, fft_size)
    T2 = T2_full[:n]
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, fft_size)
    T1_full = np.fft.irfft(W_rev * A, fft_size)
    T1 = T1_full[size - n:size][::-1]
    grad = T1 + T2
    return grad


def solve_lp_direction(a0, M, active_lags, max_lags, solver):
    """Solve LP: max sum(b) s.t. (a0*b)[k] <= M for k in active_lags, b >= 0, sum(b)=1."""
    n = len(a0)
    k_list = active_lags[:max_lags]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) == 1]
    for k in k_list:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        coeff = a0[k_int - j_idx]
        constraints.append(coeff @ b_var[j_idx] <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def line_search_conv(a, b, best_m, n_alphas=8):
    """Find best alpha in (0,1) minimizing max self-conv of (1-alpha)*a + alpha*b."""
    alphas = np.linspace(0.02, 0.5, n_alphas)
    best_alpha = 0.0
    best_peak = best_m
    for alpha in alphas:
        trial = (1.0 - alpha) * a + alpha * b
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            continue
        trial = trial / s
        peak = fast_conv(trial).max()
        if peak < best_peak:
            best_peak = peak
            best_alpha = alpha
    return best_alpha, best_peak


def lp_fw_refine(a0, time_limit, solver, top_k_init=50, top_k_max=250, verbose=True):
    """LP Frank-Wolfe with adaptive constraints and line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    current_k = top_k_init
    stall = 0
    max_stall = 200
    rng = np.random.default_rng(99)
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        c = fast_conv(a)
        M_curr = c.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        
        # Select active constraints: top-K + any recently violated
        active_lags = np.argsort(c)[::-1][:current_k]
        
        # Solve LP
        direction = solve_lp_direction(a, M_curr, active_lags, 200, solver)
        if direction is None:
            continue
        
        # Check for violated constraints in the direction
        cross = cross_conv(a, direction)
        violations = cross > M_curr * 1.001
        if violations.any():
            viol_idx = np.where(violations)[0]
            for k_new in viol_idx[:5]:
                if k_new not in active_lags:
                    active_lags = np.append(active_lags, k_new)
            # Re-solve with new constraints (limit attempts)
            direction = solve_lp_direction(a, M_curr, active_lags, 200, solver)
            if direction is None:
                continue
        
        # Line search
        alpha, trial_peak = line_search_conv(a, direction, best_m)
        if alpha > 0:
            a_new = (1.0 - alpha) * a + alpha * direction
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                if trial_peak < best_m:
                    a = a_new
                    best_m = trial_peak
                    best_a = a_new.copy()
                    stall = 0
                else:
                    a = a_new
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        # Adapt K
        if t % 30 == 0 and current_k < top_k_max:
            current_k = min(current_k + 5, top_k_max)
        
        if verbose and t % 20 == 0:
            J = 2.0 * n * best_m
            print(f"      LP-FW t={t} K={current_k}: J={J:.6f} (stall={stall})")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with transfer and swap moves."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:15]
        peak_val = c.max()
        improved = False
        
        # Transfer moves: move mass from high-contributing source to safe destination
        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find safe destination: low sensitivity to top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            safe = np.where(~in_peak)[0]
            if len(safe) < 100:
                safe = np.arange(n)
            candidates = safe[rng.permutation(len(safe))[:min(100, len(safe))]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in candidates:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add

                    best_dst = dst

                        trial[j] -= amt
                        trial[i] += amt
                    trial = np.maximum(trial, 0.0)
                    s_t = trial.sum()
                    if s_t < 1e-15:
                        continue
                    trial = trial / s_t
                    M_t = fast_conv(trial).max()
                    if M_t < best_m - 1e-15:

                        best_m = M_t

        a_out, _ = lp_fw_refine(a_in, t3, solver, top_k_init=80, top_k_max=300, verbose=True)
        try_update(a_out, "lpfw3")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord n={len(best_seq)}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, "coord4")
    
    # === Phase 5: Random restart + short refine ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Restarts, budget={remaining:.0f}s ===")
    rng = np.random.default_rng(789)
    n_restarts = 3
    if remaining > 150:
        restart_budget = remaining / (n_restarts + 1)
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            # Perturb best
            pert = rng.normal(0, 0.02, n0)
            a_pert = best_seq * (1.0 + pert)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            # Short LP refinement
            t_r = min(restart_budget, 120.0)
            if t_r > 30:
                a_out, _ = lp_fw_refine(a_pert, t_r, solver, top_k_init=80, top_k_max=200, verbose=False)
                try_update(a_out, f"restart{r}")
    
    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Coord, budget={remaining:.0f}s ===")
    if remaining > 40:
        t6 = min(remaining * 0.5, 120.0)
        if t6 > 20:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_descent(a_in, t6, verbose=True, rng_seed=1234)
            try_update(a_out, "coord_final")
    
    # === Phase 7: Very short LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t7 = min(remaining * 0.7, 80.0)
        if t7 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_fw_refine(a_in, t7, solver, top_k_init=100, top_k_max=250, verbose=False)
            try_update(a_out, "lpfw_very_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
