"""

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_max_conv(a, conv, top_k=3):
    """
    Compute subgradient of max(conv(a,a)) with softmax weighting over top peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Find top-k peaks with softmax weighting
    max_val = conv.max()
    # Softmax over peaks (higher values get more weight)
    threshold = max_val * 0.99
    active = conv >= threshold
    if not active.any():
        active = np.zeros(len(conv), dtype=bool)
        active[np.argmax(conv)] = True
    
    # Weight by how close to max each peak is
    weights = np.where(active, conv, 0.0)
    weights = weights ** 10  # Sharpen the weights toward the true max
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    # grad_i = 2 * sum_k w_k * a[k-i] (for valid k-i)
    # This is a weighted cross-correlation
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def bb_step(prev_a, prev_grad, curr_a, curr_grad, gamma_lo=1e-6, gamma_hi=1e4):
    """Barzilai-Borwein step size."""
    s = curr_a - prev_a
    y = curr_grad - prev_grad
    s_norm2 = np.dot(s, s)
    if s_norm2 < 1e-20:
        return 0.01
    ys = np.dot(y, s)
    if ys < 1e-15:
        return 0.01
    # BB+ (use larger step)
    gamma = s_norm2 / min(ys, 1e10)
    return np.clip(gamma, gamma_lo, gamma_hi)


def optimize_subgradient(a0, time_limit, seed=0, top_k=5):
    """
    Subgradient method with BB step size and adaptive perturbation.
    Input: a0 on simplex (sum=1, non-negative).
    Returns (best_a, best_max_conv).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    prev_a = a.copy()
    prev_grad = subgradient_max_conv(a, conv, top_k)

    t0 = time.monotonic()
    gamma = 0.05
    no_improve = 0
    total_iters = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_obj = conv.max()
        grad = subgradient_max_conv(a, conv, top_k)
        total_iters += 1

        # BB step size update
        if total_iters > 1 and no_improve < 10:
            gamma = bb_step(prev_a, prev_grad, a, grad)
            gamma = np.clip(gamma, 1e-5, 0.5)

        prev_a = a.copy()
        prev_grad = grad.copy()

        # Gradient step with simplex projection
        a_new = project_simplex(a - gamma * grad)
        
        # Check if step improved
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive perturbation
        if no_improve >= 50 + 20 * perturb_count:
            scale = 0.01 / (1 + 0.5 * perturb_count)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            gamma = max(0.005, gamma * 0.5)
            no_improve = 0
            perturb_count += 1

        # Decay step size slowly
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        gamma *= (1.0 - 0.001)

    return best_a, best_obj


def optimize_logsumexp(a0, time_limit, beta_init=50.0, beta_end=300.0,
                       lr_init=0.01, momentum=0.9, seed=0):
    """
    Log-sum-exp smoothed accelerated gradient descent.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)
        
        # Smoothed gradient
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum

        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]

        # Momentum update
        lr = lr_init * (1.0 - 0.5 * frac)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0

    return best_a, best_obj


def peak_deflation(a0, time_limit, seed=0):
    """
    Targeted peak reduction via coordinate descent on the most responsible pairs.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:

        conv = fast_convolve(a)
        max_val = conv.max()

        # Find top peaks
        top_k = min(3, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[conv[peak_indices] > max_val * 0.98]

        if len(peak_indices) == 0:
            break

        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            if i_start > i_end:
                continue

            # Find the most responsible index pair
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(contributions) == 0:
                continue

            # Try the top 3 most responsible indices
            idx_order = np.argsort(contributions)[::-1][:3]

            for idx in idx_order:
                i_max = i_start + int(idx)

                # Try moving mass from i_max to a "good" destination
                # Good destination: one that doesn't contribute much to any top peak
                candidates = []
                for _ in range(20):
                    j = int(rng.integers(0, n))
                    if j == i_max:
                        continue
                    # Score: how much does adding to j increase the max?
                    candidates.append(j)

                for j in candidates[:5]:
                    for frac in [0.03, 0.05, 0.08, 0.12]:
                        amt = a[i_max] * frac
                        if amt < 1e-9:
                            break
                        a_try = a.copy()
                        a_try[i_max] -= amt
                        a_try[j] += amt
                        s_try = a_try.sum()
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj - 1e-10:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved and rng.random() < 0.1:
            # Random perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = a[i] * 0.05

                    a_try[i] -= amt

                    if s_try > 1e-10:

                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()

    return best_a, best_obj


def lp_direction_improved(a, M, n, time_limit_lp=20):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)

    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum

def crossover_refine(a1, a2, time_limit, seed=0):
    """
    Create a crossover of two sequences and refine.
    Uses piecewise interpolation in the "index space".
    """
    rng = np.random.default_rng(seed)
    n = max(len(a1), len(a2))

    # Resize both to n
    a1_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a1)), a1)
    a2_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a2)), a2)
    a1_r = np.maximum(a1_r, 0)
    a2_r = np.maximum(a2_r, 0)

    best_a = None
    best_obj = np.inf

    # Try multiple crossover fractions
    fracs = [0.3, 0.4, 0.5, 0.6, 0.7]
    for frac in fracs:
        cross = (1 - frac) * a1_r + frac * a2_r
        cross = np.maximum(cross, 0)
        s = cross.sum()
        if s < 1e-10:
            continue
        cross = cross / s
        obj = fast_convolve(cross).max()
        if obj < best_obj:
            best_obj = obj
            best_a = cross.copy()

    if best_a is None:
        best_a = a1_r / a1_r.sum()

    # Refine with log-sum-exp

            best_a, remaining, beta_init=80, beta_end=400,

    return best_a, best_obj


def generate_initials(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Power law decays (asymmetric)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        pw = (1.0 - x / (n - 1)) ** p
        pw = np.maximum(pw, 1e-6)
        pw /= pw.sum()
        inits[f'power_{p}'] = pw

    # Sine-based asymmetric
    sine = np.sin(np.pi * x / (2 * (n - 1)))
    sine = np.maximum(sine, 1e-6)
    sine /= sine.sum()
    inits['sine'] = sine

    # Exponential decay
    exp_d = np.exp(-3.0 * x / n)
    exp_d /= exp_d.sum()
    inits['exp_decay'] = exp_d

    # Bimodal (concentrated at both ends)
    bi = np.exp(-20.0 * ((x - n * 0.1) / (n * 0.15)) ** 2) + \
         np.exp(-20.0 * ((x - n * 0.9) / (n * 0.15)) ** 2)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet
    inits['dirichlet_03'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['dirichlet_1'] = rng.dirichlet(np.ones(n) * 1.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_trunc'] = pp / ps
        else:
            pp = np.maximum(parent, 0)
            pp_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
            ps = pp_r.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_r / ps
            # Also try full parent
            inits['parent_full'] = pp / ps

    return inits

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_sub.copy()
                print(f"    SubGrad: {score:.6f}", flush=True)
            a_cur = a_sub

        # Step B: Log-sum-exp with high beta (25%)
        t_b = t_alloc * 0.25
        if t_b > 10:
            a_lse, _ = optimize_logsumexp(
                a_cur, t_b, beta_init=100, beta_end=500,
                lr_init=0.005, momentum=0.94, seed=int(rng.integers(0, 99999)))
            score = float(evaluate_sequence(a_lse.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lse.copy()
                print(f"    LSE: {score:.6f}", flush=True)
            a_cur = a_lse

        # Step C: LP direction (25%)
        t_c = t_alloc * 0.25
        if t_c > 10:
            a_lp, _ = optimize_lp(a_cur, t_c, seed=int(rng.integers(0, 99999)))
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_lp.copy()
                print(f"    LP: {score:.6f}", flush=True)
            a_cur = a_lp

        # Step D: Peak deflation (15%)
        t_d = t_alloc * 0.15
        if t_d > 8:
            a_pd, _ = peak_deflation(a_cur, t_d, seed=int(rng.integers(0, 99999)))
            score = float(evaluate_sequence(a_pd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pd.copy()
                print(f"    PeakDef: {score:.6f}", flush=True)
            a_cur = a_pd

        # Update pool
        if i + 1 < n_pool:
            pool[i] = a_cur

    print(f"Phase 2 done. Best: {best_score:.6f}", flush=True)

    # Phase 3: Polish (80s)
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Crossover between top candidates if we have them
        if len(scan_results) >= 2 and remaining > 40:
            a1 = best_sequence
            a2 = scan_results[1][3] if scan_results[1][3] is not best_sequence else scan_results[0][3]
            if len(a1) != len(a2):
                a2 = np.interp(np.linspace(0, 1, len(a1)), np.linspace(0, 1, len(a2)), a2)
                a2 = np.maximum(a2, 0)
                a2 = a2 / a2.sum()
            t_xo = min(30.0, remaining * 0.3)
            if t_xo > 10:
                a_cross, _ = crossover_refine(a1, a2, t_xo, seed=int(rng.integers(0, 99999)))
                score = float(evaluate_sequence(a_cross.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cross.copy()
                    print(f"  Crossover: {score:.6f}", flush=True)

        # Final subgradient polish
        remaining = deadline - time.monotonic() - 5.0
        t_fin = min(remaining * 0.5, 30.0)
        if t_fin > 5:
            a_fin, _ = optimize_subgradient(
                best_sequence, t_fin, seed=int(rng.integers(0, 99999)), top_k=3)
            score = float(evaluate_sequence(a_fin.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_fin.copy()
                print(f"  FinalSub: {score:.6f}", flush=True)

        # Final peak deflation
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 5:
            a_pd2, _ = peak_deflation(best_sequence, remaining, seed=7777)
            score = float(evaluate_sequence(a_pd2.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pd2.copy()
                print(f"  FinalPD: {score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384)

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(384)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
