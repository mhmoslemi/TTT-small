"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using fast FFT-based gradient descent
with Nesterov momentum, adaptive log-sum-exp smoothing, MCMC polish, and
multi-phase multi-length search strategy.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, w):
    """Compute gradient of sum_k w[k]*c[k] where c = a*a (linear conv).
    Returns 2 * cross_correlation(w, a) truncated to length n."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    g = np.fft.ifft(W * np.conj(A)).real[:n]
    return 2.0 * g


def gradient_optimize(a0, time_limit, seed=0, beta_init=200.0, lr_init=None,
                      momentum_init=0.90, max_beta=10000.0, verbose=False):
    """Smooth gradient descent with Nesterov momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if lr_init is None:
        lr_init = 5.0 / n

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Softmax weights over lags
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Gradient
        g = compute_smooth_grad(a, w)

        # Momentum update
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                lr *= 0.85
            if no_improve > 50:
                # Perturb and reset
                noise = rng.standard_normal(n) * (0.05 / n)
                a = project_simplex(best_a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.7
                momentum = min(0.95, momentum_init + 0.02)
                no_improve = 0

        # Gradually increase beta (sharper peak approximation)
        beta = min(max_beta, beta * 1.003)
        # Gradually increase momentum
        momentum = min(0.97, momentum + 2e-5)

    conv_final = fast_convolve(best_a)
    final_obj = conv_final.max()
    return best_a, final_obj


def mcmc_polish(a0, time_limit, seed=0, verbose=False):
    """MCMC-style random perturbation polish on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Adaptive perturbation scale
    scale = 0.5 / n
    accept_count = 0
    total_count = 0

    while time.monotonic() - t0 < time_limit:
        # Batch of random perturbations
        for _ in range(50):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            t = rng.uniform(-scale, scale)
            a_new = a.copy()
            a_new[i] -= t
            a_new[j] += t
            if a_new[i] < 0 or a_new[j] < 0:
                continue

            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            total_count += 1

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                accept_count += 1
            elif obj_new < best_obj + scale * 0.01:
                # Accept with small probability (helps exploration)
                if rng.random() < 0.05:
                    a = a_new
                    accept_count += 1

        # Adapt scale
        if total_count > 100:
            rate = accept_count / total_count
            if rate < 0.01:
                scale *= 0.8
            elif rate > 0.1:
                scale *= 1.2
            scale = max(0.01 / n, min(2.0 / n, scale))
            if verbose and accept_count % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MCMC: obj={best_obj:.8f}, scale={scale:.4f}, "
                      f"accept_rate={rate:.3f}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_ascent_polish(a0, time_limit, seed=0, verbose=False):
    """Cyclic coordinate-ascent style polish: try small transfers between pairs."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # Try transfers of various sizes
    transfer_sizes = [0.01/n, 0.005/n, 0.001/n, 0.05/n, 0.1/n]

    while time.monotonic() - t0 < time_limit:
        # Pick random pairs
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for t_scale in transfer_sizes:
                for sign in [1, -1]:
                    t = sign * t_scale
                    a_new = a.copy()
                    a_new[i] += t
                    a_new[j] -= t
                    if a_new[i] < 0 or a_new[j] < 0:
                        continue
                    conv_new = fast_convolve(a_new)
                    obj_new = conv_new.max()
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        break  # Found improvement, try next pair
                else:
                    continue
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()

    # Edge-concentrated
    edge = np.zeros(n)
    w = max(1, n // 6)
    edge[:w] = 1.0
    edge[-w:] = 1.0
    edge /= edge.sum()
    inits['bimodal'] = edge

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Random Dirichlet (concentrated)
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 0.3)
    # Random Dirichlet (spread)
    inits['random_spread'] = rng.dirichlet(np.ones(n) * 3.0)
    # Random Dirichlet (very concentrated)
    inits['random_vconc'] = rng.dirichlet(np.ones(n) * 0.1)

    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.001)
        cos_w /= cos_w.sum()
    else:
        cos_w = np.ones(1)
    inits['cosine'] = cos_w

    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10 and len(parent) == n:
            inits['parent'] = pp / ps
        elif ps > 1e-10:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), pp)
            pp_interp = np.maximum(pp_interp, 0)
            ps2 = pp_interp.sum()
            if ps2 > 1e-10:
                inits['parent_interp'] = pp_interp / ps2

    return inits


def evaluate_obj_normalized(a):

def warm_start(a0, n_steps=50, beta=50.0, lr=1.0, seed=0):
    """Quick gradient steps to 'pre-flatten' the autocorrelation."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    for _ in range(n_steps):
        conv = fast_convolve(a)
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            break
        g = compute_smooth_grad(a, w)
        a = project_simplex(a - lr * g / n)

    return a

        for i, a_start in enumerate(mcmc_starts):
            if time.monotonic() - phase3_start > phase3_time - 10:
                break
            t_alloc = min(time_per_mcmc, phase3_time - (time.monotonic() - phase3_start) - 5)
            if t_alloc < 10:
                break

            a_opt, obj_opt = mcmc_polish(a_start, time_limit=t_alloc, seed=200 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MCMC [{i}]: score={score:.6f}, t={time.monotonic()-phase3_start:.0f}s")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Coordinate / Fine Polish ==========
    print("=== Phase 4: Fine Polish ===")
    phase4_start = time.monotonic()
    phase4_time = deadline - time.monotonic() - 30.0
    phase4_time = max(50.0, min(phase4_time, 300.0))

    if best_sequence is not None:
        # Coordinate ascent polish
        t_alloc = min(phase4_time * 0.4, 100.0)
        if t_alloc > 10:
            a_opt, obj_opt = coordinate_ascent_polish(
                best_sequence, time_limit=t_alloc, seed=500)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordAscent: score={score:.6f}")

        # Another MCMC round with smaller scale
        remaining = phase4_time - (time.monotonic() - phase4_start) - 5
        if remaining > 20:
            a_opt, obj_opt = mcmc_polish(best_sequence, time_limit=remaining, seed=600)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MCMC2: score={score:.6f}")

        # Final tiny perturbations
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            n_b = len(best_sequence)
            n_tries = min(int(remaining / 0.05), 100)
            for i in range(n_tries):
                if time.monotonic() > deadline - 20:
                    break
                for scale in [0.002/n_b, 0.005/n_b, 0.01/n_b]:
                    noise = rng.standard_normal(n_b) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  TinyPert [{i}] s={scale:.6f}: score={best_score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
