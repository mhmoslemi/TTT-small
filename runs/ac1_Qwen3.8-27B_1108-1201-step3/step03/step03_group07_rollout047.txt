def smooth_max_weights(x, beta):
    """Compute smooth max and softmax weights with stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)

def grad_conv_weighted(a, w):

    N = 1 << (len(w) + n - 1 + 63).bit_length()

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

def generate_flat_spectrum_init(n, rng):
    """Initialize with flat power spectrum for low-peak convolution."""

    A[0] = np.sqrt(num_ones)
    for k in range(1, num_ones):
        phase = rng.uniform(0, 2 * np.pi)
        mag = 1.0
        A[k] = mag * np.exp(1j * phase)
        if k < fft_size - k:
            A[fft_size - k] = mag * np.exp(-1j * phase)
    if num_ones < fft_size // 2:
        A[num_ones] = np.exp(1j * rng.uniform(0, 2 * np.pi))

            unique_lengths.append(n)

    for n in unique_lengths:
        if time.monotonic() > phase2_deadline - 30:
            break

        # Get best init for this length
        best_init = None
        init_score = float('inf')
        for s, nn, a in scan_results:
            if nn == n and s < init_score:
                init_score = s
                best_init = a.copy()
        if best_init is None:
            best_init = generate_flat_spectrum_init(n, rng)

        if parent is not None:
            if len(parent) == n:
                cand = parent.copy()
            else:
                cand = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s > 1e-10:
                    cand /= s
            score_c = evaluate_sequence(cand.tolist())
            if np.isfinite(score_c) and score_c < init_score:
                best_init = cand
                init_score = score_c

        print("  Optimizing n=%d (init score=%.6f)" % (n, init_score))

        # Multi-beta L-BFGS-B with aggressive schedule
        betas = [100.0, 50.0, 30.0, 20.0, 10.0, 5.0, 3.0]
        a_cur = best_init.copy()
        for beta in betas:
            if time.monotonic() > phase2_deadline - 20:
                break
            maxiter = max(50, min(300, int((phase2_deadline - time.monotonic() - 10) / 3)))
            a_cur, _ = multi_beta_lbfgs(a_cur, [beta], maxiter=maxiter)
            update_best(a_cur)

        # Peak suppression
        if time.monotonic() < phase2_deadline - 30:
            a_ps, _ = peak_suppression(a_cur, n_rounds=30, top_k=5)
            update_best(a_ps)
            if obj_value(a_ps) < obj_value(a_cur):
                a_cur = a_ps

        # Another L-BFGS pass
        if time.monotonic() < phase2_deadline - 20:
            a_cur, _ = multi_beta_lbfgs(a_cur, [5.0, 3.0, 2.0], maxiter=200)
            update_best(a_cur)

    print("Phase 2 done. Best: %.6f" % best_score)

    # === Phase 3: Evolutionary search (400s) ===
    print("=== Phase 3: Evolutionary Search ===")
    phase3_deadline = t_start + 1150.0

    if best_sequence is not None:
        evo_budget = max(30.0, phase3_deadline - time.monotonic() - 10.0)
        print("  Evo budget: %.0fs" % evo_budget)
        best_sequence, best_score = evolutionary_search(best_sequence, evo_budget, rng)

    print("Phase 3 done. Best: %.6f" % best_score)

    # === Phase 4: Final polish (remaining time) ===
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Coordinate descent
        if remaining > 30:
            cd_iter = int(min(30, remaining / 2.0))
            a_cd, _ = coordinate_descent(best_sequence, n_sweeps=cd_iter)
            score = update_best(a_cd)
            if score <= best_score:
                best_sequence = a_cd

        # Very fine L-BFGS
        if remaining > 20:
            a_ref, _ = multi_beta_lbfgs(best_sequence, [2.0, 1.5, 1.0], maxiter=200)
            update_best(a_ref)

        # Final peak suppression
        if remaining > 15:
            a_ps, _ = peak_suppression(best_sequence, n_rounds=15, top_k=8)
            update_best(a_ps)

        # One more coordinate descent
        if remaining > 15:
            a_cd, _ = coordinate_descent(best_sequence, n_sweeps=10)
            update_best(a_cd)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final validation
    final_score = evaluate_sequence(best_sequence.tolist())
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
