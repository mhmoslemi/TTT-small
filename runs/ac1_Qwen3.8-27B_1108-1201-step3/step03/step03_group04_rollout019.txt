import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=10.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def fpw_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit:
        # Try different M values
        M_factors = [1.0, 0.95, 0.9]
        best_direction = None
        best_dir_sum = 0
        
        for mf in M_factors:
            b = lp_oracle(a, M * mf, time_limit=5.0)
            if b is not None and b.sum() > best_dir_sum:
                best_dir_sum = b.sum()
                best_direction = b
        
        if best_direction is None or best_dir_sum < 1e-15:
            # Fallback: use subgradient
            b = -logsumexp_smoothed_grad(a, tau=0.01 * M)[1]
            b = np.maximum(b, 0)
            if b.sum() < 1e-15:
                break
            best_direction = b
            best_dir_sum = b.sum()
        
        b = best_direction / best_dir_sum
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_local = 2.0 * n * fast_convolve(a).max()
            for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
                cand = (1 - t) * a + t * b
                obj = 2.0 * n * fast_convolve(cand).max()
                if obj < best_local:
                    best_local = obj
                    best_t = t
            
            if best_t < 0.05:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 12345)
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj

import numpy as np

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=8.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_subgradient(p, top_k=25):
    """Compute subgradient of max_k conv(p,p)_k using top-k peaks with softmax weights."""
    n = len(p)
    conv = fast_convolve(p)
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    # Softmax weights (temperature based on spread)
    temp = max((peak_vals.max() - peak_vals.min()) * 0.1, 1e-10)
    w = np.exp((peak_vals - peak_vals.max()) / temp)
    w /= w.sum()

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        kk = int(peaks[i])
        idx_from = max(0, kk - n + 1)
        idx_to = min(n - 1, kk)
        if idx_from <= idx_to:
            i_range = np.arange(idx_from, idx_to + 1)
            grad[i_range] += 2.0 * w[i] * p[kk - i_range]

    return grad, conv

        priority = ['parent', 'edge_8', 'edge_15', 'two_peak', 'log_edge',
                    'uniform', 'triangle', 'gauss', 'random', 'sparse', 'edge_25']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        for name in init_names[:4]:
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(15.0, deadline - time.monotonic() - 30)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = fpw_optimize(a0, time_limit=t_alloc, verbose=False, lp_time=3.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  n={n}, init={name}: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    t1_elapsed = time.monotonic() - t1_start
    print(f"After Phase 1: best={best_score:.6f} ({t1_elapsed:.0f}s)")

    # === Phase 2: Deep LP-FW on best (main optimization) ===
    if best_sequence is not None:
        print("=== Phase 2: Deep LP-FW ===")
        remaining = deadline - time.monotonic() - 40.0
        t_deep = remaining * 0.35
        if t_deep > 15:
            a_opt, obj_opt = fpw_optimize(best_sequence, time_limit=t_deep, verbose=True, lp_time=6.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Smoothed gradient (different landscape navigation) ===
    if best_sequence is not None:
        print("=== Phase 3: Smoothed Gradient ===")
        remaining = deadline - time.monotonic() - 40.0
        t_smooth = remaining * 0.2
        if t_smooth > 10:
            a_opt, obj_opt = smoothed_optimize(best_sequence, time_limit=t_smooth, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: More LP-FW (continue from phase 3) ===
    if best_sequence is not None:
        print("=== Phase 4: LP-FW Continue ===")
        remaining = deadline - time.monotonic() - 30.0
        t_cont = remaining * 0.3
        if t_cont > 10:
            a_opt, obj_opt = fpw_optimize(best_sequence, time_limit=t_cont, verbose=True, lp_time=8.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Perturbation restarts ===
    if best_sequence is not None:
        print("=== Phase 5: Perturbations ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 15.0
        n_restarts = max(1, int(remaining // 18))
        n_restarts = min(n_restarts, 5)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 12:
                break
            # Structured perturbation based on current peaks
            conv = fast_convolve(best_sequence)
            top_peaks = np.argpartition(conv, -8)[-8:]
            noise = rng.standard_normal(n_best) * 0.001 * (1 + i * 0.5)
            # Add targeted perturbation
            for pk in top_peaks:
                idx_from = max(0, int(pk) - n_best + 1)
                idx_to = min(n_best - 1, int(pk))
                if idx_from <= idx_to:
                    i_range = np.arange(idx_from, idx_to + 1)
                    noise[i_range] += 0.001 * best_sequence[int(pk) - i_range]

            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(15.0, deadline - time.monotonic() - 10)
            if t_alloc < 3:
                break
            a_opt, obj_opt = fpw_optimize(a_pert, time_limit=t_alloc, verbose=False, lp_time=4.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")
                else:
                    print(f"  Restart {i}: score={score:.6f} (no improvement)")

    # === Phase 6: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("=== Phase 6: Final Polish ===")
        remaining = deadline - time.monotonic() - 8.0
        if remaining > 5:
            a_opt, obj_opt = fpw_optimize(best_sequence, time_limit=remaining, verbose=False, lp_time=4.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 6 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
