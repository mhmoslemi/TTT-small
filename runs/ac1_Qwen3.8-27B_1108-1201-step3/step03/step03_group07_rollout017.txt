"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy combining:
1. Directed simplex gradient descent with adaptive smooth-max approximation
2. Frequency-domain spectral shaping for initialization
3. Coarse-to-fine multi-scale optimization
4. Peak-targeted coordinate descent
5. Simulated annealing with heavy-tailed perturbations
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(1.0, 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.argmax(~cond))
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_and_grad(a, beta):
    """Returns (smooth objective, gradient) for 2n * smooth_max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    x_max = conv.max()
    shifted = np.clip(beta * (conv - x_max), -500.0, 500.0)
    e = np.exp(shifted)
    total = e.sum()
    if total < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
        sm = x_max
    else:
        w = e / total
        sm = x_max + np.log(total) / beta
    N = 1 << (len(conv) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return 2.0 * n * sm, grad


def true_score_obj(a):
    """Exact objective value (not smoothed)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def optimize_directed(a0, start_beta, end_beta, deadline, rng=None):
    """
    Directed gradient descent on simplex with annealing beta.
    Uses backtracking line search with momentum.
    """
    n = len(a0)
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a0, 1e-15).copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = true_score_obj(a)
    momentum = np.zeros(n)
    mu = 0.7
    lr = 0.1

    while time.monotonic() < deadline:
        beta = start_beta * (end_beta / start_beta) ** (a.sum() * 0 + min(1.0, (time.monotonic() - (deadline - 1740.0)) / 1740.0))
        beta = max(end_beta, min(start_beta, beta))

        obj, grad = objective_and_grad(a, beta)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break

        momentum = mu * momentum + (1 - mu) * grad
        direction = momentum

        t = lr
        improved = False
        for _ in range(12):
            a_new = project_simplex(a - t * direction)
            obj_new = true_score_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            if obj_new <= cur_obj + 1e-12:
                a = a_new
                improved = True
                break
            t *= 0.4

        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_a, best_obj


def spectral_init(n, rng, n_trials=30):
    """Generate initial sequence via randomized spectral shaping."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = 1.0
        n_harm = fft_size // 2
        mags = rng.uniform(0.3, 1.0, n_harm)
        phases = rng.uniform(0, 2 * np.pi, n_harm + 1)
        for k in range(1, n_harm + 1):
            A[k] = mags[k-1] * np.exp(1j * phases[k-1])
            if k < fft_size - k:
                A[fft_size - k] = mags[k-1] * np.exp(-1j * phases[k-1])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a /= s
        obj = true_score_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a


def make_init_flat_spec(n, rng):
    """Flat-spectrum style initialization."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    for _ in range(15):
        A = np.ones(fft_size, dtype=complex)
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        A *= np.exp(1j * phases)
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a /= s
        obj = true_score_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a


def shape_from_template(n, template_type, rng):
    """Generate shaped initializations."""
    x = np.arange(n, dtype=np.float64)
    if template_type == 'asym_ramp':
        a = np.arange(n, 0, -1, dtype=np.float64)
    elif template_type == 'asym_ramp2':
        a = np.exp(-x / (n * 0.4))
    elif template_type == 'asym_ramp3':
        a = 1.0 / (1.0 + x / (n * 0.3))
    elif template_type == 'asym_ramp4':
        a = np.sqrt(1.0 - x / n) + 0.1
    elif template_type == 'u_shape':
        a = (x / n - 0.5) ** 2 * 4 + 0.3
    elif template_type == 'v_shape':
        a = np.abs(x / n - 0.5) + 0.2
    elif template_type == 'sin_asym':
        a = np.sin(np.pi * x / n * 0.7 + 0.5) + 0.1
    elif template_type == 'bump_left':
        a = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2)
    elif template_type == 'bump_right':
        a = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if rng is not None:
        a = a * rng.uniform(0.7, 1.3, n)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent(a, n_passes=3, deadline=None, rng=None):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes the peak of the convolution.
    """
    if rng is None:
        rng = np.random.default_rng(1)
    n = len(a)
    a = a.copy()
    best_obj = true_score_obj(a)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[i]
            a[i] = 0.0
            rest_sum = 1.0  # since we normalize to sum=1, setting a[i]=0 means others sum to 1
            # Binary search for best a[i]
            lo, hi = 0.0, old_val * 3.0 + 0.01
            best_local = old_val
            best_local_obj = best_obj
            for _ in range(12):
                mid = (lo + hi) / 2.0
                a[i] = mid
                a_tmp = a.copy()
                # Renormalize
                s = a_tmp.sum()
                if s < 1e-10:
                    a[i] = old_val
                    break
                a_tmp = a_tmp / s
                obj_candidate = true_score_obj(a_tmp)
                if obj_candidate < best_local_obj:
                    best_local_obj = obj_candidate
                    best_local = mid
                if mid < old_val:
                    lo = mid
                else:
                    hi = mid
            a[i] = best_local
            s = a.sum()
            if s > 1e-10:
                a = a / s
            obj_now = true_score_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now

    return a, best_obj


def noisy_restart(a, n_restarts, strength, deadline, rng, start_beta, end_beta):
    """Perturb a and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_obj = true_score_obj(a)

    for _ in range(n_restarts):
        if time.monotonic() > deadline:
            break
        alpha = rng.exponential(0.003 * strength)
        noise = rng.standard_normal(n) * alpha
        a_pert = project_simplex(a + noise)
        sub_deadline = min(deadline, time.monotonic() + 40.0)
        a_opt, obj_opt = optimize_directed(a_pert, start_beta, end_beta, sub_deadline, rng)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
    return best_a, best_obj
