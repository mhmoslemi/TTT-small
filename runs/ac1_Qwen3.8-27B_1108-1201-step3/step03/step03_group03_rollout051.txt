For the specific case of non-negative sequences with sum 1, the minimum possible max auto-convolution is... I think it's related to the "flat-top" window or similar constructions. Let me think...

If we use a "top-hat" (rectangular) window: a[i] = 1/n for all i. Then conv[k] = (n-|k-(n-1)|)/n^2 for k in [0, 2n-2]. The max is 1/n at k=n-1.

If we use a triangular window: a[i] = c*(i+1) for i < n/2, c*(n-i) for i >= n/2 (normalized). The convolution of a triangle is a "parabola" (or B-spline of order 2). The max is still at the center.

I think the optimal non-negative sequence for this problem is related to "flat-top" windows or "minimum sidelobe" sequences. In the continuous setting, the optimal would be a sinc-like function, but we're constrained to be non-negative and have finite support.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Multiplicative weight updates (mirror descent on simplex)
2. Subgradient descent with momentum
3. LP-based directional optimization
4. Coordinate descent polishing

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def multiplicative_weight_step(a, conv, eta, softmax_temp=0.0):
    """One step of multiplicative weight (mirror descent) on the simplex.
    Uses soft weighting of peaks based on softmax of convolution values."""
    n = len(a)
    if softmax_temp > 0:
        # Soft weighting: weight each lag by exp(conv[k] / temp)
        scaled = conv / softmax_temp
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()
    else:
        # Hard: only the peak
        k_star = np.argmax(conv)
        weights = np.zeros(2 * n - 1)
        weights[k_star] = 1.0
    
    # Compute subgradient: sg[m] = 2 * sum_k weights[k] * a[k-m] (if valid)
    # This is 2 * (weights_reflected * a) where weights_reflected is the reversal of weights
    # conv(weights_reversed, a) gives sum_k weights_reversed[k] * a[k-m]... 
    # Actually, we want: sg[m] = 2 * sum_k w[k] * a[k-m] * I(0<=k-m<n)
    # This is 2 * (w * a)[m] where * is cross-correlation, or equivalently:
    # sg = 2 * conv(flipped_w, a) ... let me think.
    # 
    # sg[m] = 2 * sum_k w[k] * a[k-m]
    # Let's define b[k] = w[k]. Then sg[m] = 2 * sum_k b[k] * a[k-m]
    # This is the cross-correlation of b and a, which is conv(b, reversed_a)[m + n - 1]
    # Or equivalently: conv(reversed_b, a) gives sum_k reversed_b[k] * a[k-m] = sum_k b[n_b-1-k] * a[k-m]
    # That's not quite right either.
    #
    # Let me just use: sg[m] = 2 * sum_{k: 0<=k-m<n} w[k] * a[k-m]
    # Let j = k-m, so k = j+m: sg[m] = 2 * sum_{j: 0<=j<n} w[j+m] * a[j]
    # This is 2 * (w_shifted * a)[m] where w_shifted is w shifted right by m...
    # Actually this IS the convolution of w and a!
    # conv(w, a)[m+n-1] = sum_j w[j] * a[m+n-1-j]
    # Hmm, that's not quite it.
    #
    # Let me just compute it directly using the relationship:
    # sg[m] = 2 * sum_k w[k] * a[k-m] * I(0<=k-m<n)
    # = 2 * (w * a)[m] where (w*a) is the "same" mode convolution starting at offset m
    #
    # Actually, the simplest way: sg = 2 * conv(w_full, a) but with appropriate slicing.
    # conv(w, a)[k] = sum_j w[j]*a[k-j]
    # We want: sum_k w[k]*a[k-m] = sum_k w[k]*a[k-m]
    # Let l = k-m: = sum_l w[l+m]*a[l] = conv_flipped(w, a)[m]... 
    # 
    # You know what, let me just use the fact that:
    # sum_k w[k] * a[k-m] = (w cross-correlated with a)[m] = conv(flip(w), a)[m + len(w) - 1]
    # where flip(w) = w[::-1]
    # 
    # But w has length 2n-1 and a has length n. This is getting complicated.
    # Let me just use a direct O(n) computation for each m... no, that's O(n^2).
    #
    # Better: use the FFT. Define w_padded and a_padded, do the convolution.
    
    # Actually, the cleanest way:
    # sg[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]  (where w is extended with zeros)
    # This is the cross-correlation, which equals conv(flip(w), a) with appropriate indexing.
    
    # Let w_ext = w (length 2n-1), a_ext = a (length n)
    # conv(w_ext_flipped, a_ext) has length (2n-1) + n - 1 = 3n-2
    # conv(w_ext_flipped, a_ext)[k] = sum_j w_ext_flipped[j] * a_ext[k-j]
    # = sum_j w_ext[2n-2-j] * a_ext[k-j]
    # We want sg[m]/2 = sum_j w[m+j] * a[j]
    # Let's set k-j = j' and 2n-2-j = m+j', so j = 2n-2-m-j' and k = j + j' = 2n-2-m
    # So conv(w_ext_flipped, a_ext)[2n-2-m] = sum_{j'} w[m+j'] * a[j']  ... let me verify.
    # conv(w_flipped, a)[k] = sum_j w_flipped[j] * a[k-j] = sum_j w[2n-2-j] * a[k-j]
    # Let i = 2n-2-j, so j = 2n-2-i:
    # = sum_i w[i] * a[k-2n+2+i]
    # We want this to equal sum_i w[i] * a[i-m] (i.e., k-2n+2+i = i-m, so k = 2n-2-m)
    # Yes! So sg[m]/2 = conv(w_flipped, a)[2n-2-m]
    
    w_flipped = weights[::-1]
    # conv(w_flipped, a) via FFT
    size = len(w_flipped) + len(a) - 1
    fft_size = 2 ** (size - 1).bit_length()
    W = np.fft.rfft(w_flipped, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_wa = np.fft.irfft(W * A, fft_size)[:size]
    
    # sg[m] = 2 * conv_wa[2n-2-m] for m in [0, n-1]
    # 2n-2-m ranges from 2n-2 (m=0) to n-1 (m=n-1)
    sg = 2.0 * conv_wa[2*n-2 : n-1 : -1]  # m=0 gets index 2n-2, m=n-1 gets index n-1
    # Wait, that's reversed. Let me be careful.
    # For m=0: index = 2n-2-0 = 2n-2
    # For m=n-1: index = 2n-2-(n-1) = n-1
    # So sg[m] = 2 * conv_wa[2n-2-m]
    # sg = 2 * conv_wa[n-1:n-1+(n):...] no...
    # sg[0] = 2*conv_wa[2n-2], sg[1] = 2*conv_wa[2n-3], ..., sg[n-1] = 2*conv_wa[n-1]
    # So sg = 2 * conv_wa[n-1 : 2n-1 : 1][::-1] = 2 * conv_wa[n-1:2n-1][::-1]
    
    sg = 2.0 * conv_wa[n-1:2*n-1][::-1]
    
    # Multiplicative weight update
    a_new = a * np.exp(-eta * sg)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-300:
        # Overflow/underflow, fall back to uniform
        a_new = np.ones(n) / n
    else:
        a_new = a_new / a_new_sum
    
    return a_new


def optimize_mwu(a0, time_limit, eta_init=0.5, eta_decay=0.999, 
                 softmax_temp=0.0, soft_weight=False):
    """Optimize using multiplicative weight updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        a_new = multiplicative_weight_step(a, conv, eta, softmax_temp)
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= eta_decay
        
        if no_improve > 500:
            # Perturb and restart
            rng = np.random.default_rng(hash(tuple(np.round(a[:20], 4))) % (2**31))
            a = project_simplex(a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            no_improve = 0
            eta = eta_init * 0.5
    
    return best_a, best_obj


def optimize_subgrad_momentum(a0, time_limit, lr_init=0.1, momentum=0.9,
                              top_k=20, soft_weight=False, temp=0.05):
    """Subgradient descent with momentum, using soft or hard peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if soft_weight:
            # Soft weighting of all lags
            scaled = conv / (temp * conv.max() + 1e-15)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            w = np.zeros(2 * n - 1)
            peak_vals = conv[peaks]
            # Weight peaks by their values
            w[peaks] = peak_vals
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w[np.argmax(conv)] = 1.0
        
        # Compute subgradient
        w_flipped = w[::-1]
        size = len(w_flipped) + len(a) - 1
        fft_size = 2 ** (size - 1).bit_length()
        W = np.fft.rfft(w_flipped, fft_size)
        A = np.fft.rfft(a, fft_size)
        conv_wa = np.fft.irfft(W * A, fft_size)[:size]
        sg = 2.0 * conv_wa[n-1:2*n-1][::-1]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
        
        lr *= 0.9995
        if no_improve > 100:
            lr = max(0.001, lr * 0.5)
            no_improve = 0
    
    return best_a, best_obj


def build_lp_matrix(n):
    """Build the constraint matrix for the LP: (f*g)[k] <= M for all k."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            # (f*g)[k] = sum_j f[j] * g[k-j]
            # A[k, j] = f[k-j]
            A[k, j] = 0  # Will be filled in solve
    return None  # We'll build it on the fly


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, blend_init=0.5, blend_min=0.005,
                blend_halve_every=12):
    """LP-based directional optimization."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
    
    return best_a, best_obj


def coordinate_polish(a_in, max_rounds=30, delta_frac=0.1):
    """Coordinate descent: try moving mass between pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Find the peak
        k_star = np.argmax(conv)
        # Try moving mass from positions that contribute to the peak
        # to positions that don't
        peak_contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            peak_contrib[j] = a[k_star - j]
        
        # Sort by contribution (high first)
        order = np.argsort(peak_contrib)[::-1]
        
        for i_idx in range(min(20, n)):
            i = order[i_idx]
            if a[i] < 1e-8:
                continue
            # Try moving to low-contribution positions
            for j_idx in range(n - 1, max(n - 21, -1), -1):
                j = order[j_idx]
                if j == i:
                    continue
                delta = a[i] * delta_frac
                a[i] -= delta
                a[j] += delta
                if a[i] < 0:
                    a[i] += delta
                    a[j] -= delta
                    continue
                conv_test = fast_convolve(a)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    edge = np.zeros(n)
    e = max(1, n // 6)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + 
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Concentrated at center
    g3 = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.05 * n)) ** 2)
    inits['center'] = g3 / g3.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_resized = np.interp(np.linspace(0, 1, n), 
                                      np.linspace(0, 1, len(parent)), p)
                p_resized = np.maximum(p_resized, 0)
                pps = p_resized.sum()
                if pps > 1e-10:
                    inits['parent'] = p_resized / pps
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast multiplicative weight updates (mirror descent) for broad exploration
2. Subgradient descent with soft peak weighting and momentum
3. LP-based directional optimization for precise local improvement
4. Coordinate descent polishing for final refinement

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(conv, a, soft_temp=0.0, top_k=0):
    """Compute weighted subgradient of max(conv) w.r.t. a.
    Returns sg[m] = 2 * sum_k w[k] * a[k-m] (with a[x]=0 outside [0,n-1])."""
    n = len(a)
    conv_len = 2 * n - 1
    
    if soft_temp > 0:
        scaled = conv / (soft_temp * conv.max() + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)

        if w_sum < 1e-30:
            w = np.zeros(conv_len)
            w[np.argmax(conv)] = 1.0
        else:

    elif top_k > 0 and top_k < conv_len:

        w = np.zeros(conv_len)

        ws = w.sum()
        if ws > 0:

        else:
            w[np.argmax(conv)] = 1.0
    else:

        w[np.argmax(conv)] = 1.0

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

        g = solve_lp_direction(a, M, n)

        if g_sum < 1e-15:

def coordinate_polish(a_in, max_rounds=50):
    """Coordinate descent: move mass from high-contributing to low-contributing positions."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        k_star = np.argmax(conv)
        
        # Compute contribution of each position to the peak
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[k_star - j]
        
        # Sort: high contrib first (sources), low contrib first (targets)
        order_desc = np.argsort(contrib)[::-1]
        order_asc = np.argsort(contrib)
        
        for i_idx in range(min(15, n)):
            i = order_desc[i_idx]
            if a[i] < 1e-7:
                continue
            for j_idx in range(min(15, n)):
                j = order_asc[j_idx]
                if j == i:
                    continue
                delta = a[i] * 0.08
                if delta < 1e-9:
                    continue
                a[i] -= delta
                a[j] += delta
                conv_test = fast_convolve(a)
                obj_test = 2.0 * n * conv_test.max()

                    best_obj = obj_test

                    a[i] += delta

    if best_sequence is not None and len(best_sequence) not in seen_n:
        top_list.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 250.0
    t_per = max(30.0, remaining / max(1, len(top_list)))
    
    for n, a0 in top_list:
        if time.monotonic() > deadline - 230:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 220)
        if t_alloc < 15:
            break
        
        # Subgrad with soft weighting (longer run)
        a_s, obj_s = optimize_subgrad(a0, t_alloc * 0.35, lr_init=0.02,
                                       momentum=0.92, soft_temp=0.04, top_k=30)
        # MWU
        a_m, obj_m = optimize_mwu(a0, t_alloc * 0.25, eta_init=0.25, eta_decay=0.999)
        # LP
        a_l, obj_l = optimize_lp(a0, t_alloc * 0.4, blend_init=0.25,
                                  blend_min=0.005, blend_halve_every=8)
        
        candidates = [(obj_s, a_s), (obj_m, a_m), (obj_l, a_l)]
        candidates.sort(key=lambda x: x[0])
        a_best, obj_best = candidates[0]
        
        score = float(evaluate_sequence(a_best.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_best.copy()
            print(f"  Phase2 n={n}: {score:.6f}")
    
    # Phase 3: Polish
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        
        # Coordinate descent
        a_cd = coordinate_polish(best_sequence, max_rounds=40)
        score_cd = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(score_cd) and score_cd < best_score:
            best_score = score_cd
            best_sequence = a_cd.copy()
            print(f"  CD: {score_cd:.6f}")
        
        # LP + subgrad cycles
        remaining = deadline - time.monotonic() - 50.0
        for cycle in range(5):
            if time.monotonic() > deadline - 45:
                break
            t_cyc = min(remaining / (5 - cycle), 40.0)
            if t_cyc < 6:
                break
            
            # LP step
            a_l, obj_l = optimize_lp(best_sequence, t_cyc * 0.45,
                                      blend_init=0.12, blend_min=0.002,
                                      blend_halve_every=10)
            # Subgrad polish
            a_s, obj_s = optimize_subgrad(a_l, t_cyc * 0.25, lr_init=0.004,
                                           momentum=0.93, soft_temp=0.02, top_k=40)
            a_cur = a_s if obj_s < obj_l else a_l
            
            # CD
            a_cd = coordinate_polish(a_cur, max_rounds=25)
            score = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cd.copy()
                print(f"  Cycle {cycle}: {score:.6f}")
        
        # Random perturbations and re-optimize
        if time.monotonic() < deadline - 20:
            n_b = len(best_sequence)
            for ps in [0.001, 0.003, 0.006]:
                if time.monotonic() > deadline - 15:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 12:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = optimize_subgrad(a_p, 3.0, lr_init=0.003,
                                                 momentum=0.9, soft_temp=0.03, top_k=30)
                    a_cd = coordinate_polish(a_opt, max_rounds=10)
                    score = float(evaluate_sequence(a_cd.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_cd.copy()
                        print(f"  Perturb({ps}): {score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
