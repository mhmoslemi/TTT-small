```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
1. Spectral initializations (flat power spectrum via random phases)
2. Smooth L-BFGS-B with adaptive log-sum-exp temperature
3. Direct peak-shaving with targeted mass redistribution
4. Coordinate descent fine-tuning
5. Multi-length exploration with resampling
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    result = np.real(ifft(A * A, fft_size))[:size]
    np.maximum(result, 0, out=result)
    return result


def compute_grad_from_weights(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a. Using cross-correlation."""
    return 2.0 * np.correlate(w, a, mode='valid')


def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_and_grad(x, n, beta):
    """Objective and gradient for softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    obj = 2.0 * n * sm
    grad_a = compute_grad_from_weights(a, w)
    lagrange = np.dot(a, grad_a)
    grad_x = a * (grad_a - lagrange)
    return obj, grad_x


def optimize_lbfgsb(a0, betas, maxiter_per_beta=300):
    """Multi-beta L-BFGS-B optimization with softmax parametrization."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x = np.log(a)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for beta in betas:
        def obj_func(x):
            o, _ = softmax_obj_and_grad(x, n, beta)
            return o
        
        def grad_func(x):
            _, g = softmax_obj_and_grad(x, n, beta)
            return g
        
        result = minimize(obj_func, x, jac=grad_func, method='L-BFGS-B',
                         options={'maxiter': maxiter_per_beta, 'maxls': 50,
                                  'ftol': 1e-12, 'gtol': 1e-8})
        x = result.x
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def optimize_pg(a0, beta, lr=0.05, n_iter=500, decay=0.999):
    """Projected gradient descent with smooth max."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        grad = compute_grad_from_weights(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        t = lr
        for _ in range(20):
            a_new = project_simplex(a - t * grad)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        
        a = a_new
        lr *= decay
    
    return best_a, best_obj


def flat_spectrum_init(n, rng, n_trials=30):
    """Initialize with flat (circular) power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_score = float('inf')
    
    for _ in range(n_trials):
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        A = np.exp(1j * phases)
        # Make it conjugate symmetric for real output
        A[0] = 1.0
        if fft_size % 2 == 0:
            A[fft_size // 2] = 1.0
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a


def peak_shave(a, n_rounds=50, n_peaks=5):
    """Direct peak shaving: identify peaks and redistribute mass."""
    n = len(a)
    a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for round_idx in range(n_rounds):
        conv = fast_convolve(a)
        current_max = conv.max()
        current_score = 2.0 * n * current_max
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for k in peak_indices:
            if conv[k] <= conv.mean() * 1.01:
                continue
            
            # Identify contributing pairs
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            contrib = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = k - i
                contrib[i] += a[i] * a[j]
                if i != j:
                    contrib[j] += a[i] * a[j]
            
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Reduce elements proportional to their contribution
            reduction = 0.1 * (conv[k] - conv.mean())
            for i in range(n):
                if contrib[i] > 1e-15:
                    a[i] -= reduction * contrib[i] / total_contrib
            
            # Renormalize
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a /= s
            else:
                break
            improved = True
        
        if not improved:
            break
    
    return best_a, best_score


def coordinate_descent(a, n_iters=100):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iters):
        improved = False
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        order = np.random.permutation(n) if it > 0 else np.arange(n)
        
        for i in order:
            old_val = a[i]
            # The contribution of a[i] to lag k is: a[i] * a[k-i] (if valid) + a[k-i] * a[i] (if valid)
            # For all lags, the contribution is a[i] * (sum of a[k-i] over relevant k)
            # Try different values
            for scale in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a[i] = old_val * scale
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv_new = fast_convolve(a)
                new_score = 2.0 * n * conv_new.max()
                if new_score < best_score - 1e-10:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = old_val  # restore, we'll use best_a at the end
        
        if not improved and it > 10:
            break
    
    return best_a, best_score


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
1. Spectral initializations (flat power spectrum via random phases)
2. Smooth L-BFGS-B with adaptive log-sum-exp temperature
3. Direct peak-shaving with targeted mass redistribution
4. Coordinate descent fine-tuning
5. Multi-length exploration with resampling
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    result = np.real(ifft(A * A, fft_size))[:size]
    np.maximum(result, 0.0, out=result)
    return result


def smooth_max_and_weights(x, beta):
    """Log-sum-exp approximation of max with softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_from_weights(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    So grad[i] = 2*sum_{k: valid} w[k]*a[k-i] = 2*correlate(w, a, 'valid')[i]
    """
    return 2.0 * np.correlate(w, a, mode='valid')


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lbfgs_obj(x, n, beta):
    """Objective for L-BFGS with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _lbfgs_grad(x, n, beta):
    """Gradient for L-BFGS with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_from_weights(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, betas, maxiter_per_beta=300, time_limit=None, t_start=None):
    """Multi-beta L-BFGS-B optimization with softmax parametrization."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    x = np.log(a)

    best_obj = float('inf')
    best_a = a.copy()

    for beta in betas:
        if time_limit is not None and t_start is not None:
            if time.monotonic() > t_start + time_limit:
                break

        result = minimize(_lbfgs_obj, x, jac=_lbfgs_grad, args=(n, beta),
                         method='L-BFGS-B',
                         options={'maxiter': maxiter_per_beta, 'maxls': 40,
                                  'ftol': 1e-14, 'gtol': 1e-10})
        x = result.x
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def optimize_pg(a0, beta, lr=0.05, n_iter=500, decay=0.999, time_limit=None, t_start=None):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > t_start + time_limit:
                break

        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        grad = compute_grad_from_weights(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        t = lr
        a_new = project_simplex(a - t * grad)
        for _ in range(15):
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
            a_new = project_simplex(a - t * grad)

        a = a_new
        lr *= decay

    return best_a, best_obj


def flat_spectrum_init(n, rng, n_trials=40):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_score = float('inf')

    for _ in range(n_trials):
        phases = rng.uniform(0.0, 2.0 * np.pi, fft_size)
        A = np.exp(1j * phases)
        A[0] = np.sqrt(fft_size)
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.sqrt(fft_size)
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a


def weighted_flat_spectrum_init(n, rng, n_trials=40):
    """Weighted flat spectrum init: flatten |DFT|^2 with emphasis on low frequencies."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_score = float('inf')

    for _ in range(n_trials):
        phases = rng.uniform(0.0, 2.0 * np.pi, fft_size)
        mags = np.ones(fft_size)
        # Slightly boost DC to ensure positive sum after clipping
        mags[0] = np.sqrt(n) * 2.0
        if fft_size % 2 == 0:
            mags[fft_size // 2] = np.sqrt(n)
        A = mags * np.exp(1j * phases)
        # Enforce conjugate symmetry
        for k in range(1, fft_size // 2 + 1):
            if k < fft_size - k:
                A[k] = mags[k] * np.exp(1j * phases[k])
                A[fft_size - k] = mags[k] * np.exp(-1j * phases[k])
            elif k == fft_size - k:
                A[k] = mags[k]
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a


def peak_shave(a, n_rounds=50, n_peaks=5, strength=0.15):
    """Direct peak shaving: identify peaks and reduce contributing elements."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for _ in range(n_rounds):
        conv = fast_convolve(a)
        current_max = conv.max()
        current_score = 2.0 * n * current_max

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        mean_conv = conv.mean()

        for k in peak_indices:
            if conv[k] <= mean_conv * 1.005:
                continue

            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            contrib = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = k - i
                val = a[i] * a[j]
                contrib[i] += val
                if i != j:
                    contrib[j] += val

            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            excess = (conv[k] - mean_conv) * strength
            for i in range(n):
                if contrib[i] > 1e-15:
                    a[i] -= excess * contrib[i] / total_contrib

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                return best_a, best_score

    return best_a, best_score


def multiplicative_update(a, n_iters=200, alpha=0.3, beta_smooth=50.0):
    """Multiplicative (power iteration) update for peak reduction."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for _ in range(n_iters):
        conv = fast_convolve(a)
        _, w = smooth_max_and_weights(conv, beta_smooth)
        # contribution[i] = sum_k w[k] * a[k-i] for valid k
        contrib = np.correlate(w, a, mode='valid')
        # Multiplicative update: a[i] *= (target / contrib[i])^alpha
        target = conv.max() * 0.99
        ratio = np.ones(n)
        mask = contrib > 1e-15
        ratio[mask] = (target / contrib[mask]) ** alpha
        a = a * ratio
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def coordinate_descent(a, n_iters=50, rng=None):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    for it in range(n_iters):
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved_this_iter = False
        for i in order:
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Try scaling this coordinate
            for scale in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[i] = old_val * scale
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a = a / s
                conv_new = fast_convolve(a)
                new_score = 2.0 * n * conv_new.max()
                if new_score < best_score - 1e-10:
                    best_score = new_score
                    best_a = a.copy()
                    improved_this_iter = True
                    break
            # Restore if we didn't improve (best_a is tracked separately)

        a = best_a.copy()
        if not improved_this_iter and it > 5:
            break

    return best_a, best_score


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0.0))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['flat_spec'] = flat_spectrum_init(n, rng, n_trials=30)
    inits['wflat_spec'] = weighted_flat_spectrum_init(n, rng, n_trials=20)

    g1 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g1 /= g1.sum()
    inits['gauss_025'] = g1

    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.4 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_04'] = g2

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    return inits


def score_and_track(a, state, t_start):
    """Score sequence and update state if better."""
    if a is None:
        return float('inf')
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a = a / s
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_sequence'] = a.copy()
        elapsed = time.monotonic() - t_start
        print("  New best: %.6f (n=%d, t=%.0fs)" % (score, len(a), elapsed))
    return score if np.isfinite(score) else float('inf')


def run_optimization_pipeline(a0, state, t_start, deadline, label=""):
    """Run the full optimization pipeline on a starting sequence."""
    n = len(a0)
    elapsed_limit = deadline - time.monotonic() - 10
    if elapsed_limit < 10:
        return

    # Stage 1: L-BFGS-B with decreasing beta
    betas_schedule = [80.0, 40.0, 20.0, 10.0, 5.0, 3.0, 2.0]
    t_alloc = min(200.0, elapsed_limit * 0.4)
    if t_alloc > 10:
        maxiter = int(min(300, t_alloc * 1.5))
        a_opt, _ = optimize_lbfgsb(a0, betas_schedule, maxiter_per_beta=maxiter,
                                   time_limit=t_alloc, t_start=time.monotonic())
        score_and_track(a_opt, state, t_start)

    a_cur = state['best_sequence'] if state['best_sequence'] is not None else a_opt

    # Stage 2: Multiplicative update
    elapsed_limit = deadline - time.monotonic() - 10
    if elapsed_limit > 15:
        t_alloc = min(60.0, elapsed_limit * 0.2)
        nm_iter = int(min(300, t_alloc * 10))
        a_mu, _ = multiplicative_update(a_cur, n_iters=nm_iter, alpha=0.3, beta_smooth=50.0)
        score_and_track(a_mu, state, t_start)
        a_cur = state['best_sequence']

    # Stage 3: Peak shaving
    elapsed_limit = deadline - time.monotonic() - 10
    if elapsed_limit > 10:
        n_rounds = int(min(100, elapsed_limit * 2))
        a_shaved, _ = peak_shave(a_cur, n_rounds=n_rounds, n_peaks=5, strength=0.12)
        score_and_track(a_shaved, state, t_start)
        a_cur = state['best_sequence']

    # Stage 4: PG refinement
    elapsed_limit = deadline - time.monotonic() - 10
    if elapsed_limit > 10:
        n_pg = int(min(400, elapsed_limit * 8))
        a_pg, _ = optimize_pg(a_cur, beta=30.0, lr=0.04, n_iter=n_pg,
                              time_limit=min(60.0, elapsed_limit * 0.5), t_start=time.monotonic())
        score_and_track(a_pg, state, t_start)
        a_cur = state['best_sequence']

    # Stage 5: Another L-BFGS pass with very low beta
    elapsed_limit = deadline - time.monotonic() - 10
    if elapsed_limit > 20:
        t_alloc = min(80.0, elapsed_limit * 0.4)
        maxiter = int(min(300, t_alloc * 2))
        a_opt2, _ = optimize_lbfgsb(a_cur, [3.0, 2.0, 1.5], maxiter_per_beta=maxiter,
                                    time_limit=t_alloc, t_start=time.monotonic())
        score_and_track(a_opt2, state, t_start)

    # Stage 6: Final peak shave + multiplicative
    elapsed_limit = deadline - time.monotonic() - 10
    if elapsed_limit > 10:
        a_shaved2, _ = peak_shave(state['best_sequence'], n_rounds=50, n_peaks=8, strength=0.08)
        score_and_track(a_shaved2, state, t_start)
        a_mu2, _ = multiplicative_update(state['best_sequence'], n_iters=100, alpha=0.2, beta_smooth=100.0)
        score_and_track(a_mu2, state, t_start)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining spectral initialization,
    L-BFGS-B with adaptive smooth max, multiplicative updates, direct peak
    shaving, and coordinate refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 60.0)

    state = {'best_score': float('inf'), 'best_sequence': None}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    # Phase 1: Quick scan over lengths and initializations
    print("=== Phase 1: Quick Scan ===")
    t_phase1_end = t_start + 350.0

    candidate_lengths = [128, 192, 256, 320, 384, 512]
    if parent is not None:
        if len(parent) not in candidate_lengths:
            candidate_lengths.append(len(parent))
        candidate_lengths = sorted(set(candidate_lengths))

    init_results = []

    for n in candidate_lengths:
        if time.monotonic() > t_phase1_end - 40:
            break

        inits = generate_inits(n, parent, rng)
        init_names_priority = ['parent', 'flat_spec', 'wflat_spec', 'uniform',
                               'gauss_025', 'triangle', 'random']
        init_names = [nm for nm in init_names_priority if nm in inits]
        init_names = init_names[:4]

        time_per_n = max(30.0, (t_phase1_end - time.monotonic() - 30) / max(1, len(candidate_lengths)))
        time_per_init = time_per_n / max(1, len(init_names))

        for name in init_names:
            if time.monotonic() > t_phase1_end - 20:
                break
            t_alloc = min(time_per_init, t_phase1_end - time.monotonic() - 15)
            if t_alloc < 5:
                break

            a0 = inits[name]
            print("  n=%d, init=%s (alloc=%.0fs)" % (n, name, t_alloc))

            # Quick optimization: LBFGS + peak shave
            betas_quick = [40.0, 20.0, 10.0]
            maxiter = int(min(150, t_alloc * 1.5))
            a_opt, _ = optimize_lbfgsb(a0, betas_quick, maxiter_per_beta=maxiter,
                                       time_limit=t_alloc * 0.6, t_start=time.monotonic())
            score = score_and_track(a_opt, state, t_start)

            # Quick peak shave
            a_shaved, _ = peak_shave(a_opt, n_rounds=30, n_peaks=4, strength=0.12)
            score2 = score_and_track(a_shaved, state, t_start)

            best_local = a_shaved if score2 < (score if np.isfinite(score) else float('inf')) else a_opt
            if np.isfinite(score2) or np.isfinite(score):
                init_results.append((min(score, score2) if np.isfinite(score) and np.isfinite(score2)
                                     else (score if np.isfinite(score) else score2),
                                     n, best_local.copy()))

    init_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f" % state['best_score'])

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    t_phase2_end = deadline - 400.0

    top_candidates = init_results[:4] if init_results else []
    if not top_candidates and state['best_sequence'] is not None:
        top_candidates = [(state['best_score'], len(state['best_sequence']), state['best_sequence'].copy())]

    time_per_candidate = max(60.0, (t_phase2_end - time.monotonic() - 30) / max(1, len(top_candidates)))

    for idx, (sc, n, a_start) in enumerate(top_candidates):
        if time.monotonic() > t_phase2_end - 30:
            break

        t_alloc = min(time_per_candidate, t_phase2_end - time.monotonic() - 20)
        if t_alloc < 20:
            break

        print("  Deep opt candidate %d: n=%d, score=%.6f (alloc=%.0fs)" % (idx, n, sc, t_alloc))

        # Full pipeline
        run_optimization_pipeline(a_start, state, t_start, t_phase2_end)

    print("Phase 2 done. Best: %.6f" % state['best_score'])

    # Phase 3: Intensive fine-tuning on the global best
    print("=== Phase 3: Intensive Fine-tuning ===")

    if state['best_sequence'] is None:
        state['best_sequence'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_sequence'].tolist()))

    t_phase3_end = deadline - 60.0
    n_best = len(state['best_sequence'])

    # Iterative refinement loop
    for iteration in range(8):
        if time.monotonic() > t_phase3_end - 20:
            break

        elapsed_left = t_phase3_end - time.monotonic()
        if elapsed_left < 15:
            break

        a_cur = state['best_sequence'].copy()

        # 3a: L-BFGS with low beta
        t_alloc = min(100.0, elapsed_left * 0.2)
        if t_alloc > 10:
            maxiter = int(min(300, t_alloc * 2))
            a_opt, _ = optimize_lbfgsb(a_cur, [5.0, 3.0, 2.0], maxiter_per_beta=maxiter,
                                       time_limit=t_alloc, t_start=time.monotonic())
            score_and_track(a_opt, state, t_start)

        # 3b: Multiplicative update
        elapsed_left = t_phase3_end - time.monotonic()
        if elapsed_left > 15:
            a_mu, _ = multiplicative_update(state['best_sequence'], n_iters=int(min(200, elapsed_left)),
                                            alpha=0.25, beta_smooth=80.0)
            score_and_track(a_mu, state, t_start)

        # 3c: Peak shaving (aggressive)
        elapsed_left = t_phase3_end - time.monotonic()
        if elapsed_left > 10:
            a_shaved, _ = peak_shave(state['best_sequence'], n_rounds=int(min(80, elapsed_left)),
                                     n_peaks=8, strength=0.1)
            score_and_track(a_shaved, state, t_start)

        # 3d: PG with high beta
        elapsed_left = t_phase3_end - time.monotonic()
        if elapsed_left > 15:
            n_pg = int(min(300, elapsed_left * 5))
            a_pg, _ = optimize_pg(state['best_sequence'], beta=50.0, lr=0.03, n_iter=n_pg,
                                  time_limit=min(40.0, elapsed_left * 0.4), t_start=time.monotonic())
            score_and_track(a_pg, state, t_start)

        # 3e: Random perturbation + quick optimize
        elapsed_left = t_phase3_end - time.monotonic()
        if elapsed_left > 30:
            for _ in range(3):
                if time.monotonic() > t_phase3_end - 15:
                    break
                scale = 0.005 * rng.uniform() + 0.002
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(state['best_sequence'] + noise)
                t_single = min(40.0, (t_phase3_end - time.monotonic() - 10) / 3)
                if t_single < 5:
                    break
                maxiter = int(min(150, t_single * 2))
                a_opt_p, _ = optimize_lbfgsb(a_pert, [10.0, 5.0], maxiter_per_beta=maxiter,
                                             time_limit=t_single, t_start=time.monotonic())
                a_shaved_p, _ = peak_shave(a_opt_p, n_rounds=20, n_peaks=5, strength=0.1)
                score_and_track(a_shaved_p, state, t_start)

    # Phase 4: Try alternate lengths
    print("=== Phase 4: Length Exploration ===")
    if time.monotonic() < deadline - 120:
        alt_lengths = [384, 512, 320, 192]
        for alt_n in alt_lengths:
            if time.monotonic() > deadline - 40:
                break
            if alt_n == n_best:
                continue

            a_resampled = resample_sequence(state['best_sequence'], alt_n)
            remaining = deadline - time.monotonic() - 30
            if remaining < 30:
                break

            print("  Trying n=%d (remaining=%.0fs)" % (alt_n, remaining))
            betas_alt = [30.0, 15.0, 8.0, 4.0]
            maxiter = int(min(200, remaining * 0.8))
            a_opt, _ = optimize_lbfgsb(a_resampled, betas_alt, maxiter_per_beta=maxiter,
                                       time_limit=remaining * 0.6, t_start=time.monotonic())
            score_and_track(a_opt, state, t_start)

            a_shaved, _ = peak_shave(a_opt, n_rounds=30, n_peaks=5, strength=0.1)
            score_and_track(a_shaved, state, t_start)

            a_mu, _ = multiplicative_update(state['best_sequence'], n_iters=50, alpha=0.2, beta_smooth=60.0)
            score_and_track(a_mu, state, t_start)

    # Final: one more peak shave on the absolute best
    if time.monotonic() < deadline - 10:
        a_final, _ = peak_shave(state['best_sequence'], n_rounds=50, n_peaks=10, strength=0.05)
        score_and_track(a_final, state, t_start)

    # Final verification
    if state['best_sequence'] is None:
        state['best_sequence'] = np.ones(128) / 128

    final_score = float(evaluate_sequence(state['best_sequence'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score

    print("\nFINAL SCORE: %.6f (n=%d, time=%.0fs)" % (state['best_score'],
                                                      len(state['best_sequence']),
                                                      time.monotonic() - t_start))
    return [float(x) for x in state['best_sequence']]
```
