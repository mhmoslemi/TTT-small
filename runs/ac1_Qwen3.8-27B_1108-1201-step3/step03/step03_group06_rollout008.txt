"""
FISTA + Nesterov subgradient hybrid for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive smoothing (beta annealing), multiple initializations, multiple sequence
lengths, and exploits the convex structure. Key improvements over previous approach:
- FISTA (accelerated gradient) with annealed softmax smoothing for O(1/T^2) convergence
- Direct Nesterov subgradient method for sharp refinement near the optimum
- Efficient FFT-based gradient computation
- Multiple n-value exploration with parent warm-start
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta * c[k])) where c = conv(a,a).
    Returns (gradient, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    scaled = beta * c - beta * c.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_k w[k] * a[k-i] (with a[m]=0 outside [0,n-1])
    # = 2 * (w cross-correlation a)[i]
    # Using FFT: (w cross-corr a)[i] = ifft(fft(w_padded) * conj(fft(a_padded)))[i]
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    grad = 2.0 * cross
    return grad, c


def subgradient_at_peak(a):
    """
    Compute a subgradient of max_k conv(a,a)[k] at a.
    Returns (subgradient, peak_index, convolution).
    """
    n = len(a)
    c = fast_convolve(a)
    k_star = int(np.argmax(c))
    # Subgradient: g[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
    g = np.zeros(n)
    valid = (k_star >= np.arange(n)) & (k_star - np.arange(n) < n)
    g[valid] = 2.0 * a[k_star - np.arange(n)[valid]]
    return g, k_star, c


def fista_optimize(a0, time_limit, beta_start=100.0, beta_end=2000.0,
                   momentum_init=0.9, lr_init=0.1, verbose=False, seed=0):
    """
    FISTA (accelerated proximal gradient) with annealed smoothing.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = lr_init
    no_improve = 0
    iter_count = 0
    z_prev = a.copy()
    t_mom = 1.0

    while time.monotonic() - t0 < time_limit:
        # Adaptive beta: anneal from smooth to sharp
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** elapsed_frac
        beta = max(beta_start, beta)

        # Compute gradient at current point
        grad, c_curr = smooth_grad(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Adaptive step size (simplified backtracking)
        # Lipschitz constant estimate: ~4 * beta (for the smoothed max)
        L = 4.0 * beta * max(a.max(), 0.1)
        step = min(1.0 / max(L, 1e-10), 0.5)
        step = max(step, 1e-6)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new
        t_mom = t_new

        y = a + momentum * (a - z_prev)
        z_prev = a.copy()

        # Gradient step
        a_new = project_simplex(y - step * grad)

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:

            a = a_new  # still update for momentum

                a = project_simplex(a + noise)
                z_prev = a.copy()
                t_mom = 1.0
                no_improve = 0
                beta = beta_start * 2
                if no_improve > 500:
                    break

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FISTA iter {iter_count}: obj={best_obj:.8f}, beta={beta:.0f}, t={elapsed:.1f}s")

    return best_a, 2.0 * n * best_obj

def lp_refine(a0, time_limit, top_k=50, verbose=False, seed=0):
    """
    LP-based direction finding for refinement.
    Finds direction b that maximizes sum(b) subject to b>=0 and (a*b)[k] <= M.
    Then moves toward b.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t_step = 0.5
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = c.max() * 1.001  # slight relaxation

        # Build LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for top-k peak constraints
        num_k = 2 * n - 1
        if top_k and top_k < num_k:
            peaks = np.argpartition(c, -top_k)[-top_k:]
        else:
            peaks = np.arange(num_k)

        # Build constraint matrix for selected peaks
        A_ub = np.zeros((len(peaks), n))
        for idx, k in enumerate(peaks):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            j = np.arange(i_min, i_max + 1)
            A_ub[idx, j] = a[k - j]

        b_ub = np.full(len(peaks), M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                break
            b_dir = res.x
        except Exception:
            break

        b_dir = np.maximum(b_dir, 0)
        b_sum = b_dir.sum()
        if b_sum < 1e-15:
            break
        b_dir = b_dir / b_sum

        # Move toward b_dir
        a_new = (1.0 - t_step) * a + t_step * b_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            t_step = min(1.0, t_step * 1.05)
        else:
            no_improve += 1
            t_step = max(0.05, t_step * 0.8)
            a = a_new  # partial acceptance for exploration

        if no_improve > 50:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            t_step = 0.5
            no_improve = 0

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}: obj={best_obj:.8f}, t_step={t_step:.3f}, t={elapsed:.1f}s")

    return best_a, 2.0 * n * best_obj


def mirror_descent(a0, time_limit, eta_init=0.1, decay=0.999, verbose=False, seed=0):
    """
    Mirror descent (exponential weighted / multiplicative weights) for the subproblem.
    Works in log-space: a_i = exp(x_i) / sum(exp(x_j)).
    Update: x_i <- x_i - eta * subgrad_i, then normalize.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Work in log-space
    x = np.log(a + 1e-30)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute softmax from x
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        a_curr = e / e.sum()

        # Subgradient at peak
        c = fast_convolve(a_curr)
        k_star = int(np.argmax(c))
        g = np.zeros(n)
        valid = (k_star >= np.arange(n)) & (k_star - np.arange(n) < n)
        g[valid] = a_curr[k_star - np.arange(n)[valid]]

        # Mirror descent update in log-space
        x = x - eta * g
        x -= x.max()  # normalize to prevent overflow
        x = np.maximum(x, -50)  # prevent underflow

        # Project onto simplex (handle any numerical issues)
        a_new = np.exp(x - x.max())
        a_new = a_new / a_new.sum()

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        no_improve += 1

        # Decay step size
        eta *= decay
        if eta < 1e-6:
            eta = 1e-6
            if no_improve > 100:
                break

        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.01
            a_curr = project_simplex(a_curr + noise)
            x = np.log(a_curr + 1e-30)
            no_improve = 0
            eta = eta_init * 0.5

        iter_count += 1
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter {iter_count}: obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")

    return best_a, 2.0 * n * best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Trimodal
    tr = np.zeros(n)
    seg = max(1, n // 6)
    tr[:seg] = 1.0
    tr[n//3 - seg//2:n//3 + seg//2] = 1.0
    tr[-seg:] = 1.0
    if tr.sum() > 0:
        tr /= tr.sum()
        inits['trimodal'] = tr

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n-1)) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Gaussian edge
    g2 = np.exp(-x / (0.15 * n))
    g2 /= g2.sum()
    inits['gauss_edge'] = g2

    # Cosine
    cos_w = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random sparse
    sparse = rng.random(n)
    sparse[sparse < 0.5] = 0
    if sparse.sum() > 0:
        sparse /= sparse.sum()
        inits['sparse'] = sparse

    # Random Dirichlet (concentrated)
    inits['dirichlet_conc'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['dirichlet_unif'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        if len(parent) < n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def try_length(n, parent, rng, time_budget, strategies):

    priority = ['parent_crop', 'parent_ds', 'parent_up', 'bimodal', 'uniform',

    init_names = [nm for nm in priority if nm in inits] + \
                 [nm for nm in init_names if nm not in priority]
    init_names = init_names[:4]  # Limit to 4 initializations

    best_obj = float('inf')

        noise = rng.standard_normal(best_n) * pert_scale

    num_com组合 = len(starts) * len(deep_configs)
    time_per_run = p2_budget / max(1, num_com组合)
    time_per_run = max(15.0, time_per_run)

    run_idx = 0
    for a_start in starts:
        for cfg in deep_configs:
            remaining = p2_budget - (time.monotonic() - t_p2)
            if remaining < 15:
                break
            t_alloc = min(time_per_run, remaining - 5)
            if t_alloc < 10:
                break

            if cfg['func'] == 'fista':
                a_opt, obj_opt = fista_optimize(a_start, t_alloc,
                                                 beta_start=cfg['beta_start'],
                                                 beta_end=cfg['beta_end'],
                                                 lr_init=cfg['lr_init'],
                                                 seed=run_idx * 7)
            elif cfg['func'] == 'nesterov':
                a_opt, obj_opt = nesterov_subgrad(a_start, t_alloc,
                                                   step_init=cfg['step_init'],
                                                   seed=run_idx * 7)
            elif cfg['func'] == 'mirror':
                a_opt, obj_opt = mirror_descent(a_start, t_alloc,
                                                 eta_init=cfg['eta_init'],
                                                 decay=cfg['decay'],
                                                 seed=run_idx * 7)

            if update_best(a_opt, best_n):
                print(f"  Phase2 run {run_idx}: score={best_score:.6f}")

            run_idx += 1

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: LP Refinement ==========
    print("=== Phase 3: LP Refinement ===")
    t_p3 = time.monotonic()
    p3_budget = min(350.0, (deadline - time.monotonic()) * 0.35)

    if best_sequence is not None:
        n_best = len(best_sequence)

        lp_configs = [
            {'top_k': max(30, n_best // 8), 'step': None},
            {'top_k': max(50, n_best // 5), 'step': None},
            {'top_k': max(20, n_best // 12), 'step': None},
        ]

        # Multiple starts for LP
        lp_starts = [best_sequence.copy()]
        for scale in [0.002, 0.005]:
            noise = rng.standard_normal(n_best) * scale
            lp_starts.append(project_simplex(best_sequence + noise))

        time_per_lp = p3_budget / max(1, len(lp_configs) * len(lp_starts))
        time_per_lp = max(20.0, time_per_lp)

        lp_idx = 0
        for a_start in lp_starts:
            for cfg in lp_configs:
                remaining = p3_budget - (time.monotonic() - t_p3)
                if remaining < 20:
                    break
                t_alloc = min(time_per_lp, remaining - 5)
                if t_alloc < 15:
                    break

                a_opt, obj_opt = lp_refine(a_start, t_alloc,
                                            top_k=cfg['top_k'], seed=lp_idx * 11 + 3)
                if update_best(a_opt, n_best):
                    print(f"  Phase3 LP {lp_idx}: score={best_score:.6f}")
                lp_idx += 1

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Alternating FISTA + Nesterov ==========
    print("=== Phase 4: Alternating Optimization ===")
    t_p4 = time.monotonic()
    p4_budget = min(300.0, (deadline - time.monotonic()) * 0.3)

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()

        # Alternate between FISTA (smooth) and Nesterov (sharp)
        num_rounds = 4
        time_per_round = p4_budget / max(1, num_rounds)

        for r in range(num_rounds):
            remaining = p4_budget - (time.monotonic() - t_p4)
            if remaining < 30:
                break
            t_alloc = min(time_per_round, remaining - 10)
            if t_alloc < 20:
                break

            # FISTA round (smooth, good for global progress)
            beta_s = 500 + 500 * r
            beta_e = 2000 + 1000 * r
            a_opt, obj_opt = fista_optimize(a, t_alloc * 0.6,
                                             beta_start=beta_s, beta_end=beta_e,
                                             seed=50 + r)
            if obj_opt / (2.0 * n_best) < fast_convolve(a).max():
                a = a_opt
            update_best(a, n_best)

            # Nesterov round (sharp, good for local refinement)
            remaining2 = p4_budget - (time.monotonic() - t_p4)
            t_alloc2 = min(time_per_round * 0.4, remaining2 - 5)
            if t_alloc2 > 10:
                a_opt2, obj_opt2 = nesterov_subgrad(a, t_alloc2,
                                                     step_init=0.01 / (r + 1), seed=60 + r)
                if obj_opt2 / (2.0 * n_best) < fast_convolve(a).max():
                    a = a_opt2
                update_best(a, n_best)

            if best_sequence is not None and a is not best_sequence:
                # Keep the better one
                if fast_convolve(a).max() < fast_convolve(best_sequence).max():
                    best_sequence = a.copy()

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Try tiny perturbations
        if remaining > 50:
            # Nesterov with very small steps
            t_alloc = min(60.0, remaining * 0.3)
            if t_alloc > 10:
                a_opt, obj_opt = nesterov_subgrad(best_sequence, t_alloc,
                                                   step_init=0.003, seed=42)
                if update_best(a_opt, n_best):
                    print(f"  Polish Nesterov: score={best_score:.6f}")

            # Mirror descent with tiny eta
            t_alloc = min(40.0, remaining * 0.2)
            if t_alloc > 10:
                a_opt, obj_opt = mirror_descent(best_sequence, t_alloc,
                                                 eta_init=0.01, decay=0.9999, seed=43)
                if update_best(a_opt, n_best):
                    print(f"  Polish Mirror: score={best_score:.6f}")

            # Random perturbation search
            remaining2 = deadline - time.monotonic() - 10
            if remaining2 > 20:
                n_tries = int(min(20, remaining2 / 3))
                for i in range(n_tries):
                    if time.monotonic() > deadline - 15:
                        break
                    for scale in [0.0005, 0.001, 0.002, 0.005]:
                        noise = rng.standard_normal(n_best) * scale
                        a_pert = project_simplex(best_sequence + noise)
                        score_pert = float(evaluate_sequence(a_pert.tolist()))
                        if np.isfinite(score_pert) and score_pert < best_score:
                            best_score = score_pert
                            best_sequence = a_pert.copy()
                            print(f"  Perturb [{i}] s={scale}: score={best_score:.6f}")
                            break

        # Also try one more LP polish
        remaining3 = deadline - time.monotonic() - 10
        if remaining3 > 30:
            a_opt, obj_opt = lp_refine(best_sequence, min(50.0, remaining3),
                                        top_k=max(40, n_best // 6), seed=999)
            if update_best(a_opt, n_best):
                print(f"  Final LP: score={best_score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
