"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (multiplicative weights)
with FFT-based gradients, LP-based refinement, and multi-scale exploration.
Key innovation: exponential-family (softmax) weighting of convolution peaks for
smooth, scale-invariant gradient updates that work well on the simplex.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv(a):
    """Compute linear auto-convolution."""
    return fast_convolve(a)


def compute_grad_from_weights(a, w, L):
    """
    Compute gradient of max_k c[k] w.r.t. a, given softmax weights w on c.
    grad[m] = 2 * sum_k w[k] * a[k-m]
    Uses FFT for efficiency.
    """
    n = len(a)
    a_rev = a[::-1]
    W = np.fft.rfft(w, L)
    AR = np.fft.rfft(a_rev, L)
    conv_w = np.fft.irfft(W * AR, L)
    return 2.0 * conv_w[n-1:2*n-1]


def mirror_descent(a_init, time_limit, tau_ratio_init=0.1, tau_ratio_final=0.005,
                   lr=0.5, min_tau_ratio=0.001, stagnant_limit=200):
    """
    Mirror descent (multiplicative weights) with softmax-based peak weighting.
    Optimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # FFT size for gradient computation: need to convolve w (len 2n-1) with a_rev (len n)
    # Result length: 3n-2
    conv_len = 3 * n - 2
    L = 2 ** (conv_len - 1).bit_length()
    if L < conv_len:
        L *= 2
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    stagnant = 0
    tau_init = conv.max() * tau_ratio_init
    tau_current = tau_init
    
    # Schedule: exponentially anneal tau
    n_steps_estimate = max(100, int(time_limit / max(0.001, conv.max() * 0.01)))
    
    while time.monotonic() - t0 < time_limit:
        # Update tau (exponential annealing)
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau_current = tau_init * (tau_ratio_final / tau_ratio_init) ** progress
        tau_current = max(tau_current, conv.max() * min_tau_ratio)
        
        # Compute softmax weights on convolution peaks
        c_max = conv.max()
        if c_max < 1e-15:
            break
        log_w = (conv - c_max) / tau_current
        # Clip to prevent underflow
        log_w = np.maximum(log_w, -50)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum
        
        # Compute gradient via FFT
        grad = compute_grad_from_weights(a, w, L)
        
        # Ensure gradient is non-negative (it should be since a >= 0 and w >= 0)
        grad = np.maximum(grad, 0.0)
        
        # Multiplicative update
        # Scale gradient to get reasonable step size
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        scaled_grad = grad / grad_max  # normalize to [0, 1]
        
        a_new = a * np.exp(-lr * scaled_grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a = a_new / a_new_sum
        
        # Compute new convolution and objective
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > stagnant_limit:
            # Perturb and restart
            rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.02)
            conv = fast_convolve(a)
            stagnant = 0
            # Reduce lr
            lr = max(0.05, lr * 0.7)
    
    return best_a, best_obj


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, blend_init=0.3, blend_min=0.003,
                blend_halve_every=15):
    """LP-based directional optimization."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_pairs(n)
    uniform = np.ones(n) / n
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        max_g = g.max()
        if max_g > 0.5:
            g = 0.7 * g + 0.3 * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
        
        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng(hash(tuple(np.round(a[:8], 6))) % 2**31)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0
    
    return best_a, best_obj


def optimize_subgrad(a0, time_limit, lr_init=0.01, momentum=0.9,
                     top_k=50, temp_ratio=0.03):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        temp = pmax * temp_ratio
        scaled = (peak_vals - peak_vals.max()) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.6)
        
        lr *= 0.9995
    
    return best_a, best_obj


def spectral_equalize(a_init, n_iters=50):
    """
    Spectral equalization: iteratively flatten the spectrum of the zero-padded
    sequence to reduce the convolution peak.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    L = 2 ** (2 * n - 2).bit_length()
    if L < 2 * n - 1:
        L *= 2
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iters):
        # Zero-pad and compute spectrum
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A) ** 2
        
        # Target: flat power spectrum (geometric mean)
        target = np.full_like(power, power.mean())
        
        # Partial flattening (interpolate between current and flat)
        alpha = 0.1 + 0.3 * (it / n_iters)  # increase flattening over time
        power_new = (1 - alpha) * power + alpha * target
        
        # Reconstruct: keep phase, modify amplitude
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(power_new, 0)) * np.exp(1j * phase)
        
        # Inverse FFT
        a_candidate = np.fft.irfft(A_new, L)[:n]
        
        # Project to non-negative
        a_candidate = np.maximum(a_candidate, 0.0)
        cs = a_candidate.sum()
        if cs < 1e-15:
            break
        a_candidate = a_candidate / cs
        
        # Check objective
        conv = fast_convolve(a_candidate)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_candidate.copy()
        
        a = a_candidate
    
    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.35 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Cosine-tapered (like a Kaiser window)
    cos_w = np.cos(np.pi * (x - n // 2) / n) ** 2
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # "Comb" pattern: alternating high/low
    comb = np.ones(n)
    comb[::2] = 2.0
    comb[1::2] = 0.5
    inits['comb'] = comb / comb.sum()
    
    # Random with structure (smoothed random)
    r = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 3.0) ** 2)
    kernel /= kernel.sum()
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0.01)
    inits['smooth_random'] = r_smooth / r_smooth.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p / ps)
                inits['parent_res'] = p_interp / p_interp.sum()
    
    return inits
