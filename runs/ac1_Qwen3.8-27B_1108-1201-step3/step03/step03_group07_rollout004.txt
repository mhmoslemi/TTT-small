def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-phase optimization with peak-penalized smooth objectives
- Adaptive beta scheduling for log-sum-exp approximation
- Simulated annealing for global exploration
- Coordinate descent for local refinement
- Length adaptation for finding optimal support size

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, w):
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    return np.real(ifft(np.conj(W) * A, N))[:n] * 2.0


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(u * np.arange(1, n + 1) > (cssv - 1.0))
    if rho < 1:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def exact_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_params_to_a(x):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    return a


def peak_weighted_obj(x, n, beta, peak_idx, peak_wt):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    obj = 2.0 * n * sm
    for idx, wt in zip(peak_idx, peak_wt):
        if idx < len(conv):
            obj += wt * conv[idx]
    return obj


def peak_weighted_jac(x, n, beta, peak_idx, peak_wt):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_conv(a, w)
    
    # Add peak gradient
    for idx, wt in zip(peak_idx, peak_wt):
        if idx < len(conv):
            for m in range(n):
                j = idx - m
                if 0 <= j < n:
                    grad_a[m] += wt * 2.0 * a[j]
    
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_smooth(a0, beta, maxiter=300):
    n = len(a0)
    a0_pos = np.maximum(a0, 1e-15)
    a0_pos = a0_pos / a0_pos.sum()
    x0 = np.log(a0_pos)
    
    def obj(x):
        a = softmax_params_to_a(x)
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        a = softmax_params_to_a(x)
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_conv(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    a_opt = softmax_params_to_a(result.x)
    return a_opt, result.fun


def peak_penalized_optimize(a0, n_iters_outer=10, n_peaks=20, beta_start=50.0, beta_end=5.0):
    """Iteratively add peak penalties and re-optimize."""
    n = len(a0)
    a = a0.copy() / a0.sum()
    beta = beta_start
    
    for outer in range(n_iters_outer):
        # Identify peaks
        conv = fast_convolve(a)
        # Find top peaks (excluding the very edges which are usually small)
        peak_vals = conv.copy()
        # Focus on the middle 80% of the convolution
        margin = max(1, len(conv) // 10)
        peak_vals[:margin] = -np.inf
        peak_vals[-margin:] = -np.inf
        
        top_k = np.argsort(peak_vals)[::-1][:n_peaks]
        peak_weights = np.ones(n_peaks) * 0.5
        
        # Optimize with peak penalties
        a0_pos = np.maximum(a, 1e-15)
        a0_pos = a0_pos / a0_pos.sum()
        x0 = np.log(a0_pos)
        
        def obj(x):
            a_x = softmax_params_to_a(x)
            conv_x = fast_convolve(a_x)
            sm, _ = smooth_max_and_weights(conv_x, beta)
            o = 2.0 * n * sm
            for idx, wt in zip(top_k, peak_weights):
                o += wt * conv_x[idx]
            return o
        
        def jac(x):
            a_x = softmax_params_to_a(x)
            conv_x = fast_convolve(a_x)
            sm, w = smooth_max_and_weights(conv_x, beta)
            grad_a = compute_grad_conv(a_x, w)
            for idx, wt in zip(top_k, peak_weights):
                for m in range(n):
                    j = idx - m
                    if 0 <= j < n:
                        grad_a[m] += wt * 2.0 * a_x[j]
            lagrange = np.dot(a_x, grad_a)
            return a_x * (grad_a - lagrange)
        
        maxiter = max(50, 200 - outer * 15)
        result = minimize(obj, x0, jac=jac, method='L-BFGS-B',

        a = softmax_params_to_a(result.x)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        
        # Decay beta
        beta = beta_start * (beta_end / beta_start) ** ((outer + 1) / n_iters_outer)
    
    return a


def simulated_annealing(a0, n_steps=500, t_start=0.01, t_end=0.001):
    """Simulated annealing on the simplex."""
    n = len(a0)
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_obj = exact_objective(a)
    
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(1, n_steps - 1))
        
        # Random perturbation
        noise_scale = 0.05 * (1 + step / n_steps)
        noise = np.random.standard_normal(n) * noise_scale
        a_new = project_simplex(a + noise)
        
        obj_new = exact_objective(a_new)
        obj_cur = exact_objective(a)
        
        if obj_new < obj_cur or np.random.random() < np.exp(-(obj_new - obj_cur) / max(t, 1e-10)):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
    
    return best_a, best_obj


def coordinate_descent(a0, n_sweeps=10):
    """Coordinate descent: try adjusting each element individually."""
    n = len(a0)
    a = a0.copy()
    best_obj = exact_objective(a)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = a[i]
            if old < 1e-15:
                continue
            # Try several scales
            for s in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[i] = old * s
                obj = exact_objective(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    improved = True
                    break
                a[i] = old
            if not improved:
                a[i] = old
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
    
    return a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-phase optimization with peak-penalized smooth objectives
- Adaptive beta scheduling for log-sum-exp approximation
- Simulated annealing for global exploration
- Coordinate descent for local refinement
- Length adaptation and edge suppression
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, w):
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    return np.real(ifft(np.conj(W) * A, N))[:n] * 2.0


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.sum(u * np.arange(1, n + 1) > (cssv - 1.0))
    if rho < 1:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def exact_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_params_to_a(x):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    return a


def _smooth_obj(x, n, beta):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _smooth_jac(x, n, beta):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def _peak_obj(x, n, beta, peak_idx, peak_wt):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    obj = 2.0 * n * sm
    for idx, wt in zip(peak_idx, peak_wt):
        if idx < len(conv):
            obj += wt * conv[idx]
    return obj


def _peak_jac(x, n, beta, peak_idx, peak_wt):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_conv(a, w)
    for idx, wt in zip(peak_idx, peak_wt):
        if idx < len(conv):
            for m in range(n):
                j = idx - m
                if 0 <= j < n:
                    grad_a[m] += wt * 2.0 * a[j]
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def _weighted_smooth_obj(x, n, beta, w_override):
    a = softmax_params_to_a(x)
    conv = fast_convolve(a)
    val = np.dot(w_override, conv)
    return 2.0 * n * val


def optimize_smooth(a0, beta, maxiter=300):
    n = len(a0)
    a0_pos = np.maximum(a0, 1e-15)
    a0_pos = a0_pos / a0_pos.sum()
    x0 = np.log(a0_pos)
    result = minimize(_smooth_obj, x0, args=(n, beta), jac=_smooth_jac,
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    a_opt = softmax_params_to_a(result.x)
    a_opt = np.maximum(a_opt, 0)
    s = a_opt.sum()
    if s > 1e-10:
        a_opt = a_opt / s
    return a_opt, result.fun


def peak_penalized_optimize(a0, n_iters_outer=8, n_peaks=15, beta_start=60.0, beta_end=5.0):
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 1e-10:
        a = a / s

    for outer in range(n_iters_outer):
        conv = fast_convolve(a)
        peak_vals = conv.copy()
        margin = max(1, len(conv) // 10)
        peak_vals[:margin] = -np.inf
        peak_vals[-margin:] = -np.inf
        top_k = np.argsort(peak_vals)[::-1][:n_peaks]
        peak_weights = np.ones(n_peaks) * (0.3 - 0.2 * outer / max(1, n_iters_outer))

        beta = beta_start * (beta_end / beta_start) ** ((outer + 1.0) / n_iters_outer)

        a0_pos = np.maximum(a, 1e-15)
        a0_pos = a0_pos / a0_pos.sum()
        x0 = np.log(a0_pos)

        result = minimize(_peak_obj, x0, args=(n, beta, top_k, peak_weights),
                          jac=_peak_jac, method='L-BFGS-B',
                          options={'maxiter': max(50, 200 - outer * 10), 'maxls': 40})
        a = softmax_params_to_a(result.x)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s

    return a


def simulated_annealing(a0, n_steps=500, t_start=0.005, t_end=0.0003, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 1e-10:
        a = a / s
    best_a = a.copy()
    best_obj = exact_objective(a)

    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(1, n_steps - 1))
        noise_scale = 0.03 + 0.03 * (1.0 - step / max(1, n_steps))
        noise = rng.standard_normal(n) * noise_scale
        a_new = project_simplex(a + noise)
        obj_new = exact_objective(a_new)
        obj_cur = exact_objective(a)
        if obj_new < obj_cur or (np.isfinite(obj_new) and np.isfinite(obj_cur) and
                                 rng.random() < np.exp(-min(50.0, (obj_new - obj_cur) / max(t, 1e-12)))):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
    return best_a, best_obj


def coordinate_descent(a0, n_sweeps=10, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 1e-10:
        a = a / s
    best_obj = exact_objective(a)
    scales = [0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        for i in range(n):
            old = a[i]
            if old < 1e-12:
                # Try setting to small positive
                for tiny in [0.001, 0.005]:
                    a[i] = tiny
                    obj = exact_objective(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        break
                    a[i] = 0.0
                continue
            improved = False
            for sc in scales:
                a[i] = old * sc
                obj = exact_objective(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    improved = True
                    break
            if not improved:
                a[i] = old
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
    return a, best_obj


def resample_sequence(a, new_n):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def generate_initializations(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    inits['uniform'] = np.ones(n) / n

    tri = 1.0 - np.abs(x - mid) / (mid + 1.0)
    tri = np.maximum(tri, 0)
    tri_sum = tri.sum()
    if tri_sum > 1e-10:
        inits['triangle'] = tri / tri_sum

    for sigma in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sigma * n)) ** 2)
        g_sum = g.sum()
        if g_sum > 1e-10:
            inits['gauss_%.2f' % sigma] = g / g_sum

    cos_bump = np.cos(np.pi * (x - mid) / n) ** 2
    cos_bump = np.maximum(cos_bump, 0)
    cb_sum = cos_bump.sum()
    if cb_sum > 1e-10:
        inits['cosine'] = cos_bump / cb_sum

    edge = max(1, n // 10)
    ft = np.zeros(n)
    ft[edge:n - edge] = 1.0
    ft_sum = ft.sum()
    if ft_sum > 1e-10:
        inits['flat_top'] = ft / ft_sum

    exp_dec = np.exp(-np.abs(x - mid) / (0.25 * n))
    ed_sum = exp_dec.sum()
    if ed_sum > 1e-10:
        inits['exp_center'] = exp_dec / ed_sum

    # Raised cosine with controlled edge
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - mid) / n))
    rc = np.maximum(rc, 0)
    rc_sum = rc.sum()
    if rc_sum > 1e-10:
        inits['raised_cos'] = rc / rc_sum

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent_resampled'] = pp / ps

    return inits
