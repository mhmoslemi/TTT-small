def fast_convolve(a):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
fast mirror descent for initial optimization, LP-based refinement with 
Nesterov acceleration, and local coordinate descent for final polish.
Uses FFT-based convolutions for speed and multiple initializations

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_pair(a, b):
    """Linear convolution of a and b via FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def objective_value(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def normalize(a):
    """Normalize a to have sum 1 and be non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix_sparse_info(a):
    """For the LP: the constraint matrix C has C[k,j] = a[j] if 0 <= k-j <= n-1, else 0.
    Actually C[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0.
    Wait, let me be more careful.
    
    (a*g)_k = sum_j a_{k-j} * g_j for valid indices.
    So the k-th row of C has C[k,j] = a[k-j] if 0 <= k-j < n and 0 <= j < n, else 0.
    """
    n = len(a)
    # C is (2n-1) x n
    # C[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a*g)_k includes a[k-j]*g_j for k = j, j+1, ..., j+n-1
        C[j:j+n, j] = a
    return C


def solve_lp_direction(a):
    """
    Solve LP: minimize t s.t. (a*g)_k <= t for all k, sum(g) = 1, g >= 0.
    Returns optimal g.
    """
    n = len(a)
    C = build_conv_matrix_sparse_info(a)
    
    # Variables: g_0, ..., g_{n-1}, t
    # minimize t
    # subject to: C @ g - t <= 0 (2n-1 constraints)
    #              sum(g) = 1
    #              g >= 0, t >= 0 (t is free actually, but we can bound it)
    
    # Using linprog format:
    # c @ [g, t] = t (minimize)
    # A_ub @ [g, t] <= b_ub
    # A_eq @ [g, t] = b_eq
    # bounds
    
    c = np.append(np.zeros(n), 1.0)  # minimize t
    
    # (a*g)_k - t <= 0 => C @ g - t <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # sum(g) = 1
    A_eq = np.hstack([np.ones(n), 0.0]).reshape(1, -1)
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g, result.x[n]  # g, optimal t
    return None, None


def lp_optimize(a0, time_limit, verbose=False, nesterov=True):
    """
    LP-based iterative optimization with optional Nesterov acceleration.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Nesterov momentum
    a_momentum = a.copy()
    mu = 0.0  # momentum coefficient, will increase
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit:
        # For Nesterov: solve LP with a_momentum's cross-convolution constraints
        # Actually, the LP uses a (current point) for the constraint matrix
        # Nesterov: evaluate at y = a + mu*(a - a_prev), solve LP with y
        
        if nesterov and iteration > 0:
            mu = 1.0 / (1.0 + 0.5 * np.sqrt(1.0 + 4.0 * mu * mu))
            y = a + mu * (a - a_momentum)
            y = normalize(y)
        else:
            y = a.copy()
        
        g, t_opt = solve_lp_direction(y)
        
        if g is None:
            no_improve += 1
            if no_improve > max_no_improve:
                # Perturb
                rng = np.random.default_rng(iteration * 7 + 13)
                a = best_a + rng.standard_normal(n) * 0.005
                a = normalize(a)
                c = fast_convolve(a)
                best_obj = c.max()
                no_improve = 0
                mu = 0.0
            iteration += 1
            continue
        
        a_momentum = a.copy()
        a_new = g
        
        # Check improvement
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        elif obj_new < best_obj + 1e-12:
            # Accept for Nesterov (don't update best, but move)
            a = a_new
            no_improve += 1
        else:
            no_improve += 1
            mu = max(0, mu * 0.9)
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.003 + 0.002 * min(no_improve // 100, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = normalize(a)
            c = fast_convolve(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
            no_improve = 0
            mu = 0.0
        
        iteration += 1
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, time={elapsed:.1f}s")

def mirror_descent_optimize(a0, time_limit, alpha=0.8, temp_init=0.005, temp_growth=1.005):
    """
    Fast mirror descent using FFT-based convolutions.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    temp = temp_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute weights on peaks (softmax)
        M = c.max()
        if M < 1e-15:
            break
        
        scaled = (c - M) / (temp * M + 1e-15)
        scaled = np.clip(scaled, -40, 0)
        y = np.exp(scaled)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = np.zeros_like(c)
            y[np.argmax(c)] = 1.0
        else:
            y /= y_sum
        
        # Cross-correlation: d_j = sum_k y_k * a_{k-j} for valid indices
        # This is convolution of y and reversed a, or we can compute directly
        # d_j = (y * a)_j where * is linear convolution, take first n
        d = fast_convolve_pair(y, a)[:n]
        
        # Multiplicative update
        d_clipped = np.maximum(d, 1e-18)
        a_new = a * np.power(d_clipped, alpha)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            c = fast_convolve(a)
            no_improve += 100
        else:
            a = a_new / s_new
            c = fast_convolve(a)
            obj_new = c.max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        temp *= temp_growth
        iteration += 1
        
        if no_improve > 2000:
            # Perturb and reset
            rng = np.random.default_rng(iteration)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = normalize(a)
            c = fast_convolve(a)
            no_improve = 0
            temp = temp_init
        
        if iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {iteration}, obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, num_sweeps=100):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective depends on a_i through the convolution terms
    that involve a_i. We can do a 1D optimization.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        # Random order of coordinates
        rng = np.random.default_rng(sweep)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Try different values for a[i]
            # The objective as a function of a[i] (with sum(a) = 1 fixed)
            # involves adjusting another coordinate. For simplicity, 
            # we adjust a[i] and renormalize.
            
            # Try a few multiplicative factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                a_try = a.copy()
                a_try[i] = a[i] * factor
                a_try = normalize(a_try)
                
                c_try = fast_convolve(a_try)
                obj_try = c_try.max()
                
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
        
        sweep += 1
        if sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep {sweep}, obj={best_obj:.8f}, time={elapsed:.1f}s")
        
        if sweep >= num_sweeps:
            break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Triangle (peak at center)
    tri = np.abs(x - mid) / mid + 0.5
    inits['triangle'] = tri / tri.sum()
    
    # Inverted triangle (valley at center)
    inv_tri = 1.0 / (np.abs(x - mid) / mid + 1.0)
    inits['inv_triangle'] = inv_tri / inv_tri.sum()
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - mid) / (n * 0.2)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    inits['raised_cos'] = rc / rc.sum()
    
    # Sinc-like (Prolate approximation)
    bw = 0.8  # bandwidth parameter
    sinc = np.sinc(bw * (x - mid) / n)
    sinc = np.abs(sinc) + 0.01
    inits['sinc'] = sinc / sinc.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                # Downsample
                pp = np.interp(np.linspace(0, n - 1, n), np.linspace(0, len(p) - 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                # Upsample
                pp = np.interp(np.linspace(0, n - 1, n), np.linspace(0, len(p) - 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1) Fast FFT-based mirror descent for initial optimization,
2) LP-based direction finding with adaptive relaxation,
3) Nesterov-accelerated LP iterations,
4) Coordinate descent and local search for final polish.
Uses multiple initializations and sequence lengths to avoid local optima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_pair(a, b):
    """Linear convolution of a and b via FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def normalize(a):
    """Normalize a to sum 1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix C where (C @ g)[k] = (a * g)[k] (linear conv)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_max_sum(a, M):
    """
    Solve: maximize sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g_star, sum_g_star). If infeasible or degenerate, returns (None, None).
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        g = np.maximum(result.x, 0.0)
        g_sum = g.sum()
        return g, g_sum
    return None, None


def lp_optimize(a0, time_limit, nesterov=False, adaptive_relax=True, verbose=False):

    LP-based iterative optimization.
    At each step: solve max sum(g) s.t. (a*g) <= M, g>=0.
    If sum(g*) > 1, rescale and update a.
    """

    n = len(a)

    c_conv = fast_convolve(a)
    M = c_conv.max()
    best_obj = M
    best_a = a.copy()

    if nesterov:
        a_prev = a.copy()
        mu = 0.0

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < time_limit:

        if nesterov and iteration > 2:

            y = a + mu * (a - a_prev)
            y = normalize(y)
            c_y = fast_convolve(y)
            M_y = c_y.max()
            g, g_sum = solve_lp_max_sum(y, M_y * relax_factor)

def mirror_descent_optimize(a0, time_limit, alpha=0.7, temp_init=0.003,
                            temp_growth=1.003, verbose=False):
    """
    Fast mirror descent using FFT convolutions.
    Update: a_new_j = a_j * (d_j)^alpha where d = cross-correlation with peak weights.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)

    c_conv = fast_convolve(a)
    best_obj = c_conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    temp = temp_init
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        M = c_conv.max()
        if M < 1e-15:
            break

        # Softmax weights concentrated on peaks
        scaled = (c_conv - M) / (temp * M + 1e-15)
        scaled = np.clip(scaled, -35, 0)
        y = np.exp(scaled)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = np.zeros_like(c_conv)
            y[np.argmax(c_conv)] = 1.0
        else:
            y /= y_sum

        # d_j = sum_k y_k * a_{k-j} (cross-correlation, first n entries)
        d = fast_convolve_pair(y, a)[:n]
        d = np.maximum(d, 1e-18)

        # Multiplicative update
        a_new = a * np.power(d, alpha)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            c_conv = fast_convolve(a)
            no_improve += 200
        else:
            a = a_new / s_new
            c_conv = fast_convolve(a)
            obj_new = c_conv.max()

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        temp *= temp_growth
        iteration += 1

        if no_improve > 3000:
            rng = np.random.default_rng(iteration)
            noise = rng.standard_normal(n) * 0.008
            a = best_a + noise
            a = normalize(a)
            c_conv = fast_convolve(a)
            no_improve = 0
            temp = temp_init * 0.5

        if verbose and iteration % 5000 == 0:
            print(f"      MD iter {iteration}, obj={best_obj:.8f}, "
                  f"time={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def local_search_optimize(a0, time_limit, verbose=False):

    Local search: try pairwise swaps and coordinate adjustments.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)

    c_conv = fast_convolve(a)
    best_obj = c_conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Strategy 1: Coordinate scaling
        i = rng.integers(0, n)
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.3]:
            a_try = a.copy()
            a_try[i] *= factor
            if a_try[i] < 0:
                continue
            a_try = normalize(a_try)
            c_try = fast_convolve(a_try)
            if c_try.max() < best_obj - 1e-14:
                best_obj = c_try.max()
                best_a = a_try.copy()
                a = a_try
                no_improve = 0
                break
        else:
            no_improve += 1

        # Strategy 2: Pairwise mass transfer
        if rng.random() < 0.5:
            i, j = rng.integers(0, n, 2)
            if i != j:
                for eps in [0.001, 0.005, 0.01, 0.02, -0.005, -0.01]:
                    a_try = a.copy()
                    a_try[i] += eps
                    a_try[j] -= eps
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = normalize(a_try)
                    c_try = fast_convolve(a_try)
                    if c_try.max() < best_obj - 1e-14:
                        best_obj = c_try.max()
                        best_a = a_try.copy()
                        a = a_try
                        no_improve = 0

        if no_improve > 5000:
            break

        if verbose and (iteration := int(time.monotonic() - t0)) % 30 == 0:
            print(f"      LS time={time.monotonic()-t0:.1f}s, obj={best_obj:.8f}")

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.25)

    # Bimodal (mass at edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Inverted V (valley at center)
    inv_v = 1.0 / (np.abs(x - mid) / max(1.0, mid) + 0.3)
    inits['inv_v'] = inv_v / inv_v.sum()

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    inits['raised_cos'] = rc / rc.sum()

    # Sinc-like
    bw = 1.0
    sinc = np.sinc(bw * (x - mid) / n)
    sinc = np.abs(sinc) + 0.05
    inits['sinc'] = sinc / sinc.sum()

    # Exponential decay from edges
    expd = np.exp(-np.abs(x - mid) / (n * 0.3))
    inits['exp_decay'] = expd / expd.sum()

    # Alternating
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    inits['alternating'] = alt / alt.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, n - 1, n),
                               np.linspace(0, len(p) - 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

                best_sequence = a_opt.copy()
                print(f"  Nesterov LP: score={score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # === Phase 4: Local search ===
    print("=== Phase 4: Local Search ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0
        if remaining > 20:
            a_opt, obj_opt = local_search_optimize(best_sequence, time_limit=remaining)
            score = objective_value(a_opt)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Local search: score={score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # === Phase 5: Final LP polish ===
    print("=== Phase 5: Final LP ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=remaining,
                                         nesterov=False, adaptive_relax=False)
            score = objective_value(a_opt)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = objective_value(best_sequence)

    # Final verification using the actual evaluator
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
