import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, beta):
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def optimize_adam(a0, time_limit, beta_start=20.0, beta_end=500.0,

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    perturb_count = 0
    
    t0 = time.monotonic()

        elapsed = time.monotonic() - t0

def lp_direction(a, M, n):
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 60})
    except Exception:
        return None
    if not result.success or result.x is None:
        return None
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum

def upsample(a, factor=2):
    """Upsample by factor using linear interpolation."""
    n = len(a)
    new_n = n * factor
    # Use np.interp for smooth upsample
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)


def generate_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangular (inverted)
    tri = 1.0 / (1.0 + 0.3 * ((2 * x / (n - 1) - 1)) ** 2)
    inits['tri_inv'] = tri / tri.sum()
    
    # Cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 1e-8)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    inits['gauss_c'] = g / g.sum()
    
    # Gaussian wide
    g2 = np.exp(-0.5 * ((x - n / 2) / (n / 2.5)) ** 2)
    inits['gauss_w'] = g2 / g2.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random Dirichlet
    inits['dirich1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dirich2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = upsample(np.maximum(parent, 0), factor=n // len(parent) if len(parent) > 1 else 2)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
    
    return inits
