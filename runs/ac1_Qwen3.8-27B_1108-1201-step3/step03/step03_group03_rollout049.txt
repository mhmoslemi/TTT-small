**Coordinate descent with peak awareness:**

For each peak location k in the convolution:
- The gradient w.r.t. a[j] is a[k-j] (for valid j)
- To reduce the peak, we should decrease a[j] where a[k-j] is large

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral domain amplitude
flattening, peak-directed simulated annealing, and coordinate descent.
Combines IAA-like spectral iterations with SA exploration and greedy CD polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(a, conv):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, time_limit, alpha_start=1.0, alpha_end=0.02, rng=None):
    """IAA-like spectral amplitude flattening."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    fft_size = 2 ** (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = obj_from_conv(a, fast_convolve(a))
    t0 = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, it / max(1, int(time_limit * 500)))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        target = amp.mean()
        if target < 1e-30:
            break
        
        # Modify amplitude towards flat
        ratio = target / (amp + 1e-30)
        amp_new = amp * np.power(ratio, alpha)
        phase = A / (amp + 1e-30)
        A_new = phase * amp_new
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ns
        
        conv_new = fast_convolve(a_new)
        cur_obj = obj_from_conv(a_new, conv_new)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb to escape
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
                stall = 0
        
        it += 1
    
    return best_a, best_obj


def simulated_annealing(a, time_limit, rng, T0=0.005, T_min=1e-7, cooling=0.99995):
    """Peak-directed simulated annealing on the simplex."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = obj_from_conv(a, fast_convolve(a))
    cur_obj = best_obj
    cur_conv = fast_convolve(a)
    
    T = T0
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit and T > T_min:
        r = rng.random()
        
        if r < 0.5:
            # Peak-directed: reduce the current peak
            peak_k = np.argmax(cur_conv)
            # Gradient of conv[peak_k] w.r.t. a[j]
            grad = np.zeros(n)
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            for j in range(j_start, j_end + 1):
                grad[j] = a[peak_k - j]
            
            # Move: decrease high-grad coords, increase low-grad coords
            scale = 0.005 + 0.02 * (T / T0)
            direction = -grad + rng.standard_normal(n) * scale * 0.5
            step = scale * (1.0 + T / T0)
            a_new = project_simplex(a + step * direction / (np.linalg.norm(direction) + 1e-15))
            
        elif r < 0.75:
            # Random perturbation
            scale = 0.01 + 0.03 * (T / T0)
            a_new = project_simplex(a + rng.standard_normal(n) * scale)
            
        else:
            # Pairwise transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                T *= cooling
                continue
            d = rng.random() * 0.05 * max(a[i], a[j], 1e-6)
            a_new = a.copy()
            if rng.random() < 0.5:
                a_new[i] -= d
                a_new[j] += d
            else:
                a_new[i] += d
                a_new[j] -= d
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-15:
                T *= cooling
                continue
            a_new = a_new / ns
        
        cur_conv = fast_convolve(a_new)
        cur_new = obj_from_conv(a_new, cur_conv)
        delta = cur_new - cur_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_new
            cur_obj = cur_new
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 500:
            # Larger perturbation
            a = project_simplex(a + rng.standard_normal(n) * 0.02)
            cur_conv = fast_convolve(a)
            cur_obj = obj_from_conv(a, cur_conv)
            stall = 0
        
        T *= cooling
    
    return best_a, best_obj

def build_lp_pairs(n):
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit, blend=0.3):
    """LP-based directional optimization."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = obj_from_conv(a, fast_convolve(a))
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        a_new = (1.0 - blend) * a + blend * g
        a_new = project_simplex(a_new)
        conv_new = fast_convolve(a_new)
        cur_obj = obj_from_conv(a_new, conv_new)
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 5:
                a = project_simplex(a + np.random.default_rng(stall).standard_normal(n) * 0.01)
                stall = 0
                if time.monotonic() - t0 > time_limit * 0.8:
                    break
    
    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sinc-like (inverse of flat spectrum)
    t = (x - (n - 1) / 2.0) / (n / 4.0)
    sinc = np.sinc(t)
    sinc = np.abs(sinc)
    sinc[sinc < 0.01] = 0.01
    inits['sinc'] = sinc / sinc.sum()
    
    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1 + 1e-10)) ** 2
    cos_w = np.maximum(cos_w, 0.001)
    inits['cosine'] = cos_w / cos_w.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(parent)), p)
                p_interp = np.maximum(p_interp, 0)
                pis = p_interp.sum()
                if pis > 1e-10:
                    inits['parent_interp'] = p_interp / pis
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using IAA-like spectral flattening,
peak-directed simulated annealing, and greedy coordinate descent.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(a, conv):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, time_limit, alpha_start=1.0, alpha_end=0.02, rng=None):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(0)
    fft_size = 2 ** (2 * n - 1).bit_length()
    best_a = a.copy()
    best_obj = obj_from_conv(a, fast_convolve(a))
    t0 = time.monotonic()
    stall = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, it / max(1, int(time_limit * 500)))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        target = amp[0]  # DC component is the sum, should be largest for non-neg
        # Actually, let's use the mean of non-DC components
        if len(amp) > 1:
            target = amp[1:].mean()
        else:
            target = amp.mean()
        if target < 1e-30:
            break
        ratio = 1.0
        # We want to flatten: reduce high amplitudes, boost low ones
        # ratio = target / amp, then amp_new = amp * ratio^alpha
        # This makes amp_new = amp * (target/amp)^alpha = target^alpha * amp^(1-alpha)
        # For alpha=1: amp_new = target (completely flat)
        # For alpha=0: amp_new = amp (no change)
        amp_new = np.power(amp, 1.0 - alpha) * np.power(target, alpha)
        # But we should keep the DC component fixed (it determines the sum)
        if len(amp_new) > 0:
            phase = A / (amp + 1e-30)
            A_new = phase * amp_new
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-15:

            conv_new = fast_convolve(a_new)
            cur_obj = obj_from_conv(a_new, conv_new)
            if cur_obj < best_obj - 1e-14:

                a = a_new
                stall = 0

                stall += 1
                if stall > 100:
                    a = project_simplex(a + rng.standard_normal(n) * 0.01)
                    stall = 0
        it += 1
    return best_a, best_obj
