import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -80, 80)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    sm = x_max + np.log(total) / beta
    return sm, w


def conv_gradient(a, w):
    """
    Gradient of sum_k w[k]*conv_k(a) w.r.t. a.
    Uses FFT cross-correlation: grad[m] = 2*(w * a_rev)[n-1+m]
    where a_rev = a reversed.
    """
    n = len(a)
    a_rev = a[::-1]
    N = 1 << (2 * 2 * n).bit_length()
    W = fft(w, N)
    AR = fft(a_rev, N)
    conv_val = np.real(ifft(W * AR, N))
    grad = 2.0 * conv_val[n - 1:n - 1 + n]
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_to_simplex(x):
    """Convert log-space params to simplex."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def simplex_to_softmax(a):
    """Convert simplex point to log-space params."""
    a_clipped = np.maximum(a, 1e-300)
    a_clipped = a_clipped / a_clipped.sum()
    return np.log(a_clipped)


def obj_softmax(x, n, beta):
    """Objective in softmax parameterization."""
    a = softmax_to_simplex(x)
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def jac_softmax(x, n, beta):
    """Jacobian in softmax parameterization."""
    a = softmax_to_simplex(x)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = conv_gradient(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def run_lbfgsb(a0, n, beta, maxiter, time_limit):
    """Run L-BFGS-B with time monitoring. Returns (a_opt, obj_value)."""
    x0 = simplex_to_softmax(a0)
    t0 = time.monotonic()

    def func_with_check(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        return obj_softmax(x, n, beta)

    def jac_with_check(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        return jac_softmax(x, n, beta)

    try:
        result = minimize(func_with_check, x0, jac=jac_with_check, args=(),
                          method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 40})
        a_opt = softmax_to_simplex(result.x)
        return a_opt, result.fun
    except (TimeoutError, Exception):
        return a0, obj_softmax(x0, n, beta)


def run_pg_momentum(a0, beta, lr, mu, n_iter, time_limit):
    """Projected gradient with momentum on simplex."""
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = float('inf')
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        grad_a = conv_gradient(a, w)
        # Center gradient to maintain sum=1
        grad_centered = grad_a - np.mean(grad_a)

        v = mu * v - lr * grad_centered
        a_new = project_simplex(a + v)
        a = a_new
        # Decay learning rate
        lr *= 0.999

    return best_a, best_obj


def run_exact_subgrad(a0, n_iter, lr_init, time_limit):
    """Subgradient method using exact max (single active peak)."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        k_star = int(np.argmax(conv))
        # grad[m] = 2*a[k_star - m] if 0 <= k_star - m < n
        grad = np.zeros(n)
        m_start = max(0, k_star - (n - 1))
        m_end = min(n - 1, k_star)
        for m in range(m_start, m_end + 1):
            grad[m] = 2.0 * a[k_star - m]

        lr = lr_init / (1.0 + 0.01 * it)
        grad_centered = grad - np.mean(grad)
        a = project_simplex(a - lr * grad_centered)

    return best_a, best_obj


def coordinate_descent(a0, n_sweeps, n_per_leaf, time_limit):
    """Try small multiplicative perturbations of individual coordinates."""
    n = len(a0)
    a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    scales = [0.95, 0.9, 0.8, 1.05, 1.1, 1.15]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = a[i]
            for s in scales:
                a[i] = old * s
                conv = fast_convolve(a)
                new_obj = 2.0 * n * conv.max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                else:
                    a[i] = old
                    break
            else:
                a[i] = old

    return a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    g2 = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_narrow'] = g2

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Flat spectrum init
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
    for k in range(1, fft_size // 2):
        A[k] = np.exp(1j * phases[k - 1])
        A[fft_size - k] = np.exp(-1j * phases[k - 1])
    if fft_size % 2 == 0:
        A[fft_size // 2] = np.exp(1j * phases[-1])
    a_flat = np.real(ifft(A, fft_size))[:n]
    a_flat = np.maximum(a_flat, 0.0)
    s = a_flat.sum()
    inits['flat_spec'] = a_flat / s if s > 1e-10 else np.ones(n) / n

    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(pp)), pp)
                pp_interp = np.maximum(pp_interp, 0)
                s2 = pp_interp.sum()
                inits['parent_interp'] = pp_interp / s2 if s2 > 1e-10 else np.ones(n) / n

    return inits


def evaluate_safe(a):
    """Evaluate using the provided evaluate_sequence, with safety checks."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        score = float(evaluate_sequence(a_list))
        return score if np.isfinite(score) else float('inf')
    except Exception:
        return float('inf')
