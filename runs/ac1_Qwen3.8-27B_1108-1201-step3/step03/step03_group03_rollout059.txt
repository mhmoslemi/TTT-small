"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive temperature annealing
with LSE-smoothed gradients (Adam), LP-based directional refinement, and
coordinate descent with 1D line search. Explores multiple sequence lengths
and initializations to find near-optimal sequences with flat self-convolution.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def compute_gradient(a, p):
    """Compute grad[i] = sum_j a[j] * p[i+j] for i=0..n-1."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = np.dot(a, p[i:i+n])
    return grad


def compute_gradient_fft(a, p):
    """FFT-based gradient computation."""
    n = len(a)
    size = 2 ** (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    result = np.fft.irfft(np.conj(A) * P, size)[:n]
    return result


def lse_objective(a, t):
    """Compute LSE approximation of max(conv(a,a)) with temperature t."""
    conv = fast_convolve(a)
    conv_scaled = t * conv
    max_val = conv_scaled.max()
    log_sum = max_val + np.log(np.sum(np.exp(conv_scaled - max_val)))
    return log_sum / t


def lse_gradient(a, t):
    """Compute gradient of LSE approximation."""
    conv = fast_convolve(a)
    conv_scaled = t * conv
    max_val = conv_scaled.max()
    exp_vals = np.exp(conv_scaled - max_val)
    p = exp_vals / exp_vals.sum()
    grad = compute_gradient_fft(a, p)
    return grad, conv


def adam_lse_optimize(a0, time_limit, t_init=None, t_final=None,
                      lr_init=0.005, beta1=0.9, beta2=0.999,
                      norm_factor=None, num_anneal_steps=200):
    """Adam optimizer with LSE-smoothed objective and temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if t_init is None:
        t_init = 1.0 / n
    if t_final is None:
        t_final = 50.0 * n
    if norm_factor is None:
        norm_factor = n

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        step += 1
        frac = min(1.0, step / num_anneal_steps)
        # Geometric annealing
        t = t_init * (t_final / t_init) ** frac
        lr = lr_init / (norm_factor ** 0.5)

        grad, conv = lse_gradient(a, t)
        # Scale gradient by 1/n to normalize
        grad = grad / n

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a)

        cur_conv = fast_convolve(a)
        cur_obj = 2.0 * n * cur_conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        # Check time
        if time.monotonic() - t0 > time_limit - 0.5:
            break

    return best_a, best_obj


def lp_optimize(a0, time_limit, blend_init=0.3, blend_min=0.003,
                blend_halve_every=12):

        a = a / s
    n = len(a)

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min * 1.1:
                # Try perturbation

                conv = fast_convolve(a)
                if conv.max() * 2 * n < best_obj:
                    best_obj = 2.0 * n * conv.max()
                    best_a = a.copy()
                blend = blend_min * 2
                stall = 0

    return best_a, best_obj


def coordinate_descent_1d(a_in, max_sweeps=50, time_limit=30.0):
    """Coordinate descent with 1D golden section search per coordinate."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    t0 = time.monotonic()
    golden = 0.618033988749895
    inv_golden = 1.0 - golden

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        improved = False

        for i in range(n):
            if a[i] < 1e-12:
                continue

            # Golden section search over a[i] in [0, min(2*a[i]+0.01, 0.5)]
            lo = 0.0
            hi = min(2.0 * a[i] + 0.01, 0.5)

            def obj_at_x(x):
                if x < 0 or x >= 1.0:
                    return 1e10
                a_other_sum = 1.0 - a[i]
                if a_other_sum < 1e-15:
                    return 1e10
                c = (1.0 - x) / a_other_sum
                a_test = a * c
                a_test[i] = x
                conv_test = fast_convolve(a_test)
                return conv_test.max()

            # Golden section
            c1 = hi - golden * (hi - lo)
            c2 = lo + golden * (hi - lo)
            f1 = obj_at_x(c1)
            f2 = obj_at_x(c2)

            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - golden * (hi - lo)
                    f1 = obj_at_x(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + golden * (hi - lo)
                    f2 = obj_at_x(c2)

            x_opt = (lo + hi) / 2.0
            f_opt = obj_at_x(x_opt)

            if f_opt < best_obj / (2.0 * n) - 1e-14:
                a_other_sum = 1.0 - a[i]
                if a_other_sum < 1e-15:
                    continue
                c = (1.0 - x_opt) / a_other_sum
                a = a * c
                a[i] = x_opt
                conv = fast_convolve(a)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    improved = True

        if not improved:
            break

    return a


def subgradient_refine(a0, time_limit, top_k=30):
    """Subgradient descent with momentum, targeting top-k peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.003
    momentum = 0.9
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -(top_k if top_k < len(conv) else len(conv)))[-top_k:]
        pmax = conv[peaks].max()
        if pmax < 1e-15:
            break

        # Weight peaks by their value (softer: use log)
        w = np.log1p(conv[peaks] * n)
        w = w / w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(0.0005, lr * 0.7)

        lr *= 0.9995

    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Cosine-tapered
    cos_t = np.cos(np.pi * x / (n - 1)) ** 2
    cos_t = np.maximum(cos_t, 0.001)
    inits['cosine'] = cos_t / cos_t.sum()

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                prs = p_res.sum()
                if prs > 1e-10:
                    inits['parent_res'] = p_res / prs

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining Adam+LSE annealing, LP directional optimization,
    subgradient refinement, and coordinate descent with 1D line search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass

    # Phase 1: Adam+LSE scan over lengths and initializations
    print("=== Phase 1: Adam+LSE Scan ===")
    if parent is not None:
        lengths = [len(parent)]
        if len(parent) != 256:
            lengths.append(256)
        if len(parent) != 512:
            lengths.append(512)
        if len(parent) != 768:
            lengths.append(768)
        if len(parent) != 1024:
            lengths.append(1024)
        if len(parent) != 384:
            lengths.append(384)
    else:
        lengths = [256, 384, 512, 768, 1024]

    # Sort by priority: parent length first, then others
    lengths = sorted(set(lengths), key=lambda x: -x if parent is not None and x == len(parent) else x)
    lengths = lengths[:5]  # Limit to 5 lengths

    results = []
    phase1_end = time.monotonic() + min(600.0, (deadline - time.monotonic()) * 0.4)

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break

        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_res', 'uniform', 'bimodal', 'triangle',
                'cosine', 'random', 'gauss', 'ramp']
        names = [nm for nm in prio if nm in inits]
        names = names[:4]

        t_per = max(15.0, min(80.0, 300.0 / (len(lengths) * 4)))

        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]

            # Adam + LSE with annealing
            a_opt, obj_opt = adam_lse_optimize(
                a0, t_alloc * 0.6,
                t_init=1.0 / n, t_final=30.0 * n,
                lr_init=0.01, num_anneal_steps=min(300, int(t_alloc * 5))
            )

            # Quick LP refinement
            a_opt2, obj2 = lp_optimize(a_opt, min(10.0, t_alloc * 0.25),
                                       blend_init=0.2, blend_min=0.01)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            # Quick subgradient
            a_opt3, obj3 = subgradient_refine(a_opt, min(5.0, t_alloc * 0.1))
            if obj3 < obj_opt:
                a_opt, obj_opt = a_opt3, obj3

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    if results:
        print(f"Phase 1 best: {results[0][0]:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen_keys = set()
    for score, n, name, a_opt in results[:10]:
        key = (n, name)
        if key not in seen_keys:
            top_list.append((n, a_opt.copy()))
            seen_keys.add(key)
        if len(top_list) >= 4:
            break

    if best_sequence is not None and not any(
            len(a) == len(best_sequence) for _, a in top_list):
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    t_per = max(30.0, remaining / max(1, len(top_list) * 3))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 180:
            break

        # Sub-strategy A: Deep Adam with slow annealing
        t_alloc_a = min(t_per * 0.4, deadline - time.monotonic() - 150)
        if t_alloc_a > 10:
            a_a, obj_a = adam_lse_optimize(
                a0, t_alloc_a,
                t_init=5.0 * n, t_final=200.0 * n,
                lr_init=0.003, num_anneal_steps=500
            )
            score_a = float(evaluate_sequence(a_a.tolist()))
            if np.isfinite(score_a) and score_a < best_score:
                best_score = score_a
                best_sequence = a_a.copy()
                print(f"  Deep Adam n={n}: {score_a:.6f}")

        # Sub-strategy B: LP with slow blending
        t_alloc_b = min(t_per * 0.35, deadline - time.monotonic() - 130)
        if t_alloc_b > 10:
            a_b, obj_b = lp_optimize(
                a0, t_alloc_b,
                blend_init=0.15, blend_min=0.002, blend_halve_every=15
            )
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"  Deep LP n={n}: {score_b:.6f}")

        # Sub-strategy C: Subgradient with long horizon
        t_alloc_c = min(t_per * 0.25, deadline - time.monotonic() - 110)
        if t_alloc_c > 8:
            a_c, obj_c = subgradient_refine(a0, t_alloc_c, top_k=50)
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"  Deep Subgrad n={n}: {score_c:.6f}")

    # Phase 3: Coordinate descent + final polish
    print("=== Phase 3: Coordinate Descent + Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0

        # Coordinate descent
        t_cd = min(180.0, remaining * 0.4)
        if t_cd > 10:
            print(f"  Running coordinate descent (t={t_cd:.0f}s)...")
            a_cd = coordinate_descent_1d(best_sequence, max_sweeps=100,
                                         time_limit=t_cd)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")
            else:
                print(f"  CD: {score_cd:.6f} (no improvement)")

        # Interleave LP and subgradient
        remaining = deadline - time.monotonic() - 40.0
        for cycle in range(4):
            if time.monotonic() > deadline - 35:
                break
            t_cyc = min(remaining / (4 - cycle), 50.0)
            if t_cyc < 8:
                break

            # LP step
            a_opt, _ = lp_optimize(best_sequence, t_cyc * 0.4,
                                   blend_init=0.08, blend_min=0.002,
                                   blend_halve_every=20)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cycle} LP: {best_score:.6f}")

            # Subgradient step
            a_opt2, _ = subgradient_refine(best_sequence, t_cyc * 0.3, top_k=60)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  Cycle {cycle} Subgrad: {best_score:.6f}")

            # Quick CD
            a_cd2 = coordinate_descent_1d(best_sequence, max_sweeps=30,
                                          time_limit=t_cyc * 0.2)
            score_cd = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd2.copy()
                print(f"  Cycle {cycle} CD: {best_score:.6f}")

        # Random perturbations + quick re-optimization
        if time.monotonic() < deadline - 20:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 15:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 12:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    # Quick Adam
                    a_opt, _ = adam_lse_optimize(a_p, 5.0,
                                                 t_init=10.0 * n_b,
                                                 t_final=100.0 * n_b,
                                                 lr_init=0.002,
                                                 num_anneal_steps=100)
                    # Quick subgrad
                    a_opt2, _ = subgradient_refine(a_opt, 3.0, top_k=40)
                    score = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt2.copy()
                        print(f"  Perturb(p={ps}): {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
