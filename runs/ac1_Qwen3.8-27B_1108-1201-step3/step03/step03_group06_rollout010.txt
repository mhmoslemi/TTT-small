def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def solve_lp_direction_v2(a, conv_a, t=0.1):
    """Find b on simplex minimizing max_k [(a*b)[k] + conv_a[k]/(2t)]."""
    n = len(a)
    num_k = 2 * n - 1
    
    # Build A_ub: (a*b)[k] <= z - conv_a[k]/(2t) for each k
    # Variables: b[0],...,b[n-1], z
    # Objective: minimize z
    # Constraints: for each k, sum_i a[k-i]*b[i] - z <= -conv_a[k]/(2t)
    #             sum b[i] = 1
    #             b[i] >= 0
    #             z free (or z >= 0)
    
    A_ub = np.zeros((num_k, n + 1))
    b_ub = np.zeros(num_k)
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
        A_ub[k, n] = -1.0
        b_ub[k] = -conv_a[k] / (2.0 * t)
    
    # Equality: sum b = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize z
    
    bounds = [(0, None)] * n + [(None, None)]  # b[i] >= 0, z free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]  # Return b (first n components)
    except Exception:
        pass
    return None
