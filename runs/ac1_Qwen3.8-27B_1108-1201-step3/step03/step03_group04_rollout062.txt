For non-negative sequences, one approach is to use the "water-filling" method or iterative amplitude retrieval (like Fienup's error reduction algorithm).

**Fienup's algorithm for this problem:**
1. Start with an initial $a$ (e.g., uniform).
2. Compute DFT: $A = \text{DFT}(a)$.
3. Modify the amplitude: $|A| \leftarrow \max(|A|, target)$ or $|A| \leftarrow \sqrt{|A|^2 + \epsilon}$.
4. Compute IDFT: $a' = \text{IDFT}(A \cdot A/|A|)$.
5. Project to non-negative: $a' = \max(a', 0)$.
6. Repeat.

This is similar to the "Gerchberg-Saxton" algorithm used in phase retrieval!

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=5.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def fpw_lp_optimize(a0, time_limit, verbose=False, verbose_every=200):
    """Frank-Wolfe iterations with LP oracle."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M, time_limit=min(8.0, (time_limit - (time.monotonic() - t0)) * 0.8))
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            best_t = 0.0
            best_obj_trial = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1 - t) * a + t * b
                obj_c = local_obj(candidate)
                if obj_c < best_obj_trial:
                    best_obj_trial = obj_c
                    best_t = t
            if best_t < 0.05:
                no_improve += 1
            else:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        if no_improve >= max_stall:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated (two bumps at edges)
    edge_w = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec[-edge_w:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    # Sine envelope
    sine = np.sin(np.pi * x / (n - 1 + 1e-10)) ** 2
    sine /= sine.sum()
    inits['sine'] = sine.copy()
    # Cosine edge decay
    cos_d = np.cos(np.pi * x / (2 * n)) ** 2
    cos_d[0] = max(cos_d[0], 0.1)
    cos_d[-1] = max(cos_d[-1], 0.1)
    cos_d /= cos_d.sum()
    inits['cos'] = cos_d.copy()
    # Parabolic
    mid = (n - 1) / 2.0
    para = 1.0 - (x - mid) ** 2 / (mid ** 2 + 1e-10)
    para = np.maximum(para, 0.01)
    para /= para.sum()
    inits['para'] = para.copy()
    # Gaussian
    gauss = np.exp(-0.5 * ((x - mid) / (0.4 * n)) ** 2)
    gauss /= gauss.sum()
    inits['gauss'] = gauss.copy()
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    # Random sparse
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()
    # Anti-parabola (edge-heavy smooth)
    anti_para = (x - mid) ** 2 / (mid ** 2 + 1e-10) + 0.01
    anti_para /= anti_para.sum()
    inits['anti_para'] = anti_para.copy()
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    return inits


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: optimize each element in turn.
    For element i, the objective as a function of a_i (others fixed) is:
    2n * max_k (conv_k - 2*a_i*a_{k-i_valid} + a_i^2 * delta) / (sum)^2
    Since we're on the simplex, changing a_i changes the sum.
    Simplification: fix sum=1, so a_i is determined by others: a_i = 1 - sum_others.
    Actually, let's just do: for each i, try a few values and pick the best.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    best_obj = local_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    while time.monotonic() - t0 < time_limit:
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            # Try several values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, old_val * 0.25, old_val * 2.0,
                          0.0, 0.01 / n, 0.1 / n]
            candidates = [max(0.0, c) for c in candidates]
            best_cand = old_val
            best_cobj = best_obj
            for c in candidates:
                a_copy = a.copy()
                a_copy[i] = c
                s = a_copy.sum()
                if s < 1e-15:
                    continue
                a_copy = a_copy / s
                obj_c = local_obj(a_copy)
                if obj_c < best_cobj - 1e-12:
                    best_cobj = obj_c
                    best_cand = c
            a[i] = best_cand
            s = a.sum()
            if s > 1e-15:
                a = a / s
        cur_obj = local_obj(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD it={it}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: subgradient -> spectral -> smooth gradient -> LP polish -> CD.
    Explores multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Quick screening with subgradient (fast method)
    print("=== Phase 1: Subgradient Screening ===")
    lengths = [256, 384, 512, 640, 768]

    # Determine which lengths to try based on parent
    if parent is not None:
        n_parent = len(parent)
        lengths.insert(0, n_parent)
        # Also try nearby lengths
        lengths.append(max(64, int(n_parent * 0.75)))
        lengths.append(int(n_parent * 1.25))
    lengths = sorted(set(lengths))
    lengths = [n for n in lengths if 64 <= n <= 2048]

    candidates = []  # list of (sequence, score, length)

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'edge', 'anti_para', 'sine',
                    'cos', 'uniform', 'para', 'gauss', 'random', 'sparse']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        time_per_init = max(5.0, 80.0 / max(1, len(init_names)))

        for name in init_names[:5]:
            if time.monotonic() > deadline - 250:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 200)
            if t_alloc < 3:
                break
            a0 = inits[name]
            a_opt, obj_opt = subgradient_optimize(a0, time_limit=t_alloc,
                                                  lr_init=0.008, momentum=0.92,
                                                  top_k=25, temp_init=0.15, temp_min=0.02)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                score_eval = score
            else:
                score_eval = local_obj(a_opt)
            print(f"  n={n}, init={name}, score={score_eval:.6f}")
            candidates.append((a_opt, score_eval, n))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}")

    # Phase 2: Spectral optimization on top candidates
    print("=== Phase 2: Spectral Optimization ===")
    candidates.sort(key=lambda x: x[1])
    top_cands = candidates[:3]
    for a_cand, _, _ in top_cands:
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(80.0, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break
        a_opt, obj_opt = spectral_optimize(a_cand, time_limit=t_alloc, beta=0.4)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  Spectral: score={score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # Phase 3: Smooth gradient optimization
    print("=== Phase 3: Smooth Gradient Optimization ===")
    if best_sequence is not None:
        t_alloc = min(150.0, deadline - time.monotonic() - 200)
        if t_alloc > 20:
            a_opt, obj_opt = smooth_gradient_optimize(best_sequence, time_limit=t_alloc,
                                                      tau_init=0.2, tau_min=0.01)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  SmoothGrad: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Phase 4: LP-based Frank-Wolfe polish
    print("=== Phase 4: LP Frank-Wolfe Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        t_alloc = remaining * 0.5
        if t_alloc > 20:
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_alloc,
                                             verbose=True, verbose_every=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  LP-FW: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Phase 5: Coordinate descent polish
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 5: Coordinate Descent Polish ===")
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 10:
            a_opt, obj_opt = coordinate_descent_polish(best_sequence, time_limit=remaining)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  CD: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # Phase 6: Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 6: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 15.0
        n_restarts = max(1, int(remaining // 15))
        n_restarts = min(n_restarts, 6)
        for i in range(n_restarts):
            if time.monotonic() > deadline - 12:
                break
            scale = 0.003 * (1 + i * 0.5)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(12.0, deadline - time.monotonic() - 10)
            if t_alloc < 3:
                break
            a_opt, obj_opt = subgradient_optimize(a_pert, time_limit=t_alloc,
                                                  lr_init=0.005, momentum=0.9,
                                                  top_k=20, temp_init=0.05, temp_min=0.01)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
