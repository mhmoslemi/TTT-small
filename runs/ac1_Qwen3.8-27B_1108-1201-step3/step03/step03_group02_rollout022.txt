import time
from scipy.optimize import linprog, minimize_scalar

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def sp_build_conv_matrix(a):
    """Build sparse convolution matrix."""
    from scipy.sparse import csr_matrix
    n = len(a)
    num_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(num_rows, n))

def solve_lp_max_sum(a, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b."""
    n = len(a)
    A = sp_build_conv_matrix(a)
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None

def fast_convolve_auto(a):
    """Compute full auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve_cross(a, b):
    """Compute full cross-convolution a*b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Bf = np.fft.rfft(b, fft_size)
    return np.fft.irfft(Af * Bf, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

`conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]`

`d conv[k] / d a[j]`: The variable `a[j]` appears in `conv[k]` in two ways:

So in all cases: `d conv[k] / d a[j] = 2 * a[k-j]` if `0 <= k-j < n`, else 0.

        a_new = normalize_seq(a_new)

        ca_new = fast_convolve_auto(a_new)
        val = ca_new.max()

        if val < best_val - 1e-13:
            best_val = val
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        ca = ca_new

        if no_improve > 400:
            break

    obj = 2.0 * n * best_val
    return best_a, obj


def lp_refine_adaptive(a0, time_limit, max_iters=10000):
    """LP refinement with adaptive M scaling and variant switching."""
    n = len(a0)
    a = normalize_seq(a0)
    ca = fast_convolve_auto(a)
    best_val = ca.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_scale = 1.0
    variant = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        M = ca.max() * m_scale
        b = None

        if variant == 0:
            b = solve_lp_max_sum(a, M)
        else:
            b, _ = solve_lp_min_peak(a)

        if b is None:
            m_scale *= 0.95
            continue

        if np.linalg.norm(b - a, ord=np.inf) < 1e-12:
            break

        t = exact_line_search(a, b)
        a_new = (1.0 - t) * a + t * b
        a_new = normalize_seq(a_new)

        ca_new = fast_convolve_auto(a_new)
        val = ca_new.max()

        if val < best_val - 1e-13:
            best_val = val
            best_a = a_new.copy()
            no_improve = 0
            m_scale = min(1.05, m_scale * 1.01)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                variant = 1 - variant
                m_scale = max(1.0, m_scale * 0.98)

        a = a_new
        ca = ca_new

        if no_improve > 500:
            break

    obj = 2.0 * n * best_val
    return best_a, obj


def smoothed_grad_refine(a0, time_limit, tau_init=50.0, lr_init=0.005, momentum=0.92):
    """Projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_val = fast_convolve_auto(a).max()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve_auto(a)
        scaled = tau * conv
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()

        grad = np.zeros(n)
        for j in range(n):
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo <= k_hi:
                a_idx = np.arange(k_lo - j, k_hi - j + 1)
                grad[j] = 2.0 * np.dot(p[k_lo:k_hi + 1], a[a_idx])
        grad /= tau

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve_auto(a)
        val = conv.max()

        if val < best_val - 1e-13:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
        if no_improve > 3000:
            break
        tau = min(tau * 1.02, 2000.0)

    obj = 2.0 * n * best_val
    return best_a, obj


def subgrad_refine(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=20,
                   temp=0.001, perturb_every=1000, perturb_scale=0.003):
    """Projected subgradient with multi-peak weighting."""
    n = len(a0)
    a = normalize_seq(a0)
    ca = fast_convolve_auto(a)
    best_a = a.copy()
    best_val = ca.max()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(ca):
            peaks = np.arange(len(ca))
            peak_vals = ca.copy()
        else:
            peaks = np.argpartition(ca, -top_k)[-top_k:]
            peak_vals = ca[peaks]

        sv = peak_vals / temp if temp > 1e-15 else peak_vals
        sv -= sv.max()
        w = np.exp(sv)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * a[k - idx]
        sg *= 2.0

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        ca = fast_convolve_auto(a)
        val = ca.max()

        if val < best_val - 1e-13:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            ca = fast_convolve_auto(a)
            v *= 0.5
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.2, lr * 0.5)

        if no_improve > 20000:
            break

    obj = 2.0 * n * best_val
    return best_a, obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0.0)
        else:
            xs = np.linspace(0, 1, len(parent))
            xp = np.linspace(0, 1, n)
            pp = np.interp(xp, xs, np.maximum(parent, 0.0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5, 7]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            inits[f'cos_{freq}'] = a / s

    for sigma_frac in [0.15, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma_frac * n)) ** 2)
        s = g.sum()
        if s > 1e-10:
            inits[f'gauss_{sigma_frac}'] = g / s

    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)

    for num_peaks in [2, 3, 5, 8, 13]:
        if num_peaks > n:
            continue
        peaks = rng.choice(n, size=num_peaks, replace=False)
        a = np.zeros(n)
        width = max(1, n // (num_peaks * 8))
        for p in peaks:
            a[max(0, p - width):min(n, p + width + 1)] = 1.0
        s = a.sum()
        if s > 1e-10:
            inits[f'peaks_{num_peaks}'] = a / s

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    a_edge = 0.2 + 0.8 * np.cos(np.pi * x / n) ** 2
    a_edge = np.maximum(a_edge, 0)
    inits['edge_high'] = a_edge / a_edge.sum()

    a_mid = 0.2 + 0.8 * np.sin(np.pi * x / n) ** 2
    a_mid = np.maximum(a_mid, 0)
    inits['mid_high'] = a_mid / a_mid.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: LP exploration across lengths and initializations.
    Phase 2: Deep adaptive LP refinement on best candidate.
    Phase 3: Smoothed gradient polishing.
    Phase 4: Subgradient refinement.
    Phase 5: Perturbation and re-optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a_clean = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a_clean.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_clean.copy()
            print(f"  ** New best: {score:.6f} (n={len(a_clean)}, t={time.monotonic()-deadline+1770:.0f}s)")

    # Phase 1: Exploration
    print("=== Phase 1: LP Exploration ===")
    t1_start = time.monotonic()

    lengths_to_try = [512]
    if parent is not None:
        pl = len(parent)
        if pl != 512 and 64 <= pl <= 2048:
            lengths_to_try.insert(0, pl)
    for extra in [256, 384, 768, 1024]:
        if extra not in lengths_to_try:
            lengths_to_try.append(extra)

    lengths_to_try = [l for l in lengths_to_try if 64 <= l <= 2048]

    p1_time = min(600.0, (deadline - time.monotonic()) * 0.35)
    n_total_trials = len(lengths_to_try) * 4
    time_per_trial = max(8.0, p1_time / n_total_trials)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 700:
            break
        print(f"  Trying n={n}")
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'uniform', 'cos_1', 'rand1', 'gauss_0.15',
                    'edge_high', 'mid_high', 'cos_2', 'rand2']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(time_per_trial, deadline - time.monotonic() - 700)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, _ = lp_refine(a0, time_limit=t_alloc, use_ls=True,
                                 max_iters=3000, m_scale=1.0)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # Phase 2: Deep LP refinement
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        p2_deadline = deadline - 500
        for m_scale in [1.0, 1.02, 1.01, 1.05]:
            if time.monotonic() > p2_deadline:
                break
            t_alloc = min(120.0, p2_deadline - time.monotonic() - 2)
            if t_alloc < 5:
                break
            a_opt, _ = lp_refine(best_sequence, time_limit=t_alloc, use_ls=True,
                                 max_iters=5000, m_scale=m_scale)
            try_update(a_opt)

        # Also try adaptive variant
        if time.monotonic() < p2_deadline:
            t_alloc = min(150.0, p2_deadline - time.monotonic() - 2)
            if t_alloc > 5:
                a_opt, _ = lp_refine_adaptive(best_sequence, time_limit=t_alloc,
                                              max_iters=5000)
                try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # Phase 3: Smoothed gradient polishing
    print("=== Phase 3: Smoothed Gradient ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        p3_deadline = deadline - 250
        for tau, lr in [(50.0, 0.005), (200.0, 0.003), (100.0, 0.008)]:
            if time.monotonic() > p3_deadline:
                break
            t_alloc = min(80.0, p3_deadline - time.monotonic() - 2)
            if t_alloc < 5:
                break
            a_opt, _ = smoothed_grad_refine(best_sequence, time_limit=t_alloc,
                                            tau_init=tau, lr_init=lr)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: Subgradient refinement
    print("=== Phase 4: Subgradient ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        p4_deadline = deadline - 100
        n_best = len(best_sequence)
        for lr, top_k in [(0.02, max(10, n_best // 30)), (0.01, max(15, n_best // 20))]:
            if time.monotonic() > p4_deadline:
                break
            t_alloc = min(70.0, p4_deadline - time.monotonic() - 2)
            if t_alloc < 5:
                break
            a_opt, _ = subgrad_refine(best_sequence, time_limit=t_alloc,
                                      lr_init=lr, top_k=top_k, temp=0.0008)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Perturbation and re-optimization
    print("=== Phase 5: Perturbations ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        p5_deadline = deadline - 10
        n_best = len(best_sequence)
        scales = [0.003, 0.008, 0.001, 0.015, 0.005]
        for scale in scales:
            if time.monotonic() > p5_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(30.0, p5_deadline - time.monotonic())
            if t_alloc < 3:
                break
            a_opt, _ = lp_refine(a_pert, time_limit=t_alloc, use_ls=True,
                                 max_iters=2000, m_scale=1.0)
            try_update(a_opt)

            # Also try subgradient from perturbation
            if time.monotonic() < p5_deadline:
                t_alloc2 = min(20.0, p5_deadline - time.monotonic())
                if t_alloc2 > 3:
                    a_opt2, _ = subgrad_refine(a_pert, time_limit=t_alloc2,
                                                lr_init=0.015, top_k=15, temp=0.001)
                    try_update(a_opt2)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Fallback used: score={best_score:.6f}")

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
