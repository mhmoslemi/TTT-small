def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.sum(cond)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set guided optimization,
LP-based direction finding, adaptive smooth max, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -200, 200)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a."""
    n = len(a)
    N = 1 << (n + len(w) - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(np.conj(W) * A, N))[:n]
    return grad


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.sum(cond)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max(conv, beta)
    return 2.0 * n * sm


def softmax_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max(conv, beta)
    grad_a = compute_grad(a, w)
    # Chain rule through softmax: d a / d x = a * (I - a^T)  (for normalized softmax)
    # Actually a = exp(x) / sum(exp(x)), so da_i/dx_j = a_i(δ_ij - a_j)
    # grad_x_j = sum_i (da_i/dx_j) * grad_a_i = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    lagrange = np.dot(a, grad_a)
    jac = a * (grad_a - lagrange)
    return jac


def optimize_lbfgsb(a0, beta, maxiter=500, maxls=50):
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    result = minimize(softmax_obj, x0, jac=softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': maxls,
                                                   'ftol': 1e-12, 'gtol': 1e-8})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def optimize_pg(a0, beta, lr=0.05, n_iter=500):
    """Projected gradient on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if sm - conv.max() < 1e-12:
            # Smooth max converged to true max
            break
        
        grad_a = compute_grad(a, w)
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break
        
        t = lr
        for _ in range(15):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:
                break
            t *= 0.5
        
        a = a_new
        lr *= 0.999
    
    return best_a, best_obj


def lp_direction(f, M):
    """Find g maximizing sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        C[k, i_min:i_max + 1] = f[i_min:i_max + 1]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'maxiter': 10000})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_blend_step(a, blend_factor=0.3):
    """One LP-based direction step."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    # Add small margin
    M = M * 1.001
    
    g = lp_direction(a, M)
    if g is None or g.sum() < 1e-15:
        return a
    
    g = g / g.sum()
    a_new = (1 - blend_factor) * np.maximum(a, 0) + blend_factor * g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
    for k in range(1, fft_size // 2):
        A[k] = np.exp(1j * phases[k - 1])
        A[fft_size - k] = np.exp(-1j * phases[k - 1])
    if fft_size % 2 == 0:
        A[fft_size // 2] = np.exp(1j * phases[-1])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    tri = np.minimum(x, n - 1 - x).astype(float) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    for sf in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g
    
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['flat_spec'] = flat_spectrum_init(n, rng)
    
    # Anti-palindromic: large values where mirror is small
    anti = np.zeros(n)
    for i in range(n):
        anti[i] = 1.0 / (1.0 + anti_mirror(anti, i, n) if i > n // 2 else 0.0)
    # Simpler: just use a shifted version
    shifted = np.zeros(n)
    half = n // 2
    if half > 0:
        shifted[:half] = 1.0
        shifted /= shifted.sum()
    else:
        shifted[0] = 1.0
    inits['half'] = shifted
    
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    return inits


def anti_mirror(a, i, n):
    return a[n - 1 - i] if 0 <= n - 1 - i < n else 0.0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using scaled smooth-max L-BFGS-B,

Key improvement: proper scaling of convolution values for well-conditioned smooth max.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_scaled(x, beta):
    """Smooth max with proper scaling. x is assumed O(1) scale."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -300, 300)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a (a normalized, sum=1)."""
    n = len(a)
    N = 1 << (n + len(w) - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(np.conj(W) * A, N))[:n]
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.sum(cond)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def scaled_objective(x, n, beta):
    """Objective: 2 * smooth_max(conv(a,a) * n) with sum(a)=1."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    scaled = conv * n
    sm, _ = smooth_max_scaled(scaled, beta)
    return 2.0 * sm


def scaled_jacobian(x, n, beta):
    """Jacobian of the scaled objective."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    scaled = conv * n
    sm, w = smooth_max_scaled(scaled, beta)
    grad_a = compute_grad(a, w) * n
    lagrange = np.dot(a, grad_a)
    jac = a * (grad_a - lagrange)
    return jac
