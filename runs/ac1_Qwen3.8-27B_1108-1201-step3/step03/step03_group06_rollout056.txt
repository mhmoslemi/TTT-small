Where M = max(a*a) is the current peak.

The idea is: b is a "direction" that we can move toward while keeping the convolution peak bounded. If we set a_new = (1-t)*a + t*b, then:

For small t, the dominant change is 2t(1-t)(a*b)[k] - (1-t)^2 M + t^2 (b*b)[k] ≈ M - 2t(M - (a*b)[k]) + ...

Since (a*b)[k] <= M for all k (by the LP constraint), we have M - (a*b)[k] >= 0, so the peak decreases (or stays the same) for small t.

The LP maximizes sum(b) to ensure b is "different enough" from a to make progress. If b were equal to a, we'd have (a*b) = (a*a) and no progress would be made.

"""
Hybrid LP + gradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LP-based Frank-Wolfe with warm starts, smooth gradient polishing,
and multi-phase search over diverse initializations and sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_sparse_lp_matrix(a, k_indices):
    """Build sparse constraint matrix rows for given k indices."""
    n = len(a)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    nnz = len(rows)
    if nnz == 0:
        return csr_matrix((len(k_indices), n))
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))

def solve_lp_direction(a, M, k_indices):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in k_indices."""
    n = len(a)
    A_ub = build_sparse_lp_matrix(a, k_indices)
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lp_optimize(a0, time_limit, step_init=1.0, top_k=None, relax=1.0,
                seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Adaptive constraint selection
        if top_k is not None:
            # Use top-k peaks plus some random ones
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            # Add some random constraints for diversity
            n_extra = max(5, top_k // 4)
            random_k = rng.integers(0, len(conv), size=n_extra)
            k_indices = np.unique(np.concatenate([peaks, random_k]))
        else:
            k_indices = np.arange(len(conv))
        
        b = solve_lp_direction(a, M * relax, k_indices)
        if b is None:
            # Fallback: use all constraints
            b = solve_lp_direction(a, M * relax, np.arange(len(conv)))
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Backtracking line search
        best_step = 0
        best_obj_trial = 2.0 * n * M
        for try_t in [step, step*0.8, step*0.5, step*0.25, step*0.1]:
            a_new = (1.0 - try_t) * a + try_t * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj_trial:
                best_obj_trial = obj_new
                best_step = try_t
        
        if best_step > 0:
            a = (1.0 - best_step) * a + best_step * b
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            step *= 0.7
        
        if no_improve > 30:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            step = step_init * 0.5
            no_improve = 0
        
        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, step={best_step:.3f}, t={elapsed:.1f}s")
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

def smooth_gradient(a, beta=500.0):
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = sum_k w[k] * d(a*a)/da[i] at lag k
    # d(a*a)[k]/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n
    # For the linear convolution: (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # Note: when k-i = i (i.e., k = 2i), both terms give a[i], so the contribution is 2*a[i]
    
    # Use FFT-based approach
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    
    # First term: sum_k w[k] * a[k-i] for valid k
    # = cross-correlation = ifft(A * conj(W))
    term1 = np.fft.ifft(A * np.conj(W)).real[:n]
    
    # Second term: sum_k w[k] * a[i-k] for valid k
    # = convolution = ifft(A * W)
    term2 = np.fft.ifft(A * W).real[:n]
    
    grad = term1 + term2
    return grad, conv

def smooth_gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.05,
                             momentum=0.9, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
                v *= 0.5
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = beta_init * 0.7
                no_improve = 0
        
        beta = min(5000.0, beta * 1.002)
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

Zooming Frank-Wolfe + adaptive LP + dual-averaging gradient for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Exploits symmetry, diverse initializations,
and multi-phase search with adaptive constraint management.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_sparse_lp_matrix(a, k_indices):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    nnz = len(rows)
    if nnz == 0:
        return csr_matrix((len(k_indices), n))
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))


def solve_lp_direction(a, M, k_indices):
    n = len(a)
    A_ub = build_sparse_lp_matrix(a, k_indices)
    b_ub = np.full(len(k_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def zooming_fw_optimize(a0, time_limit, step_init=1.0, top_k=None,

    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

            n_extra = max(3, top_k // 5)
            random_k = rng.integers(0, len(conv), size=n_extra)
            k_indices = np.unique(np.concatenate([peaks, random_k]))
        else:
            k_indices = np.arange(len(conv))
        b = solve_lp_direction(a, M * relax, k_indices)
        if b is None:
            b = solve_lp_direction(a, M * relax, np.arange(len(conv)))
            if b is None:
                break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        # Zooming: try several steps including t=1

        if not candidates:

    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def smooth_gradient(a, beta=500.0):
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    term1 = np.fft.ifft(A * np.conj(W)).real[:n]
    term2 = np.fft.ifft(A * W).real[:n]
    grad = term1 + term2
    return grad, conv
