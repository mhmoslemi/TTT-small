    n = len(a)
    conv = fast_convolve(a)
    
    # Scale for numerical stability
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    weights *= beta  # d/dx of (1/beta)*log(sum exp(beta*x)) has factor beta
    
    # Objective value
    obj = max_conv + np.log(exp_shifted.sum()) / beta
    
    # Gradient

    # = sum_k weights[k] * a[k-j] + a[j] * sum_k weights[k]

    size = len(a) + len(weights) - 1

    W = np.fft.rfft(weights, fft_size)

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_smooth_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (obj_value, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Numerical stability: shift by max
    max_conv = conv.max()
    scaled = beta * (conv - max_conv)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    weights = exp_scaled / sum_exp
    obj = max_conv + np.log(sum_exp) / beta

def simplex_project(x):
    """Project x onto the probability simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    # Algorithm from Duchi et al. 2008
    if x.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

    weights = exp_scaled / sum_exp  # softmax(beta * conv)
    
    obj = max_conv + np.log(sum_exp) / beta
    
    # Gradient: grad[j] = sum_k weights[k] * (a[k-j] + a[j]) * 1(0<=k-j<n)
    # Part 1: sum_k weights[k] * a[k-j] * 1(0<=k-j<n)
    #         = conv(weights, a_rev)[j + n - 1]
    a_rev = a[::-1]
    size12 = len(weights) + n - 1
    fft_size = 1 << (size12 - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_w_ar = np.fft.irfft(W * AR, fft_size)[:size12]
    part1 = conv_w_ar[n - 1:n - 1 + n]
    
    # Part 2: a[j] * S[j] where S[j] = sum_{k=j}^{j+n-1} weights[k]
    w_pad = np.zeros(3 * n)
    w_pad[:2 * n - 1] = weights
    cumsum_w = np.concatenate([[0.0], np.cumsum(w_pad)])
    S = cumsum_w[n:n + n] - cumsum_w[0:n]
    
    grad = part1 + a * S
    return obj, grad


def simplex_project(x):
    """
    Exact Euclidean projection onto the probability simplex {y >= 0, sum(y) = 1}.
    Uses the O(n log n) algorithm from Duchi et al. (2008).
    """
    n = len(x)
    if x.sum() <= 1e-30:
        return np.ones(n) / n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def true_objective(a):
    """Compute the actual objective 2*n*max(conv) for sum=1 sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def momentum_smooth_optimize(a0, time_limit, beta_start=20.0, beta_end=100000.0,
                              lr_init=1.0, momentum_init=0.85, 
                              verbose=True, verbose_every=200):
    """
    Momentum projected gradient with beta-annealed smooth max.
    a0 should be non-negative; will be normalized to sum=1.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_init
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.85))
        # Exponential beta annealing
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        
        # Compute smooth gradient
        _, grad = compute_smooth_gradient(a, beta)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a = simplex_project(a_new)
        
        # Evaluate true objective periodically
        if it % 10 == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive adjustments
        if no_improve > 200:
            # Reduce learning rate, increase momentum
            lr *= 0.7
            momentum = min(0.995, momentum + 0.01)
            no_improve = 0
            stall_count += 1
            # Occasionally add perturbation
            if stall_count > 5:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.002
                a = simplex_project(a + noise)
                stall_count = 0
                no_improve = 0
        
        # Slow decay of lr
        if it % 500 == 499:
            lr *= 0.95
        
        it += 1
        if verbose and it % verbose_every == 0:
            print(f"      it={it}, best={best_obj:.6f}, beta={beta:.0f}, lr={lr:.4f}, mom={momentum:.3f}, t={elapsed:.1f}s")
    
    # Final eval
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    return best_a, best_obj


def peak_shave(a, time_limit, num_shaves=200):
    """
    Iteratively shave the highest peak by redistributing mass from contributors.
    Works on sum=1, non-negative sequence.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    for it in range(num_shaves):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Find contributors to this peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        indices = np.arange(j_start, j_end + 1)
        contributions = a[indices] * a[peak_k - indices]
        
        if contributions.max() < 1e-15:
            break
        
        # Reduce the largest contributors, increase non-contributors
        # Move mass from indices with high contribution to indices not in the peak's support
        outside = np.setdiff1d(np.arange(n), indices)
        if len(outside) == 0:
            break
        
        # Find the index with highest contribution
        max_idx_local = np.argmax(contributions)
        max_idx = indices[max_idx_local]
        partner_idx = peak_k - max_idx
        
        # Amount to transfer (small fraction)
        transfer = 0.01 * a[max_idx]
        if transfer < 1e-15:
            break
        
        # Distribute to outside indices proportional to their current values
        outside_vals = a[outside]
        if outside_vals.sum() < 1e-15:
            # Spread uniformly
            a[outside] += transfer / len(outside)
        else:
            a[outside] += transfer * outside_vals / outside_vals.sum()
        
        a[max_idx] -= transfer
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Edge-concentrated (mass at both ends)
    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Strong edge (mass at very ends)
    edge2 = max(1, n // 8)
    ec2 = np.zeros(n)
    ec2[:edge2] = 1.0
    ec2[-edge2:] = 1.0
    ec2 /= ec2.sum()
    inits['edge_strong'] = ec2.copy()
    
    # Linear ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    
    # V-shape (high at edges, low in middle)
    x = np.arange(n, dtype=np.float64)
    v = np.minimum(x, n - 1 - x)
    v = 1.0 / (v + 1.0)
    v /= v.sum()
    inits['v_shape'] = v.copy()
    
    # Two-block: first half and last half
    tb = np.zeros(n)
    half = n // 2
    tb[:half] = 1.0
    tb[half:] = 0.5  # less mass in second block
    tb = np.zeros(n)
    tb[:edge] = 1.0
    tb[-edge:] = 1.0
    tb /= tb.sum()
    inits['two_block'] = tb.copy()
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random sparse
    sp = rng.random(n)
    sp[sp < 0.6] = 0.0
    if sp.sum() > 1e-10:
        sp /= sp.sum()
    else:
        sp = rng.dirichlet(np.ones(n))
    inits['sparse'] = sp.copy()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp.copy()
            else:
                interp = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(pp)), pp)
                interp = np.maximum(interp, 0)
                if interp.sum() > 1e-10:
                    inits['parent_interp'] = interp / interp.sum()
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: smooth gradient optimization with beta annealing,
    momentum, exact simplex projection, and peak shaving refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    # === Phase 1: Quick scan of lengths and initializations ===
    print("=== Phase 1: Quick Scan ===")
    lengths = [384, 512, 256, 768]
    # If parent exists, prioritize its length
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)
        else:
            lengths.remove(pn)
            lengths.insert(0, pn)
    
    phase1_results = []  # (score, sequence, n)
    
    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_initializations(n, parent, rng)
        # Priority order
        priority = ['parent', 'parent_interp', 'edge', 'v_shape', 'edge_strong',
                    'two_block', 'uniform', 'ramp', 'random', 'sparse']
        names = [k for k in priority if k in inits]
        
        time_per_init = max(10.0, 120.0 / len(names))
        
        for name in names[:5]:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 400)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.0f}s")
            a_opt, obj_opt = momentum_smooth_optimize(
                a0, time_limit=t_alloc, beta_start=50.0, beta_end=50000.0,
                lr_init=1.0, momentum_init=0.85, verbose=False)
            score = 2.0 * n * fast_convolve(a_opt).max()
            print(f"    local_obj={score:.6f}")
            phase1_results.append((score, a_opt.copy(), n))
    
    # Sort by score
    phase1_results.sort(key=lambda x: x[0])
    if phase1_results:
        best_score = phase1_results[0][0]
        best_sequence = phase1_results[0][1].copy()
        print(f"Phase 1 best: {best_score:.6f} (n={phase1_results[0][2]})")
    
    # === Phase 2: Deep optimization on top candidates ===
    print("\n=== Phase 2: Deep Optimization ===")
    top_candidates = phase1_results[:3] if len(phase1_results) >= 3 else phase1_results
    
    for idx, (score, seq, n) in enumerate(top_candidates):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(600.0, deadline - time.monotonic() - 100)
        if t_alloc < 30:
            break
        
        print(f"  Candidate {idx}: n={n}, current={score:.6f}, t={t_alloc:.0f}s")
        a_opt, obj_opt = momentum_smooth_optimize(
            seq, time_limit=t_alloc, beta_start=50.0, beta_end=200000.0,
            lr_init=0.8, momentum_init=0.90, verbose=True, verbose_every=500)
        score = 2.0 * n * fast_convolve(a_opt).max()
        print(f"    Deep opt: {score:.6f}")
        
        # Evaluate differently with actual evaluate_sequence
        eval_score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(eval_score):
            print(f"    eval_score={eval_score:.6f}")
            if eval_score < best_score:
                best_score = eval_score
                best_sequence = a_opt.copy()
        elif score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
    
    # === Phase 3: Peak shaving ===
    print("\n=== Phase 3: Peak Shaving ===")
    if best_sequence is not None and time.monotonic() < deadline - 60:
        n_bs = len(best_sequence)
        t_shave = min(120.0, deadline - time.monotonic() - 40)
        if t_shave > 10:
            print(f"  n={n_bs}, t={t_shave:.0f}s")
            a_shaved, obj_shaved = peak_shave(best_sequence, time_limit=t_shave, num_shaves=5000)
            score = 2.0 * n_bs * fast_convolve(a_shaved).max()
            print(f"    Peak shaved: {score:.6f}")
            eval_score = float(evaluate_sequence(a_shaved.tolist()))
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_sequence = a_shaved.copy()
            elif score < best_score:
                best_score = score
                best_sequence = a_shaved.copy()
    
    # === Phase 4: Second round of deep optimization ===
    print("\n=== Phase 4: Final Deep Optimization ===")
    if best_sequence is not None and time.monotonic() < deadline - 30:
        t_final = min(300.0, deadline - time.monotonic() - 20)
        if t_final > 20:
            n_f = len(best_sequence)
            print(f"  n={n_f}, t={t_final:.0f}s")
            a_opt, obj_opt = momentum_smooth_optimize(
                best_sequence, time_limit=t_final, beta_start=200.0, beta_end=500000.0,
                lr_init=0.3, momentum_init=0.95, verbose=True, verbose_every=500)
            score = 2.0 * n_f * fast_convolve(a_opt).max()
            print(f"    Final: {score:.6f}")
            eval_score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_sequence = a_opt.copy()
            elif score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
    
    # === Phase 5: Final peak shave ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_final_shave = min(60.0, deadline - time.monotonic() - 10)
        if t_final_shave > 5:
            a_shaved, obj_shaved = peak_shave(best_sequence, time_limit=t_final_shave, num_shaves=10000)
            score = 2.0 * len(best_sequence) * fast_convolve(a_shaved).max()
            eval_score = float(evaluate_sequence(a_shaved.tolist()))
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_sequence = a_shaved.copy()
            elif score < best_score:
                best_score = score
                best_sequence = a_shaved.copy()
            print(f"  Final shave: score={best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
