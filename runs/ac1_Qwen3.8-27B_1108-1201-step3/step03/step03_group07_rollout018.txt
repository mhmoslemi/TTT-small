```python
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    n = len(f)
    # Variables: g[0], ..., g[n-1] >= 0
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```

```python
    """Build the constraint matrix A for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A

def solve_lp_direction(f, M):

    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    return f.copy()  # Fallback
```

```python
def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    sm = x_max + np.log(total) / beta
    return sm, w

def compute_grad_smooth_conv(a, w):
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

Given f, M = max(f*f), the LP finds g that maximizes sum(g) subject to (f*g)[k] <= M for all k.

Intuitively, g will put more mass where f has less mass (to keep f*g small) and less mass where f has more mass. So g is "anti-correlated" with f in some sense.

When we mix f and g: f_new = (1-t)f + tg, the new peak (f_new * f_new)[k] should be smaller because:
(f_new * f_new)[k] = (1-t)^2 (f*f)[k] + 2t(1-t)(f*g)[k] + t^2 (g*g)[k]
<= (1-t)^2 M + 2t(1-t) M + t^2 (g*g)[k]
= M((1-t)^2 + 2t(1-t)) + t^2 (g*g)[k]
= M(1-t(2-t)) + t^2 max(g*g)

So the peak of f_new*f_new is bounded by M(1-t(2-t)) + t^2 * max(g*g).

For this to be less than M, we need:
-t(2-t) + t^2 * max(g*g)/M < 0
t(2-t) > t^2 * max(g*g)/M
(2-t) > t * max(g*g)/M

```python
def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
