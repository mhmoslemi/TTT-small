def fast_convolve(a):

    n = len(a)

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Returns (b, t_opt) or (None, None)."""
    n = len(a)
    num_rows = 2 * n - 1
    A_ub = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    # conv(a,b)[k] - t <= 0  =>  A_ub[k,:] @ b - t <= 0
    A_full = np.hstack([A_ub, -np.ones((num_rows, 1))])
    b_ub = np.zeros(num_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0.0)
        s = b.sum()
        if s < 1e-15:
            return None, None
        b = b / s
        return b, res.x[n]
    return None, None


def lp_max_sum_direction(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b>=0.
    Returns normalized b or None."""
    n = len(a)
    num_rows = 2 * n - 1
    A_ub = build_conv_matrix(a)

    c = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s < 1e-15:
            return None
        return b / s
    return None


def line_search(a, b, alpha_min=0.01, alpha_max=1.0, n_trials=8):
    """Find best alpha in [alpha_min, alpha_max] minimizing obj((1-a)*a + a*b)."""
    best_alpha = 0.0
    best_obj = obj_normalized(a)
    for i in range(n_trials):
        alpha = alpha_min + (alpha_max - alpha_min) * i / (n_trials - 1) if n_trials > 1 else 0.5
        candidate = (1.0 - alpha) * a + alpha * b
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        o = obj_normalized(candidate)
        if o < best_obj:
            best_obj = o
            best_alpha = alpha
    return best_alpha, best_obj


def subgradient_step(a, conv, top_k, temperature):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        idx = np.arange(j_lo, j_hi + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def softmax_gradient(a, t):
    """Compute gradient of soft-max smoothed objective."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp

    grad = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad[j] = (2.0 * n) * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    return grad, conv.max()


def optimize_lp_refine(a0, time_limit, max_iters=1000, step_init=0.4):
    """LP-based iterative refinement with line search."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(777)
    use_max_sum = False

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()

        if not use_max_sum:
            b, t_opt = lp_min_peak_direction(a)
            if b is None:
                b = lp_max_sum_direction(a, M * 1.05)
                if b is None:
                    break
        else:
            b = lp_max_sum_direction(a, M * 1.01)
            if b is None:
                b, _ = lp_min_peak_direction(a)
                if b is None:
                    break

        # Line search
        best_alpha, obj_at_alpha = line_search(a, b, 0.05, 1.0, 10)

        if best_alpha > 0:
            a_new = (1.0 - best_alpha) * a + best_alpha * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
        else:
            # Try smaller steps
            a_new = (1.0 - 0.05) * a + 0.05 * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try to recover: stay at current a
            a_new = a
            conv_new = conv

        a = a_new
        conv = conv_new

        if no_improve == 30:
            use_max_sum = not use_max_sum
            no_improve = 0
        elif no_improve == 60:
            # Perturb
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            use_max_sum = not use_max_sum
        elif no_improve > 150:
            break

    return best_a, best_obj


def optimize_subgrad(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=40,
                     temp_init=0.001, perturb_every=800):
    """Projected subgradient with momentum."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_step(a, conv, top_k, temp)
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            perturb_every = int(perturb_every * 1.2)
            lr = max(lr_init * 0.2, lr * 0.6)
            temp = max(0.0001, temp * 0.8)

    return best_a, best_obj


def optimize_softmax_grad(a0, time_limit, lr_init=0.01, momentum=0.93, t_init=20.0):
    """Projected gradient on soft-max smoothed objective."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    t_cur = t_init
    no_improve = 0

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        _, grad = softmax_gradient(a, t_cur)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.5
            no_improve = 0
        if no_improve > 2000:
            break
        if it % 500 == 0 and t_cur < 200:
            t_cur *= 1.5

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos_{freq}'] = a / a_sum

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g_sum = g.sum()
    if g_sum > 1e-10:
        inits['gauss_center'] = g / g_sum

    # Gaussian at edge
    g2 = np.exp(-0.5 * (x / (0.15 * n)) ** 2)
    g2_sum = g2.sum()
    if g2_sum > 1e-10:
        inits['gauss_edge'] = g2 / g2_sum

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Half-and-half
    hh = np.zeros(n)
    hh[:n//2] = 1.0
    hh_sum = hh.sum()
    if hh_sum > 0:
        inits['half'] = hh / hh_sum

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Triangle
    tri = np.minimum(np.arange(n), n - np.arange(n)) + 1
    inits['triangle'] = tri / tri.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining LP refinement,
    subgradient, and soft-max gradient methods to minimize the objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)}) at t={time.monotonic():.0f}")
        return score

    # Phase 1: Quick exploration across lengths
    print("=== Phase 1: Exploration ===")
    t1 = time.monotonic()

    explore_lengths = []
    if parent is not None:
        explore_lengths.append(len(parent))
    for L in [256, 384, 512, 640, 768, 1024]:
        if L not in explore_lengths and 64 <= L <= 2048:
            explore_lengths.append(L)

    # Sort by priority: parent length first, then common lengths
    explore_lengths = list(dict.fromkeys(explore_lengths))
    explore_lengths = sorted(explore_lengths, key=lambda x: -x if x == (len(parent) if parent is not None else 0) else x)

    candidates = {}  # n -> (a, obj)

    for n in explore_lengths[:5]:
        remaining = deadline - time.monotonic() - 600
        if remaining < 30:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'uniform', 'random', 'cos_1', 'gauss_center', 'gauss_edge']
        ordered = [nm for nm in priority if nm in inits] + \
                  [nm for nm in init_names if nm not in priority]
        ordered = ordered[:4]

        time_per = max(10.0, remaining / (len(explore_lengths[:5]) * len(ordered) * 2))

        for name in ordered:
            remaining = deadline - time.monotonic() - 500
            if remaining < 10:
                break
            t_alloc = min(time_per, remaining)
            if t_alloc < 5:
                break

            a0 = inits[name].astype(np.float64)

            # Quick LM to get a reasonable starting point
            a_opt, obj = optimize_lp_refine(a0, time_limit=t_alloc, max_iters=100)
            score = try_update(a_opt)
            if n not in candidates or obj < candidates[n][1]:
                candidates[n] = (a_opt.copy(), obj)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1:.0f}s")

    # Phase 2: Deep LP refinement on best candidates
    print("=== Phase 2: LP Refinement ===")
    t2 = time.monotonic()

    # Sort candidates by objective
    sorted_cands = sorted(candidates.items(), key=lambda x: x[1][1])

    for n, (a_cand, obj_cand) in sorted_cands[:3]:
        remaining = deadline - time.monotonic() - 300
        if remaining < 60:
            break

        # Also add parent if not already included
        if parent is not None and n == len(parent):
            pass

        # Multiple LP refinement rounds with different step sizes
        time_per_round = remaining / 3
        for step_init in [0.4, 0.2, 0.6]:
            t_alloc = min(time_per_round, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break
            a_opt, obj = optimize_lp_refine(a_cand, time_limit=t_alloc,
                                            max_iters=2000, step_init=step_init)
            try_update(a_opt)
            if obj < a_cand_obj:
                pass
            a_cand = a_opt.copy()
            a_cand_obj = obj
            a_cand_obj = obj_cand if obj_cand < obj else obj

    # Also refine from parent directly
    if parent is not None and len(parent) <= 1024:
        remaining = deadline - time.monotonic() - 200
        if remaining > 60:
            p_norm = parent / parent.sum()
            a_opt, obj = optimize_lp_refine(p_norm, time_limit=min(remaining, 200),
                                            max_iters=2000, step_init=0.35)
            try_update(a_opt)
            if obj < (best_score if best_sequence is not None else float('inf')):
                candidates[len(parent)] = (a_opt.copy(), obj)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2:.0f}s")

    # Phase 3: Subgradient polish
    print("=== Phase 3: Subgradient ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            n_best = len(best_sequence)
            for lr in [0.015, 0.008, 0.025]:
                t_alloc = min(50.0, remaining - 10)
                if t_alloc < 10:
                    break
                a_opt, _ = optimize_subgrad(best_sequence, time_limit=t_alloc,
                                            lr_init=lr, momentum=0.92,
                                            top_k=max(20, n_best // 15),
                                            temp_init=0.0008)
                try_update(a_opt)
                remaining = deadline - time.monotonic() - 100
                if remaining < 20:
                    break

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3:.0f}s")

    # Phase 4: Soft-max gradient polish
    print("=== Phase 4: Soft-max Gradient ===")
    t4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 30:
            for lr in [0.004, 0.008]:
                t_alloc = min(40.0, remaining - 10)
                if t_alloc < 10:
                    break
                a_opt, _ = optimize_softmax_grad(best_sequence, time_limit=t_alloc,
                                                 lr_init=lr, momentum=0.94, t_init=40.0)
                try_update(a_opt)
                remaining = deadline - time.monotonic() - 50
                if remaining < 20:
                    break

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4:.0f}s")

    # Phase 5: Perturbation + re-optimization
    print("=== Phase 5: Perturbations ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        n_best = len(best_sequence)
        for scale in [0.001, 0.003, 0.005, 0.002, 0.008]:
            t_alloc = min(30.0, remaining - 5)
            if t_alloc < 5:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            a_opt, _ = optimize_lp_refine(a_pert, time_limit=t_alloc,
                                          max_iters=500, step_init=0.3)
            try_update(a_opt)
            remaining = deadline - time.monotonic() - 10

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5:.0f}s")

    # Final: one more LP refinement on the absolute best
    if best_sequence is not None and (deadline - time.monotonic()) > 20:
        t_final = min(60.0, deadline - time.monotonic() - 5)
        a_opt, _ = optimize_lp_refine(best_sequence, time_limit=t_final,
                                      max_iters=1000, step_init=0.2)
        try_update(a_opt)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
