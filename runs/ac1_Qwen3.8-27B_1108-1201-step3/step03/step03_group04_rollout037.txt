```python
import numpy as np
import time

def fast_conv(a):
    """Compute linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgradient_of_max_conv(a, k_star):
    """Compute subgradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        kj = k_star - j
        if 0 <= kj < n:
            g[j] += a[kj]
            if kj != j:
                g[j] += a[j]
    return g
```

```python
def project_simplex(v):
    """Project v onto {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * (np.arange(1, n+1)) > (cumsum - 1))[0][-1]
    theta = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def optimize_subgrad(a0, time_limit, n_peaks=10, lr_init=0.002, momentum=0.9):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        top_k = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Softmax weighting
        temp = max(peak_vals.max() * 0.03, 1e-15)
        w = np.exp(peak_vals / temp - peak_vals.max() / temp)
        w /= w.sum()
        
        # Compute multi-peak subgradient
        g = multi_peak_subgrad(a, peaks, w)
        
        # Project onto tangent space (sum = 0)
        g -= g.mean()
        
        # Nesterov momentum
        v = momentum * v - lr * g
        a_new = a + v
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv = fast_conv(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9999  # slow decay
        
        # Periodic restart of momentum
        if int((time.monotonic() - t0)) % 30 == 0:
            v *= 0.5
    
    return best_a, best_obj
```

```python
"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based Frank-Wolfe, multi-peak subgradient with Nesterov momentum,
coordinate descent, and multi-length scanning. Uses parent sequence as warm start.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (css - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multi_peak_subgrad(a, peaks, weights):
    """Compute weighted sum of subgradients of conv(a,a)[k] for k in peaks."""
    n = len(a)
    g = np.zeros(n)
    j = np.arange(n)
    for k, w in zip(peaks, weights):
        if w < 1e-12:
            continue
        kj = k - j
        valid = (kj >= 0) & (kj < n)
        g[valid] += w * a[kj[valid]]
        valid2 = valid & (kj != j)
        g[valid2] += w * a[valid2]
    return g


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Sparse matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        rows.extend(np.full(len(js), k))
        cols.extend(js.tolist())
        vals.extend(a[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_oracle_reduced(a, M, max_constraints=100, time_limit=2.0):
    """LP with only top constraints (faster)."""
    n = len(a)
    # Use full matrix but only top max_constraints rows
    A_full = build_conv_matrix(a)
    b_ub_full = np.full(2 * n - 1, M)
    # Just use all constraints for simplicity (n=384 case is fast)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_ub_full, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def subgradient_optimize(a0, time_limit, n_peaks=12, lr_init=0.003, momentum=0.92,
                         verbose=False):
    """Multi-peak subgradient with Nesterov momentum and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        top_k = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

        # Softmax weighting with adaptive temperature
        peak_range = peak_vals.max() - peak_vals.min()
        temp = max(peak_range * 0.3, peak_vals.max() * 0.02, 1e-15)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        g = multi_peak_subgrad(a, peaks, w)
        g -= g.mean()  # project to tangent space

        # Nesterov momentum
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        conv_new = fast_conv(a_new)
        cur_obj = 2.0 * n * conv_new.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.99995

        # Handle stalls
        if no_improve >= max_stall:
            rng = np.random.default_rng(it % (2**31))
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            conv = fast_conv(a)
            best_obj = min(best_obj, 2.0 * n * conv.max())
            if 2.0 * n * conv.max() < best_obj:
                best_obj = 2.0 * n * conv.max()
                best_a = a.copy()
            v *= 0.1
            no_improve = 0
            max_stall = int(max_stall * 1.5)

        # Decay momentum periodically
        if it % 500 == 0 and it > 0:
            v *= 0.5

        it += 1
        if verbose and it % 500 == 0:
            print(f"      SG it={it}, obj={best_obj:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def fw_lp_optimize(a0, time_limit, verbose=False, verbose_every=200):
    """Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M, time_limit=2.0)
        if b is None:
            # Fallback: use subgradient step
            peaks = np.array([np.argmax(conv)])
            g = multi_peak_subgrad(a, peaks, np.array([1.0]))
            g -= g.mean()
            b = project_simplex(a - 0.01 * g)
            sb = b.sum()
            if sb < 1e-15:
                b = a.copy()
            else:
                b = b / sb
        else:
            sb = b.sum()
            if sb < 1e-15:
                break
            b = b / sb

        # Line search
        best_t = 0.0
        best_line_obj = best_obj
        ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for t in ts:
            candidate = (1 - t) * a + t * b
            conv_c = fast_conv(candidate)
            obj_c = 2.0 * n * conv_c.max()
            if obj_c < best_line_obj:
                best_line_obj = obj_c
                best_t = t

        if best_t < 0.05:
            no_improve += 1
        else:
            a = (1 - best_t) * a + best_t * b
            conv = fast_conv(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Stall handling
        if no_improve >= max_stall:
            rng = np.random.default_rng(hash(it) % (2**31))
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_conv(a)
            M = conv.max()
            cur = 2.0 * n * M
            if cur < best_obj:
                best_obj = cur
                best_a = a.copy()
            no_improve = 0
            max_stall = int(max_stall * 1.4)

        it += 1
        if verbose and it % verbose_every == 0:
            print(f"      FW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_trials_per_coord=4):
    """Try moving mass between coordinates to reduce peak."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        # Try random pairs of coordinates
        for _ in range(min(200, max(1, int((time_limit - (time.monotonic() - t0)) * 50))):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-10:
                continue
            # Try moving delta from i to j
            deltas = np.linspace(0.01, min(a[i] * 0.5, 0.05), n_trials_per_coord)
            for delta in deltas:
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                if a_trial[i] < 0:
                    continue
                conv_t = fast_conv(a_trial)
                obj_t = 2.0 * n * conv_t.max()
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break

        it += 1
        if it % 5 == 0:
            conv = fast_conv(a)
            cur = 2.0 * n * conv.max()
            if cur < best_obj:
                best_obj = cur
                best_a = a.copy()

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    # Edge-concentrated
    edge = max(2, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    # Gaussian
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()

    # Two-sided decay
    td = np.exp(-np.abs(x - n / 2) / (n / 4))
    td /= td.sum()
    inits['twodecay'] = td.copy()

    return inits
```
