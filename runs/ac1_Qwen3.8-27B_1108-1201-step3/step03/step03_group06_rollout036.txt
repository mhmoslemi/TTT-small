Let me think about the line search more carefully. Given a (current) and b (LP direction), both on the simplex:
- a_t = (1-t)*a + t*b for t in [0,1]
- f(t) = max_k conv(a_t, a_t)[k]

conv(a_t, a_t)[k] = sum_i ((1-t)*a[i] + t*b[i]) * ((1-t)*a[k-i] + t*b[k-i])
= (1-t)^2 * conv_a[k] + t(1-t) * (conv_ab[k] + conv_ba[k]) + t^2 * conv_b[k]

where conv_a = a*a, conv_b = b*b, conv_ab = a*b (cross-correlation), conv_ba = b*a.

Since a*b and b*a are the same (convolution is commutative), we have:
conv(a_t, a_t)[k] = (1-t)^2 * C_a[k] + 2t(1-t) * C_ab[k] + t^2 * C_b[k]

This is a quadratic in t for each k. The max over k is the max of quadratics, which is convex.

For the line search, I can:
1. Precompute C_a, C_ab, C_b
2. For a given t, compute all conv values and take the max
3. Use golden section search

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k. Returns sparse-like rows."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp(a, M, A_ub_cache=None, top_k=None, conv=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M."""
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        if A_ub_cache is not None:
            A_ub = A_ub_cache[peaks]
        else:
            A_ub_full = build_lp_matrix(a)
            A_ub = A_ub_full[peaks]
        b_ub = np.full(len(peaks), M)
    else:
        if A_ub_cache is not None:
            A_ub = A_ub_cache
        else:
            A_ub = build_lp_matrix(a)
        b_ub = np.full(num_k, M)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def line_search_conv(a, b, C_a, C_ab, C_b):
    """Golden section search for optimal t in a_t = (1-t)*a + t*b minimizing max conv."""
    def obj_at_t(t):
        conv_val = (1 - t) ** 2 * C_a + 2 * t * (1 - t) * C_ab + t ** 2 * C_b
        return conv_val.max()

    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj_at_t(c_pt)
    fd = obj_at_t(d_pt)

    for _ in range(40):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj_at_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at_t(d_pt)

    t_best = (lo + hi) / 2
    return t_best, obj_at_t(t_best)

def subgradient_optimize(a0, time_limit, lr_init=0.02, momentum=0.9, seed=0, verbose=False):

    rng = np.random.default_rng(seed)

    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    avg_grad_sq = 1.0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_max = np.argmax(conv)

        # Subgradient: d conv[k_max] / da[p] = a[k_max - p] if 0 <= k_max - p < n
        g = np.zeros(n)
        p_min = max(0, k_max - n + 1)
        p_max = min(n - 1, k_max)
        if p_max >= p_min:
            g[p_min:p_max + 1] = a[k_max - (np.arange(p_min, p_max + 1))]

        # Project gradient to tangent space of simplex
        g_mean = g.mean()
        g_proj = g - g_mean

        # Normalize
        g_norm_sq = np.sum(g_proj ** 2)
        if g_norm_sq < 1e-20:
            break
        avg_grad_sq = 0.95 * avg_grad_sq + 0.05 * g_norm_sq

        # Adaptive learning rate
        lr_eff = lr / (1 + np.sqrt(g_norm_sq / max(avg_grad_sq, 1e-20)))

        v = momentum * v - lr_eff * g_proj
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9
                v *= 0.7
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

        if verbose and no_improve % 1000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter: obj={best_obj:.6f}, lr={lr:.5f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj

                    for _ in range(15):
                        mid1 = lo + (hi - lo) / 3
                        mid2 = hi - (hi - lo) / 3
                        a_t1 = a.copy()
                        a_t1[i] -= mid1
                        a_t1[q] += mid1
                        a_t1 = np.maximum(a_t1, 0)
                        a_t1 /= a_t1.sum()
                        v1 = fast_convolve(a_t1).max()

                        a_t2 = a.copy()
                        a_t2[i] -= mid2
                        a_t2[q] += mid2
                        a_t2 = np.maximum(a_t2, 0)
                        a_t2 /= a_t2.sum()
                        v2 = fast_convolve(a_t2).max()

                        if v1 < v2:
                            hi = mid2
                        else:
                            lo = mid1

                    delta_opt = (lo + hi) / 2
                    if delta_opt > 1e-8:
                        a_new = a.copy()
                        a_new[i] -= delta_opt
                        a_new[q] += delta_opt
                        a_new = np.maximum(a_new, 0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a_new /= s_new
                        conv_new = fast_convolve(a_new)
                        obj_new = 2.0 * n * conv_new.max()
                        if obj_new < best_obj - 1e-10:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
                            best_improvement = 1
                            break

        iteration += 1
        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter {iteration}: obj={best_obj:.6f}, t={elapsed:.1f}s")

        if best_improvement == 0 and iteration % 500 == 0:
            # No improvement in a while, perturb
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def multi_peak_subgradient(a0, time_limit, lr_init=0.03, momentum=0.92,
                           num_peaks=3, seed=0):
    """Subgradient using multiple peak locations for better gradient estimation."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Get top-k peaks
        top_idx = np.argsort(conv)[-num_peaks:]
        # Weight peaks by their values (softmax-like)
        w = np.exp(100 * (conv[top_idx] - conv[top_idx].max()))
        w /= w.sum()

        # Averaged subgradient
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            p_min = max(0, ki - n + 1)
            p_max = min(n - 1, ki)
            if p_max >= p_min:
                g[p_min:p_max + 1] += wi * a[ki - np.arange(p_min, p_max + 1)]

        g_proj = g - g.mean()
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-15:
            break

        g_proj /= g_norm
        v = momentum * v - lr * g_proj
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.92
                v *= 0.8
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.002
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.4
                no_improve = 0

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()

    # Edge-concentrated
    edge = np.zeros(n)
    edge[:max(1, n // 5)] = 1.0
    edge[-max(1, n // 5):] = 1.0
    edge /= edge.sum()
    inits['edge'] = edge

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss_c'] = g

    # Two-sided decay from center
    d = np.abs(x - 0.5 * (n - 1))
    decay = np.exp(-d / (0.3 * n))
    decay /= decay.sum()
    inits['decay_c'] = decay

    # Cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random Dirichlet
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['rand_unif'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_mid'] = rng.dirichlet(np.ones(n) * 5.0)

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                       np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    return inits


def evaluate_obj_normalized(a):
    """Evaluate objective assuming sum(a)=1. Returns 2n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP with line search, multi-peak subgradient,
    coordinate descent polish. Explores multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except (NameError, Exception):
        pass

    # === Phase 1: Quick exploration across lengths (120s) ===
    print("=== Phase 1: Quick Exploration ===")
    t_p1 = time.monotonic()
    p1_time = 120.0

    lengths = [256, 320, 384, 448, 512]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)

    scan_results = []
    for n in lengths:
        remaining = t_p1 + p1_time - time.monotonic() - 10
        if remaining < 15:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        # Prioritize
        priority = ['parent', 'edge', 'gauss_c', 'ramp', 'uniform', 'rand_sparse', 'decay_c']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        t_per = max(5.0, remaining / max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            t_alloc = min(t_per, t_p1 + p1_time - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_v2(a0, time_limit=t_alloc,
                                            top_k=max(30, n // 4),
                                            seed=n + hash(name) % 1000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan n={n} {name}: score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # === Phase 2: Deep LP optimization (600s) ===
    print("=== Phase 2: Deep LP ===")
    t_p2 = time.monotonic()
    p2_time = 600.0

    # Select top candidates
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:5]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(deep_starts) >= 4:
            break

    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    # Configs
    configs = [
        {'top_k': None, 'relax': 1.0},
        {'top_k': 80, 'relax': 1.0},
        {'top_k': 200, 'relax': 0.99},
        {'top_k': 50, 'relax': 1.01},
    ]

    total_runs = len(deep_starts) * len(configs)
    t_per_run = max(10.0, p2_time / max(1, total_runs))

    run_i = 0
    for n_s, a_s in deep_starts:
        for cfg in configs:
            remaining = t_p2 + p2_time - time.monotonic() - 10
            if remaining < 8:
                break
            t_alloc = min(t_per_run, remaining)
            if t_alloc < 5:
                break

            a_opt, obj_opt = lp_optimize_v2(a_s, time_limit=t_alloc,
                                            top_k=cfg['top_k'],
                                            relax=cfg['relax'],
                                            seed=run_i * 13 + 7)
            score = float(evaluate_sequence(a_opt.tolist()))
            run_i += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP run {run_i} n={n_s}: score={score:.6f}")

    print(f"Phase 2 best: {best_score:.6f}")

    # === Phase 3: Multi-peak subgradient (350s) ===
    print("=== Phase 3: Multi-peak Subgradient ===")
    t_p3 = time.monotonic()
    p3_time = 350.0

    if best_sequence is not None:
        sg_configs = [
            {'lr_init': 0.03, 'momentum': 0.92, 'num_peaks': 3},
            {'lr_init': 0.05, 'momentum': 0.9, 'num_peaks': 5},
            {'lr_init': 0.02, 'momentum': 0.95, 'num_peaks': 2},
            {'lr_init': 0.01, 'momentum': 0.9, 'num_peaks': 1},
        ]

        t_per_sg = p3_time / (len(sg_configs) + 2)
        for i, cfg in enumerate(sg_configs):
            remaining = t_p3 + p3_time - time.monotonic() - 5
            if remaining < 5:
                break
            t_alloc = min(t_per_sg, remaining)
            if t_alloc < 5:
                break

            a_opt, obj_opt = multi_peak_subgradient(best_sequence, time_limit=t_alloc, seed=i * 31 + 3, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MP-SG [{i}]: score={score:.6f}")

        # Standard subgradient too
        remaining = t_p3 + p3_time - time.monotonic() - 5
        if remaining > 10:
            a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=min(60, remaining),
                                                  lr_init=0.02, momentum=0.9, seed=99)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SG: score={score:.6f}")

    print(f"Phase 3 best: {best_score:.6f}")

    # === Phase 4: Coordinate descent polish (300s) ===
    print("=== Phase 4: Coordinate Descent ===")
    t_p4 = time.monotonic()
    p4_time = 300.0

    if best_sequence is not None:
        # Run CD with multiple seeds
        for i in range(3):
            remaining = t_p4 + p4_time - time.monotonic() - 5
            if remaining < 10:
                break
            t_alloc = min(80, remaining)
            if t_alloc < 5:
                break
            a_opt, obj_opt = coordinate_descent_polish(best_sequence, time_limit=t_alloc, seed=100 + i * 7)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CD [{i}]: score={score:.6f}")

    print(f"Phase 4 best: {best_score:.6f}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)

        # Short LP burst
        t_lp = min(100, remaining * 0.3)
        if t_lp > 10:
            a_opt, _ = lp_optimize_v2(best_sequence, time_limit=t_lp,
                                      top_k=max(60, n_best // 4), seed=555)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

        # Short subgradient
        remaining2 = deadline - time.monotonic()
        t_sg = min(80, remaining2 * 0.4)
        if t_sg > 10:
            a_opt, _ = multi_peak_subgradient(best_sequence, time_limit=t_sg,
                                              lr_init=0.008, momentum=0.95,
                                              num_peaks=3, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final SG: score={score:.6f}")

        # Tiny perturbations
        remaining3 = deadline - time.monotonic() - 10
        if remaining3 > 10:
            for i in range(min(15, int(remaining3 / 3))):
                if time.monotonic() > deadline - 15:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_p = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_p) and score_p < best_score:
                        best_score = score_p
                        best_sequence = a_pert.copy()
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
