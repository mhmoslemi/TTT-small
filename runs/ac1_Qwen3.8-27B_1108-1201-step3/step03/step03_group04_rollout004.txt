```python
def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * (np.arange(1, n+1) - 1) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 1.0 / n
    t_temp = t_init  # temperature for log-sum-exp

        c = fast_convolve(a)

        max_tc = t_c.max()
        w = np.exp(t_c - max_tc)
```

```python
def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(v - theta, 0)

def compute_gradient(a, w):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    w has length 2n-1, a has length n.
    Returns gradient of length n.
    """
    n = len(a)
    # g[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # = 2 * convolve(w, a_rev)[n-1 + i] where a_rev = a[::-1]
    size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(a[::-1], size)
    conv_full = np.fft.irfft(W * A_rev, size)
    return 2.0 * conv_full[n - 1: 2 * n - 1]

def subgradient_polish(a0, time_limit, lr_init=0.001, momentum=0.9):
    """Polish with projected subgradient descent (exact max)."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        max_val = c.max()
        
        # Subgradient: supported on argmax
        max_idx = np.where(c >= max_val - 1e-12)[0]
        
        # Average subgradient over all max indices
        g = np.zeros(n)
        for k in max_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(max_idx)
        
        v = momentum * v - lr * g
        a_new = proj_simplex(a + v)
        
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9999
        it += 1
    
    return best_a, best_obj
```
