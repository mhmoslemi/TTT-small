    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    return np.real(np.fft.ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def grad_weighted_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (2 * n + 63).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    grad = np.real(np.fft.ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute the objective value (assuming sum(a)=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_param_obj(x, n, beta):
    a = np.exp(x - x.max())
    a /= a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def softmax_param_jac(x, n, beta):
    a = np.exp(x - x.max())
    a /= a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = grad_weighted_conv(a, w)
    lag = np.dot(a, grad_a)
    return a * (grad_a - lag)


def run_lbfgsb(a0, beta, maxiter=300):
    """Run L-BFGS-B on the softmax parameterization."""
    n = len(a0)
    a_clipped = np.maximum(a0, 1e-30)
    a_clipped /= a_clipped.sum()
    x0 = np.log(a_clipped)
    result = minimize(softmax_param_obj, x0, jac=softmax_param_jac,
                      args=(n, beta), method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    return a_opt


def subgradient_descent(a0, n_iter=1000, top_k=10, lr_init=0.05, decay=0.999):
    """Subgradient descent on the max-conv objective with momentum."""
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = obj_val(a)
    lr = lr_init
    mu = 0.9  # momentum coefficient

    for it in range(n_iter):
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Active set: top-k peaks
        idx = np.argsort(conv)[::-1][:top_k]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        g = grad_weighted_conv(a, w)

        # Project gradient to tangent space of simplex
        g = g - np.mean(g)

        # Momentum update
        v = mu * v - lr * g
        a_new = project_simplex(a + v)  # negative gradient direction

        a = a_new
        lr *= decay

        # Periodically restart momentum to avoid getting stuck
        if it % 200 == 0 and it > 0:
            v = np.zeros(n)

    return best_a


def peak_shave(a, n_steps=50):
    """Directly reduce the peak of the convolution."""
    n = len(a)
    a = a.copy()
    for step in range(n_steps):
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]
        
        # Find elements contributing to the peak
        # conv[k_peak] = sum_i a[i]*a[k_peak-i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        
        # Reduce elements proportionally to their contribution
        # We want to reduce the peak by a fraction
        reduction = 0.5 * (peak_val - conv.mean()) / max(total_contrib, 1e-15)
        reduction = min(reduction, 0.1)  # Cap the reduction
        
        a -= reduction * contrib * a  # Quadratic reduction (a[i] * contrib[i])
        # Actually, let's do: reduce a[i] by delta_i where delta_i is proportional to a[i]*contrib[i]
        # This reduces conv by sum delta_i * contrib[i] approximately
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            a = np.ones(n) / n
            break
    
    return a


def momentum_subgrad(a0, n_iter=2000, top_k=5, lr_init=0.02, momentum=0.9):
    """Improved subgradient with Nesterov-like momentum."""
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = obj_val(a)
    lr = lr_init

    for it in range(n_iter):
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if obj < 1.001:  # Can't do much better
            break

        # Adaptive top-k: include lags within 95% of max
        threshold = conv.max() * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) > 20:
            active = np.argsort(conv)[::-1][:20]
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = grad_weighted_conv(a, w)
        g = g - np.mean(g)

        # Nesterov step
        v = momentum * v
        a_nc = project_simplex(a - v * momentum)  # Nesterov look-ahead
        # Actually, let's keep it simple: just heavy ball
        v = momentum * v - lr * g
        a = project_simplex(a + v)

        # Decay learning rate
        lr = lr_init / (1 + 0.01 * it)

    return best_a


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    x = np.arange(n, dtype=np.float64)
    # Gaussian
    for sf in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (sf*n))**2)
        g /= g.sum()
        inits['gauss_%s' % sf] = g
    
    # Bimodal
    bi = np.zeros(n)
    edge = n // 3
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp / ps2
    
    return inits

from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    return np.real(np.fft.ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def grad_weighted_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    N = 1 << (2 * n + 63).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    grad = np.real(np.fft.ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj(x, n, beta):
    """Objective for scipy minimize with softmax parametrization."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def softmax_jac(x, n, beta):
    """Jacobian for scipy minimize with softmax parametrization."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = grad_weighted_conv(a, w)
    lag = np.dot(a, grad_a)
    return a * (grad_a - lag)


def run_lbfgsb(a0, beta, maxiter=300):
    """Run L-BFGS-B on softmax parameterization."""
    n = len(a0)
    a_clipped = np.maximum(a0, 1e-30)
    a_clipped = a_clipped / a_clipped.sum()
    x0 = np.log(a_clipped)
    result = minimize(softmax_obj, x0, jac=softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 100,
                                                   'ftol': 1e-15, 'gtol': 1e-10})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def run_lbfgsb_multi_beta(a0, betas, maxiter_per_beta=200):
    """Continuation method: run L-BFGS-B with decreasing beta schedule."""
    n = len(a0)
    a_cur = a0.copy()
    for beta in betas:
        a_cur = run_lbfgsb(a_cur, beta=beta, maxiter=maxiter_per_beta)
    return a_cur


def momentum_subgrad(a0, n_iter=2000, lr_init=0.01, momentum=0.9, active_frac=0.95):
    """
    Nesterov momentum subgradient descent on max(conv) objective.
    Uses adaptive active set (top lags within active_frac of max).
    """
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    lr = lr_init

    for it in range(n_iter):
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Adaptive active set
        M = conv.max()
        threshold = M * active_frac
        active = np.where(conv >= threshold)[0]
        if len(active) > 30:
            active = np.argsort(conv)[::-1][:30]

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = grad_weighted_conv(a, w)

        # Project to tangent space
        g = g - np.mean(g)

        # Nesterov momentum
        v_new = momentum * v
        a_temp = np.maximum(a + v_new, 0)
        a_temp = a_temp / max(a_temp.sum(), 1e-15)

        # Recompute gradient at look-ahead point for Nesterov
        conv_look = fast_convolve(a_temp)
        M_look = conv_look.max()
        thr_look = M_look * active_frac
        active_look = np.where(conv_look >= thr_look)[0]
        if len(active_look) > 30:
            active_look = np.argsort(conv_look)[::-1][:30]
        w_look = np.zeros(2 * n - 1)
        w_look[active_look] = 1.0 / len(active_look)
        g_look = grad_weighted_conv(a_temp, w_look)
        g_look = g_look - np.mean(g_look)

        v = v_new - lr * g_look
        a = project_simplex(a + v)

        # Learning rate schedule
        lr = lr_init / (1.0 + 0.005 * it)

        # Periodic momentum reset to avoid oscillation
        if it > 0 and it % 500 == 0:
            v *= 0.5

    return best_a


def peak_shave(a, n_steps=200, strength=0.3):
    """
    Iteratively reduce the dominant peak of the convolution by
    decreasing elements that contribute most to it.
    """
    n = len(a)
    a = a.copy()
    for step in range(n_steps):
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]
        mean_val = conv.mean()

        if peak_val <= mean_val * 1.05:
            break

        # Contribution of a[i] to conv[k_peak]: a[k_peak - i] if valid
        contrib = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = a[j] * a[i]

        total = contrib.sum()
        if total < 1e-15:
            break

        # Reduce proportionally to contribution, preserving sum
        excess = peak_val - mean_val
        reduction = strength * excess / max(total, 1e-15)
        reduction = min(reduction, 0.15)

        # Reduce a[i] proportional to its contribution to the peak
        deltas = reduction * contrib / max(a.sum(), 1e-15)
        a -= deltas
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            break

    return a


def coordinate_refine(a, n_sweeps=10):
    """
    Coordinate descent: for each coordinate, try to adjust it to
    reduce the max of the convolution.
    """
    n = len(a)
    a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = a[i]
            if old < 1e-12:
                continue
            # Try reducing
            for frac in [0.85, 0.9, 0.95]:
                a[i] = old * frac
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = 2.0 * n * fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    break
            if not improved:
                a[i] = old / a.sum() * a.sum()  # restore
                a[i] = old
                # Try small increase (might help flatten other peaks)
                for frac in [1.1, 1.05]:
                    a[i] = old * frac
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-10:
                        a[i] = old
                        break
                    a /= s
                    obj = 2.0 * n * fast_convolve(a).max()
                    if obj < best_obj:
                        best_obj = obj
                        improved = True
                        break
                    a[i] = old
        a = np.maximum(a, 0)
        a = a / a.sum()
        if not improved:
            break

    return a


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations of length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % sf] = g

    # Cosine window
    cw = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n + 1)))
    cw = np.maximum(cw, 1e-10)
    cw /= cw.sum()
    inits['cosine'] = cw

    # Bimodal (edges weighted)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Triangular (peak at center)
    tri = 1.0 - np.abs(x - mid) / (mid + 1)
    tri = np.maximum(tri, 0.1)
    tri /= tri.sum()
    inits['triangle'] = tri

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp / ps2

    return inits
