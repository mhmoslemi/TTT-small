"""
Hybrid LP-SCA + smooth gradient optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses full-constraint LP for direction finding with line search, log-sum-exp smoothing
with annealing, multi-phase search over lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build sparse-aware constraint matrix for LP: (a*b)[k] <= M for all k=0..2n-2."""
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = np.arange(len(idx))  # placeholder, we'll fill with a values
    return None  # We build it on the fly in solve_lp_direction


def solve_lp_direction(a, M, n=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    if n is None:
        n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def lp_sca_optimize(a0, time_limit, seed=0, verbose=False):
    """LP-based Successive Convex Approximation with line search."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    step_base = 1.0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        b = solve_lp_direction(a, M)
        if b is None:
            # Try relaxed
            b = solve_lp_direction(a, M * 1.01)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Line search
        best_t = 0.0
        best_f = best_obj
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            f_new = 2.0 * n * conv_new.max()
            if f_new < best_f:
                best_f = f_new
                best_t = t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            a = a_new
            if best_f < best_obj - 1e-12:
                best_obj = best_f
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            no_improve = 0
            step_base = max(0.3, step_base * 0.7)

        if no_improve > 80:
            # Larger perturbation
            noise = rng.standard_normal(n) * 0.02
            a = project_simplex(a + noise)
            no_improve = 0
            step_base = 0.8

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-SCA iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def smooth_grad_optimize(a0, time_limit, seed=0, verbose=False):
    """Smooth gradient descent with log-sum-exp annealing on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    t_total = time_limit
    beta = 100.0
    lr = 0.1
    momentum = 0.9
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / t_total)
        # Anneal beta
        beta = 100.0 * (1.0 + 20.0 * progress ** 2)
        # Decay learning rate
        lr = 0.1 * (1.0 - 0.7 * progress)

        conv = fast_convolve(a)
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Gradient computation via FFT
        fft_size = 1 << ((2 * n - 1) - 1).bit_length()
        w_padded = np.zeros(fft_size)
        w_padded[:len(w)] = w
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a

        W = np.fft.rfft(w_padded, fft_size)
        A = np.fft.rfft(a_padded, fft_size)

        # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        # First part: cross-correlation
        cross = np.fft.irfft(A * np.conj(W), fft_size).real[:n]
        # Second part: convolution
        conv_wa = np.fft.irfft(A * W, fft_size).real[:n]

        grad = cross + conv_wa
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 150:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = 0.05
                no_improve = 0

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMOOTH iter {iter_count}: obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.4f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for length n, normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Various window shapes
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        # Raised cosine-like
        t = x / (n - 1) if n > 1 else np.array([0.5])
        w = (np.cos(np.pi * t) ** 2)
        w = np.maximum(w, 1e-6)
        w = w / w.sum()
        inits[f'cos_{alpha}'] = w

    # Gaussian bumps at different positions
    for pos in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            g = np.exp(-0.5 * ((x / n - pos) / width) ** 2)
            g = g / g.sum()
            inits[f'g_{pos:.1f}_{width:.1f}'] = g

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = ramp[::-1] / ramp.sum()

    # Random sparse
    for sparsity in [0.3, 0.5, 0.7]:
        mask = rng.random(n) < sparsity
        r = np.where(mask, 1.0, 0.0)
        if r.sum() > 0:
            inits[f'sparse_{sparsity}'] = r / r.sum()

    # Random smooth (low-frequency random)
    for _ in range(4):
        freqs = rng.uniform(0, 3, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        r = np.ones(n)
        for f, p in zip(freqs, phases):
            r += np.cos(2 * np.pi * f * x / n + p)
        r = np.maximum(r, 0.0)
        rs = r.sum()
        if rs > 1e-10:
            inits[f'smooth_rand_{_}'] = r / rs

    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0).astype(np.float64)
        if len(pp) >= n:
            pp_sub = pp[:n]
            ps = pp_sub.sum()
            if ps > 1e-10:
                inits['parent_sub'] = pp_sub / ps
        else:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
            pp_interp = np.maximum(pp_interp, 0)
            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_interp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-SCA with line search + smooth gradient annealing.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST: score={score:.8f}, n={len(a)}")

    # ========== Phase 1: Quick Exploration (150s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t1 = time.monotonic()
    phase1_time = 150.0

    lengths = [160, 200, 240, 280, 320, 360, 400, 480, 560, 640]
    results = []

    for n in lengths:
        if time.monotonic() - t1 > phase1_time - 20:
            break
        inits = generate_diverse_inits(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize
        priority = ['parent_sub', 'parent_interp', 'uniform', 'bimodal',
                    'cos_1.0', 'g_0.5_0.3', 'smooth_rand_0', 'ramp']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() - t1 > phase1_time - 10:
                break
            a0 = inits[name].copy()
            t_alloc = min(12.0, phase1_time - (time.monotonic() - t1) - 5)
            if t_alloc < 3:
                break

            a_opt, obj_opt = lp_sca_optimize(a0, time_limit=t_alloc,
                                             seed=n * 31 + hash(name) % 1000)
            # Scale to max=1000
            amax = a_opt.max()
            if amax > 0:
                a_scaled = a_opt * (1000.0 / amax)
            else:
                a_scaled = a_opt
            try_update(a_scaled)
            results.append((obj_opt, n, name, a_opt.copy()))

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ========== Phase 2: Deep LP-SCA (600s) ==========
    print("=== Phase 2: Deep LP-SCA ===")
    t2 = time.monotonic()
    phase2_time = 600.0

    # Collect starting points
    deep_starts = []
    seen = set()
    for obj, n, name, a_opt in results[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
        if len(deep_starts) >= 8:
            break

    if best_sequence is not None:
        bs = best_sequence / best_sequence.max() if best_sequence.max() > 0 else best_sequence
        deep_starts.append((len(best_sequence), bs.copy()))

    # Add some fresh starts at promising lengths
    for n in [320, 360, 400, 480]:
        if len(deep_starts) >= 10:
            break
        inits = generate_diverse_inits(n, parent, rng)
        for key in ['smooth_rand_1', 'g_0.3_0.2', 'cos_2.0']:
            if key in inits:
                deep_starts.append((n, inits[key].copy()))
                break

    num_runs = len(deep_starts)
    time_per_run = phase2_time / max(1, num_runs)

    for i, (n_start, a_start) in enumerate(deep_starts):
        if time.monotonic() - t2 > phase2_time - 30:
            break
        t_alloc = min(time_per_run, phase2_time - (time.monotonic() - t2) - 15)
        if t_alloc < 10:
            break

        print(f"  Deep run {i+1}/{num_runs}: n={n_start}, alloc={t_alloc:.0f}s")
        a_opt, obj_opt = lp_sca_optimize(a_start, time_limit=t_alloc,
                                         seed=i * 137 + 7, verbose=False)
        amax = a_opt.max()
        if amax > 0:
            a_scaled = a_opt * (1000.0 / amax)
        else:
            a_scaled = a_opt
        try_update(a_scaled)

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ========== Phase 3: Smooth Gradient (400s) ==========
    print("=== Phase 3: Smooth Gradient ===")
    t3 = time.monotonic()
    phase3_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_start = best_sequence / best_sequence.max() if best_sequence.max() > 0 else best_sequence

        # Multiple gradient runs with different settings
        num_grad_runs = 5
        time_per_grad = phase3_time / num_grad_runs

        for i in range(num_grad_runs):
            if time.monotonic() - t3 > phase3_time - 20:
                break
            t_alloc = min(time_per_grad, phase3_time - (time.monotonic() - t3) - 10)
            if t_alloc < 15:
                break

            if i == 0:
                a_cur = a_start.copy()
            else:
                # Perturb previous best
                noise = rng.standard_normal(n_best) * (0.005 + 0.01 * i)
                a_cur = project_simplex(a_start + noise)

            a_opt, obj_opt = smooth_grad_optimize(a_cur, time_limit=t_alloc,
                                                  seed=i * 41 + 3)
            amax = a_opt.max()
            if amax > 0:
                a_scaled = a_opt * (1000.0 / amax)
            else:
                a_scaled = a_opt
            try_update(a_scaled)

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ========== Phase 4: Hybrid Alternating (350s) ==========
    print("=== Phase 4: Hybrid Alternating ===")
    t4 = time.monotonic()
    phase4_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence / best_sequence.max() if best_sequence.max() > 0 else best_sequence
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        best_obj_phase = 2.0 * n_best * fast_convolve(a).max()
        best_a_phase = a.copy()
        v = np.zeros(n_best)
        beta = 500.0

        t4_start = time.monotonic()
        while time.monotonic() - t4_start < phase4_time:
            conv = fast_convolve(a)
            M = conv.max()

            # LP step
            b = solve_lp_direction(a, M)
            if b is not None and b.sum() > 1e-15:
                b = b / b.sum()
                # Small step for LP
                a = (1.0 - 0.5) * a + 0.5 * b
                a = np.maximum(a, 0)
                a = a / a.sum()

            # Smooth gradient step
            beta = min(5000.0, beta * 1.01)
            conv = fast_convolve(a)
            scaled = beta * conv
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(conv))
                w[np.argmax(conv)] = 1.0
            else:
                w = w / w_sum

            fft_size = 1 << ((2 * n_best - 1) - 1).bit_length()
            w_padded = np.zeros(fft_size)
            w_padded[:len(w)] = w
            a_padded = np.zeros(fft_size)
            a_padded[:n_best] = a
            W = np.fft.rfft(w_padded, fft_size)
            A = np.fft.rfft(a_padded, fft_size)
            cross = np.fft.irfft(A * np.conj(W), fft_size).real[:n_best]
            conv_wa = np.fft.irfft(A * W, fft_size).real[:n_best]
            grad = cross + conv_wa
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm

            lr = 0.02
            v = 0.9 * v - lr * grad
            a = project_simplex(a + v)

            conv = fast_convolve(a)
            obj = 2.0 * n_best * conv.max()
            if obj < best_obj_phase - 1e-12:
                best_obj_phase = obj
                best_a_phase = a.copy()

        amax = best_a_phase.max()
        if amax > 0:
            a_scaled = best_a_phase * (1000.0 / amax)
        else:
            a_scaled = best_a_phase
        try_update(a_scaled)

    print(f"Phase 4 done. Best: {best_score:.8f}")

    # ========== Phase 5: Final Polish (remaining) ==========
    print("=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"  Remaining time: {remaining:.1f}s")

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence / best_sequence.max() if best_sequence.max() > 0 else best_sequence

        # Fine LP with very small steps
        if remaining > 60:
            t_alloc = min(remaining * 0.4, 80.0)
            a_opt, obj_opt = lp_sca_optimize(a, time_limit=t_alloc, seed=9999)
            amax = a_opt.max()
            if amax > 0:
                a_scaled = a_opt * (1000.0 / amax)
            else:
                a_scaled = a_opt
            try_update(a_scaled)

        # Fine gradient
        remaining = deadline - time.monotonic()
        if remaining > 40:
            a = best_sequence / best_sequence.max() if best_sequence.max() > 0 else best_sequence
            t_alloc = min(remaining * 0.35, 60.0)
            a_opt, obj_opt = smooth_grad_optimize(a, time_limit=t_alloc, seed=7777)
            amax = a_opt.max()
            if amax > 0:
                a_scaled = a_opt * (1000.0 / amax)
            else:
                a_scaled = a_opt
            try_update(a_scaled)

        # Tiny perturbations
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_best = len(best_sequence)
            a_base = best_sequence / best_sequence.max() if best_sequence.max() > 0 else best_sequence
            num_tries = min(int(remaining / 8), 15)
            for i in range(num_tries):
                if time.monotonic() > deadline - 15:
                    break
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(a_base + noise)
                    amax = a_pert.max()
                    if amax > 0:
                        a_scaled = a_pert * (1000.0 / amax)
                    else:
                        a_scaled = a_pert
                    score = float(evaluate_sequence(a_scaled.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_scaled.copy()
                        print(f"  Perturb [{i}] s={scale}: score={score:.8f}")
                        break

    # Final safety
    if best_sequence is None:
        best_sequence = np.ones(128) * (1000.0 / 128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) * (1000.0 / 128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}, max={best_sequence.max():.2f}")
    return [float(x) for x in best_sequence]
