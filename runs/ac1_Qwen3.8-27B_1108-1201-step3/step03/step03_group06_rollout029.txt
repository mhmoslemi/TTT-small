from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    """Build constraint matrix A_ub for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def build_lp_matrix_topk(a, conv, top_k):
    """Build constraint matrix using only top-k peak constraints."""
    n = len(a)
    num_k = 2 * n - 1
    if top_k >= num_k:
        return build_lp_matrix(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    A_ub = np.zeros((top_k, n))
    for row, k in enumerate(peaks):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[row, idx] = a[k - idx]
    return A_ub

def solve_lp(a, M, A_ub=None, top_k=None, conv=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M."""
    n = len(a)
    num_k = 2 * n - 1
    
    if A_ub is None:
        if top_k is not None and top_k < num_k and conv is not None:
            A_ub = build_lp_matrix_topk(a, conv, top_k)
            b_ub = np.full(top_k, M)
        else:
            A_ub = build_lp_matrix(a)
            b_ub = np.full(num_k, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lp_direction(a, M, top_k=None, conv=None):
    """Get LP direction and rescale to simplex."""
    n = len(a)
    b = solve_lp(a, M, top_k=top_k, conv=conv)
    if b is None:
        return None
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None
    return b / b_sum

def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    
    return cross + conv_wa, conv

def coordinate_descent_1d(a, idx, n, max_iter=1000, tol=1e-10):
    """Try to optimize a single coordinate a[idx] while keeping sum=1."""
    # If we change a[idx] by delta, we need to compensate elsewhere
    # Simplest: scale all other elements to maintain sum=1
    # But this changes the objective in a complex way
    # Alternative: just do a 1D line search on a[idx] with others proportionally scaled
    
    a_orig = a.copy()
    best_val = compute_objective(a_orig)
    best_delta = 0.0
    
    # Binary search / golden section on delta
    a_base = a.copy()
    a_base[idx] = 0  # temporary
    s_others = a_base.sum()
    
    # a[idx] = t, others scaled by (1-t)/s_others
    # t in [0, min(1, 2*a[idx])]
    t_lo = 0.0
    t_hi = min(1.0, 2.0 * a[idx])
    
    for _ in range(40):
        t_mid1 = t_lo + (t_hi - t_lo) / 3
        t_mid2 = t_hi - (t_hi - t_lo) / 3
        
        a1 = np.copy(a_orig)
        a1[idx] = t_mid1
        a1[np.arange(n) != idx] *= (1 - t_mid1) / max(s_others, 1e-15)
        obj1 = compute_objective(a1)
        
        a2 = np.copy(a_orig)
        a2[idx] = t_mid2
        a2[np.arange(n) != idx] *= (1 - t_mid2) / max(s_others, 1e-15)
        obj2 = compute_objective(a2)
        
        if obj1 < obj2:
            t_hi = t_mid2
        else:
            t_lo = t_mid1
    
    t_opt = (t_lo + t_hi) / 2
    a_new = np.copy(a_orig)
    a_new[idx] = t_opt
    a_new[np.arange(n) != idx] *= (1 - t_opt) / max(s_others, 1e-15)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    
    obj_new = compute_objective(a_new)
    if obj_new < best_val:
        return a_new, obj_new
    return a_orig, best_val

def coordinate_descent(a, time_limit, seed=0, verbose=False):

"""
Optimized LP + gradient + hybrid approach to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous:
- Use ALL convolution constraints in LP (feasible for n<=512 with HiGHS)
- Active set tightening for better LP directions
- Multi-phase optimization: LP -> gradient -> hybrid -> coordinate polish
- Diverse analytical initializations (U-shaped, bimodal, cosine)
- Adaptive step sizing with damping
- Simulated annealing restarts for escaping local minima
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix_full(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k=0..2n-2."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def build_lp_matrix_subset(a, conv, subset_size):
    """Build constraint matrix using only the most critical constraints."""
    n = len(a)
    num_k = 2 * n - 1
    if subset_size >= num_k:
        return build_lp_matrix_full(a), num_k
    # Find top constraints (where conv is largest)
    peaks = np.argpartition(conv, -subset_size)[-subset_size:]
    A_ub = np.zeros((subset_size, n))
    for row, k in enumerate(peaks):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[row, idx] = a[k - idx]
    return A_ub, subset_size


def solve_lp(a, M, top_k=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for selected k."""
    n = len(a)
    num_k = 2 * n - 1
    conv = fast_convolve(a)

    if top_k is not None and top_k < num_k:
        A_ub, n_constraints = build_lp_matrix_subset(a, conv, top_k)
        b_ub = np.full(n_constraints, M)
    else:
        A_ub = build_lp_matrix_full(a)
        b_ub = np.full(num_k, M)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_gradient(a, beta=500.0):
    """Smooth gradient of max(conv(a,a)) via softmax."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    return cross + conv_wa, conv


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # U-shaped distributions (high at edges, low in middle)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        u = (x + 1) ** alpha + (n - x) ** alpha
        u /= u.sum()
        inits[f'u_{alpha:.1f}'] = u

    # Inverted-U (high at center)
    for alpha in [0.5, 1.0, 2.0]:
        v = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
        v = v ** alpha
        v /= v.sum()
        inits[f'invu_{alpha:.1f}'] = v

    # Cosine windows
    c1 = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    c1 = np.maximum(c1, 0.01)
    c1 /= c1.sum()
    inits['cos_edge'] = c1

    c2 = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    c2 = np.maximum(c2, 0.01)
    c2 /= c2.sum()
    inits['cos_center'] = c2

    # Bimodal Gaussians at edges
    for sigma in [0.08, 0.12, 0.15, 0.2, 0.25]:
        bi = (np.exp(-0.5 * (x / (sigma * n)) ** 2) +
              np.exp(-0.5 * ((x - n + 1) / (sigma * n)) ** 2))
        bi /= bi.sum()
        inits[f'bimod_{sigma:.2f}'] = bi

    # Power-law edges
    for p in [1.0, 1.5, 2.0, 2.5]:
        pw = np.minimum(x + 1, n - x) ** (-p)
        pw /= pw.sum()
        inits[f'pow_{p:.1f}'] = pw

    # Random
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_dense'] = rng.dirichlet(np.ones(n) * 5.0)

    # Sine
    sine = np.abs(np.sin(np.pi * x / max(n - 1, 1)))
    sine = 1.0 - sine
    sine = np.maximum(sine, 0.001)
    sine /= sine.sum()
    inits['sine_edge'] = sine

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        if len(p) >= n:
            pp = p[:n]
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(p)), p)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps

    return inits


def lp_optimize(a0, time_limit, top_k=None, relax=1.0, step_init=1.0,

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max() * relax

        b = solve_lp(a, M, top_k=top_k)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if step < step_init * 0.8:
                step = min(step_init, step * 1.01)
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.92
            else:
                if no_improve > 15:
                    noise = rng.standard_normal(n) * 0.002
                    a = project_simplex(a + noise)
                    step = step_init * 0.4
                    no_improve = 0
                else:
                    step = max(0.01, step * 0.95)
                    continue

        if no_improve >= 40:
            noise = rng.standard_normal(n) * 0.008
            a = project_simplex(a + noise)
            step = step_init * 0.5
            no_improve = 0

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, t={time.monotonic()-t0:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.03,
                      momentum=0.9, seed=0, verbose=False):
    """Smooth gradient descent with momentum."""
    rng = np.random.default_rng(seed)

def fine_polish(a0, time_limit, seed=0):
    """Fine polishing with micro-perturbations and coordinate adjustments."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()

    scales = [0.0005, 0.001, 0.002, 0.005, 0.01]

    while time.monotonic() - t0 < time_limit:
        # Random micro-perturbation
        scale = scales[rng.integers(len(scales))]
        noise = rng.standard_normal(n) * scale
        a_try = project_simplex(best_a + noise)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-15:
            best_obj = obj_try
            best_a = a_try.copy()

        # Try shrinking the peak region
        conv = fast_convolve(best_a)
        peak_idx = np.argmax(conv)
        # Identify which elements contribute most to the peak
        i_min = max(0, peak_idx - n + 1)
        i_max = min(n - 1, peak_idx)
        contributors = list(range(i_min, i_max + 1))
        if contributors:
            c = rng.choice(contributors)
            a_try = best_a.copy()
            a_try[c] *= 0.95
            s_others = 1.0 - a_try[c]
            s_orig_others = 1.0 - best_a[c]
            if s_orig_others > 1e-15 and s_others > 0:
                mask = np.ones(n, dtype=bool)
                mask[c] = False
                a_try[mask] *= s_others / s_orig_others
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st > 1e-15:
                    a_try /= st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()

    return best_a, best_obj

    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
    except (NameError, Exception):
        pass

    # ========== Phase 1: Quick Scan (150s) ==========
    print("=== Phase 1: Quick Scan ===")
    t_p1 = time.monotonic()
    p1_time = 150.0

    lengths = [192, 256, 288, 320, 352, 384, 448]
    scan_results = []

    for n in lengths:
        if time.monotonic() - t_p1 > p1_time - 20:
            break

        inits = generate_initializations(n, rng, parent)
        # Prioritize promising initializations
        priority = ['parent', 'u_1.0', 'u_2.0', 'bimod_0.15', 'cos_edge',
                    'rand', 'invu_1.0', 'pow_2.0', 'u_1.5']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]

        time_per_init = max(8.0, (p1_time - (time.monotonic() - t_p1) - 15) /
                            max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() - t_p1 > p1_time - 10:
                break
            t_alloc = min(time_per_init, p1_time - (time.monotonic() - t_p1) - 8)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                         top_k=None, relax=1.0,
                                         step_init=1.0, seed=n * 7 + hash(name) % 10000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ========== Phase 2: Deep LP with Full Constraints (500s) ==========
    print("=== Phase 2: Deep LP ===")
    t_p2 = time.monotonic()
    p2_time = 500.0

    # Collect starting points
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)
    if best_sequence is not None:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    # Add analytical starts at best length
    best_n = deep_starts[0][0] if deep_starts else 320
    inits_best = generate_initializations(best_n, rng, parent)
    for extra in ['u_1.0', 'bimod_0.12', 'rand']:
        if extra in inits_best:
            deep_starts.append((best_n, inits_best[extra].copy()))

    # Deduplicate
    unique_starts = []
    seen_starts = set()
    for n_s, a_s in deep_starts:
        key = (n_s, tuple(np.round(a_s[:10], 6)))
        if key not in seen_starts:
            unique_starts.append((n_s, a_s))
            seen_starts.add(key)
    deep_starts = unique_starts[:8]

    n_starts = len(deep_starts)
    time_per_start = max(30.0, p2_time / max(1, n_starts))

    for i, (n_s, a_s) in enumerate(deep_starts):
        if time.monotonic() - t_p2 > p2_time - 20:
            break
        t_alloc = min(time_per_start, p2_time - (time.monotonic() - t_p2) - 15)
        if t_alloc < 20:
            break

        a_opt, obj_opt = lp_optimize(a_s, time_limit=t_alloc,
                                     top_k=None, relax=1.0,
                                     step_init=1.0, seed=1000 + i * 37)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP [{i}] n={n_s}: score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Hybrid LP + Gradient (450s) ==========
    print("=== Phase 3: Hybrid Optimization ===")
    t_p3 = time.monotonic()
    p3_time = 450.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        # Run hybrid from best and from perturbed versions
        hybrid_starts = [best_sequence.copy()]
        for i in range(3):
            noise = rng.standard_normal(n_best) * 0.005
            hybrid_starts.append(project_simplex(best_sequence + noise))

        time_per_hybrid = max(60.0, p3_time / max(1, len(hybrid_starts)))

        for i, a_start in enumerate(hybrid_starts):
            if time.monotonic() - t_p3 > p3_time - 30:
                break
            t_alloc = min(time_per_hybrid, p3_time - (time.monotonic() - t_p3) - 20)
            if t_alloc < 30:
                break

            a_opt, obj_opt = hybrid_optimize(a_start, time_limit=t_alloc, seed=2000 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Hybrid [{i}]: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Gradient Fine-tuning (200s) ==========
    print("=== Phase 4: Gradient Fine-tuning ===")
    t_p4 = time.monotonic()
    p4_time = 200.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        grad_configs = [
            {'beta_init': 500, 'lr_init': 0.02, 'momentum': 0.9},
            {'beta_init': 1000, 'lr_init': 0.01, 'momentum': 0.92},
            {'beta_init': 2000, 'lr_init': 0.005, 'momentum': 0.94},
        ]

        time_per_grad = max(40.0, p4_time / (len(grad_configs) + 2))

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() - t_p4 > p4_time - 20:
                break
            t_alloc = min(time_per_grad, p4_time - (time.monotonic() - t_p4) - 15)
            if t_alloc < 20:
                break

            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                               seed=3000 + i, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{i}]: score={score:.6f}")

        # Gradient from perturbed start
        if time.monotonic() - t_p4 < p4_time - 30:
            noise = rng.standard_normal(n_best) * 0.003
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(time_per_grad, p4_time - (time.monotonic() - t_p4) - 20)
            if t_alloc > 20:
                a_opt, obj_opt = gradient_optimize(a_pert, time_limit=t_alloc,
                                                   beta_init=2000, lr_init=0.008,
                                                   momentum=0.93, seed=3100)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad pert: score={score:.6f}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Fine Polish (remaining time) ==========
    print("=== Phase 5: Fine Polish ===")
    t_p5 = time.monotonic()
    p5_time = deadline - time.monotonic()

    if best_sequence is not None and p5_time > 60:
        # Fine polish with micro-perturbations
        t_alloc = min(p5_time * 0.6, 180.0)
        if t_alloc > 30:
            a_opt, obj_opt = fine_polish(best_sequence, time_limit=t_alloc, seed=5000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish: score={score:.6f}")

        # Final short LP
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=min(remaining * 0.5, 80.0),
                                         top_k=None, relax=1.0, step_init=0.3, seed=9999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

            # Final gradient
            remaining2 = deadline - time.monotonic() - 25
            if remaining2 > 20:
                a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=min(remaining2, 50.0),
                                                   beta_init=5000, lr_init=0.003,
                                                   momentum=0.95, seed=7777)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final Grad: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
