"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
(Adam), periodic LP-based direction finding, adaptive temperature, and
multi-phase search with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def compute_smooth_max_grad(a, w):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = 2.0 * w[k]
        if wk < 1e-12:
            continue
        if k < n:
            # grad[i] += wk * a[k-i] for i in [0, k]
            idx = np.arange(0, k + 1)
            grad[idx] += wk * a[k - idx]
        else:
            # grad[i] += wk * a[k-i] for i in [k-n+1, n-1]
            idx = np.arange(k - n + 1, n)
            grad[idx] += wk * a[k - idx]
    return grad

        w = exp_b / w_sum

        grad = compute_smooth_max_grad(a, w)

        # d(2n * smooth_max)/da = 2n * grad (since smooth_max = (1/tau)*log(sum(exp(tau*b))))

        # The gradient in the "free" space (before projection) is just 2n*grad.
        # The factor 2n is constant, so it just scales the learning rate.
