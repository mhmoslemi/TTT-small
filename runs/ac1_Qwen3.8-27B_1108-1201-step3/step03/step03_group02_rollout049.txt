def cross_convolve(a, b):

    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

Key innovation: parametric line search along LP directions using closed-form

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    if b is None:
        b = a
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        k_lo = j
        k_hi = min(num_rows - 1, j + n - 1)
        A[k_lo:k_hi + 1, j] = a[0:k_hi - k_lo + 1]
    return A


def solve_lp_max_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                a = project_simplex(a + noise)
                caa = fast_convolve(a)
                no_improve = 0
                perturb_scale *= 1.5
                if perturb_scale > 0.05:
                    perturb_scale = 0.05
                perturb_every = int(perturb_every * 1.3)
            if no_improve > perturb_every * 3:
                break
            continue

        # Take the step
        a = (1 - best_t) * a + best_t * best_b
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        caa = fast_convolve(a)
        cur_obj = 2.0 * n * caa.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_obj


def subgradient_refine(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=20,
                       temp=0.001, perturb_every=800, perturb_scale=0.003):
    """Projected subgradient with momentum on multi-peak objective."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    caa = fast_convolve(a)
    best_obj = 2.0 * n * caa.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient from top-k peaks
        k = min(top_k, len(caa) - 1)
        peak_idx = np.argpartition(caa, -k)[-k:]
        peak_vals = caa[peak_idx]

        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum

        # Subgradient: sg_j = 2 * sum_i w_i * a[k_i - j] (valid indices)
        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-12:
                continue
            pk = int(peak_idx[i])
            wi = 2.0 * w[i]
            j_start = max(0, pk - n + 1)
            j_end = min(n - 1, pk)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[pk - idx]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        caa = fast_convolve(a)
        cur_obj = 2.0 * n * caa.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            caa = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.2, lr * 0.6)
            temp *= 0.5
            if temp < 1e-6:
                temp = 1e-6

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-10:
            inits[f'cos_{freq}'] = a / sa

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Anti-concentrated: try to make convolution flat
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n) + 0.3 * np.cos(4.0 * np.pi * x / n)
    a = np.maximum(a, 0)
    sa = a.sum()
    if sa > 1e-10:
        inits['multi_cos'] = a / sa

    # S-shaped (sigmoid-like)
    a = 1.0 / (1.0 + np.exp(-4.0 * (x / n - 0.5)))
    sa = a.sum()
    if sa > 1e-10:
        inits['sigmoid'] = a / sa

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: LP + line search from multiple initializations
    Phase 2: Deep LP refinement on best
    Phase 3: Subgradient polishing
    Phase 4: Perturbation + re-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")

    # Phase 1: LP + Line Search from multiple initializations
    print("=== Phase 1: LP + Line Search Exploration ===")
    t1_start = time.monotonic()

    n = 512
    inits = generate_initializations(n, parent, rng)
    init_names = list(inits.keys())
    priority = ['parent', 'uniform', 'random', 'cos_1', 'gauss', 'multi_cos',
                'sigmoid', 'cos_2', 'cos_3']
    init_names = [nm for nm in priority if nm in inits] + \
                 [nm for nm in init_names if nm not in priority]

    # Allocate time: give more time to parent and uniform
    n_inits = len(init_names)
    total_time_p1 = 700
    # Weighted allocation
    weights = []
    for nm in init_names:
        if nm in ('parent', 'uniform'):
            weights.append(3.0)
        elif nm in ('random', 'cos_1', 'gauss'):
            weights.append(2.0)
        else:
            weights.append(1.0)
    w_sum = sum(weights)
    time_allocs = [total_time_p1 * w / w_sum for w in weights]

    # Track results from each init
    phase1_results = []
    for i, name in enumerate(init_names):
        remaining = deadline - time.monotonic() - 600
        if remaining < 30:
            break
        t_alloc = min(time_allocs[i], remaining)
        if t_alloc < 10:
            break

        print(f"  Init '{name}' (t={t_alloc:.0f}s)...")
        a_opt, obj_opt = optimize_lp_linesearch(
            inits[name], time_limit=t_alloc, max_iters=500,
            use_min_peak_every=5, perturb_every=35)
        phase1_results.append((obj_opt, a_opt.copy()))
        try_update(a_opt)

    # Sort phase 1 results
    phase1_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # Phase 2: Deep LP refinement on top results
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 500
        # Refine the best and second best
        top_candidates = [best_sequence]
        for obj, a_cand in phase1_results[1:3]:
            if not np.allclose(a_cand, best_sequence):
                top_candidates.append(a_cand)

        for ci, cand in enumerate(top_candidates):
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(
                max(80.0, (phase2_deadline - time.monotonic()) / (len(top_candidates) - ci) - 5),
                200.0)
            if t_alloc < 15:
                break
            print(f"  Deep refinement candidate {ci} (t={t_alloc:.0f}s)...")
            a_opt, obj_opt = optimize_lp_linesearch(
                cand, time_limit=t_alloc, max_iters=500,
                use_min_peak_every=3, perturb_every=30, perturb_scale=0.003)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # Phase 3: Subgradient polishing
    print("=== Phase 3: Subgradient Polishing ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 300
        n_best = len(best_sequence)
        for lr in [0.015, 0.008, 0.025]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(80.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 15:
                break
            print(f"  Subgrad lr={lr} (t={t_alloc:.0f}s)...")
            a_opt, obj_opt = subgradient_refine(
                best_sequence, time_limit=t_alloc, lr_init=lr,
                momentum=0.91, top_k=max(20, n_best // 20),
                temp=0.0008, perturb_every=600)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: Perturbation + LP re-optimization
    print("=== Phase 4: Perturbation + Re-optimization ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 100
        n_best = len(best_sequence)
        rng4 = np.random.default_rng(777)
        for scale in [0.003, 0.01, 0.005, 0.015, 0.002, 0.02]:
            if time.monotonic() > phase4_deadline:
                break
            noise = rng4.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(40.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            a_opt, obj_opt = optimize_lp_linesearch(
                a_pert, time_limit=t_alloc, max_iters=200,
                use_min_peak_every=4, perturb_every=25)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Final focused subgradient on best
    print("=== Phase 5: Final Polish ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        t_alloc = max(30.0, deadline - time.monotonic() - 30)
        if t_alloc > 30:
            print(f"  Final subgrad (t={t_alloc:.0f}s)...")
            a_opt, obj_opt = subgradient_refine(
                best_sequence, time_limit=t_alloc, lr_init=0.005,
                momentum=0.93, top_k=len(best_sequence),
                temp=0.0003, perturb_every=1000, perturb_scale=0.002)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        if final_score < best_score + 1e-10:
            pass

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
