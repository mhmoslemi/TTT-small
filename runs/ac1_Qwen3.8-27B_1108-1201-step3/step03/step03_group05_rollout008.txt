    cross = np.fft.irfft(P * np.conj(A), fft_size)
    return 2.0 * cross[:n]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mwu_step(a, grad, eta):
    """Multiplicative weight update: a_new[i] = a[i] * exp(-eta * grad[i]), normalized."""
    log_a = np.full(len(a), -np.inf)
    positive = a > 1e-300
    log_a[positive] = np.log(a[positive])
    log_a_new = log_a - eta * grad
    max_log = np.max(log_a_new[positive])
    a_new = np.zeros(len(a))
    a_new[positive] = np.exp(log_a_new[positive] - max_log)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def backtracking_line_search(a, grad, lr, max_iter=20):
    """Backtracking: find step size t = lr * alpha^k such that objective decreases."""
    n = len(a)
    a_cur = a
    conv_cur = fast_convolve(a_cur)
    obj_cur = conv_cur.max()
    
    for k in range(max_iter):
        t = lr * (0.5 ** k)
        a_try = project_simplex(a - t * grad)
        conv_try = fast_convolve(a_try)
        obj_try = conv_try.max()
        if obj_try <= obj_cur - 1e-12:
            return a_try, t
    return a_cur, 0.0

def optimize_mwu(a0, time_limit, eta_init=0.5, beta_init=30.0, beta_end=300.0, seed=0):
    """MWU-based optimization with adaptive parameters."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    eta = eta_init
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        
        conv = fast_convolve(a)
        grad = smooth_gradient(a, conv, beta)
        
        # MWU update
        a_new = mwu_step(a, grad, eta)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        elif obj_new < conv.max() - 1e-14:
            a = a_new
            no_improve = 0
        else:

        if no_improve >= 50:
            eta = eta * 0.5

                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                no_improve = 0
                eta = eta_init * (1.0 - 0.5 * frac)
        
        # Gradually decrease eta
        eta = eta_init * (1.0 - 0.7 * frac) + 0.01
    
    return best_a, best_obj

def peak_equalize(a0, time_limit, seed=0):

    rng = np.random.default_rng(seed)

    stuck = 0

        conv = fast_convolve(a)
        max_val = conv.max()

        num_peaks = min(5, 2 * n - 1)
        peak_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]
        peak_idx = peak_idx[conv[peak_idx] > max_val * 0.9]
        
        improved = False
        for k in peak_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            if i_start > i_end:
                continue
            
            # Contributions to this peak
            pairs = [(i, k - i) for i in range(i_start, i_end + 1)]
            if not pairs:
                continue
            
            # Find the highest contributing pair
            contribs = [(a[i] * a[k - i], i, k - i) for i in range(i_start, i_end + 1)]
            contribs.sort(reverse=True)
            
            # Try to reduce the top contributor by moving mass
            for contrib, i1, i2 in contribs[:3]:
                if contrib < 1e-15:
                    break
                # Move mass from the larger of a[i1], a[i2] to a random low element
                if a[i1] >= a[i2]:
                    src = i1
                else:
                    src = i2
                
                # Find a low element to move to
                # Prefer elements that don't contribute to any of the top peaks
                candidates = []
                for j in range(n):
                    if j == src:
                        continue
                    # Check if j contributes to any peak
                    total_contrib = 0.0
                    for pk in peak_idx:
                        pk = int(pk)
                        j_start = max(0, pk - n + 1)
                        j_end = min(n - 1, pk)
                        if j_start <= j <= j_end and 0 <= pk - j <= n - 1:
                            total_contrib += a[j] * a[pk - j]
                    candidates.append((total_contrib, j))
                candidates.sort()
                
                for _, dst in candidates[:5]:
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-10:
                            break
                        a_try = a.copy()
                        a_try[src] -= amt
                        a_try[dst] += amt
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj - 1e-12:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stuck += 1
            # Random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = a[i] * 0.05
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                if a_try[i] >= 0:
                    a = a_try
            if stuck > 200:
                break
        else:
            stuck = 0
    
    return best_a, best_obj

def generate_inits(n, parent, rng):
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    rc = np.maximum(rc, 1e-8)
    rc /= rc.sum()
    inits['cosine'] = rc
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Parent
    if parent is not None:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
        if len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
    
    return inits

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Multiplicative weight updates (MWU) with adaptive temperature
2. Nesterov gradient descent with backtracking line search
3. Peak-equalization coordinate descent targeting top convolution peaks
4. Multi-length, multi-initialization diverse search with adaptive time allocation
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv)/beta w.r.t. a.
    Returns grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i.
    """
    n = len(a)
    shifted = beta * conv - beta * conv.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p /= p_sum
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(P * np.conj(A), fft_size)
    return 2.0 * cross[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mwu_update(a, grad, eta):

    if mx < -700:
        return a.copy()
    a_new = np.exp(log_a_new - mx)
    s = a_new.sum()
    if s < 1e-300:
        a_new = np.ones(len(a)) / len(a)
    else:
        a_new /= s
    return a_new

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    eta = eta_init
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * (frac ** 0.7)

        conv = fast_convolve(a)
        grad = smooth_gradient(a, conv, beta)
        eta_cur = eta_init * (1.0 - 0.75 * frac) + 0.02
        a_new = mwu_update(a, grad, eta_cur)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        if obj_new <= conv.max() + 1e-10:
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        total_iters += 1

        if no_improve >= 80:
            noise_scale = 0.005 * (1.0 + 0.3 * np.sin(total_iters * 0.007))
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            no_improve = 0
            if total_iters > 5000:
                break

    return best_a, best_obj


def optimize_grad_bt(a0, time_limit, lr_init=0.04, beta_init=25.0, beta_end=300.0,
                     momentum=0.92, seed=0):
    """Nesterov gradient descent with backtracking line search."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * (frac ** 0.8)

        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smooth_gradient(y, conv_y, beta)

        lr = lr_init * (1.0 - 0.85 * frac) + 0.002
        obj_cur = conv.max() if iters > 0 else fast_convolve(a).max()

        # Backtracking
        a_new = None
        for k in range(10):
            t = lr * (0.5 ** k)
            trial = project_simplex(a - t * grad)
            conv_trial = fast_convolve(trial)
            if conv_trial.max() <= obj_cur + 1e-12:
                a_new = trial
                break
        if a_new is None:
            a_new = project_simplex(a - 0.001 * grad)

        v = momentum * v - lr * grad
        a = a_new
        conv = fast_convolve(a)
        obj = conv.max()
        iters += 1

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 150:
            noise = rng.standard_normal(n) * 0.008
            a = project_simplex(a + noise)
            v *= 0.2
            no_improve = 0
            if iters > 8000:
                break

    return best_a, best_obj


def peak_equalize(a0, time_limit, seed=0):
    """Targeted peak reduction via mass redistribution considering all top peaks."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()

        num_top = min(8, 2 * n - 1)
        top_idx = np.argpartition(conv, -num_top)[-num_top:]
        top_idx = top_idx[conv[top_idx] > max_val * 0.85]

        if len(top_idx) == 0:
            break

        improved = False
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue

            # Find highest contributing pair
            contribs = []
            for i in range(i_start, i_end + 1):
                c = a[i] * a[k - i]
                contribs.append((c, i, k - i))
            contribs.sort(reverse=True)

            for contrib, i1, i2 in contribs[:4]:
                if contrib < 1e-15:
                    break
                src = i1 if a[i1] >= a[i2] else i2

                # Find destination: minimize total contribution to top peaks
                best_dst = -1
                best_dst_score = float('inf')
                for _ in range(20):
                    j = int(rng.integers(0, n))
                    if j == src:
                        continue
                    total = 0.0
                    for pk in top_idx:
                        pk = int(pk)
                        ps = max(0, pk - n + 1)
                        pe = min(n - 1, pk)
                        if ps <= j <= pe and 0 <= pk - j <= n - 1:
                            total += a[j] * a[pk - j]
                    if total < best_dst_score:
                        best_dst_score = total
                        best_dst = j

                if best_dst < 0:
                    continue

                for frac in [0.03, 0.07, 0.12, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        break
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[best_dst] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-14:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stuck += 1
            # Perturb
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-6:
                amt = a[i] * 0.04
                a[i] -= amt
                a[j] += amt
            if stuck > 300:
                break
        else:
            stuck = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=0):
    """Cyclic coordinate descent: optimize one element at a time via bisection on 1D subproblem."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    order = rng.permutation(n)
    oi = 0

    while time.monotonic() - t0 < time_limit:
        j = int(order[oi % n])
        oi += 1

        # Optimize a[j]: redistribute a[j] among all elements
        # Try moving a[j] to other positions
        cur_val = a[j]
        if cur_val < 1e-10:
            continue

        # For each potential transfer, compute effect on max conv
        # Quick heuristic: try transferring to a few candidate positions
        candidates = []
        for _ in range(10):
            dst = int(rng.integers(0, n))
            if dst != j:
                candidates.append(dst)
        # Also try spreading to neighbors
        for d in [-2, -1, 1, 2]:
            nd = j + d
            if 0 <= nd < n and nd != j:
                candidates.append(nd)

        for dst in candidates[:8]:
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                amt = cur_val * frac
                a_try = a.copy()
                a_try[j] = cur_val - amt
                a_try[dst] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-14:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.4 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_wide'] = g2

    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    rc = np.maximum(rc, 1e-8)
    rc /= rc.sum()
    inits['cosine'] = rc

    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 1e-8)
    tri /= tri.sum()
    inits['triangle'] = tri

    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 1.5)

    # Tapered: high in middle, low at edges (opposite of cosine)
    tap = (x + 1.0) * (n - x + 1.0)
    tap /= tap.sum()
    inits['tapered'] = tap

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            p = np.maximum(parent[:n], 0.0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0.0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
        if len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0.0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: MWU scan -> gradient+backtracking deep opt ->
    peak equalization -> coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Quick scan with MWU
    print("=== Phase 1: Quick Scan (MWU) ===", flush=True)
    lengths = [192, 256, 320, 384, 448, 512]
    scan_results = []

    t_p1_start = time.monotonic()
    p1_budget = 350.0

    for n in lengths:
        if time.monotonic() > deadline - 500:
            break
        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'parent_ds', 'parent_up', 'gauss', 'cosine',
                    'uniform', 'gauss_wide', 'rand_a', 'triangle']
        names = [nm for nm in priority if nm in inits]
        names = names[:5]

        time_rem = p1_budget - (time.monotonic() - t_p1_start)
        per_init = max(4.0, time_rem / max(1, len(names) * (len(lengths) - lengths.index(n))))

        for name in names:
            if time.monotonic() > deadline - 450:
                break
            t_alloc = min(per_init, 25.0)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, obj_opt = optimize_mwu(a0, t_alloc, eta_init=0.35,
                                          beta_init=20, beta_end=120,
                                          seed=int(rng.integers(0, 50000)))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add one random restart at best length
    if best_sequence is not None:
        best_n = len(best_sequence)
        top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.5)))

    remaining = deadline - time.monotonic() - 150.0
    num_configs = len(top_configs)
    per_config = max(20.0, remaining / max(1, num_configs))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(per_config, 300.0)
        if t_alloc < 10:
            break

        print(f"  Config {i}/n={n}: alloc={t_alloc:.0f}s", flush=True)

        # Sub-phase A: Gradient with backtracking (45%)
        t_a = t_alloc * 0.45
        a_a, _ = optimize_grad_bt(a0, t_a, lr_init=0.035,
                                  beta_init=30, beta_end=250,
                                  momentum=0.92, seed=int(rng.integers(0, 99999)))
        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    GradBT: {score_a:.6f}", flush=True)

        # Sub-phase B: MWU refinement (25%)
        t_b = t_alloc * 0.25
        if t_b > 5:
            a_b, _ = optimize_mwu(a_a, t_b, eta_init=0.15,
                                  beta_init=80, beta_end=400,
                                  seed=int(rng.integers(0, 99999)))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    MWU: {score_b:.6f}", flush=True)
            a_current = a_b
        else:
            a_current = a_a

        # Sub-phase C: Peak equalization (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_c, _ = peak_equalize(a_current, t_c, seed=int(rng.integers(0, 99999)))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    PeakEq: {score_c:.6f}", flush=True)
            a_current = a_c

        # Sub-phase D: Fine gradient (10%)
        t_d = t_alloc * 0.10
        if t_d > 5:
            a_d, _ = optimize_grad_bt(a_current, t_d, lr_init=0.01,
                                      beta_init=300, beta_end=800,
                                      momentum=0.95, seed=int(rng.integers(0, 99999)))
            score_d = float(evaluate_sequence(a_d.tolist()))
            if np.isfinite(score_d) and score_d < best_score:
                best_score = score_d
                best_sequence = a_d.copy()
                print(f"    FineGrad: {score_d:.6f}", flush=True)

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0

        # Peak equalization
        t_ps = remaining * 0.35
        if t_ps > 8:
            a_ps, _ = peak_equalize(best_sequence, t_ps, seed=111)
            score_ps = float(evaluate_sequence(a_ps.tolist()))
            if np.isfinite(score_ps) and score_ps < best_score:
                best_score = score_ps
                best_sequence = a_ps.copy()
                print(f"  Polish_PE: {best_score:.6f}", flush=True)

        # Very fine gradient
        remaining = deadline - time.monotonic() - 10.0
        t_fg = remaining * 0.30
        if t_fg > 5:
            a_fg, _ = optimize_grad_bt(best_sequence, t_fg, lr_init=0.004,
                                       beta_init=500, beta_end=1200,
                                       momentum=0.96, seed=222)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  Polish_G: {best_score:.6f}", flush=True)

        # Coordinate descent
        remaining = deadline - time.monotonic() - 10.0
        t_cd = remaining * 0.25
        if t_cd > 5:
            a_cd, _ = coordinate_descent(best_sequence, t_cd, seed=333)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  Polish_CD: {best_score:.6f}", flush=True)

        # Final peak shave
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_fs, _ = peak_equalize(best_sequence, remaining, seed=444)
            score_fs = float(evaluate_sequence(a_fs.tolist()))
            if np.isfinite(score_fs) and score_fs < best_score:
                best_score = score_fs
                best_sequence = a_fs.copy()
                print(f"  Polish_Final: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
