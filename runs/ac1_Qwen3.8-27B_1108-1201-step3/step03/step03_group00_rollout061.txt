def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Hybrid optimizer for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Combines L_p smooth approximation with Adam, fast coordinate descent,
LP-based Frank-Wolfe, and subgradient polishing. Uses multi-length
exploration and diverse initializations to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(f):
    """Compute 2n*max(conv(f,f)) for f with sum=1, f>=0."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def compute_lp_p_gradient(f, conv, p):
    """
    Compute gradient of sum_k conv[k]^p with respect to f.
    grad[i] = 2p * sum_k f[k-i] * conv[k]^(p-1)
    (with f[k]=0 for k outside [0,n-1])
    """
    n = len(f)
    w = p * np.power(np.maximum(conv, 0.0), p - 1.0)
    # grad[i] = 2p * sum_j f[j] * w[j+i] for j=0..n-1
    # Vectorized: for each i, dot(f, w[i:i+n])
    grad = np.zeros(n)
    for i in range(n):
        jmax = min(n, len(w) - i)
        grad[i] = 2.0 * p * np.dot(f[:jmax], w[i:i + jmax])
    return grad


def adam_lp_p_optimize(f_init, p, time_limit, lr=0.05, beta1=0.9, beta2=0.999,
                       eps=1e-8, p_schedule=None):
    """
    Adam optimizer for L_p smooth approximation of the objective.
    Minimizes sum_k conv[k]^p subject to f>=0, sum(f)=1.
    """
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    conv = fast_convolve(f)
    best_obj = objective_normalized(f)
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t_count += 1

        # Update p if using schedule
        if p_schedule is not None and len(p_schedule) > 1:
            frac = (t_count) / max(1, int(time_limit / 0.001))
            idx = min(int(frac * (len(p_schedule) - 1)), len(p_schedule) - 1)
            p = p_schedule[idx]

        # Compute gradient
        conv = fast_convolve(f)
        grad = compute_lp_p_gradient(f, conv, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        f = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        f = project_simplex(f)

        # Evaluate
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate decay
        cur_lr = lr * (1.0 / (1.0 + 0.001 * t_count))

        # Restart with perturbation if stuck
        if no_improve > 500:
            rng_local = np.random.default_rng(t_count)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_f, best_obj


def coord_descent(f_init, max_sweeps=100, verbose=False):
    """
    Coordinate descent with golden section search.
    Minimizes 2n*max(conv(f,f))/(sum(f))^2 via 1D subproblems.
    """
    f = np.maximum(f_init.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    else:
        f = f / s
        s = 1.0

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            x_old = f[i]
            if x_old < 1e-15:
                continue

            # Precompute unchanged max
            # Changed positions: k from i to i+n-1 (where 0<=k-i<n)
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unchanged: k in [0, k_start-1] and [k_end+1, 2n-2]
            C = 0.0
            if k_start > 0:
                C = max(C, conv[:k_start].max())
            if k_end + 1 < 2 * n - 1:
                C = max(C, conv[k_end + 1:].max())

            # Precompute changed part coefficients
            fki = f[np.arange(k_start, k_end + 1) - i]  # f[k-i] for changed k
            base_changed = conv[k_start:k_end + 1].copy()
            # Remove old contribution
            if 2 * i >= k_start and 2 * i <= k_end:
                idx_2i = 2 * i - k_start
                base_changed[idx_2i] -= x_old * x_old
            else:
                base_changed -= 2.0 * x_old * fki
            # Now base_changed[k] is conv[k] minus contribution of x_old
            # New conv at changed pos x: base_changed + 2*x*fki (linear) or base_changed[2i]+x^2

            is_2i_in_range = (2 * i >= k_start and 2 * i <= k_end)
            idx_2i_local = 2 * i - k_start if is_2i_in_range else -1

            # Define objective at x (golden section search target)
            def _obj_at_x(x):
                S_new = s - x_old + x
                if S_new < 1e-15:
                    return np.inf
                # Changed values
                changed = base_changed + 2.0 * x * fki
                if is_2i_in_range:
                    changed[idx_2i_local] = base_changed[idx_2i_local] + x * x
                max_changed = changed.max()
                new_max = max(C, max_changed)
                if new_max < 0:
                    new_max = 0.0
                return 2.0 * n * new_max / (S_new * S_new)

            # Golden section search on [0, x_upper]
            x_upper = max(2.0 * (s - x_old) / max(1, n - 1), x_old * 3.0, 1.0 / n * 5)
            a = 0.0
            b = x_upper
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = _obj_at_x(c)
            fd = _obj_at_x(d)

            for _ in range(35):
                if b - a < 1e-13 * max(1.0, x_upper):
                    break
                if fc <= fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = _obj_at_x(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = _obj_at_x(d)

            x_opt = (a + b) / 2.0
            f_obj = _obj_at_x(x_opt)

            if f_obj < best_obj - 1e-12:
                # Update
                changed_new = base_changed + 2.0 * x_opt * fki
                if is_2i_in_range:
                    changed_new[idx_2i_local] = base_changed[idx_2i_local] + x_opt * x_opt
                conv[k_start:k_end + 1] = changed_new
                f[i] = x_opt
                s = s - x_old + x_opt
                best_obj = f_obj
                best_f = f.copy()
                improved = True

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")
        if not improved:
            break

    s_final = best_f.sum()
    if s_final > 1e-15:
        best_f = best_f / s_final
    return best_f


def solve_lp_simple(f, M):
    """Simple LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # For k=0..2n-2, the set of j where 0<=k-j<n is j in [max(0,k-n+1), min(k,n-1)]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] * g[j] appears in conv at position k where k-j is in [0,n-1]
        # i.e., k in [j, j+n-1]
        # A_ub[k, j] += f[j] for k in [j, j+n-1]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 5000:
            break

        g = solve_lp_simple(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
                no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 50:
            rng_local = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_refine(f0, time_limit, lr=0.003, mom=0.93, top_k=20):
    """Subgradient refinement with multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    temp = 0.0001

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        temp *= 1.004

        if no_improve > 300:
            rng_local = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Gaussian center
    sigma = 0.2 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits.append(g / g.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c += 0.01
    inits.append(c / c.sum())

    # Two bumps
    b1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits.append((b1 + b2) / (b1 + b2).sum())

    # Power law
    pw = 1.0 / (x + 1.0) ** 0.5
    inits.append(pw / pw.sum())

    # Symmetric random
    half = n // 2
    rh = rng.dirichlet(np.ones(half) * 0.5)
    sym = np.zeros(n)
    sym[:half] = rh
    sym[half:] = rh[::-1] if n % 2 == 0 else np.concatenate([rh[::-1], [0]])
    if n % 2 == 1:
        sym = np.concatenate([rh, rh[::-1][:-1]])
    sym = np.maximum(sym, 0)
    ss = sym.sum()
    if ss > 1e-10:
        inits.append(sym / ss)

    # From parent (resampled)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        lengths = [len(parent), 384, 256, 512, 192]
    else:
        lengths = [384, 256, 512, 128]
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 32 <= L <= 2048:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths

    print(f"=== Phase 1: L_p Smooth Optimization ===")
    print(f"Lengths to try: {lengths}")

    # Phase 1: L_p smooth optimization with Adam
    # Allocate time: more time for parent length
    total_time_p1 = min(800, (deadline - time.monotonic()) * 0.5)
    n_lengths = len(lengths)

    for li, n in enumerate(lengths):
        time_remaining = deadline - time.monotonic() - 200
        if time_remaining < 30:
            break

        # Time allocation: more for parent
        if parent is not None and n == len(parent):
            t_alloc = min(250, time_remaining * 0.4)
        else:
            t_alloc = min(100, time_remaining / max(1, n_lengths - li))

        if t_alloc < 15:
            continue

        print(f"\n  Length n={n}, time={t_alloc:.0f}s")

        inits = generate_inits(n, parent, rng)

        # Try multiple initializations with different p values
        p_values = [8, 12, 16]
        time_per_config = max(5.0, t_alloc / (len(inits) * len(p_values)))

        for p in p_values:
            for ii, f_init in enumerate(inits):
                t_rem = deadline - time.monotonic() - 150
                if t_rem < 5:
                    break
                t_this = min(time_per_config, t_rem)
                if t_this < 3:
                    continue

                a_opt, _ = adam_lp_p_optimize(f_init, p=p, time_limit=t_this,
                                               lr=0.03, beta1=0.9, beta2=0.999)
                improved = try_sequence(a_opt, best_state)
                if improved:
                    print(f"    p={p}, init={ii}: NEW BEST {best_state[0]:.6f}")

    print(f"\n  Phase 1 best: {best_state[0]:.6f}")

    if best_state[1] is None:
        best_state[1] = np.ones(384) / 384
        best_state[0] = float(evaluate_sequence(best_state[1].tolist()))

    # Phase 2: Coordinate descent on best
    print(f"\n=== Phase 2: Coordinate Descent Refinement ===")

    if best_state[1] is not None:
        n_best = len(best_state[1])
        time_remaining = deadline - time.monotonic() - 150
        t_cd = min(200, time_remaining * 0.3)

        if t_cd > 10:
            print(f"  CD on best (n={n_best}), t={t_cd:.0f}s")
            a_cd = coord_descent(best_state[1], max_sweeps=50, verbose=True)
            try_sequence(a_cd, best_state)

            # Try perturbations + CD
            for pert in [0.002, 0.005, 0.01]:
                t_rem = deadline - time.monotonic() - 100
                if t_rem < 10:
                    break
                f_pert = project_simplex(best_state[1] + rng.standard_normal(n_best) * pert)
                a_cd2 = coord_descent(f_pert, max_sweeps=30)
                try_sequence(a_cd2, best_state)

    print(f"  Phase 2 best: {best_state[0]:.6f}")

    # Phase 3: LP-FW for global improvement
    print(f"\n=== Phase 3: LP-FW Global Search ===")

    if best_state[1] is not None:
        n_best = len(best_state[1])
        time_remaining = deadline - time.monotonic() - 100
        t_fw = min(300, time_remaining * 0.5)

        if t_fw > 20:
            # Run LP-FW from best
            print(f"  FW-LP on best, t={t_fw/2:.0f}s")
            a_fw, _ = fw_lp_optimize(best_state[1], time_limit=t_fw / 2)
            try_sequence(a_fw, best_state)

            # Run from perturbed best
            t_rem = deadline - time.monotonic() - 80
            if t_rem > 20:
                f_pert = project_simplex(best_state[1] + rng.standard_normal(n_best) * 0.01)
                a_fw2, _ = fw_lp_optimize(f_pert, time_limit=min(t_fw / 3, t_rem))
                try_sequence(a_fw2, best_state)

    print(f"  Phase 3 best: {best_state[0]:.6f}")

    # Phase 4: Final subgradient polish
    print(f"\n=== Phase 4: Subgradient Polish ===")

    if best_state[1] is not None:
        n_best = len(best_state[1])
        time_remaining = deadline - time.monotonic() - 30

        if time_remaining > 15:
            t_sg = min(time_remaining, 80)
            print(f"  SG polish, t={t_sg:.0f}s")
            a_sg, _ = sg_refine(best_state[1], time_limit=t_sg,
                                lr=0.002, mom=0.94, top_k=max(5, n_best // 8))
            try_sequence(a_sg, best_state)

            # Final CD on polished result
            t_rem = deadline - time.monotonic() - 15
            if t_rem > 5:
                a_final_cd = coord_descent(best_state[1], max_sweeps=20)
                try_sequence(a_final_cd, best_state)

    # Phase 5: Quick exploration of different lengths with best structure
    print(f"\n=== Phase 5: Cross-length Exploration ===")

    if best_state[1] is not None:
        n_best = len(best_state[1])
        time_remaining = deadline - time.monotonic() - 20

        # Try resampling to slightly different lengths
        for n_try in [n_best - 32, n_best + 32, n_best * 2 // 3]:
            if time_remaining < 10 or n_try < 32 or n_try > 2048:
                continue
            t_rem = deadline - time.monotonic() - 15
            if t_rem < 5:
                break

            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_state[1])
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps < 1e-10:
                continue
            px = px / ps

            # Quick Adam + CD
            a_opt, _ = adam_lp_p_optimize(px, p=12, time_limit=min(20, t_rem * 0.5), lr=0.02)
            a_cd = coord_descent(a_opt, max_sweeps=15)
            try_sequence(a_cd, best_state)
            time_remaining = deadline - time.monotonic() - 20

    # Final verification
    print(f"\n=== Final Results ===")
    if best_state[1] is not None:
        final_score = float(evaluate_sequence(best_state[1].tolist()))
        if np.isfinite(final_score):
            best_state[0] = final_score
        print(f"FINAL SCORE: {best_state[0]:.8f} (n={len(best_state[1])})")
    else:
        best_state[1] = np.ones(384) / 384
        best_state[0] = float(evaluate_sequence(best_state[1].tolist()))
        print(f"FINAL SCORE (fallback): {best_state[0]:.8f}")

    return [float(x) for x in best_state[1]]
