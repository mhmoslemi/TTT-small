def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def lp_step(a, M, top_k=None, conv=None):
    """Single LP step: find b maximizing sum(b) s.t. b>=0, conv(a,b)[k]<=M."""
    from scipy.optimize import linprog
    
    n = len(a)
    num_k = 2 * n - 1
    
    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        A_ub = np.zeros((top_k, n))
        for ii, k in enumerate(peaks):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            A_ub[ii, idx] = a[k - idx]
        b_ub = np.full(top_k, M)
    else:
        A_ub = np.zeros((num_k, n))
        for k in range(num_k):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            A_ub[k, idx] = a[k - idx]
        b_ub = np.full(num_k, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp.copy()
    inits['ramp_dec'] = ramp[::-1].copy()
    
    # Edges (bimodal)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g
    
    # Gaussian narrow
    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.1 * n)) ** 2)
    g2 /= g2.sum()
    inits['gauss_narrow'] = g2
    
    # Cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines proximal gradient with smooth max, LP-based direction finding with
adaptive bounds, hybrid alternating optimization, and diverse restarts.
Key improvements: adaptive LP constraints, backtracking line search,
multi-scale beta scheduling, and active-set-aware optimization.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def smooth_obj_and_grad(a, beta):
    """
    Compute smooth objective logsumexp(beta * conv(a,a)) and its gradient.
    Returns (smooth_obj, gradient, conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    max_val = scaled.max()
    w = np.exp(scaled - max_val)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # Gradient of logsumexp(beta * conv(a,a)) w.r.t. a
    # = beta * sum_k w_k * (a[i-k] + a[i+k]) for valid indices
    # Use FFT for efficient computation
    conv_len = len(conv)
    fft_size2 = 1 << (conv_len + n - 1 - 1).bit_length()
    W2 = np.fft.fft(w, fft_size2)
    A2 = np.fft.fft(a, fft_size2)

    # Term 1: (w * a)[i] = sum_j w[j] * a[i-j] (linear conv, take first n)
    conv_wa = np.fft.ifft(W2 * A2).real[:n]

    # Term 2: cross-correlation sum_j a[j] * w[j-i] = ifft(conj(W) * A)
    corr = np.fft.ifft(np.conj(W2) * A2).real[:n]

    grad = beta * (conv_wa + corr)
    smooth_obj = max_val + np.log(w_sum)

    return smooth_obj, grad, conv


def build_lp_matrix_row(a, k):
    """Build one row of the LP constraint matrix for lag k."""
    n = len(a)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1)
    row = np.zeros(n)
    row[idx] = a[k - idx]
    return row


def lp_step(a, M, conv, top_k=None, adaptive_bounds=False):
    """
    LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= bound_k for selected k.
    If adaptive_bounds=True, use tighter bounds for larger conv values.
    """
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k:
        # Select top_k lags by conv value
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # Sort by conv value descending
        peaks = peaks[np.argsort(conv[peaks])[::-1]]
    else:
        peaks = np.arange(num_k)

    num_constraints = len(peaks)
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)

    for ii, k in enumerate(peaks):
        A_ub[ii] = build_lp_matrix_row(a, k)
        if adaptive_bounds:
            # Tighter for peaks, looser for lower lags
            frac = ii / max(1, num_constraints - 1)
            b_ub[ii] = M * (1.0 + frac * 0.3)
        else:
            b_ub[ii] = M

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iterate(a0, time_limit, step_init=0.8, top_k=None, relax=1.0,
               adaptive=False, seed=0, verbose=False):
    """LP-based block coordinate optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        b = lp_step(a, M * relax, conv, top_k=top_k, adaptive_bounds=adaptive)
        if b is None:
            # Retry with fewer constraints
            b = lp_step(a, M * relax, conv, top_k=min(30, n // 5))
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.85
            else:
                if no_improve > 8:
                    noise = rng.standard_normal(n) * 0.005
                    a = project_simplex(a + noise)
                    step = step_init * 0.6
                    no_improve = 0
                else:
                    step = max(0.01, step * 0.9)
                    continue

        if no_improve >= 100:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            step = step_init * 0.7
            no_improve = 0

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    return best_a, best_obj


def pg_optimize(a0, time_limit, beta_init=100.0, lr_init=0.1,
                beta_growth=1.002, max_backtrack=6, seed=0, verbose=False):
    """
    Proximal gradient descent on the simplex with smooth max approximation.
    Uses backtracking line search and adaptive beta.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient
        _, grad, conv = smooth_obj_and_grad(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            # At a critical point of smooth approx, perturb
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            continue

        # Backtracking line search
        lr_try = lr
        obj_accepted = None
        a_accepted = None

        for back in range(max_backtrack):
            a_try = project_simplex(a - lr_try * grad)
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < 2.0 * n * conv.max() * 1.001:
                obj_accepted = obj_try
                a_accepted = a_try
                break
            lr_try *= 0.5

        if a_accepted is None:
            # Accept the last try if it's not terrible
            a_try = project_simplex(a - lr_try * grad)
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj * 1.05:
                a_accepted = a_try
                obj_accepted = obj_try
            else:
                # Skip this step
                no_improve += 1
                if no_improve > 50:
                    noise = rng.standard_normal(n) * 0.005
                    a = project_simplex(a + noise)
                    lr = lr_init
                    beta = max(50.0, beta * 0.8)
                    no_improve = 0
                it += 1
                continue

        a = a_accepted
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.5)
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.9
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                lr = lr_init * 0.5
                beta = max(50.0, beta * 0.8)
                no_improve = 0

        beta = min(20000.0, beta * beta_growth)
        it += 1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG iter {it}: obj={best_obj:.6f}, beta={beta:.0f}, "
                  f"lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def hybrid_optimize(a0, time_limit, lp_step_size=0.7, lp_top_k=50,
                    pg_beta=500.0, pg_lr=0.03, adaptive_lp=False,
                    seed=0, verbose=False):
    """
    Alternating LP and proximal gradient steps.
    LP for large moves, PG for fine tuning.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = pg_beta
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # LP step (every other iteration for speed)
        if it % 2 == 0:
            conv = fast_convolve(a)
            M = conv.max()
            tk = lp_top_k if lp_top_k is not None else None
            b = lp_step(a, M, conv, top_k=tk, adaptive_bounds=adaptive_lp)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    a = (1.0 - lp_step_size) * a + lp_step_size * b
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s

        # PG step
        _, grad, conv = smooth_obj_and_grad(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            a = project_simplex(a - pg_lr * grad / g_norm)

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                no_improve = 0

        beta = min(10000.0, beta * 1.001)
        it += 1

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid iter {it}: obj={best_obj:.6f}, beta={beta:.0f}, t={elapsed:.1f}s")

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n, normalized to sum=1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp.copy()
    inits['ramp_dec'] = ramp[::-1].copy()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Three peaks
    tri = np.zeros(n)
    for center in [n // 4, n // 2, 3 * n // 4]:
        width = max(1, n // 8)
        tri[np.maximum(0, center - width):np.minimum(n, center + width)] += 1.0
    tri /= tri.sum()
    inits['three_peaks'] = tri

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[1::2] = 0.3
    alt /= alt.sum()
    inits['alternating'] = alt

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        parent = np.asarray(parent, dtype=np.float64)
        parent = np.maximum(parent, 0)
        if parent.size >= n:
            pp = parent[:n].copy()
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
            # Also try a smoothed version
            from scipy.ndimage import uniform_filter1d
            pp_smooth = uniform_filter1d(parent[:n].copy(), size=5)
            pp_smooth = np.maximum(pp_smooth, 0)
            ps = pp_smooth.sum()
            if ps > 1e-10:
                inits['parent_smooth'] = pp_smooth / ps
        if parent.size < n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if parent.size > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def score_sequence(a):

                                beta_init=200.0, lr_init=0.08,
                                beta_growth=1.003, seed=2000 + ci)
        update_best(a_opt2, f"P2 PG c{ci}")

        # Hybrid from best of the two
        a_hyb_start = a_opt1 if score_sequence(a_opt1) < score_sequence(a_opt2) else a_opt2
        a_opt3, _ = hybrid_optimize(a_hyb_start, time_limit=t_hyb,
                                    lp_step_size=0.7,
                                    lp_top_k=max(40, n_c // 4),
                                    pg_beta=800.0, pg_lr=0.02,
                                    adaptive_lp=True, seed=3000 + ci)
        update_best(a_opt3, f"P2 Hybrid c{ci}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Intensive hybrid on best ==========
    print("=== Phase 3: Intensive hybrid ===")
    t_start3 = time.monotonic()
    phase3_time = 450.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()

        # Multiple hybrid configurations with different parameters
        configs = [
            {'lp_ss': 0.9, 'lp_tk': max(50, n_best // 4), 'pg_beta': 500, 'pg_lr': 0.03, 'adapt': True},
            {'lp_ss': 1.0, 'lp_tk': max(70, n_best // 3), 'pg_beta': 1500, 'pg_lr': 0.015, 'adapt': False},
            {'lp_ss': 0.5, 'lp_tk': None, 'pg_beta': 300, 'pg_lr': 0.05, 'adapt': True},
            {'lp_ss': 0.8, 'lp_tk': max(100, n_best // 2), 'pg_beta': 3000, 'pg_lr': 0.008, 'adapt': True},
            {'lp_ss': 0.6, 'lp_tk': max(30, n_best // 6), 'pg_beta': 100, 'pg_lr': 0.08, 'adapt': False},
        ]

        time_per_cfg = phase3_time / len(configs)

        for ci, cfg in enumerate(configs):
            if time.monotonic() - t_start3 > phase3_time - 30:
                break
            t_alloc = min(time_per_cfg, phase3_time - (time.monotonic() - t_start3) - 20)
            if t_alloc < 30:
                break

            a_opt, _ = hybrid_optimize(a, time_limit=t_alloc,
                                       lp_step_size=cfg['lp_ss'],
                                       lp_top_k=cfg['lp_tk'],
                                       pg_beta=cfg['pg_beta'],
                                       pg_lr=cfg['pg_lr'],
                                       adaptive_lp=cfg['adapt'],
                                       seed=5000 + ci)
            update_best(a_opt, f"P3 hyb c{ci}")

            # Also try pure PG from the result
            t_pg_remain = min(30.0, phase3_time - (time.monotonic() - t_start3) - 20)
            if t_pg_remain > 15:
                a_opt2, _ = pg_optimize(a_opt, time_limit=t_pg_remain,
                                        beta_init=5000.0, lr_init=0.005,
                                        beta_growth=1.001, seed=6000 + ci)
                update_best(a_opt2, f"P3 PG c{ci}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Try different lengths from best ==========
    print("=== Phase 4: Length exploration ===")
    t_start4 = time.monotonic()
    phase4_time = 150.0

    if best_sequence is not None and phase4_time > 30:
        n_best = len(best_sequence)
        # Try neighboring lengths
        neighbor_lengths = [n_best - 32, n_best + 32, n_best // 2, n_best * 2]
        neighbor_lengths = [nl for nl in neighbor_lengths if 64 <= nl <= 1024]
        neighbor_lengths = list(set(neighbor_lengths))[:3]

        time_per_nl = phase4_time / max(1, len(neighbor_lengths))

        for nl in neighbor_lengths:
            if time.monotonic() - t_start4 > phase4_time - 10:
                break
            # Resample best sequence to new length
            a_resamp = np.interp(np.linspace(0, 1, nl),
                                 np.linspace(0, 1, len(best_sequence)),
                                 best_sequence)
            a_resamp = np.maximum(a_resamp, 0)
            ps = a_resamp.sum()
            if ps > 1e-10:
                a_resamp = a_resamp / ps

            t_alloc = min(time_per_nl, phase4_time - (time.monotonic() - t_start4) - 10)
            if t_alloc < 20:
                break

            a_opt, _ = hybrid_optimize(a_resamp, time_limit=t_alloc,
                                       lp_step_size=0.8,
                                       lp_top_k=max(30, nl // 5),
                                       pg_beta=500.0, pg_lr=0.03,
                                       adaptive_lp=True, seed=7000 + nl)
            update_best(a_opt, f"P4 n={nl}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # ========== Phase 5: Final polish ==========
    print("=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()

    if best_sequence is not None and remaining > 30:
        n_best = len(best_sequence)

        # Very fine PG
        t_alloc = min(remaining * 0.3, 80.0)
        if t_alloc > 15:
            a_opt, _ = pg_optimize(best_sequence, time_limit=t_alloc,
                                   beta_init=10000.0, lr_init=0.003,
                                   beta_growth=1.0005, seed=9999)
            update_best(a_opt, "P5 fine PG")

        # Fine LP
        t_alloc = min(remaining * 0.25, 60.0)
        if t_alloc > 15:
            a_opt, _ = lp_iterate(best_sequence, time_limit=t_alloc,
                                  step_init=0.2, top_k=max(60, n_best // 3),
                                  relax=1.0, adaptive=True, seed=8888)
            update_best(a_opt, "P5 fine LP")

        # Tiny perturbations
        remaining2 = deadline - time.monotonic() - 20
        if remaining2 > 10:
            n_tries = int(min(remaining2 / 2, 40))
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                for scale in [0.0003, 0.0008, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    sc = score_sequence(a_pert)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_sequence = a_pert.copy()
                        print(f"  P5 perturb [{i}] s={scale}: {sc:.6f}")
                        break

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_sc = score_sequence(best_sequence)
    if not np.isfinite(final_sc):
        best_sequence = np.ones(128) / 128
        final_sc = score_sequence(best_sequence)
    best_score = min(best_score, final_sc)

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
