"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with adaptive
temperature, FFT-accelerated subgradients, and multi-phase optimization.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, R, W_vec, n):
    """Compute subgradient using cross-correlation."""
    # W_vec has length 2n-1
    grad = 2.0 * np.correlate(W_vec, a, mode='valid')
    return grad

def mirror_descent(a0, time_limit, lr_init=0.08, lr_decay=0.9995,
                   temp_init=0.05, temp_growth=1.01, temp_max=1.0,
                   top_k_init=5, top_k_max=100, 
                   restart_patience=300, noise_scale=0.01,
                   min_lr=0.0005):
    """Mirror descent (multiplicative weights) optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    top_k = top_k_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = R.max()
        if M < 1e-15:
            break
        
        # Build weight vector W
        k_sel = min(top_k, len(R))
        peak_vals = np.partition(R, -k_sel)[-k_sel:]
        peak_idx = np.where(np.isin(R, peak_vals))[0]
        # Actually, let's just get the top-k indices properly
        sorted_idx = np.argsort(R)[::-1][:k_sel]
        peak_vals = R[sorted_idx]
        
        # Softmax weights
        scaled = (peak_vals - peak_vals.max()) / (temp * M + 1e-15)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k_sel)
            w[0] = 1.0
        else:
            w /= w_sum
        
        # Build W vector
        W_vec = np.zeros(2 * n - 1)
        for ii in range(k_sel):
            if w[ii] > 1e-12:
                W_vec[int(sorted_idx[ii])] += w[ii]
        
        # Compute subgradient
        grad = compute_subgrad(a, R, W_vec, n)
        
        # Mirror descent update: a <- a * exp(-lr * grad / M)
        # Scale grad by M for scale invarience
        log_update = -lr * grad / (M + 1e-15)
        # Clip to avoid underflow/overflow
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:

        if cur_obj < best_obj - 1e-12:

def subgrad_project(a0, time_limit, lr_init=0.05, momentum=0.9,
                    top_k=20, temp=0.1):
    """Subgradient descent with projection onto simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = R.max()
        if M < 1e-15:
            break
        
        k_sel = min(top_k, len(R))
        sorted_idx = np.argsort(R)[::-1][:k_sel]
        peak_vals = R[sorted_idx]
        
        scaled = (peak_vals - peak_vals.max()) / (temp * M + 1e-15)
        w = np.exp(scaled)
        w /= w.sum()
        
        W_vec = np.zeros(2 * n - 1)
        for ii in range(k_sel):
            if w[ii] > 1e-12:
                W_vec[int(sorted_idx[ii])] += w[ii]
        
        grad = 2.0 * np.correlate(W_vec, a, mode='valid')
        
        v = momentum * v - lr * grad / (M + 1e-15)
        a_new = project_simplex(a + v)
        a = a_new
        
        R = fast_convolve(a)
        cur_obj = 2.0 * n * R.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        lr = max(0.001, lr * 0.999)
        if stall > 200:
            noise = np.random.default_rng().standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            R = fast_convolve(a)
            v = 0.5 * v
            stall = 0
            lr = max(0.002, lr_init * 0.5)
    
    return best_a, best_obj

def coordinate_polish(a_in, max_rounds=50, delta_frac=0.03):
    """Coordinate descent: move mass between pairs to reduce peak."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Find the current peak positions
        peak_pos = int(np.argmax(R))
        
        for i in range(n):
            if a[i] < 1e-8:
                continue
            delta = a[i] * delta_frac
            if delta < 1e-10:
                continue
            for j in range(n):
                if j == i:
                    continue
                a_test = a.copy()
                a_test[i] = a[i] - delta
                a_test[j] = a[j] + delta
                if a_test[i] < 0:
                    continue
                st = a_test.sum()
                if abs(st - s) > 1e-10:
                    a_test = a_test * (s / st)
                R_test = fast_convolve(a_test)
                obj_test = 2.0 * n * R_test.max()
                if obj_test < best_obj - 1e-12:
                    a = a_test
                    best_obj = obj_test
                    R = R_test
                    improved = True
        
        if not improved:
            break
    
    return a

def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Edge-heavy (mass at edges)
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Triangle (peak in middle)
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Alternating
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alt'] = alt / alt.sum()
    
    # Sine-weighted
    sine = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits['sine'] = sine / sine.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10 and len(p) > 0:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n), 
                                  np.linspace(0, 1, len(p)), p / ps)
                p_res = np.maximum(p_res, 0)
                inits['parent_res'] = p_res / p_res.sum()
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, W_vec):
    """Compute subgradient: grad[i] = 2 * sum_k W_vec[k] * a[k-i]."""
    return 2.0 * np.correlate(W_vec, a, mode='valid')


def build_W(R, top_k, temp, M):
    """Build weight vector from top-k peaks with softmax weighting."""
    k_sel = min(top_k, len(R))
    sorted_idx = np.argsort(R)[::-1][:k_sel]
    peak_vals = R[sorted_idx]
    scaled = (peak_vals - peak_vals.max()) / (temp * M + 1e-15)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(k_sel)
        w[0] = 1.0
    else:
        w /= w.sum
    W_vec = np.zeros(len(R))
    for ii in range(k_sel):
        if w[ii] > 1e-12:
            W_vec[int(sorted_idx[ii])] += w[ii]
    return W_vec


def mirror_descent(a0, time_limit, lr_init=0.08, lr_decay=0.9995,
                   temp_init=0.05, temp_growth=1.01, temp_max=1.5,
                   top_k_init=5, top_k_max=120,
                   restart_patience=400, noise_scale=0.015,
                   min_lr=0.0003):
    """Mirror descent (multiplicative weights) with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    top_k = top_k_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = R.max()
        if M < 1e-15:
            break

        W_vec = build_W(R, top_k, temp, M)
        grad = compute_subgrad(a, W_vec)

        # Multiplicative update
        log_update = -lr * grad / (M + 1e-15)
        log_update = np.clip(log_update, -50.0, 50.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = a.copy()
        else:
            a = a_new / s_new

        R = fast_convolve(a)
        cur_obj = 2.0 * n * R.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        lr = max(min_lr, lr * lr_decay)
        if stall % 80 == 0 and stall > 0:
            temp = min(temp_max, temp * temp_growth)
            top_k = min(top_k_max, int(top_k * 1.08) + 1)

        if stall >= restart_patience:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * noise_scale
            a = np.maximum(a + noise, 1e-30)
            a = a / a.sum()
            R = fast_convolve(a)
            lr = max(min_lr * 10, lr_init * 0.4)
            temp = temp_init
            top_k = top_k_init
            stall = 0

    return best_a, best_obj


def subgrad_project(a0, time_limit, lr_init=0.04, momentum=0.9,
                    top_k=20, temp=0.1, min_lr=0.001):
    """Subgradient descent with simplex projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    R = fast_convolve(a)
    best_obj = 2.0 * n * R.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = R.max()
        if M < 1e-15:
            break

        W_vec = build_W(R, top_k, temp, M)
        grad = compute_subgrad(a, W_vec)

        v = momentum * v - lr * grad / (M + 1e-15)
        a = project_simplex(a + v)

        R = fast_convolve(a)
        cur_obj = 2.0 * n * R.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        lr = max(min_lr, lr * 0.999)
        if stall > 300:
            noise = np.random.default_rng().standard_normal(n) * 0.008
            a = project_simplex(a + noise)
            R = fast_convolve(a)
            v = 0.5 * v
            stall = 0
            lr = max(min_lr * 5, lr_init * 0.5)

    return best_a, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None
