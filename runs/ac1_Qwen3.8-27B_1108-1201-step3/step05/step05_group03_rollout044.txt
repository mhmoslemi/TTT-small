import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_for_direction(f, M):
    """
    Solve: maximize sum(g) subject to f*g <= M, g >= 0.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_i f_i * g_{k-i} where i and k-i are in [0, n-1]
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i  # index of g
            A_ub[k, j] = f[i]
    
    # Bounds: g_j >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None
