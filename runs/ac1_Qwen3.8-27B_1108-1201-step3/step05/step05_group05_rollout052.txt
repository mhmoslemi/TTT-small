from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def logsumexp(x, t):
    """Compute (1/t) * log(sum(exp(t*x))) stably."""
    x_max = x.max()
    if x_max > 1e150 / t:
        return x_max
    return x_max + np.log(np.sum(np.exp(t * (x - x_max)))) / t

def smooth_objective(x, t, n):
    """Compute smooth approximation of 2n * max(conv(a,a)) / sum(a)^2 using softmax param."""
    a = np.exp(x - x.max())  # softmax without normalization (scale invariant)
    a_sum = a.sum()
    conv = fast_convolve(a / a_sum)  # normalize to sum=1
    smooth_max = logsumexp(conv, t)
    return 2.0 * n * smooth_max  # since sum=1, objective is 2n * max_conv

def smooth_gradient(x, t, n):
    """Gradient of smooth objective w.r.t. x (log-parameterization)."""
    a = np.exp(x - x.max())
    a_sum = a.sum()
    a_norm = a / a_sum  # a_i / sum(a)
    
    conv = fast_convolve(a_norm)
    conv_max = conv.max()
    
    # Softmax weights
    scaled = t * (conv - conv_max)
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # Gradient of max_conv w.r.t. a_norm
    grad_a = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a_norm[k - idx]
    
    # Chain rule: d(a_norm_i)/dx_j
    # a_norm_i = exp(x_i) / sum(exp(x_k))
    # d(a_norm_i)/dx_j = a_norm_i * (delta_ij - a_norm_j)
    # 
    # df/dx_j = sum_i df/da_norm_i * d(a_norm_i)/dx_j
    #         = sum_i df/da_norm_i * a_norm_i * (delta_ij - a_norm_j)
    #         = a_norm_j * (df/da_norm_j - sum_i df/da_norm_i * a_norm_i)
    
    grad_fx = grad_a * a_norm  # df/da_norm * a_norm
    dot = grad_fx.sum()
    grad_x = a_norm * (grad_a - dot)
    
    # Scale by 2n
    return 2.0 * n * grad_x

    n = len(x)

    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)

    obj = 2.0 * n * smooth_max  # sum(a) = 1, so denom = 1

    # dsmooth_max/d a_j = 2 * sum_k w[k] * a[k-j] (for valid k-j)

        je = min(n - 1, k)

    w_sum = w.sum()

    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad_a = np.zeros(n)
    threshold = np.max(w) * 1e-12
    for k in range(len(w)):

        if wk < threshold:

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

    exp_x = np.exp(x_shifted)

def lbfms_single_temperature(x, t, n, max_iter=300):
    """Run L-BFGS at a single temperature."""
    state = {'count': 0}
    
    def func_and_grad(x):

def flat_spectrum_init(n, rng):
    """Flat spectrum initialization."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def smooth_init(n, rng, kind='gaussian'):
    """Smooth initialization."""
    if kind == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 2)
    elif kind == 'triangular':
        x = np.arange(n) / max(n - 1, 1)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2)**2
    elif kind == 'bimodal':
        c1 = rng.uniform(0.25, 0.35)
        c2 = rng.uniform(0.65, 0.75)
        w = 0.08
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1)**2) / (2*w**2)) + np.exp(-((x - c2)**2) / (2*w**2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_targeting_refine(a0, time_limit, rng):
    """Local search that targets the peak positions in the convolution."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peak_val = conv.max()
        
        if peak_val < best_obj:
            best_obj = peak_val
            best_a = a.copy()
        
        # Find top peak positions
        top_k = min(5, len(conv))
        peak_idxs = np.argpartition(conv, -top_k)[-top_k:]
        peak_idxs = peak_idxs[np.argsort(conv[peak_idxs])[::-1]]
        
        improved = False
        for pk in peak_idxs[:3]:
            # This peak is conv[pk] = sum_i a[i] * a[pk-i]
            # Find the pair (i, pk-i) with the largest product
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            products = a[idx] * a[pk - idx]
            
            if len(products) == 0:
                continue
            
            max_prod_idx = np.argmax(products)
            i_max = idx[max_prod_idx]
            j_max = pk - i_max
            
            if a[i_max] < 1e-10 or a[j_max] < 1e-10:
                continue
            
            # Try to reduce the product by spreading mass
            # Reduce a[i_max] and a[j_max], add to random other positions
            for frac in [0.05, 0.1, 0.15, 0.2]:
                reduction = frac * min(a[i_max], a[j_max])
                a_trial = a.copy()
                a_trial[i_max] -= reduction
                a_trial[j_max] -= reduction
                
                # Add to positions that don't contribute to this peak
                # and don't create new peaks
                candidates = []
                for pos in rng.integers(0, n, size=10):
                    if pos != i_max and pos != j_max:
                        # Check if this position contributes to the peak
                        if not (0 <= pk - pos < n and a_trial[pk - pos] > 1e-8):
                            candidates.append(pos)
                
                if len(candidates) == 0:
                    candidates = [rng.integers(0, n) for _ in range(5)]
                
                # Distribute
                for pos in candidates[:len(candidates)]:
                    a_trial[pos] += reduction / len(candidates)
                
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                
                trial_obj = fast_convolve(a_trial).max()
                if trial_obj < best_obj - 1e-12:
                    best_obj = trial_obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            # Perturb and continue
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise, target_sum=1.0)
    
    return best_a, best_obj
