    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    """
    Compute a weighted subgradient of max(b) where b = conv(a,a).
    Uses softmax weighting over the top peaks.
    Returns the subgradient vector (length n).
    """
    n = len(a)
    # Find top peaks
    k_actual = min(top_k, len(b))
    peak_indices = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peak_indices]
    
    # Softmax weights
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peak_indices))
        w[np.argmax(pv)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(peak_indices)):

        k = int(peak_indices[i])

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[k - idx]
    
    return sg, b.max()

    a_prev = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t_k = 1.0

    temp = temp_init
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        y = project_simplex(y)

            t_k = 1.0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat-spectrum (Gerchberg-Saxton) initialization
2. Nesterov-accelerated projected subgradient with adaptive peak weighting
3. Power-mean smoothed L-BFGS (unbiased smooth approximation to max)

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_max(a, b, top_k=20, temperature=0.01):
    """Compute weighted subgradient of max(b) using softmax over top peaks."""
    n = len(a)
    k_actual = min(top_k, len(b))
    peak_indices = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peak_indices]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peak_indices))
        w[np.argmax(pv)] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if w[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[k - idx]
    return sg, b.max()


def nesterov_subgradient(a0, time_limit=100.0, top_k=30, temp_init=0.005, temp_growth=1.003, base_lr=2.0):
    """Nesterov-accelerated projected subgradient for min max(conv(a,a)) on simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_obj = b.max()

    t0 = time.monotonic()
    t_k = 1.0
    temp = temp_init
    patience = 0
    lr_decay = 0.9999

    while time.monotonic() - t0 < time_limit:
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        mu = (t_k - 1.0) / t_next
        y = a + mu * (a - a_prev)
        y = project_simplex(y)
        b_y = fast_convolve(y)
        sg, obj_y = subgradient_at_max(y, b_y, top_k=top_k, temperature=temp)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        lr = base_lr / np.sqrt(t_k)
        a_new = project_simplex(y - lr * sg)
        a_prev = a
        a = a_new
        t_k = t_next
        temp *= temp_growth
        b_a = fast_convolve(a)
        obj_a = b_a.max()
        if obj_a < best_obj - 1e-14:
            best_obj = obj_a
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        if patience > 2000:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 999983)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            a_prev = a.copy()
            t_k = 1.0
            patience = 0
            temp = temp_init

    return best_a, best_obj


def power_mean_obj_and_grad(a, p):
    """Compute (sum b_k^p)^(1/p) and its gradient where b = conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    b_safe = np.maximum(b, 1e-300)
    b_p = b_safe ** p
    pm = b_p.sum() ** (1.0 / p)
    b_p_minus_1 = b_p / b_safe
    size = n + len(b_p_minus_1) - 1
    fft_size = 1 << (size - 1).bit_length()
    BP1 = rfft(b_p_minus_1, fft_size)
    A = rfft(a, fft_size)
    grad_corr = irfft(BP1 * np.conj(A), fft_size)[:n]
    prefactor = pm ** (1.0 - p)
    grad = prefactor * grad_corr
    return pm, grad


def lbgfs_power_mean(a0, p, time_limit=60.0, maxiter=5000):
    """L-BFGS on power-mean objective, maintaining simplex via re-projection."""
    n = len(a0)
    t0 = time.monotonic()
    a_curr = a0.copy().astype(np.float64)
    a_curr = a_curr / a_curr.sum()
    elapsed = [time.monotonic()]

    def fun(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros(n)
        x = project_simplex(x)
        obj, grad = power_mean_obj_and_grad(x, p)
        return obj, grad

    res = minimize(
        fun,
        x0=a_curr,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                 'maxfun': maxiter * 10, 'disp': False,
                 'maxls': 20}
    )
    a_res = project_simplex(res.x)
    return a_res


def flat_spectrum_init(n, n_iter=50, target='flat', rng=None):
    """Gerchberg-Saxton style flat spectrum initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = rng.random(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    for _ in range(n_iter):
        A = fft(a, 2 * n)
        if target == 'flat':
            magnitude = np.ones(len(A))
        else:
            magnitude = np.ones(len(A))
        magnitude[0] = 1.0
        new_A = magnitude * (A / np.maximum(np.abs(A), 1e-300) + 0j)
        a_new = np.real(ifft(new_A, 2 * n)[:n])
        a = np.maximum(a_new, 0)
        s = a.sum()
        if s < 1e-15:
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = a / s
    return a


def structured_init(n, kind='hann'):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'triangular':
        a = 1 - 2 * np.abs(x - 0.5)
    elif kind == 'gaussian':
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif kind == 'double_bump':
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif kind == 'raised_cosine':
        a = np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0)
    elif kind == 'sinc_like':
        t = 2 * x - 1
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def peak_reduction(a0, time_limit=30.0, n_steps=500):

    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        k = int(np.argmax(b))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k - idx]
        max_idx = idx[np.argmax(contributions)]
        partner = k - max_idx
        if max_idx == partner:
            factor = 0.97
            new_val = a[max_idx] * factor
            diff = a[max_idx] - new_val
            mask = np.ones(n, dtype=bool)
            mask[max_idx] = False
            others = a[mask]
            os = others.sum()
            if os < 1e-15:
                break
            a_trial = a.copy()
            a_trial[max_idx] = new_val
            a_trial[mask] = others + diff * others / os
        else:
            for target, other in [(max_idx, partner), (partner, max_idx)]:
                factor = 0.97
                new_val = a[target] * factor
                diff = a[target] - new_val
                a_trial = a.copy()
                a_trial[target] = new_val
                a_trial[other] = a[other] + diff
                if a_trial[other] < 0:
                    continue
                s_trial = a_trial.sum()
                a_trial = a_trial / s_trial
                b_trial = fast_convolve(a_trial)
                if b_trial.max() < best_obj - 1e-14:
                    a = a_trial
                    b = b_trial
                    best_obj = b.max()
                    best_a = a.copy()
                break
        else:
            continue

    return best_a, best_obj


def coordinate_descent_simple(a0, time_limit=30.0):
    """Simple coordinate descent: try scaling each coordinate."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    multipliers = np.array([0.7, 0.85, 1.15, 1.3])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            a_struct = structured_init(n_struct, kind)
            candidates.append((f'{kind}_n{n_struct}', a_struct, n_struct))

    # Flat spectrum initializations
    for n_fs in [768, 1024, 1536, 2048]:
        a_fs = flat_spectrum_init(n_fs, n_iter=80, rng=rng)
        candidates.append((f'flat_n{n_fs}', a_fs, n_fs))

    # Evaluate all candidates
    for label, a_cand, n_cand in candidates:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        try_seq(a_cand, f'init_{label}')

    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 2: Nesterov accelerated subgradient ===
    print(f"\n=== Phase 2: Nesterov subgradient (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    # Run Nesterov on the best and on a few other candidates
    nesterov_targets = [('best', best_seq.copy(), n_best)]
    # Add a few diverse ones
    for label, a_cand, n_cand in candidates[:8]:
        if a_cand is not best_seq:
            nesterov_targets.append((label, a_cand.copy(), n_cand))

    for i, (label, a_start, n_s) in enumerate(nesterov_targets[:5]):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 5 - i), 150.0)
        print(f"  Nesterov [{label}] n={n_s}, t={t_alloc:.0f}s")
        a_res, obj_res = nesterov_subgradient(
            a_start, time_limit=t_alloc, top_k=30,
            temp_init=0.003, temp_growth=1.004, base_lr=3.0
        )
        try_seq(a_res, f'nesterov_{label}')

    # Second round with different parameters
    for i, (label, a_start, n_s) in enumerate(nesterov_targets[:3]):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 3 - i), 100.0)
        print(f"  Nesterov R2 [{label}] n={n_s}, t={t_alloc:.0f}s, smaller top_k")
        a_res, obj_res = nesterov_subgradient(
            a_start, time_limit=t_alloc, top_k=10,
            temp_init=0.001, temp_growth=1.002, base_lr=5.0
        )
        try_seq(a_res, f'nesterov2_{label}')

    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 3: Power-mean L-BFGS ===
    print(f"\n=== Phase 3: Power-mean L-BFGS (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    # Try increasing powers on the best sequence
    powers = [8, 16, 32, 64, 128]
    a_curr = best_seq.copy()
    for p in powers:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(rem / max(1, len(powers) - powers.index(p)), 80.0)
        print(f"  L-BFGS p={p}, n={n_best}, t={t_alloc:.0f}s")
        a_res = lbgfs_power_mean(a_curr, p, time_limit=t_alloc, maxiter=3000)
        try_seq(a_res, f'lbfgs_p{p}')
        if best_seq is not None:
            a_curr = best_seq.copy()

    # Try on other lengths
    for target_n in [1024, 2048]:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.3, 80.0)
        print(f"  L-BFGS n={target_n}, p=32, t={t_alloc:.0f}s")
        a_res = lbgfs_power_mean(f_interp, 32, time_limit=t_alloc, maxiter=2000)
        try_seq(a_res, f'lbfgs_n{target_n}_p32')

        rem = deadline - time.monotonic() - 100
        if rem > 20 and best_seq is not None:
            f_interp2 = interpolate_sequence(best_seq, target_n)
            t_alloc2 = min(rem * 0.5, 60.0)
            a_res2 = lbgfs_power_mean(f_interp2, 64, time_limit=t_alloc2, maxiter=2000)
            try_seq(a_res2, f'lbfgs_n{target_n}_p64')

    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 4: Fine refinement ===
    print(f"\n=== Phase 4: Fine refinement (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    # Nesterov with very fine parameters on the best
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_alloc = min(rem * 0.4, 60.0)
        print(f"  Fine Nesterov, t={t_alloc:.0f}s")
        a_res, _ = nesterov_subgradient(
            best_seq, time_limit=t_alloc, top_k=5,
            temp_init=0.0005, temp_growth=1.001, base_lr=8.0
        )
        try_seq(a_res, 'fine_nesterov')

    # Peak reduction
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_pr = min(rem * 0.4, 40.0)
        print(f"  Peak reduction, t={t_pr:.0f}s")
        a_res, _ = peak_reduction(best_seq, time_limit=t_pr, n_steps=1000)
        try_seq(a_res, 'peak_reduce')

    # Coordinate descent
    rem = deadline - time.monotonic() - 40
    if rem > 10:
        t_cd = min(rem * 0.5, 30.0)
        print(f"  Coordinate descent, t={t_cd:.0f}s")
        a_res, _ = coordinate_descent_simple(best_seq, time_limit=t_cd)
        try_seq(a_res, 'coord_descent')

    # === Phase 5: Multi-restart perturbation ===
    print(f"\n=== Phase 5: Multi-restart (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    for attempt in range(10):
        rem = deadline - time.monotonic() - 30
        if rem < 20:
            break
        noise_scale = 0.001 * (1.0 / (attempt + 1))
        noise = rng.standard_normal(n_best) * noise_scale
        a_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem * 0.4, 30.0)
        a_res, _ = nesterov_subgradient(
            a_pert, time_limit=t_alloc, top_k=15,
            temp_init=0.002, temp_growth=1.003, base_lr=4.0
        )
        try_seq(a_res, f'restart_{attempt}')

    # === Phase 6: Final high-p power mean ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.5, 30.0)
            print(f"  Final L-BFGS p=256, t={t_alloc:.0f}s")
            a_res = lbgfs_power_mean(best_seq, 256, time_limit=t_alloc, maxiter=2000)
            try_seq(a_res, 'final_p256')

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.8, 20.0)
            print(f"  Final Nesterov ultra-fine, t={t_alloc:.0f}s")
            a_res, _ = nesterov_subgradient(
                best_seq, time_limit=t_alloc, top_k=3,
                temp_init=0.0001, temp_growth=1.0, base_lr=10.0
            )
            try_seq(a_res, 'final_nesterov')

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
