   import numpy as np
   from scipy.optimize import minimize, linprog
   from scipy.special import logsumexp
   import time
   
   def fast_convolve(a):
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = np.fft.rfft(a, fft_size)
       return np.fft.irfft(A * A, fft_size)[:size]
   
   def softmax(x):
       x = x - x.max()
       e = np.exp(x)
       return e / e.sum()
   
   def lse_objective_and_grad(theta, T):
       """Compute LSE smooth objective and gradient w.r.t. theta."""
       a = softmax(theta)
       n = len(a)
       conv = fast_convolve(a)
       scaled = conv / T
       m = scaled.max()
       e = np.exp(scaled - m)
       w = e / e.sum()
       val = T * (m + np.log(e.sum()))
       
       # Compute grad w.r.t. a first, then chain rule through softmax
       # grad_a[i] = 2T * sum_j a[j] * w[i+j] (as derived above)
       N = 1 << (3 * n - 2).bit_length()
       A_fft = np.fft.rfft(a, N)
       W_fft = np.fft.rfft(w, N)
       C = np.fft.irfft(A_fft * np.conj(W_fft), N)
       # g[i] = C[N-i] for i=1..n-1, C[0] for i=0
       idx = np.concatenate(([0], N - np.arange(1, n)))
       g = C[idx]
       grad_a = 2 * T * g
       
       # Chain rule: d theta / d a via softmax
       # a = softmax(theta), so da_i/dtheta_j = a_i * (delta_ij - a_j)
       # grad_theta_j = sum_i grad_a_i * da_i/dtheta_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
       # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
       dot = np.dot(grad_a, a)
       grad_theta = a * (grad_a - dot)
       
       return val, grad_theta
   
   def exact_objective(a):
       """Compute the exact objective (scale-invariant version)."""
       n = len(a)
       conv = fast_convolve(a)
       max_b = conv.max()
       sum_a = a.sum()
       if sum_a < 1e-15:
           return np.inf
       return 2.0 * n * max_b / (sum_a ** 2)
