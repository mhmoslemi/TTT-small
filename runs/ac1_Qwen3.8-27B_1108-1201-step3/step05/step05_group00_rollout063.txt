def project_simplex(v):

            grad[idx] += 2.0 * w[k] * f[k - idx]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        update = m_hat / (np.sqrt(v_hat) + eps)
        f_new = f - lr * update
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            continue
        f_new = f_new / s_new
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            if verbose and t % 100 == 0:
                print(f"    SMOX iter {t}: obj={best_obj:.10f}")
        
        f = f_new
        lr *= 0.9995
        
        if lr < 1e-6:
            break
    
    return best_f, best_obj

def multiplicative_descent_v2(f0, time_limit, verbose=False):
    """Improved multiplicative descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    temperature = 0.001
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        k_actual = min(50, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Softmax weights with adaptive temperature
        pv = conv[peaks]
        # Adaptive: temperature = (second_max - min) / 5 or similar
        sorted_pv = np.sort(pv)[::-1]
        if len(sorted_pv) > 1:
            temperature = max((sorted_pv[0] - sorted_pv[min(10, len(sorted_pv)-1)]) / 5, 1e-10)
        
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[0] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD2 iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        lr *= 0.999
        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj

sparse LP-based Frank-Wolfe optimization with adaptive M expansion.
Key insight: optimal sequences have bathtub (U-shaped) profiles with high
values at boundaries and low values in the interior to flatten the auto-convolution.
"""

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_conv_matrix(n):
    """Build the sparse convolution constraint matrix for the LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(0.0)  # placeholder, will be filled
    A = sp.csr_matrix((len(vals), (2 * n - 1, n)))
    # Actually, the matrix structure is fixed; only the values change
    # Let's precompute the structure
    row_indices = []
    col_indices = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_indices.append(k)
            col_indices.append(j)
    return np.array(row_indices), np.array(col_indices)


def solve_lp_direction(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    vals = np.zeros(len(row_idx))
    # Fill in the values: A[k,j] = f[j] for j in valid range
    for i in range(len(row_idx)):
        k = row_idx[i]
        j = col_idx[i]
        vals[i] = f[j]
    
    A_ub = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, n_precomp=None):
    """LP-based Frank-Wolfe with adaptive M expansion."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Precompute sparsity pattern
    row_idx, col_idx = build_sparse_conv_matrix(n)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break
        
        g = solve_lp_direction(f, M * m_exp, n, row_idx, col_idx)
        if g is None:
            m_exp *= 1.1
            if m_exp > 10.0:
                # Perturb and restart
                seed = int(time.monotonic() * 10000) % 100000
                rng = np.random.default_rng(seed)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003 * min(m_exp, 3.0))
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 40:
                m_exp = min(m_exp * 1.08, 6.0)
            if no_improve > 150:
                break
            continue
        else:
            if diff > 1e-5:
                no_improve = 0
            else:
                no_improve += 1
        
        # Line search: coarse + fine
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 6.0)
    
    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.008):
    """Mirror descent with multiplicative weight updates."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Top peaks with softmax weighting
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        sorted_pv = np.sort(pv)[::-1]
        temp = max((sorted_pv[0] - sorted_pv[min(k_actual-1, len(sorted_pv)-1)]) / 3, 1e-10)
        
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        lr *= 0.9992
        if no_improve > 200:
            rng = np.random.default_rng(it * 37 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj


def make_u_shaped(n, depth):
    """U-shaped sequence: high at edges, low in middle."""
    x = np.arange(n) / max(n - 1, 1)
    t = 2.0 * x - 1.0
    a = 1.0 + depth * t**2
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_bathtub(n, alpha):
    """Bathtub: p[j] proportional to (j+1)^alpha * (n-j)^alpha."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0)**alpha * (n - j)**alpha
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def make_cosine_u(n, amplitude):
    """Cosine U-shape: p[j] = 1 + amplitude * cos(2*pi*j/n)."""
    j = np.arange(n)
    a = 1.0 + amplitude * np.cos(2.0 * np.pi * j / n)
    a = np.maximum(a, 1e-10) 
    a = a / a.sum()
    return a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using U-shaped (bathtub) initializations
and sparse LP-based Frank-Wolfe optimization with adaptive M expansion.
Key structural insight: optimal sequences have high boundary values and low interior

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_indices(n):
    """Precompute sparsity pattern for the convolution constraint matrix."""
    row_idx = []
    col_idx = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row_idx.append(k)
            col_idx.append(j)
    return np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32)


def solve_lp_direction(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    vals = np.zeros(len(row_idx), dtype=np.float64)
    vals[:] = f[col_idx]
    A_ub = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.008):
    """Mirror descent with multiplicative weight updates and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Top peaks with adaptive softmax weighting
        k_actual = min(40, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        sorted_pv = np.sort(pv)[::-1]
        gap = sorted_pv[0] - sorted_pv[min(k_actual - 1, len(sorted_pv) - 1)]
        temp = max(gap / 3.0, 1e-10)

        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient computation
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.9992
        if no_improve > 200:
            rng = np.random.default_rng(it * 37 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def make_u_shaped(n, depth):
    """U-shaped sequence: high at edges, low in middle (quadratic)."""
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    t = 2.0 * x - 1.0
    a = 1.0 + depth * t ** 2
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_bathtub(n, alpha):
    """Bathtub: p[j] proportional to (j+1)^alpha * (n-j)^alpha."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0) ** alpha * (n - j) ** alpha
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def make_cosine_u(n, amplitude):
    """Cosine U-shape: p[j] = 1 + amplitude * cos(2*pi*j/n)."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + amplitude * np.cos(2.0 * np.pi * j / n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_gaussian(n, sigma_frac):
    """Gaussian bell-shaped initial sequence."""
    x = np.arange(n, dtype=np.float64) - n / 2.0
    a = np.exp(-0.5 * (x / (n * sigma_frac)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
