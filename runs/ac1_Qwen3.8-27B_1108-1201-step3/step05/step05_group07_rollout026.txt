approach with alternating LP solves, combined with gradient-based refinement

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=cp.CLARABEL):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val = b_val / b_val.sum()
    return b_val, float(M_var.value)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_refine(a0, time_limit, lr_init=0.01, top_k=50):
    """Subgradient method to refine a sequence."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv - pv.max()
        w = np.exp(scaled * 10)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - lr * sg
        a_new = project_simplex(a + v)
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv

            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.999
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = max(lr * 2, 0.001)
            no_improve = 0

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

        a_new = (1 - alpha) * a + alpha * b_new

        a_new = a_new / a_new.sum()

        if score < best_score - 1e-14:

            best_a = a_new.copy()
        a = a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a game-theoretic fictitious play
approach with alternating LP solves (exploiting the bilinear structure of the
auto-correlation), combined with damped LP refinement, subgradient polishing,
and multi-length exploration. The key insight: the problem is equivalent to
finding the symmetric Nash equilibrium of a zero-sum game where each player
chooses a probability distribution and the payoff is the peak cross-correlation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:

        for alt in [cp.ECOS, cp.GLPK, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            ma_str = f"{M_a:.6f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.6f}" if M_b is not None else "N/A"
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): M_a={ma_str} M_b={mb_str} best={best_score:.6f}")

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None, n_max_iters=2000):
    """
    Damped alternating LP refinement. Uses decreasing step size for stability.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Damped update: alpha decreases over time
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DampedLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, lr_init=0.008, top_k=50):
    """Subgradient method with momentum for fine-tuning."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.92
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Temperature for softmax weighting of peaks
        temp = 0.005 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - lr * sg

        # Project onto simplex
        a_candidate = a + v
        # Simplex projection
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv
        step_count += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.002)
            no_improve = 0

        if step_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=2000):
    """Aggressive local peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak (excluding trivial endpoints)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))
        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:3]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T=200, maxiter=3000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0

    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Fictitious play (alternating LP) at multiple sequence lengths.
    2. Damped LP refinement.
    3. L-BFGS soft-max polishing.
    4. Subgradient and peak-shaving local search.
    5. Additional FP rounds with refined initial points.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Find LP solver
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: Fictitious Play at multiple lengths ===
    print("\n=== Phase 1: Fictitious Play ===")
    n_values = [512, 768, 1024, 1280, 1536]
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in n_values:
            n_values.insert(0, n_parent)

    # Prioritize parent length, then nearby lengths
    n_values_sorted = sorted(n_values, key=lambda x: abs(x - (n_parent if parent is not None else 768)))
    n_values = n_values_sorted[:6]  # Try at most 6 lengths

    fp_results = {}
    total_fp_time = min(1000.0, (deadline - time.monotonic()) * 0.55)
    time_per_n = total_fp_time / len(n_values)

    for n_try in n_values:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_alloc = min(time_per_n, rem * 0.85)
        if t_alloc < 30:
            continue

        print(f"  FP n={n_try}, budget={t_alloc:.0f}s")
        initial = None
        if parent is not None:
            if len(parent) == n_try:
                initial = parent
            else:
                initial = interpolate_sequence(parent, n_try)
        elif best_seq is not None:
            initial = interpolate_sequence(best_seq, n_try)

        a_fp, score_fp, iters = fictitious_play_optimize(
            n_try, t_alloc, initial=initial, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_try}_it{iters}", best_seq, best_score)
        fp_results[n_try] = a_fp

    # === Phase 2: Damped LP on best result ===
    print(f"\n=== Phase 2: Damped LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.6, 250.0)
            print(f"  Damped LP n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_dl, score_dl = damped_lp_refine(best_seq, t_alloc, solver=solver)
            best_seq, best_score = check_and_update(a_dl, "damped_lp", best_seq, best_score)

    # === Phase 3: L-BFGS soft-max polish ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            for T in [50, 200, 500]:
                rem = deadline - time.monotonic() - 100
                if rem < 15:
                    break
                t_alloc = min(rem * 0.4, 80.0)
                print(f"  L-BFGS T={T}, n={n_best}, budget={t_alloc:.0f}s")
                a_lb = lbgfs_refine(best_seq, t_alloc, T=T)
                best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 4: Subgradient + Peak shaving ===
    print(f"\n=== Phase 4: Local Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_sg = min(rem * 0.4, 80.0)
            top_k = max(20, n_best // 10)
            a_sg, _ = subgradient_refine(best_seq, t_sg, lr_init=0.005, top_k=top_k)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_ps = min(rem * 0.7, 50.0)
            a_ps, _ = peak_shave_local(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: Second round of FP with refined start ===
    print(f"\n=== Phase 5: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 40:
            t_alloc = min(rem * 0.6, 200.0)
            print(f"  FP round 2: n={n_best}, budget={t_alloc:.0f}s")
            a_fp2, score_fp2, iters2 = fictitious_play_optimize(
                n_best, t_alloc, initial=best_seq, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp2, f"FP2_n{n_best}", best_seq, best_score)

            # Damped LP on new best
            if best_seq is not None:
                rem = deadline - time.monotonic() - 40
                if rem > 20:
                    a_dl2, _ = damped_lp_refine(best_seq, rem * 0.7, solver=solver)
                    best_seq, best_score = check_and_update(a_dl2, "damped_lp_2", best_seq, best_score)

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5, 2.0]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            t_alloc = min(rem * 0.5, 120.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, score_alt, iters_alt = fictitious_play_optimize(
                alt_n, t_alloc, initial=init_alt, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 30
                if rem > 15:
                    a_lp_alt, _ = damped_lp_refine(best_seq, rem * 0.6, solver=solver)
                    best_seq, best_score = check_and_update(a_lp_alt, f"lp_alt_n{alt_n}", best_seq, best_score)

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            n_best = len(best_seq)
            t_final = min(rem * 0.6, 30.0)
            a_final = lbgfs_refine(best_seq, t_final, T=500, maxiter=2000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_ps_final = min(rem * 0.7, 20.0)
            a_ps_final, _ = peak_shave_local(best_seq, t_ps_final)
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
