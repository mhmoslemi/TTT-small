import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lag_gradient(a, k):
    """Compute db[k]/da for linear convolution b = conv(a,a)."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    g = np.zeros(n)
    g[lo:hi+1] = a[lo:hi+1][::-1] + a[lo:hi+1]
    return g

def compute_topk_subgradient(a, b, top_k, temperature=0.01):
    """Compute subgradient as weighted sum of top-K lag gradients."""
    n = len(a)
    size = 2 * n - 1
    if top_k >= size:
        idx = np.argsort(b)[::-1][:size]
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
    
    vals = b[idx]
    # Softmax weights
    shifted = (vals - vals.max()) / (vals.max() * temperature + 1e-30)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    
    g = np.zeros(n)
    for i, k in enumerate(idx):
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        g[lo:hi+1] += w[i] * (a[lo:hi+1][::-1] + a[lo:hi+1])
    return g

def hankel_mv(w, a):
    """Compute H(w) @ a where H(w)_{ij} = w[i+j]."""
    # (Ha)_i = sum_j w[i+j] * a[j]
    # This is np.correlate(w, a, 'valid')
    return np.correlate(w, a, mode='valid')

    t = 0
    eta = eta0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_convolve(a)
        cur_obj = b.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > restart_interval:
            rng = np.random.default_rng(t * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            eta = eta0
            no_improve = 0
            continue
        
        g = compute_topk_subgradient(a, b, top_k, temp)
        
        # Mirror descent update (multiplicative weights)
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        ss = a_new.sum()
        if ss < 1e-300:
            a = np.ones(n) / n
        else:
            a = a_new / ss
        
        eta = max(eta * eta_decay, eta0 * 0.01)
    
    return best_a, best_obj

def two_timescale_optimize(a0, time_limit, inner_iters=50, eta_inner=0.005,
                           eta_outer=0.1, top_k_for_w=30):
    """Two-timescale: outer updates weights, inner solves QP."""
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    size = 2 * n - 1
    
    # Initialize weights
    b = fast_convolve(a)
    w = np.exp(b / (b.max() * 0.1 + 1e-30))
    w /= w.sum()
    
    best_a = a.copy()
    best_obj = b.max()
    
    t0 = time.monotonic()
    outer_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        # Inner loop: minimize a^T H(w) a s.t. sum(a)=1, a>=0
        for _ in range(inner_iters):
            if time.monotonic() - t0 > time_limit:
                break
            Ha = hankel_mv(w, a)
            grad = 2.0 * Ha
            a_new = a - eta_inner * grad
            a = project_simplex(a_new)
        
        # Evaluate
        b = fast_convolve(a)
        cur_obj = b.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Outer loop: update weights
        peak = b.max()
        w = w * np.exp(eta_outer * b / (peak + 1e-30))
        w /= w.sum()
        
        outer_iter += 1
    
    return best_a, best_obj

def spectral_flatten(a0, time_limit, alpha=0.1, n_iters_per_sec=1000):
    """Spectral-domain flattening of autocorrelation."""
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute FFT of a
        A = np.fft.rfft(a, fft_size)
        B = A * np.conj(A)  # |A|^2, this is FFT of circular autocorrelation
        
        # Inverse FFT to get autocorrelation
        b = np.fft.irfft(B, fft_size)[:2*n-1]
        
        # Flatten: reduce peak
        peak = b.max()
        b_flat = b.copy()
        # Reduce values above the mean
        b_mean = b.mean()
        mask = b_flat > b_mean
        b_flat[mask] = b_mean + (b_flat[mask] - b_mean) * (1 - alpha)
        
        # Recover magnitude spectrum
        B_new = np.fft.rfft(b_flat, fft_size)
        # Ensure B_new is non-negative (it should be since b_flat is an autocorrelation)
        # Actually B_new should be real and non-negative if b_flat is a valid autocorrelation
        # But b_flat might not be a valid autocorrelation, so let's use a different approach
        
        # Alternative: modify the magnitude spectrum directly
        # Reduce the "peak contribution" in frequency domain
        # The peak in time domain corresponds to coherent addition in frequency
        # So we want to "scramble" the phases or reduce magnitude at peaks
        
        # Simpler: just use the time-domain modification
        B_modified = B * (1 - alpha) + B.mean() * alpha
        # This reduces the dynamic range of the spectrum
        
        A_mag = np.sqrt(np.maximum(B, 0))
        A_phase = A / (A_mag + 1e-30)
        
        # Modify magnitude: compress dynamic range
        A_mag_new = A_mag ** (1 - alpha) * (A_mag.mean() ** alpha)
        # Actually this doesn't help. Let me think differently.
        
        # Better: use the "flat-top" idea
        # Target: |A|^2 should be proportional to 1 (flat)
        # Current: |A|^2 = B
        # Interpolate: B_new = (1-alpha)*B + alpha*B_mean
        B_new2 = (1 - alpha) * B + alpha * B.mean()
        A_new = np.sqrt(np.maximum(B_new2, 0)) * A_phase
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new /= ss
            a = a_new
        
        b = fast_convolve(a)
        cur_obj = b.max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
        
        count += 1
        if count % 10000 == 0:
            print(f"    Spectral iter {count}: obj={best_obj:.10f}")
    
    return best_a, best_obj

def coordinate_descent_fast(a0, time_limit, n_multipliers=6):
    """Fast coordinate descent with fewer multipliers and vectorized evaluation."""
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.6])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random order for diversity
        indices = np.random.permutation(n)
        
        for j in indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                # Try to activate this coordinate
                a[j] = best_obj / n  # small value
                a = a / a.sum()
                continue
            
            best_j_obj = best_obj
            best_j_mult = 1.0  # no change
            
            # Quick evaluation: only compute the affected part of the convolution
            # b[k] changes only for k in [j, j+n-1] (approximately)
            # For simplicity, compute full conv for each trial (n=768, this is fast)
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_mult = mult
            
            if best_j_mult != 1.0:
                new_val = old_val * best_j_mult
                denom = 1.0 - old_val
                scale = (1.0 - new_val) / denom
                a[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                improved = True
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
        
        if not improved:
            # Try larger perturbations
            rng = np.random.default_rng()
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
    
    return best_a, best_obj

def adam_with_exact_sg(a0, time_limit, lr=0.003, top_k=30, temp=0.005,
                       beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam with exact top-K subgradient."""
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve = 0
    
    t0 = time.monotonic()
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_convolve(a)
        cur_obj = b.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(t * 13 + 5)
            noise = 0.003 + 0.002 * min(no_improve // 150, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            cur_lr = lr
            no_improve = 0
            continue
        
        g = compute_topk_subgradient(a, b, top_k, temp)
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        cur_lr = max(lr * (0.99995 ** t), lr * 0.05)
    
    return best_a, best_obj

def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_symmetric_init(n, shape='bump'):
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'flat_top':
        f = np.ones(n)
        # Taper edges
        taper = n // 10
        f[:taper] = np.linspace(0.3, 1, taper)
        f[-taper:] = np.linspace(1, 0.3, taper)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()
