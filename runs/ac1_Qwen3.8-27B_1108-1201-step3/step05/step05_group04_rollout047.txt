Adam optimization, and multi-phase search strategy.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def construct_conv_matrix(f, top_k=None):
    """Construct the convolution matrix A where (A @ b)[k] = (f*b)[k].
    If top_k is given, only return the rows corresponding to the top_k largest
    values of f*f (to speed up the LP)."""
    n = len(f)
    conv = fast_convolve(f)
    
    if top_k is not None and top_k < 2*n-1:
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_indices = np.arange(2*n-1)
    
    A = np.zeros((len(top_indices), n))
    for i, k in enumerate(top_indices):
        for m in range(n):
            idx = k - m
            if 0 <= idx <= n-1:
                A[i, m] = f[idx]
    
    return A, conv


def lp_direction(f, M, top_k=None):
    """Solve LP to find direction b that maximizes sum(b) subject to f*b <= M."""
    n = len(f)
    A, conv = construct_conv_matrix(f, top_k)
    
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        iteration += 1
        conv = fast_convolve(f)
        M = conv.max()

        b_lp = lp_direction(f, M, use_top_k=max(64, n // 3))
        if b_lp is None or b_lp.sum() < 1e-10:
            rng = np.random.default_rng(iteration * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            no_improve = 0
            t_step = 0.15
            continue

        b_rescaled = b_lp / b_lp.sum()

        best_trial_f = None
        best_trial_obj = M
        for t_ratio in [t_step, t_step * 0.5, t_step * 2, 0.05, 0.3]:
            t_r = min(max(t_ratio, 0.01), 0.8)
            f_trial = (1 - t_r) * f + t_r * b_rescaled
            f_trial = project_simplex(f_trial)
            trial_obj = fast_convolve(f_trial).max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_trial_f = f_trial.copy()

        if best_trial_f is not None:
            f = best_trial_f / best_trial_f.sum()
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
                t_step = min(t_step * 1.05, 0.4)
                if verbose and iteration % 20 == 0:
                    print(f"    LP iter {iteration}: obj={best_obj:.10f}")
            else:
                no_improve += 1
                t_step = max(t_step * 0.85, 0.02)
        else:
            no_improve += 1
            t_step = max(t_step * 0.8, 0.02)

        if no_improve > 40:
            rng = np.random.default_rng(iteration * 41 + 13)
            noise = 0.003 + 0.002 * min(no_improve // 15, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            no_improve = 0
            t_step = 0.15

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            multipliers = [0.3, 0.5, 0.7, 0.8, 1.2, 1.4, 1.7, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'arch':
        x = np.linspace(0, 1, n)
        f = 1 - (2 * x - 1) ** 4
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()
