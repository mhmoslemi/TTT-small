def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto the simplex {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)] * 1) * (target_sum / max(v[0], 1e-30)) if v[0] > 0 else np.array([target_sum])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For each k, the constraint is: sum_{j: 0<=k-j<n} f[j] * g[k-j] <= M
    # Let i = k-j, so j = k-i: sum_{i: 0<=k-i<n} f[k-i] * g[i] <= M
    # Coefficient of g[i] in constraint k is: f[k-i] if 0 <= k-i <= n-1, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_step(f, M, n, m_multiplier=1.0):
    """One LP-FW step with relaxed M."""
    g = lp_direction(f, M * m_multiplier, n)
    if g is None:
        return None, None
    gs = g.sum()
    if gs < 1e-15:
        return None, None
    g = g / gs
    return g, gs

            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 300:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3, center_frac=0.5):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_structured_random(n, seed):
    """Create a structured random sequence with good initial autocorrelation properties."""
    rng = np.random.default_rng(seed)
    # Start with random
    a = rng.random(n)
    # Apply a smoothing filter to reduce high-frequency content
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_scale(n, seed):
    """Create a two-scale sequence: large-scale structure + small-scale noise."""
    rng = np.random.default_rng(seed)
    # Large scale: smooth random
    coarse_n = max(8, n // 16)
    coarse = rng.random(coarse_n)
    coarse = np.convolve(coarse, np.ones(3) / 3, mode='same')
    coarse = np.maximum(coarse, 0)
    coarse = coarse / coarse.sum()
    # Interpolate to full length
    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, coarse_n), coarse)
    # Add small-scale noise
    noise = rng.random(n) * 0.1
    a = a + noise
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining smooth-max gradient descent,
    mirror descent, alternating peak reduction, and LP-FW across multiple
    sequence lengths with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s  # Normalize for evaluation
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Generate and evaluate diverse initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []
    
    lengths_to_try = [640, 704, 768, 832, 896, 1024]
    
    if parent is not None:
        init_candidates.append(('parent', parent))
    
    for n_try in lengths_to_try:
        a_gauss = make_gaussian(n_try, 0.3)
        a_bell = make_bell(n_try, 0.7)
        a_struct = make_structured_random(n_try, seed + n_try)
        a_twoscale = make_two_scale(n_try, seed + 1000 + n_try)
        a_flat = np.ones(n_try) / n_try
        init_candidates.append((f'gauss_{n_try}', a_gauss))
        init_candidates.append((f'bell_{n_try}', a_bell))
        init_candidates.append((f'struct_{n_try}', a_struct))
        init_candidates.append((f'twoscale_{n_try}', a_twoscale))
        init_candidates.append((f'flat_{n_try}', a_flat))
    
    # Evaluate all initializations
    scores = []
    for label, a_init in init_candidates:
        sc = try_seq(a_init, f"init_{label}")
        scores.append((sc, label, a_init))
    
    scores.sort(key=lambda x: x[0])
    print(f"  Top 3 inits: {[f'{s:.4f} ({l})' for s, l, _ in scores[:3]]}")
    
    # Pick top candidates for optimization
    top_inits = [(l, a) for s, l, a in scores[:5] if np.isfinite(s)]
    if not top_inits:
        top_inits = [('default', make_gaussian(768, 0.3))]

    # === Phase 2: Smooth-max gradient descent on top initializations ===
    print("\n=== Phase 2: Smooth-Max Gradient Descent ===")
    for label, a_init in top_inits[:3]:
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem / 3, 120)
        if t_alloc < 10:
            break
        print(f"  SMD on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = smooth_max_descent(a_init, time_limit=t_alloc, T_init=0.1, T_final=0.005, verbose=True)
        try_seq(a_opt, f"smd_{label}")

    # === Phase 3: Mirror descent on current best ===
    print("\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  MD on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_focused(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "md_polish")

    # === Phase 4: LP-FW on current best ===
    print("\n=== Phase 4: LP-FW Optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 180)
            print(f"  LP-FW on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 5: Try the best initialization at a different length ===
    print("\n=== Phase 5: Length Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby lengths
        for target_n in [n_best - 64, n_best + 64]:
            if target_n < 128:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            
            t_alloc = min(rem / 2, 80)
            if t_alloc < 10:
                continue
            print(f"  SMD n={target_n}: t={t_alloc:.0f}s")
            a_opt, obj = smooth_max_descent(f_interp, time_limit=t_alloc, T_init=0.08, T_final=0.005, verbose=False)
            try_seq(a_opt, f"smd_interp_n{target_n}")

    # === Phase 6: Alternating peak reduction ===
    print("\n=== Phase 6: Alternating Peak Reduction ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  APR on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = alternating_peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "apr")

    # === Phase 7: Multiple LP-FW with different perturbations ===
    print("\n=== Phase 7: LP-FW with Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            t_alloc = min(rem / (4 - p), 60)
            if t_alloc < 10:
                continue

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise, 1.0)
            elif p == 2:
                # Frequency-based perturbation
                freq = np.fft.rfft(best_seq)
                freq[np.random.default_rng(p * 100).integers(1, len(freq))] *= (1 + 0.01)
                f_init = np.fft.irfft(freq, n_best)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                # Blend with reversed
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = f_init / f_init.sum()

            print(f"  LP-FW variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"lpfw_pert_{p}")

    # === Phase 8: Final mirror descent polish ===
    print("\n=== Phase 8: Final MD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_focused(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "md_final")

    # === Phase 9: Final LP-FW squeeze ===
    print("\n=== Phase 9: Final LP-FW Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 20)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "lpfw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
