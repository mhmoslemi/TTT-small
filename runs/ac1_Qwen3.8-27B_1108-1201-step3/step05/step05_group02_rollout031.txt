"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth optimization (log-sum-exp + Adam) for fast initial convergence
2. LP-based Frank-Wolfe for precise local optimization
3. Multi-length exploration with interpolation
4. Coordinate descent and micro-perturbation polishing
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(f, tau):
    """Compute gradient of log-sum-exp(tau * conv(f,f)) w.r.t. f."""
    n = len(f)
    c = fast_convolve(f)
    scaled = tau * c
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # Gradient: 2*tau * (p * f[::-1]) at indices [n-1 : 2n-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    corr = np.fft.irfft(P * F, fft_size)
    grad = 2.0 * tau * corr[n - 1: 2 * n - 1]
    # Clip to length n
    grad = grad[:n]
    return grad, c.max()

def smooth_optimize(f0, time_limit, tau_init=5.0, tau_growth=1.003,
                    lr_init=0.3, lr_decay=0.9995, beta1=0.9, beta2=0.999):
    """Smooth optimization using log-sum-exp and Adam."""
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    tau = tau_init
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, _ = compute_gradient(f, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = f - step
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f
        else:
            f_new /= ss
        f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-18:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        tau *= tau_growth
        lr *= lr_decay
        
        if no_improve > 5000:
            # Restart with smaller tau
            tau = max(tau * 0.5, tau_init)
            lr = max(lr * 2, lr_init * 0.5)
            no_improve = 0
    
    return best_f, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, j - (n - 1))  # Actually for (f*g)_k, row k is:
        # (f*g)_k = sum_j f[k-j] * g_j where k-j in [0, n-1]
        # So for fixed j, f[k-j] is nonzero when k-j in [0,n-1], i.e., k in [j, j+n-1]
        # A_ub[k, j] = f[k-j] for k in [j, j+n-1]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True, 'dual_feasibility_tolerance': 1e-9}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        x = np.maximum(res.x, 0)
        return x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        rem = time_limit - elapsed
        
        if rem < 5:
            break
        
        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.2, 10.0)
            if m_relax > 5:
                # Try without relaxation
                g = solve_lp_direction(f, M, n)
                if g is None:
                    m_relax = 1.0
                    no_improve += 1
                    if no_improve > 100:
                        break
                    continue
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 60:
                # Small perturbation
                rng = np.random.default_rng(it * 104729 + 7)
                scale = 0.002 + 0.001 * min(no_improve // 60, 5)
                f = project_simplex(f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0

            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-15:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-16:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
