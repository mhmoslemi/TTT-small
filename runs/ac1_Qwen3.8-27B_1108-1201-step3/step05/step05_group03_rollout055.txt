"""
Minimizes 2*n*max(auto_conv)/sum^2 using LP-based iterative method
(multiplicative weights / sustained updates), combined with FISTA
and coordinate descent polishing. Key improvement: exact best-response
computation via LP at each iteration, which provides a theoretically

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_best_response(f, M):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)_k <= M for all k.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: max sum(g) => min -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_{i: valid} f[i] * g[k-i]
    # For each k, the coefficient of g_j is f[k-j] if 0<=k-j<n and 0<=j<n
    
    # Build constraint matrix A_ub (shape: (2n-1, n))
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n and 0 <= j < n, else 0
    
    # This matrix is banded (bidiagonal-like structure)
    # For efficiency, build it row by row
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    for k in range(n_cons):
        # g_j appears in (f*g)_k with coefficient f[k-j]
        # valid j: max(0, k-n+1) <= j <= min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(n_cons, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n)


def lp_iteration(a, n_iters=100, t_init=0.1, verbose=False):
    """
    LP-based iterative method (sustained updates / multiplicative weights).
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t = t_init
    
    for it in range(n_iters):
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP for best response
        g = lp_best_response(a, M)
        g_sum = g.sum()
        if g_sum < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / g_sum
        
        # Blend
        new_a = (1 - t) * a + t * g
        new_a = np.maximum(new_a, 0)
        new_a = new_a / new_a.sum()
        
        new_score = compute_score(new_a)
        
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
        
        a = new_a
        
        # Adaptive step size
        # If score improved, keep or increase t slightly
        # If score worsened, decrease t
        # (Actually, the LP ensures (a*g) <= M, so the blend should not worsen things too much)
        t = max(0.01, t * 0.995)  # Slowly decay
        
        if verbose and it % 20 == 0:
            print(f"      LP iter {it}: score={best_score:.10f}, M={M:.10f}, t={t:.4f}")
    
    return best_a, best_score


def coord_descent(a, time_limit, verbose=False):
    """Coordinate descent with golden section search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()

    t0 = time.monotonic()
    pass_count = 0

        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            lo = 0.0
            hi = min(0.999, max(2.0 * old_val, old_val + 0.02))
            
            def obj_v(v):
                v = max(0.0, min(v, 1.0))
                new_a = a.copy()
                new_a[j] = v
                s = new_a.sum()
                if s < 1e-15:
                    return float('inf')
                new_a = new_a / s
                conv = fast_convolve(new_a)
                return conv.max()
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 25})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    s = new_a.sum()
                    if s > 1e-15:
                        new_a = new_a / s
                        sc = compute_score(new_a)
                        if sc < best_score - 1e-12:

                            improved = True
            except Exception:
                pass
        
        if improved:
            stale = 0
        else:
            stale += 1
        
        if stale >= 3:
            break
    
    return best_a, best_score

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_averaged(a, time_limit, lr0=0.01, decay=0.9999, verbose=False):
    """Averaged subgradient with top-k peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    avg_a = np.zeros(n)
    avg_count = 0
    cur_lr = lr0
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        
        # Top-k peaks with softmax weighting
        k_actual = min(50, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        temp = max(1e-6, (pv.max() - pv.min()) * 0.3)
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            for j in range(n):
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    sg[j] += 2.0 * w[i] * a[k_minus_j]
        
        # Mirror descent (multiplicative)
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()
        
        avg_a += a
        avg_count += 1
        avg_candidate = avg_a / avg_count
        
        sc = compute_score(a)
        sc_avg = compute_score(avg_candidate)
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_candidate.copy()
        
        cur_lr *= decay
        
        if verbose and step % 2000 == 0:
            print(f"      SG step {step}: score={best_score:.10f}")
        
        if step % 10000 == 0 and cur_lr < lr0 * 0.05:
            cur_lr = lr0 * 0.3
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.002)
            avg_a = a.copy()
            avg_count = 1
    
    return best_a, best_score


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def geometric_sequence(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def raised_cosine_sequence(n, alpha=0.5):
    """Raised cosine: smooth transition from 0 to 1 and back."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, np.cos(np.pi * (x - 0.5) / (0.5 * (1 + alpha))) ** 2)
    if a.max() < 1e-15:
        a = np.sin(np.pi * x) ** 2
    return normalize(a)

    """Build the convolution matrix A such that (A @ g) = convolve(f, g)."""
    n = len(f)
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n and 0 <= j < n
    # This is a sub-block of the (2n-1) x (2n-1) Toeplitz matrix

    # The first n rows form an upper-triangular Toeplitz matrix with first column f
    # The remaining n-1 rows form a lower-triangular Toeplitz matrix
    
    A = np.zeros((2*n-1, n))

    # Row k has f[k], f[k-1], ..., f[0] in columns 0 to k

    # Row k has f[n-1], f[n-2], ..., f[n-1-(k-n)] in columns (k-n+1) to n-1

        offset = k - n
        A[k, offset+1:] = f[n-1-offset::-1]
    
    return A

    Solve the LP and return:
    - The optimal g (primal)
    - The dual variables y (giving the subgradient weights)
    """

    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        # Dual variables for the inequality constraints
        # In linprog, res.e slack gives the dual variables for A_ub @ x <= b_ub
        y = res.e.slack  # Shape: (2n-1,)
        # Normalize y to sum to 1
        y_sum = y.sum()
        if y_sum > 1e-15:

        else:

        return g, y
    else:
