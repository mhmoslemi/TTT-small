def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(f, T, n_peaks=20):
    """Compute smooth gradient of max(conv(f,f)) using softmax approximation."""
    conv = fast_convolve(f)
    n_conv = len(conv)
    k = min(n_peaks, n_conv)
    
    # Get top k peaks
    top_indices = np.argpartition(conv, -k)[-k:]
    top_vals = conv[top_indices]
    
    # Softmax weights
    scaled = top_vals / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute gradient
    # conv[k] = sum_j f[j] * f[j+k]
    # d conv[k] / d f[j] = f[j+k] + f[j-k] (with boundary)
    n = len(f)
    grad = np.zeros(n)
    
    for i in range(len(top_indices)):
        if w[i] < 1e-15:
            continue
        k_idx = int(top_indices[i])
        wi = w[i]

        # f[j + k_idx] (if j + k_idx < n) + f[j - k_idx] (if j - k_idx >= 0)
        # But actually for linear convolution:
        # conv[k] = sum_j f[j] * f[j+k] where 0<=j<n and 0<=j+k<n

        # d conv[k] / d f[j] = f[j+k] (if j+k < n) + f[j-k] (if j-k >= 0)

        # = sum_j f[j+k] * f[j]
        # If we differentiate w.r.t. f[m]:
        # - First term: f[m] * f[m+k] contributes f[m+k] if m+k < n
        # - Second term: f[m-k] * f[m] contributes f[m-k] if m-k >= 0
        # So d conv[k] / d f[m] = f[m+k] + f[m-k] (with 0 for out of bounds)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with momentum,
peak-flattening, simulated annealing, and multi-phase polynomial optimization.
Key innovations: exponential-family (multiplicative) updates instead of LP-FW,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, T, n_peaks=20):
    """Compute smooth gradient of max(conv(f,f)) using softmax approximation."""
    conv = fast_convolve(f)
    n = len(f)
    k = min(n_peaks, len(conv))

    top_indices = np.argpartition(conv, -k)[-k:]
    top_vals = conv[top_indices]

    scaled = top_vals / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(top_indices))
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(top_indices)):
        if w[i] < 1e-15:
            continue
        k_idx = int(top_indices[i])
        wi = 2.0 * w[i]

        if k_idx < n:
            grad[:n - k_idx] += wi * f[k_idx:]
        if k_idx > 0:
            grad[k_idx:] += wi * f[:n - k_idx]

    return grad, conv

def peak_flatten(f0, time_limit, n_peaks=5, alpha=0.3, verbose=False):
    """Identify top autocorrelation peaks and apply targeted flattening corrections."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        it += 1

        conv = fast_convolve(f)
        current_max = conv.max()

        # Get top peaks
        k = min(n_peaks, len(conv))
        top_idx = np.argpartition(conv, -k)[-k:]
        top_vals = conv[top_idx]
        max_val = top_vals.max()

        # For each peak, compute correction direction
        # We want to reduce conv[k] by adjusting f
        # d conv[k]/d f[j] = f[j+k] + f[j-k]
        # We want to decrease f where this derivative is large at the peak
        correction = np.zeros(n)
        for i in range(len(top_idx)):
            k_idx = int(top_idx[i])
            weight = (top_vals[i] / max_val) if max_val > 0 else 1.0
            if k_idx < n:
                correction[:n - k_idx] += weight * f[k_idx:]
            if k_idx > 0:
                correction[k_idx:] += weight * f[:n - k_idx]

        # Normalize correction
        c_norm = correction.sum()
        if c_norm < 1e-15:
            break
        correction /= c_norm

        # Apply: decrease f where correction is large, maintain sum
        # f_new_j = f_j * (1 - alpha * correction_j / max_correction)
        c_max = correction.max()
        if c_max < 1e-15:
            break
        normal = correction / c_max
        f_new = f * (1.0 - alpha * normal)
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f_new /= sn

        new_obj = fast_convolve(f_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            if verbose and it % 50 == 0:
                print(f"    PF iter {it} ({time.monotonic()-t0:.0f}s): obj={best_obj:.10f}")
        else:
            # Revert and try smaller step
            f_new2 = f * (1.0 - alpha * 0.3 * normal)
            f_new2 = np.maximum(f_new2, 0)
            s2 = f_new2.sum()
            if s2 > 1e-15:
                f_new2 /= s2
                new_obj2 = fast_convolve(f_new2).max()
                if new_obj2 < best_obj - 1e-14:
                    best_obj = new_obj2
                    best_f = f_new2.copy()
                    f = f_new2

        # Decay alpha slowly
        alpha *= 0.999

    return best_f, best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    T = 0.001
    T_min = 0.00001
    cooling = 0.9997
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        it += 1

        move_type = rng.integers(0, 5)
        f_new = f.copy()

        if move_type == 0:
            # Single coordinate bump
            j = rng.integers(0, n)
            delta = rng.normal(0, 0.01)
            f_new[j] = max(0, f_new[j] + delta)
            # Redistribute
            diff = f[j] - f_new[j]
            if abs(diff) > 1e-15:
                others = np.ones(n)
                others[j] = 0
                f_new_masked = f_new.copy()
                f_new_masked[j] = 0
                f_new_masked += diff * (f_new_masked / max(f_new_masked.sum(), 1e-15))
                f_new = f_new_masked
                f_new[j] = max(0, 1.0 - f_new_masked.sum())
                if f_new[j] < 0:
                    f_new = f
        elif move_type == 1:
            # Pair swap: increase one, decrease another
            j, k = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(0.001, 0.01)
            if f[j] + f[k] > 2 * delta + 1e-10:
                f_new[j] += delta
                f_new[k] -= delta
            else:
                f_new = f
        elif move_type == 2:
            # Block perturbation
            block_len = rng.integers(5, min(50, n // 4))
            start = rng.integers(0, n - block_len)
            noise = rng.normal(0, 0.005, block_len)
            f_new[start:start + block_len] += noise
            f_new = np.maximum(f_new, 0)
        elif move_type == 3:
            # Gaussian perturbation
            noise = rng.normal(0, 0.003, n)
            f_new = f + noise
            f_new = np.maximum(f_new, 0)
        else:
            # Mirror/anti-correlation move
            # Try to create a pattern that cancels peaks
            f_new = f + 0.005 * (f[::-1] - f)
            f_new = np.maximum(f_new, 0)

        sn = f_new.sum()
        if sn < 1e-10:
            T *= cooling
            continue
        f_new /= sn

        new_obj = fast_convolve(f_new).max()
        delta_obj = new_obj - best_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            f = f_new
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cooling
        if T < T_min:
            T = T_min

        # Restart
        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            T = 0.0005
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"    SA iter {it} ({time.monotonic()-t0:.0f}s): obj={best_obj:.10f}, T={T:.6f}")

    return best_f, best_obj


def coordinate_descent_exact(f0, time_limit, verbose=False):
    """Coordinate descent with exact 1D optimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break

            old_val = f[j]
            if old_val < 1e-15:
                # Try to increase from zero
                for trial in [1e-4, 5e-4, 1e-3]:
                    if 1.0 - trial < 1e-10:
                        continue
                    scale = (1.0 - trial) / (1.0 - old_val) if old_val < 1.0 else 1.0
                    if scale <= 0:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = trial
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    sn = f_trial.sum()
                    if sn < 1e-15:
                        continue
                    f_trial /= sn
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-12:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        old_val = trial
                        break
                continue

            # Binary search for optimal value of f[j]
            lo = max(0, old_val * 0.3)
            hi = min(0.999, old_val * 3.0)
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            for _ in range(10):
                mid = (lo + hi) / 2.0
                denom = 1.0 - old_val
                if denom < 1e-12:
                    break
                scale = (1.0 - mid) / denom
                if scale <= 0:
                    hi = mid
                    continue
                f_trial = f.copy()
                f_trial[j] = mid
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                sn = f_trial.sum()
                if sn < 1e-15:
                    continue
                f_trial /= sn
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = mid
                    lo = mid
                else:
                    hi = mid

            if best_j_val != old_val and best_j_obj < best_obj - 1e-13:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 6:
            break

    return best_f, best_obj


def momentum_mirror_descent(f0, time_limit, T0=0.0003, T_growth=1.003,
                            eta0=0.12, eta_decay=0.9999, n_peaks=40,
                            beta=0.8, restart_every=200, verbose=False):
    """Mirror descent with heavy ball momentum - more aggressive than mirror prox."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    T = T0
    eta = eta0
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        it += 1

        grad, conv = compute_smooth_grad(f, T, n_peaks)
        obj = conv.max()

        # Heavy ball: v = beta*v + grad; f <- f * exp(-eta*v) / Z
        v = beta * v + grad
        exponents = np.clip(-eta * v, -25, 25)
        exponents -= exponents.max()
        new_f = f * np.exp(exponents)
        ns = new_f.sum()
        if ns < 1e-15:
            break
        new_f /= ns

        new_obj = fast_convolve(new_f).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = new_f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MMD iter {it} ({time.monotonic()-t0:.0f}s): obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Dampen if not improving
            v *= 0.5

        f = new_f
        T *= T_growth
        eta *= eta_decay

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            T = T0 * 0.6
            eta = eta0
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_one_pass(f0, n_passes=5, verbose=False):
    """Quick LP-FW for refinement (fewer iterations than full)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    for p in range(n_passes):
        g = solve_lp_direction(f, M * 1.005, n)
        if g is None:
            g = solve_lp_direction(f, M * 1.02, n)
        if g is None:
            break

        g_s = g.sum()
        if g_s < 1e-15:
            break
        g /= g_s

        if np.max(np.abs(g - f)) < 1e-12:
            break

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()

    return best_f, best_obj
