"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smoothed mirror descent
with asymmetric initializations, LP-Frank-Wolfe refinement, and multi-length
exploration. Key insight: the optimal sequence must be asymmetric (symmetric
sequences have objective >= 2 due to center = sum a_j^2 >= 1/n).

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, c, top_frac=0.15, temperature=0.01):
    """Compute weighted subgradient of max(c) using top peaks with softmax."""
    n = len(f)
    len_c = len(c)
    k_top = max(5, int(len_c * top_frac))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        # dc[k]/da[m] = a[k-m] + a[m] for m in [js, je]
        # Note: if k = 2m, both terms are a[m], giving 2*a[m]
        contrib = f[k - idx] + f[idx]
        sg[js:je + 1] += w[i] * contrib
    return sg


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=5.0, tau_final=500.0,
                            lr_init=0.01, tau_decay=1.003,
                            lr_decay=0.9998, restart_every=200):
    """
    Adaptive smoothed mirror descent with negative entropy.
    Uses log-sum-exp for smooth max approximation.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    best_tau_for_restart = tau

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Smoothed gradient via log-sum-exp weighting
        len_c = len(conv)
        k_top = max(10, int(len_c * 0.2))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib

        # Multiplicative (mirror) descent step
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Backtracking line search
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.05:
            # Dampen the step
            for damp in [0.5, 0.25, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            best_tau_for_restart = tau
            if verbose and iter_count % 100 == 0:

        else:
            no_improve += 1

        # Adaptive parameters
        tau *= tau_decay
        lr *= lr_decay

        # Periodic restart with perturbation

            rng = np.random.default_rng(iter_count * 7 + 13)
            noise_scale = 0.003 + 0.002 * (no_improve / restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.3
            tau = best_tau_for_restart
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_targeted_descent(f0, time_limit, verbose=False):
    """
    Directly target the single worst peak. For each peak position k,
    the gradient direction tells us which pairs (j, k-j) to reduce.
    """

    f = f / f.sum()
    n = len(f)

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find the argmax position(s)
        k_best = int(np.argmax(conv))

        # Compute gradient at this peak
        js = max(0, k_best - n + 1)
        je = min(n - 1, k_best)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[js:je + 1] = f[k_best - idx] + f[idx]

        # Also consider top-3 peaks for stability
        top3 = np.argsort(conv)[-3:]
        for k2 in top3:
            k2 = int(k2)
            js2 = max(0, k2 - n + 1)

            grad[js2:je2 + 1] += 0.5 * (f[k2 - np.arange(js2, je2 + 1)] + f[np.arange(js2, je2 + 1)])

        # Multiplicative step

        f_new = np.maximum(f_new, 1e-15)

        obj_new = fast_convolve(f_new).max()

def coordinate_descent_fine(f0, time_limit, verbose=False):
    """Fine coordinate descent: for each coordinate, find optimal value via 1D search."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        rng = np.random.default_rng(int((time.monotonic() - t0) * 100) % 10000 + 1)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try perturbations
            deltas = [old_val * d for d in [-0.3, -0.2, -0.1, 0.05, 0.1, 0.2, 0.3, 0.5, -0.5]]
            for delta in deltas:
                new_val = old_val + delta
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 100)

    if variant == 0:
        # Exponential decay (front-loaded)
        a = np.exp(-np.arange(n) / (n * 0.15))
    elif variant == 1:
        # Exponential decay (back-loaded)
        a = np.exp(-np.arange(n)[::-1] / (n * 0.15))
    elif variant == 2:
        # Sine-modulated (strongly asymmetric)
        a = np.exp(-np.arange(n) / (n * 0.3)) * (1 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 0)
    elif variant == 3:
        # Ramp
        a = np.linspace(0.1, 1.0, n)
    elif variant == 4:
        # Two-bump asymmetric
        x = np.arange(n) / n
        a = np.exp(-50 * (x - 0.3) ** 2) + 0.5 * np.exp(-50 * (x - 0.7) ** 2)
    elif variant == 5:
        # Cosine window (front-heavy)
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2
        a = a + 0.1 * np.ones(n)
    elif variant == 6:
        # Random with structure
        a = np.ones(n)
        a[:n//3] *= 1.5
        a[2*n//3:] *= 0.5
        a += rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0)
    elif variant == 7:
        # Logarithmic
        a = np.log1p(np.arange(n) + 1)
    elif variant == 8:
        # Power law
        a = (np.arange(n) + 1) ** (-0.5)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: asymmetric initialization -> smoothed mirror
    descent -> LP-FW -> peak-targeted descent -> coordinate descent.
    Explores multiple sequence lengths and random restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse asymmetric initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []
    test_lengths = [384, 512, 640, 768]

    for n_try in test_lengths:
        for v in range(9):
            a_init = make_asymmetric_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"asym_{v}_n{n_try}"))
            sc = try_seq(a_init, f"init_{v}_n{n_try}")

    # Also try parent
    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Find top-3 initializations by fast_convolve objective
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label))

    scored_inits.sort(key=lambda x: x[0])
    top_inits = scored_inits[:5]
    print(f"  Top inits: {[f'{l}:{o:.4f}' for o, _, l in top_inits]}")

    # === Phase 2: Smoothed Mirror Descent on top inits ===
    print("\n=== Phase 2: Smoothed Mirror Descent ===")
    for idx, (obj_init, a_init, label) in enumerate(top_inits[:3]):
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 3 - idx), 200)
        if t_alloc < 15:
            t_alloc = 15
        print(f"  SMD on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"smd_{label}")

    # === Phase 3: Peak-Targeted Descent ===
    print("\n=== Phase 3: Peak-Targeted Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  PTD on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "ptd")

    # === Phase 4: LP-Frank-Wolfe on best ===
    print("\n=== Phase 4: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  LP-FW on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [512, 640, 896, 1024]:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.01:
                t_alloc = min(rem * 0.4, 80)
                if t_alloc < 15:
                    t_alloc = 15
                print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Random restarts with SMD ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(4):
            rem = deadline - time.monotonic() - 150
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 4 - r), 80)
            if t_alloc < 15:
                t_alloc = 15

            # Different perturbation strategies
            if r == 0:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                # Blend with asymmetric
                asym = make_asymmetric_init(n_best, variant=rng.integers(0, 9), seed=seed + r)
                f_init = 0.7 * best_seq + 0.3 * asym
                f_init = f_init / f_init.sum()
            elif r == 2:
                # Reverse blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Larger perturbation
                noise = rng.standard_normal(n_best) * 0.01
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(f_init, time_limit=t_alloc,
                                                 tau_init=3.0, tau_final=200.0,
                                                 lr_init=0.008, restart_every=100)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Coordinate descent polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 8: Final LP-FW squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 50)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final")

    # === Phase 9: One more SMD with very low temperature ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem, 30)
            print(f"  Final SMD (low tau): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                 tau_init=50.0, tau_final=500.0,
                                                 lr_init=0.003, tau_decay=1.001,
                                                 restart_every=50)
            try_seq(a_opt, "final_smd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
