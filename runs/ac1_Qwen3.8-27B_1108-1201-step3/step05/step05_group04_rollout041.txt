where `conv(a, a)` is the full linear convolution of the sequence with itself.

Key observations:
1. The objective is scale-invariant (if we scale `a` by `c`, both numerator and denominator scale by `c^2`), so we can normalize `sum(a) = 1` or any convenient value.
2. The key quantity to minimize is `max(conv(a,a)) / n` when `sum(a) = 1`.
3. This is related to the "flatness" of the auto-correlation - we want the convolution to be as flat as possible.

The theoretical lower bound is 1 (achievable when conv is perfectly flat, like a "perfect sequence"), but in practice with non-negative sequences, we're limited.

Actually, let me think differently. The key insight from the LP approach is:
- Given current f with max_conv = M, solve: maximize sum(g) subject to g >= 0 and conv(f, g) <= M
- This gives a direction that stays within the "M-ball" of the convolution constraint

This is essentially a "constrained optimization" approach. The linear program is:
- Variables: g_0, ..., g_{n-1} >= 0
- Constraints: For each i = 0, ..., 2n-2: sum_{j} f_{i-j} * g_j <= M (where the sum is over valid j)
- Objective: maximize sum(j) g_j

The optimal g will have many zero components (active constraints). This gives a direction to move in.

Actually, let me think about this more carefully. The constraint conv(f, g)_i <= M for all i means:
- For i = 0: f_0 * g_0 <= M
- For i = 1: f_0 * g_1 + f_1 * g_0 <= M
- ...
- For i = n-1: f_0 * g_{n-1} + f_1 * g_{n-2} + ... + f_{n-1} * g_0 <= M
- For i = n: f_1 * g_{n-1} + f_2 * g_{n-2} + ... + f_{n-1} * g_1 <= M
- ...
- For i = 2n-2: f_{n-1} * g_{n-1} <= M

This is a linear program with n variables and 2n-1 constraints. We want to maximize sum(g_j).

After finding g*, we rescale to sum(g*) = 1 (or whatever normalization we use), and then do a convex combination f <- (1-t)*f + t*g*.

This is essentially a "projection" onto the constraint set of points with conv <= M, followed by a step.

3. Solve LP: maximize sum(g) s.t. g >= 0, conv(f,g) <= M
4. Rescale g to sum=1
5. Update: f = (1-t)*f + t*g for small t
6. Repeat

The key advantage of this approach over the soft-max subgradient is that it uses the EXACT constraint structure, not an approximation.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
Adam optimization, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M):
    """
    Solve the LP: maximize sum(g) s.t. g >= 0 and conv(f, g) <= M.
    Returns the optimal g (normalized to sum=1 if sum>0).
    """
    n = len(f)
    # conv(f, g)_i = sum_j f_{i-j} * g_j for valid indices
    # We need: for each i = 0, ..., 2n-2: sum_j f_{i-j} * g_j <= M
    
    # Build constraint matrix A_ub where A_ub[i, j] is the coefficient of g_j in constraint i
    # Constraint i: sum over valid j of f[i-j] * g_j <= M
    # g_j is valid for constraint i when 0 <= j <= n-1 and 0 <= i-j <= n-1
    # i.e., max(0, i-(n-1)) <= j <= min(n-1, i)
    
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g_j appears in constraints i where 0 <= i-j <= n-1
        # i.e., j <= i <= j + n - 1
        lo = j
        hi = min(j + n - 1, 2 * n - 2)
        A_ub[lo:hi+1, j] = f[0:i-min(lo,2*n-2)+1]  # This is wrong, let me redo
        
    # Let me redo this more carefully
    # conv(f,g)_i = sum_{j=max(0,i-n+1)}^{min(n-1,i)} f[i-j] * g_j
    # So A_ub[i, j] = f[i-j] if 0 <= i-j <= n-1, else 0
    
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        j_lo = max(0, i - (n - 1))
        j_hi = min(n - 1, i)
        for j in range(j_lo, j_hi + 1):
            A_ub[i, j] = f[i - j]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    result = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        g = result.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def lp_direction_fast(f, M):
    """
    Faster LP direction computation using the structure of the convolution matrix.
    """
    n = len(f)
    # Build A_ub more efficiently
    # A_ub[i, j] = f[i-j] if 0 <= i-j < n, else 0
    # This is a convolution matrix (Toeplitz-like)
    
    # Actually, let's just build it directly
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j of A_ub: f[i-j] for i = j, j+1, ..., j+n-1
        # But i must be in [0, 2n-2]
        start = j
        end = min(j + n, 2 * n - 1)
        A_ub[start:end, j] = f[0:end-start]
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        g = result.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        # Subgradient: sg[j] = sum_i w[i+j] * f[i] for j=0..n-1
        # This is the correlation of w with f
        sg = np.correlate(w, f, 'valid')

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj

def lp_iteration(f, time_limit, t_hop=0.1, verbose=False):
    """
    LP-based iteration: solve LP to find direction, do convex combination.
    """
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:

    else:

    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_hop = t_hop
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        g = lp_direction_fast(f, M)
        if g is None:
            # LP failed, try with slightly larger M
            g = lp_direction_fast(f, M * 1.01)
            if g is None:
                break
        
        # Try different hop sizes
        improved = False
        for t_val in [cur_hop, cur_hop * 0.5, cur_hop * 0.25, cur_hop * 2]:
            f_trial = (1 - t_val) * f + t_val * g
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-12:
                f = f_trial
                best_obj = obj
                best_f = f.copy()
                improved = True
                break
            elif obj < fast_convolve(f).max():
                f = f_trial
                improved = True
        
        if not improved:
            # Try larger M to get a different direction
            M_relaxed = M * 1.05
            g = lp_direction_fast(f, M_relaxed)
            if g is not None:
                f_trial = (1 - cur_hop * 0.5) * f + cur_hop * 0.5 * g
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-12:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
        
        if improved:
            no_improve = 0
            cur_hop = max(cur_hop * 0.95, 0.01)
        else:
            no_improve += 1
            cur_hop = max(cur_hop * 0.8, 0.005)
            
            if no_improve > 20:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                no_improve = 0
                cur_hop = t_hop
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            multipliers = [0.5, 0.7, 0.8, 1.2, 1.4, 1.7]
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_j_val) / denom
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj

def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.999,
                            temp_init=0.008, temp_decay=0.9995, top_k=None,
                            verbose=False):
    """Mirror descent (multiplicative weights)."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr_init * lr_decay ** t, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

    """
    Solve: minimize M' s.t. sum(g)=1, g>=0, conv(f,g) <= M'
    Returns (g_opt, M_opt).
    """
    n = len(f)
    # Variables: g[0..n-1], M'
    # n+1 variables total
    # Objective: minimize M' => c = [0, 0, ..., 0, 1]
    # Constraints:
    #   sum(g) = 1 (equality)
    #   conv(f,g)_i <= M' for i = 0, ..., 2n-2 (inequality)
    #   g >= 0, M' free (but will be >= 0 automatically)

    # Row i: [A[i,0], A[i,1], ..., A[i,n-1], -1] (because conv(f,g)_i - M' <= 0)
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for i in range(2 * n - 1):

            if 0 <= idx < n:
                A_ub[i, k] = f[idx]
        A_ub[i, n] = -1.0  # coefficient of M'
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M'
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]

    bounds = [(0, None)] * n + [(0, None)]

    if result.success:

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal directions (minimizing max cross-convolution),
combined with Adam, mirror descent, line search, coordinate descent polishing,
and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Cross-convolution of f and g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build (2n-1, n) matrix A where A[i,k] = f[i-k] if 0<=i-k<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k:k + n, k] = f
    return A


def lp_find_direction(f):
    """
    Solve: minimize M' s.t. sum(g)=1, g>=0, conv(f,g) <= M'
    Returns (g_normalized, M_opt) or (None, None) on failure.
    """
    n = len(f)
    A_conv = build_conv_matrix(f)

    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    if result.success:
        g = result.x[:n]
        M_opt = result.x[n]
        return g, M_opt
    return None, None


def lp_find_direction_restricted(f, top_k=None):
    """
    LP with only the most important constraints (where conv(f,f) is largest).
    Faster for large n.
    """
    n = len(f)
    if top_k is None:
        top_k = min(n, 2 * n - 1)

    conv_ff = fast_convolve(f)
    # Select the top_k indices where conv(f,f) is largest
    perm = np.argsort(conv_ff)[::-1]
    sel = perm[:top_k]

    A_conv = build_conv_matrix(f)
    A_sel = A_conv[sel, :]

    A_ub = np.hstack([A_sel, -np.ones((len(sel), 1))])
    b_ub = np.zeros(len(sel))

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        g = result.x[:n]
        M_opt = result.x[n]
        return g, M_opt
    return None, None

def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.08)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.003 * min(no_improve // 100, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.999,
                            temp_init=0.008, temp_decay=0.9995, top_k=None):
    """Mirror descent (multiplicative weights)."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr_init * lr_decay ** t, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 250:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0

def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_j_val) / denom
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-based direction finding (primary),
    Adam/mirror descent (secondary), coordinate descent polishing,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform init n=512")

    n_parent = len(parent)

    # === Phase 1: LP-based optimization on parent ===
    print(f"\n=== Phase 1: LP optimization (n={n_parent}, best={best_score:.8f}) ===")
    t_phase1_end = time.monotonic() + 300
    rem = t_phase1_end - time.monotonic()
    t_lp = min(rem * 0.85, 270)
    n_use = n_parent
    use_restr = n_use > 1024
    tk = max(100, n_use // 2) if use_restr else None
    print(f"  LP main loop, t={t_lp:.0f}s, restricted={use_restr}")
    a_opt, obj = lp_main_loop(parent, time_limit=t_lp, verbose=True,
                              use_restricted=use_restr, top_k=tk)
    try_seq(a_opt, "p1_lp")

    # === Phase 2: Adam optimization on best ===
    print(f"\n=== Phase 2: Adam optimization (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 350
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    configs = [
        {'lr': 0.005, 'temp_init': 0.006, 'temp_decay': 0.9993, 'top_k': max(20, n_base // 10)},
        {'lr': 0.003, 'temp_init': 0.010, 'temp_decay': 0.9996, 'top_k': max(30, n_base // 8)},
        {'lr': 0.008, 'temp_init': 0.004, 'temp_decay': 0.9992, 'top_k': max(15, n_base // 15)},
    ]

    for ci, cfg in enumerate(configs):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.55, 120)
        print(f"  Adam config {ci}, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"p2_adam_{ci}")

        # Mirror descent too
        rem = t_phase2_end - time.monotonic()
        if rem < 20:
            break
        t_md = min(rem * 0.4, 60)
        a_opt, _ = mirror_descent_optimize(base, time_limit=t_md,
                                           lr_init=cfg['lr'] * 2,
                                           temp_init=cfg['temp_init'],
                                           temp_decay=cfg['temp_decay'],
                                           top_k=cfg['top_k'])
        try_seq(a_opt, f"p2_mirror_{ci}")

    # === Phase 3: LP refinement on current best ===
    print(f"\n=== Phase 3: LP refinement (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 300
    if best_seq is not None:
        rem = t_phase3_end - time.monotonic()
        t_lp2 = min(rem * 0.8, 270)
        n_b = len(best_seq)
        use_r2 = n_b > 1024
        tk2 = max(100, n_b // 2) if use_r2 else None
        print(f"  LP refinement, t={t_lp2:.0f}s")
        a_opt, obj = lp_main_loop(best_seq, time_limit=t_lp2, verbose=True,
                                  use_restricted=use_r2, top_k=tk2)
        try_seq(a_opt, "p3_lp")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 250
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    target_lengths = [512, 640, 768, 1024, 1280, 1536]
    target_lengths = [t for t in target_lengths if t != n_base and 128 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - n_base))
    target_lengths = target_lengths[:3]

    for target_n in target_lengths:
        rem = t_phase4_end - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 80)
        print(f"  n={target_n}: Adam t={t_alloc:.0f}s")

        a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc,
                                 lr=0.005, temp_init=0.006,
                                 temp_decay=0.9994, top_k=max(20, target_n // 12))
        try_seq(a_opt, f"p4_n{target_n}_adam")

        rem = t_phase4_end - time.monotonic()
        if rem > 50:
            t_lp_n = min(rem * 0.4, 60)
            use_r_n = target_n > 1024
            tk_n = max(100, target_n // 2) if use_r_n else None
            print(f"  n={target_n}: LP t={t_lp_n:.0f}s")
            a_opt, _ = lp_main_loop(f_interp, time_limit=t_lp_n,
                                    use_restricted=use_r_n, top_k=tk_n)
            try_seq(a_opt, f"p4_n{target_n}_lp")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 150
    if best_seq is not None:
        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.8, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_cd")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_alloc = min(rem * 0.4, 60)
            print(f"  Fine LP, t={t_alloc:.0f}s")
            use_r6 = n_best > 1024
            tk6 = max(100, n_best // 2) if use_r6 else None
            a_opt, _ = lp_main_loop(best_seq, time_limit=t_alloc,
                                    use_restricted=use_r6, top_k=tk6)
            try_seq(a_opt, "p6_lp_fine")

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem * 0.5, 40)
            print(f"  Fine Adam, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc,
                                     lr=0.001, temp_init=0.002,
                                     temp_decay=0.9998, top_k=max(10, n_best // 25))
            try_seq(a_opt, "p6_fine_adam")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for pi in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.5, 25)
            a_opt, _ = lp_main_loop(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"p6_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
