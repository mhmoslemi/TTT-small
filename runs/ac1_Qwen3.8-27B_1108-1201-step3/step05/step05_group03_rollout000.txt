def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    s = a.sum()
    if s < 1e-15:

    conv = fast_convolve(a)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def find_impactful_coords(a, conv, n_top_peaks=5, n_top_coords=60):
    """Find coordinates that most impact the top peaks."""
    n = len(a)
    peak_indices = np.argsort(conv)[::-1][:n_top_peaks]
    # For each peak k*, coordinate j impacts it via a[j]*a[k*-j]
    impact = np.zeros(n)
    for k_star in peak_indices:
        k_star = int(k_star)
        for j in range(n):
            other = k_star - j
            if 0 <= other < n:
                impact[j] += a[j] * a[other]
    # Return top n_top_coords by impact
    top_coords = np.argsort(impact)[::-1][:n_top_coords]
    return top_coords, peak_indices

def subgradient_focused(a, time_limit, top_k=20, verbose=False):
    """Subgradient method with mirror descent, focused on top peaks."""
    n = len(a)
    a = normalize(a.copy())
    conv = fast_convolve(a)
    best_score = compute_score(a)
    best_a = a.copy()

    avg_a = a.copy()
    avg_count = 1
    cur_lr = 0.01
    decay = 0.9998
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Softmax weights
        temp = max(1e-6, pv.max() - pv.min()) * 0.3
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Mirror descent (multiplicative weights)
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()

        # Update average
        avg_a += a
        avg_count += 1

        conv = fast_convolve(a)
        sc = compute_score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        avg_candidate = avg_a / avg_count
        sc_avg = compute_score(avg_candidate)
        if sc_avg < best_score - 1e-12:
            best_score = sc_avg
            best_a = avg_candidate.copy()
            conv = fast_convolve(best_a)

        cur_lr *= decay

        # Restart if stuck
        if time.monotonic() - t0 > 50 and (time.monotonic() - t0) % 60 < 0.1:
            if cur_lr < 0.001:
                cur_lr = 0.008
                a = project_simplex(best_a + np.random.standard_normal(n) * 0.002)
                conv = fast_convolve(a)
                avg_a = a.copy()
                avg_count = 1

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

    target_ns = sorted(set([n_best - 128, n_best - 64, n_best - 32,
                            n_best + 32, n_best + 64, n_best + 128]))
    target_ns = [t for t in target_ns if 256 <= t <= 2048]

    for target_n in target_ns:
        rem = deadline - time.monotonic() - 100
        if rem < 60:
            break
        t_alloc = min(rem * 0.4, 180)
        a_interp = interpolate_sequence(best_a, target_n)
        a_opt, sc = optimize_multi_stage(a_interp, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"dim_{target_n}")

    # === Phase 4: Aggressive final polishing ===
    print(f"\n=== Phase 4: Final polishing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        # Very focused CD
        t_cd = min(rem * 0.5, 250)
        a_opt, sc = coord_descent_focused(best_a, time_limit=t_cd,
                                          n_top_peaks=10, n_top_coords=150)
        try_seq(a_opt, "final_focused_cd")

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            # Full CD
            t_cd2 = min(rem * 0.6, 150)
            a_opt, sc = full_coord_descent(best_a, time_limit=t_cd2)
            try_seq(a_opt, "final_full_cd")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Tiny perturbations + quick optimize
            for trial in range(5):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                n_b = len(best_a)
                noise = rng.standard_normal(n_b) * 0.001
                a_pert = project_simplex(best_a + noise)
                a_opt, sc = coord_descent_focused(a_pert, time_limit=min(rem, 30),
                                                  n_top_peaks=8, n_top_coords=100)
                try_seq(a_opt, f"micro_pert_{trial}")

    # === Final verification ===
    if best_a is None:
        best_a = normalize(np.ones(768))
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score
        final_list = [float(x) for x in best_a]

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(final_list)})")
    return final_list
