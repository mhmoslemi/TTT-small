import numpy as np
import scipy.sparse as sp
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build sparse matrix C such that C @ b = conv(f, b) (full)."""
    n = len(f)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    C = sp.csr_matrix((vals, (rows, cols)), shape=(m, n))
    return C

def solve_lp(f, M):
    """Solve: max sum(b) s.t. C@b <= M, b >= 0. Returns b and dual y."""
    n = len(f)
    C = build_conv_matrix(f)
    b = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b))
    constraints = [C @ b <= M, b >= 0]
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            prob.solve(solver=cp.ECOS)
    
    if b.value is None:
        return None, None
    
    b_val = np.maximum(b.value, 0)
    # Get dual variables
    y_val = None
    try:
        y_val = constraints[0].dual_value
        if y_val is not None:
            y_val = np.maximum(y_val, 0)
    except:
        pass
    
    return b_val, y_val

"""
LP-based subgradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the LP subproblem to find optimal descent directions, with line search
for step size. Includes multi-start, length exploration, and coordinate descent
polishing.

import numpy as np
import scipy.sparse as sp
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def build_conv_matrix(f):
    """Build sparse matrix C such that C @ b = full_conv(f, b)."""
    n = len(f)
    m = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)  # overestimate
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for k in range(m):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            if f[i] > 1e-15:  # skip zeros
                rows[count] = k
                cols[count] = i
                vals[count] = f[i]
                count += 1
    C = sp.csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(m, n))
    return C


def solve_lp_direction(f, M, solver='CLARABEL'):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b, s."""
    n = len(f)
    C = build_conv_matrix(f)
    b_var = cp.Variable(n)
    obj = cp.Maximize(cp.sum(b_var))
    cons = [C @ b_var <= M, b_var >= 0]
    prob = cp.Problem(obj, cons)
    
    solved = False
    for s in [solver, 'SCS', 'ECOS', 'CLARABEL']:
        if s == solver and solved:
            break
        try:
            prob.solve(solver=s)
            if b_var.value is not None:
                solved = True
                break
        except Exception as e:
            continue
    
    if not solved or b_var.value is None:
        return None, None
    
    b_val = np.maximum(b_var.value, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val, s


def line_search(f, g, M, max_iters=8):
    """Find best t in [0,1] for f_new = (1-t)*f + t*g."""
    best_t = 0.0
    best_obj = M
    for t in [1.0, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
        f_new = (1 - t) * f + t * g
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj, f_new if best_t > 0 else f.copy()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent polishing."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_M - 1e-14:
                    best_M = val
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_f, best_M


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_bump_init(n, center=0.5, sigma=0.2):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

"""
LP-based subgradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses scipy HiGHS LP solver to find optimal descent directions, with adaptive
line search, multi-start initializations, length exploration, and coordinate
descent polishing. The LP subproblem: max sum(b) s.t. conv(f,b)<=M, b>=0,
provides the exact optimal direction (dual gives exact subgradient).
"""

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def build_conv_matrix(f):
    """Build sparse matrix C such that C @ b = full_conv(f, b)."""
    n = len(f)
    m = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for k in range(m):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        length = i_max - i_min + 1
        if length > 0:
            rows[count:count + length] = k
            cols[count:count + length] = np.arange(i_min, i_max + 1)
            vals[count:count + length] = f[i_min:i_max + 1]
            count += length
    C = sp.csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(m, n))
    return C


def solve_lp_direction(f, M):
    """Solve: max sum(b) s.t. C@b <= M, b >= 0. Returns b or None."""
    n = len(f)
    C = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = opt.linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search(f, g, M):
    """Find best t for f_new = (1-t)*f + t*g to minimize max conv."""
    best_t = 0.0
    best_obj = M
    best_fn = f.copy()
    for t in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
        f_new = (1.0 - t) * f + t * g
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_fn = f_new
    return best_t, best_obj, best_fn


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_direction(f):
    """Get exact subgradient direction from argmax of conv(f,f)."""
    n = len(f)
    conv = fast_convolve(f)
    k_max = np.argmax(conv)
    g = np.zeros(n)
    i_min = max(0, k_max - n + 1)
    i_max = min(n - 1, k_max)
    g[i_min:i_max + 1] = f[k_max - i_min:k_max - i_max + 1]
    s = g.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return g / s


def lp_optimize(f0, time_limit, verbose=False, momentum=0.0):
    """LP-based optimization with line search and optional momentum averaging."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    avg_g = None

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        b = solve_lp_direction(f, M)

        if b is not None:
            s_b = b.sum()
            if s_b > 1e-15:
                g = b / s_b
            else:
                g = subgradient_direction(f)
        else:
            g = subgradient_direction(f)

        if momentum > 0 and avg_g is not None:
            g = (1.0 - momentum) * g + momentum * avg_g
            sg = g.sum()
            if sg > 1e-15:
                g = g / sg
        avg_g = g if avg_g is None else (0.7 * avg_g + 0.3 * g)

        t, obj, f_new = line_search(f, g, M)

        if obj < best_M - 1e-16:
            best_M = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and iter_count % 200 == 0:

        else:
            no_improve += 1

        f = f_new
        conv = fast_convolve(f)
        M = conv.max()

        if no_improve > 80:

            noise = 0.002 + 0.001 * min(no_improve // 40, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            avg_g = None

    return best_f, best_M


def adam_optimize(f0, time_limit, lr=0.003, temp_init=0.005, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam with soft-max subgradients (backup optimizer)."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):

    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_M

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                total = f_trial.sum()
                if total < 1e-15:
                    continue
                f_trial /= total

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    total = f.sum()
                    if total > 1e-15:
                        f /= total
                if best_j_obj < best_M - 1e-14:
                    best_M = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 1:
            break

    return best_f, best_M


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_bump_init(n, center=0.5, sigma=0.2):
    """Gaussian bump initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_power_init(n, p=2.0):
    """Power-law shaped initialization."""
    x = np.linspace(0, 1, n)
    f = (1 - np.abs(2 * x - 1)) ** p
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-based subgradient optimization with line search, multi-start,
    length exploration, Adam backup, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}", flush=True)
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_bump_init(768, 0.5, 0.22)
        print("No parent found, using bump init n=768", flush=True)

    n_parent = len(parent)

    # === Phase 1: Primary LP optimization on parent (long) ===
    print(f"\n=== Phase 1: Primary LP (n={n_parent}) ===", flush=True)
    t_p1_end = time.monotonic() + 500

    a_opt, M_opt = lp_optimize(parent, time_limit=t_p1_end - time.monotonic(),
                               verbose=True, momentum=0.0)
    try_seq(a_opt, "p1_lp_parent")

    # === Phase 2: LP with momentum ===
    print(f"\n=== Phase 2: LP with momentum (best={best_score:.8f}) ===", flush=True)
    t_p2_end = time.monotonic() + 200

    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    a_opt, _ = lp_optimize(base, time_limit=min(t_p2_end - time.monotonic(), 100),
                           verbose=False, momentum=0.3)
    try_seq(a_opt, "p2_mom0.3")

    a_opt, _ = lp_optimize(base, time_limit=min(t_p2_end - time.monotonic(), 100),
                           verbose=False, momentum=0.5)
    try_seq(a_opt, "p2_mom0.5")

    # === Phase 3: Multi-start LP from diverse initializations ===
    print(f"\n=== Phase 3: Multi-start LP (best={best_score:.8f}) ===", flush=True)
    t_p3_end = time.monotonic() + 250

    initializations = [
        ('perturb_small', project_simplex(base + rng.standard_normal(n_base) * 0.008)),
        ('perturb_med', project_simplex(base + rng.standard_normal(n_base) * 0.02)),
        ('bump_narrow', make_bump_init(n_base, 0.5, 0.15)),
        ('bump_wide', make_bump_init(n_base, 0.5, 0.35)),
        ('power2', make_power_init(n_base, 2.0)),
        ('power4', make_power_init(n_base, 4.0)),
        ('mix_uniform', project_simplex(0.4 * base + 0.6 * np.ones(n_base) / n_base)),
        ('mix_bump', project_simplex(0.5 * base + 0.5 * make_bump_init(n_base, 0.5, 0.25))),
    ]

    for name, init in initializations:
        rem = t_p3_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.35, 60)
        print(f"  LP '{name}' t={t_alloc:.0f}s", flush=True)
        a_opt, _ = lp_optimize(init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"p3_{name}")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===", flush=True)
    t_p4_end = time.monotonic() + 200

    target_lengths = [384, 512, 640, 896, 1024, 1280, 1536]
    target_lengths = [t for t in target_lengths if t != n_base and 256 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - n_base))
    target_lengths = target_lengths[:4]

    for target_n in target_lengths:
        rem = t_p4_end - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 70)
        print(f"  LP n={target_n} t={t_alloc:.0f}s", flush=True)
        a_opt, _ = lp_optimize(f_interp, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"p4_n{target_n}")

    # === Phase 5: Adam refinement on best ===
    print(f"\n=== Phase 5: Adam refinement (best={best_score:.8f}) ===", flush=True)
    t_p5_end = time.monotonic() + 150

    if best_seq is not None:
        n_best = len(best_seq)
        starts = [
            ('direct', best_seq.copy()),
            ('perturb', project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)),
            ('mix_gauss', project_simplex(0.7 * best_seq + 0.3 * make_bump_init(n_best, 0.5, 0.2))),
        ]
        for sname, sinit in starts:
            rem = t_p5_end - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 50)
            print(f"  Adam '{sname}' t={t_alloc:.0f}s", flush=True)
            a_opt, _ = adam_optimize(sinit, time_limit=t_alloc,
                                     lr=0.003, temp_init=0.004,
                                     temp_decay=0.9996,
                                     top_k=max(20, n_best // 12))
            try_seq(a_opt, f"p5_adam_{sname}")

    # === Phase 6: Coordinate descent polishing ===
    print(f"\n=== Phase 6: CD polishing (best={best_score:.8f}) ===", flush=True)
    t_p6_end = time.monotonic() + 120

    if best_seq is not None:
        rem = t_p6_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.6, 80)
            print(f"  CD t={t_cd:.0f}s", flush=True)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p6_cd")

            rem = t_p6_end - time.monotonic()
            if rem > 20 and best_seq is not None:
                n_best = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                a_opt, _ = coordinate_descent_polish(perturbed, time_limit=min(rem * 0.5, 30))
                try_seq(a_opt, "p6_cd_pert")

    # === Phase 7: Final LP refinement ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.8f}) ===", flush=True)

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 30:
            t_alloc = min(rem * 0.7, 100)
            print(f"  Final LP t={t_alloc:.0f}s", flush=True)
            a_opt, _ = lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "p7_final_lp")

            rem = deadline - time.monotonic() - 5
            if best_seq is not None and rem > 15:
                a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem * 0.6, 25))
                try_seq(a_opt, "p7_cd")

    # Fallback
    if best_seq is None:
        best_seq = make_bump_init(384, 0.5, 0.2)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
