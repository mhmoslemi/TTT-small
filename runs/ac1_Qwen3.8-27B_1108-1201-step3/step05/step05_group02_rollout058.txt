        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-15:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            # Try several ratio adjustments
            for ratio in [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]:
                new_val = old_val * ratio
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[np.arange(n) != j] *= scale
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.arange(n) != j
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_a = a.copy()
                    improved = True

        if not improved and passes > 5:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def squeeze_optimize(a0, time_limit):
    """High-precision local optimization using very tight LP steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Very tight LP with top 15 constraints
        m = 15
        peak_lags = np.argpartition(conv, -m)[-m:]
        A_ub = build_sparse_lp_matrix(a, peak_lags)
        b_ub = np.full(m, M * 1.0005)
        c_vec = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'time_limit': min(10, rem * 0.5), 'presolve': True})
        except Exception:
            res = None

        if res is None or not res.success or res.x is None:
            # Try with more relaxation
            b_ub = np.full(m, M * 1.01)
            try:
                res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                              method='highs',
                              options={'time_limit': min(10, rem * 0.5), 'presolve': True})
            except Exception:
                res = None

        if res is None or not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-14:
            break
        g = g / g_sum

        if np.max(np.abs(g - a)) < 1e-14:
            break

        # Small step
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3]:
            ft = a + alpha * (g - a)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < M - 1e-18:
                a = ft
                if val < best_obj - 1e-18:
                    best_obj = val
                    best_a = a.copy()
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining accelerated subgradient, sparse LP-FW,
    multi-length exploration, coordinate descent, and final squeezing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    print(f"\n=== Phase 1: Quick SG on parent n={n_parent} (best={best_score:.6f}) ===")

    if t1 > 5:
        print(f"  Budget: {t1:.0f}s")

    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-96, -64, -32, 32, 64, 96, 128, -128, 256, -256, 512]:
        nl = n_best + delta
        if 200 <= nl <= 4096 and nl != n_best:
            candidate_lengths.append(nl)
    # Also try some absolute lengths
    for nl in [384, 512, 640, 768, 896, 1024, 1280]:
        if 200 <= nl <= 4096 and nl != n_best and nl not in candidate_lengths:
            candidate_lengths.append(nl)

    # Sort by closeness to current best length
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    for target_n in candidate_lengths[:8]:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.15)
        if t_alloc < 10:

        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_sparse(f_interp, time_limit=t_alloc, m_constraints=40)

    # === Phase 6: Subgradient polish ===

    n_best = len(best_seq)
