def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, tau, top_k):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k_actual)[-k_actual:]

    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peak_idx))
        w[np.argmax(pv)] = 1.0

        w /= ws

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.6f}")

    # === Phase 1: Main accelerated subgradient ===
    print(f"\n=== Phase 1: Main AccSubGrad (n={len(parent)}) ===")
    t_rem = deadline - time.monotonic()
    t1 = min(t_rem * 0.28, 320)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = acc_subgrad(parent, time_limit=t1, lr=0.004, mom=0.88,
                               tau=0.0004, top_k=150, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Phase 1 best: {sc:.6f}")

    # === Phase 2: Hyperparameter sweep ===
    print(f"\n=== Phase 2: Hyperparameter sweep (best={best_score:.6f}) ===")
    configs = [
        (0.002, 0.91, 0.0002, 80),
        (0.006, 0.86, 0.0007, 200),
        (0.001, 0.94, 0.0001, 50),
        (0.003, 0.89, 0.0003, 100),
        (0.005, 0.87, 0.0005, 130),
    ]
    for ci, (lr_c, mom_c, tau_c, tk_c) in enumerate(configs):
        rem = deadline - time.monotonic() - 130
        if rem < 20:
            break
        t_alloc = min(rem * 0.22, 110)
        if t_alloc < 10:
            continue
        a_opt, _ = acc_subgrad(best_seq, time_limit=t_alloc, lr=lr_c,
                               mom=mom_c, tau=tau_c, top_k=tk_c)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Config {ci} (lr={lr_c},mom={mom_c},tau={tau_c}): {sc:.6f}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.6f}) ===")
    for delta in [64, -64, 128, -128, 192]:
        rem = deadline - time.monotonic() - 100
        if rem < 20:
            break
        tn = len(best_seq) + delta
        if tn < 64 or tn > 2048:
            continue
        fi = interp_seq(best_seq, tn)
        t_alloc = min(rem * 0.2, 80)
        if t_alloc < 10:
            continue
        a_opt, _ = acc_subgrad(fi, time_limit=t_alloc, lr=0.003, mom=0.88,
                               tau=0.0003, top_k=100)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  n={tn}: {sc:.6f}")

    # === Phase 4: Noisy restarts ===
    print(f"\n=== Phase 4: Noisy restarts (best={best_score:.6f}) ===")
    scales = [0.002, 0.004, 0.006, 0.01, 0.015]
    for trial in range(len(scales)):
        rem = deadline - time.monotonic() - 70
        if rem < 15:
            break
        scale = scales[trial]
        fi = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
        t_alloc = min(rem * 0.3, 100)
        if t_alloc < 10:
            continue
        a_opt, _ = acc_subgrad(fi, time_limit=t_alloc, lr=0.003, mom=0.88,
                               tau=0.0003, top_k=100)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Trial {trial} (scale={scale}): {sc:.6f}")
            # Quick follow-up with tighter params
            rem2 = deadline - time.monotonic() - 50
            if rem2 > 10:
                a_opt2, _ = acc_subgrad(best_seq, time_limit=min(rem2 * 0.4, 40),
                                        lr=0.001, mom=0.94, tau=0.0001, top_k=40)
                sc2 = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt2.copy()
                    print(f"    Follow-up: {sc2:.6f}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 40
    if rem > 20:
        t_cd = min(rem * 0.5, 100)
        print(f"  Budget: {t_cd:.0f}s, n={len(best_seq)}")
        a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  CD: {sc:.6f}")

    # === Phase 6: Final subgradient squeeze ===
    print(f"\n=== Phase 6: Final Squeeze (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_final = min(rem * 0.85, 60)
        a_opt, _ = acc_subgrad(best_seq, time_limit=t_final, lr=0.0008, mom=0.95,
                               tau=0.00008, top_k=30)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Final: {sc:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
