def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Fast LP Frank-Wolfe using active-set LP (top-K violated constraints).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    # Start with aggressive top_k, increase as we get closer
    base_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Dynamically adjust top_k: use more constraints as we converge
        current_top_k = min(base_top_k + it // 10, 2 * n - 1)
        
        g = fast_active_lp(f, M * m_exp, current_top_k, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                # Try full LP
                g = full_lp_direction(f, M * m_exp, n)
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.03, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Refine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      LP-FW fast iter {it}: obj={best_obj:.10f}, top_k={current_top_k}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 100:
            rng_p = np.random.default_rng(it * 17 + 3)
            scale = 0.002 + 0.003 * min(no_improve // 50, 4)
            noise = rng_p.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def smooth_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Log-sum-exp smooth optimization with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = max(T_start * (T_end / T_start) ** (elapsed / time_limit), T_end)
        lr = alpha * (0.5 ** (elapsed / (time_limit * 0.7)))
        
        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()
        
        # Gradient via cross-correlation
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(f, N)
        W_fft = np.fft.rfft(w, N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        grad = (2.0 / T) * corr
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      Smooth iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        if no_improve > 4000:
            seed_val = int(elapsed * 997) % 1000000
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj


def surgical_cd(f0, time_limit, verbose=False):
    """
    Surgical coordinate descent: focus on the most impactful coordinates.
    Optimizes one coordinate at a time using golden section, but prioritizes
    coordinates with the largest gradient magnitude.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    ORDER = 0.6180339887498949
    
    def eval_coord(f_base, j, x):
        """Evaluate max conv when f[j]=x, others scaled to maintain sum=1."""
        old_val = f_base[j]
        denom = 1.0 - old_val
        if denom < 1e-10:
            return 1e10
        if x < 0 or x >= 1.0:
            return 1e10
        scale = (1.0 - x) / denom
        f_trial = f_base.copy()
        f_trial[j] = x
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        f_trial[mask] *= scale
        return fast_convolve(f_trial).max()
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        # Compute priority scores (gradient proxy)
        conv = fast_convolve(f)
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(f, N)
        W_fft = np.fft.rfft(conv / conv.max(), N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        priority = np.abs(corr)
        
        # Process in priority order
        order = np.argsort(-priority)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-8:
                continue
            
            # Golden section search
            lo = 0.0
            hi = 0.999
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            fc = eval_coord(f, j, c)
            fd = eval_coord(f, j, d)
            
            for _ in range(16):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = eval_coord(f, j, c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = eval_coord(f, j, d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = eval_coord(f, j, opt_val)
            
            if opt_obj < best_obj - 1e-14:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True
        
        if not improved and passes > 4:
            break
    
    return best_f, best_obj


def make_chirp(n):
    """Non-negative chirp-like sequence with good autocorrelation properties."""
    x = np.arange(n, dtype=np.float64)
    # Use |cos| of quadratic phase (Zadoff-Chu inspired)
    phase = 2.0 * np.pi * x * x / (2.0 * n)
    a = np.abs(np.cos(phase)) + 0.1
    return a / a.sum()


def make_bipolar_abs(n):
    """Absolute value of a bipolar-like sequence."""
    x = np.arange(n, dtype=np.float64)
    a = np.abs(np.sin(np.pi * x / n) * np.cos(2 * np.pi * x / n)) + 0.05
    return a / a.sum()


def make_geometric(n, r):
    """Geometric sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def make_cosine(n, power=2.0):
    """Cosine window."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def make_two_peak(n, ratio=0.3):
    """Two-peak sequence."""
    a = np.zeros(n)
    center1 = n * ratio
    center2 = n * (1.0 - ratio)
    sigma = n * 0.15
    a += np.exp(-0.5 * ((np.arange(n) - center1) / sigma) ** 2)
    a += np.exp(-0.5 * ((np.arange(n) - center2) / sigma) ** 2)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer combining fast active-set LP Frank-Wolfe,
    smooth gradient optimization, and surgical coordinate descent.
    Uses diverse initializations including chirp-inspired sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass
    
    n_parent = len(parent) if parent is not None else 768
    
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")
    
    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")
    
    n_options = [n_parent, 512, 768, 1024]
    
    for n_try in n_options:
        # Geometric
        for r in [0.97, 0.98, 0.99, 0.995]:
            _try(make_geometric(n_try, r), f"geo_r{r}_n{n_try}")
        
        # Cosine
        for pw in [1.5, 2.0, 3.0, 5.0]:
            _try(make_cosine(n_try, power=pw), f"cos_p{pw}_n{n_try}")
        
        # Chirp
        _try(make_chirp(n_try), f"chirp_n{n_try}")
        
        # Bipolar abs
        _try(make_bipolar_abs(n_try), f"bipolar_n{n_try}")
        
        # Two peak
        for ratio in [0.25, 0.35, 0.40]:
            _try(make_two_peak(n_try, ratio), f"2peak_r{ratio}_n{n_try}")
    
    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # === Phase 1: Fast LP-FW on best ===
    print(f"\n=== Phase 1: Fast LP-FW (best={best_score:.6f}) ===")
    base = best_seq.copy()
    
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem, 250)
        print(f"  Fast LP-FW n={len(base)}, t={t_alloc:.0f}s, top_k=40")
        a_opt, _ = lpfw_fast(base, time_limit=t_alloc, top_k=40, verbose=True)
        _try(a_opt, "lpfw_fast_p1")
    
    # Smooth optimization
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        t_alloc = min(rem, 120)
        print(f"  Smooth opt n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize(best_seq, time_limit=t_alloc, T_start=0.2, T_end=0.008, verbose=True)
        _try(a_opt, "smooth_p1")
    
    # === Phase 2: Second round LP-FW with different top_k ===
    print(f"\n=== Phase 2: LP-FW round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        t_alloc = min(rem, 150)
        print(f"  Fast LP-FW top_k=80, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_fast(best_seq, time_limit=t_alloc, top_k=80, verbose=True)
        _try(a_opt, "lpfw_fast_p2")
    
    # LP-FW with perturbation
    rem = deadline - time.monotonic() - 300
    if rem > 15:
        noise = rng.standard_normal(len(best_seq)) * 0.004
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 100)
        a_opt, _ = lpfw_fast(f_pert, time_limit=t_alloc, top_k=40)
        _try(a_opt, "lpfw_pert")
    
    # === Phase 3: Surgical CD ===
    print(f"\n=== Phase 3: Surgical CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15:
        t_cd = min(rem * 0.5, 80)
        print(f"  Surgical CD t={t_cd:.0f}s")
        a_opt, _ = surgical_cd(best_seq, time_limit=t_cd)
        _try(a_opt, "surg_cd")
    
    # === Phase 4: Try longer sequences ===
    print(f"\n=== Phase 4: Longer sequences (best={best_score:.6f}) ===")
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 100
        if rem < 40:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        
        t1 = min(rem * 0.35, 60)
        a_opt, _ = lpfw_fast(f_interp, time_limit=t1, top_k=30)
        _try(a_opt, f"lpfw_n{target_n}")
        
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t2 = min(rem * 0.5, 50)
            a_opt, _ = smooth_optimize(f_interp, time_limit=t2, T_start=0.2, T_end=0.03)
            _try(a_opt, f"smooth_n{target_n}")
    
    # === Phase 5: Final polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Smooth with low temperature
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = smooth_optimize(best_seq, time_limit=min(rem, 50),
                                       T_start=0.03, T_end=0.003, verbose=True)
            _try(a_opt, "smooth_fine")
        
        # LP-FW final
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = lpfw_fast(best_seq, time_limit=min(rem, 40), top_k=20)
            _try(a_opt, "lpfw_final")
        
        # Surgical CD final
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = surgical_cd(best_seq, time_limit=min(rem, 25))
            _try(a_opt, "surg_cd_final")
        
        # Multiple perturbations
        for i in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            noise = rng.standard_normal(len(best_seq)) * (0.001 + 0.001 * i)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lpfw_fast(f_pert, time_limit=min(rem, 15), top_k=30)
            _try(a_opt, f"perturb_{i}")
        
        # Reversed
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = lpfw_fast(f_rev, time_limit=min(rem, 10), top_k=30)
            _try(a_opt, "reversed")
        
        # Blend with uniform
        rem = deadline - time.monotonic() - 5
        if rem > 4:
            n_best = len(best_seq)
            f_blend = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
            f_blend = project_simplex(f_blend)
            a_opt, _ = lpfw_fast(f_blend, time_limit=min(rem, 8), top_k=30)
            _try(a_opt, "blend")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
