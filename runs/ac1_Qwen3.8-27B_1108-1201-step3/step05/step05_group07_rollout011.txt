import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the normalized objective: max(conv(a,a)) / sum(a)^2 * 2n"""
    b = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * b.max() / (s * s)

        js = max(0, k - n + 1)

    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(f, M, t=0.3):
    """
    One LP-based iteration: solve LP, rescale, and blend.
    """
    g = lp_refine_step(f, M)
    if g is None or g.sum() < 1e-15:
        return f, M
    
    # Rescale g to match sum of f
    s_f = f.sum()
    s_g = g.sum()
    g = g * (s_f / s_g)
    
    # Blend
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    
    M_new = fast_convolve(f_new).max() * (f_new.sum() / s_f) ** 2
    # Actually let's keep the tracking simple
    return f_new, fast_convolve(f_new).max()


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        
        # Weights
        w = exp_Tb / exp_Tb.sum()
        
        # grad_a = 2n * cross_corr(w, a)
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i]
        # = sum_k w[k] * a[k-i]
        # Using FFT: IFFT(FFT(w) * conj(FFT(a)))? No...
        # cross_corr(w, a) = ifft(conj(fft(w)) * fft(a))
        # But we need the "non-circular" version
        
        # Actually, let's compute it directly using the convolution structure
        # (w * a)(k) = sum_i w[i] * a[k-i]
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] = (w * a_flipped.reverse)(i + len(w) - 1)
        # Hmm, let me just use a direct FFT approach
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i]
        # This is the correlation of w and a
        # = ifft(conj(FFT(w)) * FFT(a)) but circular
        # For linear, we need zero-padding
        cross = irfft(np.conj(W) * A, fft_size)[:n]
        
        grad_a = 2.0 * n * cross
        
        # Chain rule for softmax
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        
        return grad_z
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)

def structured_start(n, rng):
    """Generate a structured random sequence."""
    method = rng.integers(0, 4)
    if method == 0:
        # Random
        a = rng.random(n)
    elif method == 1:

        a = rng.random(n)

    elif method == 2:
        # Quadratic residue inspired

        for x in range(q):
            residues.add((x * x) % q)
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 if i in residues else 0.3
    elif method == 3:
        # Fourier-based: magnitude of random spectrum
        spectrum = rng.standard_normal(n) + 1j * rng.standard_normal(n)

        a = np.abs(a) + 0.1

def _is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True


def peak_shave(a, n_peaks=5, n_steps=200, time_limit=30.0):
    """Iterative peak shaving."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max() / (s * s)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = [k for k in peak_indices if b[k] > best_obj * (s*s) * 0.95]
        
        if not peak_indices:
            break
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i
            
            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                factor = 0.95
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                mask = np.ones(n, dtype=bool)
                mask[target] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[target] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max() / (a_trial.sum()**2)
                if val < best_obj - 1e-14:
                    a = a_trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent with multipliers."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max() / (s*s)
    t0 = time.monotonic()
    
    multipliers = np.array([0.5, 0.7, 1.0, 1.3, 1.5, 1.8])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max() / (a_trial.sum()**2)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
                else:
                    # Revert if no improvement
                    a = best_a.copy()
    
    return best_a


def subgradient_polish(a0, time_limit=60.0, lr=0.005, mom=0.9, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective(a)
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        co = compute_objective(a)
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.01
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_a

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement,
soft-max L-BFGS with temperature annealing, multi-length exploration,
simulated annealing, coordinate descent, and subgradient polishing.
Key innovations: LP-guided structural improvement, adaptive simulated annealing,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute normalized objective: 2n*max(conv)/(sum^2)."""
    b = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * b.max() / (s * s)


def build_conv_matrix(f):
    """Build the convolution matrix A where (A@g)[k] = (f*g)(k)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A


def lp_solve_max_sum(f, M):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(f, t=0.2, delta=0.0):
    """
    One LP-based iteration. Returns (f_new, improved).
    Solves max sum(g) s.t. f*g <= M*(1-delta), g>=0, then blends.
    """
    b = fast_convolve(a=f)
    s = f.sum()
    f_norm = f / s
    b_norm = fast_convolve(f_norm)
    M = b_norm.max()

    # Solve LP with current peak as constraint
    g = lp_solve_max_sum(f_norm, M * (1.0 - delta))
    if g is None or g.sum() < 1e-15:
        return f, False

    # Rescale g to match sum of f_norm
    g = g / g.sum()

    # Try different blend factors
    for tt in [t, 0.3 * t, 0.6 * t, 1.5 * t]:
        f_trial = (1 - tt) * f_norm + tt * g
        f_trial = np.maximum(f_trial, 0)
        ss = f_trial.sum()
        if ss < 1e-15:
            continue
        f_trial = f_trial / ss
        obj_new = fast_convolve(f_trial).max()
        obj_old = M
        if obj_new < obj_old * (1 - 1e-12):
            return f_trial * s, True

    return f, False


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp + n * np.log(n)  # add constant for normalization
    
    # Weights for each lag
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a = 2n * cross_corr(w, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(W) * A, fft_size)[:n]
    
    grad_a = 2.0 * n * cross
    
    # Chain rule: grad_z_j = a_j * (grad_a_j - a_j * sum_i grad_a_i * a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def lbgfs_run(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max objective."""
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def simulated_annealing(a0, time_limit=60.0, T_init=0.01, T_final=0.0001):
    """Simulated annealing on the sequence."""
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 999999)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    total_steps = 0

    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / time_limit
        T = T_init * (T_final / T_init) ** progress

        # Pick a random coordinate to perturb
        j = rng.integers(0, n)
        old_val = a[j]

        # Perturb: multiply by random factor
        factor = np.exp(rng.normal(0, 0.05))
        new_val = old_val * factor

        if new_val < 0 or new_val > 0.5:
            continue

        # Adjust others to maintain sum = 1
        denom = 1.0 - old_val
        if denom < 1e-10:
            continue
        scale = (1.0 - new_val) / denom
        a_trial = a.copy()
        a_trial[j] = new_val
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_trial[mask] *= scale
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss

        obj_new = fast_convolve(a_trial).max()
        delta = obj_new - best_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

        total_steps += 1
        if total_steps % 10000 == 0:
            pass  # Could print progress

    return best_a


def coordinate_descent_fast(a0, time_limit=60.0):
    """Fast coordinate descent with 3 multipliers per coordinate."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = np.array([0.6, 0.8, 1.0, 1.25, 1.5])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
                else:
                    a = best_a.copy()

        if not improved:
            # Try with different multiplier set
            break

    return best_a


def subgradient_polish(a0, time_limit=60.0, lr=0.003, mom=0.92, top_k=50):
    """Subgradient method with softmax-weighted active peaks."""
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 999999)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        co = fast_convolve(a).max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_a


def peak_shave(a0, n_peaks=5, n_steps=300, time_limit=30.0):
    """Iterative peak shaving by reducing largest contributions."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = sorted([int(k) for k in peak_indices if b[k] > best_obj * 0.95],
                              key=lambda x: -b[x])

        if not peak_indices:
            break

        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]

            # Try reducing top few contributors
            top_local = np.argsort(contributions)[::-1][:3]
            peak_improved = False
            for li in top_local:
                i = idx[li]
                j = k - i
                for target in [i, j]:
                    old_val = a[target]
                    if old_val < 1e-12:
                        continue
                    new_val = old_val * 0.92
                    diff = old_val - new_val
                    if diff < 1e-16:
                        continue

                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others = a[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[target] = new_val
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss

                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        peak_improved = True
                        break
                if peak_improved:
                    break

            improved = improved or peak_improved

        if not improved:
            break

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def random_start(n, rng):
    """Generate a random non-negative sequence (normalized to sum=1)."""
    method = rng.integers(0, 5)
    if method == 0:
        a = rng.random(n)
    elif method == 1:
        # Smoothed random
        a = np.convolve(rng.random(n), np.ones(5) / 5, mode='same')
    elif method == 2:
        # Bimodal
        a = rng.random(n)
        a[:n // 2] *= rng.uniform(0.5, 2.0)
        a[n // 2:] *= rng.uniform(0.5, 2.0)
    elif method == 3:
        # Fourier magnitude
        freq = rng.standard_normal((n + 1) // 2)
        a = np.abs(irfft(np.abs(freq) * (1 + rng.random(freq.shape) * 0.5), n)) + 0.01
    else:
        # Sparse + dense
        a = rng.random(n) * 0.3
        a[rng.choice(n, size=n // 10, replace=False)] += 1.0
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def try_seq(a, best_state, label=""):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Multi-start initialization with quick L-BFGS
    Phase 2: LP-based iterative refinement
    Phase 3: Deep L-BFGS with temperature annealing
    Phase 4: Length exploration
    Phase 5: Simulated annealing
    Phase 6: Coordinate descent + peak shaving + subgradient polish
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    best_state = [float('inf'), None]

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
    except Exception:
        pass

    # === Phase 1: Multi-start (120s) ===
    print("\n=== Phase 1: Multi-start initialization ===")
    t1_end = start_time + 120.0

    candidates = []
    if parent is not None:
        candidates.append(('parent', parent.copy()))

    # Try various lengths and random starts
    target_lengths = [512, 768, 1024, 1536, 2048]
    for tl in target_lengths:
        if time.monotonic() > t1_end:
            break
        for trial in range(2):
            if time.monotonic() > t1_end:
                break
            a_rand = random_start(tl, rng)
            candidates.append((f"rand_n{tl}_{trial}", a_rand))

    # Quick L-BFGS on each candidate
    for label, a_cand in candidates:
        if time.monotonic() > t1_end:
            break
        rem = t1_end - time.monotonic()
        if rem < 10:
            break
        n = len(a_cand)
        z0 = z_from_a(a_cand)
        t_alloc = min(rem * 0.4, 30.0)
        if t_alloc < 5:
            t_alloc = 5
        z_opt, _ = lbgfs_run(z0, 10, n, t_alloc, maxiter=1000)
        a_opt = softmax_from_z(z_opt)
        # Also try with T=50
        rem = t1_end - time.monotonic()
        if rem > 5:
            z_opt2, _ = lbgfs_run(z_opt, 50, n, min(rem * 0.5, 20.0), maxiter=500)
            a_opt2 = softmax_from_z(z_opt2)
            try_seq(a_opt2, best_state, f"{label}_T50")
        try_seq(a_opt, best_state, f"{label}_T10")

    if best_state[1] is None:
        best_state[1] = parent if parent is not None else np.ones(256) / 256
        best_state[0] = compute_obj(best_state[1])

    print(f"  Phase 1 done: best={best_state[0]:.8f}")

    # === Phase 2: LP-based iterative refinement (500s) ===
    print(f"\n=== Phase 2: LP iteration (best={best_state[0]:.8f}) ===")
    t2_end = min(start_time + 620.0, deadline - 400.0)
    a_lp = best_state[1].copy()
    a_lp = a_lp / a_lp.sum()
    n_lp = len(a_lp)

    # Pre-build the convolution matrix (will need to update as f changes)
    lp_count = 0
    lp_stuck = 0
    t_blend = 0.3

    while time.monotonic() < t2_end:
        b_norm = fast_convolve(a_lp)
        M = b_norm.max()

        # Solve LP
        g = lp_solve_max_sum(a_lp, M)
        if g is None or g.sum() < 1e-15:
            lp_stuck += 1
            if lp_stuck > 20:
                break
            # Try with relaxed constraint
            g = lp_solve_max_sum(a_lp, M * 1.001)
            if g is None or g.sum() < 1e-15:
                lp_stuck += 1
                continue

        lp_count += 1
        g = g / g.sum()

        # Try blending
        improved_this = False
        for tt in [t_blend, 0.15, 0.5, 0.05, 0.8]:
            f_trial = (1 - tt) * a_lp + tt * g
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            b_trial = fast_convolve(f_trial)
            if b_trial.max() < M * (1 - 1e-10):
                a_lp = f_trial
                M = b_trial.max()
                improved_this = True
                break

        if improved_this:
            lp_stuck = 0
            obj_lp = 2.0 * n_lp * M
            if obj_lp < best_state[0] - 1e-10:
                try_seq(a_lp, best_state, f"lp_iter{lp_count}")
            if lp_count % 50 == 0:
                print(f"    LP iter {lp_count}: M={M:.8f}, obj~{obj_lp:.8f}")
        else:
            lp_stuck += 1
            # Adjust blend
            t_blend = max(0.05, t_blend * 0.8)
            if lp_stuck > 50:
                # Try with a slightly lower M target
                g2 = lp_solve_max_sum(a_lp, M * 0.99)
                if g2 is not None and g2.sum() > 1e-15:
                    g2 = g2 / g2.sum()
                    f_trial = 0.5 * a_lp + 0.5 * g2
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    if fast_convolve(f_trial).max() < M:
                        a_lp = f_trial
                        lp_stuck = 0
                if lp_stuck > 100:
                    break

    print(f"  Phase 2 done: {lp_count} LP iterations, best={best_state[0]:.8f}")

    # === Phase 3: Deep L-BFGS with temperature annealing (350s) ===
    print(f"\n=== Phase 3: Deep L-BFGS (best={best_state[0]:.8f}) ===")
    t3_end = min(start_time + 970.0, deadline - 250.0)
    a_best = best_state[1].copy()
    a_best = a_best / a_best.sum()
    n_best = len(a_best)
    z_curr = z_from_a(a_best)

    temps = [20, 50, 100, 200, 500, 1000]
    for T in temps:
        rem = t3_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.3, 60.0)
        if t_alloc < 10:
            t_alloc = 10
        print(f"    L-BFGS n={n_best}, T={T}, t={t_alloc:.0f}s")
        z_curr, _ = lbgfs_run(z_curr, T, n_best, t_alloc, maxiter=3000)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, best_state, f"deep_T{T}")

    # Try a fresh start from best with high T
    rem = t3_end - time.monotonic()
    if rem > 20:
        z_fresh = z_from_a(best_state[1] / best_state[1].sum())
        z_fresh, _ = lbgfs_run(z_fresh, 200, len(best_state[1]), min(rem * 0.5, 40.0), maxiter=2000)
        a_fresh = softmax_from_z(z_fresh)
        try_seq(a_fresh, best_state, "fresh_T200")

    print(f"  Phase 3 done: best={best_state[0]:.8f}")

    # === Phase 4: Length exploration (200s) ===
    print(f"\n=== Phase 4: Length exploration (best={best_state[0]:.8f}) ===")
    t4_end = min(start_time + 1170.0, deadline - 150.0)
    base = best_state[1].copy()
    base_n = len(base)

    # Try scaling to nearby lengths
    for factor in [0.75, 1.25, 1.5, 2.0]:
        new_n = int(base_n * factor)
        new_n = max(256, min(4096, new_n))
        if new_n == base_n:
            continue
        rem = t4_end - time.monotonic()
        if rem < 20:
            break

        f_interp = interpolate_sequence(base, new_n)
        z_init = z_from_a(f_interp)

        t_alloc = min(rem * 0.4, 40.0)
        z_opt, _ = lbgfs_run(z_init, 10, new_n, t_alloc, maxiter=1500)
        a_opt = softmax_from_z(z_opt)
        try_seq(a_opt, best_state, f"scale_n{new_n}_T10")

        rem = t4_end - time.monotonic()
        if rem > 10:
            t_alloc2 = min(rem * 0.3, 25.0)
            z_opt2, _ = lbgfs_run(z_opt, 100, new_n, t_alloc2, maxiter=1000)
            a_opt2 = softmax_from_z(z_opt2)
            try_seq(a_opt2, best_state, f"scale_n{new_n}_T100")

    # Also try completely new lengths
    for nl in [1280, 2560, 3072]:
        rem = t4_end - time.monotonic()
        if rem < 20 or nl == base_n:
            continue
        f_interp = interpolate_sequence(base, nl)
        z_init = z_from_a(f_interp)
        t_alloc = min(rem * 0.3, 30.0)
        z_opt, _ = lbgfs_run(z_init, 20, nl, t_alloc, maxiter=1500)
        a_opt = softmax_from_z(z_opt)
        try_seq(a_opt, best_state, f"new_n{nl}")

    print(f"  Phase 4 done: best={best_state[0]:.8f}")

    # === Phase 5: Simulated annealing (150s) ===
    print(f"\n=== Phase 5: Simulated annealing (best={best_state[0]:.8f}) ===")
    t5_end = min(start_time + 1320.0, deadline - 100.0)
    a_best5 = best_state[1].copy()
    a_best5 = a_best5 / a_best5.sum()
    n_best5 = len(a_best5)

    t_sa = min(t5_end - time.monotonic(), 120.0)
    if t_sa > 10:
        print(f"    SA n={n_best5}, t={t_sa:.0f}s")
        a_sa = simulated_annealing(a_best5, time_limit=t_sa, T_init=0.005, T_final=0.0001)
        try_seq(a_sa, best_state, "sim_anneal")

    # Second SA with different parameters
    rem = t5_end - time.monotonic()
    if rem > 15:
        a_pert = best_state[1].copy()
        a_pert = a_pert / a_pert.sum()
        a_pert = project_simplex(a_pert + rng.standard_normal(n_best5) * 0.01)
        a_sa2 = simulated_annealing(a_pert, time_limit=min(rem, 60.0), T_init=0.003, T_final=0.00005)
        try_seq(a_sa2, best_state, "sim_anneal2")

    print(f"  Phase 5 done: best={best_state[0]:.8f}")

    # === Phase 6: Final polishing (remaining time) ===
    print(f"\n=== Phase 6: Final polishing (best={best_state[0]:.8f}) ===")
    a_final = best_state[1].copy()
    a_final = a_final / a_final.sum()
    n_final = len(a_final)

    # Peak shaving
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.2, 30.0)
        a_ps = peak_shave(a_final, n_peaks=5, n_steps=200, time_limit=t_ps)
        try_seq(a_ps, best_state, "peak_shave")

    # Coordinate descent
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.3, 60.0)
        a_cd = coordinate_descent_fast(a_final, time_limit=t_cd)
        try_seq(a_cd, best_state, "coord_desc")

    # Subgradient polish
    rem = deadline - time.monotonic()
    if rem > 15:
        t_sg = min(rem * 0.3, 45.0)
        a_sg = subgradient_polish(a_final, time_limit=t_sg, lr=0.002, mom=0.93,
                                  top_k=max(30, n_final // 8))
        try_seq(a_sg, best_state, "subgrad")

    # Micro-optimization: very high T L-BFGS
    rem = deadline - time.monotonic()
    if rem > 10:
        z0 = z_from_a(best_state[1] / best_state[1].sum())
        t_alloc = min(rem * 0.3, 20.0)
        z_f, _ = lbgfs_run(z0, 500, n_final, t_alloc, maxiter=2000)
        a_f = softmax_from_z(z_f)
        try_seq(a_f, best_state, "micro_T500")

    # Final LP pass on the best
    rem = deadline - time.monotonic()
    if rem > 20:
        a_lp2 = best_state[1].copy()
        a_lp2 = a_lp2 / a_lp2.sum()
        n_lp2 = len(a_lp2)
        lp2_count = 0
        while time.monotonic() < deadline - 10 and lp2_count < 200:
            b_norm = fast_convolve(a_lp2)
            M = b_norm.max()
            g = lp_solve_max_sum(a_lp2, M)
            if g is None or g.sum() < 1e-15:
                break
            g = g / g.sum()
            improved = False
            for tt in [0.2, 0.1, 0.4]:
                f_t = (1 - tt) * a_lp2 + tt * g
                f_t = np.maximum(f_t, 0)
                f_t = f_t / f_t.sum()
                if fast_convolve(f_t).max() < M * (1 - 1e-12):
                    a_lp2 = f_t
                    improved = True
                    break
            if improved:
                try_seq(a_lp2, best_state, f"final_lp{lp2_count}")
                lp2_count += 1
            else:
                break

    # Ultimate micro-perturbation + optimize
    rem = deadline - time.monotonic()
    if rem > 8:
        for attempt in range(5):
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            noise = rng.standard_normal(n_final) * 0.003
            a_pert = project_simplex(best_state[1] / best_state[1].sum() + noise)
            z_pert = z_from_a(a_pert)
            z_p, _ = lbgfs_run(z_pert, 300, n_final, min(rem * 0.4, 10.0), maxiter=500)
            a_p = softmax_from_z(z_p)
            try_seq(a_p, best_state, f"micro_pert{attempt}")

    # Final verification
    if best_state[1] is not None:
        final_score = float(evaluate_sequence(best_state[1].tolist()))
        if np.isfinite(final_score):
            best_state[0] = final_score

    print(f"\nFINAL SCORE: {best_state[0]:.8f} (n={len(best_state[1])})")
    total_time = time.monotonic() - start_time
    print(f"Total time: {total_time:.1f}s")

    return [float(x) for x in best_state[1]]
