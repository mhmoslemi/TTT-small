import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max_and_grad(a, T):
    """
    Compute smooth max of conv(a,a) and its gradient.
    Returns (smooth_max_value, gradient).
    Assumes sum(a) = 1.

    c = fast_convolve(a)
    # Softmax weights

    exp_vals = np.exp(Tc - Tc_max)
    p = exp_vals / exp_vals.sum()
    
    # Smooth max value (in units of c, i.e., the weighted average weighted by softmax)

    lse = (Tc_max + np.log(exp_vals.sum())) / T

    a_flip = a[::-1]
    
    # Compute (p * a) and (p * a_flip) via FFT
    conv_size = 2 * n - 1 + (2 * n - 1) - 1  # full convolution of two (2n-1)-vectors
    # Actually, p has length 2n-1 and a has length n.
    # (p * a) has length (2n-1) + n - 1 = 3n - 2
    # We need indices 0 to n-1 of (p*a) and indices n-1 to 2n-2 of (p*a_flip)
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    pa = np.fft.irfft(P * A, fft_size)  # length 3n-2
    
    AF = np.fft.rfft(a_flip, fft_size)
    paf = np.fft.irfft(P * AF, fft_size)  # length 3n-2
    
    # dLSE/da[m] = paf[m + n - 1] + pa[m]
    grad_lse = paf[n-1:n-1+n] + pa[:n]
    
    return lse, grad_lse

def projected_gradient_smooth_max(a0, time_limit, T_init=100, T_final=5000,

    But this is a QP (quadratic constraints), not LP.
    
    Instead, use the Frank-Wolfe approach: find the extreme point direction that
    reduces the max convolution the most.
    
    Actually, let's use a simpler approach: solve LP to find the best direction
    that reduces the current peaks.
    """
    # LP: max sum(d) s.t. (a + d) * (a + d)[k] <= M_target for all k, d >= -a, sum(d) = 0
    # This is still quadratic. Let me linearize.
    
    # Linearization: (a+d)*(a+d)[k] ≈ a*a[k] + 2*(a*d)[k] for small d
    # So: 2*(a*d)[k] <= M_target - a*a[k] for all k
    # LP: max -||d||_1 (or min -sum(d) if we want to increase some components)
    #     s.t. 2*(a*d)[k] <= M_target - (a*a)[k] for k=0..2n-2
    #          d >= -a
    #          sum(d) = 0
    
    # This LP has 2n-1 + n + 1 constraints and n variables.
    # For n=1024, that's ~3000 constraints and 1024 variables. Doable.
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    c = fast_convolve(a)
    
    # If already feasible, return a
    if c.max() <= M_target:
        return a
    
    # Build LP
    # Variables: d[0..n-1]
    # Objective: minimize -sum(d) (to make d as large as possible... hmm)
    # Actually, let's minimize ||d||_1 = sum|d_i|
    # Split d = d+ - d-, d+, d- >= 0
    # Minimize sum(d+) + sum(d-)
    # Constraints:
    #   2 * sum_j a[j] * d[k-j] <= M_target - c[k] for all k (linearized)
    #   d+ - d- >= -a (i.e., d >= -a)
    #   sum(d+) - sum(d-) = 0 (sum constraint)
    
    # Actually this is getting complicated with the linearization. Let me use a simpler approach:
    # Just solve: max sum(d) s.t. (a+d)*a[k] <= M_target/2 for all k, d >= 0, sum(d) <= epsilon
    # This ensures that after the update, the "new" peaks are reduced.
    
    # Even simpler: use the Frank-Wolfe direction.
    # At the current point a, the subgradient of max(c) is p (the softmax weights).
    # The FW direction is the solution to: min_d <p, d-a> s.t. d in simplex
    # = min_d sum_k p[k]*(d*d)[k] ... no, that's not linear.
    
    # For the smooth max: the gradient is 2n * ((p*a_flip)[m+n-1] + (p*a)[m])
    # The FW direction minimizes <grad, d> over the simplex = the minimum component of grad.
    # So d = e_j where j = argmin grad_tan[j].
    
    # This gives a simple coordinate-point direction!
    
    # Let me implement a faster FW:
    # 1. Compute gradient g = 2n * grad_lse
    # 2. Project onto tangent: g_tan = g - mean(g)
    # 3. FW direction: e_j where j = argmin g_tan
    # 4. Update: a_new = (1-t)*a + t*e_j
    # 5. Line search for t
    
    # This is O(n) per step (after computing the gradient which is O(n log n)).
    # Much faster than LP!
    
    return a  # placeholder, we'll use FW below

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_max_grad(a, T):
    """
    Compute LSE(T*c) and its gradient for c = a*a, assuming sum(a)=1.
    Returns (lse_value, gradient_of_lse).
    """
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_vals = np.exp(Tc - Tc_max)

    AF = np.fft.rfft(a_flip, fft_size)
