"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Soft-max L-BFGS with temperature annealing
2. FISTA (accelerated projected gradient) with exact/approximate max subgradient
3. Multi-length exploration
4. Coordinate descent and perturbation-based local search

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
from functools import partial

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    # Symmetrize to remove numerical artifacts
    result = (result + result[::-1]) / 2.0
    return result[:size]


def cross_corr_at_lag(a, k):
    """Compute the gradient of (a*a)_k w.r.t. a, which is 2*a_{k-i} for valid i."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        grad[j] = 2.0 * a[k - j]
    return grad


def cross_corr_at_lag_vec(a, peaks, weights):
    """Weighted sum of gradients at multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        grad[j_range] += 2.0 * w * a[k - j_range]
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_fun(T, n, z):
    """Soft-max smoothed objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    log_sum_exp = max_Tb + np.log(np.exp(Tb - max_Tb).sum())
    return (2.0 * n / T) * log_sum_exp


def soft_max_grad(T, n, z):
    """Gradient of soft-max smoothed objective w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a = 2n * sum_k w_k * a_{k-i} (cross-correlation)
    grad_a = np.zeros(n)
    # Only compute for significant weights
    significant = np.where(w > 1e-10)[0]
    for k in significant:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        grad_a[j_range] += 2.0 * n * w[k] * a[k - j_range]
    
    # Chain rule for softmax: grad_z = a * (grad_a - a * sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return grad_z


def soft_max_obj_and_grad(T, n, z):
    """Compute both objective and gradient (sharing the convolution)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    significant = np.where(w > 1e-10)[0]
    for k in significant:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        grad_a[j_range] += 2.0 * n * w[k] * a[k - j_range]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_run(z0, T, n, time_limit, maxiter=3000):
    """Run L-BFGS-B on the soft-max objective in log-space."""
    t0 = time.monotonic()
    
    class _Obj:
        def __call__(self, z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return soft_max_fun(T, n, z)
    
    class _Grad:
        def __call__(self, z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return soft_max_grad(T, n, z)
    
    try:
        res = minimize(
            fun=_Obj(),
            x0=z0,
            jac=_Grad(),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return res.x
    except Exception:
        return z0

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) LP-based blocking direction (SOTA-inspired), (2) FISTA with
Nesterov acceleration on the simplex, (3) L-BFGS with soft-max annealing,
(4) coordinate descent and multi-start exploration.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT with symmetrization."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    result = (result + result[::-1]) / 2.0
    return result[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution constraint matrix A where A[k,j] = f[k-j]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A


def lp_block_direction(f, M):
    """Solve LP: max 1^T g s.t. g >= 0, A@f g <= M. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    try:
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'dual_feasibility_tolerance': 1e-9}
        )
    except Exception:
        return None
    if not res.success:
        return None
    g = res.x
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

                no_improve = 0
    
    # Final check
    b_fin = fast_convolve(a)
    if b_fin.max() < best_obj:
        best_a = a.copy()
        best_obj = b_fin.max()
    
    return best_a


def soft_max_obj_grad(T, n, z):
    """Compute soft-max objective and gradient w.r.t. log-space z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    significant = np.where(w > 1e-10)[0]
    for k in significant:
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        j_range = np.arange(js, je + 1)
        grad_a[j_range] += 2.0 * n * w[k] * a[k - j_range]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


class LBGFSFun:
    def __init__(self, T, n, t0, time_limit):
        self.T = T
        self.n = n
        self.t0 = t0
        self.time_limit = time_limit
    def __call__(self, z):
        if time.monotonic() - self.t0 > self.time_limit:
            return 1e15
        return soft_max_obj_grad(self.T, self.n, z)[0]


class LBGFSGrad:
    def __init__(self, T, n, t0, time_limit):
        self.T = T
        self.n = n
        self.t0 = t0
        self.time_limit = time_limit
    def __call__(self, z):
        if time.monotonic() - self.t0 > self.time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(self.T, self.n, z)[1]


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS-B on soft-max objective in log-space."""
    t0 = time.monotonic()
    try:
        res = minimize(
            fun=LBGFSFun(T, n, t0, time_limit),
            x0=z0,
            jac=LBGFSGrad(T, n, t0, time_limit),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 15, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent: try scaling each element and redistribute."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.8, 2.5, 4.0]
    
    improved_any = True
    while improved_any and (time.monotonic() - t0) < time_limit:
        improved_any = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                improved_any = True
    
    return best_a


def subgradient_polish(a0, time_limit=60.0, top_k=50):
    """Subgradient with softmax-weighted peaks and momentum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = 0.01
    mom = 0.9
    cur_temp = 0.001
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - lr * sg
        a = project_simplex(a + v)
        
        b = fast_convolve(a)
        co = b.max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        cur_temp *= 1.005
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = max(lr * 0.5, 0.001)
            no_improve = 0
    
    return best_a, best_obj


def make_u_shaped(n, sharpness=2.0):
    """Create a U-shaped sequence (high at edges, low in center)."""
    x = np.linspace(0, 1, n)
    a = np.abs(x - 0.5) ** sharpness
    a = a + 0.01
    return a / a.sum()


def make_sinusoidal(n, freq=1.0):
    """Sinusoidal sequence peaking at edges."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_exponential_u(n, alpha=3.0):
    """U-shaped with exponential profile."""
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * x * (1.0 - x))
    a = a + 0.001
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization combining LP blocking, FISTA, L-BFGS soft-max,
    and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")
    
    n_parent = len(parent)
    
    # === Phase 1: LP-based blocking (most promising for improvement) ===
    print("\n=== Phase 1: LP blocking (parent) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.35, 400.0)
    if t_lp > 30:
        f_lp, _ = lp_optimize(parent, time_limit=t_lp, max_iters=5000)
        try_seq(f_lp, "lp_parent")
    
    # === Phase 2: FISTA polishing on LP result ===
    print("\n=== Phase 2: FISTA polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_fista = min(rem * 0.2, 200.0)
        if t_fista > 20:
            a_fista = fista_optimize(best_seq, time_limit=t_fista, T=1000.0)
            try_seq(a_fista, "fista_T1000")
    
    # === Phase 3: L-BFGS soft-max on best ===
    print("\n=== Phase 3: L-BFGS soft-max annealing ===")
    if best_seq is not None:
        n_best = len(best_seq)
        z0 = z_from_a(best_seq)
        temps = [20, 50, 200, 500]
        z_curr = z0
        for T in temps:
            rem = deadline - time.monotonic()
            t_alloc = min(rem * 0.12, 100.0)
            if t_alloc < 15:
                break

            z_curr = lbgfs_optimize(z_curr, T, n_best, t_alloc, maxiter=2000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"lbgfs_T{T}")
    
    # === Phase 4: Multi-start exploration ===
    print("\n=== Phase 4: Multi-start exploration ===")
    starts = {
        "u_shaped": make_u_shaped(n_parent, 2.0),
        "sinusoidal": make_sinusoidal(n_parent, 1.0),
        "exp_u": make_exponential_u(n_parent, 3.0),

    for name, start in starts.items():
        rem = deadline - time.monotonic()
        t_start = min(rem * 0.08, 60.0)

    print("\n=== Phase 6: CD + Subgradient polish ===")

    if best_seq is not None:

        t_final_lp = min(rem * 0.35, 150.0)

            f_fin, _ = lp_optimize(best_seq, time_limit=t_final_lp, max_iters=5000)
            try_seq(f_fin, "final_lp")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
