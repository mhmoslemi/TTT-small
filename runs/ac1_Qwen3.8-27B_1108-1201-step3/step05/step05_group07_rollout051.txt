"""
LP-based descent method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: at each step, solve a linear program to find the optimal
descent direction (maximizing sum of new direction subject to cross-convolution
constraints), combined with soft-max L-BFGS refinement and multi-length
exploration. This differs from pure gradient-based methods by using a global
LP to determine each step direction.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(f):
    """Build sparse matrix C where (C @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    row_parts = []
    col_parts = []
    data_parts = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        row_parts.append(np.full(je - js + 1, k, dtype=np.int64))
        col_parts.append(idx.astype(np.int64))
        data_parts.append(f[k - idx])
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_descent_direction(f, b_vec):
    """
    Solve LP: maximize sum(g) subject to C@g <= b_vec, g >= 0.
    Returns g or None if LP fails.
    """
    n = len(f)
    C = build_conv_matrix_sparse(f)
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=C, b_ub=b_vec, bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        return None
    return res.x


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def soft_max_obj_fun(z, T, n, deadline):
    """Objective function wrapper for scipy.minimize."""
    if time.monotonic() > deadline:
        return 1e15
    return soft_max_obj_grad(z, T, n)[0]


def soft_max_obj_jac(z, T, n, deadline):
    """Gradient function wrapper for scipy.minimize."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    return soft_max_obj_grad(z, T, n)[1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lbgfs_optimize(z0, T, n, deadline, maxiter=3000):
    """L-BFGS in log-space with soft-max objective."""

            fun=soft_max_obj_fun,
            x0=z0,

            method='L-BFGS-B',

                     'maxfun': maxiter * 25, 'disp': False}

    except Exception:
        return z0


def lp_descent_loop(f, deadline, max_iters=5000, verbose=True):
    """
    Main LP descent loop.
    f: initial sequence (will be normalized to sum=1)
    Returns best sequence found.
    """
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_b = fast_convolve(f)
    best_max_b = best_b.max()
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() < deadline and iter_count < max_iters:
        b = fast_convolve(f)
        max_b = b.max()
        
        # Solve LP: maximize sum(g) s.t. C@g <= b, g >= 0
        g = lp_descent_direction(f, b)
        if g is None:
            # Try with M * 1 as constraint
            g = lp_descent_direction(f, np.full(2 * n - 1, max_b))
            if g is None:
                break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # normalize to sum = 1
        
        # Line search for best t
        best_t = 0.0
        best_obj = max_b
        
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = (1 - t) * f + t * g
            b_trial = fast_convolve(f_trial)
            obj = b_trial.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_t = t
        
        if best_t == 0.0:
            no_improve += 1
            if no_improve > 20:
                break
            # Try very small steps
            for t in [0.001, 0.002, 0.005]:
                f_trial = (1 - t) * f + t * g
                b_trial = fast_convolve(f_trial)
                obj = b_trial.max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_t = t
                    break
            if best_t == 0.0:
                break
        else:
            no_improve = 0
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / f.sum()
        iter_count += 1
        
        b = fast_convolve(f)
        max_b = b.max()
        if max_b < best_max_b - 1e-14:
            best_max_b = max_b
            best_f = f.copy()
        
        if verbose and iter_count % 100 == 0:
            score = 2.0 * n * max_b  # since sum=1
            print(f"    LP iter {iter_count}: max_b={max_b:.8f}, score={score:.6f}")
    
    return best_f


def peak_shave(a, n_peaks, n_steps, deadline):
    """Peak shaving: iteratively reduce the highest peaks."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        k_actual = min(n_peaks, len(b_inner))
        peak_indices = np.argpartition(b_inner, -k_actual)[-k_actual:]
        peak_indices = [int(k) for k in peak_indices if b[k] > best_obj * 0.95]
        
        if not peak_indices:
            break
        
        improved = False
        for k in peak_indices:
            if time.monotonic() > deadline:
                break
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i
            
            for target in [int(i), int(j)]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                factor = 0.95
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                mask = np.ones(n, dtype=bool)
                mask[target] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[target] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-12:
                    a = a_trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a

"""
LP-based descent method for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: solve a linear program at each step to find the globally
optimal descent direction (maximizing sum(g) s.t. conv(f,g) <= M, g >= 0),
combined with soft-max L-BFGS refinement, multi-length exploration, and
peak shaving. The LP direction is fundamentally different from gradient
methods as it performs a global optimization over the direction at each step.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(f):
    """Build sparse matrix C where (C @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    row_parts = []
    col_parts = []
    data_parts = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        row_parts.append(np.full(je - js + 1, k, dtype=np.int64))
        col_parts.append(idx.astype(np.int64))
        data_parts.append(f[k - idx])
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, rhs):
    """
    Solve LP: maximize sum(g) subject to C@g <= rhs, g >= 0.
    Returns g or None.
    """
    n = len(f)
    C = build_conv_matrix_sparse(f)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=rhs, bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            return None
        return res.x
    except Exception:
        return None


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def soft_max_obj_fun(z, T, n, deadline):
    """Objective function wrapper for scipy.minimize."""
    if time.monotonic() > deadline:
        return 1e15
    return soft_max_obj_grad(z, T, n)[0]


def soft_max_obj_jac(z, T, n, deadline):
    """Gradient function wrapper for scipy.minimize."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    return soft_max_obj_grad(z, T, n)[1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lbgfs_optimize(z0, T, n, deadline, maxiter=3000):
    """L-BFGS in log-space with soft-max objective."""
    try:
        res = minimize(
            fun=soft_max_obj_fun,
            x0=z0,
            jac=soft_max_obj_jac,
            method='L-BFGS-B',
            args=(T, n, deadline),
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def lp_descent_loop(f_init, deadline, max_iters=10000, verbose=True,
                    use_pointwise=False):
    """
    LP descent loop. f_init should be non-negative.
    Iteratively: solve LP for direction, line search, update.
    Returns (best_f, best_max_b).
    """
    f = f_init.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)

    b = fast_convolve(f)
    best_f = f.copy()
    best_max_b = b.max()

    iter_count = 0
    no_improve = 0
    C_cached = build_conv_matrix_sparse(f)
    f_cached = f.copy()

    while time.monotonic() < deadline and iter_count < max_iters:
        b = fast_convolve(f)
        max_b = b.max()

        # Rebuild constraint matrix if f changed significantly
        if np.max(np.abs(f - f_cached)) > 1e-6:
            C_cached = build_conv_matrix_sparse(f)
            f_cached = f.copy()

        # Choose RHS
        if use_pointwise:
            rhs = b.copy()
        else:
            rhs = np.full(2 * n - 1, max_b)

        # Solve LP
        n = len(f)
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=C_cached, b_ub=rhs,
                          bounds=[(0, None)] * n, method='highs')
            if not res.success:
                # Try with pointwise constraint
                if not use_pointwise:
                    res = linprog(c_obj, A_ub=C_cached, b_ub=b,
                                  bounds=[(0, None)] * n, method='highs')
                    if not res.success:
                        break
                else:
                    break
            g = res.x
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check if direction is essentially zero
        if np.max(np.abs(g - f)) < 1e-8:
            no_improve += 1
            if no_improve > 10:
                break
            # Perturb f slightly
            f = f + np.random.default_rng(iter_count).standard_normal(n) * 1e-5
            f = project_simplex(f)
            continue

        # Adaptive line search
        best_t = 0.0
        best_obj = max_b
        t = 0.5
        while t > 0.002:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            b_trial = fast_convolve(f_trial)
            obj = b_trial.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_t = t
                break
            t *= 0.5

        if best_t == 0.0:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()

        iter_count += 1

        b = fast_convolve(f)
        max_b = b.max()
        if max_b < best_max_b - 1e-14:
            best_max_b = max_b
            best_f = f.copy()

        if verbose and iter_count % 200 == 0:
            score = 2.0 * n * max_b
            print(f"    LP iter {iter_count}: max_b={max_b:.8f}, "
                  f"score={score:.6f}, t={best_t:.4f}")

    return best_f, best_max_b


def peak_shave(a, n_peaks, n_steps, deadline):
    """Peak shaving: iteratively reduce the highest peaks."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        k_actual = min(n_peaks, len(b_inner))
        peak_indices = np.argpartition(b_inner, -k_actual)[-k_actual:]
        peak_indices = [int(k) for k in peak_indices if b[k] > best_obj * 0.95]

        if not peak_indices:
            break

        improved = False
        for k in peak_indices:
            if time.monotonic() > deadline:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]

            max_local = np.argmax(contributions)
            i = int(idx[max_local])
            j = int(k - i)

            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                new_val = old_val * 0.95
                diff = old_val - new_val
                if diff < 1e-15:
                    continue

                mask = np.ones(n, dtype=bool)
                mask[target] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[target] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-12:
                    a = a_trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a


def coordinate_descent(a0, deadline, max_rounds=3):
    """Coordinate descent for local refinement."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.8]

    for _ in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved_any = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                improved_any = True
        if not improved_any:
            break

    return best_a


def evaluate_and_update(a, state, label):
    """Evaluate sequence and update state dict with best."""

    s = a.sum()

        state['best_score'] = score
        state['best_seq'] = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score
