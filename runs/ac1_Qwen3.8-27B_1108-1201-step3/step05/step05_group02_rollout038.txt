import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient(f, tau, conv=None):
    """Compute gradient of LSE-smoothed max of auto-convolution."""
    if conv is None:
        conv = fast_convolve(f)
    n = len(f)
    # Compute softmax weights
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad = 2 * (w * f_rev)[n-1 : 2n-1]
    f_rev = f[::-1]
    conv_len = len(w) + n - 1  # (2n-1) + n - 1 = 3n-2
    N = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, N)
    FR = np.fft.rfft(f_rev, N)
    conv_result = np.fft.irfft(W * FR, N)
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad, w

        f = f / s

    it = 0

    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        grad, w = lse_gradient(f, tau, conv)
        
        # Projected gradient descent with backtracking line search
        best_alpha = 0.0
        best_val = conv.max()
        current_val = conv.max()
        
        for alpha in [lr * 0.1, lr * 0.3, lr * 0.7, lr, lr * 1.5, lr * 2, lr * 3]:
            if alpha <= 0:

            ft = f - alpha * grad
            ft = project_simplex(ft)
            val = fast_convolve(ft).max()
            if val < best_val:

                best_alpha = alpha

            f_new = project_simplex(f_new)
            conv_new = fast_convolve(f_new)

Hybrid LSE-Gradient + LP-FW optimizer for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses log-sum-exp smoothed gradient descent with momentum, adaptive tau annealing,
LP-FW for global moves, frequency-domain perturbation for diversity, and 
multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Ensure non-negative for the projection
    # Standard simplex projection: find theta such that sum(max(v-theta, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_smoothed_grad(f, tau, conv=None):
    """
    Compute gradient of tau * log(sum(exp(conv_k / tau)))
    which approximates max(conv) as tau -> 0.
    Returns (gradient, max_value, weights).
    """
    if conv is None:
        conv = fast_convolve(f)
    n = len(f)
    
    # Softmax weights (numerically stable)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # LSE value
    lse_val = tau * (conv.max() / tau + np.log(w_sum))  # = conv.max() + tau * log(w_sum)
    
    # Gradient: grad[i] = 2 * sum_j f[j] * w[i+j] (with w out of range = 0)
    # = 2 * (w * f_rev)[n-1 : 2n-1] where * is linear convolution
    f_rev = f[::-1]
    conv_len = len(w) + n - 1
    N = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, N)
    FR = np.fft.rfft(f_rev, N)
    conv_result = np.fft.irfft(W * FR, N)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return grad, lse_val, w

    tau = tau_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Total iterations plan for tau annealing
    # We'll anneal tau over the entire time limit
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed_ratio = min((time.monotonic() - t0) / time_limit, 1.0)
        
        # Anneal tau: start large, end small
        # Use exponential annealing
        tau = tau_init * (tau_final / tau_init) ** (elapsed_ratio ** 0.7)
        tau = max(tau, tau_final)
        
        conv = fast_convolve(f)
        grad, lse_val, w = lse_smoothed_grad(f, tau, conv)
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Check improvement
        new_obj = fast_convolve(f_new).max()
        
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.9
            v *= 0.5
        elif no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919 + 13)
            scale = min(0.02, 0.002 * (no_improve // 100))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init * min(1.0, no_improve / 200)
            no_improve = 0
        
        # Gradually reduce learning rate
        lr = lr_init * max(0.1, 1.0 - elapsed_ratio * 0.5)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-M iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, tau={tau:.2e}, lr={lr:.4f}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_fw_step(f, M_multiplier=1.0, time_per_lp=30):
    """Single LP-FW step: find direction and optimal step size."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return None
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max() * M_multiplier
    
    g = solve_lp_direction(f, M, n)
    if g is None:
        return None
    
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    g = g / g_sum
    
    if np.max(np.abs(g - f)) < 1e-13:
        return None
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, best_val
    
    return None


def freq_perturb(f, rng, strength=0.1):
    """Perturb f in frequency domain for diversity."""
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, N)
    
    # Perturb the magnitude and phase
    phase = np.angle(F)
    mag = np.abs(F)
    
    # Small random perturbation to phase
    phase_noise = rng.standard_normal(len(phase)) * strength * 0.1
    phase_new = phase + phase_noise
    
    # Small random perturbation to magnitude
    mag_noise = 1.0 + rng.standard_normal(len(mag)) * strength * 0.05
    mag_new = np.maximum(mag * mag_noise, 0)
    
    F_new = mag_new * np.exp(1j * phase_new)
    f_new = np.fft.irfft(F_new, N)[:n]
    
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    else:
        f_new = f.copy()
    
    return f_new


def coordinate_descent_fast(f0, time_limit, verbose=False):
    """Fast coordinate descent with multiplicative updates and random ordering."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    # Pre-compute multipliers
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.7]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-15:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f
