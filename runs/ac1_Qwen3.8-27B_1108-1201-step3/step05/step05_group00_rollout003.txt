"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated smoothed gradient
descent (Adam on log-sum-exp smoothing) with adaptive temperature, diverse
structured initializations (oscillatory, Gaussian, bell), multi-length
exploration, and coordinate-descent polish.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_grad_and_obj(a, beta):
    """
    Compute the smoothed objective (1/beta)*log(sum exp(beta*conv)) and its
    gradient w.r.t. a, using FFT-based correlation.
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    obj = (scaled.max() + np.log(exp_scaled.sum())) / beta

    # grad[j] = sum_{m=0}^{n-1} w[j+m] * a[m]  (j = 0..n-1)
    # = cross-correlation of w (len 2n-1) and a (len n), 'valid' mode
    # Computed as: corr = linear_conv(w, a_rev), grad = corr[n-1 : 2n-1]
    a_rev = a[::-1]
    N = 1 << ((2 * n - 1) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(W * Ar, N)
    grad = corr[n - 1: 2 * n - 1]
    return obj, grad, conv


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def adam_smoothed_optimize(a0, time_limit, beta_init=10.0, beta_max=200.0,
                           lr_init=0.01, lr_decay=0.999, beta_growth=1.002,
                           verbose=False, seed=42):
    """
    Adam optimization in log-space (x-space where a = softmax(x))
    minimizing the smoothed max of auto-convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    x = np.log(a) - np.log(a).mean()  # initialize x so that softmax(x) ≈ a

    beta = beta_init
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_obj = float('inf')
    best_a = a.copy()
    best_true_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax(x)

        obj, grad_a, conv = compute_smoothed_grad_and_obj(a, beta)

        # True objective for tracking
        true_obj = conv.max()
        true_score = 2.0 * n * true_obj
        if true_score < best_obj:
            best_obj = true_score
            best_a = a.copy()
            best_true_obj = true_obj
            no_improve = 0
            if verbose and t % 100 == 0:
                print(f"      Adam iter {t}: score={best_obj:.6f}, beta={beta:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Gradient in x-space: grad_x = a * (grad_a - dot(a, grad_a))
        grad_x = a * (grad_a - np.dot(a, grad_a))

        # Adam update
        m = b1 * m + (1 - b1) * grad_x
        v = b2 * v + (1 - b2) * grad_x * grad_x
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Increase beta (make smoothing sharper)
        beta = min(beta * beta_growth, beta_max)
        lr *= lr_decay

        # Periodic restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(t * 7 + seed)
            noise = rng.standard_normal(n) * 0.003
            a_pert = np.maximum(best_a + noise, 1e-15)
            a_pert = a_pert / a_pert.sum()
            x = np.log(a_pert) - np.log(a_pert).mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = lr_init * 0.5
            beta = max(beta_init, beta * 0.7)
            no_improve = 0

        if lr < 1e-7:
            break

    return best_a, best_obj


def peak_targeted_cd(a0, time_limit, verbose=False, seed=42):
    """
    Coordinate descent that specifically targets the largest convolution peaks
    by reducing the product of contributing pairs.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()
        improved = False

        # Get top peaks
        k_peaks = min(15, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        for k in peak_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            partners = k - idx
            products = a[idx] * a[partners]

            # Find the pair with maximum product
            pi = np.argmax(products)
            j_star = idx[pi]
            k_star = partners[pi]

            if a[j_star] < 1e-8:
                continue

            # Try reducing a[j_star] (and possibly a[k_star])
            best_trial = None
            best_trial_obj = current_obj

            for frac in [0.3, 0.5, 0.7, 0.8]:
                trial = a.copy()
                reduction = a[j_star] * (1.0 - frac)
                trial[j_star] = a[j_star] * frac
                # Redistribute to minimize impact on other peaks
                # Add to positions far from peak k
                dist = np.abs(np.arange(n) - j_star)
                weights = np.exp(-dist / (n * 0.1))
                weights[j_star] = 0
                wsum = weights.sum()
                if wsum > 1e-15:
                    trial += reduction * weights / wsum
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-12:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None:
                a = best_trial
                if best_trial_obj < best_obj - 1e-12:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Try a different strategy: reduce the top contributors to the global peak
                k_max = int(np.argmax(conv))
                conv = fast_convolve(a)
                k_max = int(np.argmax(conv))
                js = max(0, k_max - n + 1)
                je = min(n - 1, k_max)
                idx = np.arange(js, je + 1)
                partners = k_max - idx
                products = a[idx] * a[partners]
                # Reduce top 5 contributors
                top5 = np.argsort(products)[-5:]
                trial = a.copy()
                for pi in top5:
                    j = idx[pi]
                    k_p = partners[pi]
                    # Reduce the smaller of the two
                    if a[j] <= a[k_p]:
                        trial[j] *= 0.7
                    else:
                        trial[k_p] *= 0.7
                    # Redistribute
                    diff = a.sum() - trial.sum()
                    trial += diff / n
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    a = trial
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    # Random perturbation
                    noise = rng.standard_normal(n) * 0.002
                    a = np.maximum(best_a + noise, 0)
                    a = a / a.sum()
                    no_improve = 0
        else:
            no_improve = 0

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_oscillatory(n, freq, eps=0.1):
    """Oscillatory perturbation of uniform."""
    j = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 1e-6)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_sin_osc(n, freq, eps=0.1):
    j = np.arange(n)
    a = 1.0 + eps * np.sin(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 1e-6)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_score(a):
    """Compute the objective without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def multi_beta_sweep(a0, time_limit, verbose=False):

    # Distribute time among beta values, giving more time to middle values

            lr_init=0.004 / np.sqrt(target_n), lr_decay=0.9995,
            beta_growth=1.005, verbose=True
        )
        try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 5: Alternative Adam runs with different params ===
    print("\n=== Phase 5: Alternative Adam Configurations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {'beta_init': 5, 'beta_max': 100, 'lr_init': 0.01, 'lr_decay': 0.999, 'beta_growth': 1.005},
            {'beta_init': 50, 'beta_max': 500, 'lr_init': 0.002, 'lr_decay': 0.9998, 'beta_growth': 1.01},
            {'beta_init': 20, 'beta_max': 200, 'lr_init': 0.005, 'lr_decay': 0.9995, 'beta_growth': 1.008},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(rem / (len(configs) - ci), 120)
            print(f"  Config {ci}: t={t_alloc:.0f}s, beta_init={cfg['beta_init']}")

            # Slight perturbation for diversity
            noise = rng.standard_normal(n_best) * 0.002
            a_init = np.maximum(best_seq + noise, 1e-15)
            a_init = a_init / a_init.sum()

            a_opt, score = adam_smoothed_optimize(
                a_init, time_limit=t_alloc, verbose=(ci == 0), **cfg
            )
            try_seq(a_opt, f"cfg{ci}")

    # === Phase 6: Final Peak CD ===
    print("\n=== Phase 6: Final Peak CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, score = peak_targeted_cd(best_seq, time_limit=t_alloc, seed=123)
            try_seq(a_opt, "final_peak_cd")

    # === Phase 7: Fine Adam polish ===
    print("\n=== Phase 7: Fine Adam Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 60)
            n_best = len(best_seq)
            a_opt, score = adam_smoothed_optimize(
                best_seq, time_limit=t_alloc,
                beta_init=200, beta_max=1000,
                lr_init=0.001 / np.sqrt(n_best), lr_decay=0.9999,
                beta_growth=1.005, verbose=True
            )
            try_seq(a_opt, "fine_polish")

    # === Phase 8: One more CD round ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 30)
            a_opt, score = peak_targeted_cd(best_seq, time_limit=t_alloc, seed=456)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = compute_score(best_seq)

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Shouldn't happen, but just in case
        best_seq = np.ones(384) / 384
        best_score = compute_score(best_seq)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
