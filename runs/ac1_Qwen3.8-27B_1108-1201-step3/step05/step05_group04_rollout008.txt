```python
    """Initialize with near-flat power spectrum."""
    # Generate random phases
    m = 2 * n  # oversize for FFT
    phases = rng.uniform(0, 2*np.pi, m)
    # Flat magnitude, random phases
    A = np.exp(1j * phases)
    a = np.fft.irfft(A, m)[:n]
    a = np.abs(a)  # take positive part
    a = np.maximum(a, 1e-10)
    return a / a.sum()
```

```python
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_harp(f, conv, top_k):
    """Compute subgradient using hard top-k selection of convolution peaks."""
    n = len(f)
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]

    sg = np.zeros(n)
    for k in peak_indices:
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            m_range = np.arange(lo1, hi1 + 1)
            sg[m_range] += f[k - m_range]
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            m_range = np.arange(lo2, hi2 + 1)
            sg[m_range] += f[m_range - k]
    return sg / len(peak_indices)


def compute_subgradient_soft(f, conv, temp, top_k=None):
    """Compute subgradient using soft-max weighting."""
    n = len(f)
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_m = np.zeros(len(conv))
        w_m[idx] = w[idx]
        w_m /= w_m.sum()
        w = w_m
    return np.sum(w[:, None] * _pointwise_grad(f, conv, np.arange(len(conv))), axis=0)


def _pointwise_grad(f, conv, ks):
    """Compute gradient of conv[k] w.r.t. f for each k in ks (vectorized)."""
    n = len(f)
    grads = np.zeros((len(ks), n))
    for i, k in enumerate(ks):
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            m_range = np.arange(lo1, hi1 + 1)
            grads[i, m_range] += f[k - m_range]
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            m_range = np.arange(lo2, hi2 + 1)
            grads[i, m_range] += f[m_range - k]
    return grads


def objective_normalized(f):
    """Compute objective for a sequence normalized to sum=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def spectral_init(n, rng, n_modes=None):
    """Initialize with near-flat power spectrum using random phases."""
    m = 2 * n
    if n_modes is None:
        n_modes = m
    phases = rng.uniform(0, 2 * np.pi, m)
    A = np.exp(1j * phases)
    a = np.fft.irfft(A * m, m)[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spectral_init_tapered(n, rng, taper_power=2):
    """Spectral init with tapered magnitude for smoother time-domain signal."""
    m = 2 * n
    phases = rng.uniform(0, 2 * np.pi, m)
    freqs = np.arange(m) / m
    taper = (1 - freqs) ** taper_power
    taper[freqs > 0.5] = (1 - (freqs[freqs > 0.5] - 0.5)) ** taper_power
    A = taper * np.exp(1j * phases)
    a = np.fft.irfft(A * m, m)[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_smooth_random(n, rng, n_components=8):
    """Create a smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.03, 0.3)
        a = rng.uniform(0.5, 3.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def optimize_additive(f0, time_limit, lr_init=0.004, lr_decay=0.9997,
                      top_k_init=60, top_k_min=10, top_k_rate=0.9998,
                      temp_init=0.005, temp_decay=0.9995, temp_min=0.001,
                      use_hard=True, verbose=False):
    """Additive (projected gradient) optimization with adaptive top-k."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    lr = lr_init
    top_k = top_k_init
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        if use_hard:
            cur_k = max(int(top_k), top_k_min)
            sg = compute_subgradient_harp(f, conv, cur_k)
        else:
            sg = compute_subgradient_soft(f, conv, temp, top_k=max(int(top_k), top_k_min))

        f_new = f - lr * sg
        f = project_simplex(f_new)

        lr *= lr_decay
        lr = max(lr, lr_init * 0.05)
        top_k = max(top_k * top_k_rate, top_k_min)
        temp = max(temp * temp_decay, temp_min)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng_local = np.random.default_rng(t * 31 + 7)
            noise = rng_local.standard_normal(n) * (0.002 + 0.003 * min(no_improve // 200, 3))
            f = project_simplex(best_f + noise)
            lr = lr_init
            top_k = top_k_init
            temp = max(temp_init, 0.003)
            no_improve = 0

    return best_f, best_obj


def optimize_multiplicative(f0, time_limit, lr_init=0.01, lr_decay=0.9995,
                            top_k=40, temp_init=0.008, temp_decay=0.9994,
                            verbose=False):

    f = f0.astype(np.float64).copy()
    s = f.sum()

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:

            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.004)
            lr = lr_init
            no_improve = 0

    return best_f, best_obj


def optimize_combined(f0, time_limit, verbose=False):
    """Alternating additive and multiplicative updates."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    phase = 0
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        chunk_time = 0.5
        chunk_end = time.monotonic() + chunk_time

        if time.monotonic() + 2 > time.monotonic() + time_limit - t0:
            break

        if phase % 3 == 0:
            # Additive phase
            lr = 0.003 * (0.999 ** t)
            lr = max(lr, 0.0005)
            top_k = max(10, 60 - t // 20)

            sub_iters = 0
            while time.monotonic() < chunk_end and sub_iters < 5:
                conv = fast_convolve(f)
                sg = compute_subgradient_harp(f, conv, top_k)
                f = project_simplex(f - lr * sg)
                sub_iters += 1
                lr *= 0.998

        elif phase % 3 == 1:
            # Multiplicative phase
            lr_m = 0.008 * (0.999 ** t)
            lr_m = max(lr_m, 0.001)
            top_k = max(10, 50 - t // 20)

            sub_iters = 0
            while time.monotonic() < chunk_end and sub_iters < 5:
                conv = fast_convolve(f)
                sg = compute_subgradient_harp(f, conv, top_k)
                log_update = -lr_m * sg
                log_update -= log_update.max()
                f_new = f * np.exp(log_update)
                fs = f_new.sum()
                if fs > 1e-15:
                    f = f_new / fs
                sub_iters += 1
                lr_m *= 0.998

        else:
            # Symmetrization + small noise phase
            f_sym = (f + f[::-1]) / 2
            f_sym = np.maximum(f_sym, 0)
            ss = f_sym.sum()
            if ss > 1e-15:
                f = f_sym / ss
            conv = fast_convolve(f)
            if conv.max() > best_obj:
                f = best_f.copy()
            phase += 1
            no_improve += 1
            continue

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            rng_local = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            no_improve = 0

        phase += 1

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Fast coordinate descent: try scaling each coordinate by a factor."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-10:
                continue

            old_val = f[j]
            best_val = old_val
            best_val_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-10:
                    continue
                f_trial /= ss

                val = fast_convolve(f_trial).max()
                if val < best_val_obj - 1e-14:
                    best_val_obj = val
                    best_val = new_val

            if best_val != old_val:
                f[j] = best_val
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                ss = f.sum()
                if ss > 1e-10:
                    f /= ss
                if best_val_obj < best_obj - 1e-14:
                    best_obj = best_val_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj

            rem = t_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                n_best = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                a_opt, _ = coordinate_descent_polish(perturbed, time_limit=min(rem * 0.7, 40))
                try_seq(a_opt, "p5_cd_pert")

    # === Phase 6: Final micro-optimizations ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine additive
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem * 0.4, 40)
            a_opt, _ = optimize_additive(best_seq, time_limit=t_alloc,
                                         lr_init=0.0005, lr_decay=0.99995,
                                         top_k_init=20, top_k_min=5)
            try_seq(a_opt, "p6_fine")

        # Tiny perturbations
        for pi in range(5):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = optimize_additive(f_pert, time_limit=min(rem * 0.4, 15),
                                         lr_init=0.001, top_k_init=20, top_k_min=5)
            try_seq(a_opt, f"p6_pert_{pi}")

        # Symmetrize final
        final_sym = (best_seq + best_seq[::-1]) / 2
        final_sym = np.maximum(final_sym, 0)
        ss = final_sym.sum()
        if ss > 1e-10:
            final_sym /= ss
        try_seq(final_sym, "p6_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
