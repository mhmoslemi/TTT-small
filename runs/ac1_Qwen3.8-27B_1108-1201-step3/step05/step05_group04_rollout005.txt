def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

def soft_max_weights(conv, T, top_k=None):
    """Compute soft-max weights over convolution values with temperature T."""
    shifted = T * conv
    shifted -= shifted.max()
    e = np.exp(shifted)
    ws = e.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w = e / ws
    if top_k is not None and top_k < len(w):

        wm = np.zeros(len(w))
        wm[idx] = w[idx]
        ws2 = wm.sum()
        if ws2 > 1e-300:

        w = wm
    return w

    max_val = conv.max()
    shifted = T * (conv - max_val)
    e = np.exp(shifted)
    smooth_val = max_val + np.log(e.sum()) / T
    w = e / e.sum()
    grad = subgrad_from_weights(w, a)

def peak_reduction(a0, time_limit, n_iters=50000):
    """Iterative peak reduction: reduce the peak of the convolution.
    
    At each step, find the peak position, then multiplicatively reduce
    the coordinates that contribute to it.
    """
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.1  # step size
    momentum_buf = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        # Focus on top few peaks
        top_k = min(10, len(conv))
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]
        w = np.zeros(len(conv))
        w[peak_idx] = peak_vals ** 2
        w /= w.sum()

        sg = subgrad_from_weights(w, a)

        # Multiplicative update with momentum
        momentum_buf = 0.7 * momentum_buf + 0.3 * sg
        update = np.exp(-eta * momentum_buf)
        a_new = a * update
        sn = a_new.sum()
        if sn < 1e-30:
            a_new = a.copy()
        else:
            a_new /= sn

        cur_obj = fast_convolve(a_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()

        a = a_new
        # Decay step size slowly
        eta = max(0.01, eta * 0.9999)

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_smooth_init(n, rng, n_components=8):
    """Create a smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.08, 0.30)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def symmetric_init(n):
    """Create a symmetric (palindromic) initialization."""
    half = n // 2
    x = np.linspace(0, 1, half + 1)
    f_half = np.sin(np.pi * x) ** 2

    else:
        f = np.concatenate([f_half, f_half[-1:0:-1]])
    f = np.maximum(f, 1e-12)
    return f / f.sum()

            print(f"  L-BFGS config {li} (T={cfg['T_start']}->{cfg['T_end']}), t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_softmax_optimize(base2, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"p2_lbfgs_{li}")

    # === Phase 3: Peak reduction ===
    print(f"\n=== Phase 3: Peak reduction (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 200

    if best_seq is not None:
        rem = t3_end - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.8, 150)
            print(f"  Peak reduction, t={t_alloc:.0f}s")
            a_opt, obj = peak_reduction(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p3_peak")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 350

    if best_seq is not None:
        base4 = best_seq.copy()
        n_base = len(base4)

        # Try a range of lengths
        candidate_lengths = [384, 512, 640, 896, 1024, 1152, 1280, 1536, 2048]
        candidate_lengths = [t for t in candidate_lengths if t != n_base and 256 <= t <= 2048]
        candidate_lengths.sort(key=lambda x: abs(np.log(x / n_base)))
        candidate_lengths = candidate_lengths[:5]

        for target_n in candidate_lengths:
            rem = t4_end - time.monotonic()
            if rem < 40:
                break
            f_interp = interpolate_sequence(base4, target_n)
            t_alloc = min(rem * 0.55, 120)

            print(f"  Length n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_with_linesearch(
                f_interp, time_limit=t_alloc,
                lr_init=0.03, T_init=50.0, T_end=3000.0,
                momentum=0.88, top_k=max(20, target_n // 15)
            )
            try_seq(a_opt, f"p4_n{target_n}")

            # Quick L-BFGS polish
            rem = t4_end - time.monotonic()
            if rem > 25 and best_seq is not None and len(best_seq) == target_n:
                a_opt2, _ = lbfgs_softmax_optimize(
                    best_seq, time_limit=min(rem * 0.4, 50),
                    T_start=200.0, T_end=5000.0, n_outer=10
                )
                try_seq(a_opt2, f"p4_lbfgs_n{target_n}")

    # === Phase 5: Diverse restarts + Adam ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 200

    if best_seq is not None:
        n_best = len(best_seq)
        starts = [
            ('noise_small', project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)),
            ('noise_med', project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)),
            ('smooth_mix', project_simplex(0.7 * best_seq + 0.3 * make_smooth_init(n_best, rng))),
            ('sym_mix', project_simplex(0.6 * best_seq + 0.4 * symmetric_init(n_best))),
        ]

        for si, (sname, sinit) in enumerate(starts):
            rem = t5_end - time.monotonic()
            if rem < 25:
                break
            t_alloc = min(rem * 0.6, 60)
            print(f"  Restart '{sname}', t={t_alloc:.0f}s")
            a_opt, _ = adam_with_linesearch(
                sinit, time_limit=t_alloc,
                lr_init=0.02, T_init=50.0, T_end=3000.0,
                momentum=0.88, top_k=max(15, n_best // 12)
            )
            try_seq(a_opt, f"p5_{sname}")

    # === Phase 6: Coordinate descent polishing ===
    print(f"\n=== Phase 6: CD polishing (best={best_score:.8f}) ===")
    t6_end = time.monotonic() + 200

    if best_seq is not None:
        rem = t6_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.7, 120)
            print(f"  CD, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p6_cd")

            # Another round from perturbed
            rem = t6_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                n_best = len(best_seq)
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
                a_opt, _ = coordinate_descent_polish(pert, time_limit=min(rem * 0.7, 60))
                try_seq(a_opt, "p6_cd_pert")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 10
        if rem > 20:
            # Very fine Adam
            t_alloc = min(rem * 0.5, 80)
            print(f"  Fine Adam, t={t_alloc:.0f}s")
            a_opt, _ = adam_with_linesearch(
                best_seq, time_limit=t_alloc,
                lr_init=0.005, T_init=200.0, T_end=10000.0,
                momentum=0.93, top_k=max(10, n_best // 20)
            )
            try_seq(a_opt, "p7_fine")

            # Fine L-BFGS
            rem = deadline - time.monotonic() - 10
            if rem > 15:
                a_opt, _ = lbfgs_softmax_optimize(
                    best_seq, time_limit=min(rem * 0.5, 60),
                    T_start=500.0, T_end=10000.0, n_outer=15
                )
                try_seq(a_opt, "p7_lbfgs_fine")

            # Tiny perturbations
            for pi in range(4):
                rem = deadline - time.monotonic() - 5
                if rem < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = adam_with_linesearch(
                    f_pert, time_limit=min(rem * 0.4, 25),
                    lr_init=0.003, T_init=300.0, T_end=8000.0,
                    momentum=0.94, top_k=max(5, n_best // 30)
                )
                try_seq(a_opt, f"p7_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
