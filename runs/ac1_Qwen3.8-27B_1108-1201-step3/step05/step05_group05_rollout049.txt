For k = n-1: A[n-1, 0] = a[n-1], ..., A[n-1, n-1] = a[0].
For k = n: A[n, 1] = a[n-1], ..., A[n, n-1] = a[0].
...
For k = 2n-2: A[2n-2, n-1] = a[n-1].

So the matrix is:
Row 0:    a[0]  0     0     ...  0
Row 1:    a[1]  a[0]  0     ...  0
Row 2:    a[2]  a[1]  a[0]  ...  0
...
Row n-1:  a[n-1] a[n-2] ... a[0]
Row n:    0      a[n-1] ... a[0]
...
Row 2n-2: 0      0      ... a[n-1]

This is a "convolution matrix" (also called a Toeplitz-like matrix for the linear convolution).

Now, for the LP:
max c^T g = 1^T g
s.t. A g <= M * 1  (where 1 is a vector of ones of length 2n-1)
     g >= 0

Using scipy.optimize.linprog:
- c = -np.ones(n) (we negate because linprog minimizes)
- A_ub = A, b_ub = M * np.ones(2n-1)
- bounds = [(0, None)] * n
- method='highs' (or 'highs-ds' for sparse)

For n=768, A is 1535 x 768. This should be fast.

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the Dantzig-Wolfe style LP to find directions that reduce the
auto-convolution peak, combined with line search and active-set refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_coefficients(a, n):
    """Build the constraint matrix A for the LP.
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    Shape: (2n-1, n)
    """
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = a[k - i]
    return A


def solve_lp_direction(a, M):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns (g_star, S) where S = sum(g_star).
    """
    n = len(a)
    A = build_conv_matrix_coefficients(a, n)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g_star = result.x
        S = g_star.sum()
        return g_star, S
    else:
        return None, 0.0


def lp_guided_step(a, M, max_iters=50):
    """Perform LP-guided optimization steps.
    Returns (best_a, best_M, num_iters).
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, M, 0
    a = a / s
    
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    
    for it in range(max_iters):
        # Solve LP
        g_star, S = solve_lp_direction(a, best_M)
        
        if g_star is None or S <= 1.0 + 1e-8:
            break
        
        # Target direction
        target = g_star / S
        
        # Line search
        best_t = 0.0
        best_obj = best_M
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t) * a + t * target
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            obj = fast_convolve(a_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.05, 0.02, 0.01]:
                a_trial = (1 - t) * a + t * target
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                obj = fast_convolve(a_trial).max()
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * target
        a = np.maximum(a, 0)
        a = a / a.sum()
        current_M = fast_convolve(a).max()
        
        if current_M < best_M - 1e-12:
            best_M = current_M
            best_a = a.copy()
    
    return best_a, best_M, max_iters


def smooth_objective_and_grad(a, tau=50):
    """Compute smoothed max(conv(a,a)) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Log-sum-exp for numerical stability
    max_conv = conv.max()
    exp_conv = np.exp(tau * (conv - max_conv))
    sum_exp = exp_conv.sum()
    if sum_exp < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_conv / sum_exp
    
    obj = (max_conv + np.log(sum_exp)) / tau
    
    # Gradient: grad_j = sum_k w_k * 2 * a[k-j] (for valid k-j)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * w[k] * a[k - i]
    
    # Center the gradient (since we're on the simplex, the direction matters)
    grad = grad - grad.mean()
    
    return obj, grad


def lbfgs_smooth_optimize(a_init, time_limit, tau=50):
    """Optimize using L-BFGS on the smoothed objective with simplex constraint."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a_init)) / len(a_init)
    else:
        a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    # Use projected gradient with momentum
    v = np.zeros(n)
    momentum = 0.8
    lr = 0.05
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        obj, grad = smooth_objective_and_grad(a, tau)
        
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v, 1.0)
        
        M_new = fast_convolve(a_new).max()
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive parameters
        if no_improve > 100:
            momentum = min(0.95, momentum + 0.01)
            lr = max(0.001, lr * 0.95)
        elif no_improve > 50:
            lr = min(0.1, lr * 1.02)
        
        if no_improve > 300:
            # Restart
            rng_r = np.random.default_rng(int(time.monotonic() * 1000))
            a = project_simplex(best_a + rng_r.standard_normal(n) * 0.003, 1.0)
            v = np.zeros(n)
            no_improve = 0
            momentum = 0.8
            lr = 0.04
            tau = max(20, tau * 0.9)
        
        if no_improve % 50 == 0:
            tau = min(500, tau * 1.05)
    
    return best_a, best_M


def coordinate_descent_fast(a0, time_limit):
    """Fast coordinate descent focusing on the most impactful coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Precompute which lags are "hot" (near the max)
    conv = fast_convolve(a)
    threshold = best_M * 0.95
    hot_lags = np.where(conv >= threshold)[0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Focus on coordinates that affect hot lags
        affected_coords = set()
        for k in hot_lags:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                affected_coords.add(i)
        
        for j in sorted(affected_coords):
            if time.monotonic() - t0 > time_limit:
                break
            
            if a[j] < 1e-12:
                continue
            
            old_val = a[j]
            best_j_val = old_val
            best_j_obj = best_M
            
            for mult in [0.7, 0.85, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                scale = (1.0 - best_j_val) / old_remaining
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a /= a.sum()
                if best_j_obj < best_M - 1e-14:
                    best_M = best_j_obj
                    best_a = a.copy()
                    improved = True
                    # Update hot lags
                    conv = fast_convolve(a)
                    threshold = best_M * 0.95
                    hot_lags = np.where(conv >= threshold)[0]
        
        if not improved:
            break
    
    return best_a, best_M

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M, g >= 0.
    Returns (g_star, S, dual_y) or (None, 0, None).
    """
    n = len(a)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = a[k - i]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g_star = result.x
        S = g_star.sum()
        # Dual variables (for the inequality constraints)
        dual_y = result.ineqlin.marginals if hasattr(result, 'ineqlin') else None
        # Actually, for scipy's linprog, the duals are in result.ineqlin.marginals
        # Let me use the correct attribute
        try:
            dual_y = -result.ineqlin.marginals  # Negated because of minimization
        except:
            dual_y = None
        return g_star, S, dual_y
    else:
        return None, 0.0, None
