"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Nesterov-accelerated subgradient on the simplex with active-set peak targeting
2. Pair-transfer coordinate descent focused on peak lags
3. Multi-length interpolation and re-optimization
4. Diverse initialization with structured and random starts
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a, conv, top_k=20):
    """Compute subgradient of max(c) w.r.t. a using top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    # Weight by softmax of peak values
    temp = max(1e-6, (pv.max() - pv.min()) * 0.5)
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for idx in range(k_actual):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # Gradient of c[k] w.r.t. a
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            j = k - i
            if i == j:
                sg[i] += 2.0 * w[idx] * a[i]
            else:
                sg[i] += w[idx] * a[j]
                sg[j] += w[idx] * a[i]
    
    return sg


def nesterov_subgradient(a_init, time_limit, lr0=0.01, top_k=30, verbose=False):
    """Nesterov-accelerated subgradient with averaging."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = compute_score(a)
    
    avg_a = np.zeros(n)
    avg_count = 0
    t = 0
    beta = 0  # momentum
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        beta = (1 - 0.1) / (1 + 0.1 * t)  # Nesterov momentum
        # Actually, standard Nesterov: beta_t = (1 - sqrt(t))/... 
        # Let's use a simpler schedule
        beta = (t - 1) / t if t > 1 else 0
        
        conv = fast_convolve(y)
        sg = subgradient_of_max_conv(y, conv, top_k)
        
        # Step size: 1/sqrt(t)
        step_size = lr0 / np.sqrt(t)
        
        # Subgradient step
        new_y = project_simplex(y - step_size * sg)
        
        # Nesterov update
        a_new = beta * y + (1 - beta) * new_y
        a = a_new
        y = new_y
        
        # Track average
        avg_a += a
        avg_count += 1
        
        # Periodic evaluation
        if t % 500 == 0 or t == 1:
            sc = compute_score(a)
            sc_avg = compute_score(avg_a / avg_count)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = (avg_a / avg_count).copy()
            
            if verbose and t % 5000 == 0:
                print(f"    NS step {t}: score={best_score:.10f}")
            
            # Restart if not improving
            if t % 10000 == 0:
                a = project_simplex(best_a + np.random.standard_normal(n) * 0.001)
                y = a.copy()
                avg_a = a.copy()
                avg_count = 1
    
    # Final evaluation
    sc = compute_score(a)
    sc_avg = compute_score(avg_a / avg_count)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = (avg_a / avg_count).copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Adaptive subgradient with Nesterov acceleration and periodic restarts
2. Peak-targeted 2D coordinate descent
3. Multi-length exploration with structured initializations
   including the 'ideal flat-convolution' sequence

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def ideal_flat_conv_seq(n):
    """Generate the 'ideal' sequence whose infinite auto-convolution is flat.
    b[k] = C(2k,k) / 4^k, truncated to length n and normalized."""
    k = np.arange(n, dtype=np.float64)
    # Compute C(2k,k)/4^k using log-gamma for numerical stability
    from math import lgamma
    log_b = lgamma(2*k + 1) - lgamma(k + 1) - lgamma(k + 1) - k * np.log(4)
    b = np.exp(log_b)
    return normalize(b)


def symmetric_ideal_seq(n):
    """Symmetric version: use the ideal sequence and mirror it."""
    half = (n + 1) // 2
    bh = ideal_flat_conv_seq(half)
    if n % 2 == 0:
        a = np.concatenate([bh[::-1], bh])
    else:
        a = np.concatenate([bh[::-1][:-1], bh])
    return normalize(a)


def geometric_sequence(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    return normalize(np.sin(x) ** 2)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    return normalize(np.exp(-x**2 / (2 * sigma**2)))


def uniform_sequence(n):
    return np.ones(n) / n


def compute_subgradient(a, conv, top_k=20):
    """Subgradient of max(conv) w.r.t. a using top-k peaks with softmax weights."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    temp = max(1e-8, (pv.max() - pv.min()) * 0.3)
    scaled = (pv - pv.max()) / temp
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for idx in range(k_actual):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            j = k - i
            if i == j:
                sg[i] += 2.0 * wi * a[i]
            else:
                sg[i] += wi * a[j]
                if j >= 0 and j < n:
                    sg[j] += wi * a[i]
    
    return sg


def subgradient_nesterov(a_init, time_limit, lr0=0.05, top_k=20, restart_period=5000, verbose=False):
    """Nesterov-accelerated subgradient with averaging and restarts."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    avg_a = np.zeros(n)
    avg_count = 0
    t = 0
    t0 = time.monotonic()
    best_at_restart = best_score
    steps_since_restart = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        steps_since_restart += 1
        
        # Nesterov momentum
        beta = (t - 1) / (t + 0.5) if t > 1 else 0.0
        
        conv = fast_convolve(y)
        sg = compute_subgradient(y, conv, top_k)
        
        # Adaptive step size
        step_size = lr0 / (1 + 0.1 * np.sqrt(t))
        
        new_y = project_simplex(y - step_size * sg)
        a = beta * y + (1 - beta) * new_y
        y = new_y
        
        avg_a += a
        avg_count += 1
        
        # Periodic evaluation
        eval_period = 200
        if t % eval_period == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_at_restart = best_score
            
            if avg_count % (eval_period * 5) == 0:
                sc_avg = compute_score(avg_a / avg_count)
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = (avg_a / avg_count).copy()
                avg_a = np.zeros(n)
                avg_count = 0
            
            if verbose and t % (eval_period * 10) == 0:
                print(f"    NS t={t}: score={best_score:.10f}")
        
        # Restart if stuck
        if steps_since_restart >= restart_period and best_score >= best_at_restart * 0.999:
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.005)
            y = a.copy()
            avg_a = a.copy()
            avg_count = 1
            steps_since_restart = 0
            t = 0  # Reset momentum
            best_at_restart = best_score
            if verbose:
                print(f"    NS restart at t={t}, score={best_score:.10f}")
    
    # Final
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def coord_descent_2d(a_init, time_limit, verbose=False):
    """2D coordinate descent: optimize pairs of coordinates via mass transfer."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Focus on peak-contributing pairs
        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        
        # Generate candidate pairs
        npairs = 500
        pairs = []
        
        # Peak-targeted pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        peak_pairs = [(i, peak_k - i) for i in range(js, je + 1) if i != peak_k - i]
        if peak_pairs:
            np.random.shuffle(peak_pairs)
            pairs.extend(peak_pairs[:npairs // 2])
        
        # Random pairs
        while len(pairs) < npairs:
            i, j = rng.integers(0, n, 2)
            if i != j:
                pairs.append((i, j))
        
        for i, j in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if i == j:
                continue
            
            ai = a[i]
            aj = a[j]
            if ai + aj < 1e-14:
                continue
            
            # Optimize delta: a[i] += delta, a[j] -= delta
            # delta in [-ai, aj]
            lo = -ai
            hi = aj
            
            def obj_delta(delta):
                if delta < lo - 1e-12 or delta > hi + 1e-12:
                    return float('inf')
                delta = np.clip(delta, lo, hi)
                new_a = a.copy()
                new_a[i] = ai + delta
                new_a[j] = aj - delta
                conv_new = fast_convolve(new_a)
                return conv_new.max()
            
            best_obj = obj_delta(0)
            
            # Golden section / bounded minimize
            try:
                res = minimize_scalar(obj_delta, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 20})
                if res.success and res.fun < best_obj - 1e-14:
                    delta = res.x
                    a[i] = ai + delta
                    a[j] = aj - delta
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    sc = compute_score(a)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break
    
    return best_a, best_score


def peak_shaving(a_init, time_limit, verbose=False):
    """Aggressively shave the peaks of the auto-convolution by reducing
    the coordinates that contribute most to the peak lags."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        
        # Find top peaks
        top_k = min(10, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # For each peak, identify the coordinates that contribute most
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Get contributions
            contributions = []
            for i in range(js, je + 1):
                j = k - i
                if i == j:
                    contrib = a[i] ** 2
                else:
                    contrib = 2 * a[i] * a[j]
                contributions.append((contrib, i, j))
            
            contributions.sort(reverse=True)
            
            # Try reducing the top contributors
            for contrib, i, j in contributions[:3]:
                if time.monotonic() - t0 > time_limit:
                    break
                if contrib < 1e-14:
                    continue
                
                ai = a[i]
                if ai < 1e-12:
                    continue
                
                # Try reducing a[i] and redistributing
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    reduction = ai * frac
                    new_a = a.copy()
                    new_a[i] = ai - reduction
                    # Redistribute to random other coordinates
                    others = list(range(n))
                    others.remove(i)
                    rng.shuffle(others)
                    # Give to a few random others
                    n_others = min(5, len(others))
                    for oi in others[:n_others]:
                        new_a[oi] += reduction / n_others
                    
                    new_a = np.maximum(new_a, 0)
                    new_a /= new_a.sum()
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = new_a.copy()
                        a = new_a.copy()
                        break
    
    return best_a, best_score
