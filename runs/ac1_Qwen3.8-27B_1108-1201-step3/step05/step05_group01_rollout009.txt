def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_direction_sparse(f, M, n, active_set=None):
    """Solve LP using only a subset of active constraints for speed."""
    c = -np.ones(n)
    if active_set is None or len(active_set) == 0:
        A_ub = np.zeros((2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
    else:
        A_ub = np.zeros((len(active_set), n))
        b_ub = np.full(len(active_set), M)
        for idx, k in enumerate(active_set):
            for j in range(n):
                A_ub[idx, j] = f[j] if 0 <= k - j < n else 0.0
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smooth_obj_grad(t, beta, top_k=None):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    if top_k is not None and top_k < len(conv):
        # Use only top-k peaks for efficiency
        valid_idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_shifted = conv[valid_idx]
    else:
        conv_shifted = conv

    conv_max = conv_shifted.max()
    shifted = beta * (conv_shifted - conv_max)
    exp_shifted = np.exp(shifted)
    w_full = np.zeros(len(conv))
    if top_k is not None and top_k < len(conv):
        w_full[valid_idx] = exp_shifted
    else:
        w_full = exp_shifted
    w = w_full / w_full.sum()

    obj = conv_max + np.log(w_full.sum()) / beta

    # Gradient: grad_a[i] = 2 * sum_k w[k] * a[|k-i| adjusted for convolution]
    # For convolve(a,a)[k] = sum_i a[i]*a[k-i], d/d(a[j]) = sum_k w[k] * (a[k-j] + a[j-k])
    # Simplified via FFT cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # grad_a[j] = 2 * (w * a)[j] where (w*a) is cross-correlation
    R = np.fft.irfft(W * A, N)  # This gives sum_i w[i]*a[i+j] (correlation)
    R2 = np.fft.irfft(np.conj(W) * A, N)  # Conjugate version
    grad_a = R[:n] + R2[:n]
    # Actually let me be more careful:
    # b[k] = sum_{i+j=k} a[i]*a[j]
    # db[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n) ... 
    # Actually db[k]/da[j] = a[k-j] if 0<=k-j<n, else 0
    # So sum_k w[k]*db[k]/da[j] = sum_k w[k]*a[k-j] for valid k
    # = sum_i w[i+j]*a[i] (substituting k=i+j) -- this is cross-correlation of w and a
    # FFT cross-correlation: IFFT(conj(FFT(a)) * FFT(w))
    grad_a = np.fft.irfft(np.conj(A) * W, N)[:n] * 2.0

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-16:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use smoother interpolation
    f_interp = np.interp(new_x, old_x, f, kind='cubic')
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        try_seq(a_opt, "lbfqg_basin")

        # Try from perturbed base
        if remaining_time() > 1100:
            noise = rng.standard_normal(len(base)) * 0.02
            f_pert = project_simplex(base + noise)
            t1b = min(60, remaining_time() - 1100)
            if t1b > 20:
                a_opt, _ = lbfqg_optimize(f_pert, time_limit=t1b)
                try_seq(a_opt, "lbfqg_pert")

    # === Phase 2: LP-FW main optimization (500s) ===
    print(f"\n=== Phase 2: LP-FW Main (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(500, remaining_time() - 800)
    if t2 > 40:
        a_opt, _ = lp_fw_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_main")

        # Second LP-FW run from slightly different start
        if remaining_time() > 800 and best_seq is not None:
            noise = rng.standard_normal(len(base)) * 0.005
            f2 = project_simplex(best_seq + noise)
            t2b = min(100, remaining_time() - 800)
            if t2b > 20:
                a_opt, _ = lp_fw_optimize(f2, time_limit=t2b)
                try_seq(a_opt, "lpfw_second")

    # === Phase 3: Peak shaving (150s) ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(150, remaining_time() - 500)
    if t3 > 20:
        a_opt, _ = peak_shaving_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "peak_shave")

    # === Phase 4: Multi-length exploration (300s) ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    # Choose target lengths around base
    target_lengths = sorted(set([
        int(n_base * 0.75), int(n_base * 0.85), int(n_base * 1.1),
        int(n_base * 1.2), int(n_base * 1.35)
    ]))
    # Filter to reasonable range
    target_lengths = [t for t in target_lengths if 256 <= t <= 2048 and t != n_base]

    per_len_time = max(10, min(80, (remaining_time() - 400) / max(1, len(target_lengths))))
    for target_n in target_lengths[:4]:
        if remaining_time() < per_len_time + 300:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = per_len_time
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        # Quick L-BFGS then LP-FW
        t_half = t_alloc * 0.4
        a_opt, _ = lbfqg_optimize(f_interp, time_limit=t_half)
        try_seq(a_opt, f"len_{target_n}_lbfqg")
        t_rest = t_alloc - t_half
        if t_rest > 10:
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_rest)
            try_seq(a_opt, f"len_{target_n}_lpfw")

    # === Phase 5: Deep optimization on best (300s) ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: LP-FW deep with more time
        rem = remaining_time() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 150)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5b: L-BFGS with higher beta
        rem = remaining_time() - 120
        if rem > 20:
            t_alloc = min(rem * 0.4, 80)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 5c: Peak shaving deep
        rem = remaining_time() - 80
        if rem > 20:
            t_alloc = min(rem * 0.4, 60)
            a_opt, _ = peak_shaving_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_peak")

        # 5d: Coordinate descent
        rem = remaining_time() - 40
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

    # === Phase 6: Final exploitation (remaining time) ===
    print(f"\n=== Phase 6: Final Exploitation (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Alternating refinement loop
        attempt = 0
        while remaining_time() > 20:
            attempt += 1
            rem = remaining_time() - 10
            if rem < 10:
                break

            if attempt % 3 == 1:
                # LP-FW with tiny perturbation
                noise = rng.standard_normal(n_best) * 0.001
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.8, 60)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 2:
                # L-BFGS
                t_alloc = min(rem * 0.8, 50)
                a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")
            else:
                # Peak shaving
                t_alloc = min(rem * 0.8, 40)
                a_opt, _ = peak_shaving_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_peak_{attempt}")

        # Very final coordinate descent
        rem = remaining_time() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "absolute_final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
