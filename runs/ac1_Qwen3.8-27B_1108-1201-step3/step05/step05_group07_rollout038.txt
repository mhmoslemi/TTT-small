"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based block coordinate descent,
smooth gradient optimization with adaptive temperature, coordinate descent, and
multi-length exploration. Key innovation: direct LP formulation for the peak
constraint enables principled descent that can find sparse solutions, combined
with smooth refinement for polishing.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix A such that A @ g = np.convolve(a, g)."""
    n = len(a)
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def score_normalized(a):
    """Compute score assuming sum(a)=1. Returns 2*n*max_b."""
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max())


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_fft(w, a):
    """Cross-correlation via FFT: result[i] = sum_k w[k]*a[k-i]."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def smooth_obj_grad(a, T, n):
    """
    Smooth surrogate objective and gradient w.r.t. a (on simplex).
    Obj: (2n/T) * log(sum exp(T*b_k/sum_a^2))
    Simplified for sum_a=1: (2n/T) * logsumexp(T*b)
    """
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # Weights
    w = exp_Tb / exp_Tb.sum()

    # grad_a: d/da_i of sum_k w_k * b_k = 2 * cross_corr(w, a)_i
    grad_a = 2.0 * n * cross_corr_fft(w, a)

    # Project gradient to simplex tangent space
    S = np.dot(grad_a, a)
    grad = grad_a - a * S

    return obj, grad


def soft_obj_wrapper(z, T, n):
    """Objective in log-space."""
    a = softmax_from_z(z)
    obj, grad_a = smooth_obj_grad(a, T, n)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_optimize(a0, T, time_limit, maxiter=3000):
    """L-BFGS optimization in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        o, _ = soft_obj_wrapper(z, T, n)
        return o

    def jac(z):
        if stopped[0]:
            return np.zeros_like(z)
        _, g = soft_obj_wrapper(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return softmax_from_z(res.x)
    except Exception:
        return a0.copy()


def lp_bcd_one_step(a, t_blend=0.4):
    """Single LP-BCD step. Returns (new_a, M_old, M_new, success)."""
    n = len(a)
    b = fast_convolve(a)
    M = b.max()

    if M < 1e-15:
        return a.copy(), M, M, False

    A_mat = build_conv_matrix(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    res = linprog(c, A_ub=A_mat, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if not res.success or res.x is None:
        return a.copy(), M, M, False

    g = np.maximum(res.x, 0)
    g_sum = g.sum()

    if g_sum < 1e-10:
        return a.copy(), M, M, False

    # Normalize g to have same sum as a
    a_sum = a.sum()
    g = g * (a_sum / g_sum)

    # Blend
    a_new = (1.0 - t_blend) * a + t_blend * g
    a_new = np.maximum(a_new, 0.0)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-15:
        return a.copy(), M, M, False
    a_new = a_new / a_new_sum

    M_new = fast_convolve(a_new).max()
    return a_new, M, M_new, True


def lp_bcd(a0, time_limit=300.0, max_iter=5000, t_init=0.4):
    """LP-based Block Coordinate Descent with adaptive blend."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t_blend = t_init
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        a_new, M_old, M_new, ok = lp_bcd_one_step(a, t_blend)

        if ok and M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                t_blend = max(0.05, t_blend * 0.7)

        a = a_new

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-BCD it={it}, M={M_new:.8f}, best={best_M:.8f}, t={elapsed:.0f}s")

    return best_a, float(2.0 * n * best_M)


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent with line search on multipliers."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 3 * n:
        improved_total = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                no_improve += 1
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_total = True
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if not improved_total and no_improve > n:
            break

    return best_a, float(2.0 * len(best_a) * best_obj)


def peak_shave(a0, time_limit=60.0, n_peaks=8):
    """Targeted peak reduction."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peaks (exclude pure-energy lags 0 and 2n-2)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        if len(b_inner) > n_peaks:
            peak_idx = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        else:
            peak_idx = np.arange(len(b_inner))
        peak_idx = sorted(peak_idx, key=lambda k: -b[k])

        improved = False
        for k in peak_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if b[k] < best_obj * 0.98:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]

            if contributions.max() < 1e-15:
                continue

            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i

            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-12:
                    continue

                for factor in [0.7, 0.8, 0.9, 0.95]:
                    new_val = old_val * factor
                    diff = old_val - new_val
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others_sum = a[mask].sum()
                    if others_sum < 1e-15:
                        continue

                    a_trial = a.copy()
                    a_trial[target] = new_val
                    a_trial[mask] += diff * a[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss

                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-13:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            break

    return best_a, float(2.0 * n * best_obj)


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def subgradient_polish(a0, time_limit=60.0, lr=0.003, mom=0.9, top_k=50):
    """Subgradient method with softmax-weighted peaks."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.005

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[ii] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        co = fast_convolve(a).max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.998
        cur_temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_a, float(2.0 * n * best_obj)

        for target_n in [1024, 1536, 2048]:
            rem = deadline - time.monotonic() - 300
            if rem < 40:
                break

            f_interp = interpolate_sequence(base, target_n)
            print(f"  Exploring n={target_n}, remaining={rem:.0f}s")

            # LP-BCD on interpolated
            t_alloc = min(rem * 0.4, 120.0)
            a_test, _ = lp_bcd(f_interp, time_limit=t_alloc, t_init=0.3)
            try_seq(a_test, f"lp_n{target_n}")

            # Smooth refine
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                a_test2 = lbgfs_optimize(a_test, 100, min(rem * 0.5, 60.0), maxiter=1500)
                try_seq(a_test2, f"smooth_n{target_n}")

    # === Phase 4: Deep LP-BCD on best ===
    print(f"\n=== Phase 4: Deep LP-BCD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t4 = min(rem * 0.6, 300.0)
            print(f"  Deep LP-BCD n={n_best}, t={t4:.0f}s")
            a_deep, _ = lp_bcd(best_seq, time_limit=t4, t_init=0.2)
            try_seq(a_deep, f"deep_lp_n{n_best}")

            rem = deadline - time.monotonic() - 100
            if rem > 20:
                a_deep2 = lbgfs_optimize(a_deep, 300, min(rem * 0.5, 80.0), maxiter=2000)
                try_seq(a_deep2, f"deep_smooth_n{n_best}")

    # === Phase 5: Peak shaving + coordinate descent ===
    print(f"\n=== Phase 5: Peak shaving + CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_ps = min(rem * 0.3, 80.0)
            print(f"  Peak shaving: {t_ps:.0f}s")
            a_ps, _ = peak_shave(best_seq, time_limit=t_ps, n_peaks=10)
            try_seq(a_ps, "peak_shave")

        if best_seq is not None:
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                t_cd = min(rem * 0.5, 60.0)
                print(f"  Coordinate descent: {t_cd:.0f}s")
                a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(a_cd, "cd")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_sg = min(rem * 0.6, 80.0)
            n_best = len(best_seq)
            print(f"  Subgradient: {t_sg:.0f}s")
            a_sg, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                         lr=0.002, mom=0.92,
                                         top_k=max(30, n_best // 8))
            try_seq(a_sg, "subgrad")

            # Another round of CD after subgradient
            rem = deadline - time.monotonic() - 30
            if rem > 10 and best_seq is not None:
                a_cd2, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.5, 30.0))
                try_seq(a_cd2, "cd_final")

    # === Phase 7: Final micro-optimization ===
    print(f"\n=== Phase 7: Final micro-optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Very high T smooth
            t_f = min(rem * 0.4, 30.0)
            a_final = lbgfs_optimize(best_seq, 1000, t_f, maxiter=1000)
            try_seq(a_final, "final_T1000")

            # Perturbation + re-optimize
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                for attempt in range(3):
                    rem = deadline - time.monotonic() - 5
                    if rem < 5:
                        break
                    noise = rng.standard_normal(n_best) * 0.003
                    a_pert = project_simplex(best_seq + noise)
                    z_pert = z_from_a(a_pert)
                    t_alloc = min(rem * 0.4, 15.0)
                    a_p = lbgfs_optimize(a_pert, 500, t_alloc, maxiter=500)
                    try_seq(a_p, f"perturb_{attempt}")

            # Final peak shave
            rem = deadline - time.monotonic() - 5
            if rem > 8:
                a_ps_final, _ = peak_shave(best_seq, time_limit=min(rem * 0.6, 20.0), n_peaks=15)
                try_seq(a_ps_final, "peak_shave_final")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
