def fienup_step(a, N, rng):
    """One Fienup iteration: alternate non-negativity and flat spectrum."""
    # FFT
    A = np.fft.fft(a, N)
    # Flatten magnitude
    mag = np.abs(A)
    mag[mag < 1e-10] = 1e-10
    target_mag = np.sum(mag) / N  # average magnitude
    A_new = A * (target_mag / mag)
    # Ensure Hermitian symmetry
    # IFFT
    a_new = np.real(np.fft.ifft(A_new))[:n]
    # Non-negativity
    a_new = np.maximum(a_new, 0)
    # Normalize
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

Actually, one more thought: the problem is equivalent to finding a non-negative Toeplitz matrix (the autocorrelation) with minimum maximum entry, subject to the constraint that it's the autocorrelation of some non-negative sequence. This is related to the "minimum peak sidelobe" problem in radar signal design.

In radar, the goal is to minimize the peak sidelobe level (PSL) of the autocorrelation. The optimal sequences for this are called "minimum PSL sequences" and are typically found using the "up-down" algorithm of Terry (1965) or its variants.

The up-down algorithm:
1. Start with some sequence $a$
2. Compute the autocorrelation $R = a*a$
3. Find the peak sidelobe (the max of $R[k]$ for $k \neq 0$)
4. If the peak is above a threshold, "push it down" by modifying $a$
5. Repeat

For our problem, since all values are non-negative, the "main lobe" is the entire convolution (there's no separate main lobe and sidelobes). So we just want to minimize the overall max.

The up-down algorithm in our context:
1. Compute $(a*a)$
2. Find argmax lag $k^*$
3. Modify $a$ to reduce $(a*a)[k^*]$

The modification: we want to reduce $\sum_i a_i a_{k^*-i}$. One way is to reduce the product of the largest terms. For each $i$ where $a_i a_{k^*-i}$ is large, we can try to reduce either $a_i$ or $a_{k^*-i}$ (and add the removed mass elsewhere).

This is essentially what the subgradient method does: it pushes down the values that contribute most to the peak.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral Fienup iterations,
soft-max subgradient optimization, pairwise swap local search, and
multi-phase exploration with diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_size_for(n):
    """Next power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_fienup(a, n_iterations, N=None, alpha=1.0, soft_temp=0.0):
    """
    Fienup-style iterations alternating non-negativity and flat spectrum.
    
    alpha: exponent for magnitude flattening (1.0 = hard, <1.0 = soft)
    soft_temp: if > 0, use soft non-negativity (softplus)
    """
    n = len(a)
    if N is None:
        N = fft_size_for(n)
    
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_iterations):
        # FFT to frequency domain
        A = np.fft.fft(a, N)
        
        # Flatten magnitude spectrum
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        
        if alpha == 1.0:
            # Hard flatten: set all magnitudes to the mean
            target = np.mean(mag)
            A_new = A / mag_safe * target
        else:
            # Soft flatten: partially reduce dynamic range
            mean_mag = np.mean(mag_safe)
            factor = (mean_mag / mag_safe) ** alpha
            A_new = A * factor
        
        # IFFT back to time domain
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Enforce non-negativity
        if soft_temp > 0:
            a_new = soft_temp * np.log1p(np.exp(a_new / soft_temp))
        else:
            a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def spectral_init(n, N, rng):
    """Create non-negative sequence with approximately flat spectrum."""
    # Random phase, flat magnitude
    A = np.ones(N, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, N)
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        A[N - k] = np.exp(-1j * phases[k])
    A[0] = 1.0  # DC component
    
    a = np.real(np.fft.ifft(A))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def smooth_random_init(n, rng, n_components=8):
    """Smooth random init as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.03, 0.3)
        amp = rng.uniform(0.3, 2.0)
        f += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

    m = np.zeros(n)
    v = np.zeros(n)

    temp = temp_init

def pairwise_swap_search(f0, time_limit, n_swaps_per_pass=5000):
    """Local search via pairwise swaps to reduce max convolution."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Get current convolution
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        # Sample random pairs to try
        rng_local = np.random.default_rng()
        improved = False
        
        for _ in range(n_swaps_per_pass):
            if time.monotonic() - t0 > time_limit:
                break
            
            i, j = rng_local.integers(0, n, size=2)
            if i == j:
                continue
            if abs(f[i] - f[j]) < 1e-10:
                continue
            
            # Try swap
            f[i], f[j] = f[j], f[i]
            new_conv = fast_convolve(f)
            new_max = new_conv.max()
            
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                improved = True

                    best_obj = cur_max
                    best_f = f.copy()
                conv = new_conv
            else:
                # Revert
                f[i], f[j] = f[j], f[i]
        
        if not improved:

    return best_f, best_obj

When we swap f[i] and f[j], the convolution changes only at lags k where both k-i and k-j are in [0, n-1]. That's at most min(n, |i-j|) + ... hmm, actually it's at most n lags (the overlap of the two shifted support).

Actually, the change in convolution at lag k is:
$\Delta[k] = (f[j] - f[i])(f[k-i] - f[k-j])$ (with out-of-bounds = 0)

This is non-zero only when $k-i$ or $k-j$ is in [0, n-1], i.e., $k \in [i, i+n-1] \cup [j, j+n-1]$.

So at most $2n$ lags change. We can compute the new max by:
1. Precompute the current conv
2. For the affected lags, compute the new value

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral Fienup iterations
(alternating non-negativity and flat-spectrum projection), soft-max subgradient
Adam optimization, and multi-phase diverse exploration.

Key insight: The linear autocorrelation is IFFT(|FFT(a)|^2), so minimizing its
peak is equivalent to flattening the power spectrum. We use Fienup-style
alternating projections between {a >= 0, sum=1} and {|FFT(a)| = const}.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def get_fft_size(n):
    """Smallest power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_fienup(a, n_iter, alpha=1.0, relax=1.0, soft_temp=0.0,
                    N=None, conv_check_every=5):
    """
    Fienup-style alternating projections: non-negativity <-> flat magnitude spectrum.
    
    alpha: 1.0 = hard flatten, <1.0 = soft (reduces dynamic range partially)
    relax: relaxation factor (1.0 = full step, <1.0 = damped)
    soft_temp: >0 uses softplus for non-negativity
    """
    n = len(a)
    if N is None:
        N = get_fft_size(n)
    
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_iter):
        # FFT
        A = np.fft.fft(a, N)
        
        # Flatten magnitude
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-10)
        mean_mag = np.mean(mag_safe)
        
        if alpha >= 1.0:
            factor = mean_mag / mag_safe
        else:
            factor = (mean_mag / mag_safe) ** alpha
        
        A_new = A * factor
        
        # IFFT
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Non-negativity
        if soft_temp > 0:
            a_new = soft_temp * np.log1p(np.exp(a_new / soft_temp))
        else:
            a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Relaxation
        a = (1.0 - relax) * a + relax * a_new
        
        # Check objective periodically
        if (it % conv_check_every == 0) or (it == n_iter - 1):
            obj = fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a, best_obj


def spectral_init(n, N, rng, style='rand_phase'):
    """Create non-negative init with approximately flat spectrum."""
    if style == 'rand_phase':
        A = np.ones(N, dtype=complex)
        phases = rng.uniform(0, 2 * np.pi, N)
        for k in range(1, N // 2 + 1):
            A[k] = np.exp(1j * phases[k])
            A[N - k] = np.exp(-1j * phases[k])
        A[0] = 1.0
        a = np.real(np.fft.ifft(A))[:n]
    elif style == 'white_noise':
        a = rng.standard_normal(n)
    elif style == 'bump':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif style == 'triangle':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def smooth_random_init(n, rng, n_comp=6):
    """Sum of random Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_comp):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.03, 0.3)
        amp = rng.uniform(0.3, 2.0)
        f += amp * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def compute_subgradient(f, w):
    """
    Subgradient of max_k (f*f)[k] using weights w over lags.
    sg[j] = 2 * sum_k w[k] * f[k-j] (out-of-bounds = 0)
    """
    n = len(f)
    # cross-correlation: (w ⋆ f)[j] = sum_k w[k] * f[k-j]
    # np.correlate(w, f, 'full') gives this for j = -(n-1) ... (len(w)-1)
    # We want j = 0..n-1, which are at indices (n-1) .. (n-1+n-1) in full output
    full = np.correlate(w, f, 'full')
    return 2.0 * full[n - 1: n - 1 + n]


def subgradient_adam(f0, time_limit, lr=0.004, top_k=50,
                     temp_init=0.004, temp_decay=0.9995,
                     beta1=0.9, beta2=0.999, eps=1e-8,
                     restart_every=600, noise_scale=0.004):
    """Adam with soft-max subgradient for min max-convolution on simplex."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        # Soft-max weights
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        if top_k is not None and top_k < len(w):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(w))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked
        
        sg = compute_subgradient(f, w)
        
        # Adam
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_eff = lr * max(0.9997 ** t, 0.05)
        f_new = f - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-8)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > restart_every:
            rng_local = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng_local.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.6, 0.001)
            no_improve = 0
    
    return best_f, best_obj


def mirror_descent(f0, time_limit, lr_init=0.015, temp_init=0.004,
                   temp_decay=0.9995, top_k=50, lr_decay=0.999):
    """Multiplicative weights (mirror descent) with soft-max subgradient."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    temp = temp_init
    cur_lr = lr_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(w):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(w))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked
        
        sg = compute_subgradient(f, w)
        
        # Multiplicative update
        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new
        
        temp = max(temp * temp_decay, 1e-8)
        cur_lr = max(cur_lr * lr_decay, lr_init * 0.03)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng_local = np.random.default_rng(t * 23 + 11)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.5, 0.001)
            cur_lr = lr_init
            no_improve = 0
    
    return best_f, best_obj


def hybrid_spectral_subgradient(f0, time_limit, n_fienup=200,
                                fienup_alpha=0.8, fienup_relax=0.7,
                                subgrad_time=0.5, lr=0.003,
                                temp_init=0.003, top_k=30):
    """Alternate between spectral Fienup and subgradient Adam."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    N = get_fft_size(n)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Spectral Fienup phase
        f_obj = time.min(30.0, time_limit * 0.02)
        f, obj_f = spectral_fienup(f, n_iter=n_fienup, alpha=fienup_alpha,
                                    relax=fienup_relax, N=N, conv_check_every=10)
        if obj_f < best_obj:
            best_obj = obj_f
            best_f = f.copy()
        
        # Subgradient phase
        t_sub = time.min(20.0, max(5.0, (time_limit - (time.monotonic() - t0)) * 0.3))
        if time.monotonic() - t0 + t_sub > time_limit:

        f, obj_s = subgradient_adam(f, time_limit=t_sub, lr=lr,
                                     temp_init=temp_init, top_k=top_k,
                                     restart_every=200, noise_scale=0.003)
        if obj_s < best_obj:
            best_obj = obj_s
            best_f = f.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    if len(f) == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, len(f))
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 1e-15)
    return f_new / f_new.sum()

            best_seq, time_limit=t_hybrid,
            n_fienup=80, fienup_alpha=0.5, fienup_relax=0.5,
            subgrad_time=0.4, lr=0.002, temp_init=0.002,
            top_k=max(15, n_base // 30))
        try_seq(a_opt, "p3_hybrid_soft")
    
    # === Phase 4: Length exploration (300s) ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 300
    
    target_lengths = [384, 512, 640, 768, 896, 1024, 1152, 1280, 1536]
    target_lengths = [t for t in target_lengths if t != n_base and 256 <= t <= 1536]
    target_lengths.sort(key=lambda x: abs(x - n_base))
    target_lengths = target_lengths[:3]
    
    for target_n in target_lengths:
        rem = t_phase4_end - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        N_t = get_fft_size(target_n)
        t_alloc = min(rem * 0.4, 80.0)
        
        print(f"  Length n={target_n}, t={t_alloc:.0f}s")
        
        # Fienup at new length
        a_opt, obj = spectral_fienup(f_interp, n_iter=200, alpha=0.85,
                                       relax=0.8, N=N_t, conv_check_every=5)
        try_seq(a_opt, f"p4_fienup_n{target_n}")
        
        # Subgradient at new length
        rem = t_phase4_end - time.monotonic()
        if rem > 20:
            t_sub = min(rem * 0.4, 50.0)
            a_opt, _ = subgradient_adam(f_interp, time_limit=t_sub,
                                         lr=0.004, temp_init=0.004,
                                         top_k=max(25, target_n // 25))
            try_seq(a_opt, f"p4_subg_n{target_n}")
    
    # === Phase 5: Deep optimization on best (200s) ===
    print(f"\n=== Phase 5: Deep optimization (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 200
    n_best = len(best_seq)
    N_best = get_fft_size(n_best)
    
    # Fine Fienup with small steps
    rem = t_phase5_end - time.monotonic()
    if rem > 20:
        t_f = min(rem * 0.5, 60.0)
        n_iter = int(t_f * 20)  # roughly 20 iterations per second
        print(f"  Fine Fienup, iters~{n_iter}, t={t_f:.0f}s")
        a_opt, _ = spectral_fienup(best_seq, n_iter=n_iter, alpha=0.95,
                                     relax=0.3, N=N_best, conv_check_every=3)
        try_seq(a_opt, "p5_fine_fienup")
    
    # Very low temperature subgradient
    rem = t_phase5_end - time.monotonic()
    if rem > 20:
        t_s = min(rem * 0.6, 70.0)
        print(f"  Fine subgradient, t={t_s:.0f}s")
        a_opt, _ = subgradient_adam(best_seq, time_limit=t_s,
                                     lr=0.001, temp_init=0.001,
                                     temp_decay=0.9998, top_k=max(10, n_best // 40),
                                     restart_every=400, noise_scale=0.002)
        try_seq(a_opt, "p5_fine_subg")
    
    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    
    # Tiny perturbations + quick optimize
    for pi in range(5):
        rem = deadline - time.monotonic() - 5
        if rem < 10:
            break
        noise = rng.standard_normal(n_best) * 0.003
        f_pert = project_simplex(best_seq + noise)
        
        # Quick Fienup
        a_opt, _ = spectral_fienup(f_pert, n_iter=100, alpha=0.9,
                                     relax=0.5, N=get_fft_size(n_best),
                                     conv_check_every=5)
        try_seq(a_opt, f"p6_fienup_{pi}")
        
        rem = deadline - time.monotonic() - 5
        if rem < 8:
            break
        
        # Quick subgradient
        a_opt, _ = subgradient_adam(f_pert, time_limit=min(rem * 0.4, 15.0),
                                     lr=0.002, temp_init=0.002,
                                     temp_decay=0.9997,
                                     top_k=max(10, n_best // 40))
        try_seq(a_opt, f"p6_subg_{pi}")
    
    # One more spectral pass on the absolute best
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        n_iter = int(min(rem * 0.7, 60.0) * 25)
        a_opt, _ = spectral_fienup(best_seq, n_iter=n_iter, alpha=0.98,
                                     relax=0.2, N=get_fft_size(n_best),
                                     conv_check_every=3)
        try_seq(a_opt, "p6_final_fienup")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
