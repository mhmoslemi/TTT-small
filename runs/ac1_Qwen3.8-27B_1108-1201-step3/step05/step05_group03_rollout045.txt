```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def smooth_max_objective(theta, beta):
    """Objective using log-sum-exp approximation of max convolution."""
    n = len(theta)
    # Softmax parameterization ensures a >= 0 and we can normalize
    a = np.exp(theta - theta.max())
    s = a.sum()
    a = a / s  # normalize to sum = 1
    
    conv = fast_convolve(a)
    # Log-sum-exp for numerical stability
    m = conv.max()
    logsumexp = m + np.log(np.sum(np.exp(beta * (conv - m))))
    return logsumexp / beta  # This approximates max(conv)
```

Minimizes 2*n*max(auto_conv)/(sum)^2 using a multi-phase approach:
(1) Diverse initialization, (2) Smooth mirror descent with adaptive
log-sum-exp temperature, (3) L-BFGS refinement in softmax parameterization,
(4) Coordinate descent polishing with golden-section search, and
(5) Dimension exploration. Key innovation: smooth optimization phases
that avoid the slow convergence of first-order subgradient methods,
combined with adaptive temperature annealing for the log-sum-exp
approximation of the max operator.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, beta):
    """
    Compute the gradient of the log-sum-exp approximation of max(conv).
    Returns (smooth_obj, h) where h_i = sum_j p_{i+j} * a_j.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Log-sum-exp for numerical stability
    m = conv.max()
    shifted = beta * (conv - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    logsumexp = m + np.log(np.sum(np.exp(shifted)))
    obj = logsumexp / beta
    
    # Softmax weights
    p = np.exp(shifted)
    p = p / p.sum()
    
    # h_i = sum_j a_j * p_{i+j}
    # Using FFT: h = (a * p_rev) at specific indices
    # p has length 2n-1, a has length n
    # h_i = (a * p_rev)(2n-2-i) where p_rev = p[::-1]
    p_rev = p[::-1]
    conv_size = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    cross_conv = np.fft.irfft(A * P, fft_size)[:conv_size]
    h = cross_conv[2 * n - 2 - np.arange(n)]
    
    return obj, h


def mirror_descent_smooth(a_init, beta, time_limit, lr0=0.1, decay=0.9999,
                           verbose=False):
    """
    Mirror descent (multiplicative weights) with smooth log-sum-exp objective.
    """
    n = len(a)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr0
    step = 0
    best_step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        obj, h = compute_smooth_grad(a, beta)
        
        # Mirror descent step: a_new = a * exp(-lr * h), then normalize
        log_step = -lr * h
        log_step = np.clip(log_step, -50, 50)  # prevent overflow
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            best_step = step
        
        a = a_new
        lr *= decay
        
        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD step {step}: score={best_score:.10f}, "
                  f"lr={lr:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def lbfqs_smooth(a_init, beta, maxiter=300, verbose=False):
    """
    L-BFGS optimization in softmax parameter space with log-sum-exp objective.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    # Initialize theta = log(a)
    theta = np.log(np.maximum(a, 1e-30))
    
    def obj_and_grad(theta):
        # Compute a from theta via softmax
        theta_shifted = theta - theta.max()
        a_local = np.exp(theta_shifted)
        a_local = a_local / a_local.sum()
        
        # Compute smooth objective and gradient w.r.t. a
        obj, h = compute_smooth_grad(a_local, beta)
        
        # Gradient in theta space: a_j * (h_j - h·a)
        h_dot_a = np.dot(h, a_local)
        grad_theta = a_local * (h - h_dot_a)
        
        return obj, grad_theta
    
    try:
        result = minimize(obj_and_grad, theta, method='L-BFGS',
                         jac=True, options={'maxiter': maxiter, 'ftol': 1e-14,
                                            'gtol': 1e-10})
        theta_opt = result.x
        a_opt = np.exp(theta_opt - theta_opt.max())
        a_opt = a_opt / a_opt.sum()
    except Exception as e:
        if verbose:
            print(f"      L-BFGS error: {e}")
        a_opt = a_init.copy()
    
    score = compute_score(a_opt)
    return a_opt, score

def _coord_obj(a, j, v):
    """Objective for coordinate j: max of convolution when a[j]=v (others scaled)."""
    n = len(a)
    v = max(0.0, min(1.0, v))
    old_val = a[j]
    other_sum = 1.0 - old_val
    if other_sum < 1e-12:
        return 0.0
    scale = (1.0 - v) / other_sum
    new_a = a.copy()
    new_a[j] = v
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    new_a[mask] *= scale
    conv = fast_convolve(new_a)
    return conv.max()


def geometric_sequence(n, r):
    """Geometric sequence normalized."""
    a = r ** np.arange(n)
    return normalize(a)


def powerlaw_sequence(n, alpha):
    """Power-law decay: a[i] = (n-1-i+1)^(-alpha) (decay from left)."""
    idx = np.arange(n).astype(float)
    a = (n - idx) ** (-alpha)
    return normalize(a)


def exp_decay_sequence(n, rate):
    """Exponential decay from left: a[i] = exp(-rate * i)."""
    idx = np.arange(n).astype(float)
    a = np.exp(-rate * idx)
    return normalize(a)


def cosine_sequence(n):
    """Half-cosine window."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
