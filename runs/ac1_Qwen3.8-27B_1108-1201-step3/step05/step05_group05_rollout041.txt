"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
exact subgradient optimization, golden-section coordinate descent,
and simulated annealing local search with multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size].real

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_exact_subgrad(a, top_k=50):
    """Compute exact subgradient of max(conv(a,a)) using top-k peak positions."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Get the weights (softmax over peak values)
    pv = conv[peaks]
    max_pv = pv.max()
    # Use uniform weights for positions that attain (near) the max
    threshold = max_pv - 1e-10
    weights = np.where(pv >= threshold, 1.0, 0.0)
    if weights.sum() == 0:
        weights = np.zeros(len(peaks))
        weights[np.argmax(pv)] = 1.0
    else:
        weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        # d(conv[k])/d(a[j]) = a[j] + a[k-j] (when j != k-j), or 2*a[j] when j = k-j
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            kj = k - j
            if 0 <= kj < n:
                if j == kj:
                    grad[j] += weights[i] * 2.0 * a[j]
                else:
                    grad[j] += weights[i] * (a[j] + a[kj])
    
    return grad, conv

def lp_direction(f, M, max_time=10.0):
    """
    Solve the LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g or None if the LP fails.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A_ub: A_ub[k, j] = f[k-j] if 0 <= k-j < n
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': max_time})

        return None
    
    if time.monotonic() - t0 > max_time:
        return None
    
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        s = g.sum()
        if s > 1e-10:
            g /= s
        return g
    return None

def golden_section_coord_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent using golden section search per coordinate.
    For each coordinate j, find the optimal value using golden section
    on the interval [0, 1] (since sum=1 and all non-negative).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    # Golden ratio
    gr = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    
    def eval_coord(a_base, j, val_j):
        """Evaluate objective when a[j] = val_j, others scaled to maintain sum=1."""
        if val_j < -1e-12 or val_j > 1.0 + 1e-12:
            return float('inf')
        val_j = max(0.0, min(1.0, val_j))
        a_trial = a_base.copy()
        a_trial[j] = val_j
        remaining = 1.0 - val_j
        old_remaining = 1.0 - a_base[j]
        if old_remaining < 1e-12:
            if remaining < 1e-12:
                # All mass on j
                a_trial = np.zeros(n)
                a_trial[j] = 1.0
            else:
                return float('inf')
        else:
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_trial[mask] = a_base[mask] * (remaining / old_remaining)
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            return float('inf')
        a_trial /= ss
        return fast_convolve(a_trial).max()
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_a = a.copy()
            old_obj = best_obj
            
            # Golden section search on [0, 1]
            lo = 0.0
            hi = 1.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_coord(old_a, j, c)
            fd = eval_coord(old_a, j, d)
            
            max_iter = 30
            for _ in range(max_iter):
                if time.monotonic() - t0 > time_limit:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_coord(old_a, j, c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_coord(old_a, j, d)
                
                if hi - lo < 1e-8:
                    break
            
            opt_val = (lo + hi) / 2
            opt_obj = eval_coord(old_a, j, opt_val)
            
            if opt_obj < old_obj - 1e-12:
                a = old_a.copy()
                a[j] = opt_val
                remaining = 1.0 - opt_val
                old_remaining = 1.0 - old_a[j]
                if old_remaining > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = old_a[mask] * (remaining / old_remaining)
                a = np.maximum(a, 0)
                a /= a.sum()
                best_obj = opt_obj
                best_a = a.copy()
                improved = True
        
        if not improved and pass_count > 1:
            break
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, seed=42, verbose=False):
    """
    Simulated annealing with structured moves for escaping local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_obj = best_obj
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T_init = 0.005
    T_min = 0.0001
    total_iters = max(1, int((time_limit) / (n * 0.001)))  # rough estimate
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule
        progress = (time.monotonic() - t0) / time_limit
        T = T_init * (T_min / T_init) ** progress
        
        # Generate a move
        move_type = rng.integers(0, 4)
        
        if move_type == 0:
            # Single coordinate change
            j = rng.integers(0, n)
            delta = rng.standard_normal() * 0.02
            a_trial = a.copy()
            a_trial[j] = max(0, a[j] + delta)
            ss = a_trial.sum()
            if ss > 1e-10:
                a_trial /= ss
            else:
                continue
        elif move_type == 1:
            # Pair swap (transfer mass)
            indices = rng.integers(0, n, size=2)
            i, j = int(indices[0]), int(indices[1])
            if i == j:
                continue
            amount = min(a[i], a[j]) * rng.uniform(0.05, 0.3)
            direction = rng.choice([1, -1])
            a_trial = a.copy()
            a_trial[i] = max(0, a[i] + direction * amount)
            a_trial[j] = max(0, a[j] - direction * amount)
            ss = a_trial.sum()
            if ss > 1e-10:
                a_trial /= ss
            else:
                continue
        elif move_type == 2:
            # Block swap (swap values of two random blocks)
            block_size = min(rng.integers(2, 8), n // 4)
            i = rng.integers(0, n - block_size)
            j = rng.integers(0, n - block_size)
            if abs(i - j) < block_size:
                continue
            a_trial = a.copy()
            temp = a_trial[i:i+block_size].copy()
            a_trial[i:i+block_size] = a_trial[j:j+block_size]
            a_trial[j:j+block_size] = temp
            ss = a_trial.sum()
            if ss > 1e-10:
                a_trial /= ss
            else:
                continue
        else:
            # Gaussian perturbation
            noise = rng.standard_normal(n) * 0.01
            a_trial = project_simplex(a + noise, target_sum=1.0)
        
        # Evaluate
        new_obj = fast_convolve(a_trial).max()
        
        # Accept or reject
        delta = new_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            cur_obj = new_obj
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a.copy()
                if verbose:
                    print(f"      SA best: {best_obj:.10f} at iter {it}")
    
    return best_a, best_obj

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()

def deterministic_lobe_init(n, rng):
    """
    Initialize with a pattern that has low peak sidelobe.
    Uses a windowed sinc or similar structure.
    """
    # Use a raised cosine window as starting point
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    # Add some structure
    shift = rng.integers(0, n // 2)
    w = np.roll(w, shift)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
exact subgradient with adaptive step sizing, golden-section coordinate descent,
simulated annealing for escaping local minima, and multi-length exploration.
The LP direction maximizes sum(g) s.t. g>=0 and (f*g)_k <= M for active constraints,
providing principled descent directions that avoid the bias of smoothed subgradients.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size].real


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_exact_subgrad(a, conv=None, top_k=50):
    """Compute exact subgradient of max(conv(a,a)) using positions near the max."""
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    max_pv = pv.max()
    # Use weights based on proximity to max (soft uniform over near-max positions)
    temp = 0.01 * max_pv + 1e-10
    scaled = (pv - max_pv) / temp
    weights = np.exp(scaled)
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.zeros(len(peaks))
        weights[np.argmax(pv)] = 1.0
    else:
        weights /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            kj = k - j
            if 0 <= kj < n:
                if j == kj:
                    grad[j] += weights[i] * 2.0 * a[j]
                else:
                    grad[j] += weights[i] * (a[j] + a[kj])

    return grad, conv


def lp_direction_sparse(f, M, conv=None, active_set_size=100):
    """
    Solve reduced LP: max sum(g) s.t. g>=0, (f*g)_k <= M for active constraints.
    Active constraints are those where current conv is close to M.
    """
    n = len(f)
    if conv is None:
        conv = fast_convolve(f)

    # Find active constraints (positions where conv is close to M)
    threshold = M * 0.85
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.where(conv >= M - 1e-10)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    if len(active) > active_set_size:
        # Keep the most active ones
        active_vals = conv[active]
        top_idx = np.argsort(active_vals)[::-1][:active_set_size]
        active = active[top_idx]

    # Build reduced constraint matrix
    A_sub = np.zeros((len(active), n))
    for i, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_sub[i, j] = f[k - j]

    c = -np.ones(n)
    b_sub = np.full(len(active), M * (1 + 1e-9))
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds,
                         method='highs', options={'time_limit': 5.0})
    except Exception:
        return None

    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        s = g.sum()
        if s > 1e-10:
            g /= s
            # Verify full constraint
            fg = np.convolve(f, g)
            if fg.max() <= M * (1 + 1e-5):
                return g
            # Slightly violated; try with tighter bound
            b_sub_tight = np.full(len(active), M * (1 + 1e-12))
            try:
                result2 = linprog(c, A_ub=A_sub, b_ub=b_sub_tight, bounds=bounds,
                                  method='highs', options={'time_limit': 3.0})
                if result2.success and result2.x is not None:
                    g2 = np.maximum(result2.x, 0)
                    s2 = g2.sum()
                    if s2 > 1e-10:
                        g2 /= s2
                        fg2 = np.convolve(f, g2)
                        if fg2.max() <= M * (1 + 1e-5):
                            return g2
            except Exception:
                pass
        return None
    return None


def lp_direction_full(f, M, max_time=8.0):
    """Solve the full LP with all constraints."""
    n = len(f)
    num_constraints = 2 * n - 1

    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]

    c = -np.ones(n)
    b_ub = np.full(num_constraints, M * (1 + 1e-9))
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': max_time})
    except Exception:
        return None

    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        s = g.sum()
        if s > 1e-10:
            g /= s
            return g
    return None


def optimize_lp_subgrad(f0, time_limit, use_lp=True, lp_interval=3, verbose=False):
    """
    Combined LP-based and subgradient optimization.
    Uses LP direction every lp_interval steps, subgradient in between.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    step_size = 0.25
    momentum = 0.7
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem_time = time_limit - (time.monotonic() - t0)
        if rem_time < 3:
            break

        used_lp = False
        g_dir = None

        # Use LP direction periodically
        if use_lp and iter_count % lp_interval == 0 and rem_time > 15:
            conv = fast_convolve(f)
            try:
                g_dir = lp_direction_sparse(f, conv.max(), conv=conv,
                                            active_set_size=min(80, 2 * n // 3))
            except Exception:
                g_dir = None
            if g_dir is not None:
                used_lp = True

        if used_lp and g_dir is not None:
            # Line search along LP direction
            best_t = 0
            best_trial = None
            for t_try in [step_size, step_size * 0.5, step_size * 1.5, step_size * 0.75]:
                f_trial = (1 - t_try) * f + t_try * g_dir
                f_trial = np.maximum(f_trial, 0)
                fs = f_trial.sum()
                if fs < 1e-10:
                    continue
                f_trial /= fs
                obj = fast_convolve(f_trial).max()
                if obj < best_obj and (best_trial is None or obj < fast_convolve(best_trial).max()):
                    best_t = t_try
                    best_trial = f_trial

            if best_trial is not None:
                f = best_trial
                obj = fast_convolve(f).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    step_size = min(0.5, step_size * 1.05)
                else:
                    no_improve += 1
                    step_size = max(0.05, step_size * 0.9)
            else:
                no_improve += 1
        else:
            # Subgradient step
            grad, conv = compute_exact_subgrad(f, conv=conv, top_k=min(30, n // 4))
            grad_centered = grad - grad.mean()
            norm_g = np.linalg.norm(grad_centered)
            if norm_g < 1e-15:
                no_improve += 5
                conv = fast_convolve(f)
                continue

            # Polyak-like step with momentum
            alpha = step_size / (norm_g + 1e-10)
            v = momentum * v - alpha * grad_centered
            f_trial = project_simplex(f + v, target_sum=1.0)
            obj = fast_convolve(f_trial).max()
            conv = fast_convolve(f)

            if obj < best_obj - 1e-12:
                f = f_trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                # Try smaller step
                f_trial2 = project_simplex(f + 0.3 * v, target_sum=1.0)
                obj2 = fast_convolve(f_trial2).max()
                if obj2 < best_obj - 1e-12:
                    f = f_trial2
                    best_obj = obj2
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            # Conv always refreshed here
            conv = fast_convolve(f)

        # Adaptive parameters
        if no_improve > 50:
            momentum = min(0.9, momentum + 0.01)
            step_size = max(0.03, step_size * 0.9)
        elif no_improve > 20:
            step_size = max(0.05, step_size * 0.95)
        elif no_improve == 0:
            momentum = max(0.6, momentum - 0.02)

        # Restart if stuck
        if no_improve > 100:
            rng_r = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.008 + 0.008 * min((no_improve - 100) // 100, 3)
            f = project_simplex(best_f + rng_r.standard_normal(n) * noise_scale, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            step_size = 0.2
            momentum = 0.7
            conv = fast_convolve(f)

        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {iter_count}: obj={best_obj:.10f}, step={step_size:.3f}, "
                  f"no_imp={no_improve} ({elapsed:.0f}s)")

    return best_f, best_obj


def golden_section_coord_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent using golden section search per coordinate.
    For each coordinate j, optimizes a[j] over [0,1] with others rescaled.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    gr = (np.sqrt(5) - 1) / 2.0

    # Precompute for speed
    def eval_coord_base(base, j, val_j):
        """Compute max conv when a[j]=val_j, others scaled to maintain sum=1."""
        if val_j < -1e-10:
            val_j = 0.0
        if val_j > 1.0 + 1e-10:
            val_j = 1.0
        a_trial = base.copy()
        a_trial[j] = val_j
        remaining = 1.0 - val_j
        old_remaining = 1.0 - base[j]
        if old_remaining < 1e-12:
            if remaining < 1e-12:
                a_trial = np.zeros(n)
                a_trial[j] = 1.0
                return fast_convolve(a_trial).max()
            return float('inf')
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_trial[mask] = base[mask] * (remaining / old_remaining)
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            return float('inf')
        a_trial /= ss
        return fast_convolve(a_trial).max()

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Focus on coordinates with larger values first (more impact)
        order = np.argsort(-a)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-8:
                continue

            old_a = a.copy()

            # Golden section on [0, 1]
            lo = 0.0
            hi = 1.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_coord_base(old_a, j, c)
            fd = eval_coord_base(old_a, j, d)

            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_coord_base(old_a, j, c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_coord_base(old_a, j, d)
                if hi - lo < 1e-7:
                    break

            opt_val = (lo + hi) / 2.0
            opt_obj = eval_coord_base(old_a, j, opt_val)

            if opt_obj < best_obj - 1e-11:
                a = old_a.copy()
                a[j] = opt_val
                remaining = 1.0 - opt_val
                old_remaining = 1.0 - old_a[j]
                if old_remaining > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = old_a[mask] * (remaining / old_remaining)
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                best_obj = opt_obj
                best_a = a.copy()
                improved = True

        if not improved and pass_count > 1:
            break

        if verbose and pass_count % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GSCD pass {pass_count}: obj={best_obj:.10f} ({elapsed:.0f}s)")

    return best_a, best_obj


def pair_transfer_search(a0, time_limit, seed=42):
    """
    Focused pair-transfer local search.
    Transfers mass between pairs of coordinates, trying multiple fractions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Focus on non-zero coordinates
    support = np.where(a > 1e-7)[0]
    if len(support) == 0:
        support = np.arange(n)
    support = support[:min(len(support), 500)]

    fracs = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            min_val = min(a[i], a[j])
            if min_val < 1e-8:
                continue

            for frac in fracs:
                for direction in [1, -1]:
                    transfer = direction * frac * min_val
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-12:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        break
                if a is not best_a:
                    break

    return best_a, best_obj


def simulated_annealing(a0, time_limit, seed=42):
    """
    Simulated annealing with structured moves for escaping local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_obj = best_obj
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = 0.003
    T_min = 0.00005

    while time.monotonic() - t0 < time_limit:
        batch_end = min(time.monotonic() + 0.5, t0 + time_limit)
        while time.monotonic() < batch_end:
            progress = (time.monotonic() - t0) / time_limit
            T = T_init * (T_min / T_init) ** progress

            move_type = rng.integers(0, 5)

            if move_type == 0:
                # Single coordinate jitter
                j = rng.integers(0, n)
                delta = rng.standard_normal() * 0.015
                a_trial = a.copy()
                a_trial[j] = max(0.0, a[j] + delta)
                ss = a_trial.sum()
                if ss > 1e-10:
                    a_trial /= ss
                else:
                    continue
            elif move_type == 1:
                # Pair transfer
                indices = rng.integers(0, n, size=2)
                i, j = int(indices[0]), int(indices[1])
                if i == j:
                    continue
                amount = min(a[i], a[j]) * rng.uniform(0.05, 0.25)
                direction = 1 if rng.random() < 0.5 else -1
                a_trial = a.copy()
                a_trial[i] = max(0.0, a[i] + direction * amount)
                a_trial[j] = max(0.0, a[j] - direction * amount)
                ss = a_trial.sum()
                if ss > 1e-10:
                    a_trial /= ss
                else:
                    continue
            elif move_type == 2:
                # Block swap
                block = min(rng.integers(2, 6), n // 5)
                i = rng.integers(0, n - block)
                j = rng.integers(0, n - block)
                if abs(int(i) - int(j)) < block:
                    continue
                a_trial = a.copy()
                i, j = int(i), int(j)
                temp = a_trial[i:i + block].copy()
                a_trial[i:i + block] = a_trial[j:j + block]
                a_trial[j:j + block] = temp
                ss = a_trial.sum()
                if ss > 1e-10:
                    a_trial /= ss
                else:
                    continue
            elif move_type == 3:
                # Gaussian noise
                noise = rng.standard_normal(n) * 0.008
                a_trial = project_simplex(a + noise, target_sum=1.0)
            else:
                # Mirror perturbation
                a_rev = a[::-1]
                mix = rng.uniform(0.1, 0.3)
                a_trial = (1 - mix) * a + mix * a_rev
                a_trial = project_simplex(a_trial, target_sum=1.0)

            new_obj = fast_convolve(a_trial).max()
            delta = new_obj - cur_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_trial
                cur_obj = new_obj
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def windowed_init(n, rng):
    """Windowed initialization with random phase shifts."""
    x = np.arange(n) / n
    w = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    shift = rng.uniform(-0.1, 0.1)
    x_shifted = (x + shift) % 1.0
    w = 0.5 + 0.5 * np.cos(2 * np.pi * x_shifted)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-based direction finding,
    exact subgradient with momentum, golden-section coordinate descent,
    pair-transfer search, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [512, 640, 768, 1024, 1280]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths.sort(key=lambda x: abs(x - (len(parent) if parent is not None else 768)))

    inits = {}
    for n in target_lengths[:5]:
        best_init = None
        best_init_obj = float('inf')

        methods = [
            ("flat", flat_spectrum_init(n, rng)),
            ("random", random_init(n, rng)),
            ("window", windowed_init(n, rng)),
        ]
        if parent is not None:
            methods.append(("parent", interpolate_sequence(parent, n)))

        for name, init in methods:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init
            except Exception:
                pass

        if best_init is not None:
            inits[n] = best_init
            print(f"  n={n}: init obj={best_init_obj:.6f}")

    # Quick warm-start optimization on each init
    for n in list(inits.keys()):
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_warm = min(rem / max(1, len(inits) - len(list(inits.keys())[:0])), 25)
        a_warm, obj_warm = optimize_lp_subgrad(inits[n], time_limit=t_warm,
                                               use_lp=True, lp_interval=5)
        inits[n] = a_warm
        print(f"  n={n}: after warm-start obj={obj_warm:.6f}")

    # === Phase 2: Main LP+Subgrad optimization ===
    print(f"\n=== Phase 2: LP+Subgrad Optimization (best_score={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: fast_convolve(x[1]).max())

    for rank, (n_len, init) in enumerate(sorted_inits[:3]):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        time_fracs = [0.55, 0.30, 0.15]
        time_alloc = min(rem * time_fracs[rank], 500)
        print(f"  LP+SG n={n_len}, t={time_alloc:.0f}s")
        a_opt, obj = optimize_lp_subgrad(init, time_limit=time_alloc,
                                         use_lp=True, lp_interval=3,
                                         verbose=(rank == 0))
        try_seq(a_opt, f"lp_sg_n{n_len}_r{rank}")

    # === Phase 3: Golden section coordinate descent ===
    print(f"\n=== Phase 3: Golden Section CD (best_score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_cd = min(rem * 0.5, 180)
            print(f"  GSCD t={t_cd:.0f}s, n={len(best_seq)}")
            a_opt, obj = golden_section_coord_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "gscd")

    # === Phase 4: Pair transfer search ===
    print(f"\n=== Phase 4: Pair Transfer (best_score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_ps = min(rem * 0.4, 80)
            print(f"  PS t={t_ps:.0f}s")
            a_opt, obj = pair_transfer_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "pair_transfer")

    # === Phase 5: Simulated annealing + LP follow-up ===
    print(f"\n=== Phase 5: Simulated Annealing (best_score={best_score:.6f}) ===")
    if best_seq is not None:
        for sa_run in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_sa = min(rem * 0.35, 50)
            print(f"  SA run {sa_run}, t={t_sa:.0f}s")
            a_opt, obj = simulated_annealing(best_seq, time_limit=t_sa, seed=seed + sa_run * 200)
            try_seq(a_opt, f"sa_{sa_run}")

            # Follow up with LP optimization on the SA result
            rem = deadline - time.monotonic() - 50
            if rem > 15 and obj < best_score + 0.001:
                t_follow = min(rem * 0.5, 40)
                a_opt2, obj2 = optimize_lp_subgrad(a_opt, time_limit=t_follow,
                                                   use_lp=True, lp_interval=3)
                try_seq(a_opt2, f"sa_lp_follow_{sa_run}")

    # === Phase 6: Refined LP optimization from perturbed starts ===
    print(f"\n=== Phase 6: Refined LP (best_score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            t_alloc = min(rem / (5 - p), 60)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif p == 2:
                f_init = 0.75 * best_seq + 0.25 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            elif p == 3:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init, target_sum=1.0)
            else:
                # Random re-weighting
                weights = rng.exponential(1.0, n_best)
                f_init = best_seq * weights
                f_init = project_simplex(f_init, target_sum=1.0)

            a_opt, obj = optimize_lp_subgrad(f_init, time_limit=t_alloc,
                                             use_lp=True, lp_interval=2)
            try_seq(a_opt, f"lp_refine_{p}")

    # === Phase 7: Try nearby lengths ===
    print(f"\n=== Phase 7: Nearby Lengths (best_score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        nearby = []
        for delta in [32, 64, -32, -64, 128, -128]:
            candidate = n_best + delta
            if 256 <= candidate <= 2560 and candidate != n_best:
                nearby.append(candidate)
        for target_n in nearby[:3]:
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.5, 60)
            print(f"  Try n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_lp_subgrad(f_interp, time_limit=t_alloc,
                                             use_lp=True, lp_interval=3)
            try_seq(a_opt, f"len_n{target_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best_score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = golden_section_coord_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_gscd")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, obj = optimize_lp_subgrad(best_seq, time_limit=min(rem, 20),
                                             use_lp=True, lp_interval=2)
            try_seq(a_opt, "final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
