            fc = fd

```python
"""
LP/QP-driven iterative refinement with L-BFGS soft-max polish for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses linear programming to find optimal
perturbation directions, quadratic programming for L2-based smoothing,
and gradient-based methods for final refinement.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n <= 1:
        return a.astype(float).copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f):
    """
    Solve: min t subject to conv(f,b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (g, t_opt) or (None, None) on failure.
    """
    n = len(f)
    A_mat = build_conv_matrix(f)
    # Variables: b_0,...,b_{n-1}, t (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x[:n], 0)
            gs = g.sum()
            if gs < 1e-15:
                return None, None
            g = g / gs
            return g, res.x[-1]
    except Exception:
        pass
    return None, None


def solve_qp_direction(f):
    """
    Solve: min b^T Q b subject to sum(b)=1, b>=0.
    Where Q is the autocorrelation Toeplitz matrix of f.
    Returns g or None.
    """
    n = len(f)
    if n > 3000:
        return None  # Too slow for dense solve
    
    # Compute autocorrelation: r[l] = sum_m f[m]*f[m+l]
    r = np.zeros(n)
    for l in range(n):
        r[l] = np.dot(f[:n-l], f[l:])
    
    # Try closed-form (all-positive) solution
    # Q = Toeplitz(r), Q[i,j] = r[|i-j|]
    # b = Q^{-1} * 1 / (1^T * Q^{-1} * 1)
    try:
        # Build Q and solve
        # Use the fact that Q is symmetric Toeplitz
        Q = np.zeros((n, n))
        for l in range(n):
            idx = np.arange(n - l)
            Q[idx, idx + l] = r[l]
            Q[idx + l, idx] = r[l]
        
        Q_inv_1 = np.linalg.solve(Q, np.ones(n))
        denom = Q_inv_1.sum()
        if denom > 1e-15:
            b = Q_inv_1 / denom
            if np.all(b >= -1e-12):
                b = np.maximum(b, 0)
                bs = b.sum()
                if bs > 1e-15:
                    return b / bs
    except np.linalg.LinAlgError:
        pass
    
    # Fallback: use SLSQP
    Q = np.zeros((n, n))
    for l in range(n):
        idx = np.arange(n - l)
        Q[idx, idx + l] = r[l]
        Q[idx + l, idx] = r[l]
    
    def qp_obj(b):
        return 0.5 * b @ Q @ b
    
    def qp_grad(b):
        return Q @ b
    
    try:
        res = minimize(qp_obj, np.ones(n) / n, jac=qp_grad, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda b: b.sum() - 1,
                                     'jac': lambda b: np.ones(n)}],
                       bounds=[(0, None)] * n,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-15:
                return b / bs
    except Exception:
        pass
    return None


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_grad_a_fft(a, w):
    """
    Compute grad_a[j] = 4n * sum_{m=0}^{n-1} a[m] * w[j+m]
    using FFT. w has length 2n-1.
    """
    n = len(a)
    w_len = len(w)
    flip_a = a[::-1]
    size = n + w_len - 1
    fft_size = 1 << (size - 1).bit_length()
    FA = rfft(flip_a, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(FA * W, fft_size)[:size]
    c = conv_result[n-1:2*n-1]
    return 4.0 * n * c


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = compute_grad_a_fft(a, w)
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=5000):
    """L-BFGS optimization with time limit."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T, n)[0]
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]
    
    try:
        res = minimize(func, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': maxiter * 25})
        return res.x
    except Exception:
        return z0
