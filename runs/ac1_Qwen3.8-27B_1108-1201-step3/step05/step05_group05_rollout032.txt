```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
multi-start initialization, and adaptive line search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_cross(a, d):
    """Compute cross-convolution a * d (linear)."""
    return fast_convolve(a)  # No, this is a*a
    # Actually cross(a,d) = convolve(a, d)
    n = len(a)
    m = len(d)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    D = np.fft.rfft(d, fft_size)
    return np.fft.irfft(A * D, fft_size)[:size]

def lp_direction(a, M, conv_a):
    """
    Solve LP to find direction d that minimizes the peak of the linearized convolution.
    
    Minimize epsilon
    Subject to:
      2*cross(a,d)[k] <= M - conv_a[k] + epsilon, for all k
      d[i] >= -a[i], for all i
      sum(d) = 0
      epsilon >= 0
    """
    n = len(a)
    # Variables: d[0], ..., d[n-1], epsilon
    # Total: n + 1
    
    # c = [0, 0, ..., 0, 1] (minimize epsilon)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints (A_ub @ x <= b_ub):
    # 1) 2*cross(a,d)[k] - epsilon <= M - conv_a[k] for k = 0, ..., 2n-2
    #    cross(a,d)[k] = sum_i a[i]*d[k-i] (valid indices)
    #    So: 2*sum_i a[i]*d[k-i] - epsilon <= M - conv_a[k]
    #    
    # 2) -d[i] <= a[i] for all i (i.e., d[i] >= -a[i])
    # 3) -epsilon <= 0 (epsilon >= 0)
    
    num_conv = 2 * n - 1
    num_ineq = num_conv + n + 1
    
    A_ub = np.zeros((num_ineq, n + 1))
    b_ub = np.zeros(num_ineq)
    
    # Convolution constraints
    for k in range(num_conv):
        # cross(a,d)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*d[k-i]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] += 2.0 * a[i]
        A_ub[k, n] = -1.0  # -epsilon
        b_ub[k] = M - conv_a[k]
    
    # Non-negativity: -d[i] <= a[i]
    for i in range(n):
        A_ub[num_conv + i, i] = -1.0
        b_ub[num_conv + i] = a[i]
    
    # epsilon >= 0: -epsilon <= 0
    A_ub[-1, n] = -1.0
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
diverse initializations, adaptive multi-length search, and subgradient fallback.
The key innovation is solving a linear program at each iteration to find the
exact optimal direction for reducing the convolution peak, rather than using
approximate subgradient directions.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_lp_and_solve(a, conv_a):
    """
    Solve the LP:
      Minimize delta
      Subject to:
        conv_a[k] + 2*cross(a,d)[k] <= delta  for all k
        d[i] >= -a[i]  for all i
        sum(d) = 0
    
    Returns direction d and predicted new peak delta.
    """
    n = len(a)
    num_conv = 2 * n - 1
    M = conv_a.max()
    
    # Variables: d[0..n-1], delta -> n+1 total
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize delta
    
    # Build A_ub, b_ub
    # Constraints:
    # 1) 2*cross(a,d)[k] - delta <= M - conv_a[k]  for k=0..2n-2  [num_conv rows]
    # 2) -d[i] <= a[i]  for i=0..n-1  [n rows]
    num_ineq = num_conv + n
    A_ub = np.zeros((num_ineq, n + 1))
    b_ub = np.zeros(num_ineq)
    
    # Convolution constraints (vectorized row by row)
    for k in range(num_conv):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_ub[k, i_start:i_end + 1] = 2.0 * a[i_start:i_end + 1]
        A_ub[k, n] = -1.0
        b_ub[k] = M - conv_a[k]
    
    # Non-negativity: -d[i] <= a[i]
    A_ub[num_conv:num_conv + n, 0:n] = -np.eye(n)
    b_ub[num_conv:num_conv + n] = a
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Solve

        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=[(None, None)] * n + [(None, None)],
                         method='highs',
                         options={'presolve': True, 'time_limit': 5.0})
        if result.success:
            d = result.x[:n]
            delta = result.x[n]
            return d, delta
    except Exception:

    return None, None

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    use_subgrad_fallback = False
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Try LP first
        d = None
        if not use_subgrad_fallback:
            d, delta = build_lp_and_solve(a, conv)
            if d is None or (time.monotonic() - t0) > time_limit:
                use_subgrad_fallback = True
        
        if d is None or use_subgrad_fallback:
            # Subgradient fallback
            peaks = np.where(conv >= M * 0.95)[0]
            if len(peaks) == 0:
                peaks = np.array([np.argmax(conv)])
            grad = np.zeros(n)
            for k in peaks[:20]:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += 2.0 * a[k - idx]
            grad = grad - (a * grad).sum() * 0  # keep as is
            # Project gradient to tangent of simplex
            grad = grad - grad.mean()
            lr = 0.05 / max(1, np.sqrt(n))
            d = -lr * grad
            d = d - d.mean()  # ensure sum(d)=0
        
        # Line search: try different step sizes
        best_step_obj = best_obj
        best_step = 0
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = a + alpha * d
            if a_trial.min() < -1e-10:
                a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_step_obj:
                best_step_obj = obj_trial
                best_step = alpha
        
        if best_step == 0:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart
                rng_r = np.random.default_rng(iter_count * 17 + 3)
                noise = rng_r.standard_normal(n) * 0.01
                a = project_simplex(best_a + noise, target_sum=1.0)
                no_improve = 0
                if time.monotonic() - t0 > time_limit:
                    break
                continue
        else:
            a = a + best_step * d
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            else:
                a = best_a.copy()
        
        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: peak={best_obj:.10f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_a, best_obj


def subgrad_optimize(a0, time_limit, verbose=False):
    """Subgradient optimization with momentum (backup method)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    tau = 100.0
    lr = 0.03
    momentum = 0.85
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        if no_improve > 300:
            lr = max(0.002, lr * 0.95)
            momentum = min(0.96, momentum + 0.003)
        elif no_improve > 100:
            tau = min(500.0, tau * 1.02)
        
        conv = fast_convolve(a)
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]
        
        grad = grad - grad.mean()
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()
        a = a_new
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 13 + 5)
            noise = rng_r.standard_normal(n) * (0.003 + 0.003 * min(no_improve // 500, 5))
            a = project_simplex(best_a + noise, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.02
            tau = 80.0
            momentum = 0.85
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Fast coordinate descent polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-10:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            old_remaining = 1.0 - old_val
            if old_remaining < 1e-10:
                continue
            
            for mult in [0.5, 0.7, 0.85, 1.2, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / old_remaining
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def diverse_init(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random exponential
    a = rng.exponential(1.0, n)
    inits.append(a / a.sum())
    
    # Gaussian bump
    x = np.linspace(0, 1, n)
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            inits.append(a / a.sum())
    
    # Asymmetric: high on one side
    a = np.ones(n)
    half = n // 2
    a[:half] = 2.0
    a[half:] = 0.5
    inits.append(a / a.sum())
    
    # Cosine-modulated (flat spectrum)
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    for k in range(1, n):
        A[k] = np.exp(1j * phases[k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-10:
        inits.append(a / ss)
    
    # Two peaks (asymmetric)
    c1, c2 = 0.25, 0.75
    w = 0.15
    a = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + 0.7 * np.exp(-((x - c2) ** 2) / (2 * w ** 2))
    inits.append(a / a.sum())
    
    # Linear ramp (maximum asymmetry)
    a = np.linspace(0.01, 1.0, n)
    inits.append(a / a.sum())
    
    # Interpolated parent
    if parent is not None:
        old_n = len(parent)
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, parent)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-10:
            inits.append(a / ss)
    
    return inits


def propose_candidate(seed=42):
    """
    Multi-phase optimization: LP-based direction finding with diverse starts,
    subgradient fallback, coordinate descent polish, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    budget = 1770.0
    deadline = time.monotonic() + budget
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.size == 0 or a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # Determine lengths to try
    lengths = set()
    if parent is not None:
        lengths.add(len(parent))
    for n_try in [512, 640, 768, 896, 1024, 1152, 1280, 1536]:
        lengths.add(n_try)
    lengths = sorted(lengths)
    
    # === Phase 1: Quick diverse initialization evaluation ===
    print(f"\n=== Phase 1: Initialization (best={best_score:.6f}) ===")
    t_phase1 = 30.0
    time_left = min(t_phase1, deadline - time.monotonic())
    
    candidates = []
    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        inits = diverse_init(n, rng, parent)
        for idx, init in enumerate(inits):
            init = np.maximum(init, 0)
            ss = init.sum()
            if ss < 1e-10:
                continue
            init = init / ss
            obj = fast_convolve(init).max()
            candidates.append((obj, n, init))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates generated, best init obj={candidates[0][0]:.6f}")
    
    # === Phase 2: LP optimization on top candidates ===
    print(f"\n=== Phase 2: LP Optimization (best={best_score:.6f}) ===")
    
    # Allocate time: top candidates get more time
    n_top = min(6, len(candidates))
    time_per_lp = 180.0
    used_lengths = set()
    
    for rank in range(n_top):
        obj_init, n_len, init = candidates[rank]
        
        # Avoid repeating same length too many times
        if used_lengths.count(n_len) >= 2 if hasattr(used_lengths, 'count') else len([x for x in used_lengths if x == n_len]) >= 2:
            continue
        used_lengths.add(n_len)
        
        rem = deadline - time.monotonic() - 200
        t_alloc = min(time_per_lp, rem)
        if t_alloc < 20:
            break
        
        print(f"  LP opt [{rank}]: n={n_len}, init_obj={obj_init:.8f}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(init, time_limit=t_alloc, verbose=(rank < 2))
        try_seq(a_opt, f"lp_n{n_len}_r{rank}")
        
        # Quick subgradient refinement after LP
        rem = deadline - time.monotonic() - 150
        if rem > 30 and best_seq is not None and len(best_seq) == n_len:
            t_sg = min(40, rem)
            a_opt2, obj2 = subgrad_optimize(a_opt, time_limit=t_sg)
            try_seq(a_opt2, f"sg_n{n_len}_r{rank}")
    
    # === Phase 3: Subgradient on best result ===
    print(f"\n=== Phase 3: Subgradient Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_sg = min(200, rem)
            print(f"  SG refine n={n_best}, t={t_sg:.0f}s")
            a_opt, obj = subgrad_optimize(best_seq, time_limit=t_sg, verbose=True)
            try_seq(a_opt, "sg_refine")
    
    # === Phase 4: Multi-length LP exploration ===
    print(f"\n=== Phase 4: Multi-length LP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try slightly longer/shorter
        for n_try in [n_best + 128, n_best + 256, n_best - 128, n_best + 512]:
            if n_try < 256 or n_try > 2048:
                continue
            if n_try == n_best:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            # Interpolate best to new length
            old_x = np.linspace(0, 1, n_best)
            new_x = np.linspace(0, 1, n_try)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            ss = a_interp.sum()
            if ss > 1e-10:
                a_interp /= ss
            
            t_alloc = min(120, rem)
            print(f"  LP multi-length: n={n_try}, t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"lp_ml_n{n_try}")
    
    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd = min(120, rem)
            print(f"  CD polish t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")
    
    # === Phase 6: Final perturbations ===
    print(f"\n=== Phase 6: Final Perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for trial in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise_scale = 0.002 + 0.002 * trial
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_seq + noise, target_sum=1.0)
            t_alloc = min(30, rem)
            a_opt, obj = lp_optimize(a_pert, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{trial}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
