"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using KKT-based LP iteration.
Key insight: at a KKT point, the active set of max constraints defines
a linear system A@a = c*1. We solve an LP to find a that makes A@a
as uniform as possible, then verify and polish with coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_kkt_step(a, tol=1e-6):
    """
    One KKT-LP step: compute active set, solve LP, return new a.
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    
    # Active set: indices where b[k] is close to max
    K_mask = b >= M - tol
    K = np.where(K_mask)[0]
    
    if len(K) == 2 * n - 1:
        return a, M, K  # Can't improve with full active set
    
    # Form A[j,m] = 1 if j+m in K
    # A is n x n
    # We need: for each j, sum_m A[j,m] * a[m] >= t
    # A[j,m] = 1 if (j+m) is in K
    
    # Efficient construction:
    # For each k in K, A[j, k-j] = 1 for valid j
    # i.e., for k in K: for j in max(0, k-n+1) to min(n-1, k): A[j, k-j] = 1
    
    # Build A as sparse or dense
    # For n=768, dense is 768*768 = 590K entries, fine
    
    # Actually, let's use a more efficient approach:
    # (A@a)[j] = sum_{m: j+m in K} a[m] = sum_{k in K: 0<=k-j<=n-1} a[k-j]
    #          = sum_{k in K: j <= k <= j+n-1} a[k-j]
    
    # This is a sliding window sum of a, restricted to positions in K.
    # Let c[m] = a[m] if we think of it as... hmm.
    
    # Actually, (A@a)[j] = sum_{m=0}^{n-1} a[m] * 1_{j+m in K}
    # = sum_{m=0}^{n-1} a[m] * K_indicator[j+m]
    
    # This is the cross-correlation of a with K_indicator, restricted to valid j.
    # We can compute it via FFT!
    
    # But for the LP, we need the full A matrix (or at least Ab for the constraints).
    # Let's just build A explicitly.
    
    K_ind = np.zeros(2*n-1, dtype=np.int32)
    K_ind[K] = 1
    
    # A[j, m] = K_ind[j+m] for j=0..n-1, m=0..n-1
    # This is a Toeplitz-like matrix
    js = np.arange(n)[:, None]
    ms = np.arange(n)[None, :]
    A_mat = K_ind[js + ms].astype(np.float64)
    
    # LP: maximize t
    # subject to: (A@a)[j] >= t for all j  =>  -A@a + t <= 0
    #             sum(a) = 1
    #             a >= 0
    #             t free
    #
    # Variables: x = [a_0, ..., a_{n-1}, t]
    # Minimize: -t => c = [0, ..., 0, -1]
    # Inequalities: for each j: -A[j,:] @ a + t <= 0
    #   => [-A[j,:], 1] @ x <= 0
    # Equality: sum(a) = 1 => [1, ..., 1, 0] @ x = 1
    # Bounds: a_j >= 0, t free
    
    c = np.zeros(n + 1)
    c[-1] = -1.0  # maximize t => minimize -t
    
    A_ub = np.zeros((n, n + 1))
    A_ub[:, :n] = -A_mat
    A_ub[:, n] = 1.0
    b_ub = np.zeros(n)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return a, M, K
    
    a_new = res.x[:n]
    t_val = res.x[n]
    a_new = np.maximum(a_new, 0)
    
    return a_new, M, K


def lp_iteration(a, max_iters=100, time_limit=60):
    """
    Iterate the LP-KKT step until convergence or time limit.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_lp, M, K = lp_kkt_step(a, tol=1e-6)
        
        if a_lp is a:
            break
        
        # Evaluate the LP solution
        a_lp = normalize(a_lp)
        sc = compute_score(a_lp)
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_lp.copy()
            a = a_lp
        else:
            # Try interpolation
            for alpha in [0.5, 0.3, 0.7]:
                a_mix = (1 - alpha) * best_a + alpha * a_lp
                a_mix = normalize(a_mix)
                sc_mix = compute_score(a_mix)
                if sc_mix < best_score - 1e-12:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break
            else:
                # No improvement, break
                break
    
    return best_a, best_score


def coord_descent_fast(a, time_limit=30, n_passes=50):
    """Fast coordinate descent with exact 1D optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for _ in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            def obj(v):
                v = max(0.0, v)
                new_a = a.copy()
                new_a[j] = v
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return 1e10
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_a)
                return conv.max()
            
            try:
                res = minimize_scalar(obj, bounds=(max(0, old_val * 0.5), min(0.99, old_val * 2 + 0.01)),
                                      method='bounded', options={'xatol': 1e-8, 'maxiter': 20})
                if res.success and res.fun < obj(old_val) - 1e-13:
                    v_new = max(0.0, res.x)
                    new_a = a.copy()
                    new_a[j] = v_new
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved:
            break
    
    return best_a, best_score


def gaussian_seq(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    return normalize(a)


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x)**2
    return normalize(a)


def geometric_seq(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using KKT-based LP iteration with
active set management, combined with precise coordinate descent polishing.
Key insight: at KKT points, the active set K of max-convolution constraints
defines a linear system A@a = c*1 where A[j,m]=1_{j+m in K}. We solve an LP
to make (A@a) as uniform as possible, then verify and polish.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def solve_lp_kkt(a, k_indices):
    """
    Solve LP: minimize range of (A@a) subject to a>=0, sum(a)=1.
    A[j,m] = 1 if j+m in k_indices.
    Returns the optimal a.
    """
    n = len(a)
    k_ind = np.zeros(2 * n - 1, dtype=np.int32)
    k_ind[k_indices] = 1

    js = np.arange(n)[:, None]
    ms = np.arange(n)[None, :]
    A_mat = k_ind[js + ms].astype(np.float64)

    # LP: minimize s - t
    # s.t. (A@a)_j <= s  for all j
    #      (A@a)_j >= t  for all j
    #      sum(a) = 1
    #      a >= 0, s free, t free
    #
    # Variables: [a_0,...,a_{n-1}, s, t]
    c = np.zeros(n + 2)
    c[n] = 1.0    # minimize s
    c[n + 1] = -1.0  # minus t

    # Inequalities: A@a - s <= 0  =>  [A, -1, 0] @ x <= 0
    #               -A@a + t <= 0  =>  [-A, 0, 1] @ x <= 0
    A_ub = np.zeros((2 * n, n + 2))
    A_ub[:n, :n] = A_mat
    A_ub[:n, n] = -1.0
    A_ub[n:, :n] = -A_mat
    A_ub[n:, n + 1] = 1.0
    b_ub = np.zeros(2 * n)

    A_eq = np.zeros((1, n + 2))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None), (None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            a_new = np.maximum(res.x[:n], 0.0)
            if a_new.sum() > 1e-10:
                return a_new / a_new.sum()
    except Exception:
        pass
    return None


def lp_kkt_iteration(a, max_iters=200, time_limit=120, verbose=False):
    """
    Iterate: compute active set -> solve LP -> check improvement.
    Uses hysteresis for active set updates.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Initial active set
    b = fast_convolve(a)
    M = b.max()
    tol_high = 0.02 * M
    tol_low = 0.08 * M
    K = np.where(b >= M - tol_high)[0]

    stale_count = 0
    last_K_size = len(K)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP with current K
        a_lp = solve_lp_kkt(a, K)
        if a_lp is None:
            break

        # Evaluate LP solution
        sc_lp = compute_score(a_lp)

        if sc_lp < best_score - 1e-12:
            best_score = sc_lp
            best_a = a_lp.copy()
            a = a_lp
            stale_count = 0
        else:
            # Try interpolation
            improved = False
            for alpha in [0.5, 0.7, 0.3]:
                a_mix = (1 - alpha) * best_a + alpha * a_lp
                a_mix = normalize(a_mix)
                sc_mix = compute_score(a_mix)
                if sc_mix < best_score - 1e-12:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved = True
                    stale_count = 0
                    break
            if not improved:
                stale_count += 1

        # Update active set with hysteresis
        b = fast_convolve(a)
        M = b.max()
        # Add: elements above M - tol_high
        add_mask = b >= M - tol_high
        # Remove: elements below M - tol_low
        remove_mask = b < M - tol_low
        K_new = np.where(add_mask)[0]
        # Keep elements that are neither to add nor remove
        keep_mask = np.zeros(2 * n - 1, dtype=bool)
        keep_mask[K] = True
        keep_mask[add_mask] = False
        keep_mask[remove_mask] = False
        K_keep = np.where(keep_mask)[0]
        K = np.union1d(K_new, K_keep)

        # Ensure K is not too small or too large
        if len(K) < 3:
            K = np.where(b >= M - 0.1 * M)[0]
        if len(K) > 2 * n - 3:
            # Keep only top elements
            k_sorted = b.argsort()[::-1]
            K = k_sorted[:max(5, len(k_sorted) // 2)]

        if len(K) == last_K_size and stale_count > 5:
            # Perturb: try slightly different tolerance
            tol_high *= 1.5
            stale_count = 0
        last_K_size = len(K)

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}, |K|={len(K)}")

        if stale_count > 15:
            break

    return best_a, best_score


def coord_descent(a, time_limit=60, n_passes=100):
    """Precise coordinate descent with bounded scalar minimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False

        if p > 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]

            # Define 1D objective
            def obj_1d(v):
                v = float(max(0.0, v))
                new_a = a.copy()
                new_a[j] = v
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return 1e10
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_a)
                return conv.max()

            try:
                lo = max(0.0, old_val * 0.3)
                hi = min(0.99, old_val * 3.0 + 0.01)
                if hi - lo < 1e-6:
                    lo = max(0.0, old_val - 0.05)
                    hi = min(0.99, old_val + 0.05)

                res = minimize_scalar(obj_1d, bounds=(lo, hi),
                                      method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 25})
                if res.success and res.fun < obj_1d(old_val) - 1e-14:
                    v_new = max(0.0, res.x)
                    new_a = a.copy()
                    new_a[j] = v_new
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-14:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if verbose := False:
            pass

    return best_a, best_score


def subgradient_step(a, steps=2000, lr0=0.005, decay=0.9998, time_limit=30):
    """Entropy mirror descent with subgradient of max-conv."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    cur_lr = lr0
    t0 = time.monotonic()

    for step in range(steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Subgradient: for the argmax, direction is 2*a reversed at that lag
        k_max = int(np.argmax(b))
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        sg = np.zeros(n)
        idx = np.arange(js, je + 1)
        sg[idx] = 2.0 * a[k_max - idx]

        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        cur_lr *= decay

    return best_a, best_score


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    return normalize(np.sin(x) ** 2)


def gaussian_seq(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    return normalize(np.exp(-x ** 2 / (2 * sigma ** 2)))


def geometric_seq(n, r):
    return normalize(r ** np.arange(n))


def raised_cosine_seq(n):
    x = np.linspace(0, 2 * np.pi, n)
    return normalize(0.5 + 0.5 * np.cos(x))


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, f))

    print(f"\n=== Phase 1: LP-KKT iteration ===")

    selected = []
    seen = set()
    for sc, a, label in start_scores:
        if len(selected) >= 6:
            break
        n_val = label.rsplit('_', 1)[-1] if '_' in label else label
        base_type = label.rsplit('_', 1)[0] if '_' in label else label
        key = base_type[:4] + n_val
        if key not in seen:
            selected.append((a, label))
            seen.add(key)

    # Ensure parent included
    if not any(l == "parent" for _, l in selected):
        for sc, a, label in start_scores:
            if label == "parent" and sc < best_score * 1.02:
                selected.append((a, label))
                break

    n_sel = len(selected)
    t_per_lp = min((deadline - time.monotonic() - 400) / max(1, n_sel), 200)
    t_per_cd = 40

    for idx, (a_init, label) in enumerate(selected):
        rem = deadline - time.monotonic() - 250
        if rem < 60:
            break
        t_lp = min(t_per_lp, rem * 0.55)
        t_cd = min(t_per_cd, rem * 0.25)
        print(f"  [{label}] LP(t={t_lp:.0f}s) + CD(t={t_cd:.0f}s)")

        a_lp, sc_lp = lp_kkt_iteration(a_init, max_iters=300, time_limit=t_lp,
                                       verbose=(idx < 2))
        try_seq(a_lp, label + "_lp")

        rem = deadline - time.monotonic() - 200
        if rem > 20:
            a_cd, sc_cd = coord_descent(a_lp, time_limit=min(t_cd, rem))
            try_seq(a_cd, label + "_cd")

            # Subgradient polish
            rem = deadline - time.monotonic() - 150
            if rem > 15 and sc_cd < best_score + 1e-8:
                a_sg, sc_sg = subgradient_step(a_cd, steps=3000,
                                               lr0=0.003, time_limit=min(20, rem))
                try_seq(a_sg, label + "_sg")
                a_final, sc_f = coord_descent(a_sg, time_limit=15)
                try_seq(a_final, label + "_fin")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        targets = [384, 512, 640, 768, 896, 1024, 1280]
        targets = [t for t in targets if t != best_n]
        for target_n in targets:
            rem = deadline - time.monotonic() - 120
            if rem < 50:
                break
            a_interp = interpolate_seq(best_a, target_n)
            t_alloc = min(rem * 0.5, 100)
            a_lp, sc = lp_kkt_iteration(a_interp, max_iters=150, time_limit=t_alloc)
            if sc < best_score:
                try_seq(a_lp, f"dim{target_n}_lp")
                a_cd, sc_cd = coord_descent(a_lp, time_limit=25)
                try_seq(a_cd, f"dim{target_n}_cd")

    # === Phase 3: Aggressive final polish ===
    print(f"\n=== Phase 3: Final polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            # Final CD with more passes
            t_cd = min(rem * 0.5, 120)
            a_f, sc_f = coord_descent(best_a, time_limit=t_cd, n_passes=200)
            try_seq(a_f, "final_cd")

            rem = deadline - time.monotonic() - 25
            if rem > 20:
                # One more LP round on the polished result
                a_lp2, sc_lp2 = lp_kkt_iteration(best_a, max_iters=100,
                                                  time_limit=min(60, rem))
                try_seq(a_lp2, "lp_round2")
                a_f2, sc_f2 = coord_descent(a_lp2, time_limit=20)
                try_seq(a_f2, "final2")

            rem = deadline - time.monotonic() - 15
            if rem > 10:
                # Tiny perturbations
                for trial in range(3):
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    noise = rng.standard_normal(len(best_a)) * 0.003
                    a_pert = normalize(best_a + noise)
                    a_p, sc_p = coord_descent(a_pert, time_limit=min(20, rem))
                    try_seq(a_p, f"pert{trial}")

    # === Final verification ===
    if best_a is None:
        best_a = cosine_seq(768)
        best_score = compute_score(best_a)

    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
