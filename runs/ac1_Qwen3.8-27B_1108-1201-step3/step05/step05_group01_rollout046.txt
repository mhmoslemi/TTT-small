import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def lse_grad(a, beta):
    """Compute log-sum-exp objective and gradient using FFT-based cross-correlation."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # Gradient via cross-correlation: grad_a[i] = 2 * sum_m w[m] * a[m-i]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    return obj, grad_a, w


def nesterov_lse(f0, time_limit, verbose=True):
    """Nesterov-accelerated gradient on log-sum-exp with annealed beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    # Anneal beta from low to high
    betas = [3, 6, 12, 24, 48, 96, 192, 384, 768]
    total_beta_rounds = len(betas)

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        sub_time = remaining / (total_beta_rounds - bi)

        # Nesterov state
        x = f.copy()
        y = f.copy()
        x_prev = f.copy()
        t_k = 1.0
        eta = 1.0 / (beta * 0.5)  # conservative step size
        it = 0
        no_improve = 0

        while time.monotonic() - t0 < t0 + sub_time:
            it += 1

            # Compute gradient at y
            obj_y, grad_y, _ = lse_grad(y, beta)

            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new

            # Gradient step with projection
            r = y - eta * grad_y
            x_new = project_simplex(r)

            # Nesterov extrapolation
            y_new = x_new + momentum * (x_new - x)
            y_new = project_simplex(y_new)

            # Evaluate actual objective
            conv_check = fast_convolve(y_new)
            obj_check = conv_check.max()

            if obj_check < best_obj - 1e-14:
                best_obj = obj_check
                best_f = y_new.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"      Nesterov b={beta} it={it}: {best_obj:.10f}")
            else:
                no_improve += 1

            x_prev = x
            x = x_new
            y = y_new
            t_k = t_new

            # Adaptive step size: decrease if not improving
            if no_improve > 100:
                eta *= 0.8
                if no_improve > 500:
                    # Perturb and restart
                    rng = np.random.default_rng(bi * 1000 + it)
                    f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    x = f_pert.copy()
                    y = f_pert.copy()
                    x_prev = f_pert.copy()
                    t_k = 1.0
                    eta = 1.0 / (beta * 0.5)
                    no_improve = 0
                    break

            # Decrease step size gradually
            eta = max(eta * 0.9995, 1e-6)

        f = best_f.copy()

    # Final exact evaluation
    final_obj = fast_convolve(f).max()
    if final_obj < best_obj:
        best_obj = final_obj
        best_f = f.copy()

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res
    return None, res


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M, line search, and dual-informed restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, lp_res = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 20.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.1, 8.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.04, 0.08, 0.12, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)

        if no_improve > 60:
            # Dual-informed restart: use LP dual to identify tight constraints
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.002 + 0.004 * min(no_improve // 60, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def _smooth_obj_grad_impl(t, beta):
    """Compute smooth LSE objective and gradient in t-space for L-BFGS."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def lbfgs_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth LSE with annealed beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    betas = [4, 8, 16, 32, 64, 128, 256, 512]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 25:
            break

        try:
            res = minimize(
                _smooth_obj_grad_impl, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-10, 'maxls': 50,
                         'maxfun': 3000}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"      L-BFGS b={beta}: {best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def coordinate_descent_fast(f0, time_limit):
    """Fast coordinate descent with single best multiplier per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            for mult in [0.6, 0.8, 1.25, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.005, mom=0.93, top_k=100, temp=0.0001):
    """Subgradient polishing with top-k peak weighting and adaptive temp."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.001)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_sequence(n, shape='triangular'):
    """Generate a structured initial sequence."""
    if shape == 'triangular':
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - x * x)
    elif shape == 'gaussian':
        x = np.linspace(-3, 3, n)
        a = np.exp(-x * x / 2)
    elif shape == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif shape == 'quadratic':
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x)) ** 2
    else:
        a = np.ones(n) / n
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick length sweep + structured starts (120s)
    Phase 2: Nesterov LSE optimization (500s)
    Phase 3: LP-FW deep optimization (350s)
    Phase 4: L-BFGS polishing (200s)
    Phase 5: Final micro-optimization (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0

    parent_n = len(parent)

    # === Phase 1: Quick exploration (120s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    p1_budget = min(120, deadline - time.monotonic() - 1100)

    # Try nearby lengths via interpolation
    nearby_lengths = []
    for delta in range(-50, 51, 10):
        ln = parent_n + delta
        if 32 <= ln <= 4096:
            nearby_lengths.append(ln)

    # Also try some structured sequences
    structured_starts = {
        512: structured_sequence(512, 'triangular'),
        768: structured_sequence(768, 'gaussian'),
        1024: structured_sequence(1024, 'cosine'),
        640: structured_sequence(640, 'quadratic'),
    }

    # Quick Nesterov on interpolated parent at nearby lengths
    for ln in nearby_lengths[:8]:
        rem = deadline - time.monotonic() - 1080
        if rem < 15:
            break
        f_interp = interpolate_sequence(parent, ln)
        t_alloc = min(8, rem * 0.3)
        a_opt, _ = nesterov_lse(f_interp, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"interp_{ln}")

    # Try structured starts
    for ln, f_struct in structured_starts.items():
        rem = deadline - time.monotonic() - 1070
        if rem < 10:
            break
        t_alloc = min(10, rem * 0.2)
        a_opt, _ = nesterov_lse(f_struct, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"struct_{ln}")

    # Select best base for main optimization
    base = best_seq if best_seq is not None else parent

    # === Phase 2: Nesterov LSE main optimization (500s) ===
    print(f"\n=== Phase 2: Nesterov LSE (best={best_score:.6f}) ===")
    p2_budget = min(500, deadline - time.monotonic() - 700)
    if p2_budget > 30:
        a_opt, _ = nesterov_lse(base, time_limit=p2_budget * 0.7, verbose=True)
        try_seq(a_opt, "nesterov_main")

        # Nesterov from perturbed start
        rem = deadline - time.monotonic() - 670
        if rem > 20:
            noise = rng.standard_normal(len(base)) * 0.02
            f_pert = project_simplex(base + noise)
            a_opt, _ = nesterov_lse(f_pert, time_limit=min(rem, p2_budget * 0.3), verbose=False)
            try_seq(a_opt, "nesterov_pert")

    # === Phase 3: LP-FW deep optimization (350s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    p3_budget = min(350, deadline - time.monotonic() - 350)
    if p3_budget > 30:
        a_opt, _ = lpfw_optimize(best_seq, time_limit=p3_budget * 0.6, verbose=True)
        try_seq(a_opt, "lpfw_deep")

        # LP-FW with higher M expansion (more aggressive)
        rem = deadline - time.monotonic() - 320
        if rem > 20:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, p3_budget * 0.4), verbose=False)
            try_seq(a_opt, "lpfw_deep2")

    # === Phase 4: L-BFGS polishing (200s) ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.6f}) ===")
    p4_budget = min(200, deadline - time.monotonic() - 150)
    if p4_budget > 20:
        a_opt, _ = lbfgs_optimize(best_seq, time_limit=p4_budget * 0.6, verbose=True)
        try_seq(a_opt, "lbfgs_main")

        # L-BFGS from slightly perturbed
        rem = deadline - time.monotonic() - 130
        if rem > 15:
            noise = rng.standard_normal(len(best_seq)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lbfgs_optimize(f_pert, time_limit=min(rem, 60), verbose=False)
            try_seq(a_opt, "lbfgs_pert")

    # === Phase 5: Final micro-optimization (remaining) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: Coordinate descent
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            a_opt, _ = coordinate_descent_fast(best_seq, time_limit=min(rem, 100))
            try_seq(a_opt, "cd")

        # 5b: Subgradient polish
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 80),
                                 lr=0.003, mom=0.94,
                                 top_k=max(30, n_best // 6),
                                 temp=0.00005)
            try_seq(a_opt, "sg")

        # 5c: Short Nesterov with high beta
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = nesterov_lse(best_seq, time_limit=min(rem, 60), verbose=False)
            try_seq(a_opt, "nesterov_final")

        # 5d: Short LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 50), verbose=False)
            try_seq(a_opt, "lpfw_final")

        # 5e: Very short coordinate descent
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent_fast(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "cd_final")

        # 5f: Final SG with very small step
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 20),
                                 lr=0.001, mom=0.95,
                                 top_k=max(10, n_best // 10),
                                 temp=0.00002)
            try_seq(a_opt, "sg_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
