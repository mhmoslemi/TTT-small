Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp projected gradient with adaptive tau
2. LP-guided direction refinement (SOTA approach)
3. Coordinate descent and local search for final polish

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def smooth_max(conv, tau):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    # Shift for numerical stability
    c_max = conv.max()
    exp_vals = np.exp(tau * (conv - c_max))
    log_sum = np.log(exp_vals.sum()) + tau * c_max
    p = exp_vals / exp_vals.sum()
    return log_sum / tau, p

def compute_smooth_grad(a, p, tau):
    """Compute gradient of smooth max(conv(a,a)) using softmax weights p."""
    n = len(a)
    # p has length 2n-1
    # grad_j = 2 * sum_{m=0}^{n-1} p_{j+m} * a_m
    # Use sliding window
    p_windows = np.lib.stride_tricks.sliding_window_view(p, n)  # shape (n, n)
    grad = 2.0 * (p_windows @ a)
    return grad

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

        grad = compute_smooth_grad(a, p, tau)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp projected gradient,
LP-guided refinement, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def smooth_objective_and_grad(a, tau):
    """Compute smooth max and its gradient for auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    c_shifted = conv - conv.max()
    exp_vals = np.exp(tau * c_shifted)
    p = exp_vals / exp_vals.sum()
    # Gradient: grad_j = 2 * sum_m p_{j+m} * a_m
    p_windows = np.lib.stride_tricks.sliding_window_view(p, n)
    grad = 2.0 * (p_windows @ a)
    # Center gradient (invariant to adding constant)
    grad -= grad.mean()
    return conv.max(), p, grad

def optimize_smooth(a0, time_limit, verbose=False):

def flat_spectrum_init(n, rng):

    A = np.zeros(n, dtype=complex)
    A[0] = n  # DC component

        A[k] = np.exp(1j * phases[k])

    A_real = np.zeros(n, dtype=complex)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp projected gradient with adaptive tau and backtracking
2. LP-guided direction refinement (SOTA linear programming approach)
3. Coordinate descent with multipliers for final polish

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smooth_objective_and_grad(a, tau):
    """Compute smooth max (log-sum-exp) and gradient for auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    c_shifted = conv - conv.max()
    exp_vals = np.exp(tau * c_shifted)
    p = exp_vals / exp_vals.sum()
    p_windows = np.lib.stride_tricks.sliding_window_view(p, n)
    grad = 2.0 * (p_windows @ a)
    grad -= grad.mean()
    return conv.max(), p, grad

def build_lp_matrix(f):
    """Build constraint matrix A for conv(f,g)_k: A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiplier search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = [0.4, 0.6, 0.75, 0.85, 1.15, 1.3, 1.5, 1.8]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue
            best_mult = 1.0
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = a[j] * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                rem = 1.0 - a[j]
                if rem < 1e-12:
                    continue
                scale = (1.0 - new_val) / rem
                a_try = a.copy()
                a_try[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_try[mask] *= scale
                a_try = np.maximum(a_try, 0.0)
                ss = a_try.sum()
                if ss < 1e-12:
                    continue
                a_try /= ss
                obj = fast_convolve(a_try).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_mult = mult

            if best_mult != 1.0:
                rem = 1.0 - a[j]
                if rem > 1e-12:
                    scale = (1.0 - a[j] * best_mult) / rem
                    a_new = a.copy()
                    a_new[j] = a_new[j] * best_mult
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_new[mask] *= scale
                    a_new = np.maximum(a_new, 0.0)
                    a_new /= a_new.sum()
                a = a_new
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      CD pass done: obj={best_obj:.10f} ({elapsed:.0f}s)")

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2.0 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random exponential initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def gaussian_init(n):
    """Gaussian-shaped initialization."""
    x = np.linspace(-2.5, 2.5, n)
    a = np.exp(-x**2 / 2.0)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_seq, best_score):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-10:
        return best_seq, best_score
    a_norm = a / ss
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: smooth log-sum-exp projected gradient,
    LP-guided refinement, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    n_target = len(parent) if parent is not None else 768
    target_lengths = [n_target]
    for n_try in [512, 640, 768, 896, 1024, 1152, 1280]:
        if n_try not in target_lengths and n_try <= 1400:
            target_lengths.append(n_try)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_target))[:4]

    inits = {}
    for n in target_lengths:
        candidates = [
            flat_spectrum_init(n, rng),
            random_init(n, rng),
            gaussian_init(n),
        ]
        if parent is not None:
            candidates.append(interpolate_sequence(parent, n))
            pert = parent + rng.standard_normal(len(parent)) * 0.03
            pert = project_simplex(pert, 1.0)
            candidates.append(interpolate_sequence(pert, n))

        best_init = None
        best_init_obj = float('inf')
        for init in candidates:
            init = np.maximum(init, 0.0)
            ss = init.sum()
            if ss < 1e-10:
                continue
            init = init / ss
            obj = fast_convolve(init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init

        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init obj={best_init_obj:.8f}")

    # === Phase 2: Smooth gradient optimization ===
    print(f"\n=== Phase 2: Smooth Gradient (best={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    time_fracs = [0.45, 0.35, 0.20, 0.15]
    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:4]):
        rem = deadline - time.monotonic() - 700
        if rem < 20:
            break
        t_alloc = min(rem * time_fracs[rank], 250)
        if t_alloc < 15:
            break
        print(f"  Smooth grad n={n_len}, init_obj={init_obj:.8f}, t={t_alloc:.0f}s")
        a_opt, _ = optimize_smooth(init, time_limit=t_alloc, tau_init=40.0, verbose=(rank == 0))
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_lp = min(rem * 0.7, 350)
            print(f"  LP t={t_lp:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_refine(best_seq, time_limit=t_lp, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

    # === Phase 4: Post-LP smooth refinement ===
    print(f"\n=== Phase 4: Post-LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_sg = min(rem * 0.4, 100)
            a_opt, _ = optimize_smooth(best_seq, time_limit=t_sg, tau_init=100.0)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

            rem = deadline - time.monotonic() - 200
            if rem > 30:
                t_lp2 = min(rem * 0.5, 120)
                a_opt, _ = lp_refine(best_seq, time_limit=t_lp2, verbose=True)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_cd = min(rem * 0.6, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

    # === Phase 6: Alternating polish with perturbations ===
    print(f"\n=== Phase 6: Alternating Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for rnd in range(8):
            rem = deadline - time.monotonic() - 15
            if rem < 12:
                break
            if rnd % 2 == 0:
                noise_scale = 0.002 + 0.001 * (rnd // 2)
                f_pert = best_seq + rng.standard_normal(n_best) * noise_scale
                f_pert = project_simplex(f_pert, 1.0)
                t_alloc = min(rem * 0.4, 50)
                a_opt, _ = optimize_smooth(f_pert, time_limit=t_alloc, tau_init=80.0)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score)
            else:
                t_alloc = min(rem * 0.4, 50)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

    # === Phase 7: Try longer sequence ===
    print(f"\n=== Phase 7: Longer Sequence (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_n = min(n_best + 128, 1280)
        if target_n > n_best:
            rem = deadline - time.monotonic() - 20
            if rem > 30:
                f_interp = interpolate_sequence(best_seq, target_n)
                t_alloc = min(rem * 0.4, 80)
                print(f"  Longer n={target_n}, t={t_alloc:.0f}s")
                a_opt, _ = optimize_smooth(f_interp, time_limit=t_alloc, tau_init=40.0)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
