import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A such that (A @ g) = conv(f, g)."""
    n = len(f)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For row k, valid j are max(0, k-n+1) to min(n-1, k)
    # A[k, j] = f[k-j]
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        A[k, j_range] = f[k - j_range]
    return A

def lp_descent_step(f, M, solver='highs'):
    """Solve: max sum(g) s.t. A @ g <= M * ones, g >= 0."""
    from scipy.optimize import linprog
    n = len(f)
    A = build_conv_matrix(f)
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method=solver)
    if res.success:
        return res.x, res.fun  # res.fun = -sum(g_opt)
    return None, None

def lp_simplicial_descent(f, n_iters, time_limit):
    """LP-based simplicial descent for minimizing max(conv(f,f)) with sum(f)=1."""
    from scipy.optimize import linprog
    n = len(f)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Build constraint matrix
        A = build_conv_matrix(f)
        b = np.full(2*n-1, M)
        c = -np.ones(n)
        
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        
        g_opt = res.x
        V = sum(g_opt)
        if V <= 1 + 1e-10:
            # No improvement possible
            break
        
        g = g_opt / V  # normalize to sum 1
        
        # Line search on t
        best_t = 0
        best_new_obj = best_obj
        for t in [0.5, 0.7, 0.8, 0.9, 1.0]:
            f_trial = (1-t)*f + t*g
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_t = t
        
        if best_t > 0:
            f = (1-best_t)*f + best_t*g
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_new_obj < best_obj:
                best_obj = best_new_obj
                best_f = f.copy()
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp smoothing on softmax parameterization
2. LP-based simplicial descent
3. Coordinate descent for local polish
4. Multi-length exploration

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        A[k, j_range] = f[k - j_range]
    return A

def flat_spectrum_init(n, rng):
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def optimize_lbfgs(a0, time_limit, max_tau=5000):
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = np.log(a + 1e-30)
    best_f = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    taus = [10, 30, 100, 300, 1000, 3000, max_tau]
    
    for tau in taus:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        # Define objective
        def obj_grad(z):
            a_loc = np.exp(z - z.max())
            a_loc = a_loc / a_loc.sum()
            conv = fast_convolve(a_loc)
            shifted = tau * (conv - conv.max())
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros(len(conv))
                w[np.argmax(conv)] = 1.0
            else:
                w = w / ws
            f_val = conv.max() + np.log(ws) / tau
            
            grad_a = np.zeros(n)
            k_actual = min(300, len(conv))
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            for k in peaks:

                if wk < 1e-12:

                js = max(0, k - n + 1)

            return f_val, grad_z
        
        max_iter = max(50, int(remaining / 0.05))
        res = minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': max_iter * 3})
        
        a_final = np.exp(res.x - res.x.max())
        a_final = a_final / a_final.sum()
        f_final = fast_convolve(a_final).max()
        
        if f_final < best_f - 1e-12:
            best_f = f_final
            best_a = a_final.copy()
            z = res.x.copy()
    
    return best_a, best_f

def lp_descent(a0, time_limit, verbose=False):
    n = len(a0)
    f = np.maximum(a0, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        A = build_conv_matrix(f)
        b = np.full(2*n-1, M)
        c = -np.ones(n)
        
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                     options={'presolve': True})
        if not res.success:
            break
        
        g_opt = res.x
        V = g_opt.sum()
        if V <= 1.0 + 1e-8:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        g = g_opt / V
        
        # Line search
        best_t = 0
        best_new_obj = best_obj
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            f_trial = (1-t)*f + t*g
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            obj = fast_convolve(f_trial).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_t = t
        
        if best_t > 0:
            f = (1-best_t)*f + best_t*g
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_new_obj < best_obj - 1e-12:
                best_obj = best_new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: obj={best_obj:.10f}, V={V:.4f}, t={best_t:.2f}")
    
    return best_f, best_obj

                continue
            f_trial = f_trial / ss
            obj = fast_convolve(f_trial).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_t = t

        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_new_obj < best_obj - 1e-12:
                best_obj = best_new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: obj={best_obj:.10f}, V={V:.4f}, t={best_t:.2f} ({time.monotonic()-t0:.0f}s)")

    return best_f, best_obj


def coordinate_descent(a0, time_limit, seed=0):
    """Randomized coordinate descent with multiplier trials."""
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    multipliers = [0.4, 0.6, 0.8, 1.2, 1.4, 1.7]

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj
