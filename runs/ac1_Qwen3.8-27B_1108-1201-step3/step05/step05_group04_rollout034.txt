Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe,
subgradient mirror descent, and coordinate polishing.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_single_peak(a, k):
    """Compute gradient of q_k(a) = (a*a)[k] w.r.t. a.
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    # j ranges from max(0, k-n+1) to min(k, n-1)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    for j in range(j_start, j_end + 1):
        grad[j] = 2.0 * a[k - j]
    return grad

def compute_subgradient_vectorized(a, peaks):
    """Compute average subgradient over a set of peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peaks:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        indices = np.arange(j_start, j_end + 1)
        grad[indices] += 2.0 * a[k - indices]
    grad /= len(peaks)
    return grad

    max_val = conv.max()
    threshold = threshold_frac * max_val
    peaks = np.where(conv >= threshold)[0]
    return list(peaks)

def lp_direction(a, M):
    """Solve LP to find direction b maximizing sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Constraint matrix: for each k in 0..2n-2, A[k,j] = a[k-j] if valid
    # We build this as a sparse-like structure
    # Number of constraints: 2n-1
    num_constraints = 2 * n - 1
    
    # Build constraint matrix
    # For efficiency, build row by row
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_gaussian_init(n, center=0.5, sigma=0.15):

- `d[i] >= -a[i]` for all `i` (to maintain non-negativity)

The gradient `g_k` (for the center peak) is `2*a` (since `conv[n-1] = sum a[i]^2`, so `grad = 2a`).

For other peaks `k`, the gradient is `2 * a[shifted]` as derived earlier.

The LP is:
Minimize `t`
Subject to:
- `<g_k, d> <= t` for `k` in S (active peaks)
- `sum(d) = 0`
- `d >= -a`
- `d` is palindromic (or we can just work with half the variables)

Number of variables: `ceil(n/2) + 1` (for `t`)
Number of constraints: `|S| + 1 + n/2` (for the palindromic constraints, we just don't include them as constraints since we only have half the variables)

For `n = 768` and `|S| ≈ 20`: about 400 variables and 400+ constraints. Very fast to solve.

Actually, I realize that for the palindromic case, the gradient at the center is `2*a`, and the direction `d` must satisfy `sum(d) = 0`. So `<2a, d> = 2*sum(a[i]*d[i])`. Since `sum(d) = 0`, this is `2*sum(a[i]*d[i])` which we want to minimize.

If all `a[i] > 0`, the best direction is to move mass from high-`a` positions to low-`a` positions. This is exactly what the Frank-Wolfe step does: move mass to the position with the smallest `a` value.

So the Frank-Wolfe step (move to the vertex with minimum `a` value) is actually the steepest descent direction for the center peak! This is a nice simplification.

But we need to consider ALL active peaks, not just the center. So the full LP is still useful.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using palindromic constraint,
subgradient mirror descent, LP-based direction finding, and coordinate polishing.
Key insight: restricting to palindromic sequences halves the search space

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def symmetrize(a):
    """Make a palindromic and normalize to sum=1."""
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_peaks(conv, frac=0.999):
    """Get indices where conv is within frac of max."""
    max_val = conv.max()
    threshold = max_val * frac
    return np.where(conv >= threshold)[0]


def compute_grad_at_k(a, k):
    """Gradient of q_k(a) = (a*a)[k] w.r.t. a.
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad


def compute_subgradient(a, peaks):
    """Average subgradient over a set of peak positions."""
    n = len(a)
    if len(peaks) == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in peaks:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            grad[indices] += 2.0 * a[k - indices]
    grad /= len(peaks)
    return grad


def lp_steepest_descent(a, conv, S):
    """
    Solve LP for steepest descent direction on the simplex.
    Minimize max_{k in S} <g_k, d> subject to sum(d)=0, d >= -a.
    Returns direction d (already normalized to be a convex combo step).
    """
    n = len(a)
    # Variables: d[0..n-1], t
    # Total: n+1 variables
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    A_ub = []
    b_ub = []

    # Constraints: <g_k, d> - t <= 0 for each k in S
    for k in S:
        gk = compute_grad_at_k(a, k)
        row = np.zeros(n + 1)
        row[:n] = gk
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(0.0)

    # Constraints: d >= -a, i.e., -d <= a
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_ub.append(row)
        b_ub.append(a[j])

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    if len(A_ub) == 0:
        A_ub = None
        b_ub = None
    else:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=[(None, None)] * n + [(None, None)],
                     method='highs')

    if result.success:
        d = result.x[:n]
        return d
    return None


def mirror_descent_opt(a0, time_limit, lr_init=0.03, lr_decay=0.9997,
                       peak_frac=0.999, use_lp=False, lp_every=5,
                       verbose=False):
    """Mirror descent with palindromic constraint and subgradient from peaks."""
    a = symmetrize(a0.astype(np.float64).copy())
    n = len(a)

    cur_lr = lr_init
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"      iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 7)
            a = symmetrize(best_a + rng.standard_normal(n) * 0.003)
            best_obj = fast_convolve(a).max()
            cur_lr = lr_init
            no_improve = 0
            continue

        peaks = get_peaks(conv, peak_frac)

        if use_lp and it % lp_every == 0 and len(peaks) <= 30:
            d = lp_steepest_descent(a, conv, peaks.tolist())
            if d is not None:
                # Normalize d: make it a valid step direction
                # d should have sum=0; scale to have max |d| small
                d_max = np.abs(d).max()
                if d_max > 1e-15:
                    d = d / d_max * 0.01  # small step
                    a_new = a + d
                    a = symmetrize(a_new)
                else:
                    # Fallback to mirror descent
                    sg = compute_subgradient(a, peaks)
                    log_update = -cur_lr * sg
                    log_update -= log_update.max()
                    a_new = a * np.exp(log_update)
                    a = symmetrize(a_new)
            else:
                sg = compute_subgradient(a, peaks)
                log_update = -cur_lr * sg
                log_update -= log_update.max()
                a_new = a * np.exp(log_update)
                a = symmetrize(a_new)
        else:
            sg = compute_subgradient(a, peaks)
            log_update = -cur_lr * sg
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            a = symmetrize(a_new)

        cur_lr *= lr_decay
        cur_lr = max(cur_lr, lr_init * 0.01)

    return best_a, best_obj


def frank_wolfe_opt(a0, time_limit, peak_frac=0.999, verbose=False):
    """Frank-Wolfe with palindromic constraint."""
    a = symmetrize(a0.astype(np.float64).copy())
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"      FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            a = symmetrize(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            continue

        peaks = get_peaks(conv, peak_frac)
        sg = compute_subgradient(a, peaks)

        # FW step: find vertex e_j that minimizes sg[j]
        j_star = int(np.argmin(sg))
        e = np.zeros(n)
        e[j_star] = 1.0
        e = symmetrize(e)  # project to palindromic simplex

        # Step size: gamma = 2/(it+2) or use line search
        gamma = 2.0 / (it + 2.0)

        # Line search for better step
        best_gamma = gamma
        best_new_obj = cur_obj
        for g in [gamma * 0.1, gamma * 0.3, gamma * 0.5, gamma, gamma * 1.5, gamma * 2]:
            a_trial = (1 - g) * a + g * e
            a_trial = symmetrize(a_trial)
            new_obj = fast_convolve(a_trial).max()
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_gamma = g

        a = (1 - best_gamma) * a + best_gamma * e
        a = symmetrize(a)

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """Polish via local mass transfers between pairs of positions."""
    a = symmetrize(a0.astype(np.float64).copy())
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        n_half = (n + 1) // 2

        for j in range(n_half):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue

            # Try moving mass from j (and its mirror) to k (and its mirror)
            k_mirror = n - 1 - j

            for k in range(n_half):
                if k == j:
                    continue
                k_mirror2 = n - 1 - k

                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[j] * frac
                    a_trial = a.copy()
                    a_trial[j] -= amount
                    if k_mirror != j:
                        a_trial[k_mirror] -= amount
                    a_trial[k] += amount
                    if k_mirror2 != k:
                        a_trial[k_mirror2] += amount

                    a_trial = np.maximum(a_trial, 0)
                    a_trial = symmetrize(a_trial)

                    new_obj = fast_convolve(a_trial).max()
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break

        if not improved:
            # Random perturbations
            rng = np.random.default_rng(passes * 7 + 1)
            found = False
            for _ in range(20):
                j = rng.integers(0, n_half)
                k = rng.integers(0, n_half)
                if j == k:
                    continue
                amount = a[j] * 0.05
                if amount < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[j] -= amount
                a_trial[n-1-j] -= amount
                a_trial[k] += amount
                a_trial[n-1-k] += amount
                a_trial = np.maximum(a_trial, 0)
                a_trial = symmetrize(a_trial)
                new_obj = fast_convolve(a_trial).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    found = True
                    break

            if not improved and passes > 5:
                break

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate to a new (palindromic-friendly) length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return symmetrize(f_interp)


def make_gaussian_init(n, sigma=0.2):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return symmetrize(f)


def make_bump_init(n):
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    return symmetrize(f)


def make_power_init(n, p=1.0):
    x = np.linspace(0, 1, n)
    f = (1 - 2 * np.abs(x - 0.5)) ** p
    f = np.maximum(f, 1e-10)
    return symmetrize(f)

                                           peak_frac=0.999, verbose=False)
            try_seq(a_opt, f"p6_pert{pi}")

    # Fallback
    if best_seq is None:
        best_seq = make_bump_init(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
