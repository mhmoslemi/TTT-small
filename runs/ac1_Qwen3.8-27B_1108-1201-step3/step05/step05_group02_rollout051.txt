"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP Frank-Wolfe with Barzilai-Borwein step sizes,
peak equalization, multi-length search, and informed restarts.
Key innovations: active-set LP (faster directions), BB adaptive steps,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(f, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def solve_lp_active(f, M, n, active_fractions=(0.9, 0.95, 1.0)):
    """Solve LP direction using active set of constraints near the max."""
    conv = fast_convolve(f)
    conv_max = conv.max()
    
    for frac in active_fractions:
        threshold = M * frac
        active = np.where(conv >= threshold)[0]
        if len(active) < 3:
            active = np.where(conv >= conv_max * 0.5)[0]
        
        c = -np.ones(n)
        A_ub = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[idx, js:je + 1] = f[k - je:js - 1:-1] if js > 0 else f[k - je::-1]
        
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 30, 'presolve': True})
            if res.success and res.x is not None and res.x.sum() > 1e-10:
                return res.x
        except Exception:
            pass
    
    # Fallback: full LP
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,

        if res.success and res.x is not None and res.x.sum() > 1e-10:
            return res.x
    except Exception:
        pass
    return None

def peak_equalization(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = np.argmax(conv)
        
        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(-conv[peak_indices])]
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            if conv[k] < M * 0.7:
                break
            
            # Get the subgradient at this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad = np.zeros(n)

                grad[j] += f[k - j]
                if k - j != j:
                    grad[k - j] += f[j]
            
            # Reduce coordinates with high gradient contribution
            # and redistribute to low-gradient coordinates
            sorted_idx = np.argsort(-grad)
            top_contrib = sorted_idx[:max(2, n // 10)]
            low_contrib = sorted_idx[-max(2, n // 10):][::-1]
            
            # Try reducing top contributors and adding to low contributors
            for frac in [0.05, 0.1, 0.15, 0.2]:
                f_trial = f.copy()
                mass_removed = 0
                for j in top_contrib:
                    if f_trial[j] > 1e-8:
                        reduce = f_trial[j] * frac
                        f_trial[j] -= reduce
                        mass_removed += reduce
                # Distribute to low contributors
                n_low = len(low_contrib)
                if n_low > 0 and mass_removed > 0:
                    per = mass_removed / n_low
                    for j in low_contrib:
                        f_trial[j] += per
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < M - 1e-12:
                    f = f_trial
                    if val < best_obj:
                        best_obj = val
                        best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            # Try a different strategy: uniform small reduction of peak contributors
            f_trial = f.copy()
            for j in top_contrib[:5]:
                if f_trial[j] > 1e-8:
                    f_trial[j] *= 0.95
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_obj:
                f = f_trial
                best_obj = val
                best_f = f.copy()
                improved = True
        
        if not improved and it > 20:
            # Try adding mass to "dead zones" (low convolution contribution)
            contrib = np.zeros(n)
            for k in peak_indices[:5]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    contrib[j] += f[k - j]
            
            dead = np.where(contrib < np.median(contrib))[0]
            if len(dead) > 0:
                f_trial = f.copy()
                # Slightly boost dead zones
                boost = 0.05
                for j in dead[:10]:
                    f_trial[j] *= (1 + boost)
                f_trial = np.maximum(f_trial, 0)
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
        
        if not improved and it % 50 == 0:
            rng = np.random.default_rng(it * 13 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
    
    return best_f, best_obj


def logsumexp_polish(f0, time_limit, tau_init=50.0, lr=0.001, verbose=False):
    """Smooth objective optimization using log-sum-exp surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    tau = tau_init
    cur_lr = lr
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Softmax weights over peaks
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient: sum_k w_k * 2 * f[k-j] (for valid indices)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * w[k] * f[k - j]
        
        # Projected gradient step
        f_new = f - cur_lr * grad
        f_new = project_simplex(f_new)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        cur_lr *= 0.999
        tau *= 1.01
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            cur_lr = max(lr * 0.5, 0.0001)
            no_improve = 0
    
    return best_f, best_obj


def informative_restart(f0, rng, n):
    """Generate an informed restart based on convolution structure."""
    f = f0.copy()
    conv = fast_convolve(f)
    
    strategy = rng.integers(0, 5)
    
    if strategy == 0:
        # Smooth envelope modulation
        x = np.linspace(0, 1, n)
        freq = rng.uniform(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        envelope = 1 + 0.1 * np.sin(2 * np.pi * freq * x + phase)
        f_new = f * envelope
    elif strategy == 1:
        # Gaussian window blending
        center = n / 2
        sigma = rng.uniform(0.2, 0.5) * n
        window = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
        alpha = rng.uniform(0.1, 0.4)
        f_new = (1 - alpha) * f + alpha * window
        f_new = f_new / f_new.sum() * f.sum()
    elif strategy == 2:
        # Peak shaving: reduce the highest elements
        top = np.argsort(-f)[:max(3, n // 20)]
        f_new = f.copy()
        for j in top:
            f_new[j] *= rng.uniform(0.7, 0.9)
    elif strategy == 3:
        # Segment mirroring
        seg_start = rng.integers(0, n // 2)
        seg_len = rng.integers(n // 8, n // 3)
        f_new = f.copy()
        seg = f_new[seg_start:seg_start + seg_len]
        f_new[seg_start:seg_start + seg_len] = seg[::-1]
    else:
        # Cosine taper at edges
        edge = max(3, n // 16)
        f_new = f.copy()
        taper = np.linspace(0.5, 1.0, edge)
        f_new[:edge] *= taper
        f_new[-edge:] *= taper[::-1]
    
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    
    return f_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining active-set LP-FW, peak equalization,
    log-sum-exp smoothing, multi-length search, and informed restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Multi-length quick search ===
    print(f"\n=== Phase 1: Multi-length search (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        candidate_lengths = [n_parent]
        for delta in [-128, -96, -64, -32, -16, 16, 32, 64, 96, 128, 192, 256]:
            nl = n_parent + delta
            if 256 <= nl <= 2560 and nl != n_parent:
                candidate_lengths.append(nl)
        candidate_lengths = sorted(set(candidate_lengths))
        
        t_per = min(30, rem * 0.08)
        for nl in candidate_lengths:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            f_interp = interp_sequence(parent, nl)
            # Quick FW
            t_alloc = min(t_per, rem * 0.3)
            a_opt, _ = fw_optimize_active(f_interp, time_limit=t_alloc)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  n={nl}: score={sc:.8f} *** NEW BEST")
            else:
                print(f"  n={nl}: score={sc:.8f}")
    
    # === Phase 2: Deep LP-FW on best ===
    print(f"\n=== Phase 2: Deep LP-FW (best={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t2 = min(400, rem * 0.35)
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = fw_optimize_active(best_seq, time_limit=t2, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST after deep FW: {sc:.8f}")
    
    # === Phase 3: Peak equalization ===
    print(f"\n=== Phase 3: Peak equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(150, rem * 0.2)
        a_opt, _ = peak_equalization(best_seq, time_limit=t3)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST after peak eq: {sc:.8f}")
    
    # === Phase 4: Log-sum-exp polish ===
    print(f"\n=== Phase 4: LogSumExp polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t4 = min(100, rem * 0.2)
        n_best = len(best_seq)
        tau0 = max(20.0, 500.0 / n_best)
        a_opt, _ = logsumexp_polish(best_seq, time_limit=t4, tau_init=tau0, lr=0.0008)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST after LSE: {sc:.8f}")
    
    # === Phase 5: Informed restarts + FW ===
    print(f"\n=== Phase 5: Informed restarts (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    restarts_done = 0
    max_restarts = 12
    
    while rem > 30 and restarts_done < max_restarts:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t_alloc = min(120, rem * 0.25)
        if t_alloc < 10:
            break
        
        f_restart = informative_restart(best_seq, rng, n_best)
        a_opt, _ = fw_optimize_active(f_restart, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Restart {restarts_done}: NEW BEST {sc:.8f}")
        else:
            print(f"  Restart {restarts_done}: {sc:.8f}")
        restarts_done += 1
    
    # === Phase 6: Final multi-length fine search ===
    print(f"\n=== Phase 6: Final length tuning (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_best = len(best_seq)
        for delta in [-8, -4, 4, 8]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            nl = n_best + delta
            if nl < 64:
                continue
            f_interp = interp_sequence(best_seq, nl)
            t_alloc = min(60, rem * 0.3)
            a_opt, _ = fw_optimize_active(f_interp, time_limit=t_alloc)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  n={nl}: NEW BEST {sc:.8f}")
    
    # === Phase 7: Final squeeze ===
    print(f"\n=== Phase 7: Final squeeze (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.8, 120)
        a_opt, _ = fw_optimize_active(best_seq, time_limit=t7)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Squeeze: NEW BEST {sc:.8f}")
        
        # Peak equalization on the result
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = peak_equalization(best_seq, time_limit=min(rem * 0.7, 60))
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  PeakEq: NEW BEST {sc:.8f}")
    
    # === Phase 8: Micro perturbations ===
    print(f"\n=== Phase 8: Micro perturbations (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        n_best = len(best_seq)
        for trial in range(15):
            rem = deadline - time.monotonic()
            if rem < 3:
                break
            scales = [0.0001, 0.0002, 0.0003, 0.0005, 0.0008]
            scale = scales[trial % len(scales)]
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            sc = float(evaluate_sequence(f_pert.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f_pert.copy()
                print(f"  Perturb {trial}: NEW BEST {sc:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(fs):
        best_seq = np.ones(512) / 512
        fs = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL: score={fs:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
