# Helper functions (top-level, no closures)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv_matrix(f):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2*n-1, n))

        A[j:j+n, j] = f
    return A

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):

    for j in range(n):

    return A

            top_indices = np.argpartition(-conv, top_k)[:top_k]
            w = np.zeros(len(conv))
            w[top_indices] = 1.0 / top_k
        else:
            w = np.ones(len(conv)) / len(conv)

        # Also add soft-max component for smoothness
        soft = conv / temp
        soft -= soft.max()
        soft_w = np.exp(soft)
        soft_w /= soft_w.sum()
        w = 0.5 * w + 0.5 * soft_w
        w /= w.sum()

        # Subgradient: sg[j] = sum_i w[i+j] * f[i] for j=0..n-1
        sg = np.correlate(w, f, 'valid')

        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-6)
        cur_lr = max(lr_init * (0.9999 ** t), lr_init * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    Subgrad iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive top-k: reduce as we converge
        top_k = max(5, top_k_init - no_improve // 50)

        if no_improve > 200:
            rng = np.random.default_rng(t * 31 + 7)
            noise = rng.standard_normal(n) * 0.005
            f = project_simplex(best_f + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.5, 0.002)
            cur_lr = lr_init
            no_improve = 0
            top_k = top_k_init

    return best_f, best_obj


def mirror_descent_opt(f0, time_limit=200.0, lr_init=0.01, top_k=30,
                       temp_init=0.005, verbose=False):
    """Mirror descent (multiplicative weights) with top-k subgradients."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    n = len(f)

    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    lr = lr_init
    temp = temp_init
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        if top_k < len(conv):
            top_indices = np.argpartition(-conv, top_k)[:top_k]
            w = np.zeros(len(conv))
            w[top_indices] = 1.0 / top_k
        else:
            w = np.ones(len(conv)) / len(conv)

        sg = np.correlate(w, f, 'valid')

        # Multiplicative update
        log_update = -lr * (sg - sg.mean())
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        lr = max(lr * 0.999, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(t * 13 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_fast(f0, time_limit=100.0, verbose=False):
    """Fast coordinate descent: only adjust coordinates that matter."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Focus on coordinates with high values (they contribute most to peaks)
        order = np.argsort(-f)
        n_try = min(n, 200)

        for idx in order[:n_try]:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(idx)
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 3.0]:
                new_val = old_val * mult
                if new_val < -1e-10 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_init(n, rng):
    """Create structured initializations: cosine-modulated, etc."""
    x = np.linspace(0, 1, n)
    inits = []

    # Cosine modulated uniform
    for k in [1, 2, 3, 5]:
        for alpha in [0.2, 0.4, 0.6]:
            f = 1.0 + alpha * np.cos(2 * np.pi * k * x)
            f = np.maximum(f, 0.01)
            inits.append(f / f.sum())

    # Gaussian bump
    for sigma in [0.1, 0.15, 0.2, 0.3]:
        f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(f / f.sum())

    # Sinusoidal
    for freq in [2, 4, 8]:
        f = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x)
        f = np.maximum(f, 0.01)
        inits.append(f / f.sum())

    # Two bumps
    for sep in [0.2, 0.3, 0.4]:
        f = np.exp(-((x - 0.25) / 0.08) ** 2) + np.exp(-((x - 0.75) / 0.08) ** 2)
        inits.append(f / f.sum())

    # Random smooth
    for _ in range(5):
        f = np.zeros(n)
        for _ in range(rng.integers(3, 8)):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.05, 0.3)
            a = rng.uniform(0.5, 2.0)
            f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
        f = np.maximum(f, 1e-10)
        inits.append(f / f.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-based direction finding,
    FISTA with LSE smoothing, exact subgradient methods, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform init n=768")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, best={best_score:.8f}")

    # === Phase 1: LP-based optimization (main novelty) ===
    print(f"\n=== Phase 1: LP-based optimization (n={n_parent}) ===")
    t_phase1_end = time.monotonic() + 500

    rem = t_phase1_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem, 480)
        print(f"  LP iteration, t={t_alloc:.0f}s")
        a_lp, obj_lp = lp_iteration(parent, time_limit=t_alloc, verbose=True)
        try_seq(a_lp, "p1_lp_parent")

        # Also try LP from a structured init
        inits = make_structured_init(n_parent, rng)
        if len(inits) > 0:
            # Pick the best structured init quickly
            best_init = None
            best_init_obj = float('inf')
            for init in inits[:8]:
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init
            if best_init is not None:
                rem = t_phase1_end - time.monotonic()
                if rem > 30:
                    t_alloc2 = min(rem, 200)
                    print(f"  LP from best structured init, t={t_alloc2:.0f}s")
                    a_lp2, _ = lp_iteration(best_init, time_limit=t_alloc2, verbose=True)
                    try_seq(a_lp2, "p1_lp_struct")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: FISTA with LSE smoothing ===
    print(f"\n=== Phase 2: FISTA LSE (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 350
    n_base = len(base)

    rem = t_phase2_end - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.6, 200)
        print(f"  FISTA LSE (alpha=500->50), t={t_alloc:.0f}s")
        a_fista, _ = fista_lse(base, time_limit=t_alloc,
                               alpha_init=500.0, alpha_final=50.0,
                               alpha_decay=0.999, verbose=True)
        try_seq(a_fista, "p2_fista")

    rem = t_phase2_end - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.5, 120)
        print(f"  FISTA LSE (alpha=200->30), t={t_alloc:.0f}s")
        a_fista2, _ = fista_lse(base, time_limit=t_alloc,
                                alpha_init=200.0, alpha_final=30.0,
                                alpha_decay=0.998, verbose=False)
        try_seq(a_fista2, "p2_fista2")

    base = best_seq if best_seq is not None else base

    # === Phase 3: Exact subgradient optimization ===
    print(f"\n=== Phase 3: Exact subgradient (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 300
    n_base = len(base)

    configs = [
        {'lr_init': 0.004, 'top_k_init': 10, 'temp_init': 0.004},
        {'lr_init': 0.006, 'top_k_init': 20, 'temp_init': 0.008},
        {'lr_init': 0.003, 'top_k_init': 5, 'temp_init': 0.003},
    ]

    for ci, cfg in enumerate(configs):
        rem = t_phase3_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.55, 100)
        print(f"  Subgrad config {ci}, t={t_alloc:.0f}s")
        a_sg, _ = exact_subgrad_optimize(base, time_limit=t_alloc, verbose=False, **cfg)
        try_seq(a_sg, f"p3_sg_{ci}")

        # Mirror descent
        rem = t_phase3_end - time.monotonic()
        if rem > 15:
            a_md, _ = mirror_descent_opt(base, time_limit=min(rem * 0.4, 60),
                                         top_k=cfg['top_k_init'],
                                         temp_init=cfg['temp_init'])
            try_seq(a_md, f"p3_md_{ci}")

    base = best_seq if best_seq is not None else base

    # === Phase 4: LP + FISTA combination on current best ===
    print(f"\n=== Phase 4: LP+FISTA combo (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 250
    n_base = len(base)

    rem = t_phase4_end - time.monotonic()
    if rem > 60:
        t_lp = min(rem * 0.4, 150)
        print(f"  LP iter, t={t_lp:.0f}s")
        a_lp3, _ = lp_iteration(base, time_limit=t_lp, verbose=True)
        try_seq(a_lp3, "p4_lp")

        if best_seq is not None:
            rem = t_phase4_end - time.monotonic()
            if rem > 30:
                t_fista = min(rem * 0.5, 90)
                print(f"  FISTA polish, t={t_fista:.0f}s")
                a_fp, _ = fista_lse(best_seq, time_limit=t_fista,
                                    alpha_init=300.0, alpha_final=20.0,
                                    alpha_decay=0.999)
                try_seq(a_fp, "p4_fista")

    base = best_seq if best_seq is not None else base

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + 200
    n_base = len(base)

    # Try lengths around current and some standard ones
    candidates = set()
    for factor in [0.75, 1.0, 1.25, 1.5]:
        candidates.add(int(n_base * factor))
    for standard in [512, 640, 768, 896, 1024, 1152, 1280, 1536]:
        if 256 <= standard <= 2048:
            candidates.add(standard)
    candidates = [c for c in candidates if c != n_base and 128 <= c <= 2048]
    candidates.sort(key=lambda x: abs(x - n_base))
    candidates = candidates[:3]

    for target_n in candidates:
        rem = t_phase5_end - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 80)

        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fista_lse(f_interp, time_limit=t_alloc,
                             alpha_init=300.0, alpha_final=40.0,
                             alpha_decay=0.999)
        try_seq(a_opt, f"p5_n{target_n}")

        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            a_opt2, _ = exact_subgrad_optimize(a_opt, time_limit=min(rem * 0.5, 40),
                                               lr_init=0.003, top_k_init=10,
                                               temp_init=0.004)
            try_seq(a_opt2, f"p5_n{target_n}_sg")

    base = best_seq if best_seq is not None else base

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple short FISTA runs from slightly perturbed starts
        for pi in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = [0.003, 0.005, 0.008, 0.002][pi]
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(rem * 0.35, 60)
            a_opt, _ = fista_lse(f_pert, time_limit=max(5, t_alloc),
                                 alpha_init=400.0, alpha_final=30.0,
                                 alpha_decay=0.999)
            try_seq(a_opt, f"p6_fista_{pi}")

        # Coordinate descent
        rem = deadline - time.monotonic() - 8
        if rem > 15:
            t_cd = min(rem * 0.6, 80)
            print(f"  CD, t={t_cd:.0f}s")
            a_cd, _ = coordinate_descent_fast(best_seq, time_limit=max(5, t_cd))
            try_seq(a_cd, "p6_cd")

        # Final LP
        rem = deadline - time.monotonic() - 5
        if rem > 15 and n_best <= 1500:
            t_lp_final = min(rem * 0.5, 60)
            print(f"  Final LP, t={t_lp_final:.0f}s")
            a_lp_final, _ = lp_iteration(best_seq, time_limit=max(5, t_lp_final), verbose=True)
            try_seq(a_lp_final, "p6_lp_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
