    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)

    A = rfft(a, fft_size)

        x = x / x.sum()

        return np.sum(b**p) ** (1.0/p)
    
    def grad(x):
        x = np.maximum(x, 0)
        x = x / x.sum()
        b = fast_convolve(x)
        bp = b**p
        s = bp.sum()
        # d/dx of (sum b^p)^(1/p) = (1/p) * (sum b^p)^(1/p-1) * p * sum b^(p-1) * db/dx
        # = (sum b^p)^(1/p-1) * sum_i b_i^(p-1) * (2 * (P_i x))_j
        # For our purposes, approximate with:
        w = b**(p-1) / b.sum()
        gb = 2.0 * n * cross_corr(w, x)
        # Project gradient to simplex tangent:
        S = np.dot(gb, x)
        g = gb - x * S
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                   'jac': lambda x: np.ones_like(x)}]
    bounds = [(0, None)] * n
    
    try:
        res = minimize(
            fun=obj,
            x0=a,
            jac=grad,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
        )
        a_new = np.maximum(res.x, 0)
        a_new = a_new / a_new.sum()
        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a_new
    except Exception as e:
        pass
    
    return best_a, best_obj

def subgradient_method(a0, time_limit=60.0, lr_init=0.01, top_k=50):
    """Subgradient method with careful learning rate schedule."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    momentum = 0.9
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Soft weighting
        temp = 0.001
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        
        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = max(lr_init * 0.3, 0.001)
            no_improve = 0
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def kaiser_init(n, beta):
    """Create a Kaiser window initialized sequence."""
    w = windows.kaiser(n, beta)
    w = w / w.sum()
    return w

Let me also think about a completely different approach: **Coordinate descent with exact line search**.

For each coordinate j, the objective as a function of a[j] (with all other coordinates fixed and the sum constraint maintained) is:
- If we fix all a[i] for i ≠ j and let a[j] = t, then the sum constraint gives us the scale of the other elements.
- The convolution b[k] = sum_i a[i]*a[k-i] becomes a quadratic in t for each k.
- The max of these quadratics is a convex function of t (max of convex functions).
- So we can find the optimal t by ternary search or golden section search.

This is exact and could be very effective for the "last mile"!

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-softmax parameterized L-BFGS with adaptive temperature annealing,
multi-length exploration, diverse initializations, coordinate descent with
incremental convolution updates, and SLSQP refinement.

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
from scipy.signal import windows
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, 1e10


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def kaiser_init(n, beta):
    w = windows.kaiser(n, beta)
    w = w / w.sum()
    return w


def gaussian_init(n):
    x = np.linspace(-2.5, 2.5, n)
    g = np.exp(-x ** 2)
    return g / g.sum()


def triangular_init(n):
    t = np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
    t = 1.0 - t
    return t / t.sum()


def slsqp_refine(a0, time_limit=60.0, maxiter=150):
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    p = 30

    def obj_func(x):
        x = np.maximum(x, 0)
        ss = x.sum()
        if ss < 1e-15:
            return 1e10
        x = x / ss
        b = fast_convolve(x)
        return np.sum(b ** p) ** (1.0 / p)

    def jac_func(x):
        x = np.maximum(x, 0)
        ss = x.sum()
        if ss < 1e-15:
            return np.zeros_like(x)
        x = x / ss
        b = fast_convolve(x)
        w = b ** (p - 1)
        ws = w.sum()
        if ws < 1e-15:
            return np.zeros_like(x)
        w = w / ws
        gb = 2.0 * n * cross_corr(w, x)
        S = np.dot(gb, x)
        g = gb - x * S
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                   'jac': lambda x: np.ones_like(x)}]
    bounds = [(0.0, None)] * n

    try:
        res = minimize(
            fun=obj_func,
            x0=a,
            jac=jac_func,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        a_new = np.maximum(res.x, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
            new_obj = fast_convolve(a_new).max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new
    except Exception:
        pass

    return best_a, best_obj


def coordinate_descent(a0, time_limit=120.0):
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    b = fast_convolve(a)
    best_a = a.copy()
    best_obj = b.max()
    t0 = time.monotonic()

    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.85, 0.9, 0.95,
                            1.05, 1.1, 1.2, 1.5, 2.0, 3.0])

    for sweep in range(1000):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                candidates = np.array([1e-4, 1e-3, 5e-3, 0.01])
            else:
                candidates = old_val * multipliers
                candidates = candidates[(candidates > 1e-8) & (candidates < 0.9999)]

            best_cand = old_val
            best_cand_obj = best_obj

            for new_val in candidates:
                if new_val >= 0.9999:
                    continue
                scale = (1.0 - new_val) / (1.0 - old_val)
                if scale <= 0 or scale > 10:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_cand_obj - 1e-15:
                    best_cand_obj = obj
                    best_cand = new_val

            if best_cand != old_val:
                scale = (1.0 - best_cand) / (1.0 - old_val)
                a[j] = best_cand
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                b = fast_convolve(a)
                cur_max = b.max()
                if cur_max < best_obj - 1e-14:
                    best_obj = cur_max
                    best_a = a.copy()
                    improved = True

        if not improved and sweep > 5:
            break

    return best_a, best_obj


def subgradient_method(a0, time_limit=60.0, lr_init=0.005, top_k=50):
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    momentum = 0.9
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.002
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 400:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.4, 0.001)
            no_improve = 0

    return best_a, best_obj

        n_best = len(best_seq)
        z0 = z_from_a(best_seq)
        total_deep = min(500.0, (deadline - time.monotonic()) * 0.35)
        print(f"  Deep budget: {total_deep:.0f}s, n={n_best}")
        z_curr = temperature_anneal(z0, n_best, total_deep,
                                    temps=[50, 100, 200, 500, 1000, 2000, 5000])
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, "deep_anneal")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.5, 90.0)
            print(f"  CD budget: {t_cd:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "coord_desc")

    # === Phase 5: SLSQP refinement ===
    print(f"\n=== Phase 5: SLSQP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_sls = min(rem * 0.5, 60.0)
            print(f"  SLSQP budget: {t_sls:.0f}s")
            a_sl, _ = slsqp_refine(best_seq, time_limit=t_sls, maxiter=100)
            try_seq(a_sl, "slsqp")

    # === Phase 6: Subgradient diversification + re-optimize ===
    print(f"\n=== Phase 6: Subgradient + re-optimize (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            n_best = len(best_seq)
            # Try a few perturbations
            for attempt in range(4):
                rem = deadline - time.monotonic() - 30
                if rem < 15:
                    break
                noise = rng.standard_normal(n_best) * 0.005
                a_pert = project_simplex(best_seq + noise)
                # Quick L-BFGS
                z_pert = z_from_a(a_pert)
                t_alloc = min(rem * 0.3, 25.0)
                z_opt, _ = lbgfs_optimize(z_pert, 200.0, n_best, t_alloc, maxiter=1000)
                a_opt = softmax_from_z(z_opt)
                try_seq(a_opt, f"perturb_{attempt}")

    # === Phase 7: Final high-T polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            z0 = z_from_a(best_seq)
            t_final = min(rem * 0.7, 40.0)
            z_fin, _ = lbgfs_optimize(z0, 5000.0, n_best, t_final, maxiter=3000)
            a_fin = softmax_from_z(z_fin)
            try_seq(a_fin, "final_T5000")

            # One more coordinate descent pass
            rem = deadline - time.monotonic() - 10
            if rem > 10 and best_seq is not None:
                a_cd2, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.8, 30.0))
                try_seq(a_cd2, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
