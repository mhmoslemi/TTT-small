import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgradient(conv, a, active_set):
    """Compute subgradient of max(conv) w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        # d conv[k]/d a[j] = 2 * a[k-j] for valid j
        # j ranges from max(0, k-(n-1)) to min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            # g[j] += 2 * a[k-j] for j in [j_lo, j_hi]
            # a[k-j] for j in [j_lo, j_hi] is a[k-j_hi : k-j_lo+1] reversed
            g[j_lo:j_hi+1] += 2.0 * a[k-j_hi:k-j_lo+1][::-1]
    return g

def mirror_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t = 1.0
    f_prev = f.copy()
    f_curr = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        beta = (t_prev - 1.0) / t
        
        # Nesterov extrapolation (geometric mean)
        log_f_curr = np.log(f_curr)
        log_f_prev = np.log(f_prev)
        log_y = (1.0 - beta) * log_f_curr + beta * log_f_prev
        max_log = log_y.max()
        y = np.exp(log_y - max_log)
        y = y / y.sum()
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        # Find active set (with tolerance)
        tol = 1e-10 * M
        active = np.where(conv_y >= M - tol)[0]
        
        g = subgradient(conv_y, y, active)
        # Scale gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Step size: for convex problem with subgradient method
        # eta = 2 / (t * g_norm) works well with Nesterov
        eta = 2.0 / (t * g_norm * np.sqrt(n))

        f_new = np.maximum(f_new, 1e-15)

            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, t={t:.1f}")
        else:
            no_improve += 1
        
        if no_improve > 2000:

            rng = np.random.default_rng(it)

            f_prev = f_curr.copy()
            t = 1.0
            no_improve = 0
            best_obj = fast_convolve(f_curr).max()

**Alternative: Proximal Gradient / FISTA with Moreau Envelope**

Actually, the issue is that max(conv) is not smooth, so proximal gradient doesn't directly apply. But we can use the log-sum-exp approximation to get a smooth objective, and then use FISTA (accelerated proximal gradient).

The log-sum-exp approximation: logsumexp(conv/T) ≈ max(conv) when T is small.

Its gradient is: grad_i = (1/T) * sum_j conv_weights[j] * (a[k-j] + a[j-k]) for the active/weighted positions.

Actually, the gradient of logsumexp(conv(a,a)/T) w.r.t. a[i] is:
(1/T) * sum_k w_k * d conv[k]/d a[i]
where w_k = exp(conv[k]/T) / sum exp(conv[j]/T) (softmax weights)
and d conv[k]/d a[i] = 2*a[k-i] (as computed above)

So grad[i] = (2/T) * sum_k w_k * a[k-i]

This is a weighted sum of shifted copies of a! We can compute it efficiently using FFT (cross-correlation).

Cross-correlation: (f ⋆ w)[m] = sum_j f[j] * conj(w[j+m]) = sum_j f[j] * w[j+m] (for real signals)

    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t = 1.0
    x = f.copy()  # current point
    y = f.copy()  # extrapolated point

    T_total = time_limit
    T_start = 0.5  # initial temperature
    T_end = 0.005  # final temperature
    
    it = 0
    no_improve = 0

        it += 1
        elapsed = time.monotonic() - t0

        progress = min(elapsed / T_total, 1.0)
        T = T_start * (T_end / T_start) ** progress  # exponential decay
        T = max(T, T_end)

        # For logsumexp(conv/T), L ≈ 2/T * max(conv) roughly

        L = 2.0 / T * fast_convolve(y).max() * 1.5
        L = max(L, 1.0)
        
        # Compute gradient at y
        grad, conv_y, w = smooth_grad(y, T)
