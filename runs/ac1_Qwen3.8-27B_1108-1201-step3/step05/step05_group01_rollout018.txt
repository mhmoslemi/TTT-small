def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smooth_obj_and_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in softmax t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation via FFT)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

def peak_coord_descent(f0, time_limit, verbose=False):
    """
    Peak-elimination coordinate descent.
    At each step, find the argmax of b, compute its exact subgradient,
    and do a projected gradient step targeting that peak.
    """
    f = normalize_seq(f0)
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    lr = 0.001
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    k_smooth = 3  # number of top peaks to average over

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Get top-k peaks for a smoother subgradient
        k_actual = min(k_smooth, len(conv))
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
        # Weight by how close to max
        peak_vals = conv[peak_indices]
        weights = peak_vals - conv_max + peak_vals.max() + 1e-10
        weights = np.exp(weights * 50)
        weights /= weights.sum()

        # Compute subgradient: grad[j] = 2 * sum_k w_k * a[k-j] for valid k-j
        # Using FFT: grad = 2 * cross_correlation(w_peaks, a)
        w_full = np.zeros(len(conv))
        w_full[peak_indices] = weights
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w_full, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R

        # Momentum update
        v = momentum * v + (1 - momentum) * grad

        # Projected gradient step (stay on simplex)
        f_new = project_simplex(f - lr * v)

        # Adaptive step size
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.02, 0.01)
            if verbose and no_improve % 100 == 0:
                print(f"    PCD: obj={best_obj:.10f}, lr={lr:.5f}")
        else:
            # Try smaller step
            f_try = project_simplex(f - lr * 0.3 * v)
            obj_try = fast_convolve(f_try).max()
            if obj_try < conv_max:
                f = f_try
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.95

        # Periodic restart
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.001
            no_improve = 0
            # Occasionally increase k_smooth for smoother gradient
            k_smooth = min(k_smooth + 1, 10) if no_improve == 0 else max(3, k_smooth - 1)

        if lr < 1e-6:
            lr = 0.001
            no_improve = 0
            v = np.zeros(n)

    return best_f, best_obj


def mwu_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with smooth gradient."""
    f = normalize_seq(f0)
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    beta_ls = 50.0
    eta = 0.03
    momentum = 0.75
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            beta_ls = min(beta_ls * 1.01, 500.0)
            if verbose and best_obj < 1.51:
                print(f"    MWU: obj={best_obj:.10f}")
        else:
            no_improve += 1
            eta *= 0.98
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999991)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                eta = 0.02
                no_improve = 0
                beta_ls = 30.0

        eta = max(eta, 0.001)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, kind='gaussian'):
    """Generate structured initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        sigma = 0.25
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    elif kind == 'triangle':
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == 'parabola':
        a = 1.0 - (2.0 * x - 1.0) ** 2
    elif kind == 'bell':
        a = 1.0 / (1.0 + ((x - 0.5) / 0.15) ** 4)
    else:
        a = np.ones(n)
    return normalize_seq(a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS (high beta), LP-FW,
    peak-elimination coordinate descent, and MWU with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_seq(p)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = structured_init(768, 'gaussian')

    n_parent = len(parent)

    # === Phase 1: Quick structured init search (30s) ===
    print("\n=== Phase 1: Structured inits ===")
    t1_end = time.monotonic() + 30
    for kind in ['gaussian', 'triangle', 'raised_cosine', 'parabola', 'bell']:
        if time.monotonic() > t1_end:
            break
        for n_try in [512, 640, 768, 896, 1024]:
            if time.monotonic() > t1_end:
                break
            a_init = structured_init(n_try, kind)
            score = float(evaluate_sequence(a_init.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_init.copy()
                print(f"  Structured [{kind}, n={n_try}]: {score:.6f}")

    # Use best available as base
    base = best_seq if best_seq is not None else parent

    # === Phase 2: L-BFGS high-precision (700s) ===
    print(f"\n=== Phase 2: L-BFGS high-precision (best={best_score:.6f}) ===")
    t2 = min(700, deadline - time.monotonic() - 700)
    if t2 > 60:
        a_opt, _ = lbfqg_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # Second L-BFGS run from perturbed start
        base2 = best_seq if best_seq is not None else base
        t2b = min(200, deadline - time.monotonic() - 500)
        if t2b > 30:
            noise = rng.standard_normal(len(base2)) * 0.005
            f_pert = project_simplex(base2 + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=t2b, verbose=True)
            try_seq(a_opt, "lbfqg_perturbed")

    # === Phase 3: LP-FW (400s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(400, deadline - time.monotonic() - 400)
    if t3 > 40:
        a_opt, _ = lp_fw_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "lpfw_main")

        # LP-FW from different start
        t3b = min(150, deadline - time.monotonic() - 250)
        if t3b > 30:
            a_rev = base[::-1].copy()
            a_opt, _ = lp_fw_optimize(a_rev, time_limit=t3b)
            try_seq(a_opt, "lpfw_reversed")

    # === Phase 4: Peak-elimination coordinate descent (300s) ===
    print(f"\n=== Phase 4: Peak Coord Descent (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4 = min(300, deadline - time.monotonic() - 200)
    if t4 > 30:
        a_opt, _ = peak_coord_descent(base, time_limit=t4, verbose=True)
        try_seq(a_opt, "peak_cd")

    # === Phase 5: MWU (150s) ===
    print(f"\n=== Phase 5: MWU (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t5 = min(150, deadline - time.monotonic() - 100)
    if t5 > 20:
        a_opt, _ = mwu_optimize(base, time_limit=t5, verbose=True)
        try_seq(a_opt, "mwu_main")

    # === Phase 6: Multi-length exploration (150s) ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    target_lengths = []
    for tl in [384, 512, 640, 896, 1024, 1280, 1536, 2048]:
        if tl != n_base:
            target_lengths.append(tl)

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 50
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.3, 40)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lbfqg_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}_lbfqg")

    # === Phase 7: Final deep optimization on best (remaining time) ===
    print(f"\n=== Phase 7: Final Deep Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 7a: L-BFGS with very high beta
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem * 0.4, 120)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfqg")

        # 7b: Peak CD
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.4, 80)
            a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_peak_cd")

        # 7c: LP-FW short
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

        # 7d: L-BFGS again (it keeps improving with high beta)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.6, 60)
            base2 = best_seq if best_seq is not None else base
            a_opt, _ = lbfqg_optimize(base2, time_limit=t_alloc)
            try_seq(a_opt, "final_lbfqg2")

        # 7e: Final peak CD polish
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = peak_coord_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_peak_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
