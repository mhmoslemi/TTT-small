"""
Peak-convolution minimizer using mirror descent (multiplicative weight updates),
targeted coordinate descent, and LP-based direction finding.
Exploits scale invariance to optimize on the probability simplex.
Minimizes 2*n*max(convote(a,a)) subject to sum(a)=1, a>=0.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_projection(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent_step(a, b, T, eta):
    """One mirror descent step: a[j] <- a[j]*exp(-eta*g[j]) / Z, projected to simplex."""
    n = len(a)
    # Compute smoothed gradient of logsumexp(b/T)
    scaled = b / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad[j] = (2/T) * sum_k w[k] * a[k-j] (cross-correlation)
    N = 1 << (2 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    # Mirror descent update
    log_ratio = np.log(np.maximum(a, 1e-300)) - eta * grad
    log_ratio -= log_ratio.max()
    a_new = np.exp(log_ratio)
    s = a_new.sum()
    if s < 1e-300:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new, grad, w


def mirror_descent_optimize(a0, time_limit, T_start=1.0, T_end=0.02, eta_start=1.0,

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

        a /= s
    n = len(a)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        T = max(T, T_end)
        eta = eta_start * (eta_end / eta_start) ** (progress ** 0.5)

        b = fast_conv(a)

        cur_obj = fast_conv(a).max()

    t0 = time.monotonic()
    momentum = np.zeros(n)

        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[idx[valid]]
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        # Step size
        step = 0.01 * (1.0 / (1.0 + 0.01 * elapsed))
        momentum = beta * momentum - step * grad
        a_new = a + momentum
        a_new = simplex_projection(a_new)
        cur_obj = fast_conv(a_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 200 == 0:
                print(f"      SubGrad iter {it}: obj={best_obj:.10f}")
        else:
            a = a_new  # still update to explore

    return best_a, best_obj


def targeted_cd(a0, time_limit, verbose=False):
    """Targeted coordinate descent focusing on peak-reducing moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    def eval_pair_move(a, p, q, alpha):
        """Evaluate moving mass: a[p] gets alpha*(a[p]+a[q]), a[q] gets (1-alpha)*(a[p]+a[q])."""
        w = a[p] + a[q]
        if w < 1e-15:
            return 1e10
        a_trial = a.copy()
        a_trial[p] = alpha * w
        a_trial[q] = (1.0 - alpha) * w
        return fast_conv(a_trial).max()

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        peaks = np.sort(b)[::-1]
        n_peaks = min(20, n)
        peak_positions = np.argsort(b)[::-1][:n_peaks]

        improved = False
        # For each peak, try moving mass from nearby (hot) to far (cold) positions
        for k in peak_positions[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            # Hot positions: near k (in convolution sense, j where k-j is small)
            hot_candidates = []
            for j in range(n):
                dist = abs(k - j)
                if a[j] > 1e-8:
                    hot_candidates.append((-a[j] * np.exp(-dist / (n * 0.1)), j))
            hot_candidates.sort()
            
            # Cold positions: far from all peaks
            cold_scores = np.zeros(n)
            for k2 in peak_positions[:5]:
                dists = np.abs(k2 - np.arange(n))
                cold_scores -= np.exp(-dists / (n * 0.15))
            cold_indices = np.argsort(cold_scores)[:20]

            # Try a few hot-cold pairs
            tried = set()
            for h_idx, (h_score, h) in enumerate(hot_candidates[:5]):
                if h_idx >= 5:
                    break
                for c in cold_indices[:8]:
                    if c == h or (h, c) in tried or (c, h) in tried:
                        continue
                    tried.add((h, c))
                    if a[h] < 1e-10 or a[c] < 1e-12:
                        continue
                    # Golden section search on alpha
                    w = a[h] + a[c]
                    cur_alpha = a[h] / w
                    lo, hi = 0.0, 1.0
                    ORDER = 0.6180339887
                    c1 = hi - ORDER * (hi - lo)
                    c2 = lo + ORDER * (hi - lo)
                    f1 = eval_pair_move(a, h, c, c1)
                    f2 = eval_pair_move(a, h, c, c2)
                    for _ in range(16):
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi - ORDER * (hi - lo)
                            f1 = eval_pair_move(a, h, c, c1)
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + ORDER * (hi - lo)
                            f2 = eval_pair_move(a, h, c, c2)
                    opt_alpha = (lo + hi) / 2.0
                    opt_val = eval_pair_move(a, h, c, opt_alpha)
                    if opt_val < best_obj - 1e-13:
                        a[h] = opt_alpha * w
                        a[c] = (1.0 - opt_alpha) * w
                        best_obj = opt_val
                        best_a = a.copy()
                        improved = True

        if not improved:
            # Try random pairs
            for _ in range(200):
                if time.monotonic() - t0 > time_limit:
                    break
                p = int(rng.integers(n))
                q = int(rng.integers(n))
                if p == q or a[p] < 1e-10 or a[q] < 1e-10:
                    continue
                w = a[p] + a[q]
                cur_alpha = a[p] / w
                # Try a few ratios
                best_local = fast_conv(a).max()
                best_local_alpha = cur_alpha
                for ratio in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 0.99]:
                    if abs(ratio - cur_alpha) < 0.03:
                        continue
                    val = eval_pair_move(a, p, q, ratio)
                    if val < best_local:
                        best_local = val
                        best_local_alpha = ratio
                if best_local < best_obj - 1e-13:
                    a[p] = best_local_alpha * w
                    a[q] = (1.0 - best_local_alpha) * w
                    best_obj = best_local
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def lp_direction_find(a, M):
    """Find direction g via LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    # Build constraint matrix A_ub: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_j a[j] * g[k-j] (with bounds)
    # For efficiency, only include constraints that are "tight" (near M)
    b_conv = fast_conv(a)
    tight = b_conv > M * 0.8
    n_tight = tight.sum()
    if n_tight < 5:
        tight = b_conv > M * 0.5
        n_tight = tight.sum()
    tight_idx = np.where(tight)[0]
    
    A_ub = np.zeros((n_tight, n))
    b_ub = np.full(n_tight, M)
    for idx, k in enumerate(tight_idx):
        # (a*g)[k] = sum_j a[j] * g[k-j] where 0 <= k-j < n
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[idx, j] = a[j]
    
    bounds = [(0, 1.0)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=' highs')
        if res.success:
            return res.x
    except Exception:
        pass
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive margin."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    t0 = time.monotonic()
    margin = 1.0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = fast_conv(a).max()
        g = lp_direction_find(a, M * margin)
        if g is None:
            margin *= 1.2
            if margin > 5.0:
                margin = 1.0
            continue
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            margin = min(margin * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_conv(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a_new = (1 - best_alpha) * a + best_alpha * g
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
            a = a_new
            margin = 1.0
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            margin = min(margin * 1.1, 4.0)

        if no_improve > 80:
            rng_p = np.random.default_rng(it * 13 + 5)
            noise = rng_p.standard_normal(n) * 0.003
            a = simplex_projection(best_a + noise)
            no_improve = 0
            margin = 1.0

    return best_a, best_obj


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric (rising-falling) geometric."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * 0.5])
    elif len(a) > n:
        a = a[:n]
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    """Raised cosine window."""
    x = np.linspace(0, np.pi, n)
    a = (0.5 + 0.5 * np.cos(x)) ** power + 0.01
    return a / a.sum()


def make_ramp(n, power=1.0):
    """Power-law ramp: a[i] = (i+1)^power, normalized."""
    a = (np.arange(n, dtype=np.float64) + 1) ** power
    return a / a.sum()


def make_bimodal(n, pos1=0.3, pos2=0.7, width=0.1):
    """Bimodal Gaussian mixture."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2) + 0.01
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def decimate_seq(f, new_n):
    """Decimate to new_n by averaging blocks, normalized."""
    old_n = len(f)
    if old_n <= new_n:
        return f.copy()
    blocks = np.array_split(np.arange(old_n), new_n)
    f_dec = np.array([f[b].mean() for b in blocks])
    f_dec = np.maximum(f_dec, 0)
    f_dec /= f_dec.sum()
    return f_dec


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer: mirror descent, targeted CD, LP-FW.
    Tries multiple initializations and lengths.
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_geometric(512, 0.98)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    def try_seq(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old:.6f}]")

    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Initializations ===")
    n_parent = len(parent)
    candidates = []

    # Parent
    candidates.append((parent.copy(), "parent"))

    for n_try in [n_parent, 512, 1024]:
        for r in [0.96, 0.97, 0.98, 0.99]:
            candidates.append((make_geometric(n_try, r), f"geo_n{n_try}_r{r}"))
        for r in [0.95, 0.97, 0.99]:
            candidates.append((make_symmetric_geo(n_try, r), f"sym_n{n_try}_r{r}"))
        for pw in [1.5, 2.0, 3.0, 4.0]:
            candidates.append((make_cosine(n_try, pw), f"cos_n{n_try}_p{pw}"))
        for pw in [0.5, 1.0, 2.0, 3.0]:
            candidates.append((make_ramp(n_try, pw), f"ramp_n{n_try}_p{pw}"))
        for w in [0.05, 0.1, 0.15, 0.2]:
            candidates.append((make_bimodal(n_try, 0.3, 0.7, w), f"bi_n{n_try}_w{w}"))

    for a, label in candidates:
        try_seq(a, label)

    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq)})")

    # === Phase 1: Mirror descent on best candidates ===
    print("\n=== Phase 1: Mirror Descent ===")
    # Sort candidates by initial score, try top few
    scored_candidates = []
    for a in [parent.copy(), best_seq.copy()]:
        try:
            sc = 2 * len(a) * fast_conv(a / a.sum()).max()
            scored_candidates.append((sc, a))
        except Exception:
            pass
    scored_candidates.sort()

    for sc, a_init in scored_candidates[:3]:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(rem, 200)
        print(f"  MD: n={len(a_init)}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(a_init, time_limit=t_alloc,
                                           T_start=0.8, T_end=0.01,
                                           eta_start=2.0, eta_end=0.2,
                                           verbose=True)
        try_seq(a_opt, f"md_n{len(a_opt)}")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30 and best_seq is not None:
        t_alloc = min(rem, 180)
        print(f"  LP-FW: n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw")

    # LP-FW on a perturbed version
    rem = deadline - time.monotonic() - 250
    if rem > 20 and best_seq is not None:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        a_pert = simplex_projection(best_seq + noise)
        t_alloc = min(rem, 100)
        a_opt, _ = lp_fw_optimize(a_pert, time_limit=t_alloc)
        try_seq(a_opt, "lpfw_pert")

    # === Phase 3: Targeted CD ===
    print(f"\n=== Phase 3: Targeted CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20 and best_seq is not None:
        t_alloc = min(rem, 120)
        print(f"  TCD: n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = targeted_cd(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "tcd")

    # Targeted CD on perturbed
    rem = deadline - time.monotonic() - 150
    if rem > 15 and best_seq is not None:
        noise = rng.standard_normal(len(best_seq)) * 0.003
        a_pert = simplex_projection(best_seq + noise)
        t_alloc = min(rem, 60)
        a_opt, _ = targeted_cd(a_pert, time_limit=t_alloc)
        try_seq(a_opt, "tcd_pert")

    # === Phase 4: Longer sequences ===
    print(f"\n=== Phase 4: Longer Sequences (best={best_score:.6f}) ===")
    for target_n in [1024, 1536, 2048]:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        if best_seq is not None and len(best_seq) < target_n:
            a_interp = interpolate_seq(best_seq, target_n)
        else:
            a_interp = make_geometric(target_n, 0.99)
        
        t1 = min(rem * 0.4, 80)
        a_opt, _ = mirror_descent_optimize(a_interp, time_limit=t1,
                                           T_start=0.5, T_end=0.02,
                                           eta_start=1.5, eta_end=0.3)
        try_seq(a_opt, f"md_n{target_n}")

        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t2 = min(rem * 0.5, 50)
            a_opt, _ = targeted_cd(a_opt if len(a_opt) == target_n else a_interp,
                                   time_limit=t2)
            try_seq(a_opt, f"tcd_n{target_n}")

    # === Phase 5: Shorter sequences (decimation) ===
    print(f"\n=== Phase 5: Shorter Sequences (best={best_score:.6f}) ===")
    if best_seq is not None and len(best_seq) > 512:
        for target_n in [384, 512]:
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            a_dec = decimate_seq(best_seq, target_n)
            t1 = min(rem, 40)
            a_opt, _ = mirror_descent_optimize(a_dec, time_limit=t1,
                                               T_start=0.3, T_end=0.01,
                                               eta_start=1.0, eta_end=0.2)
            try_seq(a_opt, f"md_short_n{target_n}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine mirror descent
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t1 = min(rem, 50)
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t1,
                                               T_start=0.05, T_end=0.005,
                                               eta_start=0.5, eta_end=0.1,
                                               verbose=True)
            try_seq(a_opt, "md_fine")

        # Fine targeted CD
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t2 = min(rem, 30)
            a_opt, _ = targeted_cd(best_seq, time_limit=t2)
            try_seq(a_opt, "tcd_fine")

        # Fine LP-FW
        rem = deadline - time.monotonic() - 25
        if rem > 8:
            t3 = min(rem, 25)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3)
            try_seq(a_opt, "lpfw_fine")

        # Multiple small perturbations + short MD
        for i in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * i)
            a_pert = simplex_projection(best_seq + noise)
            t4 = min(rem, 12)
            a_opt, _ = mirror_descent_optimize(a_pert, time_limit=t4,
                                               T_start=0.03, T_end=0.003,
                                               eta_start=0.3, eta_end=0.05)
            try_seq(a_opt, f"pert_md_{i}")

        # Reversed
        rem = deadline - time.monotonic() - 8
        if rem > 6:
            a_rev = best_seq[::-1].copy()
            a_opt, _ = mirror_descent_optimize(a_rev, time_limit=min(rem, 10),
                                               T_start=0.02, T_end=0.002,
                                               eta_start=0.3, eta_end=0.05)
            try_seq(a_opt, "reversed")

        # Blend with uniform
        rem = deadline - time.monotonic() - 5
        if rem > 4:
            a_blend = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
            a_blend = simplex_projection(a_blend)
            a_opt, _ = mirror_descent_optimize(a_blend, time_limit=min(rem, 8),
                                               T_start=0.02, T_end=0.002,
                                               eta_start=0.3, eta_end=0.05)
            try_seq(a_opt, "blend")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
