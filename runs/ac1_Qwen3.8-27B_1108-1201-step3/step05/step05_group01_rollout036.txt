def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_y) where dual_y are the dual variables."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        dual_y = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else np.zeros(2 * n - 1)
        return res.x, dual_y
    return None, None


def compute_smooth_obj_grad(a, beta, top_k=0):
    """Compute smooth log-sum-exp objective and gradient.
    If top_k > 0, only consider top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    size = len(conv)

    if top_k > 0 and top_k < size:
        k_actual = min(top_k, size)
        thresh = np.partition(conv, size - k_actual)[size - k_actual]
        conv_used = np.where(conv >= thresh, conv, thresh)
    else:
        conv_used = conv

    conv_max = conv.max()
    shifted = beta * (conv_used - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * R

    return obj, grad_a


def lse_objective_for_lbfgs(a_t, beta, top_k):
    """Wrapper for scipy minimize: takes t (log-parameters), returns (f, grad)."""
    n = len(a_t)

    a = np.maximum(a, 0.0)

    obj, grad_a = compute_smooth_obj_grad(a, beta, top_k)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, mode, rng):
    """Create structured initializations based on spectral considerations."""
    k = np.arange(n)
    if mode == 0:  # uniform
        return np.ones(n) / n
    elif mode == 1:  # cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * k / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 2:  # gaussian
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
        return a / a.sum()
    elif mode == 3:  # bimodal
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 8.0
        a = np.exp(-0.5 * ((k - c1) / sigma) ** 2) + np.exp(-0.5 * ((k - c2) / sigma) ** 2)
        return a / a.sum()
    elif mode == 4:  # power law
        a = 1.0 / (1.0 + k * 0.01)
        return a / a.sum()
    elif mode == 5:  # sinusoidal modulation
        m = rng.integers(1, max(2, n // 100))
        alpha = rng.uniform(0.3, 0.9)
        a = 1.0 + alpha * np.cos(2 * np.pi * m * k / n)
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-10:
            return np.ones(n) / n
        return a / ss
    elif mode == 6:  # Zadoff-Chu inspired (real part of roots)
        root = rng.integers(1, 2 * n, 1)
        root = int(root[0])
        if root % 2 == 0:
            root += 1
        a = np.abs(np.exp(2j * np.pi * root * k * (k + 1) / (2 * n)))
        a = np.ones(n)  # |exp| = 1, so use magnitude of real part
        a = np.abs(np.cos(2 * np.pi * root * k * (k + 1) / (2 * n)))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:  # random
        a = rng.random(n)
        return a / a.sum()

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Multi-initialization exploration (300s) ===
    print("\n=== Phase 1: Multi-Init Exploration ===")
    n_base = len(parent) if parent is not None else 768
    init_time = min(300, deadline - time.monotonic() - 1500)

    if init_time > 60:
        # Try different lengths and initializations
        lengths = [512, 768, 1024]
        if n_base in lengths:
            lengths.remove(n_base)
            lengths.insert(0, n_base)

        for target_n in lengths:
            rem = init_time - (time.monotonic() - (deadline - init_time))
            if rem < 30:
                break
            t_per_init = max(5, min(30, rem / 8))

            for mode in range(7):
                if time.monotonic() > deadline - 1500:
                    break
                f_init = spectral_init(target_n, mode, rng)
                t_alloc = t_per_init
                a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
                try_seq(a_opt, f"init_n{target_n}_m{mode}")

    # === Phase 2: Deep LP-FW on best (500s) ===
    print(f"\n=== Phase 2: Deep LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else (parent if parent is not None else np.ones(768) / 768)
    t2 = min(500, deadline - time.monotonic() - 1000)
    if t2 > 60:
        a_opt, _ = lp_fw_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "deep_lpfw_1")

        # Second pass from slightly perturbed
        rem = deadline - time.monotonic() - 950
        if rem > 60:
            noise = rng.standard_normal(len(base)) * 0.005
            f_pert = project_simplex(base + noise)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 200))
            try_seq(a_opt, "deep_lpfw_2")

    # === Phase 3: L-BFGS smooth optimization (400s) ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else np.ones(768) / 768
    t3 = min(400, deadline - time.monotonic() - 600)
    if t3 > 40:
        a_opt, _ = lbfqg_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # From perturbed start
        rem = t3 - (time.monotonic() - (deadline - t3))
        if rem > 30:
            noise = rng.standard_normal(len(base)) * 0.01
            f_pert = project_simplex(base + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=min(rem, 150))
            try_seq(a_opt, "lbfqg_pert")

    # === Phase 4: Subgradient optimization (300s) ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else np.ones(768) / 768
    t4 = min(300, deadline - time.monotonic() - 300)
    if t4 > 30:
        a_opt, _ = subgradient_optimize(base, time_limit=t4, verbose=True)
        try_seq(a_opt, "subgrad_main")

    # === Phase 5: Peak-shaving coordinate descent (200s) ===
    print(f"\n=== Phase 5: Peak Shaving CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        t5 = min(200, deadline - time.monotonic() - 100)
        if t5 > 20:
            a_opt, _ = peak_shaving_cd(best_seq, time_limit=t5)
            try_seq(a_opt, "peak_cd")

    # === Phase 6: Multi-length exploration (250s) ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [1024, 1280, 1536, 2048]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 80)
            print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 7: Final deep optimization (remaining time) ===
    print(f"\n=== Phase 7: Final Deep Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 7a: LP-FW
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem * 0.4, 150)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lpfw")

        # 7b: L-BFGS
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lbfqg")

        # 7c: Subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.5, 80)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_sg")

        # 7d: CD
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.6, 60)
            a_opt, _ = peak_shaving_cd(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

    # === Phase 8: Final polishing with tiny perturbations ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for attempt in range(8):
            rem = deadline - time.monotonic() - 5
            if rem < 10:
                break

            if attempt % 3 == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_lpfw_{attempt}")
            elif attempt % 3 == 1:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_lbfqg_{attempt}")
            else:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.6, 30)
                a_opt, _ = subgradient_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_sg_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
