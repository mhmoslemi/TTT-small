import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[|k-j|] * g[j] ... 
    # Actually (f*g)[k] = sum_j f[j] * g[k-j] where valid indices
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] contributes to (f*g)[k] for k in [j, j+n-1] with coefficient f[k-j]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_reduction(f0, time_limit, verbose=False):
    """
    Targeted peak reduction: iteratively finds the largest auto-convolution
    peak and applies a multiplicative update to reduce contributions to it.
    More aggressive than simple subgradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.002
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    PR iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 500:
            # Reset with perturbation
            rng = np.random.default_rng(it * 7 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.003
            no_improve = 0

        if lr < 1e-8:
            break

        # Find top peaks
        top_k = min(15, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_indices]
        
        # Weight peaks by their excess over the median
        median_val = np.median(conv)
        weights = np.maximum(peak_vals - median_val, 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(len(peak_indices)) / len(peak_indices)
        else:
            weights /= wsum

        # Compute subgradient: for each peak k, the gradient of (a*a)_k w.r.t. a_j is a[k-j]
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            if weights[i] < 1e-10:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * f[k - idx]

        # Multiplicative update with adaptive learning rate
        # Scale subgradient to have unit mean for stability
        sg_mean = sg.mean()
        if sg_mean > 1e-15:
            sg_normalized = sg / sg_mean
        else:
            sg_normalized = sg

        f = f * np.exp(-lr * sg_normalized)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        lr *= 0.9998

    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)

    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    for c, n_c, sc in candidates:
        if np.isfinite(sc) and sc < best_start_score:
            best_start_score = sc
            best_start = c

    if best_start is None:
        best_start = parent
        best_start_score = best_score

    print(f"  Best start: score={best_start_score:.8f}")

    # Phase 2: LP-FW on best start
    print(f"\n=== Phase 2: LP-FW (n={len(best_start)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.2, 300)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(best_start, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_1")

    # Phase 3: Pressure descent
    print(f"\n=== Phase 3: Pressure Descent ===")
    base = best_seq if best_seq is not None else best_start
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = pressure_descent(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "pressure_1")

    # Phase 4: LP-FW on current best
    print(f"\n=== Phase 4: LP-FW round 2 ===")
    base = best_seq if best_seq is not None else best_start
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.2, 250)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_2")

    # Phase 5: Subgradient descent
    print(f"\n=== Phase 5: Subgradient Descent ===")
    base = best_seq if best_seq is not None else best_start
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = subgradient_descent(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "sgd_1")

    # Phase 6: Pressure descent round 2
    print(f"\n=== Phase 6: Pressure Descent round 2 ===")
    base = best_seq if best_seq is not None else best_start
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = pressure_descent(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "pressure_2")

    # Phase 7: Try different lengths with optimization
    print(f"\n=== Phase 7: Multi-length optimization ===")
    base = best_seq if best_seq is not None else best_start
    for target_n in [640, 896, 1024]:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp2_n{target_n}")

        t_alloc = min(rem * 0.1, 100)
        if t_alloc > 20:
            print(f"  LP-FW n={target_n}: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"lpfw_n{target_n}")

    # Phase 8: Stochastic local search
    print(f"\n=== Phase 8: Stochastic Local Search ===")
    base = best_seq if best_seq is not None else best_start
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 120)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = stochastic_local_search(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "sls_1")

    # Phase 9: Final LP-FW squeeze with perturbations
    print(f"\n=== Phase 9: Final LP-FW with Perturbations ===")
    base = best_seq if best_seq is not None else best_start
    n_best = len(base)

    for p in range(3):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.15, 120)

        if p == 0:
            f_init = base.copy()
        elif p == 1:
            noise = rng.standard_normal(n_best) * 0.002
            f_init = project_simplex(base + noise)
        else:
            # Blend with uniform for diversity
            f_init = 0.85 * base + 0.15 * (np.ones(n_best) / n_best)
            f_init = f_init / f_init.sum()

        print(f"  Variant {p}: {t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
        try_seq(a_opt, f"final_{p}")

    # Phase 10: Final pressure descent polish
    print(f"\n=== Phase 10: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.3, 100)
            print(f"  Pressure: {t_alloc:.0f}s")
            a_opt, obj = pressure_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "polish_1")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_alloc = min(rem * 0.5, 60)
                print(f"  LP-FW final: {t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, "polish_2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
