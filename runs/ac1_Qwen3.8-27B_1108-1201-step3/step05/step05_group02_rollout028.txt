                    conv = fast_convolve(f)
                    if conv.max() < best_obj - 1e-14:
                        best_obj = conv.max()
                        best_f = f.copy()
                        improved = True

        if not improved and passes > 3:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.12f}")

    return best_f, best_obj


def multi_peak_reweight(f0, time_limit, verbose=False):
    """
    Iterative reweighting: reduce weights at positions contributing to peaks.
    f_j <- f_j / (sum_k w_k * f_{|k-j|} + eps) where w_k = (f*f)_k / max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_f = f.copy()
    best_obj = conv.max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    beta = 0.3  # mixing parameter

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0:
            beta = max(beta * 0.98, 0.05)  # anneal

        M = conv.max()
        if M < best_obj - 1e-16:
            best_obj = M
            best_f = f.copy()
            no_improve = 0

        # Compute "load" at each position: how much does f_j contribute to peaks?
        load = np.zeros(n)
        # For each k, position j contributes f_j * f[k-j] to (f*f)[k]
        # Weight by the convolution value at k
        for k in range(2 * n - 1):
            w = conv[k] / M if M > 0 else 0
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            load[idx] += w * f[k - idx]

        # Normalize load
        load_max = load.max()
        if load_max < 1e-15:
            break
        load_norm = load / load_max

        # Update: reduce positions with high load
        f_new = f / (1.0 + beta * load_norm)
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f = f_new / ss
        conv = fast_convolve(f)

        if conv.max() >= best_obj and no_improve > 200:
            rng = np.random.default_rng(it * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            no_improve = 0
            beta = 0.3
        elif conv.max() >= best_obj:
            no_improve += 1
        else:
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, M * m_relax, n, time_limit=min(30, rem * 0.3))
        if g is None:
            m_relax = min(m_relax * 1.2, 5.0)
            if m_relax > 5.0:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.03, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 4127)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def micro_search(f0, time_limit, n_trials=30):
    """Fine-grained random search with local polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        # Random perturbation with decreasing scale
        scale = 0.001 * (1.0 - trial / n_trials * 0.5)
        f_pert = project_simplex(f + rng.standard_normal(n) * scale)
        conv_pert = fast_convolve(f_pert)
        obj_pert = conv_pert.max()
        if obj_pert < best_obj - 1e-15:
            best_obj = obj_pert
            best_f = f_pert.copy()
            f = f_pert
            # Quick local polish
            for _ in range(5):
                sg, M, _ = compute_subgradient(f, fast_convolve(f))
                sg_max = sg.max()
                if sg_max < 1e-18:
                    break
                alpha = 0.1 * np.dot(sg, f) / (np.dot(sg, sg) + 1e-15) / sg_max
                f_new = project_simplex(f - alpha * sg)
                val = fast_convolve(f_new).max()
                if val < best_obj:
                    best_obj = val
                    best_f = f_new.copy()
                    f = f_new

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining fast subgradient, LP-FW,
    coordinate descent, multi-peak reweighting, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    def update_best(f, phase_name=""):
        nonlocal best_seq, best_score
        sc = float(evaluate_sequence(f.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()
            print(f"  NEW BEST [{phase_name}]: {sc:.8f} (n={len(f)})")
            return True
        return False

    # === Phase 1: Fast subgradient on parent ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: Fast SG (n={n_parent}) ===")
    rem = total_deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = fast_sg_optimize(parent, time_limit=t1, verbose=True,
                                     lp_interval=150, lp_time=8)
        update_best(a_opt, "sg_main")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: CD polish (best={best_score:.6f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 30:
        t2 = min(200, rem * 0.15)
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t2)
        update_best(a_opt, "cd")

    # === Phase 3: Multi-peak reweighting ===
    print(f"\n=== Phase 3: Multi-peak reweight (best={best_score:.6f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 30:
        t3 = min(150, rem * 0.12)
        print(f"  Budget: {t3:.0f}s")
        a_opt, _ = multi_peak_reweight(best_seq, time_limit=t3)
        update_best(a_opt, "reweight")

    # === Phase 4: LP-FW deep optimization ===
    print(f"\n=== Phase 4: LP-FW deep (best={best_score:.6f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 40:
        t4 = min(250, rem * 0.2)
        print(f"  Budget: {t4:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
        update_best(a_opt, "lpfw_deep")

    # === Phase 5: Nearby lengths ===
    print(f"\n=== Phase 5: Nearby lengths (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-64, -32, 32, 64, -128, 128, 256, -256, 512, -512]:
        nl = n_best + delta
        if 128 <= nl <= 4096 and nl != n_best:
            candidate_lengths.append(nl)
    # Avoid duplicates
    seen = set()
    unique_lengths = []
    for nl in candidate_lengths:
        if nl not in seen:
            seen.add(nl)
            unique_lengths.append(nl)

    for target_n in unique_lengths[:6]:
        rem = total_deadline - time.monotonic()
        if rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.1)
        if t_alloc < 20:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fast_sg_optimize(f_interp, time_limit=t_alloc,
                                     lp_interval=200, lp_time=5)
        if update_best(a_opt, f"len_{target_n}"):
            # Quick CD polish on the new best
            rem2 = total_deadline - time.monotonic()
            if rem2 > 20:
                a_cd, _ = coordinate_descent(a_opt, time_limit=min(40, rem2 * 0.2))
                update_best(a_cd, f"len_{target_n}_cd")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse restarts (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    inits = []

    # Perturbed versions
    for scale in [0.003, 0.01, 0.02, 0.05]:
        noise = rng.standard_normal(n_best) * scale
        inits.append(project_simplex(best_seq + noise))

    # Structural perturbations
    # Shift
    f_shift = np.roll(best_seq, n_best // 8)
    inits.append(project_simplex(np.maximum(f_shift, 0)))
    # Mirror mix
    f_mirror = 0.7 * best_seq + 0.3 * best_seq[::-1]
    inits.append(project_simplex(f_mirror))
    # Edge-concentrated
    f_edge = np.maximum(best_seq, 0) * (1 + 0.3 * np.abs(np.linspace(-1, 1, n_best)))
    inits.append(project_simplex(f_edge))
    # Center-reduced
    center_mask = np.exp(-0.5 * ((np.linspace(0, 1, n_best) - 0.5) * 4) ** 2)
    f_center = best_seq * (1 - 0.2 * center_mask)
    inits.append(project_simplex(np.maximum(f_center, 0)))

    for idx, f_init in enumerate(inits):
        rem = total_deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(60, rem * 0.08)
        if t_alloc < 10:
            continue
        print(f"  restart {idx}: t={t_alloc:.0f}s")
        a_opt, _ = fast_sg_optimize(f_init, time_limit=t_alloc,
                                     lp_interval=300, lp_time=5)
        if update_best(a_opt, f"restart{idx}"):
            # Polish
            rem2 = total_deadline - time.monotonic()
            if rem2 > 15:
                a_cd, _ = coordinate_descent(a_opt, time_limit=min(30, rem2 * 0.2))
                update_best(a_cd, f"restart{idx}_cd")

    # === Phase 7: Final coordinate descent ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 30:
        t7 = min(150, rem * 0.3)
        print(f"  Budget: {t7:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t7)
        update_best(a_opt, "final_cd")

    # === Phase 8: Micro search ===
    print(f"\n=== Phase 8: Micro search (best={best_score:.6f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 10:
        t8 = min(80, rem * 0.3)
        a_opt, _ = micro_search(best_seq, time_limit=t8, n_trials=50)
        update_best(a_opt, "micro")

    # === Phase 9: Final fast SG squeeze ===
    print(f"\n=== Phase 9: Final SG squeeze (best={best_score:.6f}) ===")
    rem = total_deadline - time.monotonic()
    if rem > 10:
        t9 = min(100, rem * 0.5)
        a_opt, _ = fast_sg_optimize(best_seq, time_limit=t9,
                                     lp_interval=500, lp_time=5)
        update_best(a_opt, "final_sg")

    # === Phase 10: Final CD ===
    rem = total_deadline - time.monotonic()
    if rem > 15:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(40, rem * 0.6))
        update_best(a_opt, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
