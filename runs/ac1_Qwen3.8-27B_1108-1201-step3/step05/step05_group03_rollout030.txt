                    break
                old_val = a[j]
                for v_try in [0.0, old_val * 0.5, old_val * 2.0, 0.01, 0.1]:
                    if v_try < 0 or v_try > 0.999:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum < 1e-12:
                        continue
                    new_a = a.copy()
                    new_a[j] = v_try
                    new_a[mask] *= (1.0 - v_try) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-15:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved2 = True
            if not improved2:
                break

    return best_a, best_score


def peak_targeting_descent(a_init, time_limit, verbose=False):
    """Descent that specifically targets the peaks of the convolution."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(a)
        max_val = conv.max()
        threshold = max_val * 0.95

        # Find peaks above threshold
        peak_indices = np.where(conv >= threshold)[0]

        if len(peak_indices) == 0:
            break

        # Compute responsibility: for each coordinate j, how much it contributes to peaks
        resp = np.zeros(n)
        for k in peak_indices:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                resp[j] += a[k - j]

        # Perturb: reduce high-responsibility coordinates, increase low-responsibility
        top_resp = np.argsort(resp)[-max(5, n // 20):]
        bot_resp = np.argsort(resp)[:max(5, n // 20)]

        for _ in range(3):
            if time.monotonic() - t0 > time_limit:
                break
            i = top_resp[np.random.randint(len(top_resp))]
            j = bot_resp[np.random.randint(len(bot_resp))]
            if a[i] < 1e-12:
                continue
            delta = a[i] * np.random.uniform(0.02, 0.1)
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            if new_a[i] < 0:
                continue
            new_a = normalize(new_a)
            sc = compute_score(new_a)
            if sc < best_score - 1e-15:
                a = new_a
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n)
    return normalize(a)


def cosine_sequence(n):
    """Raised cosine shaped sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    """Gaussian-shaped sequence centered."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def raised_cosine_window(n, taper=0.2):
    """Raised cosine with taper on edges."""
    a = np.ones(n)
    m = int(n * taper)
    if m > 0:
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / m))
            a[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * i / m))
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining alternating LP, coordinate descent,
    peak-targeting descent, and multi-initialization exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []

    # From parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # Heuristic sequences at n=768
    n_base = 768
    for r in [0.9985, 0.999, 0.9995, 0.9997]:
        starts.append((geometric_sequence(n_base, r), f"geom_{r}"))
    starts.append((cosine_sequence(n_base), "cosine"))
    for sf in [0.25, 0.35, 0.45]:
        starts.append((gaussian_sequence(n_base, sf), f"gauss_{sf}"))
    for tap in [0.1, 0.2, 0.3]:
        starts.append((raised_cosine_window(n_base, tap), f"rcw_{tap}"))

    # Evaluate all starts quickly
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:5]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Alternating LP from best starts ===
    print(f"\n=== Phase 1: Alternating LP (best={best_score:.8f}) ===")

    # Select diverse starts for LP
    lp_starts = []
    used_types = set()
    for sc, a, label in start_scores:
        base = label.split('_')[0] if '_' in label else label
        if base not in used_types:
            lp_starts.append((a, label))
            used_types.add(base)
        if len(lp_starts) >= 4:
            break

    for a_init, label in lp_starts:
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem * 0.7, 180)
        print(f"  AltLP [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = alternating_lp(a_init, max_iter=30, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_lp")

    # === Phase 2: Coordinate descent on best ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_cd = min(rem * 0.6, 400)
            print(f"  CD: t={t_cd:.0f}s")
            a_opt, sc_opt = coord_descent(best_a, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd_best")

    # === Phase 3: Peak targeting ===
    print(f"\n=== Phase 3: Peak Targeting (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_pt = min(rem * 0.5, 150)
            print(f"  PT: t={t_pt:.0f}s")
            a_opt, sc_opt = peak_targeting_descent(best_a, time_limit=t_pt)
            try_seq(a_opt, "peak_target")

    # === Phase 4: Try other sequence lengths ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        for target_n in [640, 896, 1024, 1280]:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 120)
            a_interp = interpolate_sequence(best_a, target_n)
            # Quick CD on interpolated
            a_opt, sc = coord_descent(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim_{target_n}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_final = min(rem, 200)
            print(f"  Final CD: t={t_final:.0f}s")
            a_opt, sc_opt = coord_descent(best_a, time_limit=t_final, verbose=True)
            try_seq(a_opt, "final_cd")

        # One more peak targeting
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, sc_opt = peak_targeting_descent(best_a, time_limit=min(rem, 60))
            try_seq(a_opt, "final_pt")

    # === Final verification ===
    if best_a is None:
        best_a = cosine_sequence(n_base)
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
        if np.isfinite(final_score):
            best_score = final_score
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
