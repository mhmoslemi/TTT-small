"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute the objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        A_ub[start:end, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'feasibility_tolerance': 1e-9, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_averaging=True):
    """
    LP-based Frank-Wolfe with averaging and adaptive M.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # FW averaging
    f_avg = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP
        g = solve_lp_direction(f_avg, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                # Escape: structured perturbation
                rng = np.random.default_rng(it * 7 + 3)
                # Add a Fourier mode perturbation
                freq = rng.integers(2, max(3, n // 4))
                phase = rng.uniform(0, 2 * np.pi)
                pert = np.cos(2 * np.pi * freq * np.arange(n) / n + phase) * 0.005
                f_avg = project_simplex(best_f + pert)
                conv = fast_convolve(f_avg)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum

        # Check if we're stuck
        if np.max(np.abs(g - f_avg)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.1, 4.0)
            if no_improve > 80:
                # Escape with structured perturbation
                rng = np.random.default_rng(it * 13 + 5)
                freq = rng.integers(1, max(2, n // 3))
                phase = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.002, 0.01)
                pert = np.cos(2 * np.pi * freq * np.arange(n) / n + phase) * amp
                f_avg = project_simplex(best_f + pert)
                conv = fast_convolve(f_avg)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if np.max(np.abs(g - f_avg)) > 1e-5:
                no_improve = 0
            else:
                no_improve += 1

        # Line search
        best_alpha = 0.0
        best_val = fast_convolve(f_avg).max()

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            ft = f_avg + alpha * (g - f_avg)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine line search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f_avg + mid * (g - f_avg)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f_avg + best_alpha * (g - f_avg)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M_new = fast_convolve(f_new).max()
            f_avg = f_new
            m_exp = 1.0

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f_avg.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 150:
            break

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """
    Subgradient method with Polyak step size for minimizing max_k (a*a)[k]
    subject to a >= 0, sum(a) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Lower bound on optimal max(conv): at least 1/(2n-1)
    lb = 1.0 / (2 * n - 1)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find active set (peaks)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient as average over active peaks
        sg = np.zeros(n)
        for k in active:
            k = int(k)
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                lj = k - j  # the "other" index
                if 0 <= lj < n:
                    if j == lj:
                        sg[j] += 0.5 * f[j]
                    else:
                        sg[j] += f[lj]
        sg /= len(active)

        # Polyak step size
        gap = M - lb
        sg_norm_sq = np.dot(sg, sg)
        if sg_norm_sq < 1e-30:
            break
        eta_polyak = gap / (sg_norm_sq * n)  # scale by n for stability
        eta = max(0.001, min(0.05, eta_polyak))

        # Update: mirror descent (multiplicative) for non-negativity
        f_new = f * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        new_obj = fast_convolve(f_new).max()
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try a smaller step
            for try_eta in [eta * 0.5, eta * 0.1, eta * 0.01]:
                f_try = f * np.exp(-try_eta * sg)
                f_try = np.maximum(f_try, 1e-15)
                f_try = f_try / f_try.sum()
                try_obj = fast_convolve(f_try).max()
                if try_obj < best_obj - 1e-15:
                    best_obj = try_obj
                    best_f = f_try.copy()
                    f_new = f_try
                    no_improve = 0
                    break

        f = f_new

        if no_improve > 200:
            # Escape
            rng = np.random.default_rng(it * 3 + 1)
            freq = rng.integers(1, max(2, n // 3))
            pert = np.cos(2 * np.pi * freq * np.arange(n) / n) * 0.005
            f = project_simplex(best_f + pert)
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    stuck_epochs = 0

    while time.monotonic() - t0 < time_limit and stuck_epochs < 3:
        improved = False
        conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = conv.max()
            best_j_val = old_val

            # Try a range of values
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True

        if not improved:
            stuck_epochs += 1
        else:
            stuck_epochs = 0

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = (1.0 - t ** 4)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tukey(n, r=0.3):
    """Tukey (flat-top) window."""
    x = np.arange(n) / (n - 1) * 2 - 1  # [-1, 1]
    a = np.ones(n)
    mask = np.abs(x) > (1 - r)
    a[mask] = 0.5 * (1 + np.cos(np.pi * (np.abs(x[mask]) - (1 - r)) / r))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe
with FW averaging, adaptive M, structured perturbations, and subgradient
polishing. Focuses on sequence length 768 with occasional exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp_direction(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                rng = np.random.default_rng(it * 7 + 3)
                freq = rng.integers(2, max(3, n // 4))
                phase = rng.uniform(0, 2 * np.pi)
                pert = np.cos(2 * np.pi * freq * np.arange(n) / n + phase) * 0.005
                f = project_simplex(best_f + pert)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.1, 4.0)
            if no_improve > 80:

        else:

            ft = f + alpha * (g - f)

            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()

def subgradient_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    lb = 1.0 / (2 * n - 1)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        sg = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                lj = k - j
                if j == lj:
                    sg[j] += 0.5 * f[j]
                else:
                    sg[j] += f[lj]
        sg /= len(active)

        gap = M - lb
        sg_norm_sq = np.dot(sg, sg)
        if sg_norm_sq < 1e-30:
            break
        eta = max(0.001, min(0.05, gap / (sg_norm_sq * n)))

        f_new = f * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        new_obj = fast_convolve(f_new).max()
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            for try_eta in [eta * 0.5, eta * 0.1]:
                f_try = f * np.exp(-try_eta * sg)
                f_try = np.maximum(f_try, 1e-15)
                f_try = f_try / f_try.sum()
                try_obj = fast_convolve(f_try).max()
                if try_obj < best_obj - 1e-15:
                    best_obj = try_obj
                    best_f = f_try.copy()
                    f_new = f_try
                    no_improve = 0
                    break

        f = f_new

        if no_improve > 200:
            rng = np.random.default_rng(it * 3 + 1)
            freq = rng.integers(1, max(2, n // 3))
            pert = np.cos(2 * np.pi * freq * np.arange(n) / n) * 0.005
            f = project_simplex(best_f + pert)
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    stuck_epochs = 0

    while time.monotonic() - t0 < time_limit and stuck_epochs < 3:
        improved = False
        conv = fast_convolve(f)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = conv.max()
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True

        if not improved:
            stuck_epochs += 1
        else:
            stuck_epochs = 0

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):

    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
