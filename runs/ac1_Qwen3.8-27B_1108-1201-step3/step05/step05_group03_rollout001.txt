"""
Hybrid LP-based mirror descent + targeted coordinate descent + multi-resolution
coarse-to-fine search for minimizing 2*n*max(auto_conv)/(sum)^2.
Key innovations: exact LP-based direction finding (best response), multi-resolution
coarse-to-fine approach to escape local minima, targeted coordinate updates focused
on convolution peaks, and diverse asymmetric initializations.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def interp_upsample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return normalize(np.interp(new_x, old_x, a))


def geometric_sequence(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def two_sided_geometric(n, r, center_frac=0.5):
    c = center_frac * (n - 1)
    x = np.arange(n) - c
    a = r ** np.abs(x)
    return normalize(a)


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    return normalize(np.sin(x) ** 2)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    return normalize(np.exp(-x ** 2 / (2 * sigma ** 2)))


def triangular_sequence(n):
    x = np.linspace(0, 1, n)
    return normalize(1.0 - np.abs(2.0 * x - 1.0))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_hi = min(j + n - 1, 2 * n - 2)
        A[j:k_hi + 1, j] = a[0:k_hi - j + 1]
    return A


def solve_lp_direction(a, M, use_sparse=False):
    n = len(a)
    A_mat = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def targeted_coord_descent(a, time_limit, top_k=50, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        conv = fast_convolve(a)
        n_conv = len(conv)
        k_actual = min(top_k, n_conv)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]

            def obj_v(v):
                v = max(0.0, v)
                new_a = a.copy()
                new_a[j] = v
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= (1.0 - v) / other_sum
                cv = fast_convolve(new_a)
                return cv.max()

            lo = max(0.0, old_val * 0.2)
            hi = min(0.99, old_val * 4.0 + 0.01)
            if hi - lo < 1e-8:
                lo = max(0.0, old_val - 0.05)
                hi = min(0.99, old_val + 0.05)

            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 40})
                if res.success and res.fun < obj_v(old_val) - 1e-15:
                    v_new = max(0.0, res.x)
                    new_a = a.copy()
                    new_a[j] = v_new
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-15:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if not improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
            order = np.random.permutation(n)
        else:
            no_improve_count = 0

        if verbose and pass_count % 5 == 0:
            print(f"      TCD pass {pass_count}: score={best_score:.10f}")
        if pass_count > 500:
            break

    return best_a, best_score


def lp_mirror_descent(a, time_limit, lp_every=20, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    t_lr = 0.05
    t_decay = 0.999

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        s = a.sum()
        score = 2.0 * n * M / (s * s)

        if it % lp_every == 0:
            b = solve_lp_direction(a, M * 1.001)
            if b is not None and b.sum() > 1e-10:
                b = b / b.sum()
                t_new = t_lr
                new_a = (1.0 - t_new) * a + t_new * b
                new_a = normalize(new_a)
                sc_new = compute_score(new_a)
                if sc_new < best_score:
                    a = new_a
                    best_score = sc_new
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP it {it}: score={best_score:.10f}, b_sum={b.sum():.4f}")
                else:
                    t_lr *= 0.8
                t_lr = min(t_lr, 0.5)
            else:
                t_lr = max(t_lr * 0.7, 0.005)
        else:
            # Fast subgradient step
            k_actual = min(30, len(conv))
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            pv = conv[peaks]
            temp = max(1e-6, pv.max() - pv.min()) * 0.5
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()

            sg = np.zeros(n)
            for i in range(k_actual):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]

            a = np.maximum(a * np.exp(-t_lr * 0.1 * sg), 1e-15)
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        t_lr *= t_decay

    return best_a, best_score


def aggressive_local_search(a, time_limit, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:

        mode = it % 3

            # Pair transfer with golden section

            def obj_transfer(t):

                new_a = a.copy()
                new_a[j] = a_j - t
                new_a[i] = a_i + t
                new_a = normalize(new_a)
                cv = fast_convolve(new_a)
                return cv.max()

                                      options={'xatol': 1e-8, 'maxiter': 25})

                    new_a = a.copy()
                    new_a[j] = a_j - t_val
                    new_a[i] = a_i + t_val
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-15:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
            except Exception:
                pass

        elif mode == 1:
            # Single coordinate golden section
            j = rng.integers(0, n)
            old_val = a[j]

                continue

            def obj_coord(v):

                new_a = a.copy()
                new_a[j] = v
                other = 1.0 - old_val

                new_a = normalize(new_a)
                cv = fast_convolve(new_a)
                return cv.max()

            try:

                                      options={'xatol': 1e-8, 'maxiter': 20})

                    v_new = max(0.0, res.x)
                    new_a = a.copy()
                    new_a[j] = v_new
                    other = 1.0 - old_val
                    if other > 1e-12:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        new_a[mask] *= (1.0 - v_new) / other
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-15:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
            except Exception:
                pass

        else:
            # Block coordinate: optimize 2 adjacent coordinates
            j = rng.integers(0, n - 1)
            j2 = j + 1
            vals = [(a[j], a[j2])]

            def obj_block(v1, v2):
                v1 = max(0.0, v1)
                v2 = max(0.0, v2)
                new_a = a.copy()
                new_a[j] = v1
                new_a[j2] = v2
                old_sum = a[j] + a[j2]
                other = 1.0 - old_sum
                if other < 1e-12:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                mask[j2] = False
                new_a[mask] *= (1.0 - v1 - v2) / other
                if (new_a < -1e-10).any():
                    return float('inf')
                new_a = np.maximum(new_a, 0.0)
                new_a = normalize(new_a)
                cv = fast_convolve(new_a)
                return cv.max()

            # Try perturbations around current values
            base_v1, base_v2 = a[j], a[j2]
            for f1 in [0.5, 0.7, 1.0, 1.3, 1.5]:

                    sc_t = obj_block(v1_t, v2_t)
                    if sc_t < obj_block(base_v1, base_v2) - 1e-14:
                        new_a = a.copy()
                        new_a[j] = v1_t
                        new_a[j2] = v2_t
                        old_sum = a[j] + a[j2]
                        other = 1.0 - old_sum
                        if other > 1e-12:
                            mask = np.ones(n, dtype=bool)
                            mask[j] = False

                        new_a = np.maximum(new_a, 0.0)

def optimize_full(a, time_limit, use_lp=True, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Stage 1: LP-based mirror descent (if enabled and enough time)
    if use_lp and n <= 1024:
        t1 = min(time_limit * 0.35, 180)
        if time_limit > t1 + 30:
            a1, sc1 = lp_mirror_descent(a, time_limit=t1, lp_every=max(10, n // 40),
                                        verbose=verbose)
            if sc1 < best_score:
                best_score = sc1
                best_a = a1.copy()
                a = a1.copy()
            if verbose:
                print(f"      After LP-MD: {best_score:.10f}")

    # Stage 2: Targeted coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.35, 150)
    if rem > t2 + 20:
        a2, sc2 = targeted_coord_descent(a, time_limit=t2, top_k=max(20, n // 8),
                                         verbose=verbose)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            a = a2.copy()
        if verbose:
            print(f"      After TCD: {best_score:.10f}")

    # Stage 3: Aggressive local search
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.3, 100)
    if rem > t3 + 10:
        a3, sc3 = aggressive_local_search(a, time_limit=t3, verbose=verbose)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            a = a3.copy()
        if verbose:
            print(f"      After AGS: {best_score:.10f}")

    # Stage 4: Final TCD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 15:
        a4, sc4 = targeted_coord_descent(a, time_limit=min(rem, 80),
                                         top_k=max(30, n // 6))
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        if verbose:
            print(f"      After final TCD: {best_score:.10f}")

    return best_a, best_score
