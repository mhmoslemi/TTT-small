def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            if m_min <= m_max:
                vals = f[k - m_max:k - m_min + 1][::-1]
                sg[m_min:m_max + 1] += 2.0 * w * vals
        else:
            # k >= n: k-m in [0, n) means m in [k-n+1, k], intersect [0, n-1]
            m_min = max(0, k - n + 1)
            m_max = min(n - 1, k)
            if m_min <= m_max:
                vals = f[k - m_max:k - m_min + 1][::-1]
                sg[m_min:m_max + 1] += 2.0 * w * vals
    return sg


def compute_subgradient_fast(f, conv, temp, top_k):
    """Compute subgradient using soft-max weights on top-k convolution values."""
    n = len(f)
    # Select top-k
    if top_k is not None and top_k < len(conv):
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(conv))
    
    # Soft-max weights
    vals = conv[top_idx] / temp
    vals -= vals.max()
    w = np.exp(vals)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(top_idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    
    return compute_subgradient(f, top_idx, w)


def make_sqrt_inv_one_minus_z(n):
    """Generate coefficients of 1/sqrt(1-z): binom(2k,k)/4^k."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Symmetric initialization."""
    x = np.linspace(0, 1, n)
    if shape == 'bump':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_smooth_random(n, rng, n_comp=5):
    """Smooth random init as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_comp):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def adam_optimize(f0, time_limit, lr=0.004, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.006, temp_decay=0.9994,
                  top_k=None, restart_interval=300, verbose=False):
    """Adam optimizer with simplex projection and adaptive subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    cur_lr = lr

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        # Adaptive top_k: ensure we don't just use the center
        effective_top_k = top_k
        if effective_top_k is not None and effective_top_k < 3:
            effective_top_k = 3
        
        sg = compute_subgradient_fast(f, conv, temp, effective_top_k)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-8)
        cur_lr = max(lr * (0.99995 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > restart_interval:
            rng = np.random.default_rng(t * 31 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve // 100, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.8, 0.002)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.015, lr_decay=0.9995,
                            temp_init=0.006, temp_decay=0.9994,
                            top_k=None, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        effective_top_k = top_k
        if effective_top_k is not None and effective_top_k < 3:
            effective_top_k = 3
            
        sg = compute_subgradient_fast(f, conv, temp, effective_top_k)

        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-8)
        cur_lr = max(cur_lr * lr_decay, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.4, 1.7, 2.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break
        if passes > 50:
            break

    return best_f, best_obj


def edge_trim(f, time_limit=5.0):
    """Try to reduce edge peaks by trimming a[0] and a[n-1]."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Try reducing a[0] and a[n-1]
    for _ in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Try reducing a[0]
        for frac in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6, 0.5]:
            f_trial = f.copy()
            old = f_trial[0]
            f_trial[0] = old * frac
            rest = f_trial[1:].sum()
            if rest > 1e-15:
                f_trial[1:] *= (1.0 - f_trial[0]) / rest
                f_trial /= f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_trial.copy()
                f = f_trial.copy()
                break
        
        # Try reducing a[n-1]
        for frac in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6, 0.5]:
            f_trial = f.copy()
            old = f_trial[n - 1]
            f_trial[n - 1] = old * frac
            rest = f_trial[:-1].sum()
            if rest > 1e-15:
                f_trial[:-1] *= (1.0 - f_trial[n - 1]) / rest
                f_trial /= f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_trial.copy()
                f = f_trial.copy()
                break
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization leveraging 1/sqrt(1-z) structure,
    adaptive subgradients, and diverse optimization methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Principled initializations ===
    print(f"\n=== Phase 1: Principled initializations ===")
    t1_end = time.monotonic() + 200

    # Generate 1/sqrt(1-z) based inits at various lengths
    init_configs = []
    
    if parent is not None:
        init_configs.append(('parent', n_parent, parent.copy()))
    
    # 1/sqrt(1-z) at parent length
    n_try = n_parent if parent is not None else 768
    sqrt_init = make_sqrt_inv_one_minus_z(n_try)
    init_configs.append(('sqrt_inv', n_try, sqrt_init))
    
    # Reversed
    init_configs.append(('sqrt_rev', n_try, sqrt_init[::-1].copy()))
    
    # Mixed (average of forward and reversed)
    mixed = (sqrt_init + sqrt_init[::-1]) / 2
    mixed = np.maximum(mixed, 1e-12)
    init_configs.append(('sqrt_mix', n_try, mixed / mixed.sum()))
    
    # Gaussian
    init_configs.append(('gauss', n_try, make_gaussian_init(n_try, 0.5, 0.2)))
    
    # Bump
    init_configs.append(('bump', n_try, make_symmetric_init(n_try, 'bump')))
    
    # Smooth random
    init_configs.append(('srnd', n_try, make_smooth_random(n_try, rng, 5)))
    
    # Windowed sqrt_inv (taper at edges)
    windowed = sqrt_init.copy()
    taper = 50
    for i in range(min(taper, n_try // 4)):
        frac = (i + 1) / (taper + 1)
        windowed[i] *= frac
        windowed[n_try - 1 - i] *= frac
    windowed = np.maximum(windowed, 1e-12)
    init_configs.append(('sqrt_win', n_try, windowed / windowed.sum()))
    
    # Try each initialization with quick Adam
    p1_best = None
    p1_best_obj = float('inf')
    
    for name, n_init, init in init_configs:
        rem = t1_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.4, 20)
        
        a_opt, obj = adam_optimize(init, time_limit=t_alloc,
                                   lr=0.005, temp_init=0.005,
                                   temp_decay=0.9993, top_k=max(10, n_init // 15))
        try_seq(a_opt, f"p1_{name}")
        if obj < p1_best_obj:
            p1_best_obj = obj
            p1_best = a_opt.copy()

    if p1_best is not None:
        try_seq(p1_best, "p1_best")
    
    base = best_seq if best_seq is not None else parent

    # === Phase 2: Deep optimization with multiple methods ===
    print(f"\n=== Phase 2: Deep optimization (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + 500
    n_base = len(base)

    # Multiple Adam configs
    adam_configs = [
        {'lr': 0.004, 'temp_init': 0.004, 'temp_decay': 0.9995, 'top_k': max(15, n_base // 12)},
        {'lr': 0.006, 'temp_init': 0.006, 'temp_decay': 0.9993, 'top_k': max(10, n_base // 20)},
        {'lr': 0.003, 'temp_init': 0.008, 'temp_decay': 0.9996, 'top_k': max(20, n_base // 10)},
        {'lr': 0.005, 'temp_init': 0.003, 'temp_decay': 0.9994, 'top_k': None},

    for ci, cfg in enumerate(adam_configs):
        rem = t2_end - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.5, 90)
        
        print(f"  Adam cfg {ci} (lr={cfg['lr']}), t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(base, time_limit=t_alloc, **cfg)
        try_seq(a_opt, f"p2_adam{ci}")
        
        # Mirror descent
        rem = t2_end - time.monotonic()
        if rem < 25:
            break
        t_md = min(rem * 0.4, 60)
        a_opt, _ = mirror_descent_optimize(base, time_limit=t_md,
                                           lr_init=cfg['lr'] * 1.5,
                                           temp_init=cfg['temp_init'],
                                           temp_decay=cfg['temp_decay'],
                                           top_k=cfg['top_k'])
        try_seq(a_opt, f"p2_md{ci}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 350

    # Try different lengths, focusing on those near current best
    candidate_lengths = [512, 640, 768, 896, 1024, 1280, 1536]

    candidate_lengths.sort(key=lambda x: abs(x - n_base))
    candidate_lengths = candidate_lengths[:5]

    for target_n in candidate_lengths:
        rem = t3_end - time.monotonic()
        if rem < 50:
            break
        
        # Multiple initializations at this length
        inits = [
            interpolate_sequence_fn(base, target_n),
            make_sqrt_inv_one_minus_z(target_n),
            make_gaussian_init(target_n, 0.5, 0.2),
        ]
        
        t_per = max(10, (rem * 0.6) / len(inits))
        
        for ii, init in enumerate(inits):
            rem = t3_end - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(t_per, 50)
            a_opt, _ = adam_optimize(init, time_limit=t_alloc,
                                     lr=0.005, temp_init=0.005,
                                     temp_decay=0.9994, top_k=max(10, target_n // 15))
            try_seq(a_opt, f"p3_n{target_n}_{ii}")
            
            # Edge trim
            a_opt, _ = edge_trim(a_opt, time_limit=min(5, (t3_end - time.monotonic()) * 0.1))
            try_seq(a_opt, f"p3_n{target_n}_{ii}_trim")

    # === Phase 4: Symmetry breaking and refined search ===
    print(f"\n=== Phase 4: Symmetry breaking (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 250
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Try asymmetric perturbations
        for si in range(4):
            rem = t4_end - time.monotonic()
            if rem < 30:
                break
            
            # Asymmetric perturbation
            noise = rng.standard_normal(n_best) * 0.008
            # Make it more asymmetric
            trend = np.linspace(-0.01, 0.01, n_best)
            f_pert = project_simplex(best_seq + noise + trend)
            
            t_alloc = min(rem * 0.5, 50)
            a_opt, _ = adam_optimize(f_pert, time_limit=t_alloc,
                                     lr=0.004, temp_init=0.004,
                                     temp_decay=0.9995, top_k=max(10, n_best // 15))
            try_seq(a_opt, f"p4_asym{si}")
            
            # Mirror + perturb
            f_mirror = best_seq[::-1].copy()
            f_mirror = project_simplex(f_mirror + rng.standard_normal(n_best) * 0.005)
            rem = t4_end - time.monotonic()
            if rem > 20:
                a_opt, _ = adam_optimize(f_mirror, time_limit=min(rem * 0.4, 40),
                                         lr=0.004, temp_init=0.004,
                                         temp_decay=0.9995, top_k=max(10, n_best // 15))
                try_seq(a_opt, f"p4_mir{si}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 200
    if best_seq is not None:
        rem = t5_end - time.monotonic()

                                         lr=0.002, temp_init=0.003,

    # Fallback
    if best_seq is None:

    return [float(x) for x in best_seq]
