def fast_convolve(a):

    n = len(a)

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Numerically stable softmax."""
    b_max = b.max()
    e = np.exp(b - b_max)
    return e / e.sum()


def log_softmax_stable(b):
    """Numerically stable log-softmax."""
    b_max = b.max()
    return b - b_max - np.log(np.exp(b - b_max).sum())

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax parameterization
with mirror-gradient, adaptive temperature, coordinate descent with
incremental updates, and multi-phase local search.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    # Clean up tiny negatives from FFT
    out[np.abs(out) < 1e-15] = 0.0
    return out


def softmax_stable(b):
    """Numerically stable softmax."""
    bm = b - b.max()
    e = np.exp(bm)
    return e / e.sum()


def compute_mirror_grad(a, tau, conv, top_k):
    """
    Compute gradient of soft-max(conv) w.r.t. log-parameters b.
    grad_b[j] = 2 * a[j] * (weighted_mirror[j] - weighted_max_conv)
    """
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    weighted_conv = (w * pv).sum()
    mirror = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        mirror[idx] += w[i] * a[k - idx]

    grad_b = 2.0 * a * (mirror - weighted_conv)
    return grad_b

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent with rescaling to maintain sum=1.
    Tries multiple multiplier values per coordinate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    # Focus on non-zero coordinates
    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        support = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved_any = False

        for j in support:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                else:
                    scale = 1.0
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_any = True

        if not improved_any and pass_count > 1:
            break

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def raised_cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    return a / a.sum()


def tent_init(n):
    """Triangular (tent) shape."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def gaussian_init(n, sigma=0.3):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def pair_swap_search(a0, time_limit, seed=0):

    a = np.maximum(a0.astype(np.float64), 0.0)

    n_ref = len(best_seq) if best_seq is not None else 768
    inits = [
        ("flat_spec", flat_spectrum_init(n_ref, rng)),
        ("raised_cos", raised_cosine_init(n_ref)),
        ("tent", tent_init(n_ref)),
        ("gauss", gaussian_init(n_ref, 0.25)),
    ]

    for name, init in inits:
        rem = deadline - time.monotonic() - 30
        if rem < 20:
            break
        t_sm = min(rem * 0.3, 60)
        print(f"  {name}, t={t_sm:.0f}s")
        a_opt, obj = optimize_softmax(init, time_limit=t_sm, tau_init=20.0)
        try_seq(a_opt, f"div_{name}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_cd_final = min(rem * 0.6, 60)
            print(f"  Final CD, t={t_cd_final:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd_final)
            try_seq(a_opt, "final_cd")

        # Pair swap
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_ps = min(rem * 0.5, 30)
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "final_ps")

        # One more tiny CD
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
