    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def smooth_obj_and_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[i] = 2 * cross_corr(a, w)[i]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    # grad_t = a * (grad_a - <a, grad_a>)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def peak_targeted_grad(a, top_k=20, temperature=0.001):
    """Compute subgradient targeting the top-k peaks of the convolution."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg, conv.max()


def adam_optimize(f0, time_limit, beta_schedule=None, lr_max=0.005, lr_min=0.0001,
                  beta1=0.9, beta2=0.999, eps=1e-8, restart_interval=100,
                  top_k_polish=30, polish_every=50, verbose=False, label=""):
    """Adam optimizer with annealed beta and cosine LR schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    if beta_schedule is None:
        beta_schedule = [10.0, 30.0, 80.0, 200.0, 500.0]

    t0 = time.monotonic()
    total_time = time_limit
    best_t = np.log(f + 1e-15)

    for bi, beta in enumerate(beta_schedule):
        remaining = total_time - (time.monotonic() - t0)
        if remaining < 20:
            break
        frac = (bi + 1) / len(beta_schedule)
        sub_time = remaining * frac  # spend proportionally more time on later (higher beta) stages
        sub_time = min(sub_time, remaining)

        if verbose:
            print(f"    [{label}] Adam beta={beta:.0f}, time={sub_time:.0f}s")

        m = np.zeros(n)
        v = np.zeros(n)
        step = 0
        t_cur = best_t.copy()

        while time.monotonic() - t0 < t0 + sub_time:
            step += 1
            obj, grad = smooth_obj_and_grad(t_cur, beta)

            # Peak-targeted polishing every polish_every steps
            if polish_every > 0 and step % polish_every == 0:
                a_cur = softmax_fn(t_cur)
                sg, _ = peak_targeted_grad(a_cur, top_k=top_k_polish, temperature=0.0005)
                sg_t = a_cur * (sg - np.dot(a_cur, sg))
                grad = 0.5 * grad + 0.5 * sg_t

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)

            # Cosine LR with warm restarts
            if step % restart_interval == 0:
                m = np.zeros(n)
                v = np.zeros(n)
                step_in_restart = 0
            else:
                step_in_restart = step % restart_interval

            progress = step_in_restart / float(restart_interval)
            lr = lr_min + 0.5 * (lr_max - lr_min) * (1.0 + np.cos(np.pi * progress))

            t_cur = t_cur - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Periodic check
            if step % 10 == 0:
                a_check = softmax_fn(t_cur)
                a_check = np.maximum(a_check, 0.0)
                sc = a_check.sum()
                if sc > 1e-15:
                    a_check = a_check / sc
                obj_check = fast_convolve(a_check).max()
                if obj_check < best_obj - 1e-12:
                    best_obj = obj_check
                    best_f = a_check.copy()
                    best_t = t_cur.copy()
                    if verbose and step % 100 == 0:
                        print(f"      [{label}] step={step}, obj={best_obj:.10f}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit=60, verbose=False):
    """LP-based Frank-Wolfe with adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 5})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def sg_polish_optimize(f0, time_limit=30, lr=0.003, mom=0.93, top_k=40, temp=0.0003):
    """Subgradient polishing with momentum and peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        sg, _ = peak_targeted_grad(f, top_k=top_k, temperature=cur_temp)
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = lr * 0.7
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def palindromic_project(a):
    """Project to palindromic simplex: a[j] = a[n-1-j], sum=1, a>=0."""
    n = len(a)
    # Average with reverse
    a_pal = 0.5 * (a + a[::-1])
    # Project to simplex preserving palindrome structure
    # Effective dimension is ceil(n/2)
    if n % 2 == 0:
        half = a_pal[:n // 2]
        h_proj = project_simplex(half * 2.0) * 0.5
        a_pal = np.concatenate([h_proj, h_proj[::-1]])
    else:
        mid = n // 2
        half = a_pal[:mid]
        mid_val = a_pal[mid]
        # Project (half, mid_val) to simplex with proper weighting
        effective = np.concatenate([half * 2.0, [mid_val]])
        e_proj = project_simplex(effective)
        half_proj = e_proj[:mid] * 0.5
        mid_proj = e_proj[mid]
        a_pal = np.concatenate([half_proj, [mid_proj], half_proj[::-1]])

    a_pal = np.maximum(a_pal, 0)
    ss = a_pal.sum()
    if ss > 1e-15:
        a_pal /= ss
    return a_pal


def gaussian_init(n, sigma_frac=0.3, seed=0):
    """Create a Gaussian-shaped initial sequence."""
    rng = np.random.default_rng(seed)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def cosinit_init(n, k=1, seed=0):
    """Create a cosine-bump shaped initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, np.pi, n)
    a = np.cos(k * (x - np.pi / 2.0)) ** 2
    a = np.maximum(a, 0)
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick multi-length, multi-init exploration
    Phase 2: Deep Adam optimization on best candidates
    Phase 3: LP-FW auxiliary
    Phase 4: Palindromic track
    Phase 5: Multi-restart
    Phase 6: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(640) / 640
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent)
    print(f"Starting best: {best_score:.6f} (n={n_parent})")

    # === Phase 1: Quick exploration (120s) ===
    print("\n=== Phase 1: Quick exploration ===")
    t_phase1 = 120
    t1_end = time.monotonic() + t_phase1

    target_lengths = [512, 640, 768, 896, 1024, 1280]
    candidates = {}  # length -> (sequence, score)

    if n_parent not in target_lengths:
        target_lengths.append(n_parent)
    target_lengths = sorted(set(target_lengths))

    for tgt_n in target_lengths:
        if time.monotonic() > t1_end:
            break
        rem = t1_end - time.monotonic()
        if rem < 15:
            break

        f_init = interpolate_sequence(parent, tgt_n)
        # Quick Adam with low beta
        t_alloc = min(rem * 0.6, 15)
        a_opt, obj = adam_optimize(f_init, time_limit=t_alloc,
                                   beta_schedule=[15.0, 40.0],
                                   lr_max=0.003, restart_interval=50,
                                   verbose=False, label=f"explore_n{tgt_n}")
        score = try_seq(a_opt, f"explore_n{tgt_n}")
        if score < best_score:
            candidates[tgt_n] = (a_opt, score)
        elif tgt_n not in candidates:
            candidates[tgt_n] = (a_opt, score)

    # Also try Gaussian and cosine inits at a couple lengths
    for init_type in ['gauss', 'cos']:
        if time.monotonic() > t1_end:
            break
        rem = t1_end - time.monotonic()
        if rem < 15:
            break
        for tgt_n in [768, 1024]:
            if time.monotonic() > t1_end:
                break
            if init_type == 'gauss':
                f_init = gaussian_init(tgt_n, sigma_frac=0.3, seed=seed + tgt_n)
            else:
                f_init = cosinit_init(tgt_n, k=1, seed=seed + tgt_n)
            t_alloc = min(rem * 0.3, 10)
            a_opt, obj = adam_optimize(f_init, time_limit=t_alloc,
                                       beta_schedule=[15.0, 40.0],
                                       lr_max=0.003, restart_interval=50,
                                       verbose=False, label=f"{init_type}_{tgt_n}")
            try_seq(a_opt, f"{init_type}_{tgt_n}")

    # === Phase 2: Deep Adam (500s) ===
    print(f"\n=== Phase 2: Deep Adam (best={best_score:.6f}) ===")
    t_phase2 = 500
    t2_end = time.monotonic() + t_phase2

    # Pick top 2 candidates by score for deep optimization
    sorted_candidates = sorted(candidates.items(), key=lambda x: x[1][1])
    deep_lengths = [sorted_candidates[i][0] for i in range(min(2, len(sorted_candidates)))]
    if len(best_seq) not in deep_lengths:
        deep_lengths.append(len(best_seq))
    deep_lengths = deep_lengths[:3]

    for dl in deep_lengths:
        if time.monotonic() > t2_end:
            break
        rem = t2_end - time.monotonic()
        if rem < 30:
            break

        if dl in candidates:
            f_start = candidates[dl][0].copy()
        else:
            f_start = interpolate_sequence(parent, dl)

        t_alloc = min(rem * 0.7, 180)
        print(f"  Deep Adam on n={dl}, time={t_alloc:.0f}s")
        a_opt, obj = adam_optimize(
            f_start, time_limit=t_alloc,
            beta_schedule=[10.0, 30.0, 80.0, 200.0, 500.0, 1000.0],
            lr_max=0.004, lr_min=0.00005,
            restart_interval=80, top_k_polish=40, polish_every=30,
            verbose=True, label=f"deep_n{dl}")
        try_seq(a_opt, f"deep_n{dl}")

        # Follow up with SG polish
        rem2 = t2_end - time.monotonic()
        if rem2 > 20:
            a_opt2, _ = sg_polish_optimize(a_opt, time_limit=min(rem2 * 0.5, 30),
                                           lr=0.002, mom=0.94, top_k=50, temp=0.0002)
            try_seq(a_opt2, f"deep_sg_n{dl}")

    # === Phase 3: LP-FW (200s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    t_phase3 = 200
    t3_end = time.monotonic() + t_phase3

    if best_seq is not None:
        rem = t3_end - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.6, 100)
            print(f"  LP-FW on best, time={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_main")

        rem = t3_end - time.monotonic()
        if rem > 20:
            # LP-FW from perturbed start
            noise = rng.standard_normal(len(best_seq)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.7, 50)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, "lpfw_perturbed")

    # === Phase 4: Palindromic track (150s) ===
    print(f"\n=== Phase 4: Palindromic (best={best_score:.6f}) ===")
    t_phase4 = 150
    t4_end = time.monotonic() + t_phase4

    for pal_n in [768, 1024]:
        if time.monotonic() > t4_end:
            break
        rem = t4_end - time.monotonic()
        if rem < 30:
            break

        # Create palindromic start from current best
        f_base = interpolate_sequence(best_seq if best_seq is not None else parent, pal_n)
        f_pal = palindromic_project(f_base)

        t_alloc = min(rem * 0.5, 50)
        a_opt, obj = adam_optimize(f_pal, time_limit=t_alloc,
                                   beta_schedule=[20.0, 60.0, 150.0],
                                   lr_max=0.003, restart_interval=60,
                                   verbose=False, label=f"pal_{pal_n}")

        # Ensure palindromicity is maintained (Adam may break it)
        # Try again with palindromic projection at each step via a wrapper
        # For now, just project the result
        a_opt_pal = palindromic_project(a_opt)
        try_seq(a_opt_pal, f"pal_{pal_n}")

    # === Phase 5: Multi-restart (350s) ===
    print(f"\n=== Phase 5: Multi-restart (best={best_score:.6f}) ===")
    t_phase5 = 350
    t5_end = time.monotonic() + t_phase5

    n_restarts = 12
    per_restart_time = t_phase5 / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > t5_end:
            break
        rem = t5_end - time.monotonic()
        if rem < 15:
            break

        # Different perturbation strategies
        if r % 3 == 0:
            # Small Gaussian noise
            noise_scale = 0.003 + 0.002 * (r // 3)
            f_start = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * noise_scale)
        elif r % 3 == 1:
            # Random coordinate blending
            f_start = best_seq.copy()
            k = len(best_seq) // 4
            idx = rng.choice(len(best_seq), size=k, replace=False)
            f_start[idx] = rng.dirichlet(np.ones(k))
            f_start = f_start / f_start.sum()
        else:
            # Interpolation with a different-length version
            other_n = len(best_seq) + rng.choice([-128, 128, -64, 64])
            other_n = max(256, min(1536, other_n))
            f_interp = interpolate_sequence(best_seq, other_n)
            f_start = project_simplex(f_interp + rng.standard_normal(other_n) * 0.005)

        t_alloc = min(rem * 0.7, per_restart_time)
        a_opt, obj = adam_optimize(
            f_start, time_limit=t_alloc,
            beta_schedule=[30.0, 80.0, 200.0],
            lr_max=0.003, lr_min=0.0002,
            restart_interval=60, top_k_polish=30, polish_every=25,
            verbose=False, label=f"restart_{r}")
        try_seq(a_opt, f"restart_{r}")

    # === Phase 6: Final polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")

    if best_seq is not None:
        # 6a: Very high beta Adam (essentially targeting the true max)
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 80)
            print(f"  Final high-beta Adam, time={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(
                best_seq, time_limit=t_alloc,
                beta_schedule=[200.0, 500.0, 1000.0, 2000.0],
                lr_max=0.001, lr_min=0.00005,
                restart_interval=40, top_k_polish=50, polish_every=15,
                verbose=True, label="final_high_beta")
            try_seq(a_opt, "final_high_beta")

        # 6b: SG polish with very fine parameters
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.6, 40)
            a_opt, _ = sg_polish_optimize(best_seq, time_limit=t_alloc,
                                          lr=0.001, mom=0.95, top_k=60, temp=0.0001)
            try_seq(a_opt, "final_sg")

        # 6c: LP-FW final
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.7, 30)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

        # 6d: One more Adam round with fresh perturbation
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            noise = rng.standard_normal(len(best_seq)) * 0.002
            f_start = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.8, 30)
            a_opt, _ = adam_optimize(f_start, time_limit=t_alloc,
                                     beta_schedule=[300.0, 800.0],
                                     lr_max=0.001, restart_interval=30,
                                     verbose=False, label="final_restart")
            try_seq(a_opt, "final_restart")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
