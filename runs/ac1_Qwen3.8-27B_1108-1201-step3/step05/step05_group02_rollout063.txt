import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_toeplitz_matrix(f, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_direction(f, M, n, tol=1e-10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'presolve': True, 'dual_feasibility_tolerance': tol,
               'primal_feasibility_tolerance': tol, 'ipm_optimality_tolerance': 1e-12}
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=options)
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def analytic_init(n, style, rng):
    """Generate analytic initial sequences of various styles."""
    x = np.linspace(0, 1, n)
    if style == 'uniform':
        f = np.ones(n)
    elif style == 'parabola':
        a = 1.0 + rng.uniform(0, 2)
        f = np.maximum(1 - a * (2*x - 1)**2, 0)
    elif style == 'gaussian':
        sigma = 0.15 + rng.uniform(0, 0.3)
        f = np.exp(-0.5 * ((2*x - 1) / sigma)**2)
    elif style == 'cosine':
        phase = rng.uniform(0, 2*np.pi)
        freq = rng.uniform(0.5, 3)
        f = np.maximum(1 + rng.uniform(0.2, 0.8) * np.cos(2*np.pi*freq*x + phase), 0)
    elif style == 'raised_cosine':
        f = 0.5 * (1 + np.cos(2*np.pi*x))
    elif style == 'triangular':
        f = 1 - np.abs(2*x - 1)
    elif style == 'wider_gaussian':
        sigma = 0.3 + rng.uniform(0, 0.2)
        f = np.exp(-0.5 * ((2*x - 1) / sigma)**2)
    elif style == 'plateau':
        w = 0.3 + rng.uniform(0, 0.4)
        f = np.where(np.abs(2*x - 1) < w, 1.0, 0.0)
        # Smooth edges
        f = np.convolve(f, np.ones(5)/5, mode='same')
    elif style == 'bimodal':
        center1 = 0.3 + rng.uniform(-0.1, 0.1)
        center2 = 0.7 + rng.uniform(-0.1, 0.1)
        sigma = 0.1 + rng.uniform(0, 0.1)
        f = np.exp(-0.5*((x-center1)/sigma)**2) + np.exp(-0.5*((x-center2)/sigma)**2)
    elif style == 'smooth_step':
        t = rng.uniform(0.1, 0.4)
        f = 1 / (1 + np.exp(-(x - t) / 0.05)) - 1 / (1 + np.exp(-(x - 1 + t) / 0.05))
    else:
        f = np.ones(n)
    
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n):
    """Build the (2n-1) x n Toeplitz constraint matrix efficiently."""
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = f[diff[valid]]
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    c = -np.ones(n)
    A = build_constraint_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {
        'presolve': True,
        'dual_feasibility_tolerance': 1e-11,
        'primal_feasibility_tolerance': 1e-11,
        'ipm_optimality_tolerance': 1e-12
    }
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, max_m_relax=8.0):
    """Enhanced LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.15
            if m_relax > max_m_relax:
                m_relax = 1.0
                no_improve += 1
                if no_improve > 40:
                    # Try with larger relaxation
                    g = solve_lp_direction(f, M * 1.5, n)
                    if g is None:
                        break
                    m_relax = 1.5
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.03, 5.0)
            continue

        # Line search: coarse then fine bisection
        best_alpha = 0.0
        best_val = M

        alphas_coarse = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas_coarse:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best
        if best_alpha > 0.0:
            if best_alpha >= 1.0:
                lo, hi = 0.7, 1.0
            elif best_alpha <= 0.02:
                lo, hi = 0.0, 0.08
            else:
                lo = max(0.0, best_alpha - 0.12)
                hi = min(1.0, best_alpha + 0.12)

            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    iter {it} ({elapsed:.0f}s): obj={best_obj:.15f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

        # Periodic restart with diverse perturbation
        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(no_improve // 80, 5)
            perturbation_type = no_improve % 3

            if perturbation_type == 0:
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
            elif perturbation_type == 1:
                # Frequency domain perturbation
                F = np.fft.rfft(f)
                phases = np.exp(1j * rng.standard_normal(len(F)) * 0.3)
                F_new = F * phases
                f_real = np.fft.irfft(F_new, n)
                f = project_simplex(np.maximum(f_real, 0))
            else:
                # Smooth perturbation
                f_smooth = np.convolve(best_f, np.ones(5) / 5, mode='same')
                f = project_simplex(f_smooth + rng.standard_normal(n) * scale * 0.5)

            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, multipliers=None):
    """Coordinate descent with multiplicative updates and re-normalization."""
    if multipliers is None:
        multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def peak_targeted_optimize(f0, time_limit, top_k=15):
    """Targeted optimization: identify peaks and try to reduce them via moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()

        k_actual = min(top_k, len(conv))
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]

        improved = False
        rng = np.random.default_rng(it * 31 + 7)

        for k in sorted(peak_indices.tolist(), key=lambda x: -conv[x])[:5]:
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(k)
            # Find the j that contributes most to this peak
            j_lo = max(0, k_int - n + 1)
            j_hi = min(n - 1 - k_int, n - 1)

            if j_lo > j_hi:
                continue

            j_range = np.arange(j_lo, j_hi + 1)
            contributions = f[j_range] * f[j_range + k_int]
            j_max = j_range[np.argmax(contributions)]

            if f[j_max] < 1e-8:
                continue

            # Try moving a small fraction from j_max to random positions
            delta = f[j_max] * 0.02
            n_candidates = min(30, n)
            candidates = rng.choice(n, size=n_candidates, replace=False)

            for dst in candidates:
                if dst == j_max or f[dst] > 0.99:
                    continue
                f_trial = f.copy()
                f_trial[j_max] -= delta
                f_trial[dst] += delta
                if f_trial[j_max] < 0 or f_trial[dst] < 0:
                    continue
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < M - 1e-14:
                    f = f_trial
                    improved = True
                    break

        if not improved:
            # Small random perturbation to escape
            f = project_simplex(f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def analytic_init(n, style, rng):
    """Generate diverse analytic initial sequences."""
    x = np.linspace(0, 1, n)
    if style == 'parabola':
        a = 1.0 + rng.uniform(0, 3)
        f = np.maximum(1 - a * (2 * x - 1) ** 2, 0)
    elif style == 'gaussian':
        sigma = 0.15 + rng.uniform(0, 0.35)
        f = np.exp(-0.5 * ((2 * x - 1) / sigma) ** 2)
    elif style == 'cosine':
        freq = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.9)
        f = np.maximum(1 + amp * np.cos(2 * np.pi * freq * x + phase), 0)
    elif style == 'raised_cosine':
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 'triangular':
        f = 1 - np.abs(2 * x - 1)
    elif style == 'u_shaped':
        a = 1.0 + rng.uniform(0, 2)
        f = np.maximum(a * (2 * x - 1) ** 2 + 0.1, 0)
    elif style == 'bimodal':
        c1 = 0.25 + rng.uniform(-0.1, 0.1)
        c2 = 0.75 + rng.uniform(-0.1, 0.1)
        sigma = 0.1 + rng.uniform(0, 0.15)
        f = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif style == 'plateau':
        w = 0.3 + rng.uniform(0, 0.4)
        f = np.where(np.abs(2 * x - 1) < w, 1.0, 0.0)
        f = np.convolve(f, np.ones(7) / 7, mode='same')
    elif style == 'skewed':
        alpha = 1.0 + rng.uniform(0, 3)
        f = x ** alpha + (1 - x) ** (2 * alpha)
    elif style == 'sinc_like':
        freq = 1.0 + rng.uniform(0, 5)
        f = np.maximum(np.sinc(freq * (x - 0.5)) + 1.5, 0)
    else:
        f = np.ones(n)

    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f


def interpolate_sequence(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def freq_perturb(f, rng, strength=0.2):
    """Frequency domain perturbation: modify phases and project back."""
    n = len(f)
    F = np.fft.rfft(f)
    magnitudes = np.abs(F)
    phases = np.angle(F)
    new_phases = phases + rng.standard_normal(len(F)) * strength
    F_new = magnitudes * np.exp(1j * new_phases)
    f_new = np.fft.irfft(F_new, n)
    f_new = project_simplex(np.maximum(f_new, 0))
    return f_new

    styles = ['parabola', 'gaussian', 'u_shaped', 'bimodal', 'skewed',
              'sinc_like', 'plateau', 'cosine']

    for style in styles:
        t_rem = deadline - time.monotonic() - 350
        if t_rem < 15:
            break
        t_alloc = min(55, t_rem * 0.12)
        if t_alloc < 10:
            continue
        f_init = analytic_init(n_parent, style, rng)
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  [{style}] NEW BEST: {sc:.8f}")
        else:
            print(f"  [{style}] {sc:.8f}")

    # === Phase 3: Multi-length search ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    candidate_lengths = set()
    # Relative to best
    for factor in [0.5, 0.6, 0.7, 0.8, 1.2, 1.4, 1.6, 2.0]:
        nl = int(n_best * factor)
        nl = max(64, min(4096, nl))
        candidate_lengths.add(nl)
    # Fixed candidates
    for nl in [128, 192, 256, 320, 384, 448, 512, 640, 768, 896, 1024, 1280, 1536, 2048, 3072]:
        if 64 <= nl <= 4096:
            candidate_lengths.add(nl)

    candidate_lengths.discard(n_best)
    # Sort by proximity to current best length
    sorted_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))
    sorted_lengths = sorted_lengths[:10]  # Limit to 10

    for target_n in sorted_lengths:
        t_rem = deadline - time.monotonic() - 250
        if t_rem < 15:
            break
        t_alloc = min(70, t_rem * 0.15)
        if t_alloc < 10:
            continue

        f_interp = interpolate_sequence(best_seq, target_n)
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  n={target_n} NEW BEST: {sc:.8f}")
        else:
            print(f"  n={target_n} {sc:.8f}")

    # === Phase 4: Deep LP-FW with diverse perturbations ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    # Generate diverse perturbations
    perturbations = []
    for i, scale in enumerate([0.002, 0.005, 0.01, 0.015, 0.025]):
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        perturbations.append(f_pert)

    # Frequency domain perturbations
    for i in range(3):
        f_freq = freq_perturb(best_seq, rng, strength=0.1 + 0.1 * i)
        perturbations.append(f_freq)

    # Power-law transform perturbations
    for p in [0.7, 1.3, 0.5]:
        f_pow = best_seq ** p
        f_pow = project_simplex(f_pow / f_pow.sum())
        perturbations.append(f_pow)

    # Smoothing perturbation
    f_smooth = np.convolve(best_seq, np.ones(5) / 5, mode='same')
    perturbations.append(project_simplex(np.maximum(f_smooth, 0)))

    for idx, f_pert in enumerate(perturbations):
        t_rem = deadline - time.monotonic() - 180
        if t_rem < 12:
            break
        t_alloc = min(100, t_rem * 0.18)
        if t_alloc < 10:
            continue
        a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  pert[{idx}] NEW BEST: {sc:.8f}")
        else:
            if sc < best_score + 0.001:
                print(f"  pert[{idx}] {sc:.8f} (close)")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic() - 100
    if t_rem > 25:
        t_cd = min(t_rem * 0.5, 120)
        print(f"  t={t_cd:.0f}s, n={n_best}")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
        else:
            print(f"  Score: {sc:.8f}")

    # === Phase 6: Peak targeted optimization ===
    print(f"\n=== Phase 6: Peak Targeting (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic() - 60
    if t_rem > 15:
        t_pt = min(t_rem * 0.6, 50)
        a_opt, _ = peak_targeted_optimize(best_seq, time_limit=t_pt, top_k=25)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")
        else:
            print(f"  Score: {sc:.8f}")

    # === Phase 7: Final LP-FW squeeze ===
    print(f"\n=== Phase 7: Final Squeeze (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic() - 10
    if t_rem > 10:
        t_sq = min(t_rem * 0.9, 50)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sq)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL LP-FW: {sc:.8f}")

    # === Phase 8: Micro-perturbation search ===
    print(f"\n=== Phase 8: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        t_rem = deadline - time.monotonic()
        if t_rem < 3:
            break
        scale = 0.0002 + trial * 0.00015
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  micro[{trial}] NEW BEST: {sc:.8f}")
            # Quick polish
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 5:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(t_rem2 * 0.4, 8))
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  micro[{trial}]+fw: {sc2:.8f}")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
