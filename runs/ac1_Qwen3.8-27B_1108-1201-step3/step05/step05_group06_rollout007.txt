"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with log-sum-exp smoothing, peak-targeted coordinate descent,
diverse structured initializations, and adaptive temperature scheduling.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_mirror(f, T):
    """Compute gradient for mirror descent with log-sum-exp smoothing."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))

    # grad[i] = (2/T) * sum_j f[j] * w[i+j] (cross-correlation)
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def mirror_descent_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Mirror descent (multiplicatively weighted update) with log-sum-exp smoothing and AdaGrad."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # AdaGrad state
    sq_grad = np.zeros(n)
    t_adagrad = 0
    base_lr = 0.05

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        # Cosine annealing for temperature
        progress = min(elapsed / time_limit, 1.0)
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        T = max(T, T_end)
        lr = base_lr / (1.0 + elapsed * 0.001)

        grad, lse_val, conv, w = smooth_grad_mirror(f, T)

        # Mirror descent step: a_new ∝ a * exp(-lr * grad / sqrt(cumulative_grad^2))
        t_adagrad += 1
        sq_grad += grad ** 2
        eff_lr = lr / (np.sqrt(sq_grad) + 1e-8)

        log_step = -eff_lr * grad
        log_step = np.clip(log_step, -50, 50)  # prevent overflow/underflow
        f_new = f * np.exp(log_step)
        s_new = f_new.sum()
        if s_new < 1e-15 or not np.isfinite(s_new):
            f_new = f.copy()
        else:
            f_new = f_new / s_new
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new = f_new / s_new

        cur_obj = fast_convolve(f_new).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Mirror iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1

        if no_improve > 500:
            # Restart with perturbation
            scale = 0.005 + 0.005 * min(no_improve // 500, 3)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            sq_grad = np.zeros(n)
            t_adagrad = 0
            no_improve = 0

    return best_f, best_obj


def adam_smooth_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Adam-based optimization with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    T = T_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        T = max(T, T_end)
        lr = alpha * (0.3 ** (elapsed / (time_limit * 0.7)))

        grad, lse_val, conv, w = smooth_grad_mirror(f, T)

        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)

        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 300 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1

        if no_improve > 400:
            scale = 0.003 + 0.004 * min(no_improve // 400, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best_f, best_obj


def peak_shaving_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting the peak lags of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    rng = np.random.default_rng(77777)

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top K peak lags
        conv = fast_convolve(f)
        K = min(5, 2 * n - 1)
        peak_lags = np.argsort(conv)[::-1][:K]
        M = conv[peak_lags[0]]

        # For each peak lag, find the most contributing pairs and adjust
        for k_star in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break

            # Find contributing pairs: j and k_star - j must both be in [0, n-1]
            j_min = max(0, k_star - (n - 1))
            j_max = min(n - 1, k_star)

            if j_min > j_max:
                continue

            indices = np.arange(j_min, j_max + 1)
            partners = k_star - indices
            products = f[indices] * f[partners]

            # Target the top contributing pairs
            top_pairs = np.argsort(products)[::-1][:min(20, len(indices))]

            for idx_pos in top_pairs:
                j = indices[idx_pos]
                partner = k_star - j
                if partner < 0 or partner >= n:
                    continue
                if j == partner:
                    continue

                # Try to redistribute mass between j and partner
                old_j = f[j]
                old_p = f[partner]
                total = old_j + old_p
                if total < 1e-12:
                    continue

                # Try different splits
                cur_conv_max = conv.copy()
                # Effect on lag k_star: (new_j * new_p - old_j * old_p)
                # We want to reduce this, so make the product smaller
                # Try making one larger and the other smaller
                best_ratio = old_j / total
                best_delta = 0

                for frac in [0.8, 0.9, 0.95, 0.1, 0.05, 0.0]:
                    new_j = frac * total
                    new_p = (1 - frac) * total
                    delta = (new_j * new_p - old_j * old_p)
                    if delta < best_delta - 1e-15:
                        # Check if this helps overall
                        f_trial = f.copy()
                        f_trial[j] = new_j
                        f_trial[partner] = new_p
                        new_conv = fast_convolve(f_trial)
                        new_max = new_conv.max()
                        if new_max < M - 1e-14:
                            best_delta = delta
                            best_ratio = frac

                if best_ratio != old_j / total:
                    f[j] = best_ratio * total
                    f[partner] = (1 - best_ratio) * total
                    new_conv = fast_convolve(f)
                    new_max = new_conv.max()
                    if new_max < best_obj - 1e-14:
                        best_obj = new_max
                        best_f = f.copy()
                        improved = True
                        conv = new_conv
                        M = conv.max()

        if not improved and passes > 5:
            # Perturb and continue
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)

    return best_f, best_obj

def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric (rising-falling) geometric sequence."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()


def make_gaussian(n, sigma_ratio=0.3, mu_ratio=0.5):
    """Gaussian window."""
    j = np.arange(n, dtype=np.float64)
    mu = mu_ratio * (n - 1)
    sigma = sigma_ratio * (n - 1)
    a = np.exp(-0.5 * ((j - mu) / sigma) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_bimodal(n, width=0.25):
    """Bimodal distribution (mass at both ends)."""
    j = np.arange(n, dtype=np.float64)
    mu1 = width * (n - 1)
    mu2 = (1 - width) * (n - 1)
    sigma = 0.08 * (n - 1)
    a = np.exp(-0.5 * ((j - mu1) / sigma) ** 2) + np.exp(-0.5 * ((j - mu2) / sigma) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_cosine_mod(n, beta=0.5, freq=1.0):
    """Cosine-modulated sequence: 1 + beta*cos(2*pi*freq*j/n)."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + beta * np.cos(2 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_powerlaw(n, alpha=0.5):
    """Power-law decay: (j+1)^(-alpha)."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** (-alpha)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    for target_n in [896, 1024, 1280, 1536, 2048]:
        if abs(target_n - len(best_seq)) < 50:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break

        f_interp = interpolate_seq(best_seq, target_n)
        _try(f_interp, f"interp_n{target_n}")

        t1 = min(rem * 0.3, 60)
        if t1 > 10:
            a_opt, obj_opt = mirror_descent_optimize(f_interp, time_limit=t1,
                                                      T_start=0.15, T_end=0.01)
            _try(a_opt, f"mirror_n{target_n}")

        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t2 = min(rem * 0.3, 50)
        if t2 > 10:
            a_opt, obj_opt = adam_smooth_optimize(f_interp, time_limit=t2,
                                                  T_start=0.1, T_end=0.01)
            _try(a_opt, f"adam_n{target_n}")

    # === Phase 5: Final polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.3, 80)
        if t_alloc > 8:
            a_opt, obj_opt = mirror_descent_optimize(best_seq, time_limit=t_alloc,
                                                      T_start=0.02, T_end=0.001, verbose=True)
            _try(a_opt, "mirror_final")

        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 50)
        if t_alloc > 8:
            a_opt, obj_opt = single_cd_optimize(best_seq, time_limit=t_alloc)
            _try(a_opt, "cd_final")

        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 40)
        if t_alloc > 8:
            a_opt, obj_opt = peak_shaving_cd(best_seq, time_limit=t_alloc)
            _try(a_opt, "peak_final")

        # Multiple small perturbations
        for i in range(6):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * i)
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.15, 25)
            if t_alloc > 5:
                a_opt, obj_opt = mirror_descent_optimize(f_pert, time_limit=t_alloc,
                                                          T_start=0.02, T_end=0.001)
                _try(a_opt, f"perturb_{i}")

        # Reversed
        rem = deadline - time.monotonic()
        if rem > 15:
            f_rev = best_seq[::-1].copy()
            a_opt, obj_opt = mirror_descent_optimize(f_rev, time_limit=min(rem, 25),
                                                      T_start=0.02, T_end=0.001)
            _try(a_opt, "reversed")

        # Blend with uniform
        rem = deadline - time.monotonic()
        if rem > 10:
            for blend in [0.9, 0.85, 0.8]:
                f_blend = blend * best_seq + (1 - blend) * (np.ones(n_best) / n_best)
                f_blend = project_simplex(f_blend)
                t_alloc = min(rem, 15)
                if t_alloc > 5:
                    a_opt, obj_opt = mirror_descent_optimize(f_blend, time_limit=t_alloc,
                                                              T_start=0.015, T_end=0.001)
                    _try(a_opt, f"blend_{blend}")
                rem = deadline - time.monotonic()
                if rem < 8:
                    break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
