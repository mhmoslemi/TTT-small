def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Central binomial coefficient initialization (flat autocorrelation approximation)
2. LP-based Frank-Wolfe with adaptive constraints
3. L-BFGS on annealed log-sum-exp in softmax parameterization
4. Peak-targeted subgradient shaving

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def binomial_init(n):
    """Initialize with central binomial coefficient sequence (flat autocorrelation approx)."""
    a = np.zeros(n)
    for j in range(n):
        a[j] = comb(2 * j, j) / (4.0 ** j)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_init(n):
    """Initialize with a cosine-squared window."""
    j = np.arange(n)
    a = np.cos(np.pi * j / (2.0 * n)) ** 2
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0] / new_n * old_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_direction_sparse(f, M, n):

    c = -np.ones(n)

    for j in range(n):

        end = min(j + n, n_constraints)
        A_ub[start:end, j] = f[:end - j]
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub.tocsr(), b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None
