"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with spectral projection, coordinate descent, and multi-length search.
"""

import numpy as np
import time
import scipy.sparse
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def spectral_refine(f, time_limit=50, verbose=False):
    """Iterative spectral flattening with non-negativity projection."""
    f = np.maximum(f.astype(np.float64), 0)
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    alpha = 0.3  # blending parameter for spectral modification
    
    while time.monotonic() - t0 < time_limit:
        # Compute FFT
        F = np.fft.rfft(f)
        
        # Target: flat magnitude
        target_mag = np.full(len(F), np.sqrt(np.mean(np.abs(F)**2)))
        
        # Blend current and target magnitude
        current_mag = np.abs(F)
        new_mag = np.power(current_mag, 1-alpha) * np.power(target_mag, alpha)
        
        # Preserve phase
        phase = np.angle(F)
        F_new = new_mag * np.exp(1j * phase)
        
        # Ensure conjugate symmetry (rfft already handles this)
        f_new = np.fft.irfft(F_new, n)
        
        # Project to non-negative simplex
        f_new = project_simplex(f_new, 1.0)
        
        obj = fast_convolve(f_new).max()
        if obj < best_M - 1e-14:
            best_M = obj
            best_f = f_new.copy()
        else:
            # Reduce alpha
            alpha *= 0.9
            if alpha < 0.01:
                break
        
        f = f_new
    
    return best_f, best_M


def coordinate_descent(a0, time_limit=50, verbose=False):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj

            continue
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        M_new = conv.max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            M = M_new
            no_improve = 0
            step_decay = 1.0
        else:
            no_improve += 1
            M = min(M, M_new * 1.001)
            step_decay *= 0.95
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: M={M:.10f}, best={best_M:.10f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_f, best_M


def spectral_flat(f, alpha=0.3):
    """One step of spectral flattening."""
    n = len(f)
    F = np.fft.rfft(f)
    current_mag = np.abs(F)
    target_mag = np.full(len(F), np.sqrt(np.mean(current_mag**2)))
    new_mag = np.power(current_mag, 1 - alpha) * np.power(target_mag, alpha)
    phase = np.angle(F)
    F_new = new_mag * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    return f_new


def spectral_refine(f, time_limit, verbose=False):
    """Iterative spectral flattening with non-negativity projection."""
    f = np.maximum(f.astype(np.float64), 0)
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    alpha = 0.2
    
    while time.monotonic() - t0 < time_limit:
        f_new = spectral_flat(f, alpha)
        f_new = project_simplex(f_new, 1.0)
        
        obj = fast_convolve(f_new).max()
        if obj < best_M - 1e-14:
            best_M = obj
            best_f = f_new.copy()
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.9
            if alpha < 0.02:
                break
        f = f_new
    
    return best_f, best_M


def coordinate_descent(a0, time_limit=30):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.7, 0.85, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val <= 0 or new_val >= 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
