def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def logsumexp(x, axis=None):
    """Numerically stable log-sum-exp."""
    x_max = np.max(x, axis=axis, keepdims=True)
    out = np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True)) + x_max
    return np.squeeze(out, axis=axis) if axis is not None else np.squeeze(out)


def softmax_weights(c, t):
    """Compute softmax weights over convolution values."""
    shifted = t * c
    shifted -= shifted.max()
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return w / s


def compute_gradient(a, p):
    """Compute gradient of logsumexp(t*c) w.r.t. a, given softmax weights p."""
    n = len(a)
    size = 2 * n - 1
    corr_size = size + n - 1
    fft_size = 1 << (corr_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    grad = np.fft.irfft(np.conj(P) * A, fft_size)[:n] * 2.0
    return grad


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        x = np.linspace(0, 1, n)

        f = np.maximum(f, 1e-15)

        c = fast_convolve(f)

        c = fast_convolve(f)

    if res.success or res.fun < best_score:
        a = make_gaussian(n, res.x[0], res.x[1])

            best_a = a.copy()

        f = f / f.sum()
        score = 2 * n * res.fun
        if score < best_score:

            best_a = f.copy()

    # Cosine modulated
    res = scipy_minimize(obj_mod, x0=[0.3, -0.1], method='Nelder-Mead',

def lbfags_optimize(a0, budget_s, t_init=50.0, t_end=500.0, n_steps=200):
    """
    L-BFGS optimization on softmax parameterization with annealed temperature.
    Minimize logsumexp(t * conv(a, a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    # Work in log-space: a = exp(x) / sum(exp(x))
    x0 = np.log(a + 1e-30)
    best_x = x0.copy()
    best_obj = fast_convolve(a).max()

    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        t = t_init + (t_end - t_init) * frac
        x0 = best_x.copy()

        def obj_and_grad(x):
            ex = np.exp(x - x.max())
            a_ = ex / ex.sum()
            c = fast_convolve(a_)
            val = logsumexp(t * c)
            p = softmax_weights(c, t)
            # Gradient in x-space: chain rule through softmax
            grad_a = compute_gradient(a_, p)
            # dx/dx -> da/dx: da_i/dx_j = a_i*(delta_ij - a_j)
            grad_x = a_ * (grad_a - a_ * grad_a.sum())
            return val, grad_x

        def obj_only(x):
            ex = np.exp(x - x.max())
            a_ = ex / ex.sum()
            c = fast_convolve(a_)
            return logsumexp(t * c)

        time_per_step = budget_s / n_steps
        res = scipy_minimize(obj_only, x0, method='L-BFGS-B',
                             jac=lambda x: obj_and_grad(x)[1],
                             options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-8,
                                      'maxfun': 200})

        if res.fun < best_obj / max(t, 1e-6):
            ex = np.exp(res.x - res.x.max())
            a_new = ex / ex.sum()
            obj_new = fast_convolve(a_new).max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_x = res.x.copy()

    ex = np.exp(best_x - best_x.max())
    a_out = ex / ex.sum()
    final_obj = fast_convolve(a_out).max()
    return a_out, 2 * n * final_obj


def spectral_sharpen(a0, budget_s, n_iter=5000):
    """
    Iterative frequency-domain sharpening.
    Alternates between:
    1. Enforcing non-negativity (time domain)
    2. Sharpening the magnitude spectrum (frequency domain)
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = 1
    a = a / s

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    alpha = 0.3  # sharpening strength
    for it in range(n_iter):
        if time.monotonic() - t0 > budget_s:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Sharpen: boost peaks, suppress tails
        mag_mean = mag.mean()
        if mag_mean > 1e-15:
            normalized = mag / mag_mean
            # Apply power sharpening
            new_mag = mag * (normalized ** alpha)
            # Renormalize to preserve DC (sum of a)
            dc = A[0].real
            new_dc = new_mag[0]
            if new_dc > 1e-15:
                new_mag = new_mag * (dc / new_dc)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Anneal alpha (start strong, end gentle)
        frac = it / n_iter
        alpha = 0.5 * (1 - frac) + 0.1 * frac

    return best_a, 2 * n * best_obj


def subgradient_optimize(a0, budget_s, lr_init=0.01, temp_init=0.01,
                         temp_decay=0.9995, top_k=None):
    """Improved subgradient method with soft-max weights."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    t = 0
    temp = temp_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < budget_s:
        t += 1
        c = fast_convolve(a)
        p = softmax_weights(c, 1.0 / temp)

        if top_k is not None and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            p_full = np.zeros(len(c))
            p_full[idx] = p[idx]
            p = p_full / p_full.sum()

        grad = compute_gradient(a, p)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        a_new = a - lr * grad
        a = project_simplex(a_new)

        temp = max(temp * temp_decay, 1e-6)
        lr = lr_init * (0.999 ** t)
        lr = max(lr, lr_init * 0.05)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            temp = max(temp_init * 0.5, 0.003)
            lr = lr_init
            no_improve = 0

    return best_a, 2 * n * best_obj

                break
            if a[j] < 1e-14:
                continue
            old_val = a[j]
            best_val = old_val
            best_local_obj = best_obj

            # Try several multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                # Renormalize others to keep sum=1
                scale = (1.0 - new_val) / (1.0 - old_val)
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss

                obj = fast_convolve(a_trial).max()
                if obj < best_local_obj - 1e-15:
                    best_local_obj = obj
                    best_val = new_val

            if best_val != old_val:
                a[j] = best_val
                scale = (1.0 - best_val) / (1.0 - old_val)
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, 2 * n * best_obj


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a)
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-15:
        f = f / ss
    return f

        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
