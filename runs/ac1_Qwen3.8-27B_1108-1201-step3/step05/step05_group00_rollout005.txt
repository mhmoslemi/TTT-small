from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_banded(f, M, n):
    """Solve the LP exploitin the banded structure."""
    # max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # This is a banded matrix system

    # Build the constraint matrix efficiently
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A_ub[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)

        end = j + n

    b_ub = np.full(2*n-1, M)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-Frank-Wolfe, smoothed gradient descent, and peak shaving
with diverse initializations and adaptive length selection.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def smoothed_obj_grad(a, tau):
    """
    Compute the smoothed objective and its gradient.
    F(a) = (1/tau) * log(sum_k exp(tau * (a*a)[k]))
    grad_j = (1/tau) * sum_k p_k * a[k-j] for valid k-j
    where p_k = exp(tau * (a*a)[k]) / sum_l exp(tau * (a*a)[l])
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    
    smoothed_val = (1.0 / tau) * (scaled.max() + np.log(wsum))
    
    # Gradient: grad[j] = (1/tau) * sum_m a[m] * w[j+m] for m=0..n-1, j+m < 2n-1
    grad = np.zeros(n)
    for j in range(n):
        # sum_m a[m] * w[j+m] for m = 0 to min(n-1, 2n-2-j)
        m_end = min(n, 2 * n - 1 - j)
        if m_end > 0:
            grad[j] = (1.0 / tau) * np.dot(a[:m_end], w[j:j + m_end])
    
    return smoothed_val, grad

    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:

        f_new = project_simplex(f - lr * grad)

        if obj_new < best_obj - 1e-15:

            no_improve = 0
            lr = min(lr * 1.1, 1.0)
        else:

                no_improve = 0

def peak_shave(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        f = f / s

    best_f = f.copy()

        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Find top peaks
        top_k = min(5, len(conv))
        peak_idx = np.argsort(conv)[-top_k:]

            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            j_max = idx[np.argmax(products)]
            
            old_val = f[j_max]
            if old_val < 1e-12:
                continue
            
            best_trial = None
            best_trial_obj = M
            
            for mult in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                trial = f.copy()
                trial[j_max] = new_val
                diff = old_val - new_val
                if abs(diff) > 1e-15:
                    weights = f.copy()
                    weights[j_max] = 0
                    wsum = weights.sum()
                    if wsum > 1e-15:
                        trial += diff * weights / wsum
                    else:
                        trial = f.copy()
                        trial[j_max] = new_val
                        trial /= trial.sum()
                        continue
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-15:
                    best_trial_obj = obj
                    best_trial = trial
            
            if best_trial is not None:
                f = best_trial
                if best_trial_obj < best_obj - 1e-15:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Perturb and restart
                rng = np.random.default_rng(it * 19 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
                if it > 200:
                    break
        else:
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"      PeakShave iter {it}: obj={2*n*best_obj:.10f}")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
