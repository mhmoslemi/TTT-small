```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with log-sum-exp
smoothing, temperature annealing, diverse initializations, and coordinate
descent polishing. Key improvement over previous: smooth objective enables
faster-converging quasi-Newton methods (L-BFGS-B) instead of subgradient
methods (Adam), plus more aggressive temperature annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_grad(a, tau):
    """
    Compute the log-sum-exp smoothed objective and gradient.
    F(a) = 2n * tau * logsumexp(c/tau) / (sum(a))^2
    where c = conv(a,a).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    c = fast_convolve(a)
    
    # Numerically stable log-sum-exp
    c_max = c.max()
    c_shifted = (c - c_max) / tau
    exp_shifted = np.exp(c_shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    
    g = tau * (c_max / tau + np.log(sum_exp))
    # g = c_max + tau * log(sum(exp((c - c_max)/tau)))
    
    # Softmax weights
    w = exp_shifted / sum_exp
    
    # h[j] = sum_{i=0}^{n-1} a[i] * w[j+i]
    h = np.correlate(w, a, 'valid')
    
    # Ensure h has length n
    if len(h) != n:
        # Pad or trim if necessary
        h_new = np.zeros(n)
        h_new[:min(len(h), n)] = h[:min(len(h), n)]
        h = h_new
    
    # Objective
    F = 2.0 * n * g / (S * S)
    
    # Gradient
    grad = (4.0 * n / (S * S)) * (h - g / S)
    
    return F, grad


def run_lbfgs(a0, tau_schedule, time_limit, max_iter_per_tau=200):
    """
    Run L-BFGS-B with decreasing tau values.
    Returns the best sequence found.
    """
    a = a0.astype(np.float64).copy()
    # Normalize
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a0)) / len(a0)
        S = 1.0
    else:
        a = a / S
    
    best_a = a.copy()
    best_true_obj = float('inf')
    
    t0 = time.monotonic()
    
    for tau_idx, tau in enumerate(tau_schedule):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break
        
        max_iter = min(max_iter_per_tau, max(10, int(rem * 5)))
        
        # Define objective and gradient for this tau
        def obj_func(x):
            val, _ = smoothed_obj_grad(x, tau)
            return val
        
        def grad_func(x):
            _, g = smoothed_obj_grad(x, tau)
            return g
        
        bounds = [(0.0, None)] * len(a0)
        
        try:
            result = minimize(
                obj_func, a, method='L-BFGS-B',
                jac=grad_func, bounds=bounds,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxls': 50, 'maxfun': max_iter * 10}
            )
            a = np.maximum(result.x, 0.0)
            S = a.sum()
            if S > 1e-15:
                a = a / S
        except Exception:
            pass
        
        # Evaluate true objective
        c = fast_convolve(a)
        true_obj = 2.0 * len(a) * c.max() / (a.sum() ** 2)
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a.copy()
    
    # Final evaluation
    c = fast_convolve(a)
    true_obj = 2.0 * len(a) * c.max() / (a.sum() ** 2)
    if true_obj < best_true_obj:
        best_true_obj = true_obj
        best_a = a.copy()
    
    return best_a, best_true_obj


def coordinate_descent_polish(a0, time_limit):
    """
    Coordinate descent: for each coordinate, try scaling it and redistributing.
    """
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:

def make_gaussian_init(n, center=0.5, sigma=0.2, asym=0.0):
    """Create a (possibly asymmetric) Gaussian initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2 + asym * (x - center))
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_linear_init(n, slope=1.0):
    """Create a linearly varying initialization."""
    x = np.linspace(0, 1, n)
    f = 1.0 + slope * x
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_arcosine_init(n):
    """Acosine distribution."""
    x = np.linspace(0, 1, n)
    f = np.sqrt(1.0 - (2*x - 1)**2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-10)
    return f_interp / f_interp.sum()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with log-sum-exp
smoothing and temperature annealing, diverse initializations, length
exploration, and coordinate descent polishing.

Key improvement over previous: replaces subgradient Adam with smooth
quasi-Newton L-BFGS-B optimization of a log-sum-exp smoothed objective,
enabling much faster convergence per unit time.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_and_grad(a, tau):
    """
    Compute the log-sum-exp smoothed objective F(a) and its gradient.
    F(a) = 2n * tau * logsumexp(conv(a,a)/tau) / (sum(a))^2
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return 1e10, np.zeros(n)

    c = fast_convolve(a)

    # Numerically stable log-sum-exp
    c_max = c.max()
    c_shifted = (c - c_max) / tau
    exp_shifted = np.exp(c_shifted)
    sum_exp = float(exp_shifted.sum())
    if sum_exp < 1e-300:
        sum_exp = 1e-300

    g = c_max + tau * np.log(sum_exp)

    # Softmax weights
    w = exp_shifted / sum_exp

    # h[j] = sum_{i=0}^{n-1} a[i] * w[j+i]
    # Using np.correlate: correlate(w, a, 'valid') gives n values
    # output[k] = sum_i w[i+k] * a[i] for k=0,...,n-1
    h = np.correlate(w, a, 'valid')

    # Objective
    F = 2.0 * n * g / (S * S)

    # Gradient: dF/da[j] = 4n/S^2 * (h[j] - g/S)
    grad = (4.0 * n / (S * S)) * (h - g / S)

    return F, grad


def obj_wrapper(a, tau):
    val, _ = smoothed_obj_and_grad(a, tau)
    return val


def grad_wrapper(a, tau):
    _, g = smoothed_obj_and_grad(a, tau)
    return g


def run_lbfgs_anneal(a0, tau_list, time_limit, max_iter_per_tau=300):
    """
    Run L-BFGS-B with decreasing tau values (annealing schedule).
    Returns the best (sequence, true objective) found.
    """
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / S
    # Ensure all positive
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    n = len(a0)
    best_a = a.copy()
    best_true = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()

    for tau in tau_list:
        elapsed = time.monotonic() - t0
        rem = time_limit - elapsed
        if rem < 2.0:
            break

        max_iter = min(max_iter_per_tau, max(20, int(rem * 3)))

        bounds = [(0.0, None)] * n

        try:
            result = minimize(
                obj_wrapper, a,
                method='L-BFGS-B',
                jac=grad_wrapper,
                args=(tau,),
                bounds=bounds,
                options={'maxiter': max_iter, 'ftol': 1e-14,
                         'gtol': 1e-10, 'maxls': 40}
            )
            a_new = np.maximum(result.x, 1e-12)
            S_new = a_new.sum()
            if S_new > 1e-15:
                a = a_new / S_new
        except Exception:
            pass

        # Evaluate true (unsmoothed) objective
        c = fast_convolve(a)
        true_obj = 2.0 * n * c.max() / (a.sum() ** 2)
        if true_obj < best_true:
            best_true = true_obj
            best_a = a.copy()

    return best_a, best_true


def coord_descent_polish(a0, time_limit):
    """
    Coordinate descent: for each coordinate, try scaling it and
    redistributing mass to others to minimize max convolution.
    """
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_max_c = fast_convolve(a).max()

    t0 = time.monotonic()
    multipliers = [0.4, 0.6, 0.8, 1.25, 1.5, 2.0]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]

            if old_val < 1e-12:
                candidates = [1e-4 / n, 1e-3 / n, 5e-3 / n]
            else:
                candidates = [old_val * m for m in multipliers]

            best_cand = old_val
            best_cand_mc = best_max_c

            for cand in candidates:
                if cand < 0 or cand >= 1.0:
                    continue
                remaining = 1.0 - cand
                if remaining < 1e-10:
                    continue

                a_trial = a.copy()
                a_trial[j] = cand
                others_sum = a_trial.sum() - cand
                if others_sum < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] = a_trial[mask] * remaining / others_sum

                c_val = fast_convolve(a_trial).max()
                if c_val < best_cand_mc - 1e-16:
                    best_cand_mc = c_val
                    best_cand = cand

            if best_cand != old_val:
                remaining = 1.0 - best_cand
                others_sum = a.sum() - best_cand
                if others_sum > 1e-15:
                    a[j] = best_cand
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = a[mask] * remaining / others_sum
                    a = a / a.sum()

                if best_cand_mc < best_max_c - 1e-15:
                    best_max_c = best_cand_mc
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 1:
            break

    final_score = 2.0 * n * best_max_c
    return best_a, final_score


def make_init_gaussian(n, center=0.5, sigma=0.2, asym=0.0):
    """Create a (possibly asymmetric) Gaussian initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2 + asym * (x - center))
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_linear(n, slope=1.0):
    """Create a linearly varying initialization."""
    x = np.linspace(0, 1, n)
    f = 1.0 + slope * x
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_arc(n):
    """Arc-shaped (semi-circle) initialization."""
    x = np.linspace(0, 1, n)
    f = np.sqrt(np.maximum(1.0 - (2.0 * x - 1.0) ** 2, 0.0))
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_bump(n):
    """Sinusoidal bump initialization."""
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_init_smooth_random(n, rng, n_components=6):
    """Smooth random initialization as sum of Gaussians."""
    x = np.linspace(0, 1, n)
    f = np.zeros(n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.3)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interp_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-10)
    return f_interp / f_interp.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: L-BFGS-B with log-sum-exp smoothing and temperature
    annealing, diverse initializations, length exploration, CD polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        print("No parent found, using uniform init n=512")

    n_parent = len(parent)

    # === Phase 1: L-BFGS-B from parent with aggressive annealing ===
    print(f"\n=== Phase 1: L-BFGS-B from parent (n={n_parent}) ===")
    tau_sched_main = [
        1.0 / n_parent,
        0.3 / n_parent,
        0.1 / n_parent,
        0.03 / n_parent,
        0.01 / n_parent,
        0.003 / n_parent,
        0.001 / n_parent,
    ]
    a_opt, _ = run_lbfgs_anneal(parent, tau_sched_main, time_limit=180)
    try_seq(a_opt, "p1_parent")

    # === Phase 2: Diverse initializations at parent length ===
    print(f"\n=== Phase 2: Diverse inits (n={n_parent}) ===")

    inits = {
        'gauss_c': make_init_gaussian(n_parent, 0.5, 0.2, 0.0),
        'gauss_asym_r': make_init_gaussian(n_parent, 0.5, 0.15, 2.0),
        'gauss_asym_l': make_init_gaussian(n_parent, 0.5, 0.15, -2.0),
        'linear_r': make_init_linear(n_parent, 1.5),
        'linear_l': make_init_linear(n_parent, -1.5),
        'arc': make_init_arc(n_parent),
        'bump': make_init_bump(n_parent),
        'smooth_rand': make_init_smooth_random(n_parent, rng, 6),
    }

    tau_sched_div = [
        0.5 / n_parent,
        0.1 / n_parent,
        0.02 / n_parent,
        0.005 / n_parent,
        0.001 / n_parent,
    ]

    for name, init in inits.items():
        if time.monotonic() > deadline - 400:
            break
        rem = min(deadline - time.monotonic() - 400, 60)
        if rem < 15:
            break
        a_opt, _ = run_lbfgs_anneal(init, tau_sched_div, time_limit=rem)
        try_seq(a_opt, f"p2_{name}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration ===")

    base = best_seq if best_seq is not None else parent
    target_lengths = [384, 512, 640, 768, 896, 1024, 1280, 1536, 1920]
    target_lengths = [t for t in target_lengths if t != len(base) and 128 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - len(base)))
    target_lengths = target_lengths[:5]

    for target_n in target_lengths:
        if time.monotonic() > deadline - 300:
            break
        rem = min(deadline - time.monotonic() - 300, 80)
        if rem < 20:
            break
        f_interp = interp_seq(base, target_n)
        tau_sched_len = [
            0.5 / target_n, 0.1 / target_n, 0.02 / target_n,
            0.005 / target_n, 0.001 / target_n
        ]
        a_opt, _ = run_lbfgs_anneal(f_interp, tau_sched_len, time_limit=rem)
        try_seq(a_opt, f"p3_n{target_n}")

    # === Phase 4: Deep optimization from current best ===
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine annealing
        rem = min(deadline - time.monotonic() - 200, 120)
        if rem > 20:
            tau_fine = [
                0.01 / n_best, 0.003 / n_best,
                0.001 / n_best, 0.0003 / n_best, 0.0001 / n_best
            ]
            a_opt, _ = run_lbfgs_anneal(best_seq, tau_fine, time_limit=rem)
            try_seq(a_opt, "p4_fine")

        # Perturbed restarts with different noise scales
        for pi in range(5):
            if time.monotonic() > deadline - 150:
                break
            rem = min(deadline - time.monotonic() - 150, 50)
            if rem < 10:
                break
            noise_scale = 0.005 * (pi + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            tau_pert = [0.05 / n_best, 0.01 / n_best, 0.002 / n_best, 0.0005 / n_best]
            a_opt, _ = run_lbfgs_anneal(f_pert, tau_pert, time_limit=rem)
            try_seq(a_opt, f"p4_pert{pi}")

        # Mix with different shapes
        for mi in range(3):
            if time.monotonic() > deadline - 120:
                break
            rem = min(deadline - time.monotonic() - 120, 40)
            if rem < 10:
                break
            mix_ratio = 0.2 + 0.2 * mi
            if mi == 0:
                other = make_init_gaussian(n_best, 0.5, 0.2, 1.0)
            elif mi == 1:
                other = make_init_linear(n_best, 1.0)
            else:
                other = make_init_smooth_random(n_best, rng, 4)
            f_mix = (1.0 - mix_ratio) * best_seq + mix_ratio * other
            f_mix = project_simplex(f_mix)
            tau_mix = [0.1 / n_best, 0.02 / n_best, 0.005 / n_best, 0.001 / n_best]
            a_opt, _ = run_lbfgs_anneal(f_mix, tau_mix, time_limit=rem)
            try_seq(a_opt, f"p4_mix{mi}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: CD polishing (best={best_score:.8f}) ===")

    if best_seq is not None:
        rem = min(deadline - time.monotonic() - 60, 150)
        if rem > 20:
            a_opt, _ = coord_descent_polish(best_seq, time_limit=rem)
            try_seq(a_opt, "p5_cd")

    # === Phase 6: Final L-BFGS + CD cycle ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # L-BFGS with very low temp
        rem = min(deadline - time.monotonic() - 30, 60)
        if rem > 10:
            tau_final = [0.001 / n_best, 0.0003 / n_best, 0.0001 / n_best]
            a_opt, _ = run_lbfgs_anneal(best_seq, tau_final, time_limit=rem)
            try_seq(a_opt, "p6_lbfgs")

        # CD again
        if best_seq is not None:
            rem2 = min(deadline - time.monotonic() - 10, 60)
            if rem2 > 10:
                a_opt, _ = coord_descent_polish(best_seq, time_limit=rem2)
                try_seq(a_opt, "p6_cd")

        # One more L-BFGS from CD result
        if best_seq is not None:
            n_best2 = len(best_seq)
            rem3 = min(deadline - time.monotonic() - 5, 30)
            if rem3 > 8:
                tau_final2 = [0.0005 / n_best2, 0.0001 / n_best2]
                a_opt, _ = run_lbfgs_anneal(best_seq, tau_final2, time_limit=rem3)
                try_seq(a_opt, "p6_final_lbfgs")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
