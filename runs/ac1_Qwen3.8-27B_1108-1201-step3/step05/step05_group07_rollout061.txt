"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_obj_and_grad(a, T):
    """
    Compute smoothed objective and gradient.
    Objective: (2n/T) * log(sum_k exp(T * b[k])) where b = conv(a,a)
    This approximates 2n * max(b) as T -> infinity.
    Returns (obj, grad) where grad is w.r.t. a (on the simplex).
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: grad_a[i] = 2n * sum_k w[k] * a[k-i]
    # This is 2n * (cross-correlation of w and a)[i]
    # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] (with a indexed 0..n-1)
    # For k from 0 to 2n-2, a[k-i] is valid when 0 <= k-i <= n-1
    
    # Use FFT for efficiency
    # We need: g[i] = sum_{k=0}^{2n-2} w[k] * a[k-i] for i = 0..n-1
    # This is the cross-correlation of w (len 2n-1) and a (len n)
    # g[i] = sum_k w[k] * a[(k-i) mod ...] but with linear (not circular)
    
    # Actually, let's think of it as: 
    # g = ifft(conj(fft(w, N)) * fft(a, N))[:n] where N >= 2n-1
    # Wait, cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    # In FFT: ifft(conj(fft(w)) * fft(a)) gives cross-correlation
    
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g = irfft(np.conj(W) * A, fft_size)[:n]
    
    grad = 2.0 * n * g
    return obj, grad

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    a_safe = np.maximum(a, 1e-15)
    return np.log(a_safe)

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_optimize_z(z0, T, n, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer in log-space."""
    t0 = time.monotonic()
    z = z0.copy().astype(np.float64)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_obj = np.inf
    best_z = z0.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax_from_z(z)
        obj, grad_a = compute_obj_and_grad(a, T)
        
        # Chain rule for log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        
        # Skip if gradient is too large (numerical instability)
        if np.any(np.isnan(grad_z)) or np.max(np.abs(grad_z)) > 1e6:
            lr = max(lr * 0.5, 1e-6)
            continue
        
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if obj < best_obj:
            best_obj = obj
            best_z = z.copy()
    
    return best_z, best_obj

def adam_optimize_simplex(a0, T, n, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer directly on the simplex (with projection)."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_obj = np.inf
    best_a = a0.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad = compute_obj_and_grad(a, T)
        
        # Project gradient to tangent space of simplex
        S = np.dot(grad, a)
        grad_tangent = grad - a * S
        
        m = beta1 * m + (1 - beta1) * grad_tangent
        v = beta2 * v + (1 - beta2) * grad_tangent**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent on the simplex."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[np.arange(n) != j] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    a = a_trial
                    best_a = a.copy()
                    improved = True
                    old_val = a[j]  # Update for next multiplier
    
    return best_a, best_obj

def peak_shave(a0, time_limit=60.0):
    """Iteratively reduce the highest peaks."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        if M <= best_obj * 1.001:
            break
        
        # Find the lag with the maximum
        k = np.argmax(b)
        
        # Find which pair contributes most
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k - idx]
        max_local = np.argmax(contributions)
        i = idx[max_local]
        j = k - i
        
        # Reduce the larger of a[i], a[j]
        if a[i] >= a[j]:
            target = i
        else:
            target = j
        
        old_val = a[target]
        if old_val < 1e-10:
            break
        
        # Try reducing
        for factor in [0.9, 0.95, 0.98]:
            new_val = old_val * factor
            diff = old_val - new_val
            if diff < 1e-15:
                continue
            mask = np.ones(n, dtype=bool)
            mask[target] = False
            others = a[mask]
            if others.sum() < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[target] = new_val
            a_trial[mask] = others + diff * others / others.sum()
            a_trial = a_trial / a_trial.sum()
            
            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-12:
                a = a_trial
                best_obj = val
                best_a = a.copy()
                break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
smoothed gradient optimization (soft-max objective) with L-BFGS and Adam,
temperature annealing from low to very high T, diverse initializations,

active-set subgradient). Key improvements over prior: better gradient (fixed factor),
Adam for robustness, warm-started temperature schedule, and targeted local search.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_w_a(w, a):
    """
    Compute r[j] = sum_k w[k] * a[k-j] for j=0..n-1.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    N = len(w)
    fft_size = 1 << (N - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-15))


def smooth_obj_grad_z(z, T):
    """
    Compute smoothed objective and gradient w.r.t. z (log-space).
    f_T = (2n/T) * log(sum_k exp(T * b[k]))
    df/dz_j = a_j * (4n * r_j - 4n * sum_i a_i * r_i)
    where r = cross_corr(w, a), w_k = exp(T*b_k)/sum exp(T*b_k)
    """
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    S = exp_Tb.sum()
    w = exp_Tb / S
    log_sum_exp = max_Tb + np.log(S)
    obj = (2.0 * n / T) * log_sum_exp

    # Gradient w.r.t. a: grad_a = 4n * cross_corr(w, a)
    r = cross_corr_w_a(w, a)
    grad_a = 4.0 * n * r

    # Chain rule for log-space
    S_dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S_dot)

    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_logspace(z0, T, time_limit, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer in log-space for a fixed temperature T."""
    t0 = time.monotonic()
    z = z0.copy().astype(np.float64)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_obj = np.inf
    best_z = z0.copy()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad_z = smooth_obj_grad_z(z, T)

        if np.any(np.isnan(grad_z)) or np.max(np.abs(grad_z)) > 1e4:
            lr = max(lr * 0.5, 1e-7)
            continue

        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_z = z.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive learning rate
        if stall > 200:
            lr *= 0.9
            stall = 0
        elif stall > 50:
            # Small random kick
            z += np.random.standard_normal(len(z)) * 0.001

    return best_z, best_obj


def lbfgs_logspace(z0, T, time_limit, maxiter=2000):
    """L-BFGS in log-space for a fixed temperature T."""
    t0 = time.monotonic()

    def func(z):
        return smooth_obj_grad_z(z, T)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smooth_obj_grad_z(z, T)[1]

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        if time.monotonic() - t0 < time_limit:
            return res.x, res.fun
    except Exception:
        pass
    return z0, func(z0)


def temperature_schedule_optimize(a0, time_limit, optimizer='adam',
                                   T_list=None, lr=0.005, maxiter=2000):
    """
    Multi-temperature optimization (warm-started).
    """
    t0 = time.monotonic()
    n = len(a0)
    if T_list is None:
        T_list = [5, 10, 20, 50, 100, 200, 500]

    z = z_from_a(a0)
    best_a = a0.copy()
    best_score = np.inf

    per_T = time_limit / len(T_list)
    for i, T in enumerate(T_list):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = min(per_T * 1.5, rem)

        if optimizer == 'adam':
            z, _ = adam_logspace(z, T, t_alloc, lr=lr)
        else:
            z, _ = lbfgs_logspace(z, T, t_alloc, maxiter=maxiter)

        a = softmax_from_z(z)
        a = np.maximum(a, 0)
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    T={T}: score={score:.6f}")

def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent cycling through elements."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.8]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12 or old_val > 0.999:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try reverse order
            for j in range(n - 1, -1, -1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[j]
                if old_val < 1e-12 or old_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                best_j_obj = best_obj
                best_j_val = old_val
                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.999:
                        continue
                    scale = (1.0 - new_val) / denom
                    if scale <= 0:
                        continue
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                if best_j_val != old_val:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True
            if not improved:
                break

    return best_a, best_obj


def peak_shave(a0, time_limit=60.0):
    """Iteratively reduce the highest convolution peaks."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find top peaks (exclude edges)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_k = np.argmax(b_inner)
        peak_val = b[peak_k]
        if peak_val <= best_obj * 1.001:
            break

        # Find contributing pair
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_k - idx]
        max_local = np.argmax(contributions)
        i = idx[max_local]
        j = peak_k - i

        # Reduce the larger element
        target = i if a[i] >= a[j] else j
        old_val = a[target]
        if old_val < 1e-10:
            break

        for factor in [0.9, 0.95, 0.98]:
            new_val = old_val * factor
            diff = old_val - new_val
            if diff < 1e-15:
                continue
            mask = np.ones(n, dtype=bool)
            mask[target] = False
            others = a[mask]
            if others.sum() < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[target] = new_val
            a_trial[mask] = others + diff * others / others.sum()
            a_trial = a_trial / a_trial.sum()
            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-12:
                a = a_trial
                best_obj = val
                best_a = a.copy()
                break

        # Check if we've made no progress
        if best_obj >= peak_val * 0.999:
            break

    return best_a, best_obj


def subgradient_step(a, T_sg, top_k=20):
    """One subgradient step using softmax-weighted peaks."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    scaled = pv / T_sg
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg


def subgradient_polish(a0, time_limit=60.0, lr=0.003, mom=0.9, top_k=30):
    """Subgradient method with momentum."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_T = 0.001

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_step(a, cur_T, top_k)
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = b.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.999
        cur_T *= 1.01

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: diverse initializations, temperature-annealed
    smooth gradient (Adam + L-BFGS), multi-length exploration, and local
    refinement (CD, peak shaving, subgradient).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    n_parent = len(parent)

    # === Phase 1: Diverse starts with short optimization ===
    print("\n=== Phase 1: Diverse starts ===")
    starts = []
    labels = []

    starts.append(parent.copy())
    labels.append("parent")

    starts.append(np.ones(n_parent) / n_parent)
    labels.append("uniform")

    r = rng.random(n_parent) ** 0.5
    starts.append(r / r.sum())
    labels.append("sqrt_rand")

    # U-shaped (edge-concentrated)
    x = np.linspace(-1, 1, n_parent)
    u = x ** 4 + 0.05
    starts.append(u / u.sum())
    labels.append("U4")

    # Triangle
    tri = np.maximum(0, 1 - np.abs(x))
    if tri.sum() > 0:
        starts.append(tri / tri.sum())
        labels.append("triangle")

    # Bimodal
    bim = np.exp(-((x - 0.5) ** 2) / 0.05) + np.exp(-((x + 0.5) ** 2) / 0.05)
    starts.append(bim / bim.sum())
    labels.append("bimodal")

    # Cosine window
    cos_w = np.cos(np.pi * x / 2) ** 2
    cos_w = np.maximum(cos_w, 1e-6)
    starts.append(cos_w / cos_w.sum())
    labels.append("cosine")

    # Periodic modulation
    per = 0.5 + 0.4 * np.cos(2 * np.pi * np.arange(n_parent) * 5.0 / n_parent)
    per = np.maximum(per, 0.01)
    starts.append(per / per.sum())
    labels.append("periodic5")

    #ifferent period
    per2 = 0.5 + 0.4 * np.cos(2 * np.pi * np.arange(n_parent) * 7.0 / n_parent)
    per2 = np.maximum(per2, 0.01)
    starts.append(per2 / per2.sum())
    labels.append("periodic7")

    phase1_budget = (deadline - time.monotonic()) * 0.25
    per_start = max(10.0, phase1_budget / len(starts))

    for i, (sp, lbl) in enumerate(zip(starts, labels)):
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(per_start, rem)
        print(f"  [{lbl}] t={t_alloc:.0f}s")
        a_opt, _ = temperature_schedule_optimize(
            sp, t_alloc, optimizer='adam',
            T_list=[5, 20, 50], lr=0.008)
        try_seq(a_opt, f"adam_{lbl}")

    # === Phase 2: Deep refinement on best ===
    print(f"\n=== Phase 2: Deep refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            print(f"  Adam deep: T=[10,20,50,100,200,500], t={rem:.0f}s")
            a_opt, _ = temperature_schedule_optimize(
                best_seq, rem * 0.8, optimizer='adam',
                T_list=[10, 20, 50, 100, 200, 500], lr=0.005)
            try_seq(a_opt, "deep_adam")

    # === Phase 3: L-BFGS refinement ===
    print(f"\n=== Phase 3: L-BFGS refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            n_b = len(best_seq)
            z0 = z_from_a(best_seq)
            T_list_lbfgs = [20, 50, 100, 300]
            per_T = rem * 0.7 / len(T_list_lbfgs)
            z_curr = z0
            for T in T_list_lbfgs:
                rem2 = deadline - time.monotonic() - 200
                if rem2 < 10:
                    break
                t_alloc = min(per_T, rem2)
                z_curr, _ = lbfgs_logspace(z_curr, T, t_alloc, maxiter=3000)
                a_curr = softmax_from_z(z_curr)
                try_seq(a_curr, f"lbfgs_T{T}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in sorted([n_best * 2, max(128, n_best // 2), 1024, 2048]):
            if target_n < 64 or target_n > 4096:
                continue
            if target_n == n_best:
                continue
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.35, 90.0)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = temperature_schedule_optimize(
                f_interp, t_alloc, optimizer='adam',
                T_list=[10, 30, 100, 300], lr=0.006)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Local refinement ===
    print(f"\n=== Phase 5: Local refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_ps = min(rem * 0.35, 50.0)
            a_ps, _ = peak_shave(best_seq, time_limit=t_ps)
            try_seq(a_ps, "peak_shave")

        if best_seq is not None:
            rem = deadline - time.monotonic() - 80
            if rem > 15:
                t_cd = min(rem * 0.5, 45.0)
                a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(a_cd, "cd")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            n_b = len(best_seq)
            t_sg = min(rem * 0.6, 50.0)
            a_sg, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                         lr=0.002, mom=0.92,
                                         top_k=max(15, n_b // 15))
            try_seq(a_sg, "sg")

    # === Phase 7: Final high-T Adam ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = temperature_schedule_optimize(
                best_seq, rem, optimizer='adam',
                T_list=[100, 300, 800], lr=0.003)
            try_seq(a_opt, "final_adam")

        # One more CD pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_cd, _ = coordinate_descent(best_seq, time_limit=rem)
            try_seq(a_cd, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
