   ```
       # Solve: max sum(b) s.t. (a*b)(j) <= M for all j, b >= 0
       # Using linprog: minimize -sum(b) s.t. (a*b)(j) <= M, b >= 0
       
       n = len(a)
       # Constraints: for each j in [0, 2n-2], sum_i a_i * b_{j-i} <= M
       # The coefficient of b_k in constraint j is a_{j-k} (if valid)
       
       # Build constraint matrix A (shape: (2n-1, n))
       # A[j, k] = a[j-k] if 0 <= j-k < n, else 0
       
       A = np.zeros((2*n-1, n))
       for j in range(2*n-1):
           for k in range(n):
               idx = j - k
               if 0 <= idx < n:
                   A[j, k] = a[idx]
       
       # Objective: minimize -sum(b) => c = -ones(n)
       c = -np.ones(n)
       
       # Bounds: b >= 0
       bounds = [(0, None)] * n
       
       # RHS
       b_rhs = np.full(2*n-1, M)
       
       res = linprog(c, A_ub=A, b_ub=b_rhs, bounds=bounds, method='highs')
       
       if res.success:

       return None
   ```

```python
"""
LP-based optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Uses iterative linear programming (cutting-plane linearization of the
quadratic convolution constraint) combined with coordinate descent polishing.
The key innovation is solving an LP at each step: max sum(b) s.t. (a*b)(j) <= M,
which provides a principled improvement direction for the auto-convolution peak.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
from scipy.sparse import coo_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_lp_matrix(a):
    """Build sparse constraint matrix A where A[j,k] = a[j-k] if 0<=j-k<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        k_start = max(0, j - n + 1)
        k_end = min(n - 1, j)
        for k in range(k_start, k_end + 1):
            idx = j - k
            if a[idx] > 1e-15:
                rows.append(j)
                cols.append(k)
                vals.append(a[idx])
    if not rows:
        return csr_matrix((2 * n - 1, n))
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return csr_matrix(A)


def lp_solve(a, M, max_bound=None):
    """Solve LP: max sum(b) s.t. A @ b <= M, b >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    if max_bound is None:
        min_pos = a[a > 1e-10].min() if (a > 1e-10).any() else 1e-10
        max_bound = max(10.0 * M / max(min_pos, 1e-10), 10.0)
    bounds = [(0.0, max_bound)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iterate(a, time_limit, verbose=False, max_iters=5000):
    """Iterative LP optimization: repeatedly solve LP and blend."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        b = lp_solve(a, M)
        if b is None:
            break

        if b.sum() <= a.sum() * (1 + 1e-8):
            stagnant += 1
            if stagnant > 20:
                break
            # Try with slightly larger M
            b = lp_solve(a, M * 1.001)
            if b is None or b.sum() <= a.sum() * (1 + 1e-8):
                if stagnant > 50:
                    break
                continue
        else:
            stagnant = 0

        b = b / b.sum()

        # Line search: try several blend ratios
        found_improvement = False
        for t in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            c = (1.0 - t) * a + t * b
            c = normalize(c)
            sc = compute_score(c)
            if sc < best_score - 1e-10:
                a = c
                best_score = sc
                best_a = a.copy()
                found_improvement = True
                break

        if not found_improvement:
            # Try pure LP solution with different scaling
            b_scaled = b * (a.sum() / b.sum())
            sc = compute_score(b_scaled)
            if sc < best_score - 1e-10:
                a = normalize(b_scaled)
                best_score = sc
                best_a = a.copy()
            else:
                stagnant += 1
                if stagnant > 100:
                    break

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}")

    return best_a, best_score


def coord_descent(a, time_limit, verbose=False, max_passes=50):
    """Coordinate descent with golden-section search per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            lo = max(0.0, old_val * 0.5)
            hi = min(0.99, old_val * 2.0)
            if hi - lo < 1e-8:
                lo = max(0.0, old_val - 0.02)
                hi = min(0.99, old_val + 0.02)

            def obj_v(v):
                v = max(0.0, min(1.0, v))
                new_a = a.copy()
                new_a[j] = v
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_a)
                return conv.max()

            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 25})
                if res.success:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        new_a[mask] *= (1.0 - v_new) / other_sum
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-12:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if not improved:
            break

    return best_a, best_score


def pair_swap_local(a, time_limit, n_trials=5000):
    """Random pair transfer local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(999)
    t0 = time.monotonic()

    for _ in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        i, j = rng.integers(0, n, 2)
        if i == j or a[i] < 1e-10 or a[j] < 1e-10:
            continue

        best_delta = 0.0
        best_sc = best_score
        for f in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = a[j] * f
            new_a = a.copy()
            new_a[j] -= delta
            new_a[i] += delta
            sc = compute_score(new_a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_delta = delta

        if best_delta != 0.0:
            a[j] -= best_delta
            a[i] += best_delta
            a = normalize(a)
            best_score = best_sc
            best_a = a.copy()

    return best_a, best_score


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    return normalize(a)


def symmetric_geometric(n, r):
    """Symmetric geometric (bilateral exponential) sequence."""
    c = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64) - c
    a = r ** np.abs(x)
    return normalize(a)


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def power_law_sequence(n, p):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-p)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def full_optimize(a, time_limit, verbose=False):
    """Multi-stage: LP iterate -> CD -> pair swap -> CD polish."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Stage 1: LP iteration (main workhorse)
    t1 = min(time_limit * 0.5, 600)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_lp, sc_lp = lp_iterate(a, time_limit=min(t1, rem), verbose=verbose)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
            a = a_lp.copy()
        if verbose:
            print(f"      After LP: {best_score:.10f}")

    # Stage 2: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.3, 300)
    if rem > 20:
        a_cd, sc_cd = coord_descent(a, time_limit=min(t2, rem), verbose=verbose)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
        if verbose:
            print(f"      After CD: {best_score:.10f}")

    # Stage 3: Pair swap
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.2, 120)
    if rem > 15:
        a_ps, sc_ps = pair_swap_local(a, time_limit=min(t3, rem))
        if sc_ps < best_score:
            best_score = sc_ps
            best_a = a_ps.copy()
            a = a_ps.copy()
        if verbose:
            print(f"      After PS: {best_score:.10f}")

    # Stage 4: More LP (now from a better starting point)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 30:
        a_lp2, sc_lp2 = lp_iterate(a, time_limit=min(rem * 0.5, rem - 20), verbose=verbose)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()
            a = a_lp2.copy()
        if verbose:
            print(f"      After LP2: {best_score:.10f}")

    # Stage 5: Final CD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_final, sc_final = coord_descent(a, time_limit=min(rem, 60))
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        if verbose:
            print(f"      After final CD: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses LP-based iterative optimization as the core method,
    with diverse initializations, multiple dimensions, and multi-stage polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    print("=== LP-Based Optimization ===")
    print(f"Budget: {budget_s}s, deadline at {deadline - time.monotonic():.0f}s")

    # Generate diverse starting points
    print("\n--- Generating starts ---")
    starts = []

    # Try the parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    # Geometric sequences (various ratios and lengths)
    for n in [512, 640, 768, 896, 1024]:
        for r in [0.997, 0.998, 0.9985, 0.999, 0.9993]:
            starts.append((geometric_sequence(n, r), f"geom_r{r}_n{n}"))

    # Symmetric geometric
    for n in [512, 768, 1024]:
        for r in [0.995, 0.997, 0.998, 0.999]:
            starts.append((symmetric_geometric(n, r), f"symgeom_r{r}_n{n}"))

    # Power law
    for n in [768, 1024]:
        for p in [0.5, 0.7, 1.0, 1.3, 1.5]:
            starts.append((power_law_sequence(n, p), f"pow_p{p}_n{n}"))

    # Cosine
    for n in [768, 1024]:
        starts.append((cosine_sequence(n), f"cos_n{n}"))

    # Random smooth
    for n in [768, 1024]:
        a = rng.random(n)
        kernel = np.ones(20) / 20
        a = np.convolve(a, kernel, mode='same')
        starts.append((normalize(a), f"rand_n{n}"))

    # Quick evaluation of all starts
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:8]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # Select diverse top starts for optimization
    print("\n--- Phase 1: Optimize diverse starts ---")
    selected = []
    selected_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 8:
            break
        # Extract type prefix
        parts = label.split('_')
        base = parts[0]
        if base not in selected_types:
            selected.append((a, label, sc))
            selected_types.add(base)

    # Ensure parent is included
    if not any(label == "parent" for _, label, _ in selected):
        for sc, a, label in start_scores:
            if label == "parent":
                selected.insert(0, (a, label, sc))
                break

    # Allocate time
    n_sel = len(selected)
    total_avail = deadline - time.monotonic() - 400
    t_per = max(30.0, min(300.0, total_avail / max(1, n_sel)))

    for a_init, label, sc_init in selected:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}] (init={sc_init:.6f}): t={t_alloc:.0f}s")
        a_opt, sc_opt = full_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # Phase 2: Dimension exploration on best
    print(f"\n--- Phase 2: Dimension exploration (best={best_score:.8f}) ---")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = [512, 640, 768, 896, 1024, 1280]
        target_ns = [tn for tn in target_ns if tn != best_n]
        for target_n in target_ns[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 180)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = full_optimize(a_interp, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"dim_{target_n}")

    # Phase 3: Intensive polishing on the best
    print(f"\n--- Phase 3: Intensive polishing (best={best_score:.8f}) ---")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            # LP again with more iterations
            t_lp = min(rem * 0.4, 300)
            print(f"  Intensive LP: t={t_lp:.0f}s")
            a_final, sc = lp_iterate(best_a, time_limit=t_lp, verbose=True, max_iters=10000)
            try_seq(a_final, "intensive_lp")

            rem = deadline - time.monotonic() - 40
            if rem > 15 and best_a is not None:
                # CD polish
                t_cd = min(rem * 0.4, 120)
                a_cd, sc = coord_descent(best_a, time_limit=t_cd)
                try_seq(a_cd, "intensive_cd")

            rem = deadline - time.monotonic() - 30
            if rem > 10 and best_a is not None:
                # Local perturbations + reoptimize
                n_best = len(best_a)
                for trial in range(5):
                    rem = deadline - time.monotonic() - 20
                    if rem < 15:
                        break
                    noise = rng.standard_normal(n_best) * 0.003
                    a_pert = np.maximum(best_a + noise, 0.0)
                    a_pert = normalize(a_pert)
                    t_small = min(rem * 0.5, 60)
                    a_opt, sc = full_optimize(a_pert, time_limit=t_small)
                    try_seq(a_opt, f"perturb_{trial}")

    # Final verification
    if best_a is None:
        best_a = geometric_sequence(768, 0.9985)
        best_score = compute_score(best_a)

    # Clip to valid range and verify
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return [float(x) for x in best_a]
```
