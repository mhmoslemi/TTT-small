def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a, conv=None, top_k=5):
    """Compute subgradient of max(conv(a,a)) using top-k active positions.
    Returns (gradient, active_indices, conv_values_at_active).
    """
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    
    # Find top-k positions
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Compute subgradient as average of gradients at active positions
    # grad of (a*a)_k at position i is 2*a[k-i] (with a[m]=0 for m out of range)
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        # (a*a)_k = sum_j a_j * a_{k-j}
        # d/d a_i = 2*a_{k-i} (for valid i)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(peaks)
    
    return grad, peaks, conv


def mirror_descent_step(a, grad, eta):
    """Mirror descent (multiplicative weights) step.
    a_new = a * exp(-eta * grad) / sum(a * exp(-eta * grad))

    grad_clipped = np.clip(grad, -100, 100)
    log_update = np.exp(-eta * grad_clipped)
    a_new = a * log_update
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def newton_equalization(a0, time_limit, verbose=False):
    """Newton-based equalization of top convolution peaks.
    Tries to make the top-k convolution values equal while reducing the max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        curr_obj = conv.max()
        
        # Find top-k peaks to equalize
        k_eq = min(5, len(conv))
        peaks = np.argpartition(conv, -k_eq)[-k_eq:]
        peaks = np.sort(peaks)
        
        # Check if peaks are already close
        peak_vals = conv[peaks]
        if np.std(peak_vals) < 1e-10 * peak_vals[0]:
            # Peaks are already equalized, try a subgradient step
            break
        
        # Newton step: find direction d that equalizes the peaks
        # We want: (a+td)*(a+td) at peaks to be equal
        # Linearize: conv(a)_k + t * (gradient at k) . d = constant for all k in peaks
        # This gives: (grad_k - grad_l) . d = conv_l - conv_k for all pairs (k,l)
        
        # Build the system
        m = len(peaks)
        if m < 2:
            break
            
        # Use the first peak as reference
        ref = peaks[0]
        A_sys = np.zeros((m - 1, n))
        b_sys = np.zeros(m - 1)
        
        for idx in range(m - 1):
            k = peaks[idx + 1]
            # grad of (conv)_k at position i is 2*a[k-i]
            # grad of (conv)_ref at position i is 2*a[ref-i]
            for i in range(n):
                jk = k - i
                jr = ref - i
                val_k = 2.0 * a[jk] if 0 <= jk < n else 0.0
                val_r = 2.0 * a[jr] if 0 <= jr < n else 0.0
                A_sys[idx, i] = val_k - val_r
            b_sys[idx] = conv[ref] - conv[k]
        
        # Solve: minimize ||d||^2 subject to A_sys @ d = b_sys and 1^T d = 0
        # Using Lagrange multipliers: d = A_sys^T * lambda + nu * 1
        # This is a linear system in (lambda, nu)
        
        # Simpler: just solve the least squares with the constraint sum(d) = 0
        # d = pinv(A_sys) @ b_sys, then project to sum(d)=0
        d, _, _, _ = np.linalg.lstsq(A_sys, b_sys, rcond=None)
        d -= d.mean()  # ensure sum(d) = 0
        
        # Step size: try a few values
        best_t = 0
        best_obj_trial = curr_obj
        for t in [0.1, 0.05, 0.02, 0.01, 0.5]:
            a_trial = project_simplex(a + t * d, target_sum=1.0)
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_obj_trial:
                best_obj_trial = obj_trial
                best_t = t
        
        if best_t > 0:
            a = project_simplex(a + best_t * d, target_sum=1.0)
            if best_obj_trial < best_obj - 1e-14:
                best_obj = best_obj_trial
                best_a = a.copy()
        else:
            # No improvement from equalization, do a small subgradient step
            grad, _, _ = subgradient_of_max_conv(a, conv, top_k=k_eq)
            grad_c = grad - grad.mean()
            a = project_simplex(a - 0.01 * grad_c, target_sum=1.0)
            obj_check = fast_convolve(a).max()
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a.copy()
            else:
                break  # stuck
        
    return best_a, best_obj

                if new_val < 0 or new_val > 0.99:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 1:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
