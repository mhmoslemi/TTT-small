"""
Multi-phase optimizer for minimizing 2n*max(conv(p,p))/(sum p)^2.
Uses soft-max L-BFGS with aggressive temperature annealing, active-set
refinement, SLSQP polishing, and diverse initializations including
structured windows and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft, fft, ifft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient(w, p):
    """Compute grad[j] = sum_k w[k] * p[k-j] for j=0..n-1 via FFT cross-correlation."""
    n = len(p)
    size = 2 * n - 1
    M = 1 << (size - 1).bit_length()
    W = fft(w, M)
    P = fft(p, M)
    c = np.real(ifft(W * np.conj(P)))[:n]
    return c


def soft_max_obj_grad_z(z, T):
    """
    Compute soft-max smoothed objective and gradient w.r.t. z (log-space).
    p = softmax(z).
    Objective: (2n/T) * log(sum_k exp(T * B_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    p = exp_z / exp_z.sum()
    
    b = fast_convolve(p)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_p = 2.0 * n * compute_gradient(w, p)
    
    S = np.dot(grad_p, p)
    grad_z = p * (grad_p - p * S)
    
    return obj, grad_z


def lbgfs_optimize(z0, T, time_limit, maxiter=5000):
    """L-BFGS optimization in log-space with time limit."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad_z(z, T)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad_z(z, T)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x, res.fun

def z_from_a(a):
    """Convert probability vector to log-space."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def softmax_from_z(z):
    """Convert log-space to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step(p, T=0.01, lr=0.002, top_k=50):
    """One subgradient step to reduce peak of convolution."""
    n = len(p)
    b = fast_convolve(p)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    scaled = pv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Build weighted w vector for gradient computation
    w_full = np.zeros(2 * n - 1)
    for i in range(len(peaks)):
        w_full[int(peaks[i])] += w[i]
    
    grad = 2.0 * n * compute_gradient(w_full, p)
    
    S = np.dot(grad, p)
    grad_simplex = grad - S  # project gradient to tangent space of simplex
    
    p_new = p - lr * grad_simplex
    p_new = project_simplex(p_new)
    return p_new


def active_set_optimize(p0, time_limit, max_iter=200):
    """
    Active set method: identify top peaks and minimize their maximum
    using projected gradient with adaptive learning rate.
    """
    t0 = time.monotonic()
    n = len(p0)
    p = p0.copy()
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    lr = 0.005
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(p)
        cur_max = b.max()
        
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            lr *= 0.8
            if no_improve > 200:
                break
            continue
        
        # Find top peaks
        top_k = min(20, len(b))
        peak_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_vals = b[peak_idx]
        max_val = peak_vals.max()
        
        # Weight peaks by how close they are to the max
        w = np.exp((peak_vals - max_val) / 0.001)
        w /= w.sum()
        
        # Compute gradient
        w_full = np.zeros(2 * n - 1)
        for i in range(len(peak_idx)):
            w_full[int(peak_idx[i])] += w[i]
        
        grad = 2.0 * n * compute_gradient(w_full, p)
        S = np.dot(grad, p)
        grad_tan = grad - S
        
        # Adaptive step size (backtracking)
        step = lr
        for _ in range(20):
            p_trial = project_simplex(p - step * grad_tan)
            b_trial = fast_convolve(p_trial)
            if b_trial.max() < cur_max - 1e-15:
                break
            step *= 0.5
        
        p = p_trial
        lr = max(step * 1.05, 0.0001)
    
    return best_p, best_obj


def slsqp_refine(p0, max_val, time_limit=60.0):
    """
    SLSQP refinement: minimize max(conv(p,p)) subject to the constraint
    that all B[k] <= max_val (slightly relaxed).
    """
    t0 = time.monotonic()
    n = len(p0)
    
    def obj(p_inner):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        b = fast_convolve(p_inner)
        return b.max()
    
    def grad_obj(p_inner):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(p_inner)
        b = fast_convolve(p_inner)
        k_max = np.argmax(b)
        # Gradient of B[k_max]
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] += p_inner[k_max - idx]
        if k_max - idx != idx:
            grad[k_max - idx] += p_inner[idx]
        # Symmetrize: dB[k]/dp[j] = p[k-j] + p[j-k]... 
        # Actually B[k] = sum_i p[i]*p[k-i], dB[k]/dp[j] = sum_i [delta(i,j)*p[k-i] + delta(k-i,j)*p[i]]
        # = p[k-j] + p[k-j] (if k-2j is valid... no)
        # = p[k-j] (from i=j term) + p[k-j] (from k-i=j, i.e., i=k-j term)
        # But if j = k-j, i.e., k=2j, we get p[j] (only once since i=j=k-j)
        # Actually: sum_i [delta(i,j)*p[k-i]] = p[k-j]
        #           sum_i [delta(k-i,j)*p[i]] = p[k-j] (when i=k-j, which is valid if k-j is in [0,n-1])
        # So dB[k]/dp[j] = 2*p[k-j] if j != k-j, and p[k-j] = p[j] if j = k-j
        # Wait no: if j = k-j (i.e., k = 2j), then i=j gives p[k-j]=p[j], and i=k-j=j also gives p[j].
        # But these are the same term! So it's just p[j], not 2*p[j].
        # Let me redo: B[k] = sum_i p[i]*p[k-i]. d/dp[j]:
        # = sum_i [delta(i,j)*p[k-i] + p[i]*delta(k-i,j)]
        # = p[k-j] + sum_i p[i]*delta(k-i,j)
        # = p[k-j] + p[k-j]  (from i = k-j)
        # = 2*p[k-j]
        # But wait, if j = k-j (k=2j), the two terms are the same: p[j] + p[j]... 
        # No! The first sum gives p[k-j] = p[j]. The second sum: i ranges over valid indices, 
        # and we need k-i=j, so i=k-j=j. So we get p[i]=p[j]. Total: p[j]+p[j] = 2*p[j].
        # Hmm but B[k] = sum_i p[i]*p[k-i]. If k=2j, the term i=j is p[j]*p[j] = p[j]^2.
        # d/dp[j] of p[j]^2 = 2*p[j]. So yes, it's 2*p[k-j] always.
        
        # Let me just compute it properly
        grad = np.zeros(n)
        # For each j in valid range:
        for j in idx:
            k_minus_j = k_max - j
            if 0 <= k_minus_j < n:
                grad[j] += 2.0 * p_inner[k_minus_j]
        
        # Project to tangent space
        S = np.dot(grad, p_inner)
        grad = grad - S
        return grad
    
    def constraint(p_inner):
        b = fast_convolve(p_inner)
        return max_val - b.max()
    
    constraints = [
        {'type': 'ineq', 'fun': constraint},
        {'type': 'eq', 'fun': lambda p: np.sum(p) - 1.0}
    ]
    
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(
            obj, p0, jac=grad_obj, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
        )
        if res.success or (hasattr(res, 'fun') and res.fun < fast_convolve(p0).max()):
            p_res = project_simplex(np.maximum(res.x, 0))
            p_res = p_res / p_res.sum()
            return p_res
    except Exception:
        pass
    
    return p0


def slsqp_direct(p0, time_limit=60.0):
    """
    Direct SLSQP: minimize max(conv(p,p)) with sum=1 and p>=0.
    Uses a smooth upper bound as proxy.
    """
    t0 = time.monotonic()
    n = len(p0)
    
    def smooth_obj(p_inner):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        b = fast_convolve(p_inner)
        T = 200.0
        Tb = T * b
        max_Tb = Tb.max()
        return (2.0 * n / T) * (max_Tb + np.log(np.exp(Tb - max_Tb).sum())) - (2.0 * n / T) * np.log(2.0 * n - 1)
    
    def smooth_grad(p_inner):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(p_inner)
        b = fast_convolve(p_inner)
        T = 200.0
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        grad = 2.0 * n * compute_gradient(w, p_inner)
        S = np.dot(grad, p_inner)
        return grad - S
    
    try:
        res = minimize(
            fun=smooth_obj,
            x0=p0,
            jac=smooth_grad,
            method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda p: np.sum(p) - 1.0}],
            options={'maxiter': 500, 'ftol': 1e-14, 'disp': False}
        )
        if hasattr(res, 'x'):
            p_res = project_simplex(np.maximum(res.x, 0))
            if p_res.sum() > 1e-15:
                p_res = p_res / p_res.sum()
                b_res = fast_convolve(p_res)
                if b_res.max() < fast_convolve(p0).max() * 1.001:
                    return p_res
    except Exception:
        pass
    
    return p0


def coordinate_descent(p0, time_limit=60.0):
    """Coordinate descent with line search on each element."""
    t0 = time.monotonic()
    n = len(p0)
    p = p0.copy()
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    multipliers = np.array([0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.8])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = p[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                scale = (1.0 - new_val) / denom
                p_trial = p.copy()
                p_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                p_trial[mask] *= scale
                p_trial = p_trial / p_trial.sum()
                
                val = fast_convolve(p_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                p[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                p[mask] *= scale
                p = p / p.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_p = p.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_p, best_obj


def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular (Bartlett)
    if n % 2 == 1:
        tri = np.concatenate([np.linspace(0, 1, (n+1)//2), np.linspace(1, 0, (n-1)//2)])
    else:
        tri = np.concatenate([np.linspace(0, 1, n//2), np.linspace(1, 0, n//2 - 1)])
    tri += 0.01
    inits.append(tri / tri.sum())
    
    # Cosine
    x = np.linspace(0, np.pi, n)
    cos_w = 0.5 * (1 + np.cos(x))
    cos_w += 0.01
    inits.append(cos_w / cos_w.sum())
    
    # Gaussian
    x = np.linspace(0, 1, n)
    gauss = np.exp(-5 * (x - 0.5)**2)
    gauss += 0.01
    inits.append(gauss / gauss.sum())
    
    # Two-lobe (bimodal)
    x = np.linspace(0, 1, n)
    bi = np.exp(-30 * (x - 0.25)**2) + np.exp(-30 * (x - 0.75)**2)
    bi += 0.01
    inits.append(bi / bi.sum())
    
    # Edge-heavy
    x = np.linspace(0, 1, n)
    edge = np.exp(5 * (2*x - 1)**2)
    edge += 0.01
    inits.append(edge / edge.sum())
    
    # Random
    r = rng.random(n)
    r += 0.01
    inits.append(r / r.sum())
    
    # Near-uniform with small perturbation
    nu = np.ones(n) + rng.standard_normal(n) * 0.1
    nu = np.maximum(nu, 0.01)
    inits.append(nu / nu.sum())
    
    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Broad exploration with diverse initializations and lengths
    Phase 2: Deep L-BFGS with temperature annealing
    Phase 3: Active-set refinement + SLSQP
    Phase 4: Final polish with coordinate descent and high-T L-BFGS
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        print("No parent, using uniform n=512")
    
    # === Phase 1: Broad exploration ===
    print("\n=== Phase 1: Broad exploration ===")
    t1_start = time.monotonic()
    t1_budget = min(300.0, (deadline - t1_start) * 0.18)
    
    target_lengths = [256, 384, 512, 640, 768, 1024, 1280, 1536]
    # Always include parent length
    if len(parent) not in target_lengths:
        target_lengths.append(len(parent))
    target_lengths.sort()
    
    candidates = []
    
    for n in target_lengths:
        if time.monotonic() - t1_start > t1_budget - 30:
            break
        
        # Generate initializations
        inits = make_initializations(n, rng)
        
        # If we have a parent, add interpolated version
        if parent is not None and len(parent) != n:
            inits.append(interpolate_sequence(parent, n))
        elif parent is not None:
            inits.append(parent.copy())
        
        for idx, p_init in enumerate(inits):
            if time.monotonic() - t1_start > t1_budget - 20:
                break
            
            t_alloc = min(30.0, (t1_budget - (time.monotonic() - t1_start)) / max(1, len(inits) - idx))
            if t_alloc < 10:
                t_alloc = 10
            
            z0 = z_from_a(p_init)
            z_opt, _ = lbgfs_optimize(z0, T=30.0, time_limit=t_alloc, maxiter=1500)
            p_opt = softmax_from_z(z_opt)
            
            score = try_seq(p_opt, f"P1_n{n}_init{idx}")
            if np.isfinite(score):
                candidates.append((score, p_opt.copy()))
    
    # Sort candidates and keep top few
    if candidates:
        candidates.sort(key=lambda x: x[0])
        print(f"Phase 1 complete: top score = {candidates[0][0]:.6f}")
    
    # Select starting point for Phase 2
    if best_seq is not None:
        p_start = best_seq.copy()
    else:
        p_start = parent.copy()
    
    # === Phase 2: Deep L-BFGS with temperature annealing ===
    print(f"\n=== Phase 2: Deep L-BFGS (current best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t2_budget = min(900.0, (deadline - t2_start) * 0.45)
    
    n_start = len(p_start)
    z_curr = z_from_a(p_start)
    
    # Temperature schedule: start moderate, increase aggressively
    temps = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    
    # Allocate time: more time for higher temperatures
    weights = [1, 1, 1.5, 2, 2, 2.5, 3, 3, 3]
    total_w = sum(weights)
    
    for i, T in enumerate(temps):
        rem = deadline - time.monotonic() - 400  # reserve for phases 3,4
        if rem < 15:
            break
        
        t_alloc = min(t2_budget * weights[i] / total_w, rem / max(1, len(temps) - i), 120.0)
        if t_alloc < 10:
            continue
        
        print(f"  T={T}, n={n_start}, t={t_alloc:.0f}s")
        z_curr, obj = lbgfs_optimize(z_curr, T, t_alloc, maxiter=3000)
        p_curr = softmax_from_z(z_curr)
        try_seq(p_curr, f"P2_T{T}")
        
        # Periodically try intermediate evaluations
        if i % 2 == 1 and time.monotonic() - t2_start < t2_budget:
            rem = deadline - time.monotonic() - 350
            if rem > 20:
                t_quick = min(20.0, rem * 0.3)
                z_check, _ = lbgfs_optimize(z_curr, temps[min(i+1, len(temps)-1)], t_quick, maxiter=1000)
                p_check = softmax_from_z(z_check)
                score_check = try_seq(p_check, f"P2_check_T{temps[min(i+1,len(temps)-1)]}")
                if score_check < best_score:
                    z_curr = z_check
    
    # Try a different length if time permits
    rem = deadline - time.monotonic() - 350
    if rem > 60 and best_seq is not None:
        # Try scaling up
        new_n = len(best_seq) * 2
        if new_n <= 4096:
            print(f"  Trying n={new_n} (2x upscale)")
            p_up = interpolate_sequence(best_seq, new_n)
            z_up = z_from_a(p_up)
            t_alloc = min(60.0, rem * 0.4)
            z_up, _ = lbgfs_optimize(z_up, 50, t_alloc, maxiter=2000)
            p_up = softmax_from_z(z_up)
            try_seq(p_up, f"P2_upscale_n{new_n}")
            
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                t_alloc2 = min(40.0, rem * 0.5)
                z_up, _ = lbgfs_optimize(z_up, 200, t_alloc2, maxiter=2000)
                p_up = softmax_from_z(z_up)
                try_seq(p_up, f"P2_upscale_n{new_n}_T200")
    
    # === Phase 3: Active-set + SLSQP refinement ===
    print(f"\n=== Phase 3: Active-set + SLSQP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            # Active set optimization
            t_as = min(rem * 0.35, 180.0)
            print(f"  Active set: t={t_as:.0f}s")
            p_as, obj_as = active_set_optimize(best_seq, time_limit=t_as)
            try_seq(p_as, "P3_active_set")
            
            # SLSQP direct
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                t_sl = min(rem * 0.4, 120.0)
                print(f"  SLSQP direct: t={t_sl:.0f}s")
                p_sl = slsqp_direct(best_seq, time_limit=t_sl)
                try_seq(p_sl, "P3_slsqp")
            
            # SLSQP with constraint
            rem = deadline - time.monotonic() - 100
            if rem > 15 and best_seq is not None:
                b_best = fast_convolve(best_seq)
                max_best = b_best.max() * 1.001  # slightly relaxed bound
                t_sl2 = min(rem * 0.5, 60.0)
                print(f"  SLSQP constrained: t={t_sl2:.0f}s")
                p_sl2 = slsqp_refine(best_seq, max_best, time_limit=t_sl2)
                try_seq(p_sl2, "P3_slsqp_constrained")
    
    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Coordinate descent
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_cd = min(rem * 0.3, 45.0)
            print(f"  Coordinate descent: t={t_cd:.0f}s")
            p_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(p_cd, "P4_coord_descent")
        
        # High-T L-BFGS
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            z_final = z_from_a(best_seq)
            t_final = min(rem * 0.4, 30.0)
            print(f"  Final L-BFGS T=5000: t={t_final:.0f}s")
            z_f, _ = lbgfs_optimize(z_final, 5000, t_final, maxiter=2000)
            p_f = softmax_from_z(z_f)
            try_seq(p_f, "P4_final_T5000")
        
        # Subgradient polish
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            p = best_seq.copy()
            t_sg = min(rem * 0.5, 20.0)
            t0 = time.monotonic()
            lr = 0.001
            while time.monotonic() - t0 < t_sg:
                p = subgradient_step(p, T=0.005, lr=lr, top_k=30)
                b_p = fast_convolve(p)
                if b_p.max() < fast_convolve(best_seq).max() - 1e-14:
                    try_seq(p, "P4_subgrad")
                lr *= 0.999
                if lr < 0.0001:
                    lr = 0.0005
                    break
        
        # Final perturbation attempts
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            for attempt in range(5):
                rem = deadline - time.monotonic() - 5
                if rem < 4:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                p_pert = project_simplex(best_seq + noise)
                z_pert = z_from_a(p_pert)
                t_p = min(rem * 0.4, 10.0)
                z_p, _ = lbgfs_optimize(z_pert, 1000, t_p, maxiter=1000)
                p_r = softmax_from_z(z_p)
                try_seq(p_r, f"P4_perturb_{attempt}")
        
        # One more active set pass
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_as2 = min(rem * 0.6, 20.0)
            p_as2, _ = active_set_optimize(best_seq, time_limit=t_as2, max_iter=100)
            try_seq(p_as2, "P4_final_active_set")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
