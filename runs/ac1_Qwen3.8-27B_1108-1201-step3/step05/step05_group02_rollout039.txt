import time

# Helper functions (fast_convolve, project_simplex, solve_lp_direction, etc.)

def smooth_gd_nesterov(f0, time_limit, T0=0.001, T_min=1e-6, beta=0.9, **kwargs):
    """Smooth gradient descent with Nesterov acceleration and temperature annealing."""
    f = normalize(f0)
    n = len(f)
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(n)
    
    T = T0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute smooth gradient
        conv = fast_convolve(y)
        # Log-sum-exp gradient
        shifted = (conv - conv.max()) / T
        weights = np.exp(np.clip(shifted, -50, 50))
        weights /= weights.sum()
        
        # Compute gradient
        grad = compute_grad(y, weights)
        
        # Nesterov update
        beta = 0.5 * (1 + np.sqrt(1 + 4 * it * it)) / (1 + np.sqrt(1 + 4 * it * it))
        # Actually, standard Nesterov: beta_t = (1 + sqrt(1+4t^2)) / (2t)
        v = beta * v + (1 - beta) * grad  # momentum form
        # Or: v = momentum * v + grad; y = f - lr * v
        
        lr = 1.0 / (np.linalg.norm(grad) * np.sqrt(it + 1) + 1e-10)
        y = project_simplex(y - lr * v)
        
        # Refresh y from f periodically
        # (Standard Nesterov: y_{t+1} = (1-beta) f_{t+1} + beta * y_{t+1}... this is getting complex)
        
        f = y  # Simplified
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        T = max(T * 0.999, T_min)
        
        if no_improve > 200:
            # Restart
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            T = T0 * 0.5
            no_improve = 0
    
    return best_f, best_obj

For `f(a) = max_k (a*a)[k]`, the subgradient at point a is:
`sum_{k in argmax} grad_k(a)` (or average over argmax)

where `grad_k(a)[i] = a[k-i] + a[i]` (with appropriate range handling).

`grad f_T(a)[i] = sum_k p_k * (a[k-i] + a[i])` where `p_k = exp((a*a)[k]/T) / sum_l exp((a*a)[l]/T)`

where `(p * a)[i]_cross = sum_k p_k * a[k-i]` (with a[j] = 0 outside [0,n-1]).

    grad[i] = sum_k weights[k] * (a[k-i] + a[i]) with range handling.

def smooth_gd(f0, time_limit, T0=0.0008, T_min=1e-8,
              momentum=0.85, lr_scale=1.0, verbose=False, label=""):
    """
    Smooth gradient descent with momentum and temperature annealing.
    Minimizes max(auto-conv(a)) for a on the simplex.
    """
    f = normalize(f0.copy())
    n = len(f)
    conv_len = 2 * n - 1

    v = np.zeros(n)
    T = T0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    momentum_cur = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        max_conv = conv.max()

        # Compute log-sum-exp weights (soft attention over peaks)
        shifted = (conv - max_conv) / T
        weights = np.exp(np.clip(shifted, -60, 0))
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.zeros(conv_len)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= ws

        # Compute gradient via FFT
        grad = compute_smooth_grad_fft(f, weights)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break

        # Adaptive learning rate
        lr = lr_scale / (grad_norm * np.sqrt(1 + 0.01 * it))
        lr = min(lr, 0.5)

        # Momentum update
        v = momentum_cur * v - lr * grad
        f_new = project_simplex(f + v)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} it={it} t={elapsed:.0f}s obj={best_obj:.12f} T={T:.2e}")
        else:
            no_improve += 1

        f = f_new

        # Anneal temperature
        T = max(T * 0.9997, T_min)

        # Slowly increase momentum
        momentum_cur = min(momentum_cur + 0.0001, 0.97)

        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            scale = 0.002 * min(no_improve // 300, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            T = max(T0 * 0.3, T)
            no_improve = 0
            momentum_cur = momentum

    return best_f, best_obj


def solve_lp_direction(f, M, n, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # For each k, the coefficients of g in (f*g)[k] are:
    # (f*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j] * g[k-j]
    # Let i = k-j, then j = k-i: (f*g)[k] = sum_i f[k-i] * g[i]
    # So the row for constraint k has entry f[k-i] at column i (where valid)
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[k, i] = f[k - i]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': time_limit}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_step(f, M, n, relax=1.01):
    """Single LP-FW step: find LP direction and do line search."""
    g = solve_lp_direction(f, M * relax, n, time_limit=20)
    if g is None:
        g = solve_lp_direction(f, M * 1.05, n, time_limit=20)
    if g is None:
        return f, M

    g_sum = g.sum()
    if g_sum < 1e-15:
        return f, M
    g = g / g_sum

    if np.max(np.abs(g - f)) < 1e-12:
        return f, M

    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = np.maximum(f + alpha * (g - f), 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-10:
        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        return f_new, best_val
    return f, M


def coordinate_descent(f0, time_limit, verbose=False):

    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def structured_init(n, seed=0, pattern="uniform"):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)

    if pattern == "uniform":
        f = np.ones(n) / n
    elif pattern == "bimodal":
        f = np.zeros(n)
        f[:n//4] = 1.0
        f[3*n//4:] = 1.0
        f = normalize(f)
    elif pattern == "gaussian":
        x = np.arange(n) / n
        f = np.exp(-4 * (x - 0.5) ** 2)
        f = normalize(f)
    elif pattern == "ramp":
        f = np.arange(1, n + 1, dtype=float)
        f = normalize(f)
    elif pattern == "combo":
        x = np.arange(n) / n
        f = 1.0 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.sin(8 * np.pi * x)
        f = np.maximum(f, 0.01)
        f = normalize(f)
    elif pattern == "noise":
        f = rng.random(n) + 0.1
        f = normalize(f)
    elif pattern == "alternating":
        f = np.ones(n)
        f[::2] *= 1.5
        f[1::2] *= 0.5
        f = normalize(f)
    else:
        f = np.ones(n) / n

    return f


def hybrid_optimize(f0, time_limit, verbose=False, label=""):
    """
    Hybrid optimizer: smooth GD + periodic LP-FW + CD.
    The main workhorse is smooth GD (fast), with LP-FW for escaping plateaus.
    """
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    phase_cd_counter = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # Determine which method to use
        elapsed = time.monotonic() - t0
        if elapsed < time_limit * 0.6:
            # Main phase: smooth GD
            t_gd = min(rem * 0.4, 120)
            if t_gd < 5:
                break
            f_opt, obj = smooth_gd(f, time_limit=t_gd,
                                   T0=max(0.001, fast_convolve(f).max() * 0.1),
                                   T_min=1e-8, momentum=0.85,
                                   lr_scale=1.0, verbose=verbose, label=label)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_opt.copy()
                if verbose:
                    print(f"  {label} GD: obj={best_obj:.12f} t={time.monotonic()-t0:.0f}s")
            f = f_opt
        elif elapsed < time_limit * 0.8:
            # Middle phase: LP-FW steps
            M = fast_convolve(f).max()
            f_new, M_new = lp_fw_step(f, M, n, relax=1.02)
            if M_new < best_obj - 1e-12:
                best_obj = M_new
                best_f = f_new.copy()
                if verbose:
                    print(f"  {label} LP-FW: obj={best_obj:.12f} t={time.monotonic()-t0:.0f}s")
            f = f_new
            phase_cd_counter += 1
            if phase_cd_counter % 3 == 0:
                # Occasional CD
                rem2 = time_limit - (time.monotonic() - t0)
                if rem2 > 10:
                    f_cd, obj_cd = coordinate_descent(f, time_limit=min(8, rem2 * 0.2))
                    if obj_cd < best_obj - 1e-12:
                        best_obj = obj_cd
                        best_f = f_cd.copy()
                        f = f_cd
                        if verbose:
                            print(f"  {label} CD: obj={best_obj:.12f} t={time.monotonic()-t0:.0f}s")
        else:
            # Final phase: coordinate descent polish
            rem2 = time_limit - (time.monotonic() - t0)
            if rem2 > 5:
                f_cd, obj_cd = coordinate_descent(f, time_limit=rem2 * 0.7)
                if obj_cd < best_obj - 1e-12:
                    best_obj = obj_cd
                    best_f = f_cd.copy()
                    if verbose:
                        print(f"  {label} CD final: obj={best_obj:.12f}")
                f = f_cd
                # One more LP-FW at the end
                rem3 = time_limit - (time.monotonic() - t0)
                if rem3 > 10:
                    M = fast_convolve(f).max()
                    f_new, M_new = lp_fw_step(f, M, n, relax=1.0)
                    if M_new < best_obj - 1e-12:
                        best_obj = M_new
                        best_f = f_new.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Try parent sequence with hybrid optimization
    Phase 2: Explore nearby lengths
    Phase 3: Diverse structured initializations
    Phase 4: Deep refinement on best found
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Hybrid optimize parent ===
    print(f"\n=== Phase 1: Hybrid parent n={n_parent} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = hybrid_optimize(parent, time_limit=t1, verbose=True, label="P1")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: Explore nearby lengths ===
    print(f"\n=== Phase 2: Nearby lengths (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    target_lengths = []
    for delta in [-128, -64, -32, 32, 64, 128, 256, -256, 512]:
        nl = n_best + delta
        if 128 <= nl <= 3000 and nl != n_best:
            target_lengths.append(nl)
    # Also try some "natural" lengths
    for nl in [256, 512, 768, 1024, 1536]:
        if nl not in target_lengths and 128 <= nl <= 3000:
            target_lengths.append(nl)

    for target_n in sorted(target_lengths)[:6]:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(120, rem * 0.25)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = hybrid_optimize(f_interp, time_limit=t_alloc, verbose=False, label=f"L{target_n}")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")
        else:
            print(f"  n={target_n}: score={sc:.6f}")

    # === Phase 3: Diverse structured initializations ===
    print(f"\n=== Phase 3: Diverse inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    patterns = ["gaussian", "bimodal", "combo", "alternating", "ramp", "noise"]

    for pat in patterns:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        f_init = structured_init(n_best, seed=seed + hash(pat) % 1000, pattern=pat)
        # Blend with current best for warm start
        f_init = normalize(0.7 * best_seq + 0.3 * f_init)
        t_alloc = min(100, rem * 0.2)
        if t_alloc < 15:
            continue
        print(f"  pattern={pat}, t={t_alloc:.0f}s")
        a_opt, _ = hybrid_optimize(f_init, time_limit=t_alloc, verbose=False, label=f"{pat}")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{pat}]: {sc:.6f}")
        else:
            print(f"  {pat}: score={sc:.6f}")

    # === Phase 4: Deep refinement ===
    print(f"\n=== Phase 4: Deep refinement (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 30:
        t_phase4 = min(rem * 0.6, 300)
        print(f"  t={t_phase4:.0f}s")
        a_opt, _ = hybrid_optimize(best_seq, time_limit=t_phase4, verbose=True, label="P4")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 5: Multiple short smooth-GD runs with different temperatures ===
    print(f"\n=== Phase 5: T-sweep (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for T_try in [0.0002, 0.0005, 0.001, 0.003]:
        rem = deadline - time.monotonic() - 40
        if rem < 10:
            break
        f_pert = normalize(best_seq + rng.standard_normal(n_best) * 0.001)
        t_alloc = min(60, rem * 0.3)
        if t_alloc < 8:
            continue
        a_opt, obj = smooth_gd(f_pert, time_limit=t_alloc, T0=T_try,
                               T_min=T_try * 0.01, momentum=0.88,
                               lr_scale=1.5, verbose=False, label=f"T{T_try}")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [T={T_try}]: {sc:.6f}")
        else:
            print(f"  T={T_try}: score={sc:.6f}")

    # === Phase 6: Final coordinate descent + LP-FW ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_cd = min(rem * 0.5, 80)
        a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  CD: {sc:.6f}")

    rem = deadline - time.monotonic() - 10
    if rem > 15:
        M = fast_convolve(best_seq).max()
        for relax in [1.0, 1.005, 1.01, 1.02]:
            f_new, M_new = lp_fw_step(best_seq, M, n_best, relax=relax)
            if M_new < fast_convolve(best_seq).max() - 1e-12:
                best_seq = f_new
                M = M_new
                sc = float(evaluate_sequence(best_seq.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    print(f"  LP-FW relax={relax}: {sc:.6f}")
            else:
                break

    # === Phase 7: Final micro-perturbations ===
    print(f"\n=== Phase 7: Micro-perturb (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.0001
        f_pert = normalize(best_seq + rng.standard_normal(n_best) * scale)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  pert{trial}: {sc:.6f}")
            # Quick GD polish
            rem2 = deadline - time.monotonic()
            if rem2 > 8:
                a_opt, _ = smooth_gd(f_pert, time_limit=min(rem2 * 0.4, 8),
                                     T0=0.0001, T_min=1e-8,
                                     momentum=0.9, lr_scale=1.0, verbose=False, label="micro")
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  pert{trial}_gd: {sc2:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
