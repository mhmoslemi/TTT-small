def project_simplex(v):
    """ Euclidean projection onto the simplex {x >= 0, sum(x)=1} """
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cum_sum = np.cumsum(u) - 1
    candidate_indices = np.arange(1, n+1)
    non_cond_idx = candidate_indices - 1
    cond = candidate_indices * u[non_cond_idx] > cum_sum
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(non_cond_idx[cond])
    theta = (cum_sum[rho-1] + 1) / rho if rho > 0 else 0.0
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M=1.001, n=None):
    """ Solve the linear program: max{sum(g)} s.t. (convolve(f, g))[k] <= M for all k, and g >= 0"""
    if n is None:
        n = len(f)
    # Build the constraint matrix: A_ub has rows corresponding to convolution shifts (0 to 2*n-2) and columns corresponding to indices in f
    # Each row i corresponds to the convolution coefficient at shift i, and involves a contiguous block of the current sequence
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        A_ub[j+j:j+j+n] = f[j]  # j+j is the shift for j, and we take a block of length n starting at j+j

    b_ub = np.ones(2*n-1) * M

    # Solve the LP: minimize -sum(g) subject to binding constraints and simplex projection
    c = [-1] * n  # Coefficients for objective minimization (negative because we want to maximize sum(g))
    opt = {'time_limit': 60, 'maxiter': 200, 'disp': False}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options=opt)
    if res.success and res.x is not None:

        # Check if the binding constraints are not trivially satisfied by zero
        if g_sum > 1e-15:
