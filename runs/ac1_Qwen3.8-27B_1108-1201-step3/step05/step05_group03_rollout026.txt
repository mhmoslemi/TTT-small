import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a, conv_cache=None):
    """Compute 2*n*max_conv/sum^2 (same as evaluate_sequence but faster)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    if conv_cache is None:
        conv = fast_convolve(a)
    else:
        conv = conv_cache
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a, conv, top_k=50):
    """Compute subgradient of max_conv with respect to a."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    # Use softmax weighting for smoothness
    pv = conv[peaks]
    temp = max(1e-8, (pv.max() - pv.min()) * 0.1)
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for i in range(k_actual):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        # conv_k = sum_j a_j * a_{k-j}, dconv_k/da_j = 2*a_{k-j} if valid
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        reversed_idx = k - idx
        sg[idx] += 2.0 * w[i] * a[reversed_idx]
    
    return sg


def mirror_descent(a, time_limit, lr0=0.01, decay=0.9998, top_k=50, restart_every=3000, verbose=False):
    """Mirror descent (multiplicative weights) with averaging and restarts."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    
    avg_a = np.zeros(n)
    avg_count = 0
    cur_lr = lr0
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        conv = fast_convolve(a)
        sg = subgradient_of_max_conv(a, conv, top_k)
        
        # Multiplicative update
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-18)
        a = a / a.sum()
        
        # Update average
        avg_a += a
        avg_count += 1
        
        # Check current and averaged
        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        
        if avg_count % 100 == 0:
            avg_candidate = avg_a / avg_count
            sc_avg = compute_score(avg_candidate)
            if sc_avg < best_score - 1e-14:
                best_score = sc_avg
                best_a = avg_candidate.copy()
        
        cur_lr *= decay
        
        # Restart
        if step % restart_every == 0:
            a = normalize(best_a + np.random.standard_normal(n) * 0.005)
            avg_a = a.copy()
            avg_count = 1
            cur_lr = lr0 * 0.7
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD step {step}: score={best_score:.10f}, lr={cur_lr:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def coord_descent(a, time_limit, verbose=False):
    """Precise coordinate descent with exact 1D optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    no_improve_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            # Define objective: set a[j] = v, rescale others to maintain sum=1
            # After rescaling, compute max_conv
            other_sum = 1.0 - old_val
            if other_sum < 1e-15:
                continue
            
            def obj_v(v):
                if v < 0 or v > 1.0:
                    return float('inf')
                v = float(v)
                scale = (1.0 - v) / other_sum
                new_a = a * scale
                new_a[j] = v
                conv = fast_convolve(new_a)
                return conv.max()
            
            # Determine search bounds
            if old_val < 1e-14:
                lo, hi = 0.0, min(0.05, 1.0)
            else:
                lo = max(0.0, old_val * 0.5)
                hi = min(1.0 - 1e-10, old_val * 2.0)
                if hi <= lo:
                    lo = max(0.0, old_val - 0.02)
                    hi = min(1.0 - 1e-10, old_val + 0.02)
            
            if hi <= lo:
                continue
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 50})
                if res.success:
                    v_new = res.x
                    scale = (1.0 - v_new) / other_sum
                    new_a = a * scale
                    new_a[j] = v_new
                    new_a = np.maximum(new_a, 0.0)
                    new_a = new_a / new_a.sum()
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-15:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
        else:
            no_improve_count = 0
    
    return best_a, best_score


def block_descent(a, time_limit, block_size=16, verbose=False):
    """Block coordinate descent: optimize groups of coordinates jointly."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()

    iteration = 0

def simulated_annealing(a, time_limit, verbose=False):
    """Simulated annealing for escaping local minima."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    T = 0.01  # Initial temperature
    cooling = 0.999
    cur_score = best_score
    steps = 0
    
    while time.monotonic() - t0 < time_limit:
        T *= cooling
        if T < 1e-12:
            T = best_score * 0.001
            # Restart from best
            a = normalize(best_a + rng.standard_normal(n) * 0.01)
        
        # Random perturbation
        pert_type = rng.random()
        if pert_type < 0.5:
            # Single coordinate
            j = rng.integers(0, n)
            delta = rng.standard_normal() * 0.01
            a[j] = max(0, a[j] + delta)
        elif pert_type < 0.8:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                delta = min(a[j], 0.05) * rng.random()
                a[j] -= delta
                a[i] += delta
        else:
            # Small Gaussian noise
            a = a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
        
        a = a / a.sum()
        new_score = compute_score(a)
        steps += 1
        
        # Accept/reject
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a.copy()
        else:
            accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-15))
            if rng.random() < accept_prob:
                cur_score = new_score
            else:
                # Reject: revert (we already modified a, so recompute)
                # For simplicity, just continue from current state
                pass
        
        if verbose and steps % 10000 == 0:
            print(f"      SA step {steps}: best={best_score:.10f}, T={T:.6f}")
    
    return best_a, best_score


def parametric_init(n, method='auto', **kwargs):
    """Generate parametric initial sequences."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'exponential':
        r = kwargs.get('r', 0.999)
        a = r ** np.arange(n)
        return a / a.sum()
    elif method == 'cosine':
        a = np.sin(np.pi * np.arange(n) / n) ** 2
        return a / a.sum()
    elif method == 'gaussian':
        sigma = kwargs.get('sigma', 0.3)
        x = (np.arange(n) - n/2) / (sigma * n)
        a = np.exp(-x**2 / 2)
        return a / a.sum()
    elif method == 'power_law':
        alpha = kwargs.get('alpha', 0.5)
        a = (np.arange(n) + 1) ** (-alpha)
        return a / a.sum()
    elif method == 'raised_cosine':
        beta = kwargs.get('beta', 2.0)
        a = (1.0 + np.cos(2 * np.pi * np.arange(n) / n)) ** beta
        return a / a.sum()
    elif method == 'superposition':
        # Sum of shifted Gaussians
        n_bumps = kwargs.get('n_bumps', 3)
        a = np.zeros(n)
        for i in range(n_bumps):
            center = n * (i + 0.5) / n_bumps
            sigma = n / (n_bumps * 4)
            a += np.exp(-((np.arange(n) - center) ** 2) / (2 * sigma ** 2))
        return a / a.sum()
    elif method == 'bipolar_pos':
        # Take absolute value of a "noise-like" sequence
        rng = np.random.default_rng(kwargs.get('seed', 42))
        a = np.abs(rng.standard_normal(n))
        # Smooth
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    elif method == 'chirp_pos':
        # Positive part of a chirp-like pattern
        a = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n) ** 2 / n)
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def optimize_full(a, time_limit, verbose=False):
    """Full multi-stage optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    def remaining():
        return max(0, time_limit - (time.monotonic() - t0))
    
    # Stage 1: Mirror descent (exploratory)
    t1 = min(remaining() * 0.25, 200)
    if t1 > 10:
        a_md, sc_md = mirror_descent(a, time_limit=t1, lr0=0.005, decay=0.9999,
                                      top_k=max(20, n // 5), restart_every=2000, verbose=verbose)
        if sc_md < best_score:
            best_score = sc_md
            best_a = a_md.copy()
            a = a_md.copy()
        if verbose:
            print(f"    After MD: {best_score:.10f}")
    
    # Stage 2: Coordinate descent
    t2 = min(remaining() * 0.35, 300)
    if t2 > 10:
        a_cd, sc_cd = coord_descent(a, time_limit=t2, verbose=verbose)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
        if verbose:
            print(f"    After CD: {best_score:.10f}")
    
    # Stage 3: Block descent
    t3 = min(remaining() * 0.2, 100)
    if t3 > 10:
        a_bd, sc_bd = block_descent(a, time_limit=t3, block_size=max(8, n // 32), verbose=verbose)
        if sc_bd < best_score:
            best_score = sc_bd
            best_a = a_bd.copy()
            a = a_bd.copy()
        if verbose:
            print(f"    After BD: {best_score:.10f}")
    
    # Stage 4: Simulated annealing
    t4 = min(remaining() * 0.15, 80)
    if t4 > 10:
        a_sa, sc_sa = simulated_annealing(a, time_limit=t4, verbose=verbose)
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa.copy()
            a = a_sa.copy()
        if verbose:
            print(f"    After SA: {best_score:.10f}")
    
    # Stage 5: Final CD polish
    t5 = remaining()
    if t5 > 5:
        a_final, sc_final = coord_descent(a, time_limit=t5, verbose=verbose)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        if verbose:
            print(f"    After final CD: {best_score:.10f}")
    
    return best_a, best_score
