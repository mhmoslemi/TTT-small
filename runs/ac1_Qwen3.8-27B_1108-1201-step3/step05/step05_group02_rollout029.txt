def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n

For the log-sum-exp approximation, the gradient is a weighted average of the gradients of the quadratic forms $(f*f)_k$. Each $\nabla (f*f)_k$ has magnitude at most $2\|f\|_1 = 2$ (since $f$ is on the simplex). The second derivative (Hessian) of $(f*f)_k$ is related to the shift operators, and its operator norm is at most 2.

So the gradient of $S(f)$ has Lipschitz constant at most $2\beta \cdot 2 = 4\beta$ (very rough bound). So I should use $\eta \leq 1/(4\beta)$ or so.

But this is very conservative. In practice, the effective Lipschitz constant is much smaller because the weights $w_k$ are concentrated on a few peaks.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with continuation
on the log-sum-exp smooth approximation, combined with LP-FW refinement,
coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(f, beta):
    """Compute gradient of log-sum-exp smooth approximation using FFT."""
    n = len(f)
    conv = fast_convolve(f)  # length 2n-1
    shifted = conv - conv.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All weights equal (beta effectively 0)
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= w_sum
    # grad_i = 2 * sum_k w_k * f_{k-i} for valid indices
    # = 2 * (w cross-correlate f)_i
    # Using: (w cross-correlate f)_i = (w * f_flip)_{n-1+i}
    f_flip = f[n-1::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flip, fft_size)
    h = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * h[n-1:n-1+n]
    return grad


def fista_smooth(f0, time_limit, beta=50.0, verbose=False):
    """FISTA on the smooth log-sum-exp approximation."""
    n = len(f0)
    f = np.maximum(f0.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    t = 1.0
    f_curr = f.copy()
    f_prev = f.copy()
    eta = 1.0 / (4.0 * beta)  # Conservative initial step
    
    no_improve = 0
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = f_curr + (t - 1.0) / t_new * (f_curr - f_prev)
        
        # Gradient step + projection
        grad = smooth_grad_fft(y, beta)
        x_new = y - eta * grad
        f_curr_new = project_simplex(x_new)
        
        # Evaluate true objective
        obj = fast_convolve(f_curr_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_curr_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f_prev = f_curr
        f_curr = f_curr_new
        t = t_new
        
        # Adaptive step size
        if no_improve > 200:
            eta *= 0.7
            no_improve = 0
    
    return best_f, best_obj


def fista_continuation(f0, time_limit, verbose=False):
    """FISTA with continuation on beta."""
    n = len(f0)
    f = np.maximum(f0.copy(), 0.0)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    # Start with small beta, increase over time
    betas = [5, 10, 20, 50, 100, 200, 500]
    
    for i, beta in enumerate(betas):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        t_alloc = rem / (len(betas) - i) * 1.3  # Slight overlap
        f, obj = fista_smooth(f, t_alloc, beta=beta, verbose=verbose)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
        if verbose:
            print(f"    beta={beta}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M, n)
        if g is None:
            g = solve_lp_direction(f, M * 1.05, n)
        if g is None:
            g = solve_lp_direction(f, M * 1.15, n)
        if g is None:
            no_improve += 5
            if no_improve > 100:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                # Add noise to escape
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max()
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            multipliers = [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]
            best_j_val = old_val
            best_j_obj = cur_conv.max()
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

LP-FW refinement, coordinate descent, and multi-length exploration.
Key innovation: FISTA (O(1/k^2) convergence) on smooth approximation of max-convolution
with adaptive beta continuation for faster convergence than plain subgradient or FW.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(f, beta):
    """Compute gradient of log-sum-exp smooth approximation using FFT.
    
    S(f) = (1/beta) * log(sum_k exp(beta * (f*f)_k))
    grad_S(f)_i = 2 * sum_k w_k * f_{k-i}
    where w_k = softmax(beta * (conv - max_conv))
    """
    n = len(f)
    conv = fast_convolve(f)
    shifted = conv - conv.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    # grad_i = 2 * (w cross-correlate f)_i = 2 * (w * f_flip)_{n-1+i}
    f_flip = f[n - 1::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flip, fft_size)
    h = np.fft.irfft(W * F, fft_size)
    grad = 2.0 * h[n - 1:n - 1 + n]
    return np.maximum(grad, -1e15)


def fista_smooth(f0, time_limit, beta=50.0, verbose=False):

    n = len(f0)

    t = 1.0
    f_curr = f.copy()
    f_prev = f.copy()
    # Safe step size based on Lipschitz bound ~ 2 + 2*beta
    eta = 1.0 / (2.0 * beta + 4.0)
    no_improve = 0
    next_report = 50

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        grad = smooth_grad_fft(y, beta)
        x_new = y - eta * grad
        f_curr_new = project_simplex(x_new)
