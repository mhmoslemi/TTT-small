Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp
continuation, softmax parametrization, flat-spectrum initialization,
and LP-based refinement steps.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def fast_crosscorr(b, a, n):
    """Compute y[j] = sum_m b[m+j]*a[m] for j=0..n-1 using FFT."""
    size = 1 << (len(b) + n - 1).bit_length()
    B = np.fft.rfft(b, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(B * np.conj(A), size)[:n]

def softmax_from_x(x):
    """Compute softmax(x) with numerical stability."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    return a / a.sum()

    """
    Computes the log-sum-exp of conv(softmax(x), softmax(x)) and its gradient w.r.t. x.
    """

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def fast_crosscorr(b, a_int, n):
    """Compute y[j] = sum_m b[m+j]*a_int[m] for j=0..n-1."""
    size = 1 << (len(b) + n - 1).bit_length()
    B = np.fft.rfft(b, size)
    A = np.fft.rfft(a_int, size)
    return np.fft.irfft(B * np.conj(A), size)[:n]

def softmax_from_x(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()

    s = a.sum()

        return np.ones(len(x)) / len(x)

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u - (cssv - target_sum) / rho) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1)
    return np.maximum(v - theta, 0.0)

def lp_refine(f, time_limit):
    """
    LP-based refinement step.
    Given f (normalized), solve LP to find b that maximizes sum(b)
    subject to (f*b)_k <= max(f*f) for all k, b >= 0.
    """
    n = len(f)
    M = fast_convolve(f).max()
    
    # Build constraint matrix: (f*b)_k = sum_i f_i * b_{k-i} <= M
    # For k = 0, ..., 2n-2:
    #   sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} f[i] * b[k-i] <= M
    
    # Using scipy's linprog
    from scipy.optimize import linprog
    
    # c = -1 (minimize -sum(b))
    c_vec = -np.ones(n)
    
    # Inequality constraints: A_ub @ b <= b_ub
    # For each k, the constraint is: sum_i f[i] * b[k-i] <= M
    # This is: (row of A_ub) dot b <= M
    
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, k - i] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if result.success:
            b = result.x
            b = np.maximum(b, 0)
            s_b = b.sum()
            if s_b > 1e-10:
                b = b / s_b
                obj_b = fast_convolve(b).max()
                # Mix: f_new = (1-t)*f + t*b
                t = 0.5
                f_new = (1 - t) * f + t * b
                f_new = np.maximum(f_new, 0)
                s_new = f_new.sum()
                if s_new > 1e-10:
                    f_new = f_new / s_new
                    obj_new = fast_convolve(f_new).max()
                    return f_new, obj_new
    except Exception as e:
        print(f"    LP failed: {e}")
    
    return f, fast_convolve(f).max()

                scale = (1.0 - new_val) / remaining

                if best_j_obj < best_obj:
