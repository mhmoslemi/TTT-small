"""
Multi-stage convex optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: FFT-based convolution, log-sum-exp smooth Adam with cyclic temperature,
LP-Frank-Wolfe with adaptive constraints, hot-spot coordinate descent,
and multi-length search with structured restarts.

import numpy as np
from scipy.optimize import linprog
import time as _time


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Gradient and value of logsumexp(conv(f,f)/T)."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    # grad_i = (2/T) * sum_j f[j] * w[i+j] for i=0..n-1
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def smooth_optimize_cyclic(f0, time_limit, T_start=0.3, T_end=0.005, n_cycles=3, verbose=False):
    """Smooth Adam optimization with cyclic temperature schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0

    alpha_base = 0.006
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = _time.monotonic()
    total_time = time_limit
    no_improve = 0
    it = 0
    patience = 4000

    for cycle in range(n_cycles):
        cycle_start = cycle * total_time / n_cycles
        cycle_dur = total_time / n_cycles
        # Temperature for this cycle: start higher, end lower
        T_lo = max(T_end, T_start * (0.3 ** (cycle + 1)))
        T_hi = max(T_lo * 1.5, T_start * (0.3 ** cycle))

        while _time.monotonic() - t0 < total_time:
            it += 1
            elapsed = _time.monotonic() - t0
            cycle_elapsed = max(0.0, elapsed - cycle_start)
            cycle_frac = min(1.0, cycle_elapsed / cycle_dur)
            # Temperature within cycle
            T = T_lo + (T_hi - T_lo) * (1 - cycle_frac)
            T = max(T, T_end)
            # Learning rate: higher at start of cycle, decay within
            lr = alpha_base * ((1.0 - 0.5 * cycle_frac) * (0.8 ** cycle))

            grad, lse_val, conv, w = smooth_grad(f, T)

            t_adam += 1
            m_state = beta1 * m_state + (1 - beta1) * grad
            v_state = beta2 * v_state + (1 - beta2) * grad ** 2
            m_hat = m_state / (1 - beta1 ** t_adam)
            v_hat = v_state / (1 - beta2 ** t_adam)

            f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
            f = project_simplex(f)

            cur_obj = fast_convolve(f).max()

            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"      Smooth cyc{cycle} it{it}: obj={best_obj:.10f}, T={T:.4f}")
            else:
                no_improve += 1

            if no_improve > patience:
                # Structured restart
                rng_p = np.random.default_rng(int(elapsed * 7919) % 999999)
                scale = 0.003 + 0.004 * min(no_improve // 2000, 4)
                # Mix of Gaussian noise and geometric perturbation
                geo = 0.98 ** np.arange(n, dtype=np.float64)
                pert = rng_p.standard_normal(n) * scale + 0.002 * (geo - geo.mean())
                f = project_simplex(best_f + pert)
                m_state = np.zeros(n)
                v_state = np.zeros(n)
                t_adam = 0
                no_improve = 0

    return best_f, best_obj

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True, "time_limit": 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def hot_spot_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting hot-spot convolution indices."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = _time.monotonic()
    passes = 0
    ORDER = 0.6180339887

    def _convolve_max_trial(f_trial, j, x):
        """Compute max conv when f[j] changed to x, others scaled to keep sum=1."""
        old_val = f_trial[j]
        denom = 1.0 - old_val
        if denom < 1e-12:
            return 1e10
        if x < 0 or x > 0.9999:
            return 1e10
        scale = (1.0 - x) / denom
        f_t = f_trial.copy()
        f_t[j] = x
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        f_t[mask] *= scale
        return fast_convolve(f_t).max()

    while _time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        # Find hot spots (top convolution indices)
        top_k = np.argsort(conv)[::-1][:5]

        # Focus on coordinates that contribute to hot spots
        hot_coords = set()
        for k in top_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n and f[i] > 1e-8 and f[j] > 1e-8:
                    if f[i] * f[j] > conv[k] * 0.05:
                        hot_coords.add(i)
                        hot_coords.add(j)

        hot_list = list(hot_coords)
        if len(hot_list) < 10:
            hot_list = list(range(n))
        # Shuffle for diversity
        rng_p = np.random.default_rng(passes * 31 + 7)
        rng_p.shuffle(hot_list)

        for j in hot_list:
            if _time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                continue

            # Golden section search
            lo = 0.0
            hi = 0.999
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            fc = _convolve_max_trial(f, j, c)
            fd = _convolve_max_trial(f, j, d)

            for _ in range(16):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = _convolve_max_trial(f, j, c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = _convolve_max_trial(f, j, d)

            opt_val = (lo + hi) / 2.0
            opt_obj = _convolve_max_trial(f, j, opt_val)

            if opt_obj < best_obj - 1e-13:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 4:
            break

    return best_f, best_obj

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def fw_subgradient(f0, time_limit, verbose=False):
    """Subgradient Frank-Wolfe for the convex problem."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = _time.monotonic()
    it = 0
    no_improve = 0

    while _time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        k_star = np.argmax(conv)
        M_cur = conv[k_star]

        # Subgradient at active constraint: g[i] = 2*f[k_star-i] (valid indices)
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * f[j]
            elif i == k_star and k_star < n:
                g[i] = 2.0 * f[i]  # diagonal term

        # Also consider second and third max for a combined subgradient
        top_indices = np.argsort(conv)[::-1][:3]
        weights = np.zeros(2 * n - 1)
        total = 0
        for rank, ki in enumerate(top_indices):
            w = (0.6 ** rank)
            weights[ki] = w
            total += w
        weights /= total

        g_combined = np.zeros(n)
        for ki in top_indices:
            wk = weights[ki]
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    g_combined[i] += wk * 2.0 * f[j]

        # Use combined subgradient
        g = g_combined

        # FW direction: minimize g^T x over simplex => x = e_{argmin g}
        j_star = np.argmin(g)
        s_vec = np.zeros(n)
        s_vec[j_star] = 1.0

        # Also try "blended" direction toward uniform
        s_uniform = np.ones(n) / n
        s_blended = 0.7 * s_vec + 0.3 * s_uniform

        # Line search over multiple directions
        candidates = [s_vec, s_uniform, s_blended]
        best_alpha = 0.0
        best_val = M_cur
        best_s = None

        for s_dir in candidates:
            for alpha in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                ft = (1 - alpha) * f + alpha * s_dir
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_alpha = alpha
                    best_s = s_dir

        if best_s is not None:
            f_new = (1 - best_alpha) * f + best_alpha * best_s
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            f = f_new
            if best_val < best_obj - 1e-13:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      SubgradFW it{it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 300:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer: diverse initializations, smooth Adam with cyclic
    temperature, LP-Frank-Wolfe, subgradient FW, hot-spot CD, block CD,
    and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = _time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    print(f"=== Phase 0: Diverse Initializations ===")
    # Try various lengths and shapes
    n_candidates = [n_parent, 768]
    if n_parent != 640:
        n_candidates.append(640)
    if n_parent != 900:
        n_candidates.append(900)

    for n_try in n_candidates:
        for r in [0.96, 0.97, 0.98, 0.99, 0.995]:
            _try(make_geometric(n_try, r), f"geo_{r}_{n_try}")
        for r in [0.97, 0.98, 0.99]:
            _try(make_symmetric_geo(n_try, r), f"symgeo_{r}_{n_try}")
        for pw in [1.5, 2.0, 3.0, 4.0]:
            _try(make_cosine(n_try, power=pw), f"cos_{pw}_{n_try}")
        for w in [0.2, 0.25, 0.3, 0.35]:
            _try(make_bell(n_try, width=w), f"bell_{w}_{n_try}")
        _try(make_dual_lobe(n_try), f"dual_{n_try}")

    print(f"  After Phase 0: best={best_score:.6f}")

    # Select top candidates for optimization
    # Use the best as primary, and maybe one diverse one
    primary = best_seq.copy() if best_seq is not None else np.ones(768) / 768
    n_primary = len(primary)

    # Also prepare a diverse candidate
    diverse = make_cosine(n_primary, power=2.5)
    if parent is not None and len(parent) == n_primary:
        diverse = 0.6 * primary + 0.4 * make_bell(n_primary, width=0.25)
        diverse = project_simplex(diverse)

    # === Phase 1: Smooth optimization (primary) ===
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.6f}) ===")
    rem = deadline - _time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 300)
        print(f"  Cyclic smooth n={n_primary}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_cyclic(primary, time_limit=t_alloc,
                                           T_start=0.25, T_end=0.004, n_cycles=3, verbose=True)
        _try(a_opt, "smooth_cyclic_p1")

    rem = deadline - _time.monotonic() - 550
    if rem > 20:
        t_alloc = min(rem, 150)
        print(f"  Standard smooth (diverse init) t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_standard(diverse, time_limit=t_alloc,
                                             T_start=0.3, T_end=0.005)
        _try(a_opt, "smooth_diverse")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "lpfw_p2")

        # LP-FW with perturbation
        rem = deadline - _time.monotonic() - 350
        if rem > 15:
            noise = rng.standard_normal(len(best_seq)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 120)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc)
            _try(a_opt, "lpfw_pert")

    # === Phase 3: Subgradient FW ===
    print(f"\n=== Phase 3: Subgradient FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 250
        if rem > 15:
            t_alloc = min(rem, 120)
            print(f"  SubgradFW t={t_alloc:.0f}s")
            a_opt, _ = fw_subgradient(best_seq, time_limit=t_alloc)
            _try(a_opt, "subgrad_fw")

    # === Phase 4: Targeted coordinate descent ===
    print(f"\n=== Phase 4: Hot-Spot CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - _time.monotonic() - 200
        if rem > 10:
            t_cd = min(rem * 0.5, 80)
            print(f"  Hot-spot CD t={t_cd:.0f}s")
            a_opt, _ = hot_spot_cd(best_seq, time_limit=t_cd)
            _try(a_opt, "hotspot_cd")

        rem = deadline - _time.monotonic() - 160
        if rem > 10:
            t_blk = min(rem * 0.6, 70)
            print(f"  Block CD t={t_blk:.0f}s")
            a_opt, _ = block_cd(best_seq, time_limit=t_blk)
            _try(a_opt, "block_cd")

    # === Phase 5: Longer sequences ===
    print(f"\n=== Phase 5: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [900, 1024, 1200]:
            rem = deadline - _time.monotonic() - 100
            if rem < 40:
                break
            f_interp = interpolate_seq(best_seq, target_n)

            t1 = min(rem * 0.4, 80)
            a_opt, _ = smooth_optimize_cyclic(f_interp, time_limit=t1,
                                               T_start=0.2, T_end=0.01, n_cycles=2)
            _try(a_opt, f"smooth_n{target_n}")

            rem = deadline - _time.monotonic() - 80
            if rem > 25:
                t2 = min(rem * 0.5, 60)
                a_opt, _ = lpfw_optimize(a_opt if a_opt is not None else f_interp,
                                         time_limit=t2)
                _try(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine smooth optimization
        rem = deadline - _time.monotonic() - 50
        if rem > 10:
            a_opt, _ = smooth_optimize_standard(best_seq, time_limit=min(rem, 80),
                                                 T_start=0.03, T_end=0.002)
            _try(a_opt, "smooth_fine")

        # Final LP-FW
        rem = deadline - _time.monotonic() - 30
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 50))
            _try(a_opt, "lpfw_final")

        # Small perturbations and re-optimize
        for trial in range(3):
            rem = deadline - _time.monotonic() - 15
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize_standard(f_pert, time_limit=min(rem, 30),
                                                 T_start=0.02, T_end=0.003)
            _try(a_opt, f"perturb_{trial}")

        # Reversed
        rem = deadline - _time.monotonic() - 10
        if rem > 8:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = smooth_optimize_standard(f_rev, time_limit=min(rem, 20),
                                                 T_start=0.02, T_end=0.002)
            _try(a_opt, "reversed")

        # Blend with uniform
        rem = deadline - _time.monotonic() - 8
        if rem > 6:
            f_blend = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
            f_blend = project_simplex(f_blend)
            a_opt, _ = smooth_optimize_standard(f_blend, time_limit=min(rem, 15),
                                                 T_start=0.015, T_end=0.002)
            _try(a_opt, "blend")

        # Hot-spot CD one more time
        rem = deadline - _time.monotonic() - 5
        if rem > 8:
            a_opt, _ = hot_spot_cd(best_seq, time_limit=min(rem, 15))
            _try(a_opt, "hotspot_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
