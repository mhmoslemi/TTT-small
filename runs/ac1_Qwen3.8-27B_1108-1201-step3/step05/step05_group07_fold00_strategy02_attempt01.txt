def gaussian_pulses(center, std, amplitude, length):
    """
    Generate a sequence of length L representing a Gaussian pulse centered at 'center'
    with standard deviation 'std', amplitude 'amplitude'
    """
    x = np.linspace(0, length-1, length)
    dx = x - center
    g = amplitude * np.exp(-0.5 * (dx / std)**2)
    return g

def density_function_centers(x, centers, radii, gaps, total_shift=40):
