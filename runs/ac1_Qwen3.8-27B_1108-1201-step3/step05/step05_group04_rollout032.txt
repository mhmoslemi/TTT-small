"""
LP-based active set optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Core idea: at each step, solve an LP to find the best direction b (on the simplex)
that minimizes the peak of conv(f, b), then take a step towards b.
This is a cutting-plane style method for the non-smooth non-convex objective.
Additional polishing via coordinate descent and L-BFGS on smooth approximation.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, b):
    """Cross-convolution of f and b via FFT."""
    size = len(f) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(F * B, fft_size)[:size]


def compute_obj_normalized(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    conv = fast_convolve(f)
    return 2.0 * len(f) * conv.max()


def build_conv_matrix(f):
    """Build matrix A (2n-1 x n) such that A @ b = convolve(f, b)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            rows.append(j + i)
            cols.append(j)
            vals.append(f[i])
    A = lil_matrix((2 * n - 1, n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A


def solve_lp_direction(f, M):
    """
    Solve: min M' subject to conv(f,b)_k - M' <= 0, sum(b)=1, b>=0, M'>=0.
    Returns (b_opt, M_opt).
    """
    n = len(f)
    # A matrix for constraints conv(f,b)_k <= M'
    A_conv = build_conv_matrix(f)
    
    # Full A_ub: [A_conv | -1] with b_ub = 0
    # Variables: [b_0,...,b_{n-1}, M']
    n_vars = n + 1
    A_ub = np.zeros((2 * n - 1, n_vars))
    A_ub[:, :n] = A_conv.toarray()
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M'
    c = np.zeros(n_vars)
    c[n] = 1.0
    
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_opt = res.x[:n]
    M_opt = res.x[n]
    
    # Clean up tiny negatives
    b_opt = np.maximum(b_opt, 0)
    s = b_opt.sum()
    if s < 1e-15:
        return None, None
    b_opt /= s
    
    return b_opt, M_opt


def adam_softmax_optimize(f0, time_limit, lr=0.003, temp_init=0.005,
                          temp_decay=0.9995, top_k=None, verbose=False):
    """Adam optimizer with soft-max subgradients on the simplex."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    cur_lr = lr

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        # Soft-max weights
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        # Subgradient: sg[j] = sum_i w[i+j] * f[i]
        sg = np.correlate(w, f, 'valid')

        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            f = np.ones(n) / n
        else:
            f = f_new / fs

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 0)
            f /= f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent: try scaling each coordinate and renormalize others."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    multipliers = [0.5, 0.7, 0.85, 0.9, 1.1, 1.15, 1.3, 1.5]

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Shuffle order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj

                    # Let me just directly compute:
                    # conv_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} x_i * x_{k-i}
                    # d/dx_j of this sum:
                    #   For each term x_i * x_{k-i}:
                    #     if i = j: contributes x_{k-j} (derivative of x_j)
                    #     if k-i = j, i.e., i = k-j: contributes x_{k-j} (derivative of x_j)
                    #     if i = j AND k-i = j (i.e., k=2j): contributes 2*x_j
                    # So:
                    if 0 <= k - j and k - j < n:
                        if k - j == j:
                            grad_x[j] += 2.0 * x[j] * w[k]
                        else:
                            grad_x[j] += x[k - j] * w[k]
                            # Also the "other" term where i = k-j
                            if 0 <= k - j and k - j < n:
                                grad_x[j] += x[k - j] * w[k]
                    # Hmm wait, I think I'm double counting. Let me be very explicit.
                    # conv_k = sum_i x_i * x_{k-i} (sum over valid i)
                    # This is a polynomial in x. d/dx_j picks out terms containing x_j.
                    # Terms containing x_j:
                    #   (a) i = j: term is x_j * x_{k-j}. d/dx_j = x_{k-j}
                    #   (b) k-i = j, i.e., i = k-j: term is x_{k-j} * x_j. d/dx_j = x_{k-j}
                    #   If (a) and (b) are the same term (j = k-j, i.e., k=2j): total derivative = 2*x_j
                    # So:
                    #   if k-j is valid (0 <= k-j < n):
                    #     if k == 2*j: derivative = 2*x_j
                    #     else: derivative = x_{k-j} + x_{k-j} = 2*x_{k-j}
                    #   Wait that's not right either. Let me think again.
                    # 
                    # The sum is: conv_k = x_0*x_k + x_1*x_{k-1} + ... (valid indices only)
                    # Each term x_i * x_{k-i} is a product of two components of x.
                    # d/dx_j of x_i * x_{k-i}:
                    #   = delta(i,j) * x_{k-i} + x_i * delta(k-i, j)
                    #   = delta(i,j) * x_{k-j} + x_i * delta(i, k-j)
                    # Summing over i:
                    #   = x_{k-j} (from i=j, if j is valid) + x_{k-j} (from i=k-j, if k-j is valid and k-j != j)
                    #   = 2*x_{k-j} if k-j != j
                    #   = 2*x_j if k-j == j (i.e., k=2j)
                    # 
                    # But wait, we also need to check that i=j is a valid index (0<=j<n, always true)
                    # and that k-i = k-j is valid (0<=k-j<n).
                    # And for the second term, i=k-j must be valid (0<=k-j<n) and j must be valid.
                    #
                    # So the final answer: d(conv_k)/d(x_j) = 
                    #   2 * x_{k-j}  if 0 <= k-j < n and k-j != j
                    #   2 * x_j      if k-j == j (i.e., k=2j) [which is the same as 2*x_{k-j} since k-j=j]
                    #   0            if k-j < 0 or k-j >= n
                    #
                    # Wait, that simplifies to: 2 * x_{k-j} if 0 <= k-j < n, else 0.
                    # Let me verify with a small example.
                    # n=2, x=[a,b]. conv = [a^2, 2ab, b^2].
                    # d(conv_0)/d(a) = 2a. Formula: 2*x_{0-0} = 2*x_0 = 2a. Correct.
                    # d(conv_1)/d(a) = 2b. Formula: 2*x_{1-0} = 2*x_1 = 2b. Correct.
                    # d(conv_1)/d(b) = 2a. Formula: 2*x_{1-1} = 2*x_0 = 2a. Correct.
                    # d(conv_2)/d(b) = 2b. Formula: 2*x_{2-1} = 2*x_1 = 2b. Correct.
                    # Great!
                    pass  # Already handled above

        # Redo gradient computation cleanly
        grad_x = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            # d(conv_k)/d(x_j) = 2*x_{k-j} if 0 <= k-j < n, else 0
            # So grad_x[j] += 2 * x[k-j] * w[k] for valid k-j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad_x[j] += 2.0 * x[k - j] * w[k]

        # Now chain rule: params -> x (softmax) -> grad
        # x_i = exp(params_i) / sum(exp(params))
        # dx_i/dparams_j = x_i * (delta(i,j) - x_j)
        # grad_params_j = sum_i grad_x_i * dx_i/dparams_j = sum_i grad_x_i * x_i * (delta(i,j) - x_j)
        # = x_j * grad_x_j - x_j * sum_i (grad_x_i * x_i)
        grad_x_weighted = np.sum(grad_x * x)
        grad_params = x * (grad_x - grad_x_weighted)

        return obj, grad_params

    # Initial params: log(f)
    f_safe = np.maximum(f, 1e-15)
    params0 = np.log(f_safe / f_safe.sum())

    t0 = time.monotonic()
    best_obj = compute_obj_normalized(f)
    best_f = f.copy()

    res = minimize(objective_and_grad, params0, method='L-BFGS-B',
                   options={'maxiter': 1000, 'maxfun': 5000, 'ftol': 1e-12,
                            'gtol': 1e-10, 'maxls': 50})

    # Convert back
    x = np.exp(res.x - res.x.max())
    x /= x.sum()
    obj = compute_obj_normalized(x)
    if obj < best_obj:
        best_obj = obj
        best_f = x.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_init(n, rng, idx):
    """Create diverse initializations."""
    x = np.linspace(0, 1, n)
    if idx == 0:
        f = np.ones(n) / n
    elif idx == 1:
        f = np.sin(np.pi * x) ** 2
    elif idx == 2:
        f = np.exp(-0.5 * ((x - 0.5) / 0.15) ** 2)
    elif idx == 3:
        f = np.exp(-0.5 * ((x - 0.3) / 0.1) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.1) ** 2)
    elif idx == 4:
        # Smooth random
        f = np.zeros(n)
        for _ in range(4):
            c = rng.uniform(0.15, 0.85)
            s = rng.uniform(0.08, 0.25)
            a = rng.uniform(0.5, 2.0)
            f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    else:
        f = 1.0 - np.abs(2 * x - 1)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def lp_optimize_main(f0, time_limit, verbose=True):
    """
    Main LP-based optimization loop.
    At each iteration:
    1. Compute M = max(conv(f,f))
    2. Solve LP: min M' s.t. conv(f,b) <= M', sum(b)=1, b>=0
    3. If M' < M, take a step towards b
    4. Backtracking line search
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    current_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    best_obj = current_obj
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    if verbose:
        print(f"    LP start: obj={current_obj:.10f}, n={n}")

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        current_obj = 2.0 * n * M

        # Solve LP
        b_dir, M_prime = solve_lp_direction(f, M)
        
        if b_dir is None:
            # Fallback: use soft-max direction
            temp = 0.01
            shifted = conv / temp
            shifted -= shifted.max()
            w = np.exp(shifted)
            w /= w.sum()
            b_dir = np.correlate(w, f, 'valid')
            b_dir = np.maximum(b_dir, 0)
            bs = b_dir.sum()
            if bs < 1e-15:
                break
            b_dir /= bs
            M_prime = M

        # Check if we've converged
        if M_prime >= M - 1e-12:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 17)
                f = np.maximum(f + rng.standard_normal(n) * 0.003, 0)
                f /= f.sum()
                no_improve = 0
                continue
        else:
            no_improve = 0

        # Line search
        t = 0.5
        accepted = False
        for _ in range(10):
            f_new = (1 - t) * f + t * b_dir
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                t *= 0.5
                continue
            f_new /= fs
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < current_obj - 1e-15:
                f = f_new
                current_obj = obj_new
                accepted = True
                break
            t *= 0.5

        if not accepted and M_prime < M - 1e-10:
            # Take a very small step
            t = 0.01
            f = (1 - t) * f + t * b_dir
            f = np.maximum(f, 0)
            f /= f.sum()
            current_obj = 2.0 * n * fast_convolve(f).max()

        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: obj={best_obj:.10f} (M'={M_prime:.6f})")

    if verbose:
        print(f"    LP done after {it} iters: best={best_obj:.10f}")
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-based active set method (main workhorse)
    Phase 2: Adam soft-max refinement
    Phase 3: Coordinate descent polishing
    Phase 4: L-BFGS on smooth approximation
    Phase 5: Length exploration
    Phase 6: Final micro-polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        # Scale to have sum in a reasonable range for evaluate_sequence
        a_eval = a * 1.0  # already normalized
        try:
            score = float(evaluate_sequence(a_eval.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            try:
                p_score = float(evaluate_sequence(parent.tolist()))
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
                if np.isfinite(p_score):
                    best_score = p_score
                    best_seq = parent.copy()
            except Exception:
                pass
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        print(f"No parent found, using uniform init n={n_init}")
        try_seq(parent, "uniform_init")

    n_parent = len(parent)
    base = best_seq.copy() if best_seq is not None else parent

    # === Phase 1: LP-based optimization on parent length ===
    print(f"\n=== Phase 1: LP optimization (n={n_parent}, best={best_score:.8f}) ===")
    t1_end = time.monotonic() + 600

    a_opt, obj = lp_optimize_main(base, time_limit=min(300, t1_end - time.monotonic()), verbose=True)
    try_seq(a_opt, "p1_lp")

    # Second LP run from the result (with perturbation for diversity)
    rem = t1_end - time.monotonic()
    if rem > 30:
        pert = np.maximum(base + rng.standard_normal(n_parent) * 0.01, 0)
        pert /= pert.sum()
        a_opt2, _ = lp_optimize_main(pert, time_limit=min(150, rem), verbose=False)
        try_seq(a_opt2, "p1_lp_pert")

    # Also try from a diverse init at same length
    rem = t1_end - time.monotonic()
    if rem > 30:
        diverse = make_diverse_init(n_parent, rng, 3)  # two-bump
        a_opt3, _ = lp_optimize_main(diverse, time_limit=min(100, rem), verbose=False)
        try_seq(a_opt3, "p1_diverse")

    base = best_seq.copy() if best_seq is not None else base
    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: Adam refinement ===
    print(f"\n=== Phase 2: Adam refinement (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + 300
    n_base = len(base)

    configs = [
        {'lr': 0.003, 'temp_init': 0.004, 'temp_decay': 0.9996, 'top_k': max(10, n_base // 20)},
        {'lr': 0.005, 'temp_init': 0.008, 'temp_decay': 0.9994, 'top_k': max(20, n_base // 10)},
        {'lr': 0.002, 'temp_init': 0.003, 'temp_decay': 0.9997, 'top_k': max(5, n_base // 30)},
    ]

    for ci, cfg in enumerate(configs):
        rem = t2_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.6, 100)
        print(f"  Adam cfg {ci}, t={t_alloc:.0f}s")
        a_opt, _ = adam_softmax_optimize(base, time_limit=t_alloc, verbose=False, **cfg)
        try_seq(a_opt, f"p2_adam_{ci}")
        base = best_seq.copy() if best_seq is not None else base
        n_base = len(base)

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Coordinate descent polishing ===
    print(f"\n=== Phase 3: CD polishing (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 200

    if best_seq is not None:
        rem = t3_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.7, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p3_cd")

        # CD from perturbed start
        rem = t3_end - time.monotonic()
        if rem > 15 and best_seq is not None:
            n_b = len(best_seq)
            pert = np.maximum(best_seq + rng.standard_normal(n_b) * 0.005, 0)
            pert /= pert.sum()
            a_opt, _ = coordinate_descent_polish(pert, time_limit=min(rem * 0.7, 60))
            try_seq(a_opt, "p3_cd_pert")

    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: L-BFGS on smooth approximation ===
    print(f"\n=== Phase 4: L-BFGS smooth (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 150

    if best_seq is not None:
        rem = t4_end - time.monotonic()
        if rem > 20:
            print(f"  L-BFGS t={rem:.0f}s")
            a_opt, _ = lgbfs_smooth_optimize(best_seq, time_limit=min(rem * 0.8, 100), temp=0.02)
            try_seq(a_opt, "p4_lgbfs")

            # Lower temperature
            rem = t4_end - time.monotonic()
            if rem > 15:
                a_opt, _ = lgbfs_smooth_optimize(best_seq, time_limit=min(rem * 0.8, 50), temp=0.005)
                try_seq(a_opt, "p4_lgbfs_cold")

    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 250

    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = [n_best // 2, int(n_best * 1.25), int(n_best * 1.5), 1024, 1536, 2048]
        target_lengths = sorted(set([t for t in target_lengths if 256 <= t <= 3000 and t != n_best]))
        target_lengths = target_lengths[:3]  # Try at most 3

        for target_n in target_lengths:
            rem = t5_end - time.monotonic()
            if rem < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  n={target_n}, rem={rem:.0f}s")
            # LP optimization
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = lp_optimize_main(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"p5_lp_n{target_n}")

            # Quick Adam
            rem = t5_end - time.monotonic()
            if rem > 20:
                a_opt, _ = adam_softmax_optimize(interpolate_sequence(best_seq, target_n),
                                                 time_limit=min(rem * 0.4, 60),
                                                 lr=0.004, temp_init=0.006,
                                                 temp_decay=0.9995, top_k=max(10, target_n // 15))
                try_seq(a_opt, f"p5_adam_n{target_n}")

    print(f"  After Phase 5: best={best_score:.8f}")

    # === Phase 6: Final micro-polishing ===
    print(f"\n=== Phase 6: Final polishing (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine Adam
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem * 0.4, 60)
            print(f"  Fine Adam, t={t_alloc:.0f}s")
            a_opt, _ = adam_softmax_optimize(best_seq, time_limit=t_alloc,
                                             lr=0.001, temp_init=0.002,
                                             temp_decay=0.9998, top_k=max(5, n_best // 40))
            try_seq(a_opt, "p6_fine")

        # LP with very small step
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = lp_optimize_main(best_seq, time_limit=min(rem * 0.5, 60), verbose=False)
            try_seq(a_opt, "p6_lp")

        # Tiny perturbations + quick CD
        rem = deadline - time.monotonic() - 10
        for pi in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = np.maximum(best_seq + noise, 0)
            f_pert /= f_pert.sum()
            a_opt, _ = coordinate_descent_polish(f_pert, time_limit=min(rem * 0.4, 30))
            try_seq(a_opt, f"p6_pert_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        try_seq(best_seq, "fallback")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
