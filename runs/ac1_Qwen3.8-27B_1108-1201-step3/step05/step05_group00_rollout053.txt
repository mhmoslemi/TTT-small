from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, T):
    """Compute gradient of soft_max(conv(a,a))/T and the soft-max weights."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = T * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # = 2 * cross_corr(w, a)[m]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
    using constraint generation (lazy constraints).
    """
    # Start with top-20 peak constraints
    conv_test = fast_convolve(f) * M  # Just for reference
    # Initial constraint set: top K peaks
    K = min(50, 2 * n - 1)
    active = set(np.argsort(fast_convolve(np.ones(n) / n))[-K:])
    # Actually, the constraints depend on f, not on the test sequence
    # Start with all constraints where f contributes most to the peak
    # Use a heuristic: include constraints k where sum_j f_j * f_{k-j} is large
    f_conv = fast_convolve(f)
    active = set(np.argsort(f_conv)[-K:].tolist())

    for iteration in range(30):
        # Build constraint matrix for active constraints
        constraints = sorted(active)
        m_c = len(constraints)
        A_ub = np.zeros((m_c, n))
        b_ub = np.full(m_c, M)
        for i, k in enumerate(constraints):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[i, j] = f[k - j]

        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        g = res.x

        # Check which inactive constraints are violated
        f_g = fast_convolve(g)
        violations = np.where(f_g > M + 1e-10)[0]
        if len(violations) == 0:
            return g

        # Add the most violated constraint
        worst = violations[np.argmax(f_g[violations])]
        active.add(int(worst))

        # Keep only the most important constraints (trim if too many)
        if len(active) > max_constraints:
            # Keep top constraints by violation amount
            f_g = fast_convolve(g)
            sorted_active = sorted(active, key=lambda k: -f_g[k])
            active = set(sorted_active[:max_constraints])

    # If we didn't converge, return best effort
    return g


def lp_fw_optimize(f0, time_limit, verbose=False, max_constraints=100):
    """LP-based Frank-Wolfe with lazy constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_lazy(f, M * m_exp, n, max_constraints=max_constraints)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 150:
                rng = np.random.default_rng(it * 19 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = fast_convolve(f).max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.004):
    """Mirror descent with multiplicative weight updates."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    T = 0.0005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Soft-max weights over top peaks
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient via FFT
        # Build w_ext (length 2n-1) from peak weights
        w_ext = np.zeros(2 * n - 1)
        for i in range(len(peaks)):
            w_ext[int(peaks[i])] = w[i]

        # grad[m] = 2 * sum_j w_ext[m+j] * f[j]
        a_rev = f[::-1]
        L = 1 << (3 * n).bit_length()
        conv_full = np.fft.irfft(np.fft.rfft(w_ext, L) * np.fft.rfft(a_rev, L), L)
        sg = 2.0 * conv_full[n - 1:n - 1 + n]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"      MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.9996
        T *= 1.01

        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with scaling trick."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            multipliers = [0.5, 0.7, 0.85, 0.93, 1.07, 1.15, 1.3, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.7):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = (1.0 - t) ** 3
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n):
    """Raised cosine window."""
    x = np.arange(n) / (n - 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * x - np.pi))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_hump(n, sep=0.25):
    """Two-hump distribution."""
    x = np.arange(n) / n
    a = np.exp(-((x - 0.5 + sep) / 0.08) ** 2) + np.exp(-((x - 0.5 - sep) / 0.08) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat(n):
    """Uniform distribution."""
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse inits -> smooth gradient descent ->
    LP-FW -> multi-length -> multiplicative descent -> coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations (30s) ===
    print("=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()
    init_candidates = []
    for n_try in [384, 512, 640, 768, 896]:
        init_candidates.append(make_gaussian(n_try, 0.25))
        init_candidates.append(make_bell(n_try, 0.6))
        init_candidates.append(make_cosine(n_try))
        init_candidates.append(make_two_hump(n_try, 0.2))
        init_candidates.append(make_flat(n_try))

    if parent is not None:
        init_candidates.append(parent)

    # Quick evaluate all
    for a_init in init_candidates:
        sc = try_seq(a_init, f"init_n{len(a_init)}")

    # Find best init by fast objective
    best_init = None
    best_init_obj = float('inf')
    for a_init in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        obj = 2 * len(a_norm) * fast_convolve(a_norm).max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(512, 0.3)

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Smooth gradient descent (400s) ===
    print("\n=== Phase 2: Smooth Gradient Descent ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  SGD on n={len(best_init)}: t={t_alloc:.0f}s")
        a_opt, obj = smooth_grad_descent(best_init, time_limit=t_alloc,
                                         T_init=10.0, T_min=0.3,
                                         lr_init=0.5, decay=0.9998,
                                         restart_every=150, verbose=True)
        try_seq(a_opt, "sgd_main")

        # Also try on parent if different
        if parent is not None and len(parent) != len(best_init):
            rem = deadline - time.monotonic() - 800
            if rem > 40:
                t_alloc = min(rem, 100)
                print(f"  SGD on parent n={len(parent)}: t={t_alloc:.0f}s")
                a_opt2, obj2 = smooth_grad_descent(parent, time_limit=t_alloc,
                                                    T_init=8.0, T_min=0.5,
                                                    lr_init=0.3, verbose=False)
                try_seq(a_opt2, "sgd_parent")

    # === Phase 3: LP-FW polishing (350s) ===
    print("\n=== Phase 3: LP-FW Polishing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem, 350)
            print(f"  LP-FW on n={len(best_seq)}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc,
                                        verbose=True, max_constraints=80)
            try_seq(a_opt, "lpfw_main")

    # === Phase 4: Multi-length exploration (250s) ===
    print("\n=== Phase 4: Multi-length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [512, 768, 1024]:
        if len(base) == target_n:
            continue

            break

                                             T_init=5.0, T_min=1.0,

    print("\n=== Phase 5: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  MD on n={len(best_seq)}: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                verbose=True, lr_init=0.003)
            try_seq(a_opt, "md")

    # === Phase 6: Coordinate descent (100s) ===
    print("\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:

            print(f"  CD on n={len(best_seq)}: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 7: Second round SGD with different params (100s) ===
    print("\n=== Phase 7: Second-round SGD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            # Try with different temperature schedule
            t_alloc = min(rem, 100)
            print(f"  SGD round2: t={t_alloc:.0f}s")
            a_opt, obj = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                             T_init=3.0, T_min=0.1,
                                             lr_init=0.2, decay=0.9999,
                                             restart_every=100, verbose=False)
            try_seq(a_opt, "sgd_r2")

            # Perturb and retry
            if rem - t_alloc > 30:
                n_best = len(best_seq)
                rng2 = np.random.default_rng(seed + 999)
                f_pert = np.maximum(best_seq + rng2.standard_normal(n_best) * 0.004, 0)
                f_pert = project_simplex(f_pert)
                t_alloc2 = min(rem - t_alloc - 10, 60)

                                                    T_init=5.0, T_min=0.5,
                                                    lr_init=0.3, verbose=False)
                try_seq(a_opt2, "sgd_pert")

    # === Phase 8: Final LP-FW squeeze (50s) ===
    print("\n=== Phase 8: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_alloc = min(rem, 50)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc,
                                        max_constraints=60)
            try_seq(a_opt, "final_lpfw")

    # === Phase 9: Final coordinate descent squeeze ===
    print("\n=== Phase 9: Final CD Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(512, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
