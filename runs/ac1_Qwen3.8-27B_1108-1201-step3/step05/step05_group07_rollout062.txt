def lbgfs_step(z0, T, n, time_limit, maxiter=2000):
    """Run L-BFGS for a short time."""
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_grad(z, T, n)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_for_direction(f, M, n_active=30):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top n_active peaks.
    Returns optimal g or None if failed.
    """
    n = len(f)
    b = fast_convolve(f)
    
    # Find top n_active peaks
    k_actual = min(n_active, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    
    # Build constraint matrix
    # (f*g)[k] = sum_j f[j] * g[k-j] for 0 <= j < n, 0 <= k-j < n
    # So for constraint k: sum_j A[k,j] * g[j] <= M where A[k,j] = f[j] if 0 <= k-j < n
    num_constraints = len(peaks)
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.full(num_constraints, M)
    
    for i, k in enumerate(peaks):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[i, j_start:j_end+1] = f[j_start:j_end+1]
    
    c = -np.ones(n)  # minimize -sum(g)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    
    return None


def lp_blend_loop(f, time_limit, n_active=30, t_init=0.3, tol=1e-7):
    """
    Iterate LP-based blending to reduce max autocorrelation.
    f is normalized (sum=1).
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_M = M
    best_f = f.copy()
    no_improve = 0
    t = t_init
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if iteration % 500 == 0:
            print(f"    LP iter {iteration}: M={M:.8f}, best_M={best_M:.8f}, t={t:.4f}")
        
        g = solve_lp_for_direction(f, M, n_active)
        if g is None:
            # Try with fewer active peaks
            g = solve_lp_for_direction(f, M, max(5, n_active // 2))
            if g is None:
                break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        # Normalize g to have same sum as f
        g = g * (f.sum() / g_sum)
        
        # Check if g is meaningfully different from f
        if np.allclose(g, f, atol=1e-10):
            # Try perturbing
            break
        
        # Blend with backtracking
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            t_eff = t * factor
            f_new = (1 - t_eff) * f + t_eff * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            
            b_new = fast_convolve(f_new)
            M_new = b_new.max()
            
            if M_new <= M + 1e-12:
                f = f_new
                M = M_new
                improved = True
                if M < best_M - 1e-12:
                    best_M = M
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
        
        if not improved:
            t *= 0.5
            no_improve += 1
        
        if no_improve > 100:
            # Perturb to escape
            rng = np.random.default_rng(iteration)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            b = fast_convolve(f)
            M = b.max()
            t = t_init
            no_improve = 0
        
        if M < 1.0 / n * 1.01:
            # Very close to theoretical lower bound for this approach
            break
    
    return best_f, best_M


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent on the simplex."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.3, 0.5, 0.7, 1.3, 1.5, 2.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. L-BFGS with soft-max for quick initialization
    2. LP-based blending for the main optimization
    3. Coordinate descent for final polish
    
    The LP approach solves for the optimal direction at each step,
    guaranteeing monotone decrease of the peak autocorrelation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
    
    n_parent = len(parent)
    
    # === Phase 1: Quick L-BFGS initialization ===
    print("\n=== Phase 1: L-BFGS initialization ===")
    t1_budget = min(120.0, (deadline - time.monotonic()) * 0.08)
    
    if t1_budget > 20:
        z0 = np.log(np.maximum(parent, 1e-300))
        for T in [5, 20, 100]:
            rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.85
            if rem < 10:
                break
            t_alloc = min(rem / max(1, 3 - [5, 20, 100].index(T)), 40.0)
            z0 = lbgfs_step(z0, T, n_parent, t_alloc, maxiter=1000)
            a_curr = np.exp(z0 - z0.max())

    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * np.conj(A), fft_size)[:n]


def softmax_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    S = compute_cross_corr(w, a)
    grad_a = 4.0 * n * S
    S_total = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S_total)
    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_run(z0, T, n, time_limit, maxiter=2000):
    """Run L-BFGS optimization with time limit."""
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return softmax_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return softmax_obj_grad(z, T, n)[1]

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def solve_lp_direction(f, M, n_active, relax=1.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M*relax for active peaks.
    Returns optimal g vector or None.
    """
    n = len(f)
    b = fast_convolve(f)
    k_actual = min(n_active, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    peaks = sorted([int(p) for p in peaks])

    num_c = len(peaks)
    A_ub = np.zeros((num_c, n))
    b_ub = np.full(num_c, M * relax)

    for i, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[js:je + 1]

    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_blend_loop(f, time_limit, n_active=30, t_init=0.2, relax=1.0):
    """
    Main LP blending loop. f is normalized (sum=1).
    Returns (best_f, best_M).
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy().astype(np.float64)
    b = fast_convolve(f)
    M = b.max()
    best_M = M
    best_f = f.copy()
    no_improve = 0
    t = t_init
    iteration = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if iteration % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {iteration}: M={M:.10f} best={best_M:.10f} t={t:.4f} elapsed={elapsed:.0f}s")

        g = solve_lp_direction(f, M, n_active, relax)
        if g is None:
            g = solve_lp_direction(f, M, max(5, n_active // 3), relax)
            if g is None:
                if iteration > 100:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * (f.sum() / g_sum)

        # Check if g is meaningfully different
        diff = np.linalg.norm(g - f)
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                b = fast_convolve(f)
                M = b.max()
                t = t_init
                no_improve = 0
            continue

        # Backtracking line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            t_eff = t * factor
            f_new = (1.0 - t_eff) * f + t_eff * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new /= sn
            M_new = fast_convolve(f_new).max()
            if M_new <= M + 1e-13:
                f = f_new
                M = M_new
                improved = True
                if M < best_M - 1e-13:
                    best_M = M
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break

        if not improved:
            t = max(t * 0.5, 0.01)
            no_improve += 1

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            b = fast_convolve(f)
            M = b.max()
            t = t_init
            no_improve = 0

        # Early stop if M is very close to 1/n (theoretical min for uniform)
        if M < 1.0 / n * 0.76:
            break

    return best_f, best_M


def subgradient_polish(f0, time_limit=60.0, lr=0.002, top_k=20):
    """Subgradient method with exact max."""
    t0 = time.monotonic()
    n = len(f0)
    f = f0.copy().astype(np.float64)
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    v = np.zeros(n)
    no_improve = 0
    cur_lr = lr
    mom = 0.9
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        k_act = min(top_k, len(b))
        peaks = np.argpartition(b, -k_act)[-k_act:]
        active = peaks[b[peaks] > best_M * 0.999]
        if len(active) == 0:
            active = peaks[:1]

        sg = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        M_new = fast_convolve(f).max()

        if M_new < best_M - 1e-13:
            best_M = M_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_M


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent on simplex for final polish."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = np.array([0.4, 0.6, 0.8, 1.25, 1.5, 2.0])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. L-BFGS warm start
    2. LP-based blending (main workhorse)
    3. Subgradient polish
    4. Coordinate descent final polish
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: L-BFGS warm start ===
    print("\n=== Phase 1: L-BFGS warm start ===")
    t1_budget = min(80.0, (deadline - time.monotonic()) * 0.05)
    if t1_budget > 15:
        z0 = np.log(np.maximum(parent, 1e-300))
        for T in [10, 50, 200]:
            rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.9
            if rem < 10:
                break
            t_alloc = min(rem, 30.0)
            z0 = lbgfs_run(z0, T, n_parent, t_alloc, maxiter=1000)
            a_curr = np.exp(z0 - z0.max())
            a_curr /= a_curr.sum()
            try_seq(a_curr, f"lbgfs_T{T}")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: LP-based blending (main) ===
    print(f"\n=== Phase 2: LP blending (best={best_score:.6f}) ===")

    # Multiple LP configurations with different parameters
    lp_configs = [
        (30, 0.2, 1.0),
        (50, 0.15, 1.0),
        (20, 0.3, 0.95),
        (40, 0.1, 1.0),
    ]

    for cfg_idx, (n_act, t_init, relax) in enumerate(lp_configs):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 300.0)
        print(f"  LP cfg{cfg_idx}: n_active={n_act}, t0={t_init}, relax={relax}, time={t_alloc:.0f}s")
        f_best, M_best = lp_blend_loop(base, time_limit=t_alloc,
                                       n_active=n_act, t_init=t_init, relax=relax)
        try_seq(f_best, f"lp_cfg{cfg_idx}")
        if best_seq is not None:
            base = best_seq.copy()

    # === Phase 3: Continue LP on best with fine parameters ===
    print(f"\n=== Phase 3: Fine LP polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 300.0)
            print(f"  Fine LP: n_active=60, t0=0.1, time={t_alloc:.0f}s")
            f_best, M_best = lp_blend_loop(best_seq, time_limit=t_alloc,
                                           n_active=60, t_init=0.1, relax=1.0)
            try_seq(f_best, "lp_fine")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_sg = min(rem * 0.4, 80.0)
            f_sg, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                         lr=0.001, top_k=30)
            try_seq(f_sg, "sg_polish")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_cd = min(rem * 0.5, 60.0)
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 6: One more LP round after all polish ===
    print(f"\n=== Phase 6: Final LP round (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 60.0)
            f_best, _ = lp_blend_loop(best_seq, time_limit=t_alloc,
                                       n_active=40, t_init=0.05, relax=1.0)
            try_seq(f_best, "lp_final")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [1024, 2048]:
            rem = deadline - time.monotonic() - 30
            if rem < 40:
                break
            f_interp = np.interp(
                np.linspace(0, 1, target_n),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            f_interp = np.maximum(f_interp, 0)
            f_interp = f_interp / f_interp.sum()
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  LP at n={target_n}, time={t_alloc:.0f}s")
            f_best, _ = lp_blend_loop(f_interp, time_limit=t_alloc,
                                       n_active=30, t_init=0.2)
            try_seq(f_best, f"lp_n{target_n}")
            if best_seq is not None:
                base = best_seq.copy()

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
