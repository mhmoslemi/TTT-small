"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines log-sum-exp gradient descent, coordinate descent with golden section,
and spectral reshaping for global exploration.
"""

import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_nonneg_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def spectral_reshape(a, alpha):
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.exp(1j * np.angle(A))
    new_mag = mag ** alpha
    new_A = new_mag * phase
    new_a = np.fft.irfft(new_A, fft_size)[:n]
    return project_nonneg_sum1(new_a)

def compute_grad_lse(a, p):
    """Compute gradient of LSE w.r.t. a, given softmax weights p."""
    n = len(a)
    # grad[j] = 2 * sum_m a[m] * p[m+j]
    windows = sliding_window_view(p, n)  # shape (n, n): windows[j, m] = p[j+m]
    return 2.0 * (windows @ a)

def lse_descent(a0, time_limit=60.0, verbose=False):
    a = a0.copy()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        tau = 30.0 + 170.0 * frac  # 30 to 200
        lr = 0.02 * (0.01) ** frac  # 0.02 to 0.0002
        # Actually, let me use a different schedule
        lr = 0.01 * (1.0 - 0.8*frac)
        
        conv = fast_convolve(a)
        cmax = conv.max()
        
        # Softmax weights
        scaled = tau * (conv - cmax)
        p = np.exp(scaled)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= ps
        
        grad = compute_grad_lse(a, p)
        # Project to tangent space
        grad = grad - grad.mean()
        
        # Step
        a_new = a + lr * (-grad)  # negative gradient
        a = project_nonneg_sum1(a_new)
        
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Spectral kick every 100 iterations
        if it % 100 == 0 and no_improve > 50:
            alpha = 0.85 + 0.1 * np.random.default_rng(it).random()
            a = spectral_reshape(a, alpha)
            no_improve = 0
        
        if no_improve > 300:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 7)
            noise = rng.standard_normal(n) * 0.005
            a = project_nonneg_sum1(best_a + noise)
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"      LSE iter {it}: obj={best_obj:.8f}, tau={tau:.1f}")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit=120.0, verbose=False, max_passes=200):
    a = a0.copy()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_num = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and pass_num < max_passes:
        pass_num += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_t = a[j]
            if old_t < 1e-10:
                # Try small positive values
                candidates = [1e-4, 1e-3, 5e-3, 0.01, 0.02]
            else:
                # Golden section search around current value
                # Search range: [max(0, old_t - 0.05), min(0.5, old_t + 0.05)]
                lo = max(0.0, old_t * 0.5)
                hi = min(0.5, old_t * 2.0 + 0.01)
                
                # Golden section
                gr = (np.sqrt(5) - 1) / 2  # ~0.618
                c = hi - gr * (hi - lo)
                d = lo + gr * (hi - lo)
                
                def eval_at(t_val):
                    if t_val < 0 or t_val >= 1.0:
                        return float('inf')
                    s = (1.0 - t_val) / (1.0 - old_t) if old_t < 1.0 else 0.0
                    if s < 0 or s > 2.0:
                        return float('inf')
                    a_new = a.copy()
                    a_new[j] = t_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_new[mask] *= s
                    a_new = np.maximum(a_new, 0)
                    ss = a_new.sum()
                    if ss < 1e-15:
                        return float('inf')
                    a_new /= ss
                    return 2.0 * n * fast_convolve(a_new).max()
                
                fc = eval_at(c)
                fd = eval_at(d)
                
                for _ in range(15):
                    if fc < fd:
                        hi = d
                        d = c
                        fd = fc
                        c = hi - gr * (hi - lo)
                        fc = eval_at(c)
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + gr * (hi - lo)
                        fd = eval_at(d)
                
                t_opt = (c + d) / 2
                candidates = [t_opt, old_t * 0.5, old_t * 1.5, 0.01, 0.1]
            
            best_cand_obj = best_obj
            best_cand_t = old_t
            
            for t_val in candidates:
                if t_val < 0 or t_val >= 0.9:
                    continue
                s = (1.0 - t_val) / (1.0 - old_t) if old_t < 1.0 else 0.0
                if s < -0.1 or s > 3.0:
                    continue
                a_new = a.copy()
                a_new[j] = t_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_new[mask] *= s
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                if ss < 1e-15:
                    continue
                a_new /= ss
                obj = 2.0 * n * fast_convolve(a_new).max()
                if obj < best_cand_obj - 1e-12:
                    best_cand_obj = obj
                    best_cand_t = t_val
            
            if best_cand_t != old_t:
                s = (1.0 - best_cand_t) / (1.0 - old_t)
                a[j] = best_cand_t
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= s
                a = project_nonneg_sum1(a)
                if best_cand_obj < best_obj - 1e-12:
                    best_obj = best_cand_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj

a[k-j]*(1-2t)*c - R[k]*(1-t) = 0
a[k-j]*c - 2*a[k-j]*c*t - R[k] + R[k]*t = 0
t*(R[k] - 2*a[k-j]*c) = R[k] - a[k-j]*c
t = (R[k] - a[k-j]*c) / (R[k] - 2*a[k-j]*c)

This gives the optimal t if a single peak dominates. In practice, multiple peaks compete, so the true optimum is different. But this gives a good starting point for a local search.

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines log-sum-exp gradient descent, coordinate descent with golden section search,
spectral reshaping for global exploration, and multi-length/multi-start search.
Key innovations: vectorized gradient via sliding window, adaptive tau schedule,
analytical warm-start for coordinate descent, and periodic spectral kicks.
"""

import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_nonneg_sum1(a):
    """Project to {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_reshape(a, alpha):
    """Reshape spectrum by raising magnitude to power alpha."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.exp(1j * np.angle(A))
    new_mag = mag ** alpha
    new_A = new_mag * phase
    new_a = np.fft.irfft(new_A, fft_size)[:n]
    return project_nonneg_sum1(new_a)


def spectral_broaden(a, strength=0.3):
    """Broaden the spectrum towards uniform (anti-peak in time domain)."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.exp(1j * np.angle(A))
    mean_mag = mag.mean()
    new_mag = mag + strength * (mean_mag - mag)
    new_mag = np.maximum(new_mag, 1e-10)
    new_A = new_mag * phase
    new_a = np.fft.irfft(new_A, fft_size)[:n]
    return project_nonneg_sum1(new_a)


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return project_nonneg_sum1(a)


def make_blackman(n):
    n_actual = n - 1
    if n_actual <= 0:
        return np.ones(n) / n
    x = np.arange(n) / n_actual
    a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    return project_nonneg_sum1(a)


def make_wendland(n, eps=0.7):
    """Wendland-like smooth bump: (1 - r)^2 * (2*eps + 1) type."""
    x = np.arange(n) / (n - 1) * 2 - 1  # [-1, 1]
    a = (1 - x ** 2) ** 2 * (1 + 2 * x ** 2)
    return project_nonneg_sum1(a)


def make_sinc_envelope(n, beta=3.0):
    """Sinc-like with Gaussian envelope."""
    x = np.arange(n) - n / 2.0
    env = np.exp(-0.5 * (x / (n * 0.2)) ** 2)
    a = np.sinc(x / n * beta) * env
    return project_nonneg_sum1(a)


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return project_nonneg_sum1(a_interp)


def eval_coordinate_change(a, j, t_val, old_t, n):
    """Evaluate objective after changing a[j] to t_val with proportional rescaling."""
    if t_val < -1e-10 or t_val > 0.99:
        return float('inf')
    if abs(old_t - 1.0) < 1e-15:
        return float('inf')
    s = (1.0 - t_val) / (1.0 - old_t)
    if s < -0.5 or s > 3.0:
        return float('inf')
    a_new = a.copy()
    a_new[j] = t_val
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    a_new[mask] *= s
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return float('inf')
    a_new /= ss
    return 2.0 * n * fast_convolve(a_new).max()


def compute_lse_grad(a, p):
    """Compute gradient of LSE(conv(a,a)) w.r.t. a given softmax weights p."""
    n = len(a)
    windows = sliding_window_view(p, n)
    return 2.0 * (windows @ a)


def lse_descent(a0, time_limit=60.0, verbose=False):
    """Log-sum-exp gradient descent with adaptive temperature."""
    a = a0.copy()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        tau = 20.0 + 200.0 * frac ** 1.5
        lr = 0.008 * (1.0 - 0.9 * frac) + 0.0002

        conv = fast_convolve(a)
        cmax = conv.max()

        scaled = tau * (conv - cmax)
        p = np.exp(scaled)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= ps

        grad = compute_lse_grad(a, p)
        grad = grad - grad.mean()

        a_new = a - lr * grad
        a = project_nonneg_sum1(a_new)

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 80 == 0 and no_improve > 40:
            rng = np.random.default_rng(it * 13)
            alpha = 0.8 + 0.15 * rng.random()
            a = spectral_reshape(a, alpha)
            no_improve = 0

        if no_improve > 250:
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = project_nonneg_sum1(best_a + noise)
            no_improve = 0

        if verbose and it % 40 == 0:
            print(f"      LSE iter {it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}")

    return best_a, best_obj


def multiplicative_descent(a0, time_limit=60.0, verbose=False):
    """Mirror descent with negative entropy (multiplicative weights)."""
    a = np.maximum(a0.copy(), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    tau = 0.02
    lr = 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        tau = 0.02 + 0.08 * frac
        lr = 0.5 * (1.0 - 0.85 * frac)

        conv = fast_convolve(a)
        cmax = conv.max()

        scaled = conv / tau
        scaled -= scaled.max()
        p = np.exp(scaled)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= ps

        grad = compute_lse_grad(a, p)
        grad = grad - grad.mean()

        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 11)
            alpha = 0.85 + 0.1 * rng.random()
            a = spectral_reshape(best_a, alpha)
            no_improve = 0

        if verbose and it % 40 == 0:
            print(f"      MD iter {it}: obj={best_obj:.8f}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit=120.0, verbose=False, max_passes=300):
    """Coordinate descent with golden section and analytical warm-start."""
    a = a0.copy()
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_num = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and pass_num < max_passes:
        pass_num += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_t = a[j]
            if old_t < 1e-12:
                # Try small activations
                best_t = old_t
                best_val = best_obj
                for t_try in [5e-5, 2e-4, 1e-3, 3e-3, 1e-2]:
                    val = eval_coordinate_change(a, j, t_try, old_t, n)
                    if val < best_val:
                        best_val = val
                        best_t = t_try
            else:
                # Golden section with analytical warm-start
                c = 1.0 - old_t
                conv = fast_convolve(a)
                # Find dominant peak
                k_peak = np.argmax(conv)
                kj = k_peak - j
                if 0 <= kj < n and kj != j:
                    R_approx = conv[k_peak] - 2.0 * old_t * a[kj]
                    a_kj = a[kj]
                    disc = a_kj * c
                    denom = R_approx - 2.0 * a_kj * c
                    if abs(denom) > 1e-10:
                        t_guess = (R_approx - disc) / denom
                    else:
                        t_guess = old_t
                    t_guess = np.clip(t_guess, 0.0, 0.5)
                else:
                    t_guess = old_t

                # Search range around current and guess
                lo = max(0.0, min(old_t, t_guess) * 0.3)
                hi = min(0.5, max(old_t, t_guess) * 2.0 + 0.02)

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)

                fc = eval_coordinate_change(a, j, c_pt, old_t, n)
                fd = eval_coordinate_change(a, j, d_pt, old_t, n)

                for _ in range(12):
                    if fc < fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        fc = eval_coordinate_change(a, j, c_pt, old_t, n)
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        fd = eval_coordinate_change(a, j, d_pt, old_t, n)

                t_opt = (c_pt + d_pt) / 2.0
                # Final candidates
                best_t = old_t
                best_val = best_obj
                for t_try in [t_opt, lo, hi, old_t * 0.7, old_t * 1.3, 0.01, 0.05]:
                    if t_try < 0 or t_try > 0.5:
                        continue
                    val = eval_coordinate_change(a, j, t_try, old_t, n)
                    if val < best_val:
                        best_val = val
                        best_t = t_try

            if best_t != old_t and best_t > 0:
                s = (1.0 - best_t) / (1.0 - old_t)
                a[j] = best_t
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= s
                a = project_nonneg_sum1(a)
                if best_val < best_obj - 1e-12:
                    best_obj = best_val
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and pass_num % 20 == 0:
            print(f"      CD pass {pass_num}: obj={best_obj:.8f}")

    return best_a, best_obj


def combined_optimize(a0, time_total=300.0, verbose=False):
    """Run LSE descent then coordinate descent on a given initialization."""
    n = len(a0)
    t_start = time.monotonic()

    # Phase A: LSE gradient descent (60% of time)
    t_a = time_total * 0.55
    a_opt, obj_a = lse_descent(a0, time_limit=t_a, verbose=verbose)
    if verbose:
        print(f"      After LSE: obj={obj_a:.8f}, time={time.monotonic()-t_start:.1f}s")

    # Phase B: Multiplicative descent (15% of time)
    rem = time_total - (time.monotonic() - t_start)
    t_b = rem * 0.3
    a_opt2, obj_b = multiplicative_descent(a_opt, time_limit=t_b, verbose=verbose)
    if obj_b < obj_a:
        a_opt = a_opt2
        obj_a = obj_b
    if verbose:
        print(f"      After MD: obj={obj_a:.8f}, time={time.monotonic()-t_start:.1f}s")

    # Phase C: Coordinate descent (remaining time)
    rem = time_total - (time.monotonic() - t_start)
    t_c = max(1.0, rem * 1.0)
    a_opt3, obj_c = coordinate_descent(a_opt, time_limit=t_c, verbose=verbose, max_passes=500)
    if obj_c < obj_a:
        a_opt = a_opt3
        obj_a = obj_c
    if verbose:
        print(f"      After CD: obj={obj_a:.8f}, time={time.monotonic()-t_start:.1f}s")

    return a_opt, obj_a

        if rem > 20 and best_seq is not None:
            t_alloc = min(rem, 60.0)
            print(f"  Final LSE: t={t_alloc:.0f}s")
            a_opt, obj_l = lse_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lse")

    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    else:
        # Should not happen, but just in case
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t0:.0f}s")
    return [float(x) for x in best_seq]
