def fast_convolve(a):
    n = len(a)
    size = 2*n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    if s < 1e-15:
        return float('inf')

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):

    n = len(a)
    s = a.sum()
    if s < 1e-15:

    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

    if s < 1e-15:

    return a / s

def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    a[0] = 1.0
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    """Sinusoidal shaped sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    """Gaussian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def triangular_sequence(n):
    """Triangular sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_single(a_init, time_limit, verbose=False):
    """
    Multi-stage optimization for a single sequence.
    Stage 1: Subgradient (exploratory)
    Stage 2: Coordinate descent (refinement)
    Stage 3: Repeated CD passes (polish)
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Stage 1: Subgradient (short, exploratory)
    t1 = min(time_limit * 0.25, 60)
    rem = time_limit - (time.monotonic() - t0)
    if rem > t1 + 30:
        a_sg, sc_sg = subgradient_descent(a, time_limit=t1, lr0=0.03)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
            a = a_sg.copy()
        if verbose:
            print(f"      After SG: {best_score:.10f}")
    
    # Stage 2: Coordinate descent (main refinement)
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.5, 120)
    if rem > t2 + 20:
        a_cd, sc_cd = coordinate_descent(a, time_limit=t2, verbose=verbose)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
        if verbose:
            print(f"      After CD: {best_score:.10f}")
    
    # Stage 3: More CD (polish)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 15:
        a_cd2, sc_cd2 = coordinate_descent(a, time_limit=rem - 5, verbose=verbose)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_a = a_cd2.copy()
        if verbose:
            print(f"      After CD2: {best_score:.10f}")
    
    return best_a, best_score
