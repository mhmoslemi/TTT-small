"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast O(n) Frank-Wolfe with
dual LP exploitation for direction finding (exploiting the interval cover
structure of the dual), combined with multiplicative descent polish and
multi-length exploration with diverse restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_obj(f):
    """Compute 2*n*max(conv) for normalized f (sum=1)."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()


def solve_dual_direction(f, M, n):
    """
    Solve the dual LP in O(n) exploiting the interval-cover structure.
    Returns the FW direction g (normalized to sum=1).
    
    Dual: min M * sum(y_k) s.t. sum_{k=j}^{j+n-1} y_k >= 1/f[j], y_k >= 0
    Solved via forward pass on prefix sums S.
    """
    # Forward pass: S[i] for i = 0, ..., 2n-2
    # S[i-1] = 0 for i=0; constraint at i>=n-1: S[i] >= S[i-n] + 1/f[i-n+1]
    S = np.zeros(2 * n - 1)
    for i in range(n - 1, 2 * n - 1):
        j = i - (n - 1)  # j = 0, ..., n-1
        prev = S[j - 1] if j > 0 else 0.0
        S[i] = max(S[i - 1], prev + 1.0 / f[j])

    # Dual variables y_k = S[k] - S[k-1]
    y = np.empty(2 * n - 1)
    y[0] = S[0]
    y[1:] = np.diff(S)

    # Identify active set: j where window constraint is tight
    tol = 1e-9
    A_list = []
    for j in range(n):
        ws = S[j + n - 1] - (S[j - 1] if j > 0 else 0.0)
        target = 1.0 / f[j]
        if ws >= target * (1.0 - tol):  # tight (within tolerance)
            A_list.append(j)

    # If active set is empty or too small, relax
    if len(A_list) < 2:
        tol = 1e-6
        A_list = []
        for j in range(n):
            ws = S[j + n - 1] - (S[j - 1] if j > 0 else 0.0)
            target = 1.0 / f[j]
            if ws >= target * (1.0 - tol):
                A_list.append(j)

    if not A_list:
        return np.ones(n) / n

    # Support of positive dual variables
    K_list = [k for k in range(2 * n - 1) if y[k] > 1e-10]
    if not K_list:
        K_list = [int(np.argmax(y))]  # at least one

    s = len(A_list)
    opt_sum = M * S[2 * n - 2]

    # Build and solve the linear system for g on active set
    # Equation for k: sum_{j in A, valid} f[k-j] * g[j] = M
    # where valid means 0 <= k-j <= n-1

    if s > 200:
        # Too many active variables; use a simpler heuristic
        g = np.zeros(n)
        for j in A_list:
            g[j] = 1.0 / f[j]
        g_sum = g.sum()
        if g_sum > 1e-15:
            g /= g_sum
        else:
            g = np.ones(n) / n
        return g

    # Use min(s, len(K_list)) equations + potentially sum constraint
    n_eq = min(len(K_list), s)
    use_K = K_list[:n_eq]

    if n_eq == s:
        # Square system
        B = np.zeros((s, s))
        for p, k in enumerate(use_K):
            for q, j in enumerate(A_list):
                idx = k - j
                if 0 <= idx <= n - 1:
                    B[p, q] = f[idx]
        rhs = np.full(s, M)
        try:
            sol = np.linalg.solve(B, rhs)
        except np.linalg.LinAlgError:
            sol, _, _, _ = np.linalg.lstsq(B, rhs, rcond=None)
    elif n_eq < s:
        # Underdetermined: add sum(g) = opt_sum
        B = np.zeros((n_eq + 1, s))
        for p, k in enumerate(use_K):
            for q, j in enumerate(A_list):
                idx = k - j
                if 0 <= idx <= n - 1:
                    B[p, q] = f[idx]
        B[n_eq, :] = 1.0
        rhs = np.empty(n_eq + 1)
        rhs[:n_eq] = M
        rhs[n_eq] = opt_sum
        sol, _, _, _ = np.linalg.lstsq(B, rhs, rcond=None)
    else:
        # Overdetermined (shouldn't happen since n_eq <= s)
        B = np.zeros((s, s))
        for p, k in enumerate(use_K[:s]):
            for q, j in enumerate(A_list):
                idx = k - j
                if 0 <= idx <= n - 1:
                    B[p, q] = f[idx]
        rhs = np.full(s, M)
        try:
            sol = np.linalg.solve(B, rhs)
        except np.linalg.LinAlgError:
            sol, _, _, _ = np.linalg.lstsq(B, rhs, rcond=None)

    sol = np.maximum(sol, 0.0)
    g = np.zeros(n)
    for q, j in enumerate(A_list):
        g[j] = sol[q]
    g_sum = g.sum()
    if g_sum > 1e-15:
        g /= g_sum
    else:
        # Fallback: subgradient-based direction
        conv = fast_convolve(f)
        peak = int(np.argmax(conv))
        sg = np.zeros(n)
        js = max(0, peak - n + 1)
        je = min(n - 1, peak)
        idx = np.arange(js, je + 1)
        sg[idx] = f[peak - idx]
        g = np.ones(n) / n
        g[int(np.argmin(sg))] = 1.0
        g /= g.sum()

    return g


def fw_optimize(f0, time_limit, verbose=False, max_iters=500000):
    """Fast Frank-Wolfe with O(n) dual direction finding."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        # Get FW direction
        g = solve_dual_direction(f, M, n)

        # Check convergence: if g is close to f, we're at a (local) optimum
        gap = np.max(np.abs(g - f))
        if gap < 1e-10:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 7 + 13)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 0)
                f /= f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                it = 0
            continue
        else:
            no_improve = 0

        # Line search: try a few step sizes
        best_alpha = 0.0
        best_val = M

        # Coarse grid
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha >= 0.1 and best_alpha < 1.0:
            # Fine bisection
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(4):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 1000 == 0:
                    print(f"      FW iter {it}: obj={best_obj:.10f}")

        if it % 5000 == 0 and verbose:
            print(f"      FW iter {it}: obj={best_obj:.10f}, gap={gap:.2e}")

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.003, decay=0.9997):
    """Mirror descent with multiplicative updates for peak shaving."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks for subgradient
        k_actual = min(15, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= decay
        temperature *= 1.01

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 17)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.7
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.7):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n):
    """Cosine window."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_shaving_local(f0, time_limit, verbose=False):
    """Targeted coordinate updates on the largest peak contributors."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        improved = False

        # Find the pair with maximum product contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[peak_k - idx]

        # Try reducing the element with max product and redistributing
        j_max = idx[np.argmax(products)]
        old_val = f[j_max]
        if old_val < 1e-12:
            j_max = idx[np.argmin(products)]
            old_val = f[j_max]
            if old_val < 1e-12:
                break

        best_trial = None
        best_trial_obj = M

        for frac in [0.3, 0.5, 0.7, 0.85, 1.3, 1.6]:
            new_val = old_val * frac
            if new_val < 0 or new_val > 0.5:
                continue
            trial = f.copy()
            trial[j_max] = new_val
            diff = old_val - new_val
            if abs(diff) > 1e-15:
                # Redistribute to elements with smallest contribution
                weights = f.copy()
                weights[j_max] = 0
                wsum = weights.sum()
                if wsum > 1e-15:
                    trial += diff * weights / wsum
            trial = np.maximum(trial, 0)
            tsum = trial.sum()
            if tsum < 1e-15:
                continue

            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-13:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-13:
                best_obj = best_trial_obj
                best_f = f.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            # Perturb
            rng = np.random.default_rng(int((time.monotonic() - t0) * 1000))
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0)
            f /= f.sum()
            no_improve = 0

    return best_f, best_obj

    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Evaluate diverse initializations ===
    print("=== Phase 1: Initializations ===")
    init_candidates = []
    for n_try in [384, 512, 640, 768]:
        init_candidates.append((make_gaussian(n_try, 0.25), f"gauss_{n_try}"))
        init_candidates.append((make_gaussian(n_try, 0.35), f"gauss_w_{n_try}"))
        init_candidates.append((make_bell(n_try, 0.6), f"bell_{n_try}"))
        init_candidates.append((make_cosine(n_try), f"cos_{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        # Interpolated versions of parent
        for tn in [512, 768]:
            if tn != len(parent):
                init_candidates.append((interpolate_sequence(parent, tn), f"parent_interp_{tn}"))

    # Evaluate all inits
    best_init = None
    best_init_obj = float('inf')
    for a_init, label in init_candidates:
        sc = try_seq(a_init, label)
        a_norm = a_init / a_init.sum()
        obj = fast_convolve(a_norm).max()
        if 2 * len(a_norm) * obj < best_init_obj:
            best_init_obj = 2 * len(a_norm) * obj
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(512, 0.3)

    print(f"  Best init obj: {best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Fast FW on best init ===
    print(f"\n=== Phase 2: Fast FW (n={len(best_init)}) ===")
    f_work = best_init.copy()
    if best_seq is not None and len(best_seq) == len(f_work):
        f_work = best_seq.copy()

    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  FW: t={t_alloc:.0f}s")
        a_opt, obj = fw_optimize(f_work, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "fw_main")

    # === Phase 3: Try different lengths with interpolated best ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [512, 640, 768, 896]:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_{target_n}")

        t_alloc = min(rem * 0.4, 80)
        print(f"  FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"fw_n{target_n}")

    # === Phase 4: Multiplicative descent polish ===
    print(f"\n=== Phase 4: MD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  MD: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                verbose=True, lr_init=0.002, decay=0.9997)
            try_seq(a_opt, "md_polish")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Peak shave: t={t_alloc:.0f}s")
            a_opt, obj = peak_shaving_local(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peak_shave")

    # === Phase 6: Restart FW with perturbations ===
    print(f"\n=== Phase 6: Restarted FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alloc = min((rem - 30) / (4 - p), 60)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = np.maximum(best_seq + noise, 0)
                f_init /= f_init.sum()
            elif p == 2:
                # Reverse
                f_init = best_seq[::-1].copy()
                f_init /= f_init.sum()
            else:
                # Blend with Gaussian
                gauss = make_gaussian(n_best, 0.3)
                f_init = 0.7 * best_seq + 0.3 * gauss
                f_init /= f_init.sum()

            print(f"  Restart FW variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"restart_{p}")

    # === Phase 7: Final FW squeeze ===
    print(f"\n=== Phase 7: Final FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final FW: t={t_alloc:.0f}s")
            a_opt, obj = fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_fw")

    # === Phase 8: Final MD micro-polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 10)
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                lr_init=0.001, decay=0.9998)
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
