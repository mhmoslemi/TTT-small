import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Build the matrix A where (A @ b)[k] = (a*b)(k) for k=0,...,2n-2."""

        js = max(0, k - n + 1)
        je = min(n-1, k)

Uses the linear programming approach: at each iteration, fix the current
sequence a, solve an LP to find the best direction g (maximizing sum(g)
subject to (a*g)(k) <= M for all k), then take a step in the direction g.
Combines with coordinate descent and random restarts for robustness.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    return A


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    One LP iteration: find best direction g.
    Returns (g, A_new) where g is the rescaled LP solution.
    """
    # Build constraint matrix
    A = build_conv_matrix(a)
    M = fast_convolve(a).max()
    
    # Solve LP: maximize sum(b) s.t. A @ b <= M, b >= 0
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n

        return None, A
    
    g = np.maximum(res.x, 0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, A
    g /= g_sum
    
    return g, A

def coordinate_descent(a0, time_limit=60, verbose=False):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def block_swap_search(a0, time_limit=60, seed=0):
    """Block swap local search: swap mass between blocks of elements."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    block_sizes = [1, 2, 4, 8, 16, 32]
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            block_size = block_sizes[rng.integers(0, len(block_sizes))]
            if block_size >= n:
                block_size = 1
            n_blocks = n // block_size
            if n_blocks < 2:
                continue
            
            bi = rng.integers(0, n_blocks)
            bj = rng.integers(0, n_blocks)
            if bi == bj:
                continue
            
            si = bi * block_size
            ei = si + block_size
            sj = bj * block_size
            ej = sj + block_size
            
            block_i = a[si:ei].sum()
            block_j = a[sj:ej].sum()
            if block_i < 1e-10 or block_j < 1e-10:
                continue
            
            for frac in [0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(block_i, block_j)
                    # Redistribute: add to block i, subtract from block j
                    a_trial = a.copy()
                    # Distribute transfer evenly within blocks
                    a_trial[si:ei] += transfer / block_size
                    a_trial[sj:ej] -= transfer / block_size
                    a_trial = np.maximum(a_trial, 0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    a_trial /= s_trial
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
    
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        return np.ones(n) / n
    return a / ss


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)

"""
LP-based alternating minimization for 2*n*max(conv(a,a))/(sum(a))^2.
Uses the linear programming direction-finding approach (solving an LP to
find the best descent direction at each iteration), combined with
coordinate descent refinement, block-swap local search, and multi-length
exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    return A


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a):
    n = len(a)
    A = build_conv_matrix(a)
    M = fast_convolve(a).max()
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    return g / g_sum
