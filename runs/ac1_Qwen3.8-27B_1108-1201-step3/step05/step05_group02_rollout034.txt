import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def sg_step(a, lr, top_k, temp):
    """One subgradient step."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    
    # Compute weights (softmax over peaks)
    pv = b[peaks]
    if temp > 0:
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws > 0:
            w /= ws
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        # contribution[j] = 2 * a[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j in [j_start, j_end] means a[k-j_end] to a[k-j_start]
            # which is a[k-j_end : k-j_start+1] reversed
            idx_start = k - j_end
            idx_end = k - j_start
            contrib = 2.0 * w[i] * a[idx_start : idx_end + 1][::-1]
            sg[j_start : j_end + 1] += contrib
    
    return a - lr * sg, b.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast subgradient descent with
top-K peak weighting, adaptive learning rates, coordinate descent, structured
restarts, and multi-length exploration. Key improvements over previous:
faster SG computation via vectorized indexing, better diversity in restarts,
larger sequence lengths, and simulated-annealing-style temperature scheduling.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Ensure non-negative before projecting
    v = np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sg(a, peaks, weights):
    """Compute subgradient of max_k b[k] at a using given peaks and weights."""
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        if w < 1e-15:
            continue
        if k < n:
            # sg[0:k+1] += 2*w * a[0:k+1][::-1]
            sg[0:k + 1] += 2.0 * w * a[0:k + 1][::-1]
        else:
            # sg[k-n+1:n] += 2*w * a[k-n+1:n][::-1]
            sg[k - n + 1:n] += 2.0 * w * a[k - n + 1:n][::-1]
    return sg


def sg_optimize(a0, time_limit, verbose=False):
    """
    Subgradient optimization with adaptive learning rate, top-K peaks,
    momentum, and periodic restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = 0.9
    v = np.zeros(n)
    lr = 0.3
    top_k = max(5, n // 20)
    temp = best_obj * 0.1

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        b = fast_convolve(a)
        cur_obj = b.max()

        # Find top-K peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]

        # Softmax weights
        scaled = pv / max(temp, 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = compute_sg(a, peaks, w)

        # Update with momentum
        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        # Check improvement
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        a = a_new
        b = b_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            # Decay lr
            lr *= 0.995
            # Increase temperature (smooth out)
            temp *= 1.01

        if verbose and it % 500 == 0:
            print(f"    SG iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, lr={lr:.4f}, top_k={top_k}")

        # Restart logic
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            scale = min(0.001 + 0.002 * (no_improve // 200), 0.02)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = max(0.1, lr * 0.5)
            top_k = max(3, n // 30)
            temp = fast_convolve(a).max() * 0.05
            no_improve = 0

        # Periodically adjust top_k
        if it % 500 == 0:
            top_k = max(3, n // (15 + it // 1000))
            temp = b.max() * (0.05 + 0.05 * np.sin(it * 0.01))

    return best_a, best_obj


def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates and rescaling."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    passes = 0
    multipliers = [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        b_cur = fast_convolve(a)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-10:
                # Try setting it to a small value
                for target in [1.0 / n, 2.0 / n]:
                    a_try = a.copy()
                    a_try[j] = target
                    rest_sum = 1.0 - target
                    if rest_sum < 1e-10:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_try[mask] *= rest_sum / a_try[mask].sum()
                    obj = fast_convolve(a_try).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                continue

            best_j_val = old_val
            best_j_obj = b_cur.max()

            for mult in multipliers:
                new_val = old_val * mult
                rest_sum = 1.0 - new_val
                if rest_sum < 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = rest_sum / denom
                a_try = a.copy()
                a_try[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_try[mask] *= scale
                ss = a_try.sum()
                if ss < 1e-15:
                    continue
                a_try /= ss
                obj = fast_convolve(a_try).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                new_val = best_j_val
                rest_sum = 1.0 - new_val
                denom = 1.0 - old_val
                if denom > 1e-12 and rest_sum > 1e-10:
                    scale = rest_sum / denom
                    a[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                    if best_j_obj < obj_max_obj(best_a) - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True

        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

        if not improved and passes > 10:
            break
        if not improved:
            continue

    return best_a, best_obj


def obj_max_obj(a):
    return fast_convolve(a).max()


def structured_perturb(a, rng, n):
    """Generate structurally diverse perturbations."""
    choices = []

    # Shift
    shift = rng.integers(1, max(2, n // 20))
    choices.append(np.roll(a, shift))

    # Reverse
    choices.append(a[::-1].copy())

    # Interpolate with uniform
    t = rng.uniform(0.1, 0.4)
    choices.append((1 - t) * a + t * np.ones(n) / n)

    # Gaussian window modulation
    center = n / 2.0
    sigma = n * rng.uniform(0.2, 0.5)
    window = np.exp(-0.5 * (np.arange(n) - center) ** 2 / sigma ** 2)
    w_sum = window.sum()
    choices.append(a * window / w_sum)

    # Bistable (two bumps)
    w1 = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / (n * 0.1)) ** 2)
    w2 = np.exp(-0.5 * ((np.arange(n) - n * 0.75) / (n * 0.1)) ** 2)
    choices.append((w1 + w2) / (w1 + w2).sum())

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    cos_w = np.maximum(cos_w, 1e-6)
    choices.append(a * cos_w / cos_w.sum())

    # Random phase modulation (keep non-negative)
    phase = 2 * np.pi * rng.random(n)
    mod = 0.5 + 0.5 * np.cos(phase)
    mod = np.maximum(mod, 0.01)
    choices.append(a * mod / mod.sum())

    # Select random perturbation
    choice = choices[rng.integers(0, len(choices))]
    return project_simplex(np.maximum(choice, 0))


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_window(n, window_type='raised_cosine'):
    """Generate various window functions as initializations."""
    x = np.arange(n) / max(n - 1, 1)
    if window_type == 'raised_cosine':
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif window_type == 'gaussian':
        w = np.exp(-4 * (x - 0.5) ** 2)
    elif window_type == 'flat_top':
        w = 1.0 - 1.93 * np.abs(x - 0.5) ** 5 + 1.26 * np.abs(x - 0.5) ** 3
        w = np.maximum(w, 0)
    elif window_type == 'triangular':
        w = 1 - np.abs(2 * x - 1)
    elif window_type == 'hamming':
        w = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    elif window_type == 'blackman':
        w = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif window_type == 'uniform':
        w = np.ones(n)
    else:
        w = np.ones(n)
    w = np.maximum(w, 1e-6)
    return w / w.sum()

        f_interp = interpolate_to_length(best_seq, nl)
        t_alloc = min(30, rem * 0.3)
        if t_alloc < 8:
            continue
        a_opt, obj = sg_optimize(f_interp, time_limit=t_alloc)
        try_sequence(a_opt, f"final-len-{nl}")

    # === Final: one more CD pass ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt, obj = coord_descent(best_seq, time_limit=min(rem * 0.8, 60))
        try_sequence(a_opt, "FINAL-CD")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
