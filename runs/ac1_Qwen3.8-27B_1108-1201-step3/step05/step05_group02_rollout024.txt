import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, temp=0.001):
    """Soft subgradient of max_k (a*a)(k) using exponential peak weighting."""
    n = len(a)
    # Use top peaks weighted by exponential of (conv - max)/temp
    cmax = conv.max()
    scaled = (conv - cmax) / temp
    scaled = np.maximum(scaled, -50)  # prevent underflow
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= wsum

    g = np.zeros(n)
    # Only process peaks with significant weight
    significant = w > 1e-10
    peaks = np.where(significant)[0]
    for k in peaks:
        wk = w[k]
        if wk < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += wk * 2.0 * a[k - idx]
    return g


def mirror_descent(f0, time_limit, verbose=False):
    """Mirror descent (exponential gradient) on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    # Adaptive parameters
    temp = 0.002
    lr_base = 0.01
    decay = 0.9998

    while time.monotonic() - t0 < time_limit:
        it += 1
        lr = lr_base / (1.0 + 0.01 * it)

        g = compute_subgradient(f, conv, temp)

        # Mirror descent update: f_j *= exp(-lr * g_j), then normalize
        log_update = -lr * g
        # Clip for numerical stability
        log_update = np.clip(log_update, -5, 5)
        f = f * np.exp(log_update)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"  MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Slowly increase temperature (anneal)
        temp *= 1.0005

        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(it * 131 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            no_improve = 0
            temp = 0.001
            lr_base = max(lr_base * 0.7, 0.001)

    return best_f, best_obj


def subgradient_polish(f0, time_limit, verbose=False):
    """Direct subgradient method with Polyak-like step size and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    n = len(f)

    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    momentum = np.zeros(n)
    mom = 0.9
    lr = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        cur_obj = conv.max()

        # Use top-K peaks for smoother subgradient
        k_top = min(20, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Exponential weighting
        scaled = (pv - pv.max()) / 0.0005
        w = np.exp(scaled)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:

            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * 2.0 * f[k - idx]

        # Momentum update
        momentum = mom * momentum - lr * g
        f_new = f + momentum
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            momentum = np.zeros(n)
            no_improve = 0
            lr = max(lr * 0.5, 0.0003)

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.1
            if m_relax > 5.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.02, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.75, 1.0
            elif best_alpha <= 0.02:
                lo, hi = 0.0, 0.06
            else:
                lo = max(0.0, best_alpha - 0.12)
                hi = min(1.0, best_alpha + 0.12)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 4.0)

        if no_improve > 100:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(no_improve // 100, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_equalize(f0, time_limit):
    """Targeted optimization: reduce the tallest peaks by adjusting contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Find top peaks
        n_peaks = min(10, len(conv))
        peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_idx = peak_idx[conv[peak_idx] > cur_max * 0.95]  # only near-max peaks

        if len(peak_idx) == 0:
            break

        # For each peak, find the coordinates that contribute most
        # and try to reduce them
        improved = False
        for k in peak_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = f[idx] * f[k - idx]

            # Try reducing the largest contributor
            worst_local = idx[np.argmax(contributions)]
            old_val = f[worst_local]

            for frac in [0.8, 0.6, 0.4, 0.7, 0.9]:
                new_val = old_val * frac
                if new_val < 0:
                    continue
                # Renormalize by scaling others
                denom = 1.0 - old_val
                if denom < 1e-12:
                    break
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[worst_local] = new_val
                mask = np.ones(n, dtype=bool)
                mask[worst_local] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < cur_max - 1e-14:
                    f = f_trial
                    cur_max = val
                    improved = True
                    break

            if cur_max < best_obj - 1e-14:
                best_obj = cur_max
                best_f = f.copy()

        if not improved and it > 2:
            # Try increasing some coordinates to boost low-convolution regions
            min_idx = np.argmin(conv)
            js = max(0, min_idx - n + 1)
            je = min(n - 1, min_idx)
            idx = np.arange(js, je + 1)
            # Try increasing a coordinate that contributes to a low region
            for j in idx:
                for frac in [1.5, 2.0, 1.3]:
                    new_val = f[j] * frac
                    if new_val >= 0.99:
                        continue
                    denom = 1.0 - f[j]
                    if denom < 1e-12:
                        break
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
                if improved:
                    break

        if not improved and it > 20:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization combining mirror descent, LP-FW,
    subgradient polish, peak equalization, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Mirror Descent on parent ===
    print(f"\n=== Phase 1: Mirror Descent (n={n_parent}) ===")
    t_rem = deadline - time.monotonic()
    t1 = min(200, t_rem * 0.15)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = mirror_descent(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (MD): {sc:.6f}")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}, best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t2 = min(300, t_rem * 0.2)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_opt, obj = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (LP-FW): {sc:.6f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t3 = min(120, t_rem * 0.1)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a_opt, obj = subgradient_polish(best_seq, time_limit=t3)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (SG): {sc:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-128, -64, -32, 32, 64, 128, 256, -256]:
        nl = n_best + delta
        if 256 <= nl <= 2048 and nl != n_best:
            candidate_lengths.append(nl)
    # Also try some "natural" lengths
    for nl in [512, 640, 768, 896, 1024, 1280]:
        if nl not in candidate_lengths and 256 <= nl <= 2048:
            candidate_lengths.append(nl)

    # Sort by distance to best_n
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    for target_n in candidate_lengths[:6]:  # Try at most 6 lengths
        t_rem = deadline - time.monotonic() - 200
        if t_rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, t_rem * 0.15)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")

        # Run both MD and LP-FW
        a1, _ = mirror_descent(f_interp, time_limit=t_alloc * 0.4)
        sc1 = float(evaluate_sequence(a1.tolist()))
        a2, _ = lp_fw_optimize(f_interp, time_limit=t_alloc * 0.5)
        sc2 = float(evaluate_sequence(a2.tolist()))

        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  NEW BEST (n={target_n}, MD): {sc1:.6f}")
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  NEW BEST (n={target_n}, LP): {sc2:.6f}")

    # === Phase 5: Combined MD + LP-FW on best ===
    print(f"\n=== Phase 5: Combined optimization (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic() - 100
    if t_rem > 30:
        # Alternate between MD and LP-FW
        f_cur = best_seq.copy()
        for round_idx in range(4):
            t_rem = deadline - time.monotonic() - 80
            if t_rem < 15:
                break
            t_half = min(80, t_rem * 0.25)
            if t_half < 10:
                break

            a_opt, _ = mirror_descent(f_cur, time_limit=t_half)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                f_cur = a_opt.copy()
                print(f"  [Round {round_idx} MD] {sc:.6f}")

            t_rem = deadline - time.monotonic() - 60
            if t_rem < 15:
                break
            t_half = min(80, t_rem * 0.25)
            if t_half < 10:
                break

            a_opt, _ = lp_fw_optimize(f_cur, time_limit=t_half)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                f_cur = a_opt.copy()
                print(f"  [Round {round_idx} LP] {sc:.6f}")

    # === Phase 6: Peak equalization ===
    print(f"\n=== Phase 6: Peak Equalization (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic() - 50
    if t_rem > 20:
        t_pe = min(60, t_rem * 0.4)
        print(f"  Budget: {t_pe:.0f}s")
        a_opt, obj = peak_equalize(best_seq, time_limit=t_pe)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (PE): {sc:.6f}")

    # === Phase 7: Final micro-search ===
    print(f"\n=== Phase 7: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        t_rem = deadline - time.monotonic()
        if t_rem < 4:
            break
        scale = 0.0002 + trial * 0.00015
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick polish
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 5:
                a_opt, _ = subgradient_polish(f_pert, time_limit=min(t_rem2 * 0.3, 8))
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_sg]: {sc2:.6f}")

    # === Final squeeze: short MD + LP-FW ===
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        t_final = min(40, t_rem * 0.8)
        a_opt, _ = mirror_descent(best_seq, time_limit=t_final)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL MD: {sc:.6f}")

    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        t_final = min(30, t_rem * 0.8)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL LP: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
