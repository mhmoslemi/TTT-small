$\min t$
s.t. $b_k + 2 \sum_i \delta_i a_{k-i} \leq t$ for $k \in \mathcal{A}$
$\sum_i (a_i + \delta_i) = 1$
$a_i + \delta_i \geq 0$

This is a linear program! We can solve it with any LP solver.

If $\alpha = 1$ and the new $a$ satisfies all constraints (not just the linearized ones), we're at a local optimum. Otherwise, we take a smaller step.

This is essentially a **sequential linear programming (SLP)** approach!

```python
"""
SLP-based optimizer for minimum peak-to-average ratio sequences.
Uses sequential linear programming to minimize the max of the auto-convolution,
with subgradient polishing and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def slp_step(a, active_lags=None, top_k=50):
    """
    One SLP iteration: solve LP to find descent direction for max(conv(a,a)).
    Returns (delta, t_star, active_from_lp).
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    
    if active_lags is None:
        # Use top-k peaks
        k_actual = min(top_k, len(b))
        active_lags = np.argpartition(b, -k_actual)[-k_actual:]
    
    # Build LP: min t
    # s.t. for each k in active_lags: 2*sum_j a[k-j]*delta_j - t <= -b[k]
    #      sum_j delta_j = 0
    #      delta_j >= -a_j
    
    n_vars = n + 1  # delta_0..delta_{n-1}, t
    t_idx = n  # index of t variable
    
    # Objective: minimize t
    c = np.zeros(n_vars)
    c[t_idx] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k: 2*sum_j a[k-j]*delta_j - t <= -b[k]
    A_rows = []
    b_ub_list = []
    
    for k in active_lags:
        row = np.zeros(n_vars)
        # For each j, coefficient of delta_j is 2 * a[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = 2.0 * a[k - j]
        row[t_idx] = -1.0
        A_rows.append(row)
        b_ub_list.append(-b[k])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_ub_list)
    
    # Equality: sum delta_j = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: delta_j >= -a_j, t >= 0
    bounds = []
    for j in range(n):
        bounds.append((-a[j], None))  # delta_j >= -a_j
    bounds.append((0.0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, M, active_lags
    
    delta = res.x[:n]
    t_star = res.x[n]
    
    return delta, t_star, active_lags


def slp_optimize(a0, time_limit=120.0, top_k=50, max_iter=200):
    """
    SLP optimization: iteratively solve LPs to reduce max(conv(a,a)).
    """
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        delta, t_star, active = slp_step(a, top_k=top_k)
        
        if delta is None:
            print(f"  SLP it {it}: LP infeasible, stopping")
            break
        
        if t_star >= M - 1e-14:
            print(f"  SLP it {it}: t*={t_star:.10f} >= M={M:.10f}, converged")
            break
        
        # Line search
        best_alpha = 0
        best_M_trial = M
        
        for alpha in [1.0, 0.75, 0.5, 0.25, 0.125, 0.0625]:
            a_trial = a + alpha * delta
            if (a_trial < -1e-12).any():
                continue
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            b_trial = fast_convolve(a_trial)
            M_trial = b_trial.max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_alpha = alpha
        
        if best_alpha == 0:
            print(f"  SLP it {it}: no improvement found, stopping")
            break
        
        a = a + best_alpha * delta
        a = np.maximum(a, 0)
        a = a / a.sum()
        b = fast_convolve(a)
        M = b.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
        
        if it % 20 == 0:
            print(f"  SLP it {it}: M={M:.10f}, alpha={best_alpha}, t*={t_star:.10f}")
        
        # Check convergence
        if M - best_M > M * 0.001:
            # Too much degradation, revert
            a = best_a.copy()
            M = best_M
    
    return best_a, best_M


def subgradient_refine(a0, time_limit=60.0, lr=0.01, momentum=0.9, n_peaks=30, temp_decay=0.9997, T0=0.01):
    """Subgradient method with softmax-weighted peaks."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    best_a = a.copy()
    best_M = M
    v = np.zeros(n)
    t0 = time.monotonic()
    T = T0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        
        # Select top peaks
        k_actual = min(n_peaks, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        
        # Softmax weights
        scaled = pv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute gradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * w[i] * a[k - j]
        
        # Update with momentum
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        M_new = b_new.max()
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        T *= temp_decay
        lr *= 0.9999
        
        if no_improve > 500:
            break
    
    return best_a, best_M


def try_sequence(seq, best_score_holder):
    """Try a sequence and update best if improved."""
    if seq is None:
        return
    a = np.asarray(seq, dtype=np.float64)
```
