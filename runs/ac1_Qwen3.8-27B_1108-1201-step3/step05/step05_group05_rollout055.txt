        # Actually, this is just the convolution matrix. Let me use a more efficient construction.
        # Row k of A_lp: A_lp[k, :] = a[k-0], a[k-1], ..., a[k-(n-1)] (padded with 0)
        # = the "autocorrelation" of a at lag k
        # A_lp[k, j] = a[k-j] if 0 <= k-j < n else 0
        
        # Build more efficiently
        for j in range(n):
            A_lp[j:j+n, j] = a
        
        c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
        b_ub = np.full(2*n-1, M)
        
        res = linprog(c, A_ub=A_lp, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b = b / b_sum  # normalize
        
        # Line search: find best t in a_new = (1-t)*a + t*b
        best_t = 0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1-t)*a + t*b
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()

                continue

            obj = fast_convolve(a_trial).max()

            if best_t_obj < best_obj:
                best_obj = best_t_obj
                best_a = a.copy()

            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            pv = conv[peaks]
            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
            
            obj = n * (w @ pv)
            
            grad = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-15:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += 2.0 * w[i] * a_x[k - idx]
            
            dot = a_x @ grad
            grad_x = (grad - a_x * dot) / sx
            return obj, grad_x
        
        max_iter = max(50, int(200 * time_per_temp / max(remaining / len(temps), 1)))
        
        result = minimize(
            fun=fun_and_grad,
            x0=a,
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': min(max_iter, 500), 'maxfun': 1500,
                     'ftol': 1e-18, 'gtol': 1e-14}
        )
        
        a_new = np.maximum(result.x, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new /= s_new
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            elif obj_new < best_obj:
                a = a_new.copy()
        if verbose:
            print(f"    L-BFGS temp={temp}: obj={best_obj:.10f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj


def lp_refine(a0, time_limit=60, verbose=False):

            break

        improved = False

            # a_new[j] = a[j] * c
            # a_new[k] = a[k] * (1 - a[j]*c) / (1 - a[j]) for k != j
            # c in [0, c_max] where c_max = (1 - 1e-12) / a[j]

                if c < 0 or c > c_max:
                    return 1e10
                a_trial = a.copy()
                a_trial[j] = a[j] * c
                remaining = 1.0 - a[j] * c
                old_remaining = 1.0 - a[j]
                if old_remaining < 1e-12:
                    return 1e10
                scale = remaining / old_remaining
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                return fast_convolve(a_trial).max()
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
            lo = 0.0
            hi = c_max
            
            # Check endpoints and current
            f_lo = f_of_c(lo + 1e-8)
            f_hi = f_of_c(hi - 1e-8)
            f_cur = f_of_c(1.0)
            
            if f_lo < f_cur and f_lo < f_hi:
                hi = 1.0
            elif f_hi < f_cur and f_hi < f_lo:
                lo = 1.0
            else:
                lo = max(0.5, lo)
                hi = min(1.5, hi)
            
            # If current is best, try a narrow range
            best_c = 1.0
            best_f = f_cur
            if f_lo < best_f:
                best_f = f_lo
                best_c = lo + 1e-8
            if f_hi < best_f:
                best_f = f_hi
                best_c = hi - 1e-8
            
            # Golden section in [lo, hi]
            if hi - lo > 0.01:
                a1 = hi - gr * (hi - lo)
                a2 = lo + gr * (hi - lo)
                f_a1 = f_of_c(a1)
                f_a2 = f_of_c(a2)
                
                for _ in range(25):
                    if hi - lo < 0.001:
                        break
                    if f_a1 < f_a2:
                        hi = a2
                        a2 = a1
                        f_a2 = f_a1
                        a1 = hi - gr * (hi - lo)
                        f_a1 = f_of_c(a1)
                    else:
                        lo = a1
                        a1 = a2
                        f_a1 = f_a2
                        a2 = lo + gr * (hi - lo)
                        f_a2 = f_of_c(a2)
                
                c_mid = (lo + hi) / 2
                f_mid = f_of_c(c_mid)
                if f_mid < best_f:
                    best_f = f_mid
                    best_c = c_mid
            
            # Apply best
            if best_c > 0 and best_c < c_max and best_f < best_obj - 1e-14:
                a[j] = a[j] * best_c
                remaining = 1.0 - a[j]
                old_remaining = 1.0 - a[j] / best_c if best_c > 0 else 1.0
                # Actually let me redo this properly
                a_trial = a.copy()
                a_trial[j] = a[j] / best_c * best_c  # = a[j], no wait
                
                # Let me just recompute from scratch
                a[j] = a[j] * best_c
                new_rem = 1.0 - a[j]
                old_rem = 1.0 - a[j] / best_c
                if old_rem > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = a[mask] * (new_rem / old_rem)
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                
                if best_f < best_obj:
                    best_obj = best_f
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            if pass_count > 2:
                break
            else:
                # Reset to best and try again with shuffled order
                a = best_a.copy()
    
    return best_a, best_obj

This is simpler but less powerful than golden section. Let me use a hybrid: try the multipliers first, and if we find an improvement, do a golden section around the best multiplier.

Actually, for the purposes of squeezing out the last bit of performance, let me use a different strategy:

1. **Primary**: L-BFGS with decreasing temperature (this is the main workhorse)
2. **Secondary**: LP refinement (this gives exact directions)
3. **Tertiary**: Coordinate descent with fine multipliers (polish)

And I'll interleave them in a smart way.

Let me also think about what's special about the problem that I might be missing. 

Key insight: The objective `2n * max(conv) / sum^2` is scale-invariant. So we can fix sum=1 without loss of generality. The problem reduces to: minimize `2n * max_k c_k` where `c_k = sum_i a_i * a_{k-i}`, subject to `a_i >= 0`, `sum a_i = 1`.

There's a lower bound: By AM-GM or other inequalities, what's the minimum possible value?

For uniform a: `c_{n-1} = sum (1/n)^2 = 1/n`, so objective = `2n * 1/n = 2`.

For a "flat" autocorrelation (which is what we want), the average of `c_k` over all k is... let me compute.

`sum_k c_k = sum_k sum_i a_i * a_{k-i} = sum_i a_i * sum_i a_i = (sum a_i)^2 = 1` (when sum=1).

There are 2n-1 values of k, so the average is `1/(2n-1)`. Since the max must be >= the average, we have `max_k c_k >= 1/(2n-1)`.

So the lower bound on the objective is `2n/(2n-1) ≈ 1` for large n. We're at 1.505, so there's room for improvement in principle, but the non-negativity constraint makes it much harder to achieve.

Actually, I realize the bound `max >= average` gives `max >= 1/(2n-1)`, so objective >= `2n/(2n-1) ≈ 1 + 1/(2n-1)`. For n=768, this is about 1.00065. So in principle, we could get much lower than 1.505, but the non-negativity constraint is the binding issue.

For a non-negative sequence, what additional constraints apply? The convolution at k=0 is `a_0^2`, and at k=2n-2 is `a_{n-1}^2`. So `max_k c_k >= max(a_0^2, a_{n-1}^2)`. If the sequence is concentrated, these endpoint values are large.

Also, `c_{n-1} = sum_i a_i * a_{n-1-i}`. If a is symmetric (a_i = a_{n-1-i}), then `c_{n-1} = sum a_i^2`. Otherwise, it's a "cross term."

The key tradeoff is: making the sequence flat reduces the center peak but makes the endpoints more significant. Making the sequence concave (like a cosine) can reduce the center while keeping endpoints small.

I think the optimal shape is something like a raised cosine or a "smooth window function" - concave in the middle, tapering to zero at the edges. This is consistent with the "flat spectrum" initialization (IFFT of flat magnitude gives a sinc-like shape, which when rectified gives a concave shape).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with adaptive
softmax temperature annealing, LP-based local refinement, and fine-grained
coordinate descent. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_grad(a, temp, top_k=150):
    """Compute smoothed max objective and its gradient w.r.t. a (unnormalized)."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    obj = n * (w @ pv)
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return obj, grad


def lbfqgs_optimize(a0, time_limit, verbose=False):
    """L-BFGS-B with temperature annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    temps = [60, 40, 30, 20, 15, 12, 10, 8, 7, 6, 5, 4.5, 4, 3.5, 3, 2.5, 2, 1.8, 1.5, 1.3, 1.1, 1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.25]
    
    for ti, temp in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        frac_per_temp = 1.0 / (len(temps) - ti)
        
        def fun_and_grad(x):
            a_x = np.maximum(x, 0)
            sx = a_x.sum()
            if sx < 1e-10:
                return 1e10, np.zeros(n)
            a_x = a_x / sx
            obj, grad = softmax_grad(a_x, temp, top_k=150)
            # Chain rule for normalization: a_x = x/sx
            dot = a_x @ grad
            grad_x = (grad - a_x * dot) / sx
            return obj, grad_x
        
        max_iter = max(30, int(400 * frac_per_temp))
        
        try:
            result = minimize(
                fun=fun_and_grad,
                x0=a,
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': min(max_iter, 600), 'maxfun': 2000,
                         'ftol': 1e-18, 'gtol': 1e-14, 'factr': 1e12}
            )
        except Exception:
            break
        
        a_new = np.maximum(result.x, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new /= s_new
            obj_new = fast_convolve(a_new).max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new.copy()
        
        if verbose:
            print(f"    temp={temp:.2f}: obj={best_obj:.10f} ({time.monotonic()-t0:.1f}s)")
    
    return best_a, best_obj
