import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_reduced(f, M, n, active_indices=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0.
    Uses constraint generation - only active constraints.
    """
    c = -np.ones(n)
    num_constraints = len(active_indices)
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.full(num_constraints, M)

    for row, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]

    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def solve_lp_full(f, M, n):
    """Solve LP with all constraints (fallback)."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 45, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 4.0)

    return best_f, best_obj


def smooth_optimize(f0, time_limit, verbose=False):
    """
    Smooth max approximation with adaptive temperature and Nesterov momentum.
    Uses softmax-weighted subgradient of the smooth approximation of max(c).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    mom = 0.92
    tau = 50.0  # temperature (inverse)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax weights on convolution values
        conv = fast_convolve(f)
        c_max = conv.max()
        scaled = tau * (conv - c_max)
        weights = np.exp(scaled)
        # Only keep significant weights for efficiency
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= w_sum

        # Compute weighted subgradient
        # sg[j] = 2 * sum_k weights[k] * a[max(0,k-n+1):min(n,k+1)] shifted
        # More precisely: d(c[k])/d(a[j]) = 2*a[k-j] (for valid k-j, k≠2j) or 2*a[j] (for k=2j)
        # sg[j] = 2 * sum_k weights[k] * a[k-j] (with bounds)
        sg = np.zeros(n)
        # Use only top-weight indices for efficiency
        top_k = min(50, len(conv))
        peak_indices = np.argsort(weights)[-top_k:]
        peak_indices = peak_indices[weights[peak_indices] > 1e-10]

        for k in peak_indices:
            wk = weights[k]
            if wk < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            vals = f[k - idx]  # f[k-j] for j in [js, je]
            # Handle the double counting at k=2j
            sg[idx] += 2.0 * wk * vals

        # Nesterov lookahead
        f_look = f + mom * v
        f_look = project_simplex(f_look)
        # Recompute gradient at lookahead point (optional, expensive)
        # For efficiency, use gradient at current point
        v_new = mom * v - lr * sg
        f = project_simplex(f + v_new)
        v = v_new

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-16:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate and temperature
        lr *= 0.9998
        tau *= 1.005  # gradually sharpen the softmax

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.003
            tau = 50.0
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      smooth iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent_peaks(f0, time_limit):
    """
    Coordinate descent that specifically targets the peak coordinates.
    For each peak position k, optimize the coordinates that most affect c[k].
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find top peaks
        num_peaks = min(10, len(conv))
        peak_indices = np.argsort(conv)[-num_peaks:]

        for k in sorted(peak_indices, key=lambda x: -conv[x]):
            if time.monotonic() - t0 > time_limit:
                break

            # Coordinates that affect c[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            # Try adjusting each coordinate in the affected range
            for j in range(js, je + 1, max(1, (je - js) // 10)):
                old_val = f[j]
                if old_val < 1e-10:
                    continue

                # Try multiplicative changes
                for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial /= ss
                    val = fast_convolve(f_trial).max()
                    if val < M - 1e-12:
                        f = f_trial
                        M = val
                        improved = True
                        if M < best_obj - 1e-14:
                            best_obj = M
                            best_f = f.copy()
                        break

        if not improved and passes > 3:
            break

    return best_f, best_obj


def make_linear_init(n, power=1.0):
    """Create a linearly-decreasing initialization: a[j] ∝ (n-j)^power."""
    j = np.arange(n)
    a = (n - j) ** power
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exp_init(n, alpha=1.0):
    """Create an exponentially-decreasing initialization."""
    j = np.arange(n)
    a = np.exp(-alpha * j / n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal_init(n, width=0.3):
    """Create a bimodal initialization (high at edges)."""
    j = np.linspace(0, 1, n)
    a = np.exp(-((j - 0) / width) ** 2) + np.exp(-((j - 1) / width) ** 2) + 0.1
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining diverse initializations,
    LP-FW with constraint generation, smooth optimization,
    coordinate descent, and length sweeping.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def update_best(f, label=""):
        nonlocal best_seq, best_score
        sc = float(evaluate_sequence(f.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(f)})")
            return True
        return False

    # === Get parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        parent = make_linear_init(768, power=1.0)
        update_best(parent, "default_linear")
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Try diverse initializations at n=768 ===
    print(f"\n=== Phase 1: Diverse inits (best={best_score:.8f}) ===")
    n_base = len(parent)

    inits = {}
    inits["parent"] = parent.copy()
    inits["linear1"] = make_linear_init(n_base, 1.0)
    inits["linear2"] = make_linear_init(n_base, 1.5)
    inits["linear05"] = make_linear_init(n_base, 0.5)
    inits["exp1"] = make_exp_init(n_base, 1.0)
    inits["exp2"] = make_exp_init(n_base, 2.0)
    inits["exp05"] = make_exp_init(n_base, 0.5)
    inits["bimodal"] = make_bimodal_init(n_base, 0.3)

    # Add noisy variants of parent
    for sc in [0.005, 0.01, 0.02]:
        noise = rng.standard_normal(n_base) * sc
        inits[f"parent_noise{sc}"] = project_simplex(parent + noise)

    # Quick screening: 15s LP-FW each
    for label, f_init in inits.items():
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(15, rem * 0.08)
        if t_alloc < 5:
            break
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
        update_best(a_opt, f"screen_{label}")

    # === Phase 2: Deep LP-FW on best ===
    print(f"\n=== Phase 2: Deep LP-FW (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t2 = min(rem, 500)
        print(f"  Budget: {t2:.0f}s, n={n_best}")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        update_best(a_opt, "deep_lp_fw")

    # === Phase 3: Smooth optimization ===
    print(f"\n=== Phase 3: Smooth opt (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t3 = min(rem * 0.5, 150)
        print(f"  Budget: {t3:.0f}s")
        a_opt, _ = smooth_optimize(best_seq, time_limit=t3, verbose=True)
        update_best(a_opt, "smooth")

    # === Phase 4: Further LP-FW from smooth result ===
    print(f"\n=== Phase 4: LP-FW after smooth (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        t4 = min(rem * 0.5, 150)
        print(f"  Budget: {t4:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
        update_best(a_opt, "lp_fw_after_smooth")

    # === Phase 5: Coordinate descent on peaks ===
    print(f"\n=== Phase 5: Coord descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15:
        t5 = min(rem * 0.4, 80)
        print(f"  Budget: {t5:.0f}s")
        a_opt, _ = coordinate_descent_peaks(best_seq, time_limit=t5)
        update_best(a_opt, "coord_desc")

    # === Phase 6: Length sweeping ===
    print(f"\n=== Phase 6: Length sweep (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-128, -64, -32, -16, 16, 32, 64, 128, 256]:
        nl = n_best + delta
        if 200 <= nl <= 2500:
            candidate_lengths.append(nl)

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 100
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(40, rem * 0.15)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        update_best(a_opt, f"len_{target_n}")

    # === Phase 7: Multi-start from different shapes at best length ===
    print(f"\n=== Phase 7: Multi-start (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    shapes = [
        ("lin1", make_linear_init(n_best, 1.0)),
        ("lin15", make_linear_init(n_best, 1.5)),
        ("exp1", make_exp_init(n_best, 1.0)),
        ("exp3", make_exp_init(n_best, 3.0)),
    ]
    # Blend with current best
    for label, shape in shapes:
        blended = 0.7 * best_seq + 0.3 * shape
        blended = project_simplex(blended)
        rem = deadline - time.monotonic() - 80
        if rem < 15:
            break
        t_alloc = min(30, rem * 0.2)
        if t_alloc < 8:
            continue
        a_opt, _ = lp_fw_optimize(blended, time_limit=t_alloc)
        update_best(a_opt, f"blend_{label}")

    # === Phase 8: Final polishing ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        # Quick smooth with very fine steps
        t8a = min(rem * 0.4, 60)
        a_opt, _ = smooth_optimize(best_seq, time_limit=t8a)
        update_best(a_opt, "final_smooth")

        # Then LP-FW squeeze
        rem2 = deadline - time.monotonic() - 10
        if rem2 > 15:
            t8b = min(rem2 * 0.8, 60)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t8b)
            update_best(a_opt, "final_lp_fw")

    # === Phase 9: Micro-perturbations ===
    print(f"\n=== Phase 9: Micro-search (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        scale = 0.0002 + trial * 0.00015
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [perturb{trial}]: {sc:.8f}")
            # Quick polish
            rem2 = deadline - time.monotonic()
            if rem2 > 8:
                a_opt, _ = smooth_optimize(f_pert, time_limit=min(rem2 * 0.3, 15))
                update_best(a_opt, f"perturb{trial}_polish")

    # Fallback
    if best_seq is None:
        best_seq = make_linear_init(512, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = make_linear_init(512, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
