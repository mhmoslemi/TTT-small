LP-guided optimization for minimizing 2*n*max(auto_conv)/sum^2.
Uses the state-of-the-art LP direction-finding approach with line search,
multiple initializations, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import fft, ifft


def fast_autocorr(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum=1) or None if infeasible.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    # (a*g)[k] = sum_j A[k,j] * g[j]
    A_rows = []
    for k in range(num_constraints):
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a[k - j]
        A_rows.append(row)
    A_mat = np.array(A_rows)
    
    # Maximize sum(g) = - minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A_mat @ g <= M (all k)
    b = np.full(num_constraints, M)
    
    # g >= 0 handled by bounds
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = normalize(res.x)
            return g
        else:
            return None
    except Exception:
        return None

def subgradient_step(a, lr, top_k=50):
    """One subgradient step with projection."""
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    
    # Get top-k peaks
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    
    # Softmax weights
    temp = max(1e-6, (pv.max() - pv.min())) * 0.5
    scaled = pv / temp - (pv / temp).max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Subgradient
    sg = np.zeros(n)
    for i in range(k_actual):
        if w[i] < 1e-10:

        k = int(peaks[i])

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    
    # Multiplicative update (mirror descent)
    a_new = np.maximum(a * np.exp(-lr * sg), 1e-15)
    a_new = a_new / a_new.sum()
    
    return a_new

def coord_descent(a, time_budget, verbose=False):
    """Coordinate descent with golden-section-like search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_budget:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_budget:
                break
            
            old_val = a[j]
            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue
            
            def obj_v(v):
                if v < -1e-10 or v > 1.0 - 1e-12:
                    return float('inf')
                v = max(0.0, v)
                new_a = a.copy()
                new_a[j] = v
                new_a = new_a / new_a.sum()  # renormalize
                return fast_autocorr(new_a).max()
            
            # Sample several points
            candidates = [0.0, old_val * 0.3, old_val * 0.7, old_val,
                          old_val * 1.3, old_val * 1.7, min(0.3, old_val * 2.5)]
            best_cand = old_val
            best_obj = obj_v(old_val)
            for v in candidates:
                o = obj_v(v)
                if o < best_obj - 1e-13:
                    best_obj = o
                    best_cand = v
            
            if best_cand != old_val:
                new_a = a.copy()
                new_a[j] = best_cand
                new_a = normalize(new_a)
                sc = 2.0 * n * fast_autocorr(new_a).max()
                if sc < best_score - 1e-13:
                    a = new_a
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")
        if not improved:
            break
    
    return best_a, best_score


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_seq(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    return normalize(a)


def geometric_seq(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def triangle_seq(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def super_gaussian_seq(n, p=2.0, width_frac=0.8):
    """a[i] = exp(-((i - center)/(n*width_frac/2))^p)"""
    center = (n - 1) / 2.0
    x = (np.arange(n) - center) / (n * width_frac / 2.0)
    a = np.exp(-np.abs(x) ** p)
    return normalize(a)

"""
LP-guided optimization for minimizing 2*n*max(auto_conv)/sum^2.
Uses LP direction finding (from state-of-the-art), subgradient descent,
coordinate descent, and diverse initializations. The key innovation is
iteratively solving max sum(g) s.t. (a*g)[k]<=M, g>=0 to find descent
directions that flatten the auto-convolution.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def fast_autocorr(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2 (scale invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a, k_indices=None):
    """Build convolution submatrix A where (A@g)[i] = (a*g)[k_indices[i]]."""
    n = len(a)
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    A = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        length = je - js + 1
        A[idx, js:je+1] = a[k - js : k - je : -1]
    return A


def solve_lp_direction(a, M, n_top=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    num_k = 2 * n - 1
    if n_top is not None and n_top < num_k:
        conv = fast_autocorr(a)
        k_idx = np.argpartition(conv, -n_top)[-n_top:]
    else:
        k_idx = np.arange(num_k)
    
    A_mat = build_conv_matrix(a, k_idx)
    c = -np.ones(n)
    b = np.full(len(k_idx), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerances': 1e-9})
        if res.success and res.fun < -1e-10:
            g = res.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_optimize(a, time_budget, verbose=False, top_k=None):
    """Iterative LP-guided optimization with line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stuck_count = 0
    
    if top_k is None:
        top_k = min(500, 2 * n - 1)
    
    while time.monotonic() - t0 < time_budget:
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Solve LP with current M
        g = solve_lp_direction(a, M, n_top=top_k)
        if g is None:
            stuck_count += 1
            if stuck_count > 5:
                break
            # Perturb and retry
            noise = np.random.standard_normal(n) * 0.001
            a = project_simplex(a + noise)
            continue
        
        # Check meaningful difference
        diff = np.abs(g - a).sum()
        if diff < 1e-5:
            # Try tighter constraint
            g2 = solve_lp_direction(a, M * 0.98, n_top=top_k)
            if g2 is not None and np.abs(g2 - a).sum() > diff:
                g = g2
            else:
                stuck_count += 1
                if stuck_count > 10:
                    # Add noise and restart
                    noise = np.random.standard_normal(n) * 0.005
                    a = project_simplex(best_a + noise)
                    stuck_count = 0
                    continue
                if stuck_count > 20:
                    break
                continue
        else:
            stuck_count = 0
        
        # Line search: try multiple blend parameters
        best_t = 0.0
        best_obj = M
        t_values = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]
        for t in t_values:
            new_a = (1.0 - t) * a + t * g
            ns = new_a.sum()
            if ns < 1e-15:
                continue
            new_a = np.maximum(new_a, 0.0) / ns
            obj = fast_autocorr(new_a).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = normalize(a)
            sc = 2.0 * n * fast_autocorr(a).max()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
        else:
            stuck_count += 1
            if stuck_count > 15:
                noise = np.random.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise)
                stuck_count = 0
        
        iter_count += 1
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: score={best_score:.10f}, t={best_t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def subgradient_optimize(a, time_budget, lr0=0.005, decay=0.9995, top_k=80, verbose=False):
    """Multiplicative subgradient descent with top-k peak weighting."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    lr = lr0
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_budget:
        conv = fast_autocorr(a)
        M = conv.max()
        
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        temp = max(1e-10, (pv.max() - pv.min())) * 0.5
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        # Multiplicative update
        a = np.maximum(a * np.exp(-lr * sg), 1e-15)
        a = a / a.sum()
        
        sc = 2.0 * n * fast_autocorr(a).max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
        
        lr *= decay
        step += 1
        
        # Periodic restart to escape local minima
        if step % 3000 == 0 and lr < lr0 * 0.1:
            lr = lr0 * 0.3
            noise = np.random.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
        
        if verbose and step % 2000 == 0:
            print(f"    SG step {step}: score={best_score:.10f}, lr={lr:.6f}")
    
    return best_a, best_score


def coord_descent(a, time_budget, verbose=False):
    """Coordinate descent with candidate sampling."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_budget:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            other_sum = 1.0 - old_val
            if other_sum < 1e-12:
                continue
            
            # Try several candidates
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2,
                          old_val * 1.5, 0.0, min(0.2, old_val * 2.0)]
            best_cand = old_val
            best_obj = fast_autocorr(a).max()
            
            for v in candidates:
                v = max(0.0, v)
                if v > 1.0 - 1e-10:
                    continue
                new_a = a.copy()
                new_a[j] = v
                ns = new_a.sum()
                if ns < 1e-12:
                    continue
                new_a = new_a / ns
                obj = fast_autocorr(new_a).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_cand = v
            
            if best_cand != old_val:
                new_a = a.copy()
                new_a[j] = best_cand
                new_a = normalize(new_a)
                sc = 2.0 * n * fast_autocorr(new_a).max()
                if
