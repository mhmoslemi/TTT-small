"""
Fast coordinate-descent optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses incremental convolution updates for O(n) per-coordinate evaluation,
diverse geometric/structured initializations, multi-length exploration,
simulated-annealing global search, and log-sum-exp smooth refinement.

import numpy as np
import time


def fft_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 internally (no evaluate call)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.maximum(v, 0.0)
    sv = v.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    return v / sv

def smooth_optimize(f0, time_limit, T_start=0.3, T_end=0.01):
    """Log-sum-exp smooth optimization with Adam on the simplex."""
    f = project_simplex(f0.astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_obj = fft_convolve(f).max()

    m_st = np.zeros(n)
    v_st = np.zeros(n)
    t_adam = 0
    alpha = 0.01
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t0 = time.monotonic()
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        el = time.monotonic() - t0
        T = max(T_start * (T_end / T_start) ** (el / time_limit), T_end)
        lr = alpha * (0.3 ** (el / time_limit))

        conv = fft_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()

        # Gradient: grad[i] = (2/T) * sum_j f[j] * w[i+j] (cross-correlation)
        N_fft = 1 << (3 * n - 1).bit_length()
        A_f = np.fft.rfft(f, N_fft)
        W_f = np.fft.rfft(w, N_fft)
        corr = np.fft.irfft(A_f * np.conj(W_f), N_fft)[:n]
        grad = (2.0 / T) * corr

        t_adam += 1
        m_st = beta1 * m_st + (1 - beta1) * grad
        v_st = beta2 * v_st + (1 - beta2) * grad ** 2
        m_hat = m_st / (1 - beta1 ** t_adam)
        v_hat = v_st / (1 - beta2 ** t_adam)

        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)

        cur_obj = fft_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 5000:
            rng_p = np.random.default_rng(int(el * 1000) % 1000000)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            m_st = np.zeros(n)
            v_st = np.zeros(n)
            t_adam = 0
            no_imp = 0

    return best_f, best_obj


def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def make_asc_geo(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def make_desc_geo(n, r):
    a = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * 0.1])
    elif len(a) > n:
        a = a[:n]
    return a / a.sum()


def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()


def make_two_peak(n, pos1_frac=0.25, pos2_frac=0.75, width=0.1):
    """Two Gaussian peaks."""
    x = np.linspace(0, 1, n)
    p1 = pos1_frac
    p2 = pos2_frac
    a = np.exp(-((x - p1) ** 2) / (2 * width ** 2)) + np.exp(-((x - p2) ** 2) / (2 * width ** 2)) + 0.002
    return a / a.sum()


def make_alternating(n, ratio=0.5):
    """Alternating high/low values."""
    a = np.ones(n)
    a[::2] = 1.0
    a[1::2] = ratio
    return a / a.sum()


def make_sinusoidal(n, freq=2.0, asym=0.3):
    """Asymmetric sinusoidal."""
    x = np.linspace(0, 1, n)
    a = 1.0 + asym * np.sin(2 * np.pi * freq * x)
    a = np.maximum(a, 0.0) + 0.005
    return a / a.sum()


def make_comet(n, decay=0.95, pos=0.0):
    """Peak at position pos, decaying both directions."""
    pos_idx = int(pos * (n - 1))
    a = np.ones(n) * 0.05
    for i in range(n):
        d = abs(i - pos_idx)
        a[i] = decay ** d
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def shuffle_coordinates(f, rng, frac=0.3):
    """Randomly permute a fraction of coordinates (global search move)."""
    n = len(f)
    k = max(1, int(frac * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = f[idx].copy()
    rng.shuffle(vals)
    f[idx] = vals
    return f

            scored_inits.append((sc, a))
    scored_inits.sort(key=lambda x: x[0])
    print(f"  Generated {len(inits)} inits, best init score={scored_inits[0][0]:.6f}")

    # Try top initializations directly
    for sc, a in scored_inits[:20]:
        _try(a, "init")

    # Also try parent
    if parent is not None:
        _try(parent, "parent")

    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    print(f"  Phase 0 time: {time.monotonic()-t0:.1f}s")

    # === Phase 1: Coordinate descent on best initializations ===
    print(f"\n=== Phase 1: Coordinate Descent (best={best_score:.6f}) ===")
    t1_start = time.monotonic()

    # Pick top diverse initializations for CD
    cd_candidates = []
    seen_n = set()
    for sc, a in scored_inits:
        n_a = len(a)
        if n_a not in seen_n or sc < 1.52:
            cd_candidates.append(a)
            seen_n.add(n_a)
        if len(cd_candidates) >= 8:
            break

    # Also include the current best
    if best_seq is not None:
        cd_candidates.insert(0, best_seq)

    # Time allocation for CD
    rem = deadline - time.monotonic() - 600
    per_cd = max(20, rem / len(cd_candidates))
    print(f"  {len(cd_candidates)} CD candidates, {per_cd:.0f}s each")

    for idx, a_init in enumerate(cd_candidates):
        rem = deadline - time.monotonic() - 500
        if rem < 15:
            break
        t_alloc = min(per_cd, rem)
        print(f"  CD [{idx}] n={len(a_init)}, t={t_alloc:.0f}s")
        a_opt, _ = gauss_elimination_cd(a_init, time_limit=t_alloc)
        _try(a_opt, f"cd_{idx}")

    print(f"  Phase 1 time: {time.monotonic()-t1_start:.1f}s, best={best_score:.6f}")

    # === Phase 2: Simulated annealing / random restarts ===
    print(f"\n=== Phase 2: Global Search (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        temp = 0.02
        cooldown = 0.997
        min_temp = 0.001

        rem = deadline - time.monotonic() - 400
        sa_iter = 0
        while time.monotonic() - t2_start < rem and temp > min_temp:
            if time.monotonic() > deadline - 380:
                break
            sa_iter += 1

            # Perturb
            pert_scale = temp * 5
            noise = rng.standard_normal(n_best) * pert_scale
            a_pert = np.maximum(best_seq + noise, 0.0)

            # Occasionally do a structural perturbation
            if sa_iter % 10 == 0:
                a_pert = shuffle_coordinates(a_pert, rng, frac=0.1)
                a_pert = np.maximum(a_pert, 0.0)

            a_pert = project_simplex(a_pert)

            # Quick CD (short time)
            t_cd_quick = min(5.0, (deadline - time.monotonic() - 370) * 0.1)
            if t_cd_quick < 1.0:
                break
            a_opt, _ = gauss_elimination_cd(a_pert, time_limit=t_cd_quick)
            _try(a_opt, f"sa_{sa_iter}")

            temp *= cooldown

    print(f"  Phase 2 time: {time.monotonic()-t2_start:.1f}s, best={best_score:.6f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-Length (best={best_score:.6f}) ===")
    t3_start = time.monotonic()

    if best_seq is not None:
        for target_n in [512, 768, 1024, 1536, 2048]:
            if abs(target_n - len(best_seq)) < 50:
                continue
            rem = deadline - time.monotonic() - 250
            if rem < 40:
                break
            f_interp = interpolate_seq(best_seq, target_n)

            # CD on interpolated
            t_cd = min(60, rem * 0.4)
            a_opt, _ = gauss_elimination_cd(f_interp, time_limit=t_cd)
            _try(a_opt, f"len_{target_n}")

            # Also try fresh geometric at this length
            for r in [0.97, 0.99, 0.995]:
                geo = make_asc_geo(target_n, r)
                rem2 = deadline - time.monotonic() - 230
                if rem2 < 20:
                    break
                a_opt2, _ = gauss_elimination_cd(geo, time_limit=min(30, rem2 * 0.3))
                _try(a_opt2, f"geo_{target_n}_{r}")

    print(f"  Phase 3 time: {time.monotonic()-t3_start:.1f}s, best={best_score:.6f}")

    # === Phase 4: Smooth optimization refinement ===
    print(f"\n=== Phase 4: Smooth Refinement (best={best_score:.6f}) ===")
    t4_start = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_smooth = min(120, rem * 0.6)
            print(f"  Smooth opt n={len(best_seq)}, t={t_smooth:.0f}s")
            a_opt, _ = smooth_optimize(best_seq, time_limit=t_smooth, T_start=0.2, T_end=0.008)
            _try(a_opt, "smooth_p4")

        # CD after smooth
        rem = deadline - time.monotonic() - 80
        if rem > 20 and best_seq is not None:
            a_opt2, _ = gauss_elimination_cd(a_opt, time_limit=min(60, rem * 0.5))
            _try(a_opt2, "cd_after_smooth")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    t5_start = time.monotonic()

    if best_seq is not None:
        # Long CD run
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            print(f"  Final CD, t={rem:.0f}s")
            a_opt, _ = gauss_elimination_cd(best_seq, time_limit=rem * 0.5)
            _try(a_opt, "final_cd1")

        # More perturbations + short CD
        for _ in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 8:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = gauss_elimination_cd(f_pert, time_limit=min(15, rem * 0.4))
            _try(a_opt, "perturb_cd")

        # Shift-based perturbation
        rem = deadline - time.monotonic() - 10
        if rem > 8 and best_seq is not None:
            a_shift = np.roll(best_seq, 1)
            a_shift = np.maximum(a_shift, 0)
            a_shift = project_simplex(a_shift)
            a_opt, _ = gauss_elimination_cd(a_shift, time_limit=min(10, rem * 0.5))
            _try(a_opt, "shift_cd")

    print(f"  Phase 5 time: {time.monotonic()-t5_start:.1f}s, best={best_score:.6f}")

    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(512) / 512
        _try(best_seq, "fallback_uniform")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
