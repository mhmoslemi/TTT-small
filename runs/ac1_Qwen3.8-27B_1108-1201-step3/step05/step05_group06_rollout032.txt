import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_active(f, T, K=50):
    """Compute gradient of logsumexp(conv/T) using only top K lags."""
    n = len(f)
    conv = fast_convolve(f)
    K = min(K, len(conv))
    top_idx = np.argpartition(conv, -K)[-K:]
    top_vals = conv[top_idx]
    
    # Softmax weights
    scaled = top_vals / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for t in range(K):
        k = int(top_idx[t])
        wk = float(w[t])
        if wk < 1e-15:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += (2.0 / T) * wk * f[k-hi : k-lo+1][::-1]
    
    return grad, conv, float(conv.max())

def adam_optimize(f0, time_limit, T_start=0.1, T_end=0.005, K=50, verbose=False):
    """Adam optimizer on log-sum-exp of convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)
        T = max(T, T_end)
        lr = alpha * (0.8 ** (elapsed / (time_limit * 0.5)))
        
        grad, conv, cur_max = compute_grad_active(f, T, K)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:

            best_f = f_new.copy()
            f = f_new

        else:

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(f, T, K):
    """Compute LSE gradient using top K lags. Returns (grad, conv, max_val)."""
    n = len(f)
    conv = fast_convolve(f)
    K = min(K, len(conv))
    top_idx = np.argpartition(conv, -K)[-K:]
    top_vals = conv[top_idx]
    
    scaled = top_vals / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    grad = np.zeros(n)
    for t in range(K):
        k = int(top_idx[t])
        wk = float(w[t])
        if wk < 1e-15:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += (2.0 / T) * wk * f[k-hi : k-lo+1][::-1]
    
    return grad, conv, float(conv.max())
