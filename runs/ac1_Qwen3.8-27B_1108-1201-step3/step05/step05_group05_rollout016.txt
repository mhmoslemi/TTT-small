```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based descent on active lags,
with subgradient fallback, coordinate descent, and pair-swap polishing.
Key innovation: uses a linear program at each step to find the best descent
direction that simultaneously reduces all active (high) convolution lags.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = 0.0
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()

def lp_descent_step(a, n_active=50, max_iter=1):
    """
    One LP-based descent step.
    Returns (new_a, improvement) or (a, 0) if LP fails.
    """
    n = len(a)
    conv_aa = fast_convolve(a)
    current_max = conv_aa.max()
    
    # Identify active lags
    k_active = min(n_active, len(conv_aa))
    active_indices = np.argpartition(conv_aa, -k_active)[-k_active:]
    active_indices = active_indices[conv_aa[active_indices] > 0.5 * current_max]
    
    if len(active_indices) == 0:
        return a.copy(), 0.0
    
    # Build LP:
    # Variables: d[0..n-1], M  (total n+1 variables)
    # Minimize: M
    # Subject to:
    #   For each active lag k: sum_j 2*a[k-j]*d[j] - M <= -conv_aa[k]
    #   For each j: -d[j] <= a[j]  (i.e., a[j]+d[j] >= 0)
    #   sum(d) = 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = []
    b_ub = []
    
    # Convolution constraints
    for k in active_indices:
        row = np.zeros(n + 1)
        row[-1] = -1.0  # -M
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row[j] = 2.0 * a[idx]
        A_ub.append(row)
        b_ub.append(-conv_aa[k])
    
    # Non-negativity constraints: -d[j] <= a[j]
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_ub.append(row)
        b_ub.append(a[j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    
    if res.status != 0:
        return a.copy(), 0.0
    
    d = res.x[:n]
    
    # Line search
    best_a = a.copy()
    best_obj = current_max
    
    for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:
        a_trial = a + alpha * d
        if (a_trial < -1e-12).any():
            continue
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        obj = fast_convolve(a_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_trial.copy()
    
    return best_a, current_max - best_obj

def subgradient_step(a, n_top=30):
    """
    Subgradient step using top-K lags.
    """
    n = len(a)
    conv_aa = fast_convolve(a)
    
    k_top = min(n_top, len(conv_aa))
    top_indices = np.argpartition(conv_aa, -k_top)[-k_top:]
    
    # Weight by how much each exceeds the median of top-K
    top_vals = conv_aa[top_indices]
    weights = np.maximum(top_vals - top_vals.min(), 0)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(top_indices)) / len(top_indices)
    else:
        weights /= w_sum
    
    # Compute subgradient
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * a[k - idx]
    
    # Center and normalize
    grad = grad - grad.mean()
    return grad

def multiplicative_update(a, n_top=30, eta=0.1):
    """
    Multiplicative update to maintain non-negativity.
    """
    n = len(a)
    conv_aa = fast_convolve(a)
    
    k_top = min(n_top, len(conv_aa))
    top_indices = np.argpartition(conv_aa, -k_top)[-k_top:]
    
    top_vals = conv_aa[top_indices]
    max_val = top_vals.max()
    
    # Pressure: for each coordinate j, how much it contributes to high lags
    pressure = np.zeros(n)
    for k in top_indices:
        k = int(k)
        w = (conv_aa[k] / max_val) ** 2
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        pressure[idx] += w * a[k - idx]
    
    # Multiplicative update
    a_new = a / (1.0 + eta * pressure / (a + 1e-30))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    
    return a_new

def coordinate_descent(a0, time_limit, verbose=False, n_mult=8):
    """Coordinate descent with refined multiplier search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    converged = False

    while time.monotonic() - t0 < time_limit and not converged:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            # Geometric multipliers
            multipliers = np.geomspace(0.3, 3.0, n_mult)
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 1:
            converged = True

    return best_a, best_obj

def interpolate_sequence(f, new_n):

    old_n = len(f)

    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()

    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based active-set descent,
multiplicative updates, subgradient fallback, coordinate descent, and
pair-swap polishing. Key innovation: at each step, solves a linear program
to find the best direction that simultaneously reduces all active (high)
auto-convolution lags, exploiting the linearization of the quadratic
convolution around the current point.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = 0.0
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()

    a = rng.exponential(1.0, n)
    if parent is not None:

        p_interp = np.maximum(p_interp, 0)
        ps = p_interp.sum()
        if ps > 1e-10:
            p_interp /= ps
        a = 0.5 * a + 0.5 * p_interp
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

    return a / s


def lp_descent_step(a, n_active=50, lp_timeout=20.0):
    """
    One LP-based descent step. Solves LP to find direction d minimizing
    the maximum of linearized auto-convolution at active lags.
    Returns (new_a, improvement).
    """
    n = len(a)
    conv_aa = fast_convolve(a)
    current_max = conv_aa.max()

    # Identify active lags (top values)
    k_active = min(n_active, len(conv_aa))
    active_indices = np.argpartition(conv_aa, -k_active)[-k_active:]
    # Only keep lags significantly above zero
    threshold = 0.3 * current_max
    active_indices = active_indices[conv_aa[active_indices] > threshold]

    if len(active_indices) == 0:
        return a.copy(), 0.0

    # Build LP:
    # Variables: d[0..n-1], M  (total n+1 variables)
    # Minimize: M
    # Subject to:
    #   For each active lag k: sum_j 2*a[k-j]*d[j] - M <= -conv_aa[k]
    #   For each j: -d[j] <= a[j]  (i.e., a[j]+d[j] >= 0)
    #   sum(d) = 0
    c = np.zeros(n + 1)
    c[-1] = 1.0

    rows = []
    bounds_ub = []

    # Convolution constraints (one per active lag)
    for k in active_indices:
        k = int(k)
        row = np.zeros(n + 1)
        row[-1] = -1.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = 2.0 * a[k - j]
        rows.append(row)
        bounds_ub.append(-conv_aa[k])

    # Non-negativity: -d[j] <= a[j]
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        rows.append(row)
        bounds_ub.append(a[j])

    A_ub = np.array(rows)
    b_ub = np.array(bounds_ub)

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    t_start = time.monotonic()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': lp_timeout})
    except Exception:
        return a.copy(), 0.0

    if time.monotonic() - t_start > lp_timeout * 1.5:
        return a.copy(), 0.0

    if res.status != 0:
        return a.copy(), 0.0

    d = res.x[:n]

    # Line search: try multiple step sizes
    best_a = a.copy()
    best_obj = current_max

    for alpha in [1.0, 0.9, 0.75, 0.6, 0.4, 0.25, 0.1]:
        a_trial = a + alpha * d
        if (a_trial < -1e-10).any():
            continue
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial /= s
        obj = fast_convolve(a_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_trial.copy()

    return best_a, current_max - best_obj


def subgradient_step(a, n_top=50):
    """Compute exact subgradient as weighted average of top-K lag gradients."""
    n = len(a)
    conv_aa = fast_convolve(a)

    k_top = min(n_top, len(conv_aa))
    top_indices = np.argpartition(conv_aa, -k_top)[-k_top:]

    top_vals = conv_aa[top_indices]
    max_val = top_vals.max()
    # Exponential weighting for emphasis
    weights = np.exp(5.0 * (top_vals - max_val) / (max_val + 1e-30))
    w_sum = weights.sum()
    weights /= w_sum

    # Compute subgradient
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        if weights[i] < 1e-15:
            continue
        k = int(top_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * a[k - idx]

    grad = grad - grad.mean()
    return grad


def multiplicative_update(a, n_top=50, eta=0.08):
    """Multiplicative update maintaining non-negativity."""
    n = len(a)
    conv_aa = fast_convolve(a)

    k_top = min(n_top, len(conv_aa))
    top_indices = np.argpartition(conv_aa, -k_top)[-k_top:]

    top_vals = conv_aa[top_indices]
    max_val = top_vals.max()

    pressure = np.zeros(n)
    for k in top_indices:
        k = int(k)
        w = (conv_aa[k] / (max_val + 1e-30)) ** 3
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        pressure[idx] += w * a[k - idx]

    a_new = a / (1.0 + eta * pressure / (a + 1e-30))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()

    return a_new


def coordinate_descent(a0, time_limit, n_mult=8):
    """Coordinate descent with geometric multiplier search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = np.geomspace(0.3, 3.0, n_mult)
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 1:
            break

    return best_a, best_obj


def pair_swap_search(a0, time_limit, seed=0):
    """Enhanced pair-swap local search with adaptive perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)

    base_fracs = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue

            found = False
            for frac in base_fracs:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        found = True
                        no_improve = 0
                        break
                if found:
                    break

            if not found:
                no_improve += 1

        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise, target_sum=1.0)
            no_improve = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_update(best_score, best_seq, a, label=""):
    """Try updating best sequence. Returns (best_score, best_seq)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_score, best_seq
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score, a.copy()
    return best_score, best_seq


def optimize_lp(a0, time_limit, n_active=50, seed=0):
    """LP-based optimization with subgradient fallback."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    rng = np.random.default_rng(seed + 9999)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Adapt active set size
        if no_improve > 40:
            n_act_eff = min(150, n_active + no_improve // 3)
        else:
            n_act_eff = n_active

        # Try LP step
        t_lp_start = time.monotonic()
        a_new, improvement = lp_descent_step(a, n_active=n_act_eff,
                                             lp_timeout=min(15.0, rem * 0.5))
        lp_time = time.monotonic() - t_lp_start

        if improvement > 1e-14:
            a = a_new
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Fallback: subgradient when LP is slow or stuck
        if no_improve > 3 or lp_time > 10.0:
            grad = subgradient_step(a, n_top=50)
            found = False
            for lr in [0.06, 0.04, 0.03, 0.02, 0.01, 0.005]:
                a_trial = project_simplex(a - lr * grad, target_sum=1.0)
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_obj - 1e-13:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if not found and no_improve > 10:
                # Try multiplicative update
                a_mu = multiplicative_update(a, n_top=50, eta=0.05)
                obj_mu = fast_convolve(a_mu).max()
                if obj_mu < best_obj - 1e-13:
                    a = a_mu
                    best_obj = obj_mu
                    best_a = a.copy()
                    no_improve = 0

        # Periodic perturbation to escape local optima
        if no_improve > 60:
            noise_scale = 0.003 + 0.002 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(best_a + noise, target_sum=1.0)
            no_improve = 0

        if iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.10f}, no_imp={no_improve} ({elapsed:.0f}s)")

    return best_a, best_obj


def optimize_multiplicative(a0, time_limit, seed=0):
    """Multiplicative update optimization with annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    iter_count = 0
    eta = 0.1
    n_top = 50
    rng = np.random.default_rng(seed + 7777)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        a_new = multiplicative_update(a, n_top=n_top, eta=eta)
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(0.01, eta * 0.998)
        else:
            # Blend towards best
            blend = min(1.0, 0.3 + 0.01 * iter_count)
            a = blend * a_new + (1.0 - blend) * best_a
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a /= ss
            eta = max(0.005, eta * 0.95)

        # Periodic restart
        if iter_count % 500 == 0:
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(best_a + noise, target_sum=1.0)

        if iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MU iter {iter_count}: obj={best_obj:.10f}, eta={eta:.4f} ({elapsed:.0f}s)")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based active-set descent, multiplicative updates,
    subgradient fallback, coordinate descent, and pair-swap search.
    Exploits the linearization of quadratic convolution for efficient global
    direction finding via linear programming.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = flat_spectrum_init(768, rng)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Generated init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 1: LP-based optimization on parent ===
    print(f"\n=== Phase 1: LP Descent on parent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.35, 120)
        print(f"  LP t={t_alloc:.0f}s, n_active=50")
        a_opt, obj = optimize_lp(parent, time_limit=t_alloc, n_active=50, seed=seed)
        best_score, best_seq = try_update(best_score, best_seq, a_opt, "lp_parent")

    # === Phase 2: Multiplicative updates ===
    print(f"\n=== Phase 2: Multiplicative Updates (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20 and best_seq is not None:
        t_alloc = min(rem * 0.25, 80)
        print(f"  MU t={t_alloc:.0f}s")
        a_opt, obj = optimize_multiplicative(best_seq, time_limit=t_alloc, seed=seed)
        best_score, best_seq = try_update(best_score, best_seq, a_opt, "mu")

    # === Phase 3: LP with different active set sizes ===
    print(f"\n=== Phase 3: LP different params (best={best_score:.6f}) ===")
    if best_seq is not None:
        for n_act in [30, 80, 120]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.2, 60)
            print(f"  LP n_active={n_act}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_lp(best_seq, time_limit=t_alloc, n_active=n_act, seed=seed + n_act)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, f"lp_na{n_act}")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse Inits (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        inits = [flat_spectrum_init(n_best, rng), random_init(n_best, rng)]
        names = ["flat", "rand"]
        for idx, (init_a, name) in enumerate(zip(inits, names)):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 60)
            print(f"  {name} init + LP, t={t_alloc:.0f}s")
            a_opt, obj = optimize_lp(init_a, time_limit=t_alloc, n_active=50, seed=seed + 100 + idx)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, f"{name}_lp")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd = min(rem * 0.4, 100)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd, n_mult=10)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, "cd")

    # === Phase 6: Pair swap search ===
    print(f"\n=== Phase 6: Pair Swap (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_ps = min(rem * 0.5, 60)
            print(f"  PS t={t_ps:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, "ps")

    # === Phase 7: Longer sequences ===
    print(f"\n=== Phase 7: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best + 128, n_best + 256]:
            if target_n > 1536 or target_n < 256:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 25:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 50)
            print(f"  Long n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = optimize_lp(f_interp, time_limit=t_alloc, n_active=50, seed=seed + 200)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, f"long_{target_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_cd = min(rem * 0.5, 40)
            print(f"  Final CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd, n_mult=8)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, "final_cd")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_ps = min(rem, 20)
            print(f"  Final PS t={t_ps:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed + 1)
            best_score, best_seq = try_update(best_score, best_seq, a_opt, "final_ps")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
