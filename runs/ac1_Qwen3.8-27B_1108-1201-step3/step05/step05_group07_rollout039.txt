"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1) High-temperature soft-max L-BFGS for smooth optimization
2) Active-set subgradient polishing using KKT optimality conditions
3) Diverse asymmetric initializations and multi-length exploration

polishing directly targets the flatness of the auto-correlation.

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_weighted(w, a):
    """Compute sum_k w[k] * a[k-j] for all j (the subgradient of weighted b)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # conv(w, a)[k] = sum_i w[i]*a[k-i], we want sum_k w[k]*a[k-j] for each j
    # This is the cross-correlation: (w~ * a)[j] where w~ is reversed
    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    result = irfft(W_rev * A, fft_size)[:n]
    return result


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_grad(z, T, n):
    """
    Compute the soft-max smoothed objective and gradient w.r.t. z.
    Objective: (2n/T) * log(sum_k exp(T * b_k)) where b = conv(a,a), a = softmax(z).
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(sum_exp)
    obj = (2.0 * n / T) * log_sum_exp

    # Weights for each lag
    w = exp_Tb / sum_exp

    # grad_a_j = 2n * sum_k w_k * a_{k-j}
    grad_a = 2.0 * n * cross_corr_weighted(w, a)

    # Chain rule for softmax: grad_z_j = a_j * (grad_a_j - a_j * sum_i grad_a_i * a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_optimize(z0, T, n, time_limit, maxiter=5000):
    """L-BFGS optimization in log-space with time limit."""
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def active_set_polish(a0, time_limit=60.0, top_k=30, lr_init=0.01):
    """
    Active-set subgradient polishing using KKT conditions.
    At optimum: sum_{k in active} w_k * grad(b_k) = const (on tangent space).
    We iteratively find optimal weights and take steps toward satisfaction.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, fast_convolve(a).max()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init

    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find active set (top-k lags)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        peaks = peaks[np.argsort(b[peaks])[::-1]]

        # Check KKT: find weights w making sum w_k * P_{k}a as constant as possible
        # Build matrix P: P[i,j] = a[peaks[i]-j] if valid
        m = len(peaks)
        P = np.zeros((m, n))
        for i, k in enumerate(peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                P[i, js:je + 1] = a[k - je:k - js + 1][::-1]

        # Solve for weights: minimize ||P^T w - c*1||^2 s.t. w>=0, sum w=1
        # Use projected gradient on the weights
        w = np.ones(m) / m
        for inner in range(50):
            v = P.T @ w  # n-vector
            v_centered = v - v.mean()
            # Gradient of ||v_centered||^2 w.r.t. w is P @ v_centered
            grad_w = P @ v_centered
            w = w - 0.05 * grad_w
            w = project_simplex(w)

        # Compute the subgradient direction
        v = P.T @ w  # weighted sum of shifted a's
        direction = v - v.mean()  # projected to tangent space

        # Check if we're near optimal (direction should be ~0)
        dir_norm = np.linalg.norm(direction) / (np.linalg.norm(v) + 1e-300)
        if dir_norm < 1e-6:
            break

        # Take a step
        a_new = a - lr * 2 * direction
        a_new = project_simplex(a_new)

        # Line search
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_trial = a - trial_lr * 2 * direction
            a_trial = project_simplex(a_trial)
            b_trial = fast_convolve(a_trial)
            if b_trial.max() < M - 1e-15:
                a = a_trial
                M = b_trial.max()
                improved = True
                break

        if not improved:
            lr *= 0.7
            if lr < 1e-10:
                break
        else:
            lr = min(lr * 1.05, 0.1)

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()

    return best_a, best_obj


def subgradient_method(a0, time_limit=60.0, top_k=20):
    """
    Subgradient method with adaptive step size for minimizing max(b_k).
    Uses the minimum-norm subgradient (weighted average of active peaks).
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, fast_convolve(a).max()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)  # momentum
    mom = 0.85
    lr = 0.005
    no_improve = 0

    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find top-k peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]

        # Compute softmax-weighted subgradient
        pv = b[peaks]
        scale = 50.0  # temperature for peak weighting
        scaled = (pv - pv.max()) * scale
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient: sum_k w_k * 2 * P_k a
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Project to tangent space
        sg = sg - sg.mean()

        # Update with momentum
        v = mom * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        b_new = fast_convolve(a_new)
        M_new = b_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
                v *= 0.5
            if no_improve > 500:
                # Restart from best with perturbation
                rng = np.random.default_rng(iteration)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = 0.003
                no_improve = 0

    return best_a, best_obj


def coordinate_descent_max(a0, time_limit=60.0):
    """Coordinate descent targeting the max convolution value."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, fast_convolve(a).max()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.8, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def power_norm_lbgfs(z0, p, n, time_limit, maxiter=3000):
    """Optimize using power norm (sum b_k^p)^(1/p) which approximates max for large p."""
    t0 = time.monotonic()

    func_obj = (2.0 * n) * (fast_convolve(softmax_from_z(z0)) ** p).sum() ** (1.0 / p)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        b = fast_convolve(a)
        # (sum b^p)^(1/p)
        bp = b ** p
        s = bp.sum()
        if s < 1e-300:
            return 1e15
        return (2.0 * n) * s ** (1.0 / p)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        bp = b ** p
        s = bp.sum()
        if s < 1e-300:
            return np.zeros_like(z)
        # d/dz [(sum b^p)^(1/p)] = (1/p) * (sum b^p)^(1/p-1) * sum p*b^(p-1) * db/dz
        # = (sum b^p)^(1/p-1) * sum b^(p-1) * db/dz
        # db_k/dz_j = a_j * (2*a_{k-j} - a_j * sum_i 2*a_{k-i}*a_i)
        # Simplify: grad_a_j = 2 * sum_k b_k^(p-1) * a_{k-j}
        weights = b ** (p - 1)
        grad_a = 2.0 * cross_corr_weighted(weights / weights.sum(), a)
        # Scale factor
        factor = (s ** (1.0 / p - 1.0)) * 2.0 * n * (weights.sum())
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S) * factor / (weights.sum() + 1e-300)
        # Normalize
        return grad_z

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_init(n, rng, style="ramp"):
    """Create diverse asymmetric initializations."""
    if style == "ramp":
        a = np.arange(1, n + 1, dtype=np.float64)
    elif style == "exp_ramp":
        a = np.exp(np.linspace(0, 2, n))
    elif style == "two_peak":
        a = np.zeros(n)
        a[0] = 1.0
        a[n - 1] = 1.0
        a += 0.1 + rng.random(n) * 0.1
    elif style == "sine_asym":
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1 * np.cos(x) + 0.5
        a = np.maximum(a, 0.01)
    elif style == "quadratic":
        x = np.linspace(0, 1, n)
        a = (x ** 2 + 0.1)
    else:
        a = 1.0 + rng.standard_normal(n) * 0.3
        a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Soft-max L-BFGS with diverse initializations at multiple lengths
    Phase 2: Active-set KKT polishing
    Phase 3: Subgradient refinement
    Phase 4: Coordinate descent micro-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_init(768, rng, "ramp")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 1: Multi-init soft-max L-BFGS ===
    print("\n=== Phase 1: Multi-init L-BFGS ===", flush=True)
    n_parent = len(parent)

    # Define initializations to try
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("ramp", make_asymmetric_init(n_parent, rng, "ramp")))
    inits.append(("exp_ramp", make_asymmetric_init(n_parent, rng, "exp_ramp")))
    inits.append(("quadratic", make_asymmetric_init(n_parent, rng, "quadratic")))

        if nl == n_parent:
            continue
        interp = interpolate_sequence(parent, nl)
        inits.append((f"parent_n{nl}", interp))

        if rem < 15:

        z0 = z_from_a(a0)
        z_opt = lbgfs_optimize(z0, 50, n, t_alloc, maxiter=2000)
        a_opt = softmax_from_z(z_opt)
        candidates.append((a_opt, label))
        try_seq(a_opt, f"lbfgs_T50_{label}")

    print("\n--- Phase 1b: Higher T refinement ---", flush=True)
