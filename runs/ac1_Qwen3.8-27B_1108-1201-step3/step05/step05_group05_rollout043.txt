    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_lp_matrix_indices(n):
    """Precompute row/col indices for the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp_direction(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g)_k <= M."""
    # Values for the sparse matrix
    # A[k,j] = f[k-j] where k-j is the appropriate index
    vals = np.empty(len(row_idx))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        vals[idx] = f[k - j]
    
    n_cons = 2 * n - 1
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(n_cons, n))
    
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 15})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception as e:
        pass
    
    return None

def lp_optimize(f0, time_limit, verbose=False):
    """Main LP-guided optimization loop."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    row_idx, col_idx = compute_lp_matrix_indices(n)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    t_step = 0.1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M, n, row_idx, col_idx)
        
        if g is None:
            # Fallback: use gradient direction
            conv = fast_convolve(f)
            k_max = np.argmax(conv)
            js = max(0, k_max - n + 1)
            je = min(n - 1, k_max)
            grad = np.zeros(n)
            idx = np.arange(js, je + 1)
            grad[idx] = 2.0 * f[k_max - idx]
            # Move away from the peak
            g = project_simplex(f - 0.01 * grad, 1.0)
        
        # Try different step sizes
        best_t = t_step
        best_M_trial = M
        for t in [t_step, 0.5 * t_step, 0.25 * t_step, 0.75 * t_step, t_step * 1.5]:
            if t <= 0 or t > 0.9:
                continue
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial /= ss
            M_trial = fast_convolve(f_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_t = t
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f / f.sum()
        M = fast_convolve(f).max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_improve = 0
            t_step = min(0.3, t_step * 1.1)
        else:
            no_improve += 1
            t_step = max(0.02, t_step * 0.95)
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: M={M:.8f}, best={best_M:.8f}, t={t_step:.3f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_f, best_M

def coordinate_descent(a0, time_limit):
    """Simple coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.8, 1.2, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                remaining = 1.0 - old_val
                if remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[np.arange(n) != j] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                remaining = 1.0 - old_val
                if remaining > 1e-10:
                    scale = (1.0 - best_j_val) / remaining
                    a[j] = best_j_val
                    a[np.arange(n) != j] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def flat_spectrum_init(n, rng):
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP direction-finding (from SOTA) with coordinate descent polishing

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_conv_indices(n):

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the LP direction-finding approach (solve max sum(g) s.t. conv(f,g)<=M, g>=0)
combined with subgradient refinement, coordinate descent polishing, and
multi-start exploration for robustness.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_conv_indices(n):
    """Precompute row/col indices for the convolution constraint sparse matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, M, n, row_idx, col_idx, use_top_k=None):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k.
    Returns normalized g (sum=1) or None on failure.
    If use_top_k is set, only enforce the top-k most constraining inequalities.
    """
    vals = np.empty(len(row_idx))
    for i in range(len(row_idx)):
        k = row_idx[i]
        j = col_idx[i]
        vals[i] = f[k - j]

    n_cons = 2 * n - 1

    if use_top_k is not None and use_top_k < n_cons:
        conv_f = fast_convolve(f)
        peaks = np.argpartition(conv_f, -use_top_k)[-use_top_k:]
        peak_set = set(peaks.tolist())
        mask = np.array([row_idx[i] in peak_set for i in range(len(row_idx))])
        A_ub = csr_matrix((vals[mask], (row_idx[mask], col_idx[mask])),
                          shape=(n_cons, n))
    else:
        A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(n_cons, n))

    b_ub = np.full(n_cons, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass

    return None

def coordinate_descent(a0, time_limit):

        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:

                if new_val < 0 or new_val > 0.999:

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the LP direction-finding approach (solve max sum(g) s.t. conv(f,g)<=M, g>=0)
combined with subgradient refinement, coordinate descent polishing, and

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_conv_indices(n):
    """Precompute row/col indices for the convolution constraint sparse matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, M, n, row_idx, col_idx, use_top_k=None):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k.
    Returns normalized g (sum=1) or None on failure.
    """
    vals = np.empty(len(row_idx))
    for i in range(len(row_idx)):
        k = row_idx[i]
        j = col_idx[i]
        vals[i] = f[k - j]

    n_cons = 2 * n - 1

    if use_top_k is not None and use_top_k < n_cons:
        conv_f = fast_convolve(f)
        peaks = np.argpartition(conv_f, -use_top_k)[-use_top_k:]
        peak_set = set(peaks.tolist())
        mask = np.array([row_idx[i] in peak_set for i in range(len(row_idx))])
        A_ub = csr_matrix((vals[mask], (row_idx[mask], col_idx[mask])),
                          shape=(n_cons, n))
    else:
        A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(n_cons, n))

    b_ub = np.full(n_cons, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass

    return None
