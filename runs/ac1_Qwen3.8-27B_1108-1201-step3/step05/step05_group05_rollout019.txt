    y_hist = []  # gradient differences
    rho_hist = []
    
    tau = tau_init

        log_w_max = log_w.max()

        cross = np.fft.irfft(W * np.conj(A), preL)
        grad = 2.0 * cross[:n]

        grad_centered = grad - grad.mean()
        
        # L-BFGS two-loop recursion
        q = grad_centered.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient optimization with
adaptive temperature, exact subgradient polishing, multi-start stratified initializations,
and targeted peak deflation. Key improvements: clean FFT gradient formula,
log-sum-exp smoothing without top-k truncation, L-BFGS-inspired memory,
and a final exact-subgradient deflation phase.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, w, L):
    """Compute gradient of smoothed objective. w is softmax weight on c=a*a.
    grad[m] = 2 * sum_k w[k] * a[k-m] (valid indices only)"""
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)
    return 2.0 * cross[:len(a)]


def softmax_weights(c, tau):
    """Compute softmax weights with numerical stability."""
    log_w = tau * c
    log_w -= log_w.max()
    w = np.exp(log_w)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

def peak_deflation(a0, time_limit, verbose=False):
    """Targeted local search: reduce contribution to top peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    L = 1 << (3 * n).bit_length()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        # Find top peaks
        k_peaks = min(10, len(c))
        peak_indices = np.argpartition(c, -k_peaks)[-k_peaks:]
        peak_indices = peak_indices[np.argsort(c[peak_indices])[::-1]]

        improved = False
        for k in peak_indices[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            # Find which a[j] contribute most to c[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            # Try reducing the top contributor
            top_local = idx[np.argmax(contrib)]
            
            for frac in [0.05, 0.1, 0.15, 0.25]:
                old_val = a[top_local]
                transfer = frac * old_val
                if transfer < 1e-10:
                    continue
                # Redistribute to a random position (preferring low-contribution areas)
                # Choose target: position that minimizes increase in nearby peaks
                target = rng.integers(0, n)
                if target == top_local:
                    continue
                
                a_trial = a.copy()
                a_trial[top_local] = old_val - transfer
                a_trial[target] += transfer
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            # Try small random perturbations
            best_pert = a.copy()
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                j = rng.integers(0, n)
                l = rng.integers(0, n)
                if j == l or a[j] < 1e-10:
                    continue
                transfer = 0.1 * a[j]
                a_trial = a.copy()
                a_trial[j] = a[j] - transfer
                a_trial[l] = a[l] + transfer
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True

    return best_a, best_obj

                if new_val < 0 or new_val > 0.999:
                    continue
                rem = 1.0 - old_val
                if rem < 1e-10:
                    continue
                scale = (1.0 - new_val) / rem
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                rem = 1.0 - old_val
                if rem > 1e-10:
                    scale = (1.0 - best_j_val) / rem
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj

        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)

        a_clipped = np.minimum(a, 1000.0)
        score = float(evaluate_sequence(a_clipped.tolist()))
        if np.isfinite(score) and score < best_score:

            best_seq = a_clipped.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    parent = None
    try:

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
