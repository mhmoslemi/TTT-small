import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_normalized(f):
    """Evaluate objective for normalized sequence (sum=1, non-negative)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):

    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

    return a

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)

    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        for k in peak_indices:
            k = int(k)
            if conv[k] < M * 0.9:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]

            # Try multiple candidate coordinates to modify
            top_pairs = np.argsort(products)[-5:]

            for pi in top_pairs:
                j_max = idx[pi]
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                best_trial = None
                best_trial_obj = M

                for mult in [0.2, 0.4, 0.6, 0.8, 1.3, 1.6, 2.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    trial = f.copy()
                    trial[j_max] = new_val
                    diff = old_val - new_val
                    if abs(diff) > 1e-15:
                        # Redistribute to non-contributing elements for this peak
                        contrib_mask = np.zeros(n, dtype=bool)
                        contrib_mask[idx] = True
                        non_contrib = ~contrib_mask
                        if non_contrib.sum() > 0:
                            nc_weights = f[non_contrib].copy()
                            nc_sum = nc_weights.sum()
                            if nc_sum > 1e-15:
                                trial[non_contrib] += diff * nc_weights / nc_sum
                            else:
                                trial[non_contrib] += diff / non_contrib.sum()
                        else:
                            w = f.copy()
                            w[j_max] = 0
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial += diff * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = fast_convolve(trial).max()
                    if obj < best_trial_obj - 1e-14:
                        best_trial_obj = obj
                        best_trial = trial

                if best_trial is not None:
                    f = best_trial
                    if best_trial_obj < best_obj - 1e-14:
                        best_obj = best_trial_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Create a bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP-FW, multiplicative descent with Nesterov
    momentum, peak surgery, Fourier-based structured perturbations, and
    adaptive restarts. Focuses on the parent length with exploration of
    nearby lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)

    n_parent = len(parent)

    # === Phase 1: Quick LP-FW warmup on parent ===
    print(f"\n=== Phase 1: LP-FW Warmup (n={n_parent}) ===", flush=True)
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 120)
        print(f"  Time: {t_alloc:.0f}s", flush=True)
        a_opt, obj = lp_fw_optimize(parent, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_warmup")

    # === Phase 2: MD with Nesterov momentum ===
    print(f"\n=== Phase 2: MD Nesterov ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  Time: {t_alloc:.0f}s", flush=True)
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True,
                                                lr_init=0.003, decay=0.9996)
            try_seq(a_opt, "md_nesterov")

    # === Phase 3: Peak Surgery ===
    print(f"\n=== Phase 3: Peak Surgery ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 25:
            t_alloc = min(rem, 60)
            print(f"  Time: {t_alloc:.0f}s", flush=True)
            a_opt, obj = peak_surgery(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "surgery1")

    # === Phase 4: Deeper LP-FW with better starting point ===
    print(f"\n=== Phase 4: Deep LP-FW ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem, 150)
            print(f"  Time: {t_alloc:.0f}s", flush=True)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                        m_relax_max=12.0)
            try_seq(a_opt, "lpfw_deep")

    # === Phase 5: Structured Perturbation Cycle ===
    print(f"\n=== Phase 5: Perturbation Cycle ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 60:
            n_perts = min(4, int(rem / 50))
            for p in range(n_perts):
                if time.monotonic() > deadline - 50:
                    break
                t_alloc = min((deadline - time.monotonic() - 50) / (n_perts - p), 70)

                if p == 0:
                    # Fourier perturbation - low frequency
                    freq = rng.integers(1, max(2, n_best // 8))
                    amp = 0.005
                    phase = rng.uniform(0, 2 * np.pi)
                    pert = amp * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                    f_init = project_simplex(best_seq + pert)
                elif p == 1:
                    # Fourier perturbation - high frequency
                    freq = rng.integers(n_best // 8, max(n_best // 4 + 1, n_best // 6))
                    amp = 0.003
                    phase = rng.uniform(0, 2 * np.pi)
                    pert = amp * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                    f_init = project_simplex(best_seq + pert)
                elif p == 2:
                    # Asymmetric shift: more mass on one side
                    asym = np.zeros(n_best)
                    asym[:n_best // 3] = 0.008
                    asym[2 * n_best // 3:] = -0.004
                    f_init = project_simplex(best_seq + asym)
                else:
                    # Random coordinate swap
                    i = rng.integers(0, n_best // 2)
                    j = rng.integers(n_best // 2, n_best)
                    f_init = best_seq.copy()
                    delta = (f_init[i] - f_init[j]) * 0.5
                    f_init[i] -= delta
                    f_init[j] += delta
                    f_init = np.maximum(f_init, 0)
                    f_init = f_init / f_init.sum()

                print(f"  Pert {p}: t={t_alloc:.0f}s", flush=True)
                a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc)
                try_seq(a_opt, f"pert_{p}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration ===", flush=True)
    if best_seq is not None:
        for target_n in [n_parent + 128, n_parent - 128, n_parent + 256]:
            if target_n < 128:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.4, 50)
                print(f"  Optimize n={target_n}: t={t_alloc:.0f}s", flush=True)
                a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 7: Final MD polish ===
    print(f"\n=== Phase 7: Final MD Polish ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 50)
            print(f"  Time: {t_alloc:.0f}s", flush=True)
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True,
                                                lr_init=0.001, decay=0.9998)
            try_seq(a_opt, "md_final")

    # === Phase 8: Final peak surgery + LP-FW squeeze ===
    print(f"\n=== Phase 8: Final Squeeze ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 25:
            t1 = min(rem * 0.5, 20)
            t2 = min(rem * 0.5, 20)
            print(f"  Surgery: {t1:.0f}s, LP-FW: {t2:.0f}s", flush=True)
            a_opt, obj = peak_surgery(best_seq, time_limit=t1)
            try_seq(a_opt, "surgery_final")
            if best_seq is not None:
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t2)
                try_seq(a_opt, "lpfw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
