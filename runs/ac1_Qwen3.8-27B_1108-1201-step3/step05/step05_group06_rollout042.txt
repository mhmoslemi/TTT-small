def project_simplex(v):

    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n

    a = np.maximum(a0.astype(np.float64), 0.0)

                if no_improve > 80:
                    break
            continue
        g = g / g.sum()

            no_improve += 1
            if no_improve > 50:
                break
            continue

        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_t = a + alpha * (g - a)
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-15:

            val = fast_convolve(a_t).max()

                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            M = fast_convolve(a_new).max()
            m_mult = 1.0
            a = a_new

                best_a = a.copy()

def make_various_initializations(n, rng):
    """Generate diverse initializations for length n."""
    inits = []
    
    # Geometric
    for r in [0.97, 0.98, 0.99, 0.995]:
        a = r ** np.arange(n, dtype=np.float64)
        inits.append(a / a.sum())
    
    # Symmetric geometric
    for r in [0.96, 0.98, 0.99]:
        half = (n + 1) // 2
        asc = r ** np.arange(half, dtype=np.float64)
        desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
        a = np.concatenate([asc, desc])
        if len(a) < n:
            a = np.concatenate([a, np.ones(n - len(a))])
        inits.append(a / a.sum())
    
    # Cosine windows with different powers
    for p in [1.5, 2.0, 3.0, 5.0, 8.0]:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** p + 0.001
        inits.append(a / a.sum())
    
    # Scaled cosine (narrow peak)
    for width in [0.3, 0.5, 0.7, 1.0]:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 / (2 * width**2))
        inits.append(a / a.sum())
    
    # Random (log-uniform)
    for _ in range(3):
        a = 10.0 ** rng.uniform(-2, 0, n)
        inits.append(a / a.sum())
    
    # "Barker-like": alternating with decay
    for decay in [0.99, 0.995]:
        signs = np.ones(n)
        # Create a pattern that has low autocorrelation
        a = decay ** np.arange(n) * (0.5 + 0.5 * np.random.default_rng(42).random(n))
        inits.append(a / a.sum())
    
    return inits

def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact FFT-based subgradient, mirror descent, LSE-Adam, LP Frank-Wolfe,
and diverse initializations with adaptive length exploration.
Key innovations: exact subgradient via FFT cross-correlation, multiplicative
(mirror) updates for better simplex handling, and aggressive multi-phase
search with frequency-informed restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgrad_fft(a, rel_tol=0.01):
    """
    Compute exact subgradient of max(conv(a,a)) using FFT.
    Uses the 'max set' (positions within rel_tol of max) for a specific subgradient.
    Returns (gradient, conv, max_val).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    tol = max(rel_tol * max_val, max_val * 1e-10)
    S = np.where(conv >= max_val - tol)[0]

    w = np.zeros(2 * n - 1)
    w[S] = 1.0 / len(S)

    N = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)

    g1 = np.fft.irfft(np.conj(A_fft) * W_fft, N)[:n]
    g2 = np.fft.irfft(W_fft * A_fft, N)[:n]
    g = g1 + g2
    return g, conv, max_val


def smooth_grad_fft(a, T):
    """Compute gradient of log-sum-exp smoothing of max(conv) via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()

    N = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)

    g1 = np.fft.irfft(np.conj(A_fft) * W_fft, N)[:n]
    g2 = np.fft.irfft(W_fft * A_fft, N)[:n]
    g = g1 + g2
    return g, conv, max_val


def subgradient_optimize(a0, time_limit, verbose=False):
    """
    Projected subgradient method with exact subgradient and adaptive step size.
    Uses backtracking: decrease eta on no-improve, increase on sustained improvement.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    eta = 0.8
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    consecutive_decrease = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g, conv, cur_max = exact_subgrad_fft(a, rel_tol=cur_max * 0.01)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        a_new = a - eta * g / g_norm
        a_new = project_simplex(a_new)

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            consecutive_decrease = 0
            eta = min(eta * 1.03, 3.0)
            if verbose and it % 100 == 0:
                print(f"    SubGrad it {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            consecutive_decrease += 1
            if consecutive_decrease > 3:
                eta *= 0.7
                consecutive_decrease = 0

        if no_improve > 150:
            scale = 0.002 + 0.003 * min(no_improve // 150, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            eta = 0.5
            no_improve = 0

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, verbose=False):
    """
    Mirror descent (multiplicative update) for the simplex-constrained problem.
    Uses exponential family natural gradient, which handles non-negativity naturally.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    eta = 0.15
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, conv, cur_max = exact_subgrad_fft(a, rel_tol=cur_max * 0.02)

        log_update = -eta * g
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.01, 1.5)
            if verbose and it % 200 == 0:
                print(f"    Mirror it {it}: obj={best_obj:.10f}, eta={eta:.4f}")
        else:
            no_improve += 1
            eta *= 0.85

        if no_improve > 250:
            scale = 0.003 + 0.002 * min(no_improve // 250, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            eta = 0.2
            no_improve = 0

    return best_a, best_obj


def lse_adam_optimize(a0, time_limit, T_start=0.08, T_end=0.001, verbose=False):
    """
    Log-sum-exp smooth optimization with Adam using FFT-based gradients.
    Uses a more aggressive T schedule (down to 0.001) for sharper gradients.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    total_time = time_limit
    no_improve = 0
    it = 0
    rng = np.random.default_rng(77777)

    while time.monotonic() - t0 < total_time:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_time, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.6)
        T = max(T, T_end)
        alpha = 0.006 * (1.0 - 0.4 * progress)

        g, conv, _ = smooth_grad_fft(a, T)

        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * g
        v_state = beta2 * v_state + (1 - beta2) * g ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)

        a = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)

        cur_obj = fast_convolve(a).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    LSE-Adam it {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1

        if no_improve > 1500:
            rng_p = np.random.default_rng(int(elapsed * 1000) % 1000000)
            a = project_simplex(best_a + rng_p.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation and perturbation restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_mult = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_mult)
        bounds = [(0, None)] * n

        lp_time = max(0.5, remaining * 0.2)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': lp_time})
            g = res.x if res.success else None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            m_mult = min(m_mult * 1.1, 5.0)
            if m_mult >= 5.0:
                no_improve += 1
                if no_improve > 60:
                    break
            continue
        g = g / g.sum()

        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_t = a + alpha * (g - a)
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-15:
                continue
            a_t = a_t / st
            val = fast_convolve(a_t).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a_t = a + mid * (g - a)
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    break
                a_t = a_t / st
                val = fast_convolve(a_t).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            M = fast_convolve(a_new).max()
            m_mult = 1.0
            a = a_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW it {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_mult = min(m_mult * 1.05, 4.0)

        if no_improve > 80:
            rng_p = np.random.default_rng(it * 13 + 3)
            a = project_simplex(best_a + rng_p.standard_normal(n) * 0.004)
            M = fast_convolve(a).max()
            m_mult = 1.0
            no_improve = 0

    return best_a, best_obj


def make_diverse_initializations(n, rng):
    """Generate a variety of structured initializations."""
    inits = []

    for r in [0.97, 0.98, 0.99, 0.995]:
        a = r ** np.arange(n, dtype=np.float64)
        inits.append(a / a.sum())

    for r in [0.96, 0.98, 0.99]:
        half = (n + 1) // 2
        asc = r ** np.arange(half, dtype=np.float64)
        desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
        a = np.concatenate([asc, desc])
        if len(a) < n:
            a = np.concatenate([a, np.ones(n - len(a))])
        inits.append(a / a.sum())

    for p in [1.5, 2.0, 3.0, 5.0, 8.0]:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** p + 0.001
        inits.append(a / a.sum())

    for width in [0.3, 0.5, 0.7, 1.0, 1.5]:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 / (2 * width ** 2))
        inits.append(a / a.sum())

    for _ in range(3):
        a = 10.0 ** rng.uniform(-2, 0, n)
        inits.append(a / a.sum())

    # Frequency-informed: try to make |FFT(a)|^2 flat
    for _ in range(3):
        a = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(a)

    return inits


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer combining exact subgradient, mirror descent,
    LSE-Adam, and LP Frank-Wolfe with diverse initializations and
    adaptive length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")

    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")
    for n_try in [n_parent, max(256, n_parent // 2), min(1200, int(n_parent * 1.5))]:
        inits = make_diverse_initializations(n_try, rng)
        for i, init in enumerate(inits):
            _try(init, f"init_n{n_try}_{i}")

    print(f"  After Phase 0: best={best_score:.6f}")

    # === Phase 1: LSE-Adam on best ===
    print(f"\n=== Phase 1: LSE-Adam (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 20:
        t_alloc = min(rem, 200)
        print(f"  LSE-Adam n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lse_adam_optimize(best_seq, time_limit=t_alloc,
                                     T_start=0.08, T_end=0.001, verbose=True)
        _try(a_opt, "lse_adam_p1")

    # === Phase 2: Exact subgradient ===
    print(f"\n=== Phase 2: Exact Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 550
    if rem > 15:
        t_alloc = min(rem, 120)
        print(f"  SubGrad n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "subgrad_p2")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 450
    if rem > 15:
        t_alloc = min(rem, 100)
        print(f"  Mirror n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "mirror_p3")

    # === Phase 4: LP-FW ===
    print(f"\n=== Phase 4: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 15:
        t_alloc = min(rem, 130)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p4")

    # === Phase 5: Alternating refinement ===
    print(f"\n=== Phase 5: Alternating Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    cycle = 0
    while rem > 25:
        cycle += 1
        if cycle % 4 == 1:
            t_alloc = min(rem * 0.25, 50)
            a_opt, _ = lse_adam_optimize(best_seq, time_limit=t_alloc,
                                         T_start=0.03, T_end=0.002)
            _try(a_opt, f"refine_lse_{cycle}")
        elif cycle % 4 == 2:
            t_alloc = min(rem * 0.25, 45)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc)
            _try(a_opt, f"refine_subg_{cycle}")
        elif cycle % 4 == 3:
            t_alloc = min(rem * 0.25, 45)
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc)
            _try(a_opt, f"refine_mirr_{cycle}")
        else:
            t_alloc = min(rem * 0.25, 50)
            a_opt, _ = lp_frank_wolfe(best_seq, time_limit=t_alloc)
            _try(a_opt, f"refine_lpfw_{cycle}")
        rem = deadline - time.monotonic() - 180

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.6f}) ===")
    for target_n in [n_parent + 32, n_parent + 64, n_parent + 128, n_parent - 32]:
        if target_n < 64 or target_n > 2500:
            continue
        rem = deadline - time.monotonic() - 100
        if rem < 25:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        t_alloc = min(rem * 0.35, 45)
        a_opt, _ = lse_adam_optimize(f_interp, time_limit=t_alloc,
                                     T_start=0.04, T_end=0.004)
        _try(a_opt, f"len{target_n}_lse")

        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_alloc = min(rem * 0.5, 35)
            a_opt, _ = lp_frank_wolfe(f_interp, time_limit=t_alloc)
            _try(a_opt, f"len{target_n}_lpfw")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final Fine-tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = lse_adam_optimize(best_seq, time_limit=min(rem, 35),
                                         T_start=0.008, T_end=0.0005)
            _try(a_opt, "final_lse")

        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 20))
            _try(a_opt, "final_subg")

        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 12))
            _try(a_opt, "final_mirr")

        # Small perturbations with quick optimization
        for _ in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 6:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lse_adam_optimize(f_pert, time_limit=min(rem, 10),
                                         T_start=0.005, T_end=0.0005)
            _try(a_opt, "final_perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
