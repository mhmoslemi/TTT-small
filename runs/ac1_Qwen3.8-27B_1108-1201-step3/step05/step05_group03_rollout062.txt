Minimizes 2*n*max(auto_conv)/(sum)^2 using FFT-based evaluation, 
softmax-weighted subgradient with averaging, multi-candidate coordinate descent,
simulated annealing, and multi-length exploration.
Key improvements: asymmetric power-law initializations, wider coordinate search,
adaptive subgradient weights, and multi-n exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return np.array([a[0] * a[0]])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def power_law_seq(n, alpha):
    """Power law decay: p_i proportional to (i+1)^(-alpha)."""
    i = np.arange(1, n + 1)
    a = i ** (-alpha)
    return normalize(a)


def exp_correction_seq(n, beta, gamma):
    """Exponential with quadratic correction."""
    x = np.arange(n) / n
    a = np.exp(-beta * x + gamma * x * x)
    return normalize(a)


def sorted_random_seq(n, seed):
    """Random sequence sorted in decreasing order."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.sort(a)[::-1]
    return normalize(a)

        step += 1
        conv = fast_convolve(a)

        max_conv = conv.max()
        # Use temperature based on the spread
        temp = max(1e-6, max_conv * 0.1)
        scaled = conv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            wk = w[k]
            if wk < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * wk * a[k - idx]
        
        # Multiplicative update
        a = np.maximum(a * np.exp(-cur_lr * sg), 1e-15)
        a = a / a.sum()
        
        # Average
        avg_a += a
        avg_count += 1
        
        conv_new = fast_convolve(a)
        sc = 2.0 * n * conv_new.max()
        sc_avg = 2.0 * n * fast_convolve(avg_a / avg_count).max()
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        if sc_avg < best_score - 1e-12:
            best_score = sc_avg
            best_a = (avg_a / avg_count).copy()
        
        cur_lr *= decay
        
        if verbose and step % 5000 == 0:
            print(f"      SG step {step}: best={best_score:.10f}, lr={cur_lr:.6e}")
        
        # Adaptive restart
        if step % 10000 == 0 and cur_lr < lr0 * 0.05:
            cur_lr = lr0 * 0.3
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.001)
            avg_a = a.copy()
            avg_count = 1
    
    return best_a, best_score
