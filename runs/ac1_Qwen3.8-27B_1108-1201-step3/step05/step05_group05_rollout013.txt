import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def smoothed_objective_and_grad(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    # Log-sum-exp for numerical stability
    conv_max = conv.max()
    shifted = tau * (conv - conv_max)
    log_sum_exp = conv_max + np.log(np.sum(np.exp(shifted))) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
    # This is 2 * cross_correlation(weights, a)[:n]
    # Using FFT
    fft_size = 1 << (len(weights) + n - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: ifft(conj(W) * A)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * corr[:n]
    
    return log_sum_exp, grad

def softmax_objective(b, tau):
    """Objective in softmax parameterization."""
    # Compute a from b
    b_max = b.max()
    exp_b = np.exp(b - b_max)
    a = exp_b / exp_b.sum()
    n = len(a)
    obj, grad_a = smoothed_objective_and_grad(a, tau)
    # Chain rule: df/db[j] = a[j] * (df/da[j] - sum_i a[i]*df/da[i])
    mean_grad = np.sum(a * grad_a)
    grad_b = a * (grad_a - mean_grad)
    return 2.0 * n * obj, grad_b

    a = a.copy()

    for i in range(n_iter):
        alpha = alpha_start + (alpha_end - alpha_start) * i / max(n_iter - 1, 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping initialization (iterative magnitude flattening)
2. Projected gradient with full log-sum-exp smoothing + backtracking + Nesterov momentum
3. L-BFGS-B in softmax parameterization for fast local convergence
4. Smart coordinate descent with gradient-guided coordinate selection
5. Multi-length exploration (n=640..1280)
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smooth_obj_and_grad(a, tau):
    """Compute log-sum-exp smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = tau * (conv - conv_max)
    # Numerically stable log-sum-exp
    max_shifted = shifted.max()
    log_sum_exp = conv_max + (max_shifted + np.log(np.exp(shifted - max_shifted).sum())) / tau
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()

    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (valid k-j in [0,n-1])
    # = 2 * cross_correlation(weights, a)[:n]
    fft_size = 1 << (len(weights) + n - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * corr[:n]

    return log_sum_exp, grad


def softmax_obj_grad(b, tau):
    """Objective and gradient in softmax parameterization b -> a=softmax(b)."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum()
    n = len(a)
    obj_val, grad_a = smooth_obj_and_grad(a, tau)
    mean_grad = np.dot(a, grad_a)
    grad_b = a * (grad_a - mean_grad)
    return 2.0 * n * obj_val, grad_b


def spectral_shaping(a, n_iter=100, alpha=0.7):
    """Iteratively flatten the power spectrum to reduce auto-convolution peak."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        phase = np.angle(A)
        mag = np.abs(A)
        # Flatten: push magnitudes toward the median (excluding DC)
        if len(mag) > 2:
            target = np.median(mag[1:])
            mag_new = mag.copy()
            mag_new[1:] = (mag[1:] + alpha * target) / (1 + alpha)
            mag_new[0] = mag[0]  # Preserve DC (sum)
            # Also n/K bin if present
            if n % 2 == 0 and len(mag_new) > n // 2:
                pass  # Nyquist kept as-is
        else:
            mag_new = mag
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def projected_gradient_opt(a_init, time_limit, tau_start=200.0, tau_end=50.0,
                           verbose=False):
    """Nesterov accelerated projected gradient with backtracking on simplex."""
    a = project_simplex(a_init.copy(), 1.0)
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive tau
        frac = min(1.0, it / max(200, time_limit * 2))
        tau = tau_start * (tau_end / tau_start) ** frac

        # Nesterov lookahead
        a_look = project_simplex(a + momentum * v, 1.0)
        _, grad = smooth_obj_and_grad(a_look, tau)

        # Update velocity
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v, 1.0)

        # Backtracking line search
        obj_cur = fast_convolve(a).max()
        obj_new = fast_convolve(a_new).max()
        step = 1.0
        while obj_new > obj_cur - 1e-14 and step > 1e-6:
            step *= 0.5
            a_trial = project_simplex(a + step * v, 1.0)
            obj_new = fast_convolve(a_trial).max()
            if obj_new < obj_cur - 1e-14:
                a_new = a_trial
                break
        a_new = project_simplex(a + step * v, 1.0)
        a = a_new

        obj_a = fast_convolve(a).max()
        if obj_a < best_obj - 1e-14:
            best_obj = obj_a
            best_a = a.copy()
            no_improve = 0
            if obj_a < best_obj * 1.001:
                lr = min(0.05, lr * 1.02)
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.95
            if no_improve > 200:
                momentum = min(0.95, momentum + 0.005)
            if no_improve > 800:
                # Restart with perturbation
                rng_r = np.random.default_rng(it)
                a = project_simplex(best_a + rng_r.standard_normal(n) * 0.003, 1.0)
                v = np.zeros(n)
                no_improve = 0
                lr = 0.02
                momentum = 0.9

        if verbose and it % 100 == 0:
            print(f"    PG iter {it}: obj={best_obj:.10f}, lr={lr:.4f}, tau={tau:.0f}, no_imp={no_improve}")

    return best_a, best_obj


def lbfgs_softmax_opt(a_init, time_limit, tau=30.0, verbose=False):
    """L-BFGS-B optimization in softmax parameterization."""
    a = project_simplex(a_init.copy(), 1.0)
    n = len(a)
    # Initialize b from a (with small floor to avoid log(0))
    a_floored = np.maximum(a, 1e-12)
    a_floored = a_floored / a_floored.sum()
    b0 = np.log(a_floored)
    # Fix gauge: set b[0] = 0
    b0 = b0 - b0[0]

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    def func_wrapped(b):
        val, _ = softmax_obj_grad(b, tau)
        return val

    def grad_wrapped(b):
        _, g = softmax_obj_grad(b, tau)
        return g

    try:
        res = minimize(func_wrapped, b0, jac=grad_wrapped, method='L-BFGS-B',
                       options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8,
                                'maxfun': 1000,
                                'callback': None})
        # Convert back
        b_opt = res.x - res.x[0]
        exp_b = np.exp(b_opt - b_opt.max())
        a_opt = exp_b / exp_b.sum()
        obj = fast_convolve(a_opt).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    L-BFGS: obj={best_obj:.10f} (iters={res.nit})")
    except Exception as e:
        if verbose:
            print(f"    L-BFGS failed: {e}")

    return best_a, best_obj


def smart_coordinate_descent(a_init, time_limit, verbose=False):
    """Coordinate descent with gradient-guided coordinate selection and local search."""
    a = project_simplex(a_init.copy(), 1.0)
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Get gradient to prioritize coordinates
        _, grad = smooth_obj_and_grad(a, tau=50.0)
        # Sort by gradient magnitude (most impactful first)
        order = np.argsort(-np.abs(grad))
        # Also include random coordinates
        rng_p = np.random.default_rng(pass_count)
        random_idx = rng_p.integers(0, n, size=max(10, n // 20))
        candidates = np.concatenate([order[:min(n, 300)], random_idx])

        for j in candidates:
            j = int(j)
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-10:
                # Try adding a small amount
                for frac in [0.001, 0.005, 0.01]:
                    new_val = frac
                    remaining = 1.0
                    scale = (1.0 - new_val) / remaining if remaining > 1e-10 else 1.0
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= scale
                    a_trial = np.maximum(a_trial, 0)
                    obj = fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                continue

            # Try several multipliers
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                remaining = 1.0 - old_val
                if remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                remaining = 1.0 - old_val
                if remaining > 1e-10:
                    scale = (1.0 - best_j_val) / remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-15:
                        a /= s
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 1:
            break

    return best_a, best_obj


def pair_swap_search(a_init, time_limit, seed=0):
    """Randomized pair transfer local search."""
    a = project_simplex(a_init.copy(), 1.0)
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    support = np.where(a > 1e-6)[0]
    if len(support) == 0:
        support = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j or a[i] < 1e-10 or a[j] < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining spectral shaping, projected gradient
    with log-sum-exp smoothing, L-BFGS-B in softmax space, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Initialization and spectral shaping ===
    print("\n=== Phase 1: Initialization & Spectral Shaping ===")
    candidates = {}

    if parent is not None:
        n_parent = len(parent)
        # Apply spectral shaping to parent
        shaped = spectral_shaping(parent, n_iter=200, alpha=0.5)
        candidates[f'shaped_n{n_parent}'] = (n_parent, shaped)
        candidates[f'parent_n{n_parent}'] = (n_parent, parent.copy())

    # Generate candidates at various lengths
    target_ns = [512, 640, 768, 896, 1024]
    if parent is not None:
        target_ns.insert(0, len(parent))
    # Remove duplicates and sort
    seen = set()
    unique_ns = []
    for n_try in target_ns:
        if n_try not in seen and n_try <= 1280:
            seen.add(n_try)
            unique_ns.append(n_try)

    for n_try in unique_ns:
        if n_try in [k for k, (_, v) in candidates.items()]:
            continue
        # Multiple initializations
        inits = {
            'flat': flat_spectrum_init(n_try, rng),
            'gauss': None,
            'tri': None,
        }
        # Gaussian
        x = np.linspace(-3, 3, n_try)
        g = np.exp(-x**2 / 2)
        inits['gauss'] = g / g.sum()
        # Triangular
        x = np.arange(n_try) / max(n_try - 1, 1)
        t = 1.0 - np.abs(2.0 * x - 1.0)
        inits['tri'] = t / t.sum()

        # Pick best initial
        best_init = None
        best_init_obj = float('inf')
        for name, init in inits.items():
            init = project_simplex(init, 1.0)
            obj = fast_convolve(init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init

        if best_init is not None:
            # Shape it
            shaped = spectral_shaping(best_init, n_iter=100, alpha=0.5)
            candidates[f'shaped_n{n_try}'] = (n_try, shaped)

    # Evaluate all candidates
    for name, (n_c, a_c) in list(candidates.items()):
        score = try_seq(a_c, f"init_{name}")

    # Sort by objective
    def get_key(item):
        return fast_convolve(item[1]).max()
    sorted_cands = sorted(candidates.items(), key=lambda x: get_key(x))

    # === Phase 2: Projected gradient on top candidates ===
    print(f"\n=== Phase 2: Projected Gradient (best={best_score:.6f}) ===")
    n_optimize = min(3, len(sorted_cands))
    for rank, (name, (n_c, a_c)) in enumerate(sorted_cands[:n_optimize]):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        time_frac = [0.5, 0.3, 0.2][rank] if rank < 3 else 0.1
        t_alloc = min(rem * time_frac, 300)
        print(f"  PG optimize {name} (n={n_c}), t={t_alloc:.0f}s")
        a_opt, obj = projected_gradient_opt(a_c, time_limit=t_alloc,
                                            tau_start=300.0, tau_end=30.0,
                                            verbose=(rank == 0))
        try_seq(a_opt, f"pg_{name}")

    # === Phase 3: L-BFGS-B in softmax space ===
    print(f"\n=== Phase 3: L-BFGS-B Softmax (best={best_score:.6f}) ===")
    if best_seq is not None:
        for tau_val in [50.0, 20.0, 10.0]:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 60)
            print(f"  L-BFGS tau={tau_val}, t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_softmax_opt(best_seq, time_limit=t_alloc, tau=tau_val, verbose=True)
            try_seq(a_opt, f"lbfgs_tau{tau_val}")

    # === Phase 4: Spectral shaping + re-optimization ===
    print(f"\n=== Phase 4: Spectral Re-shaping (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alpha_val in [0.3, 0.5, 0.7]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            shaped = spectral_shaping(best_seq, n_iter=100, alpha=alpha_val)
            t_pg = min(rem * 0.3, 60)
            a_opt, obj = projected_gradient_opt(shaped, time_limit=t_pg,
                                                tau_start=200.0, tau_end=30.0)
            try_seq(a_opt, f"reshape_a{alpha_val}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = smart_coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: Pair swap ===
    print(f"\n=== Phase 6: Pair Swap (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_ps = min(rem * 0.4, 60)
            print(f"  PS t={t_ps:.0f}s")
            a_opt, obj = pair_swap_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "pair_swap")

    # === Phase 7: Try longer sequences ===
    print(f"\n=== Phase 7: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best + 128, n_best + 256]:
            if target_n > 1536 or target_n <= n_best:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                continue
            a_interp = interpolate_sequence(best_seq, target_n)
            shaped = spectral_shaping(a_interp, n_iter=50, alpha=0.5)
            t_alloc = min(rem * 0.4, 80)
            print(f"  Longer n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = projected_gradient_opt(shaped, time_limit=t_alloc,
                                                tau_start=200.0, tau_end=30.0)
            try_seq(a_opt, f"long_n{target_n}")

            rem = deadline - time.monotonic() - 25
            if rem > 15:
                a_opt2, obj2 = smart_coordinate_descent(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt2, f"cd_long_n{target_n}")

    # === Phase 8: Final polish with perturbations ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, 1.0)
            elif p == 2:
                f_init = project_simplex(best_seq + best_seq[::-1] * 0.1, 1.0)
            elif p == 3:
                f_init = project_simplex(best_seq * 0.9 + np.ones(n_best) * 0.1 / n_best, 1.0)
            elif p == 4:
                # Flip a random subset
                f_init = best_seq.copy()
                flip_idx = rng.integers(0, n_best, size=n_best // 4)
                f_init[flip_idx] *= 0.5
                f_init = project_simplex(f_init, 1.0)
            else:
                # Sharpen
                f_init = best_seq ** 2
                f_init = project_simplex(f_init, 1.0)

            t_alloc = min(rem * 0.3, 20)
            a_opt, obj = projected_gradient_opt(f_init, time_limit=t_alloc,
                                                tau_start=100.0, tau_end=20.0)
            try_seq(a_opt, f"polish_{p}")

            # Quick CD after each
            rem = deadline - time.monotonic() - 10
            if rem > 5:
                a_opt2, obj2 = smart_coordinate_descent(best_seq, time_limit=min(rem, 10))
                try_seq(a_opt2, f"cd_polish_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
