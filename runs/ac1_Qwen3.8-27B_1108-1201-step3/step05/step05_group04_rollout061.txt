"""
LP-based and spectral optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal descent directions, combined with
Adam polishing, coordinate descent, spectral shaping, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_A(f):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def build_A_sparse(f, top_k):
    """Build sparse constraint matrix using only top_k most active constraints."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    # Select top_k positions where conv is largest
    if top_k < 2 * n - 1:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        idx = np.arange(2 * n - 1)
    A_full = build_A(f)
    return A_full[idx, :], idx


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w):
    """Compute subgradient: result[j] = sum_i w[i+j] * f[i]."""
    n = len(f)
    return np.correlate(w, f, 'valid')


def get_max_weights(conv, temp=None, top_k=None):
    """Get weights concentrated on the maximum positions."""
    if temp is None:
        # Hard max: uniform over argmax positions
        M = conv.max()
        w = (conv >= M - 1e-10).astype(float)
        ws = w.sum()
        if ws > 0:
            w /= ws
        return w
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w

        obj = 2.0 * n * fast_convolve(f_new).max()

def adam_optimize(f0, time_limit, lr=0.005, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = get_max_weights(conv, temp=temp, top_k=top_k)
        sg = compute_subgradient(f, w)
        
        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)
        
        cur_obj = 2.0 * n * fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.5, 0.002)
            cur_lr = lr
            no_improve = 0
    
    return best_f, best_obj / (2.0 * n)


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with per-coordinate optimization."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            for mult in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj / (2.0 * n) - 1e-14:
                    best_obj = 2.0 * n * best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj / (2.0 * n)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

"""
LP-based water-filling optimization combined with Adam, spectral shaping,
and coordinate descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal descent directions, parametric
families, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_A(f):
    """Build constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w):
    """Compute subgradient: result[j] = sum_i w[i+j] * f[i]."""
    n = len(f)
    return np.correlate(w, f, 'valid')


def get_max_weights(conv, temp=None, top_k=None):
    """Get weights concentrated on max positions."""
    if temp is None:

        w = (conv >= M - 1e-10).astype(float)
        ws = w.sum()
        if ws > 0:
            w /= ws
        return w
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w


def lp_optimize(f0, time_limit, full_LP=True, sparse_k=None, verbose=False):
    """LP-based optimization using water-filling directions."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t_step = 0.5
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if full_LP:
            A = build_A(f)
            b_ub = np.full(2 * n - 1, M)
        else:
            k = sparse_k if sparse_k else max(20, n // 5)
            idx = np.argpartition(conv, -k)[-k:]
            A_full = build_A(f)
            A = A_full[idx, :]
            b_ub = np.full(len(idx), M)

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                         method='highs', options={'presolve': True, 'time_limit': max(1.0, time_limit * 0.01)})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        b = res.x.copy()
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        # Line search with multiple step sizes
        best_t = 0.0
        best_t_obj = best_obj
        candidates = [t_step, t_step * 2.0, t_step * 0.5, t_step * 0.25,
                      t_step * 0.1, 0.05, 0.01, 0.8, 0.3]
        candidates = [t for t in candidates if 0 < t <= 1.0]
        for t in candidates:
            f_trial = (1 - t) * f + t * b
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            obj = 2.0 * n * fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-14:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            t_step = max(t_step * 0.7, 0.02)
            if no_improve > 25:
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                no_improve = 0
                t_step = 0.3
                continue
            continue

        f = (1 - best_t) * f + best_t * b
        f = np.maximum(f, 0)
        f = f / f.sum()

        cur_obj = 2.0 * n * fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            t_step = min(t_step * 1.05, 1.0)
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: score={best_obj / (2 * n):.10f}, t={best_t:.4f}")
        else:
            no_improve += 1
            t_step = max(t_step * 0.9, 0.05)

    return best_f, best_obj / (2.0 * n)


def spectral_optimize(f0, time_limit, alpha_init=0.3, verbose=False):
    """Spectral shaping optimization via power spectrum flattening."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_alpha = alpha_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        fft_size = 1 << (2 * n - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        P = np.abs(F) ** 2

        P_uniform = P.mean()
        P_new = (1 - cur_alpha) * P + cur_alpha * P_uniform
        P_new = np.maximum(P_new, 1e-30)

        phase = np.angle(F)
        F_new = np.sqrt(P_new) * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_size)[:n]

        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            no_improve += 1
            cur_alpha = max(cur_alpha * 0.5, 0.01)
            if no_improve > 50:
                break
            continue
        f_new = f_new / ss

        obj = 2.0 * n * fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            cur_alpha = min(cur_alpha * 0.97, 0.6)
        else:
            no_improve += 1
            cur_alpha = max(cur_alpha * 0.9, 0.01)
            if no_improve > 40:
                cur_alpha = min(cur_alpha * 2.5, 0.8)
                no_improve = 0

    return best_f, best_obj / (2.0 * n)


def adam_optimize(f0, time_limit, lr=0.005, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = get_max_weights(conv, temp=temp, top_k=top_k)
        sg = compute_subgradient(f, w)

        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        cur_obj = 2.0 * n * fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 250:
            rng = np.random.default_rng(t * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.5, 0.002)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj / (2.0 * n)


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with per-coordinate optimization."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj / (2.0 * n) - 1e-14:
                    best_obj = 2.0 * n * best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj / (2.0 * n)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Create a Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Create a symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Create a smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def parametric_search(f0, time_limit):
    """Try parametric families and return the best one."""
    n = len(f0)
    x = np.linspace(0, 1, n)
    best_f = f0.copy()
    best_obj = 2.0 * n * fast_convolve(f0).max()
    t0 = time.monotonic()

    families = []

    # Cosine bump: f = 1 + eps*cos(pi*x)
    for eps in np.linspace(-0.3, 0.3, 30):
        f = 1.0 + eps * np.cos(np.pi * x)
        f = np.maximum(f, 0)
        ss = f.sum()
        if ss > 1e-10:
            f /= ss
            obj = 2.0 * n * fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()

    # Power bump: f = (1 - |2x-1|)^p
    for p in np.linspace(0.3, 4.0, 40):
        f = (1.0 - np.abs(2 * x - 1)) ** p
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()

    # Two-parameter: a*cos(pi*x) + b*cos(2*pi*x)
    for a in np.linspace(-0.2, 0.3, 15):
        for b in np.linspace(-0.15, 0.2, 15):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj / (2.0 * n)
            f = 1.0 + a * np.cos(np.pi * x) + b * np.cos(2 * np.pi * x)
            f = np.maximum(f, 0)
            ss = f.sum()
            if ss > 1e-10:
                f /= ss
                obj = 2.0 * n * fast_convolve(f).max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()

    return best_f, best_obj / (2.0 * n)


def lp_optimize_symmetric(f0, time_limit, verbose=False):
    """LP optimization exploiting symmetry: f[j] = f[n-1-j]."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    nh = n // 2 + 1  # half length (including center)

    # Work with half sequence
    fh = f[:nh].copy()
    if n % 2 == 0:
        # Even: f[j] = f[n-1-j], so first n/2 determine the rest
        fh = f[:nh - 1].copy()  # n/2 elements

    def rebuild(hh):
        if n % 2 == 0:
            half = len(hh)
            out = np.zeros(n)
            out[:half] = hh
            out[half:] = hh[::-1]
        else:
            half = len(hh)
            out = np.zeros(n)
            out[:half] = hh
            out[half:] = hh[-2::-1]
        return out

    f_cur = rebuild(fh)
    f_cur = np.maximum(f_cur, 0)
    f_cur = f_cur / f_cur.sum()

    best_obj = 2.0 * n * fast_convolve(f_cur).max()
    best_fh = fh.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f_cur)
        M = conv.max()

        # Build constraint matrix for full problem but solve in half-space
        # Actually, let's just solve the full LP and symmetrize the result
        A = build_A(f_cur)
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                         method='highs', options={'presolve': True})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        b = res.x.copy()
        # Symmetrize
        b_sym = 0.5 * (b + b[::-1])
        b_sym = np.maximum(b_sym, 0)
        bs = b_sym.sum()
        if bs < 1e-15:
            break
        b_sym = b_sym / bs

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.5, 1.0, 0.25, 0.75, 0.1, 0.8, 0.3]:
            f_trial = (1 - t) * f_cur + t * b_sym
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial = f_trial / ss
            obj = 2.0 * n * fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-14:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        f_cur = (1 - best_t) * f_cur + best_t * b_sym
        f_cur = np.maximum(f_cur, 0)
        f_cur = f_cur / f_cur.sum()

        cur_obj = 2.0 * n * fast_convolve(f_cur).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_fh = f_cur[:nh].copy()
            no_improve = 0
        else:
            no_improve += 1

    best_f = rebuild(best_fh) if n % 2 == 0 else rebuild(best_fh)
    # Actually just return the full sequence
    f_final = f_cur.copy()
    # Use the best we found
    if best_obj < 2.0 * n * fast_convolve(f_cur).max():
        # Reconstruct from best_fh
        if n % 2 == 0:
            half = len(best_fh)
            f_final = np.zeros(n)
            f_final[:half] = best_fh
            f_final[half:] = best_fh[::-1]
        else:
            half = len(best_fh)
            f_final = np.zeros(n)
            f_final[:half] = best_fh
            f_final[half:] = best_fh[-2::-1]
        f_final = np.maximum(f_final, 0)
        ss = f_final.sum()
        if ss > 1e-15:
            f_final /= ss

    return f_final, best_obj / (2.0 * n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP water-filling, spectral shaping, Adam,
    coordinate descent, parametric search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_symmetric_init(768, 'bump')
        print("No parent found, using bump init n=768")

    n_parent = len(parent)

    # === Phase 0: Quick parametric search ===
    print(f"\n=== Phase 0: Parametric search (n={n_parent}) ===")
    t0 = time.monotonic()
    a_param, _ = parametric_search(parent, time_limit=60)
    try_seq(a_param, "p0_parametric")
    t_elapsed = time.monotonic() - t0
    print(f"  Parametric done in {t_elapsed:.1f}s")

    base = best_seq if best_seq is not None else parent

    # === Phase 1: LP-based optimization at parent length ===
    print(f"\n=== Phase 1: LP optimization (n={len(base)}, best={best_score:.8f}) ===")
    t_phase1_end = time.monotonic() + 350

    lp_starts = [
        ('parent', base.copy()),
        ('gauss_02', make_gaussian_init(len(base), 0.5, 0.2)),
        ('gauss_025', make_gaussian_init(len(base), 0.5, 0.25)),
        ('bump', make_symmetric_init(len(base), 'bump')),
    ]

    for si, (sname, sinit) in enumerate(lp_starts):
        rem = t_phase1_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.55, 90)

        a_opt, _ = lp_optimize(sinit, time_limit=t_alloc, full_LP=True, verbose=False)
        try_seq(a_opt, f"p1_lp_{sname}")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Contin
