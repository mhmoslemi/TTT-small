import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_pnrm_obj_grad(a, p):
    """Compute p-norm of auto-convolution and its gradient.
    
    F(a) = (sum_k conv[k]^p)^(1/p)
    grad[i] = dF/da[i] = S^(1/p-1) * (T[i] + T[2n-2-i])
    where S = sum(conv^p), T[i] = sum_j a[j] * conv[i+j]^(p-1)
    """
    n = len(a)
    conv = fast_convolve(a)  # length 2n-1
    
    S = np.sum(conv ** p)
    F = S ** (1.0 / p)
    coeff = S ** (1.0 / p - 1.0)
    
    conv_p = conv ** (p - 1)
    
    # T[i] = sum_j a[j] * conv_p[j+i] for i = 0, ..., 2n-2
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    Cp = np.fft.rfft(conv_p, N)
    T = np.fft.irfft(np.conj(A) * Cp, N)[:2 * n - 1]
    
    # grad[i] = coeff * (T[i] + T[2n-2-i]) for i = 0, ..., n-1
    T_rev = T[n - 1:2 * n - 1][::-1]
    grad = coeff * (T[:n] + T_rev)
    
    return F, grad


def compute_lse_obj_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    
    return obj, grad_a


def lse_obj_grad_t(t, beta):
    """LSE objective and gradient in log-parameterization (for L-BFGS)."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t


def pnrm_obj_grad_t(t, p):
    """p-norm objective and gradient in log-parameterization (for L-BFGS)."""
    n = len(t)
    a = softmax_fn(t)
    F, grad_a = compute_pnrm_obj_grad(a, p)
    
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return F, grad_t


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def adam_optimize(f0, time_limit, verbose=False):
    """Adam optimizer with simplex projection for minimizing max conv."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.01
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    
    t0 = time.monotonic()
    no_improve = 0
    beta_ls = 50.0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient using LSE
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R
        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            t += 1
            if verbose and t % 500 == 0:
                print(f"    Adam step {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new  # Still move, but track best
            t += 1
        
        # Decay learning rate and anneal beta
        lr = max(0.001, lr * 0.9995)
        beta_ls = min(beta_ls * 1.002, 200.0)
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 1
            lr = 0.005
            no_improve = 0
    
    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing on the simplex for exploring the landscape."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    temperature = 0.0001
    cool_rate = 0.9998
    it = 0
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Random perturbation: pick a few coordinates and adjust
        if it % 10 == 0:
            k = min(n, rng.integers(5, 50))
            idx = rng.choice(n, size=k, replace=False)
            delta = rng.standard_normal(k) * 0.01
            f_pert = f.copy()
            f_pert[idx] += delta
            f_pert = project_simplex(f_pert)
        else:
            # Single coordinate perturbation
            j = rng.integers(n)
            f_pert = f.copy()
            f_pert[j] += rng.standard_normal() * 0.005
            f_pert = project_simplex(f_pert)
        
        obj_pert = fast_convolve(f_pert).max()
        delta_obj = obj_pert - cur_obj
        
        # Accept or reject
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(temperature, 1e-10)):
            f = f_pert
            cur_obj = obj_pert
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
        
        temperature = max(temperature * cool_rate, 1e-8)
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            for mult in [0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f, kind='cubic' if old_n >= 4 else 'linear')
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_structured_init(n, kind='cosine'):
    """Generate a structured initialization of length n."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        f = np.ones(n)
    elif kind == 'cosine':
        f = np.cos(np.pi * x) ** 2
    elif kind == 'gaussian':
        f = np.exp(-2.0 * (x - 0.5) ** 2)
    elif kind == 'triangular':
        f = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == 'power':
        f = x ** 0.5 * (1 - x) ** 0.5
    elif kind == 'bell':
        f = np.exp(-4.0 * (x - 0.5) ** 2) * 0.5 + 0.5 * np.exp(-16.0 * (x - 0.5) ** 2)
    else:
        f = np.ones(n)
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f

        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = generate_structured_init(768, 'cosine')
    
    # === Phase 1: Quick multi-length exploration (120s) ===
    print("\n=== Phase 1: Multi-length exploration ===")
    t1_end = time.monotonic() + 120
    candidates = {}
    
    # Try different lengths with structured inits
    for target_n in [512, 768, 1024, 1280, 1536]:
        if time.monotonic() > t1_end:
            break
        for kind in ['cosine', 'gaussian', 'bell']:
            if time.monotonic() > t1_end:
                break
            f_init = generate_structured_init(target_n, kind)
            # Quick LP-FW to get a feel
            t_alloc = min(8, t1_end - time.monotonic())
            if t_alloc > 2:
                a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc)
                score = try_seq(a_opt, f"init_n{target_n}_{kind}")
                key = (target_n, kind)
                if key not in candidates or score < candidates[key][1]:
                    candidates[key] = (a_opt, score)
    
    # Also try parent at its native length and interpolated
    t1b = t1_end - time.monotonic()
    if t1b > 5:
        a_opt, _ = lp_fw_optimize(parent, time_limit=min(t1b, 20))
        try_seq(a_opt, "parent_quick_lpfw")
    
    # Get current best for further optimization
    base = best_seq if best_seq is not None else parent
    
    # === Phase 2: P-norm L-BFGS (500s) ===
    print(f"\n=== Phase 2: P-norm L-BFGS (best={best_score:.6f}) ===")
    t2_end = time.monotonic() + 500
    
    # P-norm on current best
    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 30:
        a_opt, _ = pnrm_lbfqg_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "pnrm_main")
    
    base = best_seq if best_seq is not None else base
    
    # P-norm on interpolated to longer length
    for target_n in [1024, 1280]:
        t_alloc = min(120, t2_end - time.monotonic())
        if t_alloc < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        a_opt, _ = pnrm_lbfqg_optimize(f_interp, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"pnrm_len{target_n}")
        base = best_seq if best_seq is not None else base
    
    # === Phase 3: LSE L-BFGS (300s) ===
    print(f"\n=== Phase 3: LSE L-BFGS (best={best_score:.6f}) ===")
    t3_end = time.monotonic() + 300
    
    base = best_seq if best_seq is not None else parent
    t_alloc = min(150, t3_end - time.monotonic())
    if t_alloc > 30:
        a_opt, _ = lse_lbfqg_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lse_main")
    
    # LSE on perturbed
    t_alloc = min(100, t3_end - time.monotonic())
    if t_alloc > 20:
        noise = rng.standard_normal(len(base)) * 0.01
        f_pert = project_simplex(base + noise)
        a_opt, _ = lse_lbfqg_optimize(f_pert, time_limit=t_alloc)
        try_seq(a_opt, "lse_perturbed")
    
    # === Phase 4: LP-FW deep refinement (350s) ===
    print(f"\n=== Phase 4: LP-FW deep (best={best_score:.6f}) ===")
    t4_end = time.monotonic() + 350
    
    base = best_seq if best_seq is not None else parent
    t_alloc = min(180, t4_end - time.monotonic())
    if t_alloc > 20:
        a_opt, _ = lp_fw_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_deep")
    
    # LP-FW on reverse
    t_alloc = min(80, t4_end - time.monotonic())
    if t_alloc > 15:
        a_rev = base[::-1].copy()
        a_opt, _ = lp_fw_optimize(a_rev, time_limit=t_alloc)
        try_seq(a_opt, "lpfw_reverse")
    
    # LP-FW on interpolated longer
    for target_n in [1536, 2048]:
        t_alloc = min(60, t4_end - time.monotonic())
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"lpfw_len{target_n}")
        base = best_seq if best_seq is not None else base
    
    # === Phase 5: Adam optimization (200s) ===
    print(f"\n=== Phase 5: Adam (best={best_score:.6f}) ===")
    t5_end = time.monotonic() + 200
    
    base = best_seq if best_seq is not None else parent
    t_alloc = min(120, t5_end - time.monotonic())
    if t_alloc > 20:
        a_opt, _ = adam_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "adam_main")
    
    # Adam on perturbed
    t_alloc = min(60, t5_end - time.monotonic())
    if t_alloc > 15:
        noise = rng.standard_normal(len(base)) * 0.008
        f_pert = project_simplex(base + noise)
        a_opt, _ = adam_optimize(f_pert, time_limit=t_alloc)
        try_seq(a_opt, "adam_perturbed")
    
    # === Phase 6: Simulated annealing (150s) ===
    print(f"\n=== Phase 6: Simulated Annealing (best={best_score:.6f}) ===")
    t6_end = time.monotonic() + 150
    
    base = best_seq if best_seq is not None else parent
    t_alloc = min(100, t6_end - time.monotonic())
    if t_alloc > 30:
        a_opt, _ = simulated_annealing(base, time_limit=t_alloc)
        try_seq(a_opt, "sa_main")
    
    # SA on the SA result (continued cooling)
    t_alloc = min(40, t6_end - time.monotonic())
    if t_alloc > 10 and best_seq is not None:
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "sa_continued")
    
    # === Phase 7: Coordinate descent (150s) ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.6f}) ===")
    t7_end = time.monotonic() + 150
    
    base = best_seq if best_seq is not None else parent
    t_alloc = min(80, t7_end - time.monotonic())
    if t_alloc > 10:
        a_opt, _ = coordinate_descent(base, time_limit=t_alloc)
        try_seq(a_opt, "cd_main")
    
    # CD on perturbed
    t_alloc = min(50, t7_end - time.monotonic())
    if t_alloc > 10:
        noise = rng.standard_normal(len(base)) * 0.003
        f_pert = project_simplex(base + noise)
        a_opt, _ = coordinate_descent(f_pert, time_limit=t_alloc)
        try_seq(a_opt, "cd_perturbed")
    
    # === Phase 8: Final intensive polishing (remaining time) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Alternate between LP-FW and P-norm with tiny perturbations
        for attempt in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            
            if attempt % 3 == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            elif attempt % 3 == 1:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = pnrm_lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_pnrm_{attempt}")
            else:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = lse_lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lse_{attempt}")
        
        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
