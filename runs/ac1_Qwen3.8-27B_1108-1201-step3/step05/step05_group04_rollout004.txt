from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(f, b):
    """Linear cross-convolution via FFT."""
    n1, n2 = len(f), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(F * B, fft_size)[:size]

def lp_best_response(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k."""
    # Build constraint matrix A such that A @ b = (f*b)_0, ..., (f*b)_{2n-2}
    # (f*b)_k = sum_{i} f[i] * b[k-i]

    rows = []
    cols = []
    vals = []

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the "best response" LP at each iteration, combined with Adam fine-tuning.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_soft_max_weights(conv, temp, top_k=None):
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w

def compute_subgradient(f, w):
    n = len(f)
    return np.correlate(w, f, 'valid')

def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,
                  top_k=None):

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)
        conv = fast_convolve(f)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj

def lp_iteration(f, M, n, verbose=False):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k."""
    c = -np.ones(n)
    # Build constraint matrix
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(m, j + n)
        A[j:end, j] = f[:end - j]
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.status != 0:
            return None
        return result.x
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
        return None
