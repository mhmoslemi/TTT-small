Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive log-softmax L-BFGS,
peak-deflation heuristics, and multi-scale coordinate descent.
Key innovations: aggressive temperature sweep, structured peak reduction,
and multi-length exploration with interpolation-based scaling.

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_v_a(v, a):
    """
    Compute g[j] = sum_{m=0}^{n-1} a[m] * v[j+m] for j=0,...,n-1.
    v has length 2n-1, a has length n.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    A = rfft(a, L)
    V = rfft(v, L)
    h = irfft(np.conj(A) * V, L)
    return h[:n]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def smooth_obj_grad_z(z, T):
    """
    Log-softmax smooth objective and gradient w.r.t. z.
    a = softmax(z), b = conv(a,a).
    F = (2n/T) * log(sum exp(T*b))
    """
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Softmax weights
    p = exp_Tb / exp_Tb.sum()
    
    # grad_a = 2n * 2 * cross_corr(p, a) = 4n * cross_corr(p, a)
    grad_a = 2.0 * 2.0 * n * cross_corr_v_a(p, a)
    
    # Chain rule for log-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def power_obj_grad_z(z, p):
    """

    Actually let's use: 2n * (sum b_k^p)^(1/p)

    = 2n * (sum b_k^p)^(1/p - 1) * sum_k b_k^(p-1) * 2 * a_{k-j}
    """
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    b_p = b ** p
    sum_b_p = b_p.sum()
    obj = 2.0 * n * sum_b_p ** (1.0 / p)
    
    # weights q_k = b_k^(p-1) / (sum b^p)^(1 - 1/p) * ... 
    # Actually: d obj / da_j = 2n * (sum b^p)^(1/p - 1) * sum_k b_k^(p-1) * 2 * a_{k-j}
    # = 4n * (sum b^p)^(1/p - 1) * sum_k b_k^(p-1) * a_{k-j} [0<=k-j<n]
    # = 4n * (sum b^p)^(1/p - 1) * cross_corr(b_^(p-1), a)_j
    
    q = b ** (p - 1)
    factor = 4.0 * n * sum_b_p ** (1.0 / p - 1.0)
    grad_a = factor * cross_corr_v_a(q, a)
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def lbgfs_logspace(z0, T, time_budget, maxiter=5000, use_power=False, power_p=50):
    """Run L-BFGS in log-space with time limit."""
    t0 = time.monotonic()
    elapsed = [0.0]
    
    def func(z):
        elapsed[0] = time.monotonic()
        if elapsed[0] - t0 > time_budget:
            return 1e15
        if use_power:
            return power_obj_grad_z(z, power_p)[0]
        return smooth_obj_grad_z(z, T)[0]
    
    def jac(z):
        if time.monotonic() - t0 > time_budget:
            return np.zeros_like(z)
        if use_power:
            return power_obj_grad_z(z, power_p)[1]
        return smooth_obj_grad_z(z, T)[1]
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def project_simplex(v):
    """Project onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_deflate(a, time_budget, n_peaks=5, n_iters=500):
    """
    Peak deflation: iteratively reduce the top peaks by adjusting
    the elements that contribute most to them.
    """
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    b = fast_convolve(a)
    best_peak = b.max()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        # Find top peaks
        k_peaks = min(n_peaks, len(b))
        peak_idx = np.argpartition(b, -k_peaks)[-k_peaks:]
        peak_idx = sorted(peak_idx.tolist(), key=lambda x: -b[x])
        
        improved = False
        for k in peak_idx:
            if time.monotonic() - t0 > time_budget:
                break
            if b[k] < best_peak * 0.99:
                continue
            
            # Compute contributions to b[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # Find the max contributor
            max_c = np.argmax(contribs)
            i = idx[max_c]
            j_complement = k - i
            
            # Try reducing a[i]
            old_val = a[i]
            if old_val < 1e-12:
                continue
            
            # Try several reduction factors
            best_factor = None
            best_new_peak = best_peak
            for factor in [0.85, 0.9, 0.95, 0.97, 0.99]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute proportionally to non-peak-contributing elements
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                # Reduce redistribution to elements that contribute to peaks
                other_contrib = np.zeros(n)
                for pk in peak_idx:
                    pjs = max(0, pk - n + 1)
                    pje = min(n - 1, pk)
                    for ii in range(pjs, pje + 1):
                        other_contrib[ii] += a[ii] * a[pk - ii]
                
                # Add more weight to non-contributing elements
                weights = mask.astype(float) * (1.0 + 0.1 * other_contrib / max(other_contrib.max(), 1e-15))
                weights[i] = 0
                w_sum = weights.sum()
                if w_sum < 1e-15:
                    continue
                a_trial[mask] += diff * weights[mask] / w_sum
                a_trial = a_trial / a_trial.sum()
                
                trial_peak = fast_convolve(a_trial).max()
                if trial_peak < best_new_peak:
                    best_new_peak = trial_peak
                    best_factor = factor
                    best_trial = a_trial.copy()
            
            if best_factor is not None:
                a = best_trial
                best_peak = best_new_peak
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Try smaller perturbations
            b = fast_convolve(a)
            k = int(np.argmax(b))
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            max_c = np.argmax(contribs)
            i = idx[max_c]
            old_val = a[i]
            if old_val < 1e-12:
                break
            
            for factor in [0.995, 0.998, 0.999, 0.9995]:
                new_val = old_val * factor
                diff = old_val - new_val
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] += diff * a[mask] / a[mask].sum()
                a_trial = a_trial / a_trial.sum()
                trial_peak = fast_convolve(a_trial).max()
                if trial_peak < best_peak - 1e-15:
                    a = a_trial
                    best_peak = trial_peak
                    best_a = a.copy()
                    break
    
    return best_a


def coord_descent(a0, time_budget, n_multipliers=8):
    """Coordinate descent with fine multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    # Fine multipliers around 1.0
    multipliers = np.array([0.7, 0.8, 0.85, 0.9, 0.95, 0.97, 0.99, 1.01, 1.03, 1.05, 1.1, 1.15, 1.2, 1.3, 1.5])
    
    converged = False
    while not converged and time.monotonic() - t0 < time_budget:
        improved = False
        converged = True
        for j in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_peak = best_peak
            best_j_val = old_val
            best_j_scale = 1.0
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                trial_peak = fast_convolve(a_trial).max()
                if trial_peak < best_j_peak - 1e-14:
                    best_j_peak = trial_peak
                    best_j_val = new_val
                    best_j_scale = scale
            
            if best_j_val != old_val:
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= best_j_scale
                a = a / a.sum()
                
                if best_j_peak < best_peak - 1e-14:
                    best_peak = best_j_peak
                    best_a = a.copy()
                    improved = True
                    converged = False
    
    return best_a


def subgradient_step(a, lr, mom, T, n_top=20, v=None):
    """One subgradient step with momentum."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(n_top, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    scaled = T * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Gradient of weighted sum of peaks
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    
    if v is None:
        v = np.zeros(n)
    v = mom * v - lr * grad
    a_new = project_simplex(a + v)
    return a_new, v


def subgradient_polish(a0, time_budget, lr=0.002, mom=0.9, T=5.0, n_top=30):
    """Subgradient polishing with adaptive parameters."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    v = np.zeros(n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        a, v = subgradient_step(a, lr, mom, T, n_top, v)
        peak = fast_convolve(a).max()
        
        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % 2**31)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr *= 0.7
            T *= 1.5
            no_improve = 0
            if lr < 0.0001:
                lr = 0.001
                T = 5.0
    
    return best_a
