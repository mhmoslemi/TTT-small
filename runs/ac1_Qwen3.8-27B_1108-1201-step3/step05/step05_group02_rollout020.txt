```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_topk(f, M, n, top_k=None):
    """Solve LP with only top-K constraints for speed."""
    if top_k is None or top_k >= 2*n-1:
        top_k = 2*n-1
    
    conv = fast_convolve(f)
    # Get the top_k largest conv values
    if top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        idx = np.arange(len(conv))
    
    c = -np.ones(n)
    A_ub = np.zeros((len(idx), n))
    for k in idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_idx = np.arange(js, je + 1)
        A_ub[list(idx).index(k), j_idx] = f[k - j_idx]
    
    b_ub = np.full(len(idx), M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Fast smoothed gradient descent (log-sum-exp surrogate) with momentum
2. LP-Frank-Wolfe with constraint generation (fast, many iterations)
3. Multi-length exploration with interpolation
4. Coordinate descent polishing
5. Final micro-perturbations
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv[k])) w.r.t. a.
    Uses FFT for efficiency: grad[m] = 2 * sum_j w[j+m]*a[j]
    where w = softmax(tau*conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_scaled = tau * conv
    conv_scaled -= conv_scaled.max()
    w = np.exp(conv_scaled)
    w /= w.sum()

    # c[m] = sum_j w[j+m]*a[j] for m=0..n-1
    # Using FFT cross-correlation
    L = 1 << (len(w) + n - 2).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    R = np.fft.irfft(W * np.conj(A), L)
    grad = 2.0 * R[:n]
    return grad, conv


def smoothed_gd_descent(f0, time_limit, tau_init=5.0, tau_final=500.0,
                        lr_init=0.01, lr_final=0.0001, momentum=0.9):
    """Smoothed gradient descent on the simplex with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))

        # Anneal tau and learning rate
        tau = tau_init * (tau_final / tau_init) ** progress
        lr = lr_init * (lr_final / lr_init) ** (0.5 * progress)

        # Nesterov lookahead
        f_look = f + momentum * v
        grad, conv = compute_lse_gradient(f_look, tau)

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Evaluate
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 500:
            # Perturb and restart
            rng = np.random.default_rng(it * 127 + 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            no_improve = 0

        if it % 5000 == 0:
            print(f"    SGD iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, tau={tau:.1f}")

    return best_f, best_obj


def solve_lp_cg(f, M, n, max_cg_iter=30, initial_k=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    using constraint generation for speed.
    """
    # Start with the peak constraints (from f*f as proxy)
    conv_ff = fast_convolve(f)
    k_indices = list(np.argpartition(conv_ff, -initial_k)[-initial_k:])

    for _ in range(max_cg_iter):
        n_k = len(k_indices)
        c = -np.ones(n)
        A_ub = np.zeros((n_k, n))
        # Build constraint matrix: (f*g)[k] = sum_j f[k-j]*g[j]
        for i, k in enumerate(k_indices):
            js_lo = max(0, k - n + 1)
            js_hi = min(n - 1, k)
            if js_lo > js_hi:
                continue
            j_arr = np.arange(js_lo, js_hi + 1)
            A_ub[i, j_arr] = f[k - j_arr]

        b_ub = np.full(n_k, M)
        bounds = [(0, None)] * n
        options = {'time_limit': 20, 'presolve': True}
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options=options)

        if not res.success or res.x is None:
            return None

        g = res.x
        if g.sum() < 1e-15:
            return None

        # Check all constraints via FFT
        fg = fast_convolve_cross(f, g)
        violations = fg > M + 1e-9
        if not violations.any():
            return g

        # Add the most violated constraint
        worst_k = int(np.argmax(fg))
        if worst_k not in k_indices:
            k_indices.append(worst_k)

    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with constraint generation and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g = solve_lp_cg(f, M * m_relax, n, max_cg_iter=25, initial_k=12)
        if g is None:
            m_relax *= 1.1
            if m_relax > 10.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                # Perturb
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.002 + 0.003 * min(no_improve // 50, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            else:
                m_relax = min(m_relax * 1.03, 8.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best_alpha
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 8.0)

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_obj = best_obj

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                continue

            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]
            best_j_val = old_val
            best_j_obj = cur_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break
        if passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(budget_s=1800):
    """
    Multi-phase optimization combining smoothed GD, LP-FW with constraint
    generation, multi-length exploration, and coordinate descent.
    """
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Smoothed Gradient Descent (fast exploration) ===
    print(f"\n=== Phase 1: Smoothed GD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(120, rem * 0.08)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = smoothed_gd_descent(best_seq, time_limit=t1,
                                       tau_init=3.0, tau_final=200.0,
                                       lr_init=0.008, lr_final=0.0001)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: LP-FW with constraint generation ===
    print(f"\n=== Phase 2: LP-FW CG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(500, rem * 0.33)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 3: LP-FW restarts with different initializations ===
    print(f"\n=== Phase 3: LP-FW restarts (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    restarts = []
    # Small noise
    restarts.append(("noise2k", project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)))
    # Medium noise
    restarts.append(("noise5k", project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)))
    # Reverse
    f_rev = 0.7 * best_seq + 0.3 * best_seq[::-1]
    restarts.append(("reverse", project_simplex(f_rev)))
    # Mix with uniform
    f_mix = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
    restarts.append(("mix_unif", project_simplex(f_mix)))
    # Shift
    f_shift = np.roll(best_seq, 3)
    restarts.append(("shift", project_simplex(f_shift)))

    for label, f_init in restarts:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(200, rem * 0.2)
        if t_alloc < 15:
            continue
        print(f"  {label}: t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    candidate_lengths = set()
    for delta in [-128, -64, -32, 32, 64, 128, 256, -256, 512, -512]:
        nl = n_best + delta
        if 256 <= nl <= 2048 and nl != n_best:
            candidate_lengths.add(nl)

    # Sort by proximity to current best
    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))
    candidate_lengths = candidate_lengths[:4]

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        # Also try a fresh initialization at this length
        t_alloc = min(150, rem * 0.25)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (n={target_n}): {sc:.6f}")

    # === Phase 5: Coordinate descent on best ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 30:
        t_cd = min(rem * 0.5, 250)
        print(f"  t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 6: Final LP-FW squeeze ===
    print(f"\n=== Phase 6: Final LP-FW squeeze (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_sq = min(rem * 0.7, 150)
        print(f"  t={t_sq:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sq)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 7: Micro-perturbations ===
    print(f"\n=== Phase 7: Micro-perturbations (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        scale = 0.0002 + trial * 0.0001
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick polish
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem2 * 0.4, 30))
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_lp]: {sc2:.6f}")

    # Final LP-FW with remaining time
    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL LP: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
