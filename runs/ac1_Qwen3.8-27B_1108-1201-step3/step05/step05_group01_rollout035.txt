Let me also add a new idea: **spectral shaping**. Since the convolution is related to the squared magnitude of the FFT, we can think of this problem in the frequency domain. Specifically:

So max(conv(a,a)) is the maximum of the inverse FFT of |fft(a)|^2. This is the peak of the "power spectrum" in the time domain (which is the autocorrelation).

In the frequency domain, |fft(a)|^2 is the power spectrum. The constraint "max of ifft of power spectrum <= M" means that the autocorrelation is "flat" (bounded by M).

This is related to the "minimum peak-to-average power ratio" (PAPR) problem, but for the autocorrelation rather than the signal itself.

One approach: **iteratively modify the spectrum** to flatten the autocorrelation. In the frequency domain, we can:
1. Compute the current spectrum |A(f)|^2
2. Compute the autocorrelation r = ifft(|A(f)|^2)
3. Identify the peaks in r
4. Modify |A(f)| to reduce the peaks: e.g., reduce |A(f)| at frequencies that contribute most to the peaks

This is similar to the "successive overrelaxation" method for PAPR reduction.

def freq_domain_optimize(a, time_limit):
    n = len(a)
    a = a / a.sum()  # normalize
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    gamma = 0.5  # relaxation parameter
    
    while time.monotonic() - t0 < time_limit:
        # Compute spectrum
        fft_size = 1 << (2*n-1).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2
        
        # Compute autocorrelation
        r = np.fft.irfft(power, fft_size)[:2*n-1]
        r_max = r.max()
        
        # Compute "excess" at peaks
        # Modify spectrum: reduce power at frequencies that contribute to peaks
        # The contribution of frequency f to r[k] is |A(f)|^2 * exp(2*pi*i*f*k/fft_size)
        # So to reduce r[k] for specific k, we should reduce |A(f)|^2 * cos(2*pi*f*k/fft_size)
        # where this is positive.
        
        # Simpler: use iterative water-filling
        # r_new[k] = r[k] - gamma * max(0, r[k] - target)
        # then A_new = sqrt(rfft_inverse(r_new)) ... but this doesn't preserve non-negativity
        
        # Alternative: directly modify a
        # grad_k = 2 * a[k-i] for the peak lag k
        # Step: a -= step * projected_grad, then project to simplex
        
        # Let's use the subgradient approach with adaptive step
        peaks = np.where(r >= r_max * 0.99)[0]
        sg = np.zeros(n)
        for k in peaks:
            weight = r[k] - r_max * 0.99
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sg[i] += weight * a[j]
        
        # Project gradient to tangent space
        sg_tan = sg - sg.mean()
        
        # Adaptive step size
        step = 0.1 / (np.linalg.norm(sg_tan) + 1e-10)
        
        a_new = a - step * sg_tan
        a_new = project_simplex(a_new)
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            gamma *= 0.95
    
    return best_a, best_obj

def feasibility_check(M, n, time_limit, max_iter=5000):
    """Check if there exists a >= 0, sum a = 1, with conv(k) <= M for all k."""
    # Use projected gradient on the penalty function
    # P(a) = sum_k max(0, conv_k(a) - M)^2
    
    rng = np.random.default_rng(42)
    a = np.ones(n) / n
    best_penalty = np.inf
    t0 = time.monotonic()
    
    lr = 0.1
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        excess = np.maximum(0, conv - M)
        penalty = np.sum(excess**2)
        
        if penalty < best_penalty:
            best_penalty = penalty
            if penalty < 1e-12:
                return True, a
        
        # Gradient of penalty
        # d/d a_i [sum_k max(0, conv_k - M)^2] = 2 * sum_k max(0, conv_k - M) * d(conv_k)/da_i
        # = 2 * sum_k excess[k] * 2 * a[k-i] * [0<=k-i<n]
        # = 4 * sum_j a[j] * sum_k excess[k] * [k = i+j]
        # = 4 * sum_j a[j] * excess[i+j] (where excess[k]=0 for k outside range)

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_smooth_grad(conv, a, beta):
    """Compute gradient of smooth approximation obj = max_k conv[k] approximated by log-sum-exp.
    Returns (obj, grad_a) where grad_a[i] = 2 * sum_j a[j] * w[i+j]."""
    n = len(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta + conv_max - conv_max  # simplify
    obj = conv_max + np.log(np.exp(beta * (conv - conv_max)).sum()) / beta

    # grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # = 2 * conv(a[::-1], w)[i + n - 1]
    a_rev = a[::-1]
    N = 1 << (3 * n - 2).bit_length()
    A_rev = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(A_rev * W, N)[:3 * n - 2]
    grad_a = 2.0 * cross[n - 1:2 * n - 1]
    return obj, grad_a


def smooth_lbfqg(f0, time_limit, betas=None, verbose=False):
    """L-BFGS optimization on smooth approximation with annealed beta in log-softmax space."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    if betas is None:
        betas = [3.0, 6.0, 12.0, 24.0, 48.0, 96.0, 192.0, 384.0, 768.0]

    eps = 1e-15
    t = np.log(f + eps)
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 15:
            break
        sub_time = remaining / (len(betas) - bi)

        def obj_grad(t_val, beta_val):
            a = softmax_fn(t_val)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-15:
                a = np.ones(len(a)) / len(a)
            else:
                a = a / a_sum
            conv = fast_convolve(a)
            obj, grad_a = compute_smooth_grad(conv, a, beta_val)

            # Convert grad_a to grad_t: grad_t = a * (grad_a - <a, grad_a>)
            dot = np.dot(a, grad_a)
            grad_t = a * (grad_a - dot)
            return obj, grad_t

        try:
            res = minimize(
                obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'gtol': 1e-10, 'maxls': 50, 'maxfun': 5000}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            if verbose:
                print(f"    L-BFGS beta={beta} failed")

    return best_f, best_obj


def penalty_refine(a0, M_target, time_limit, verbose=False):
    """Try to find a feasible point (sum a=1, a>=0, max conv <= M_target) using penalty method.
    Returns (success, best_a, best_obj)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    current_max = conv.max()
    best_a = a.copy()
    best_obj = current_max

    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        excess = np.maximum(0.0, conv - M_target)
        penalty = np.sum(excess ** 2)

        if penalty < 1e-18:
            return True, a, conv.max()

        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            # Try to reduce M_target constraint by using a slightly relaxed target
            relaxed = M_target * (1.0 + 0.001 * (no_improve // 200))
            excess = np.maximum(0.0, conv - relaxed)
            lr *= 0.5
            if lr < 0.01:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                lr = 0.3
                no_improve = 0
                continue

        # Compute gradient of penalty: grad[i] = 4 * sum_j a[j] * excess[i+j]
        a_rev = a[::-1]
        N = 1 << (3 * n - 2).bit_length()
        A_rev = np.fft.rfft(a_rev, N)
        E = np.fft.rfft(excess, N)
        cross = np.fft.irfft(A_rev * E, N)[:3 * n - 2]
        grad = 4.0 * cross[n - 1:2 * n - 1]

        # Project to tangent space of simplex
        grad = grad - grad.mean()

        # Line search
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = lr * mult
            a_new = project_simplex(a - step * grad)
            conv_new = fast_convolve(a_new)
            excess_new = np.maximum(0.0, conv_new - M_target)
            penalty_new = np.sum(excess_new ** 2)
            if penalty_new < penalty * 0.999:
                a = a_new
                improved = True
                break

        if not improved:
            lr *= 0.7

    return best_obj <= M_target * 1.0001, best_a, best_obj


def peak_subgradient(a0, time_limit, verbose=False):
    """Subgradient method targeting the top peaks in the convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = 0.05
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    top_k = max(5, n // 32)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        conv_max = conv.max()

        if conv_max < best_obj - 1e-14:
            best_obj = conv_max
            best_a = a.copy()
            no_improve = 0
            if verbose and best_obj < 1.51:
                print(f"    PeakSG: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Select top peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Compute weighted subgradient
        pv = conv[peaks]
        temp = 0.01 / (beta if (beta := 1.0) else 1.0)
        # Use temperature based on gap
        gap = pv.max() - pv.min()
        if gap > 1e-15:
            temp = gap * 0.1
        else:
            temp = 0.01
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Momentum + projection
        v = momentum * v - lr * sg
        v = v - v.mean()
        a = project_simplex(a + v)

        lr *= 0.9995
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 7919)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.03
            no_improve = 0
            top_k = min(top_k * 2, n // 8)

    return best_a, best_obj


def coordinate_descent_exact(a0, time_limit):
    """Coordinate descent with exact (discrete) line search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    mults = [0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in mults:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def lpfw_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def parametric_starts(n, rng):
    """Generate diverse parametric initializations."""
    candidates = []
    candidates.append(np.ones(n) / n)

    # Triangular
    t = np.linspace(-1, 1, n)
    a = np.maximum(0, 1 - np.abs(t))
    a /= a.sum()
    candidates.append(a)

    # Various bell shapes
    for p in [1.5, 2.0, 3.0, 4.0, 6.0]:
        x = np.linspace(-1, 1, n)
        a = np.abs(x) ** p
        a = a - a.min()
        if a.sum() > 1e-15:
            a /= a.sum()
            candidates.append(a.copy())

    # Gaussian-like
    for sigma in [0.25, 0.35, 0.5, 0.7]:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 / (2 * sigma ** 2))
        a /= a.sum()
        candidates.append(a)

    # Two humps
    for gap in [0.3, 0.5, 0.7]:
        x = np.linspace(-1, 1, n)
        a = np.exp(-(x - gap) ** 2 / 0.1) + np.exp(-(x + gap) ** 2 / 0.1) + 0.1
        a = np.maximum(a, 0)
        a /= a.sum()
        candidates.append(a)

    # Random
    for _ in range(4):
        a = rng.random(n)
        a = np.maximum(a, 0)
        a /= a.sum()
        candidates.append(a)

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            noise = rng.standard_normal(n_base) * scale
            starts.append(project_simplex(parent + noise))

    for si, start in enumerate(starts):
        if time.monotonic() > t1_end:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 20:
            break
        t_alloc = min(remaining * 0.45, 60)
        if t_alloc < 10:
            t_alloc = 10
        a_opt, _ = smooth_lbfqg(start, time_limit=t_alloc,
                                 betas=[5.0, 20.0, 80.0, 320.0])
        try_seq(a_opt, f"start_{si}")

    # === Phase 2: LP-FW on best (300s) ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    t2 = min(300, deadline - time.monotonic() - 900)
    if t2 > 30 and best_seq is not None:
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_main")

        # Also try from a slightly different starting point
        t2b = min(100, deadline - time.monotonic() - 800)
        if t2b > 20:
            noise = rng.standard_normal(len(best_seq)) * 0.01
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=t2b)
            try_seq(a_opt, "lpfw_pert")

    # === Phase 3: Deep smooth optimization (350s) ===
    print(f"\n=== Phase 3: Deep Smooth (best={best_score:.6f}) ===")
    t3 = min(350, deadline - time.monotonic() - 700)
    if t3 > 40 and best_seq is not None:
        a_opt, _ = smooth_lbfqg(best_seq, time_limit=t3,
                                 betas=[10.0, 40.0, 100.0, 200.0, 400.0, 800.0, 1600.0],
                                 verbose=True)
        try_seq(a_opt, "deep_smooth")

    # === Phase 4: Peak subgradient (150s) ===
    print(f"\n=== Phase 4: Peak Subgradient (best={best_score:.6f}) ===")
    t4 = min(150, deadline - time.monotonic() - 400)
    if t4 > 20 and best_seq is not None:
        a_opt, _ = peak_subgradient(best_seq, time_limit=t4, verbose=True)
        try_seq(a_opt, "peak_sg")

    # === Phase 5: Penalty refinement (200s) ===
    print(f"\n=== Phase 5: Penalty Refinement (best={best_score:.6f}) ===")
    t5 = min(200, deadline - time.monotonic() - 300)
    if t5 > 30 and best_seq is not None:
        n_b = len(best_seq)
        # Current max_b in normalized (sum=1) space
        cur_sum = best_seq.sum()
        best_norm = best_seq / cur_sum
        conv_now = fast_convolve(best_norm)
        cur_max_b = conv_now.max()

        # Try to reduce M slightly
        for reduction in [0.0002, 0.0005, 0.001, 0.002]:
            if time.monotonic() > deadline - 250:
                break
            M_target = cur_max_b * (1.0 - reduction)
            t_alloc = min(40, deadline - time.monotonic() - 250)
            if t_alloc < 10:
                break
            success, a_trial, obj_trial = penalty_refine(best_norm, M_target, t_alloc)
            if success:
                try_seq(a_trial, f"penalty_{reduction}")
                cur_max_b = obj_trial
            else:
                # Still check if we got closer
                if obj_trial < cur_max_b:
                    try_seq(a_trial, f"penalty_close_{reduction}")
                    cur_max_b = obj_trial

    # === Phase 6: Coordinate descent (150s) ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    t6 = min(150, deadline - time.monotonic() - 100)
    if t6 > 20 and best_seq is not None:
        a_opt, _ = coordinate_descent_exact(best_seq, time_limit=t6)
        try_seq(a_opt, "coord_descent")

    # === Phase 7: Multi-length exploration (200s) ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    t7 = min(200, deadline - time.monotonic() - 100)
    if t7 > 40 and best_seq is not None:
        for target_n in [640, 896, 1024]:
            if time.monotonic() > deadline - 80:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(t7 * 0.3, 50)
            if t_alloc < 15:
                break
            a_opt, _ = smooth_lbfqg(f_interp, time_limit=t_alloc,
                                     betas=[10.0, 40.0, 160.0, 640.0])
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 8: Final deep optimization (remaining time) ===
    print(f"\n=== Phase 8: Final Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20

        # 8a: More LP-FW
        if remaining > 60:
            t_alloc = min(remaining * 0.4, 120)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lpfw")

        # 8b: More smooth with very high beta
        remaining = deadline - time.monotonic() - 20
        if remaining > 40:
            t_alloc = min(remaining * 0.5, 100)
            a_opt, _ = smooth_lbfqg(best_seq, time_limit=t_alloc,
                                     betas=[500.0, 1000.0, 2000.0, 4000.0],
                                     verbose=True)
            try_seq(a_opt, "final_smooth_high")

        # 8c: Another penalty pass
        remaining = deadline - time.monotonic() - 15
        if remaining > 20:
            best_norm = best_seq / best_seq.sum()
            conv_now = fast_convolve(best_norm)
            cur_max_b = conv_now.max()
            M_target = cur_max_b * 0.9995
            t_alloc = min(remaining * 0.7, 50)
            success, a_trial, obj_trial = penalty_refine(best_norm, M_target, t_alloc)
            if success:
                try_seq(a_trial, "final_penalty")
            elif obj_trial < cur_max_b:
                try_seq(a_trial, "final_penalty_close")

        # 8d: Final coordinate descent
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            a_opt, _ = coordinate_descent_exact(best_seq, time_limit=min(remaining, 40))
            try_seq(a_opt, "final_cd")

        # 8e: Final peak subgradient
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, _ = peak_subgradient(best_seq, time_limit=min(remaining, 30))
            try_seq(a_opt, "final_peak")

        # 8f: One more L-BFGS from perturbed start
        remaining = deadline - time.monotonic() - 3
        if remaining > 15:
            noise = rng.standard_normal(len(best_seq)) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_lbfqg(f_pert, time_limit=min(remaining, 30),
                                     betas=[100.0, 400.0, 1600.0])
            try_seq(a_opt, "final_pert")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
