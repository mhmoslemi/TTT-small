"""
Multi-phase optimization for minimizing normalized peak convolution of
non-negative step functions. Uses subgradient FISTA with structured
initializations, adaptive step sizes, and multi-restart local search.
"""

import numpy as np
import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)

def compute_subgradient(a):
    """Compute subgradient of the score function."""
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a)
    M = np.max(c)
    # Find all lags achieving the max
    K = np.where(np.isclose(c, M, rtol=1e-10))[0]
    if len(K) == 0:
        K = np.array([np.argmax(c)])
    
    # Compute dM/da[j] for each j
    grad_M = np.zeros(n)
    for k in K:
        for j in range(n):
            # dc[k]/da[j] = a[k-j] + a[j-k] with a[m]=0 outside [0,n-1]
            kj = k - j
            jk = j - k
            val = 0.0
            if 0 <= kj < n:
                val += a[kj]
            if 0 <= jk < n:
                val += a[jk]
            grad_M[j] += val
    
    grad_M /= len(K)
    
    # df/da[j] = 2n/S^2 * (dM/da[j] - 2*M*a[j]/S)
    grad = (2.0 * n / (S * S)) * (grad_M - 2.0 * M * a / S)
    return grad, M, S, c

def fista_optimize(a_init, time_limit, lr=None, verbose=False):
    """FISTA subgradient optimization with projection."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    k = 1  # FISTA momentum counter
    z = a.copy()
    
    if lr is None:
        max_a = np.max(a)
        lr = S * S / (2.0 * n * 2.0 * max(a) * 2.0)
        lr = min(lr, 0.1)
    
    iter_count = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        beta = (k - 1.0) / t_new if k > 1 else 0.0
        z = a + beta * (a - z)
        z = np.clip(z, 0.0, None)
        
        # Compute subgradient at z
        grad, M, S_z, c = compute_subgradient(z)
        
        # Gradient step
        a_new = z - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = z
        else:
            # Don't renormalize; the score is scale-invariant but we want
            # to keep the scale reasonable for numerical stability
            pass
        
        score_new = compute_score(a_new)
        iter_count += 1
        k = t_new
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stagnant = 0
            # Adapt learning rate
            if lr < 0.5:
                lr *= 1.02
        else:
            stagnant += 1
            if stagnant > 50:
                lr *= 0.8
                stagnant = 0
            if stagnant > 200:
                # Restart with perturbation
                a_new = best_a + 0.01 * np.random.default_rng(iter_count).standard_normal(n)
                a_new = np.clip(a_new, 0.0, None)
                lr = 0.05
                stagnant = 0
                k = 1
                z = a_new.copy()
        
        a = a_new
        
        if verbose and iter_count % 100 == 0:
            print(f"  FISTA iter={iter_count}, lr={lr:.4f}, score={score_new:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def compute_subgradient(a):
    """
    Compute a subgradient of f(a) = 2n*max_k c[k] / S^2.
    Returns (grad, M, S, c).
    """
    n = len(a)
    S = np.sum(a)

    grad_M = np.zeros(n)

    best_score = compute_score(a)

    z = a.copy()

    stagnant = 0

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * k_mom * k_mom)) / 2.0
            beta = (k_mom - 1.0) / t_new

            z = np.clip(z, 0.0, None)

            k_mom = 2

            k_mom = 1

        a_new = z - lr * grad
        a_new = np.clip(a_new, 0.0, None)

            a_new = z

                S_new = S

def peak_directed_search(a_init, time_limit, rng, verbose=False):
    """
    Local search that directly targets the peak convolution values.
    At each step, identify elements contributing most to the max lag,
    reduce them, and increase elements contributing to non-peak lags.
    """
    a = np.clip(a_init.astype(np.float64), 0.0, None)
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf

    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a)
        M = np.max(c)
        peak_idx = int(np.argmax(c))

        # Find elements contributing to the peak
        contribs = np.zeros(n)
        for j in range(n):
            kj = peak_idx - j
            if 0 <= kj < n:
                contribs[j] += a[j] * a[kj]
            jk = j - peak_idx
            if 0 <= jk < n:
                contribs[j] += a[j] * a[jk]

        # Also track near-peak lags
        near_peaks = np.where(c >= M * 0.95)[0]

        candidate = a.copy()
        r = rng.random()

        if r < 0.5:
            # Reduce top contributors to peak
            top_k = max(3, n // 16)
            top_idx = np.argsort(contribs)[-top_k:]
            factors = np.exp(rng.normal(-0.04, 0.015, size=top_k))
            candidate[top_idx] *= factors
            # Increase some non-top elements
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.7:
            # Anti-palindromic push: reduce a[i] where a[n-1-i] is also large
            prod = a * a[::-1]
            top_k = max(3, n // 10)
            top_idx = np.argsort(prod)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=top_k))
            # Boost the "opposite" side
            opp_idx = n - 1 - top_idx
            valid = (opp_idx >= 0) & (opp_idx < n)
            if np.any(valid):
                candidate[opp_idx[valid]] *= np.exp(rng.normal(0.03, 0.015, size=np.sum(valid)))
        elif r < 0.85:
            # Windowed perturbation
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (S / cand_sum)

        new_score = compute_score(candidate)
        iter_count += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling
        if temperature < 1e-5:
            temperature = 0.001  # Reheat occasionally

    return best_a, best_score

def make_cosine_init(n, alpha, rng):

def make_ramp_window_init(n, rng):

    a += 0.5 * rng.choice([-1, 1]) * (t - 0.5)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_provided_init(provided_seq, n, rng):
    """Interpolate provided sequence to target length."""
    if len(provided_seq) == n:
        return provided_seq.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(provided_seq))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided_seq)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Generate diverse structured initializations.
    Phase 2: FISTA subgradient optimization on each.
    Phase 3: Peak-directed local search refinement.
    Phase 4: Final coordinate/FISTA polish on best.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = compute_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # Define search configurations
    n_prov = len(provided_seq) if provided_seq is not None else 256

    search_configs = [
        (n_prov, 'provided', 0.22),
        (n_prov, 'cosine', 0.15),
        (n_prov, 'ramp_window', 0.12),
        (n_prov, 'chirp', 0.10),
        (n_prov, 'multitone', 0.08),
        (n_prov, 'polynomial', 0.08),
        (512, 'cosine', 0.10),
        (512, 'ramp_window', 0.08),
        (384, 'cosine', 0.07),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = deadline - time.monotonic()

    print("\n" + "=" * 60)
    print("Phase 1+2: FISTA optimization on diverse initializations")
    print("=" * 60)

    for n, init_name, frac in search_configs:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(20.0, frac / total_frac * available_time * 0.88)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 20)
        if time_alloc < 15:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_provided_init(provided_seq, n, rng)
        elif init_name == 'cosine':
            alpha = 0.4 + 0.5 * rng.random()
            a0 = make_cosine_init(n, alpha, rng)
        elif init_name == 'ramp_window':
            a0 = make_ramp_window_init(n, rng)
        elif init_name == 'chirp':
            alpha = 0.3 + 0.5 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng)
        elif init_name == 'polynomial':
            a0 = make_polynomial_init(n, rng)
        else:
            a0 = np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.01, None)

        score0 = compute_score(a0)
        print(f"\n  n={n}, init='{init_name}': initial score={score0:.6f}, time_alloc={time_alloc:.0f}s")

        if score0 < best_score:
            best_score = score0
            best_seq = a0.copy()
            print(f"  *** New best from init: {best_score:.6f} ***")

        # Run FISTA
        t_start = time.monotonic()
        f_best, score = optimize_fista(a0, time_alloc, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  FISTA done: score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # Phase 3: Peak-directed search on best
    print("\n" + "=" * 60)
    print("Phase 3: Peak-directed local search")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        pd_time = min(300.0, remaining * 0.4)
        if pd_time > 20:
            print(f"  Peak-directed search, n={len(best_seq)}, time={pd_time:.0f}s")
            a_pd, score_pd = peak_directed_search(best_seq, pd_time, rng)
            print(f"  PD result: {score_pd:.6f}")
            if score_pd < best_score:
                best_score = score_pd
                best_seq = a_pd.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Final FISTA polish
    print("\n" + "=" * 60)
    print("Phase 4: Final FISTA polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        final_time = min(200.0, remaining * 0.7)
        if final_time > 15:
            print(f"  Final FISTA, n={len(best_seq)}, time={final_time:.0f}s")
            a_fin, score_fin = optimize_fista(best_seq, final_time, lr_init=None)
            print(f"  Final FISTA result: {score_fin:.6f}")
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 5: One more peak-directed pass if time
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n  Final PD pass, time={remaining:.0f}s")
            a_fin2, score_fin2 = peak_directed_search(best_seq, remaining * 0.8, rng)
            if score_fin2 < best_score:
                best_score = score_fin2
                best_seq = a_fin2.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
