For the simplex projection (project onto {x >= 0, sum(x) = S}):
```
def simplex_project(x, S):
    """Project x onto {x >= 0, sum(x) = S}."""
    # Algorithm from projection onto the simplex
    n = len(x)
    if n == 0:
        return x
    b = np.sort(x)[::-1]
    cssv = np.cumsum(b)
    rho = np.where(b * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(x - theta, 0)
```

Yes! So the average of the autocorrelation is (sum(a))^2 / (2n-1). If the autocorration were perfectly flat, max = (sum(a))^2/(2n-1), giving objective = 2n/(2n-1) ≈ 1.

The "excess" is how much the max exceeds the average. We want to minimize this excess.

This gives another perspective: we want the autocorrelation to be as flat as possible. The gradient should push down the peaks and (implicitly) allow the valleys to rise (since the sum is fixed).

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses Nesterov-accelerated projected gradient descent on soft-max autocorrelation,
LP-based block refinement, simplex projection, and multi-start with perturbation.
Starts from height_sequence_1 if available, plus diverse random/structured starts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (S * S))


def normalize_sum(a, S):
    """Scale a to have sum S."""
    s = a.sum()
    if s < 1e-12:
        return a * (S / max(n, 1)) if (n := len(a)) > 0 else a
    return a * (S / s)


def simplex_project(x, S):
    """Project x onto {x >= 0, sum(x) = S}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def autocorr_fft(a):
    """Compute autocorrelation (full) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    Af = np.fft.rfft(a, fsize)
    C = np.fft.irfft(Af * np.conj(Af), fsize)[:size]
    return C


def soft_max_grad(a, tau):
    """
    Gradient of (1/tau)*log(sum exp(tau*c[k])) w.r.t. a.
    grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    where w[k] = exp(tau*(c[k]-c.max())) / sum(exp(tau*(c-k)-c.max()))
    Computed via FFT cross-correlation.
    """
    n = len(a)
    c = autocorr_fft(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # = 2 * (w * a_rev)[n-1-j] where a_rev = a[::-1]
    a_rev = a[::-1].copy()
    fsize = 1
    while fsize < (2 * n - 1) + n - 1:
        fsize <<= 1
    Wf = np.fft.rfft(w, fsize)
    Af = np.fft.rfft(a_rev, fsize)
    conv_res = np.fft.irfft(Wf * Af, fsize)[:2 * n - 1 + n - 1]
    # We need conv_res[n-1+j] for j=0..n-1? Let me re-derive.
    # (w * a_rev)[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-(k-i)]
    # = sum_i w[i] * a[n-1-k+i]
    # We want sum_m w[j+m] * a[m]. Let i = j+m, so m = i-j:
    # = sum_i w[i] * a[i-j] (for valid i, i-j)
    # This equals (w * a_rev)[k] where n-1-k+i = i-j, so n-1-k = -j, k = n-1+j.
    # So grad[j] = 2 * (w * a_rev)[n-1+j]
    grad = 2.0 * conv_res[n - 1:n - 1 + n]
    return grad


def hard_max_grad(a):
    """Gradient of max_k conv(a,a)[k] w.r.t. a (subgradient)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_solve(a, M):
    """
    Solve: max sum(g) s.t. conv(a,g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    if n > 512:
        return None

    c_obj = -np.ones(n)
    # A[k, i] = a[k-i] if 0 <= k-i < n else 0, for k=0..2n-2, i=0..n-1
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # a[k-i] is non-zero when 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        A[k_lo:k_hi + 1, i] = a[0:k_hi - k_lo + 1]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 5.0, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_sequence(a0, time_limit, verbose=True):
    """
    Full optimization pipeline: Nesterov GD + LP + gradient polish.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    S_target = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None).copy()
    a = simplex_project(a, S_target)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    if verbose:
        print(f"    n={n}, init_score={score:.6f}")

    deadline = time.monotonic() + time_limit

    # Phase 1: Nesterov accelerated gradient descent with soft-max
    # Use increasing tau (soft to hard)
    momentum = 0.9
    lr = 0.5
    tau = 0.5
    v = np.zeros(n)
    no_improve = 0
    check_interval = max(1, n // 10)

    phase1_deadline = time.monotonic() + time_limit * 0.45
    it = 0
    while time.monotonic() < phase1_deadline and it < 100000:
        it += 1
        # Adapt hyperparameters
        progress = it / 5000.0
        tau = min(0.5 * (1.05 ** it), 80.0)
        lr = 0.5 * (0.999 ** it)
        lr = max(lr, 0.01)
        momentum = min(0.9 + 0.05 * (it / 1000.0), 0.99)

        grad = soft_max_grad(a, tau)
        # Project gradient: remove mean component (tangent to sum constraint)
        grad -= grad.mean()

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = simplex_project(a_new, S_target)

        # Check score periodically
        if it % check_interval == 0:
            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                best_score = s
                best = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 30:
                    # Reset and try different tau
                    v = np.zeros(n)
                    no_improve = 0
            a = a_new
        else:
            a = a_new

    if verbose:
        print(f"    Phase1: score={best_score:.6f}, iters={it}")

    a = best.copy()

    # Phase 2: LP-based block refinement
    phase2_deadline = time.monotonic() + time_limit * 0.30
    if n <= 512:
        no_improve_lp = 0
        lp_it = 0
        while time.monotonic() < phase2_deadline and lp_it < 80:
            lp_it += 1
            c = np.convolve(a, a)
            M = float(c.max())

            g = lp_solve(a, M)
            if g is None or g.sum() < 1e-6:
                break

            g = simplex_project(g, S_target)
            s_g = compute_obj(g)

            # Try blending
            improved = False
            blend_candidates = [1.0, 0.7, 0.5, 0.3, 0.15]
            if s_g < best_score - 1e-8:
                blend_candidates.append(1.0)  # already included

            for t_b in blend_candidates:
                a_new = (1.0 - t_b) * a + t_b * g
                a_new = simplex_project(a_new, S_target)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    a = a_new
                    best = a_new.copy()
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                no_improve_lp += 1
                # Small gradient nudge
                grad = hard_max_grad(a)
                grad -= grad.mean()
                a_new = a - 0.02 * grad
                a_new = simplex_project(a_new, S_target)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a_new.copy()
                    best_score = s_new
                    no_improve_lp = 0
                if no_improve_lp > 8:
                    break
            else:
                no_improve_lp = 0

        if verbose:
            print(f"    Phase2 (LP): score={best_score:.6f}, iters={lp_it}")

    a = best.copy()

    # Phase 3: Fine gradient polish with hard max
    phase3_deadline = min(time.monotonic() + time_limit * 0.20, deadline - 5)
    v = np.zeros(n)
    lr = 0.05
    it = 0
    while time.monotonic() < phase3_deadline and it < 20000:
        it += 1
        lr = max(0.05 * (0.9995 ** it), 0.002)
        grad = hard_max_grad(a)
        grad -= grad.mean()
        v = 0.95 * v - lr * grad
        a_new = a + v
        a_new = simplex_project(a_new, S_target)
        s = compute_obj(a_new)
        if s < best_score - 1e-10:
            best_score = s
            best = a_new.copy()
            a = a_new
        else:
            a = a_new
            if it % 100 == 0:
                v *= 0.5  # damp momentum to escape

    if verbose:
        print(f"    Phase3 (polish): score={best_score:.6f}, iters={it}")

    return best, best_score


def generate_diverse_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)

    # Cosine modulations
    for freq in [2, 3, 5, 7, 11]:
        starts.append(0.5 + 0.4 * np.cos(2 * np.pi * i * freq / n))

    # Sinusoidal absolute
    starts.append(np.abs(np.sin(2 * np.pi * i * 3.0 / n)) + 0.1)

    # Gaussian bump
    starts.append(np.exp(-0.5 * ((i - n * 0.5) / (n * 0.25)) ** 2))

    # Two bumps
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.12)) ** 2) +
                  np.exp(-0.5 * ((i - n * 0.75) / (n * 0.12)) ** 2))

    # Linear ramp + noise
    starts.append(i / n * 0.5 + 0.25 + rng.random(n) * 0.1)

    # Alternating pattern
    starts.append(np.where(np.arange(n) % 2 == 0, 0.8, 0.3))

    return [np.maximum(s, 0.01) for s in starts]


def perturb_sequence(a, rng, scale=0.1):
    """Add Gaussian perturbation and re-project."""
    n = len(a)
    S = a.sum()
    noise = rng.normal(0, scale, n)
    a_new = np.maximum(a + noise, 0)
    return simplex_project(a_new, S)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-start optimization with Nesterov GD,
    LP refinement, and perturbation cycles.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_sequence(a0, time_alloc, label=""):
        nonlocal best_seq, best_score
        if time.monotonic() > deadline - 20:
            return
        try:
            seq, score = optimize_sequence(a0, time_alloc, verbose=False)
            # Validate
            score_check = compute_obj(seq)
            if score_check < best_score:
                best_score = score_check
                best_seq = seq.copy()
                print(f"  [{label}] New best: score={best_score:.6f}, n={len(seq)}, t={time.monotonic()-t_start:.0f}s")
            else:
                print(f"  [{label}] score={score_check:.6f}, n={len(seq)} (not better)")
        except Exception as e:
            print(f"  [{label}] Error: {e}")

    print(f"=== Starting optimization, budget={budget_s}s ===")
    print(f"Target: 1.5027, Previous best: 1.510181")

    # --- Strategy 1: Start from height_sequence_1 ---
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size >= 16 and np.all(np.isfinite(hs)) and hs.sum() > 0.01:
            n_hs = len(hs)
            a_hs = np.maximum(hs, 0.0)
            t_alloc = min(300.0, (deadline - time.monotonic()) * 0.25)
            try_sequence(a_hs, t_alloc, "height_seq")

            # Perturbed versions
            for p in range(3):
                if time.monotonic() > deadline - 60:
                    break
                a_pert = perturb_sequence(a_hs, rng, scale=0.05 * (a_hs.sum() / n_hs))
                t_alloc = min(200.0, (deadline - time.monotonic()) * 0.15)
                try_sequence(a_pert, t_alloc, f"height_pert{p}")
    except NameError:
        print("  height_sequence_1 not available, using other starts.")
    except Exception as e:
        print(f"  Error with height_sequence_1: {e}")

    # --- Strategy 2: Diverse random/structured starts at various lengths ---
    lengths_to_try = [320, 352, 384, 416, 448]
    if best_seq is not None:
        # Focus on length near current best
        bn = len(best_seq)
        lengths_to_try = [bn] + [l for l in lengths_to_try if l != bn][:2]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        starts = generate_diverse_starts(rng, n_try)
        # Sort by initial score (quick eval)
        scored_starts = []
        for s in starts:
            s_proj = simplex_project(np.maximum(s, 0), np.sqrt(2.0 * n_try))
            sc = compute_obj(s_proj)
            scored_starts.append((sc, s_proj))
        scored_starts.sort(key=lambda x: x[0])

        # Try top 3 starts
        for idx, (sc, s) in enumerate(scored_starts[:3]):
            if time.monotonic() > deadline - 45:
                break
            remaining = deadline - time.monotonic() - 30
            t_alloc = min(250.0, remaining / max(1, 3 - idx + 1))
            t_alloc = max(60.0, t_alloc)
            try_sequence(s, t_alloc, f"n{n_try}_start{idx}")

    # --- Strategy 3: Perturb current best and re-optimize ---
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Perturbation phase (current best={best_score:.6f}) ===")
        for p in range(5):
            if time.monotonic() > deadline - 30:
                break
            scale = 0.03 + 0.02 * p
            a_pert = perturb_sequence(best_seq, rng, scale=scale * (best_seq.sum() / len(best_seq)))
            remaining = deadline - time.monotonic() - 20
            t_alloc = min(200.0, remaining / max(1, 5 - p))
            t_alloc = max(40.0, t_alloc)
            try_sequence(a_pert, t_alloc, f"perturb{p}_s{scale:.3f}")

    # --- Strategy 4: Final intensive optimization on best ---
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Final polish (current best={best_score:.6f}) ===")
        remaining = deadline - time.monotonic() - 10
        t_alloc = min(remaining, 300.0)
        if t_alloc > 50:
            try_sequence(best_seq, t_alloc, "final_polish")

    # --- Final validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            ev_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {ev_score:.6f}")
        except Exception as e:
            print(f"evaluate error: {e}")
        # Ensure non-negative
        best_seq = np.maximum(best_seq, 0.0)
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback to uniform sequence.")
    return [1.0] * 384
