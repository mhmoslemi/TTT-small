"""
Hybrid LP + smooth gradient optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding (from SOTA), log-sum-exp smooth gradients,
active set management, multi-phase search, and structured initializations.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix A_ub for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M, top_k=None, conv=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all (or top-k) k."""
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k and conv is not None:
        # Only use top-k peak constraints for speed
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[peaks]
        b_ub = np.full(len(peaks), M)
    else:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(num_k, M)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, step_init=1.0, top_k=None, relax=1.0,
                verbose=False, seed=0):
    """LP-based optimization on the simplex. Returns (best_a, best_obj)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    max_no_improve = 80
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        b = solve_lp_direction(a, M * relax, top_k=top_k, conv=conv)
        if b is None:
            # Try without top_k restriction
            b = solve_lp_direction(a, M * relax, top_k=None)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        b = b / b_sum

        # Try the update
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if step > 0.05:
                step *= 0.85
                if no_improve > 3:
                    step = max(0.01, step * 0.5)
            else:
                # Reduce step further or perturb
                if no_improve > 5:
                    noise = rng.standard_normal(n) * 0.005
                    a = project_simplex(a + noise)
                    step = step_init * 0.5
                    no_improve = 0
                    max_no_improve = int(max_no_improve * 1.2)
                    iter_count += 10  # artificial increment to change seed behavior
                else:
                    step = max(0.01, step * 0.9)
                    continue

        if no_improve >= max_no_improve:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            step = step_init * 0.7
            no_improve = 0
            max_no_improve = min(200, int(max_no_improve * 1.3))

        iter_count += 1
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    # Compute final objective properly
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax approximation."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Gradient of (a*a)[k] w.r.t. a[i]
    # = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n (with factor 2 when k-i=i)
    grad = np.zeros(n)
    # Use convolution structure efficiently
    # sum_k w_k * a[k-i] = (w * a)[-i] for valid i... let me just do it directly
    # For efficiency, use convolution:
    # sum_k w_k * a[k-i] for fixed i = sum_j w[j+i] * a[j] where j = k-i
    # This is cross-correlation of w and a
    # sum_k w_k * a[k-i] = (w R a)[i] where R is reversal... 
    
    # Actually: sum_k w[k] * a[k-i] = cross-correlation
    # Let's use FFT for this
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
    
    # First term: sum_k w[k] * a[k-i]
    # Let j = k-i, k = j+i: sum_j w[j+i] * a[j] (where 0<=j<n and 0<=j+i<len(w))
    # This is a cross-correlation. Use FFT:
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    
    # cross_corr[i] = sum_j w[j+i] * a[j] = ifft(w_flipped * a)[...] 
    # Actually: sum_j w[j+i]*a[j] = sum_j w[j+i]*a[j]
    # = (w * flip(a))[convolution]... let me think differently
    # sum_j w[j+i] * a[j] for j=0..n-1 (needs j+i < len(w))
    # = sum_j a[j] * w[j+i]
    # This is the cross-correlation of a and w evaluated at lag i
    # = ifft(fft(a) * conj(fft(w)))[i]
    
    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    
    # Second term: sum_k w[k] * a[i-k]
    # Let j = i-k, k = i-j: sum_j w[i-j] * a[j] (where 0<=j<n and 0<=i-j<len(w))
    # = sum_j a[j] * w[i-j]
    # This is the convolution of a and w at position i
    # = ifft(fft(a) * fft(w))[i]
    conv_wa = np.fft.ifft(A * W).real[:n]
    
    grad = cross + conv_wa
    return grad, conv


def smooth_gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.05,
                             momentum=0.9, verbose=False, seed=0):
    """Smooth gradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.8
                v *= 0.5
            if no_improve > 200:
                # Perturb and reset
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = beta_init * 0.7
                no_improve = 0

        # Slowly increase beta (sharper max approximation)
        beta = min(5000.0, beta * 1.001)

        if verbose and no_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SM iter: obj={best_obj:.6f}, beta={beta:.0f}, lr={lr:.5f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()

    # Triangular (peaked at edges)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Triangular (peaked at center)
    tri_c = np.minimum(x, n - 1 - x)[::-1] + 1.0
    tri_c = n - 1 - np.minimum(x, n - 1 - x)
    tri_c = tri_c.astype(np.float64)
    tri_c /= tri_c.sum()
    inits['triangle_c'] = tri_c

    # Gaussians of various widths
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # One-sided bump
    bump = np.exp(-x / (0.3 * n))
    bump /= bump.sum()
    inits['bump_left'] = bump
    inits['bump_right'] = bump[::-1].copy()

    # Cosine window
    cos_w = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(1)
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    # Mixture of Gaussians
    for _ in range(3):
        centers = rng.uniform(0, n, size=2)
        widths = rng.uniform(0.05, 0.3, size=2) * n
        mix = np.zeros(n)
        for c, w in zip(centers, widths):
            mix += np.exp(-0.5 * ((x - c) / w) ** 2)
        mix /= mix.sum()
        inits[f'mix_{_}'] = mix

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

    num_configs = len(lp_configs)
    total_runs = num_starts * num_configs
    time_per_run = phase2_time / max(1, total_runs)
    time_per_run = max(10.0, time_per_run)

    run_count = 0
    for i, (n_start, a_start) in enumerate(deep_starts):
        for j, cfg in enumerate(lp_configs):
            if time.monotonic() + 10 > phase2_start + phase2_time:
                break
            t_alloc = min(time_per_run, phase2_start + phase2_time - time.monotonic() - 10)
            if t_alloc < 8:
                break

            cfg = dict(cfg)
            # Adjust top_k for different lengths
            if cfg['top_k'] is not None:
                cfg['top_k'] = max(10, int(cfg['top_k'] * n_start / best_n))

            a_opt, obj_opt = lp_optimize(a_start, time_limit=t_alloc,
                                         seed=run_count * 7 + 3, verbose=False, **cfg)

            score = float(evaluate_sequence(a_opt.tolist()))
            run_count += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP [{run_count}]: n={n_start}, cfg{j}, score={score:.6f}")

    print(f"Phase 2 done. Best so far: {best_score:.6f}")

    # ========== Phase 3: Smooth Gradient Fine-tuning ==========
    print("=== Phase 3: Smooth Gradient Fine-tuning ===")
    phase3_start = time.monotonic()
    phase3_time = 300.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        grad_configs = [
            {'beta_init': 300, 'lr_init': 0.03, 'momentum': 0.9},
            {'beta_init': 800, 'lr_init': 0.02, 'momentum': 0.92},
            {'beta_init': 1500, 'lr_init': 0.01, 'momentum': 0.93},
            {'beta_init': 200, 'lr_init': 0.05, 'momentum': 0.85},
        ]

        time_per_grad = phase3_time / max(1, len(grad_configs) + 2)

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            t_alloc = min(time_per_grad, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a_opt, obj_opt = smooth_gradient_optimize(
                best_sequence, time_limit=t_alloc, seed=i * 13 + 1, **cfg)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{i}]: score={score:.6f}")

        # Also try gradient from perturbed starts
        for i in range(3):
            if time.monotonic() + 10 > phase3_start + phase3_time:
                break
            noise = rng.standard_normal(n_best) * 0.005
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(time_per_grad / 2, phase3_start + phase3_time - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a_opt, obj_opt = smooth_gradient_optimize(
                a_pert, time_limit=t_alloc, beta_init=1000, lr_init=0.02,
                momentum=0.92, seed=100 + i)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad pert [{i}]: score={score:.6f}")

    print(f"Phase 3 done. Best so far: {best_score:.6f}")

    # ========== Phase 4: Hybrid Alternating Optimization ==========
    print("=== Phase 4: Hybrid Alternating ===")
    phase4_start = time.monotonic()
    phase4_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()

        # Alternate between LP steps and gradient steps
        hybrid_cfgs = [
            {'lp_step': 0.8, 'lp_top_k': max(40, n_best // 6), 'grad_beta': 1000,
             'grad_lr': 0.02, 'grad_mom': 0.9},
            {'lp_step': 1.0, 'lp_top_k': max(60, n_best // 4), 'grad_beta': 500,
             'grad_lr': 0.03, 'grad_mom': 0.88},
            {'lp_step': 0.5, 'lp_top_k': None, 'grad_beta': 2000,
             'grad_lr': 0.01, 'grad_mom': 0.93},
        ]

        for cfg_i, hcfg in enumerate(hybrid_cfgs):
            if time.monotonic() + 60 > phase4_start + phase4_time:
                break
            t_alloc = min(phase4_time / len(hybrid_cfgs),
                          phase4_start + phase4_time - time.monotonic() - 30)
            if t_alloc < 30:
                break

            a_curr = a.copy()
            best_in_phase = a.copy()
            best_obj_phase = evaluate_obj_normalized(a)
            v = np.zeros(n_best)

            phase4_t0 = time.monotonic()
            step = hcfg['lp_step']

            while time.monotonic() - phase4_t0 < t_alloc:
                # Alternate: LP step, then gradient step
                conv = fast_convolve(a_curr)
                M = conv.max()

                # LP step
                b = solve_lp_direction(a_curr, M, top_k=hcfg['lp_top_k'], conv=conv)
                if b is not None:
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        b = b / b_sum
                        a_curr = (1.0 - step) * a_curr + step * b
                        a_curr = np.maximum(a_curr, 0.0)
                        s = a_curr.sum()
                        if s > 1e-15:
                            a_curr = a_curr / s

                # Gradient step
                grad, conv = smooth_gradient(a_curr, hcfg['grad_beta'])
                g_norm = np.linalg.norm(grad)
                if g_norm > 1e-15:
                    grad = grad / g_norm
                v = hcfg['grad_mom'] * v - hcfg['grad_lr'] * grad
                a_curr = project_simplex(a_curr + v)

                conv = fast_convolve(a_curr)
                obj = 2.0 * n_best * conv.max()

                if obj < best_obj_phase - 1e-12:
                    best_obj_phase = obj
                    best_in_phase = a_curr.copy()

            score = float(evaluate_sequence(best_in_phase.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = best_in_phase.copy()
                a = best_in_phase.copy()
                print(f"  Hybrid [{cfg_i}]: score={score:.6f}")

    print(f"Phase 4 done. Best so far: {best_score:.6f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Final Polish ===")
    phase5_start = time.monotonic()
    phase5_time = deadline - time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Very fine-tuned LP
        t_alloc = min(phase5_time * 0.4, 100.0)
        if t_alloc > 10:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                         step_init=0.3, top_k=max(50, n_best // 5),
                                         relax=1.0, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

        # Very fine-tuned gradient
        t_alloc = min(phase5_time * 0.3, 80.0)
        if t_alloc > 10:
            a_opt, obj_opt = smooth_gradient_optimize(
                best_sequence, time_limit=t_alloc,
                beta_init=3000, lr_init=0.005, momentum=0.95, seed=777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish Grad: score={score:.6f}")

        # Tiny random perturbations
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 20:
            n_tries = int(remaining / 5)
            n_tries = min(n_tries, 20)
            for i in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                for scale in [0.001, 0.003, 0.008]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Tiny perturb [{i}] s={scale}: score={score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
