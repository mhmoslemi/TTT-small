import scipy.sparse as sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

def solve_lp(f, M, rows, cols, n, x0=None):
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M)

import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

def normalize_vec(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

def lp_indices(n):
    """Precompute row/col indices for LP constraint matrix."""
    i_idx = np.repeat(np.arange(n), n)
    j_idx = np.tile(np.arange(n), n)
    return (i_idx + j_idx).astype(np.int32), i_idx.astype(np.int32)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided alternating optimization
with warm starts, adaptive M perturbation, subgradient polishing, and diverse
restarts. Focused on deep optimization from the parent sequence with supplementary

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize_vec(a):
    """Normalize non-negative vector to sum=1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix (2n-1 x n)."""
    i_idx = np.repeat(np.arange(n), n)
    j_idx = np.tile(np.arange(n), n)
    return (i_idx + j_idx).astype(np.int32), i_idx.astype(np.int32)


def solve_lp(f, M, r, c, n, tl=10):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (r, c)), shape=(2 * n - 1, n))
    res = optimize.linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                           bounds=[(0, None)] * n, method='highs',
                           options={'disp': False, 'time_limit': tl})
    if res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

            if t2b > 8:
                f2b, _ = subgrad_polish(f2, time_limit=t2b, eta0=0.015, rng=rng)
                update_best(f2b)

    # === Phase 3: Diverse LP starts at n=512 ===
    print("\n=== Phase 3: Diverse starts ===")
    n3 = 512
    x = np.arange(n3, dtype=np.float64)
    diverse_inits = []

    # Gaussian shapes
    for sf in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n3) / (sf * n3)) ** 2)
        diverse_inits.append(g / g.sum())

    # Cosine shape
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n3))
    diverse_inits.append(cb / cb.sum())

    # Bimodal
    bi = np.zeros(n3)
    w = n3 // 5
    bi[:w] = 1.0
    bi[-w:] = 1.0
    diverse_inits.append(bi / bi.sum())

    # Perturbed parent
    if parent is not None and len(parent) == n3:
        for sc in [0.02, 0.05]:
            pp = normalize_vec(parent + sc * rng.standard_normal(n3))
            diverse_inits.append(pp)

    rem = deadline - time.monotonic()
    if rem > 60:
        per_init = min(120.0, rem / (len(diverse_inits) + 1))
        for idx, init in enumerate(diverse_inits):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            ta = min(per_init, rem - 15)
            if ta < 15:
                break
            print(f"  Diverse init {idx+1}/{len(diverse_inits)}, {ta:.0f}s")
            fa, _ = lp_optimize(init, time_limit=ta, rng=rng)
            update_best(fa)

    # === Phase 4: Final polishing ===
    if best_seq is not None:
        print("\n=== Phase 4: Final polishing ===")
        rem = deadline - time.monotonic()
        if rem > 15:
            # Subgradient with very small eta
            t4a = min(60.0, rem * 0.3)
            if t4a > 5:
                f4a, _ = subgrad_polish(best_seq, time_limit=t4a, eta0=0.005, rng=rng)
                update_best(f4a)

            # Mirror with high temperature
            rem = deadline - time.monotonic()
            t4b = min(50.0, rem * 0.3)
            if t4b > 5:
                f4b, _ = mirror_polish(best_seq, time_limit=t4b, t_param=150.0, eta0=0.08)
                update_best(f4b)

            # One more LP pass
            rem = deadline - time.monotonic()
            t4c = min(80.0, rem * 0.4)
            if t4c > 10:
                f4c, _ = lp_optimize(best_seq, time_limit=t4c, rng=rng)
                update_best(f4c)

            # Final subgradient
            rem = deadline - time.monotonic()
            t4d = min(40.0, rem * 0.5)
            if t4d > 5:
                f4d, _ = subgrad_polish(best_seq, time_limit=t4d, eta0=0.003, rng=rng)
                update_best(f4d)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"  FALLBACK: score={best_score:.6f}")

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
