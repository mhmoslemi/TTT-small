"""
Multi-phase optimizer for non-negative low-autocorrelation sequences.
Combines efficient coordinate descent (with O(n) convolution updates),
LP-guided direction finding, and simulated annealing for escaping local minima.
Key innovation: incremental convolution updates avoid redundant FFTs.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def coordinate_descent(f_init, time_limit, t_start=None, verbose=False):
    """
    Coordinate descent with golden-section 1D subproblem solving.
    Uses O(n) incremental convolution updates per coordinate.
    """
    if t_start is None:
        t_start = time.monotonic()

    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    c = fft_convolve(f, f).copy()
    S = float(np.sum(f))
    best_score = 2.0 * n * np.max(c) / (S * S)
    best_f = f.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    cycle = 0
    no_improve = 0

    while time.monotonic() - t_start < time_limit:
        if time.monotonic() - t_start > time_limit - 2.0:
            break

        cycle += 1
        cycle_improved = False

        # Periodically recompute convolution via FFT to prevent drift
        if cycle % 10 == 0:
            c = fft_convolve(f, f).copy()
            S = float(np.sum(f))

        for i in range(n):
            if time.monotonic() - t_start > time_limit - 1.0:
                break

            old_val = f[i]
            if old_val < 1e-8:
                # Try setting to a small positive value
                x_try = max(0.1 * S / n, 0.01)
                f_corr = np.zeros(2 * n - 1)
                for k in range(i, min(i + n, 2 * n - 1)):
                    f_corr[k] = f[k - i]
                E = np.zeros(2 * n - 1)
                if 2 * i < 2 * n - 1:
                    E[2 * i] = 1.0
                c_test = c + 2.0 * x_try * f_corr + x_try * x_try * E
                S_test = S + x_try
                s_test = 2.0 * n * np.max(c_test) / (S_test * S_test)
                s_curr = 2.0 * n * np.max(c) / (S * S)
                if s_test < s_curr - 1e-10:
                    c = c_test
                    f[i] = x_try
                    S = S_test
                    if s_test < best_score:
                        best_score = s_test
                        best_f = f.copy()
                        cycle_improved = True
                continue

            # Compute correction vectors for changing f[i] from old_val to x
            f_corr = np.zeros(2 * n - 1)
            for k in range(i, min(i + n, 2 * n - 1)):
                f_corr[k] = f[k - i]

            E = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                E[2 * i] = 1.0

            # c_new[x][k] = c[k] + 2*(x - old_val)*f_corr[k] + (x^2 - old_val^2)*E[k]
            #             = (c[k] - 2*old_val*f_corr[k] - old_val^2*E[k]) + 2*x*f_corr[k] + x^2*E[k]
            base = c - 2.0 * old_val * f_corr - old_val * old_val * E
            S_base = S - old_val

            # Search range for x
            x_max = max(old_val * 4.0, 3.0 * S_base / max(1, n), 0.5)
            x_max = min(x_max, 500.0)
            x_min = 0.0

            def obj_1d(x):
                fk = base + 2.0 * x * f_corr + x * x * E
                Sx = S_base + x
                if Sx < 0.01:
                    return 1e10
                m = np.max(fk)
                return 2.0 * n * m / (Sx * Sx)

            # Golden section search
            lo, hi = x_min, x_max
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = obj_1d(x1)
            f2 = obj_1d(x2)

            for _ in range(20):
                if f1 < f2:
                    hi = x2
                    x2, f2 = x1, f1
                    x1 = hi - gr * (hi - lo)
                    f1 = obj_1d(x1)
                else:
                    lo = x1
                    x1, f1 = x2, f2
                    x2 = lo + gr * (hi - lo)
                    f2 = obj_1d(x2)

            x_opt = (lo + hi) / 2.0

            # Update c, f, S
            delta = x_opt - old_val
            c += 2.0 * delta * f_corr + delta * (x_opt + old_val) * E
            f[i] = x_opt
            S = S_base + x_opt

            curr_score = 2.0 * n * np.max(c) / (S * S)
            if curr_score < best_score - 1e-12:

                best_f = f.copy()
                cycle_improved = True

        if cycle_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                # Perturb to escape local minimum
                rng = np.random.default_rng(cycle * 31 + 7)
                f += rng.normal(0, 0.01 * max(1.0, np.mean(f)), size=n)
                f = np.clip(f, 0.01, 1000.0)
                c = fft_convolve(f, f).copy()
                S = float(np.sum(f))
                no_improve = 0

        if verbose and cycle % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"        CD [{elapsed:6.1f}s] cycle={cycle:4d} best={best_score:.6f}")

    return best_f, best_score


def lp_optimize(f_init, time_limit, t_start=None, verbose=False):
    """LP-guided optimization with line search."""
    if t_start is None:
        t_start = time.monotonic()

    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    it = 0
    stall = 0

    while time.monotonic() - t_start < time_limit:
        if time.monotonic() - t_start > time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))
        S = float(np.sum(f))

        b = solve_lp_direction(f, M * 1.01, timeout=3.0)
        if b is None:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        # Line search
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.1, 0.05, 0.02, 0.08, 0.15]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"        LP [{elapsed:6.1f}s] it={it:3d} score={best_s:.6f} best={best_score:.6f}")

        if stall >= 50:
            rng = np.random.default_rng(100 + it * 13)
            f += 0.02 * rng.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0

    return best_f, best_score


def anneal_refine(f_init, time_limit, t_start=None):
    """Simulated annealing with peak-targeted mutations."""
    if t_start is None:
        t_start = time.monotonic()

    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    rng = np.random.default_rng(777)
    temp = 0.003
    cooling = 0.9997
    it = 0

    while time.monotonic() - t_start < time_limit:
        if time.monotonic() - t_start > time_limit - 1.0:
            break

        cand = f.copy()
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))

        r = rng.random()
        if r < 0.35:
            # Reduce contributors to the peak
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            top_k = max(2, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            cand[top_idx] *= np.exp(rng.normal(-0.05, 0.03, size=len(top_idx)))
            # Increase non-peak areas
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = max(3, len(other) // 8)
                inc_idx = rng.choice(other, size=min(n_inc, len(other)), replace=False)
                cand[inc_idx] *= np.exp(rng.normal(0.05, 0.03, size=len(inc_idx)))
        elif r < 0.6:
            # Smooth perturbation
            width = max(2, int(rng.integers(1, n // 6)))
            start = int(rng.integers(0, max(1, n - width)))
            cand[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        elif r < 0.8:
            # Global scaling perturbation
            cand *= np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            # Swap-based: move mass from high to low
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
            if src != dst:
                amt = 0.1 * f[src]
                cand[src] = max(0.0, f[src] - amt)
                cand[dst] = f[dst] + amt

        cand = np.clip(cand, 0.01, 1000.0)
        new_score = score_fast(cand)

        if new_score < best_score:
            best_score = new_score
            best_f = cand.copy()
            f = cand
        elif rng.random() < np.exp(-(new_score - score_fast(f)) / max(temp, 1e-15)):
            f = cand

        temp *= cooling
        it += 1

    return best_f, best_score


def make_hann(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bartlett(n, rng):
    t = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.2 + 0.8 * w
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t * t / (3.0 * n))
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_smooth(n, rng):
    """Random but smooth sequence."""
    n_ctrl = max(4, n // 16)
    ctrl = rng.uniform(0.3, 1.0, size=n_ctrl)
    t_ctrl = np.linspace(0, 1, n_ctrl)
    t = np.linspace(0, 1, n)
    a = np.interp(t, t_ctrl, ctrl)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

        f0, s0, n = task_results[idx]
        t_alloc = min(100.0, remaining * 0.4)

        print(f"  LP on candidate {idx} (n={n}, score={s0:.6f}), t={t_alloc:.0f}s")
        f_lp, s_lp = lp_optimize(f0, t_alloc, t_start=time.monotonic(), verbose=False)
        print(f"  -> LP score={s_lp:.6f}")

        if s_lp < best_score:
            best_score = s_lp
            best_seq = f_lp.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    print(f"\n=== Phase 3: Annealing on best candidate ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 20.0:
            t_alloc = min(200.0, remaining * 0.6)
            print(f"  Annealing (n={len(best_seq)}, score={best_score:.6f}), t={t_alloc:.0f}s")
            f_ann, s_ann = anneal_refine(best_seq, t_alloc, t_start=time.monotonic())
            print(f"  -> Anneal score={s_ann:.6f}")
            if s_ann < best_score:
                best_score = s_ann
                best_seq = f_ann.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    print(f"\n=== Phase 4: Final coordinate descent polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10.0:
            t_alloc = min(150.0, remaining * 0.8)
            print(f"  Final CD (n={len(best_seq)}), t={t_alloc:.0f}s")
            f_fin, s_fin = coordinate_descent(best_seq, t_alloc, t_start=time.monotonic(), verbose=True)
            print(f"  -> Final score={s_fin:.6f}")
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 5: Try interpolation to different n values
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20.0:
            print(f"\n=== Phase 5: Try other n values via interpolation ===")
            n_orig = len(best_seq)
            for n_try in [n_orig * 2, n_orig // 2, n_orig + 64, n_orig - 64]:
                if n_try < 32 or n_try > 2048 or remaining < 10.0:
                    continue
                x_new = np.linspace(0, 1, n_try)
                x_old = np.linspace(0, 1, n_orig)
                f_try = np.interp(x_new, x_old, best_seq)
                f_try = np.clip(f_try, 0.01, None)
                t_alloc = min(30.0, remaining * 0.4)
                f_r, s_r = coordinate_descent(f_try, t_alloc, t_start=time.monotonic(), verbose=False)
                print(f"  n={n_try}: score={s_r:.6f}")
                if s_r < best_score:
                    best_score = s_r
                    best_seq = f_r.copy()
                    print(f"  *** NEW BEST: {best_score:.6f} ***")
                remaining = deadline - time.monotonic() - 10.0

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    return [1.0] * 256
