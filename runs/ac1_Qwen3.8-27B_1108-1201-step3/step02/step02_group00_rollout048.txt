```python
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient-based optimization
with log-sum-exp approximation, L-BFGS in softmax parameterization, multi-phase
search over lengths and initializations, and dual-guided direction refinement.
Key improvements: proper smooth gradients (not subgradients), quasi-Newton
optimization via scipy, adaptive temperature annealing, and active-set tracking.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def smooth_obj_and_grad_a(a, T):
    """
    Compute smooth objective and gradient w.r.t. a (on simplex, sum(a)=1).
    f(a) = 2n * LSE_T(conv(a,a)) where LSE_T(v) = (1/T)*log(sum exp(T*v))
    """
    n = len(a)
    conv = fast_convolve(a)
    c_max = conv.max()
    if c_max < 1e-30:
        return 0.0, np.zeros_like(a)

    # Softmax weights for the log-sum-exp
    shifted = T * (conv - c_max)
    # All values <= 0, so numerically stable
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        w = exp_shifted / w_sum

    # Objective: 2n * (c_max + log(w_sum) / T)
    obj = 2.0 * n * (c_max + np.log(w_sum) / T)

    # Gradient: 2n * sum_k w[k] * d(conv_k)/da
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += wk * a[k - idx]

    grad *= 2.0 * n
    return obj, grad


def softmax_obj(x, T, n):
    """Objective in softmax parameterization."""
    a = softmax(x)
    obj, _ = smooth_obj_and_grad_a(a, T)
    return obj


def softmax_grad(x, T, n):
    """Gradient in softmax parameterization: g_x = a * (g_a - a^T g_a)."""
    a = softmax(x)
    _, grad_a = smooth_obj_and_grad_a(a, T)
    grad_x = a * (grad_a - np.dot(a, grad_a))
    return grad_x


def optimize_lBFGS_softmax(a0, T, time_limit, max_iter=500):
    """
    Optimize using L-BFGS-B on the softmax-parameterized smooth objective.
    Returns (best_a_normalized, best_objective).
    """
    n = len(a0)
    a0 = np.maximum(a0, 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s

    # Initialize x from a0: x_i = log(a_i) with a floor
    a_floor = 1e-30
    x0 = np.log(np.maximum(a0, a_floor))

    # Add small random perturbation to break symmetry
    x0 += 0.01 * np.random.default_rng(42).standard_normal(n)

    best_a = a0.copy()
    best_obj = float('inf')

    t0 = time.monotonic()

    def fun(x):
        if time.monotonic() - t0 > time_limit:
            raise ValueError("Time limit exceeded")
        return softmax_obj(x, T, n)

    def jac(x):
        if time.monotonic() - t0 > time_limit:
            raise ValueError("Time limit exceeded")
        return softmax_grad(x, T, n)

    # Add callback to track best
    state = {'best_a': best_a, 'best_obj': best_obj, 'count': 0}

    def callback(xk):
        if time.monotonic() - t0 > time_limit:
            raise ValueError("Time limit exceeded")
        a = softmax(xk)
        obj, _ = smooth_obj_and_grad_a(a, T)
        if obj < state['best_obj']:
            state['best_obj'] = obj
            state['best_a'] = a.copy()
        state['count'] += 1
        return time.monotonic() - t0 > time_limit

    try:
        result = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': 2000,
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'callback': callback})
    except (ValueError, Exception):
        pass

    a_final = softmax(result.x if hasattr(result, 'x') else x0)
    obj_final, _ = smooth_obj_and_grad_a(a_final, T)
    if obj_final < state['best_obj']:
        state['best_obj'] = obj_final
        state['best_a'] = a_final.copy()

    return state['best_a'], state['best_obj']


def optimize_projected_smooth(a0, T, time_limit, lr_init=0.01, momentum_init=0.7):
    """
    Projected gradient descent with momentum on the simplex.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    best_obj = float('inf')
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    mu = momentum_init

    while time.monotonic() - t0 < time_limit:
        obj, grad = smooth_obj_and_grad_a(a, T)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Momentum update
        v = mu * v - lr * grad
        a_new = project_simplex(a + v)

        # Backtracking line search (simplified: just accept if improved)
        obj_new, _ = smooth_obj_and_grad_a(a_new, T)
        if obj_new > obj:
            # Try smaller step
            a_new = project_simplex(a - 0.5 * lr * grad)
            obj_new, _ = smooth_obj_and_grad_a(a_new, T)
            if obj_new > obj:
                a_new = a  # reject step
                obj_new = obj
                v *= 0.5
            else:
                v = 0.5 * v
        a = a_new
        obj = obj_new

        # Schedule
        lr *= 0.999
        mu = min(0.95, mu + 0.001)

    return best_a, best_obj


def optimize_active_set(a0, time_limit, T_base=50.0):
    """
    Active-set method: identify which conv index is maximal, then
    minimize that single quadratic using L-BFGS, periodically re-checking.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]
        obj = 2.0 * n * max_val

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Build the quadratic for this k_star
        # Minimize a^T Q a where Q[j,l] = 1 if j+l=k_star (off-diag) or 2j=k_star (diag)
        # i.e., conv_k_star = sum_j a[j]*a[k_star-j]

        # Use L-BFGS to minimize this single quadratic on the simplex
        # via softmax parameterization
        a_nondegenerate = np.maximum(a, 1e-30)
        x0 = np.log(a_nondegenerate / a_nondegenerate.sum())

        def quad_fun(x):
            if time.monotonic() - t0 > time_limit:
                raise ValueError("timeout")
            aa = softmax(x)
            val = 0.0
            for j in range(n):
                lj = k_star - j
                if 0 <= lj < n:
                    val += aa[j] * aa[lj]
            return 2.0 * n * val

        def quad_grad(x):
            if time.monotonic() - t0 > time_limit:
                raise ValueError("timeout")
            aa = softmax(x)
            g = np.zeros(n)
            for j in range(n):
                lj = k_star - j
                if 0 <= lj < n:
                    g[j] += aa[lj]
            g *= 2.0 * n
            return aa * (g - np.dot(aa, g))

        def quad_cb(xk):
            if time.monotonic() - t0 > time_limit:
                raise ValueError("timeout")
            return time.monotonic() - t0 > time_limit - 5

        try:
            res = minimize(quad_fun, x0, jac=quad_grad, method='L-BFGS-B',
                           options={'maxiter': 100, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'callback': quad_cb})
            a_new = softmax(res.x)
        except (ValueError, Exception):
            break

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Mix to avoid getting stuck
        a = 0.7 * a_new + 0.3 * a
        a = project_simplex(a)

        # Check if the active constraint changed
        conv_check = fast_convolve(a)
        if int(np.argmax(conv_check)) != k_star:
            # Active set changed, re-identify
            pass

    return best_a, best_obj


def lp_direction(a, M, n, max_iter=100):
    """
    Find direction b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0.
    Uses simple projected gradient on the dual.
    """
    # We want max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    # Dual: min M * sum(y) s.t. a^T M_y a >= 1, y >= 0 (roughly)
    # This is complex, so use a simpler approach:
    # Iteratively reduce b where conv(a,b) is high
    b = np.ones(n) / n
    t0 = time.monotonic()

    for _ in range(min(max_iter, 200)):
        if time.monotonic() - t0 > 1.0:
            break
        conv_ab = np.convolve(a, b, mode='full')
        # Find violations
        violations = np.maximum(0, conv_ab - M)
        # Gradient of violation w.r.t. b
        grad = np.zeros(n)
        for k in range(len(conv_ab)):
            if violations[k] > 0:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += violations[k] * a[k - idx]
        # Update: increase b where gradient is low, decrease where high
        b = b - 0.01 * (grad - grad.mean())
        b = np.maximum(b, 0)
        bs = b.sum()
        if bs > 1e-10:
            b = b / bs

    return b


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.1, 0.15, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    # Bimodal (edges)
    edge = max(1, n // 3)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Periodic
    for period in [max(2, n // 8), max(2, n // 4), max(2, n // 2)]:
        per = np.zeros(n)
        per[::period] = 1.0
        per /= per.sum()
        inits[f'period_{period}'] = per

    # Random Dirichlet with different concentrations
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        inits[f'dirichlet_{alpha}'] = rng.dirichlet(np.ones(n) * alpha)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def multi_T_anneal(a0, time_limit, T_values=None):

    a = a0.copy()
    n = len(a)
    t0 = time.monotonic()

    best_a = a.copy()

    for T in T_values:

        if obj_opt < best_obj:

            top_configs.append((n, a_opt))
            seen_keys.add(key)
        if len(top_configs) >= 10:
            break

    # Always include current best
    if best_sequence is not None:
        if (len(best_sequence),) not in [(c[0],) for c in top_configs]:
            top_configs.append((len(best_sequence), best_sequence.copy()))

    # Add some at the best length
    if best_sequence is not None:
        best_n = len(best_sequence)
        inits_b = generate_initializations(best_n, parent, rng)
        for extra in ['dirichlet_0.1', 'random', 'gauss_0.15']:
            if extra in inits_b:
                top_configs.append((best_n, inits_b[extra]))

    # Also try a few larger lengths
    for n_large in [2048]:
        inits_l = generate_initializations(n_large, parent, rng)
        for nm in ['uniform', 'gauss_0.3', 'dirichlet_1.0']:
            if nm in inits_l:
                top_configs.append((n_large, inits_l[nm]))

    t_per_deep = max(20.0, (phase2_deadline - t_phase2_start - 30) / len(top_configs))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(t_per_deep, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break

        print(f"  Deep {i+1}/{len(top_configs)}: n={n}, t_alloc={t_alloc:.0f}s")

        # T annealing with L-BFGS
        a_opt, obj_opt = multi_T_anneal(a0, t_alloc * 0.6,
                                        T_values=[20.0, 50.0, 150.0, 500.0, 2000.0])
        update_best(a_opt, f"anneal n={n}")

        # Active set method
        if time.monotonic() < phase2_deadline - 30:
            a_opt2, obj_opt2 = optimize_active_set(a_opt, time_limit=t_alloc * 0.3)
            update_best(a_opt2, f"active n={n}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # === Phase 3: Fine polishing ===
    print("=== Phase 3: Fine Polishing ===")
    t_phase3_start = time.monotonic()
    phase3_deadline = deadline

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Very high T L-BFGS
        for T in [5000.0, 20000.0, 50000.0]:
            if time.monotonic() > phase3_deadline - 20:
                break
            t_alloc = min(60.0, phase3_deadline - time.monotonic() - 15)
            if t_alloc < 5:
                break
            a_opt, obj_opt = optimize_lBFGS_softmax(best_sequence, T, t_alloc, max_iter=500)
            update_best(a_opt, f"polish T={T}")

        # Projected gradient with very small lr
        if time.monotonic() < phase3_deadline - 20:
            t_alloc = min(60.0, phase3_deadline - time.monotonic() - 15)
            a_opt, obj_opt = optimize_projected_smooth(best_sequence, T=10000.0,
                                                       time_limit=t_alloc, lr_init=0.005)
            update_best(a_opt, "polish proj")

        # Active set final
        if time.monotonic() < phase3_deadline - 20:
            t_alloc = min(60.0, phase3_deadline - time.monotonic() - 15)
            a_opt, obj_opt = optimize_active_set(best_sequence, time_limit=t_alloc)
            update_best(a_opt, "polish active")

        # Random perturbation + re-optimize
        for attempt in range(5):
            if time.monotonic() > phase3_deadline - 15:
                break
            scale = 0.005 / (attempt + 1)
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(40.0, phase3_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            a_opt, obj_opt = optimize_lBFGS_softmax(a_pert, T=1000.0,
                                                    time_limit=t_alloc, max_iter=300)
            update_best(a_opt, f"perturb {attempt}")

    # === Phase 4: Try different lengths if time remains ===
    print("=== Phase 4: Length Exploration ===")
    if time.monotonic() < deadline - 60:
        # Try lengths near the best that we haven't fully explored
        if best_sequence is not None:
            n_best = len(best_sequence)
            nearby = [n_best // 2, n_best * 2, n_best * 3 // 2, n_best * 5 // 4]
            nearby = [n for n in nearby if 64 <= n <= 4096 and n != n_best]
            for n_try in nearby:
                if time.monotonic() > deadline - 30:
                    break
                inits_t = generate_initializations(n_try, parent, rng)
                if 'uniform' in inits_t:
                    a0 = inits_t['uniform']
                    # Also try parent-derived
                    if 'parent' in inits_t:
                        a0 = inits_t['parent']
                    elif 'parent_ds' in inits_t:
                        a0 = inits_t['parent_ds']
                    t_alloc = min(80.0, deadline - time.monotonic() - 20)
                    if t_alloc > 10:
                        a_opt, _ = multi_T_anneal(a0, t_alloc,
                                                  T_values=[50.0, 200.0, 1000.0, 5000.0])
                        update_best(a_opt, f"length n={n_try}")

    # Final verification
    print(f"\nFinal best score: {best_score:.6f}")
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
