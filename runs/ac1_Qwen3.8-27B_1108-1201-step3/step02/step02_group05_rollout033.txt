```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using surrogate gradient descent
with FFT-based gradient computation and LP-based polishing.
Uses multiple sequence lengths and diverse initializations.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_surrogate_gradient(a, p=4):
    """
    Compute gradient of F(a) = sum_k (conv(a,a)[k])^p.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    C = fast_convolve(a)  # length 2n-1
    C = np.maximum(C, 1e-30)  # numerical safety
    
    D = C ** (p - 1)  # length 2n-1
    
    # We need: grad[i] = 2p * sum_{m=0}^{n-1} a[m] * D[m+i]
    # This is the cross-correlation of a with D, evaluated at lags 0, 1, ..., n-1
    # (specifically, the "anti-diagonal" of the correlation matrix)
    
    # Using FFT: corr[k] = ifft(conj(FFT(a)) * FFT(D))[k] = sum_m a[m] * D[m-k] (circular)
    # With sufficient zero-padding, this becomes linear correlation.
    # We want sum_m a[m] * D[m+i] = corr[-i] = corr[fft_size - i] (for i > 0)
    # and corr[0] for i = 0.
    
    fft_size = 1 << (2 * n - 1 + n - 1).bit_length()  # >= 3n-2
    A = np.fft.rfft(a, fft_size)
    Df = np.fft.rfft(D, fft_size)
    corr = np.fft.irfft(np.conj(A) * Df, fft_size)
    
    # corr[k] = sum_m a[m] * D[m-k] (linear, due to padding)
    # We want grad[i] = 2p * sum_m a[m] * D[m+i] = 2p * corr[-i mod fft_size]
    # = 2p * corr[fft_size - i] for i >= 1, and 2p * corr[0] for i = 0
    
    grad = np.zeros(n)
    grad[0] = corr[0]
    if n > 1:
        grad[1:] = corr[fft_size - 1 : fft_size - n : -1]
    
    return 2.0 * p * grad


def surrogate_gradient_descent(a0, time_limit, p=4, verbose=False):
    """
    Minimize sum(C[k]^p) where C = conv(a,a), subject to sum(a)=1, a>=0.
    Uses Adam optimizer with projection onto the simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit:
        if t % 10 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= max_no_improve:
                # Restart with smaller learning rate
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
        
        # Compute gradient
        grad = compute_surrogate_gradient(a, p=p)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex: a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a_new / s
        
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
    
    # Final check
    obj = fast_convolve(a).max()
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Surrogate gradient descent (sum of p-th powers of convolution) with Adam optimizer
2. LP-based directional optimization (Frank-Wolfe style)
3. Perturbation-based restarts to escape local minima
4. Multiple sequence lengths and diverse initializations
Each gradient step is O(n log n) via FFT, enabling thousands of iterations.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_surrogate_gradient(a, p=4):
    """
    Compute gradient of F(a) = sum_k (conv(a,a)[k])^p w.r.t. a.
    Uses FFT for O(n log n) computation.
    grad[i] = 2p * sum_{m=0}^{n-1} a[m] * C[m+i]^(p-1)
    """
    n = len(a)
    C = fast_convolve(a)
    C_safe = np.maximum(C, 1e-30)
    D = C_safe ** (p - 1)

    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Df = np.fft.rfft(D, fft_size)
    corr = np.fft.irfft(np.conj(A) * Df, fft_size)

    grad = np.zeros(n)
    grad[0] = corr[0]
    if n > 1:
        grad[1:] = corr[fft_size - 1: fft_size - n: -1]

    return 2.0 * p * grad


def compute_max_gradient(a):
    """
    Compute subgradient of max_k C[k] where C = conv(a,a).
    Returns the subgradient w.r.t. a (2*a[k*-i] for the argmax k*).
    """
    n = len(a)
    C = fast_convolve(a)
    k_star = int(np.argmax(C))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s


def surrogate_gd(a0, time_limit, p=4, lr_init=0.003, verbose=False):
    """
    Surrogate gradient descent with Adam to minimize sum(C[k]^p).
    Constraint: sum(a)=1, a>=0 enforced via projection.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    t0 = time.monotonic()
    no_improve = 0
    check_every = 20

    while time.monotonic() - t0 < time_limit:
        if t % check_every == 0 and t > 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 15:
                lr *= 0.6
                m *= 0.5
                v *= 0.5
                no_improve = 0
                if lr < 1e-10:
                    break

        grad = compute_surrogate_gradient(a, p=p)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

    obj = fast_convolve(a).max()
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    return best_a, best_obj


def surrogate_gd_mirror(a0, time_limit, p=4, lr_init=0.01, verbose=False):
    """
    Mirror descent (exponential family) on the surrogate objective.
    Update: a[i] <- a[i] * exp(-lr * grad[i]) / Z
    Naturally maintains positivity.
    """
    n = len(a0)
    log_a = np.log(np.maximum(a0.astype(np.float64), 1e-30))
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    t = 0
    check_every = 50

    while time.monotonic() - t0 < time_limit:
        if t % check_every == 0 and t > 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 10:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-10:
                    break

        grad = compute_surrogate_gradient(a, p=p)
        t += 1

        # Mirror descent step in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= lr * grad / np.max(np.abs(grad) + 1e-15)
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a = project_simplex(a_new)

    obj = fast_convolve(a).max()
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Find g >= 0, sum(g)=1 minimizing max_k conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit, max_iters=200):
    """LP-based iterative optimization (Frank-Wolfe style)."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    f = project_simplex(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    no_improve = 0

    for _ in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        C = fast_convolve(f)
        M = C.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(f, M)

        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            # Line search: try multiple step sizes
            best_t = 0.0
            best_obj_t = best_obj
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_t = (1 - t) * f + t * g
                a_t = project_simplex(a_t)
                obj_t = fast_convolve(a_t).max()
                if obj_t < best_obj_t - 1e-12:
                    best_obj_t = obj_t
                    best_t = t

            if best_t > 0:
                f = (1 - best_t) * f + best_t * g
                f = project_simplex(f)
                if best_obj_t < best_obj - 1e-12:
                    best_obj = best_obj_t
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 40:
            break

    return best_f, best_obj


def local_search(a0, time_limit):
    """Targeted local search: move mass from peak-contributing to safe indices."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = project_simplex(a)
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < 200:
        C = fast_convolve(a)
        k_star = int(np.argmax(C))

        if max_c < 1e-15:
            break

        thr_high = 0.5 * max_c
        thr_low = 0.3 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        for di in danger[:10]:
            if found:
                break
            if a[di] < 1e-10:
                continue
            for si in safe[:10]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.15, 0.3, 0.5, 0.7]:
                    amount = a[di] * frac
                    if amount < 1e-12:

                    a_trial = a.copy()
                    a_trial[di] -= amount

                    a_trial = project_simplex(a_trial)
                    o = fast_convolve(a_trial).max()
                    if o < best_obj - 1e-11:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        stuck = 0 if found else stuck + 1

    return a, best_obj

def make_gaussian(n, sigma):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma * n)) ** 2)
    g = np.maximum(g, 1e-10)
    return g / g.sum()


def make_hann(n):
    """Hann window initialization."""
    if n <= 1:
        return np.ones(n)
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    h = np.maximum(h, 1e-10)
    return h / h.sum()


def make_random_dirichlet(n, rng):
    """Random Dirichlet initialization."""
    return rng.dirichlet(np.ones(n) * 0.5)


def perturbed_best(a, rng, scale=0.05):
    """Create a perturbed version of the best sequence."""
    a_pert = a.copy()
    a_pert += rng.normal(0, scale, size=len(a))
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s

    # Ensure all values are valid for evaluate_sequence
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
