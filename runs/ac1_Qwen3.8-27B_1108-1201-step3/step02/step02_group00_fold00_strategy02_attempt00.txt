def generate_spectral_sequence(n, parent_ratio=0.85, initial_sun=1.0):

        base_sequence = np.roll(base_sequence, parent_rotation)
        base_sequence = base_sequence * (initial_sun / np.sum(base_sequence))

        factors = factors * (1-0.45*|i|/40)**2
        factors = factors[range(n)]
    return base_sequence
