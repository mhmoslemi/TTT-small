"""
Multi-phase optimization for minimizing normalized peak self-convolution
of non-negative step functions. Uses LP-based alternating optimization,
gradient polishing with adaptive soft-max temperature, and diverse
multi-start initialization with crossover.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_constraint_A(f, n):
    """Build constraint matrix for LP: A[k,j] = f[j] if 0 <= k-j < n."""
    # Actually, the constraint is: for each k in 0..2n-2, sum_j g[j] * f[k-j] <= M
    # where f[k-j] is valid when 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in c[k] for k = j, j+1, ..., j+n-1
        # c[k] = sum_i a[i] * a[k-i], and g[j] plays the role of a[j]
        # So c[k] includes g[j] * f[k-j] when 0 <= k-j < n
        # i.e., k-j in [0, n-1], so k in [j, j+n-1]
        A[j:j+n, j] = f
    return A

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. sum_j g[j]*f[k-j] <= M for all k, g >= 0."""
    A = build_constraint_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 20})
    if result.success:
        return result.x
    return None

def lp_optimize(f_init, time_limit, t_init=0.4, verbose=False):
    """LP-based iterative optimization with adaptive step size and backtracking."""
    f = f_init.astype(np.float64).copy()

    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)

        if g is None:

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        # Backtracking line search
        accepted = False
        for trial_t in [t, t*0.7, t*0.5, t*0.3, t*0.1]:
            f_new = (1.0 - trial_t) * f + trial_t * g * (sum_f / sum_g)
            f_new = np.clip(f_new, 0.0, None)
            sum_new = np.sum(f_new)
            if sum_new < 1e-10:

            f_new = f_new * (sum_f / sum_new)
            score_new = local_score(f_new)
            if score_new <= best_score + 1e-10:
                f = f_new
                t = trial_t

    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))

    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 1.0
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    
    # Adaptive temperature: start high, decrease
    tau_max = 20.0 / n
    tau_min = 0.1 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Nonlinear temperature schedule
        tau = tau_max * (tau_min / tau_max) ** (elapsed_frac ** 0.7)

        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        
        # Soft-max gradient
        shifted = (conv - conv_max) / tau
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50, 0)
        exp_conv = np.exp(shifted)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp

        # Gradient of soft_max(conv(a,a)) w.r.t. a
        # d/d a_i [sum_k w_k * conv(a,a)[k]] where w_k = exp((conv[k]-max)/tau) / sum
        # conv(a,a)[k] = sum_j a[j] * a[k-j]
        # d/d a_i conv[k] = a[k-i] (when valid) + a[i] (when k-i valid, i.e., the conjugate term)
        # Actually for self-convolution: d/d a_i sum_k w_k * c[k] = sum_k w_k * (a[k-i] + a[i] if applicable)
        # More precisely: c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n... wait)
        # c[k] = sum_j a[j] * a[k-j]. dc[k]/da[i] = a[k-i] (from j=i term) + a[i] (from k-j=i, i.e., j=k-i term)
        # But when i = k-i, i.e., k=2i, it's 2*a[i].
        # Let me just compute the gradient as: grad[i] = sum_k w_k * (a[k-i] if 0<=k-i<n else 0)
        # But this is only half. The full gradient is: grad[i] = sum_k w_k * (a[k-i] * I(0<=k-i<n) + a[i] * I(0<=k-i<n))
        # Wait, that's not right either. Let me be more careful.
        # c[k] = sum_j a[j] * a[k-j]
        # dc[k]/da[i] = a[k-i] (when 0 <= k-i <= n-1, from the term j=i)
        #              + a[i] (when 0 <= k-i <= n-1, from the term k-j=i, i.e., j=k-i)
        # But wait, when j=i and k-j=i, we need i = k-i, i.e., k=2i. In that case, the term is a[i]*a[i], and the derivative is 2*a[i].
        # So dc[k]/da[i] = a[k-i] * I(0<=k-i<n) + a[i] * I(0<=k-i<n)
        # = (a[k-i] + a[i]) * I(0<=k-i<n) when k != 2i
        # = 2*a[i] when k = 2i (which is consistent since a[k-i] = a[i])
        # Actually it's always (a[k-i] + a[i]) * I(0<=k-i<n), and when k=2i this gives 2*a[i].
        
        # So grad[i] = sum_k w[k] * (a[k-i] + a[i]) * I(0<=k-i<n)
        #            = a[i] * sum_k w[k] * I(0<=k-i<n) + sum_k w[k] * a[k-i] * I(0<=k-i<n)
        #            = a[i] * (sum of w[k] for k in [i, i+n-1]) + (correlation of a and w reversed)
        
        # Let me compute this efficiently:
        # First term: a[i] * sum_{k=i}^{i+n-1} w[k] (clipped to valid range)
        # Second term: sum_{k: 0<=k-i<n} w[k] * a[k-i] = sum_{j=0}^{n-1} w[i+j] * a[j] (where i+j < 2n-1)
        
        # Actually, let me just use FFT:
        # The second term is (a * w_rev)[i] where w_rev is the reverse of w... 
        # Hmm, let me think again.
        # sum_k w[k] * a[k-i] * I(0<=k-i<n) = sum_j w[i+j] * a[j] * I(0<=j<n) = (w convolved with a) at position i+n-1... 
        # This is getting complicated. Let me just use the direct approach with FFT.
        
        # Actually, the simplest correct approach:
        # grad = sum_k w[k] * (a shifted by k) + sum_k w[k] * (a reversed shifted)
        # Let me use: grad[i] = sum_k w[k] * (a[(k-i) % n] if 0<=k-i<n else 0) + ... 
        # This is O(n^2) which is too slow for n=256? Actually 256^2 = 65536, that's fine.
        
        # But let me use the FFT approach:
        # The gradient of sum_k w[k] * c[k] where c = conv(a,a) is:
        # = conv(w, a) + conv(w_rev, a)  (by the chain rule and the structure of convolution)
        # Wait, d/d a [c[k]] = a reversed shifted... 
        # Let me just do it the straightforward way:
        
        # grad[i] = sum_{k=0}^{2n-2} w[k] * (a[k-i] if 0<=k-i<n else 0) + a[i] * sum_{k=max(0,i)}^{min(2n-2,i+n-1)} w[k]
        # Hmm, this is getting complicated. Let me just use a simpler (but correct) approach:
        
        # The Jacobian of c = conv(a,a) w.r.t. a is: J[i,k] = dc[k]/da[i]
        # J[i,k] = 1 if k-i is in [0,n-1] (contribution from the first a)
        #          + 1 if k-i is in [0,n-1] (contribution from the second a)
        # Wait no, dc[k]/da[i]:
        # c[k] = sum_j a[j] * a[k-j]
        # The term with a[i] appears when j=i (giving factor a[k-i]) or when k-j=i (giving factor a[j]=a[i], but this is the same as j=k-i)
        # So dc[k]/da[i] = a[k-i] if 0<=k-i<n, plus a[i] if 0<=k-i<n (from the other factor)
        # Wait, that double counts. Let me be very explicit:
        # c[k] = a[0]*a[k] + a[1]*a[k-1] + ... (terms where both indices are in [0,n-1])
        # dc[k]/da[i]: differentiate each term that contains a[i].
        # Term a[j]*a[k-j] contains a[i] if j=i or k-j=i.
        # If j=i: term is a[i]*a[k-i], derivative is a[k-i]. Valid when 0<=k-i<n.
        # If k-j=i (j=k-i): term is a[k-i]*a[i], derivative is a[k-i]. Valid when 0<=k-i<n.
        # When j=i and k-j=i (i.e., k=2i): the term is a[i]*a[i], derivative is 2*a[i].
        # So dc[k]/da[i] = 2*a[k-i] if k=2i and 0<=k-i<n, else a[k-i] (counted once) if 0<=k-i<n.
        # Wait, I think I'm double counting. When j=i, the term is a[i]*a[k-i]. When k-j=i, j=k-i, the term is a[k-i]*a[i]. These are the SAME term when j=i and j=k-i, i.e., when i=k-i. Otherwise they're different terms in the sum.
        # So: dc[k]/da[i] = a[k-i] (from j=i) + a[k-i] (from j=k-i) = 2*a[k-i] when i ≠ k-i and both are valid
        # And = 2*a[i] when i = k-i (i.e., k=2i).
        # Wait no. If i ≠ k-i, then the term with j=i is a[i]*a[k-i] and the term with j=k-i is a[k-i]*a[i]. These are two different terms in the sum (different values of j), so both contribute to the derivative. Each gives a[k-i]. Total: 2*a[k-i].
        # If i = k-i, then there's only one term: a[i]*a[i], derivative is 2*a[i].
        # So in all cases: dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
        
        # Therefore: grad[i] = sum_k w[k] * 2*a[k-i] * I(0<=k-i<n)
        #             = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (where i+j < 2n-1)
        #             = 2 * (correlation of w and a)
        
        # This is: grad = 2 * (w_full * a) where * denotes appropriate cross-correlation
        # Let me compute: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] for valid i+j
        # This is a cross-correlation. I can compute it with FFT.
        
        # w has length 2n-1, a has length n.
        # grad[i] = 2 * sum_j w[i+j]*a[j] for j=0..n-1 and i+j < 2n-1
        # = 2 * (w ⊛ a)[i] where ⊛ is cross-correlation
        
        # Using FFT: cross-correlation can be computed as ifft(fft(w) * conj(fft(a)))
        # But the lengths are different, so I need to zero-pad.
        
        size_w = 1
        while size_w < (2*n-1) + n:
            size_w <<= 1
        W = np.fft.rfft(w, size_w)
        A = np.fft.rfft(a, size_w)
        corr = np.fft.irfft(W * np.conj(A), size_w)
        # corr[i] = sum_j w[i+j] * conj(a[j]) = sum_j w[i+j]*a[j] (since a is real)
        # We want grad[i] = 2 * corr[i] for i = 0..n-1
        # But we need to check the indexing...
        # Actually, the standard cross-correlation (w ⊛ a)[i] = sum_j w[i+j]*a[j]
        # In numpy/scipy, this would be: for each i, sum over j of w[i+j]*a[j]
        # With FFT: pad both to same length, compute ifft(FFT(w) * conj(FFT(a)))
        # This gives the circular cross-correlation, which for zero-padded signals equals the linear one.
        
        # Let me verify: if w has length L1 and a has length L2, and we pad both to length L >= L1+L2-1,
        # then ifft(FFT(w_padded) * conj(FFT(a_padded))) gives a result of length L where
        # result[i] = sum_j w_padded[(i+j) % L] * conj(a_padded[j])
        # For linear cross-correlation, we want result[i] = sum_j w[i+j]*a[j] for i = 0, ..., L1+L2-2
        # With zero-padding, this is correct for i in the valid range.
        
        # So grad[i] = 2 * corr[i] for i = 0, ..., n-1

        g_a = 2.0 * corr[:n]
        
        # Now project onto the constraint that sum(a) = S (the normalization)
        # We're parameterizing a = softmax(theta) * S, so the gradient w.r.t. theta is:
        # d a[i] / d theta[j] = a[i] * (delta_{ij} - a[j]/sum(a)) = a[i] * (delta_{ij} - a[j]/S)
        # grad_theta[j] = sum_i (d score / d a[i]) * (d a[i] / d theta[j])
        # score = 2n * max(conv) / sum(a)^2 (but we're using soft_max as proxy)
        # Let's just use: grad_theta = a * (g_a - mean(g_a * a) / S * S) 
        # Actually: grad_theta[j] = sum_i g_a[i] * a[i] * (delta_{ij} - a[j]/S)
        #            = g_a[j]*a[j] - a[j]/S * sum_i g_a[i]*a[i]
        #            = a[j] * (g_a[j] - dot(g_a, a)/S)
        
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)
        
        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.005 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose and step_count % 100 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}")

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing with peak-targeting moves."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0a = time.monotonic()
    temperature = 0.005
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0a < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.4:
            # Peak-targeting: reduce elements that contribute most to the peak
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(10, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.01, size=n_inc))
        elif r < 0.65:
            # Block modification
            width = max(2, int(rng.integers(2, max(3, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Single element perturbation
            idx = rng.integers(0, n)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.03))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            f = candidate
            current_score = new_score

        temperature *= cooling

    return best_f, best_score

def make_chirp_init(n, rng):
    """Generate chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = rng.uniform(0.3, 0.8)
    freq = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_multitone_init(n, rng):
    """Generate multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(rng.integers(4, 10)):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_splash_init(n, rng):
    """Generate sparse 'splash' initialization."""
    a = np.full(n, 0.1)
    n_spikes = rng.integers(n//10, n//4)
    spike_idx = rng.choice(n, size=n_spikes, replace=False)
    a[spike_idx] = rng.uniform(1.0, 3.0, size=n_spikes)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_binary_init(n, rng):
    """Generate binary-like initialization."""
    a = rng.choice([0.1, 1.0, 1.5], size=n, p=[0.3, 0.5, 0.2])
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng):
    a = 0.5 + rng.random(n)
    return np.clip(a, 0.01, None)

def make_provided_init(provided_seq, n, rng):
    if len(provided_seq) == n:
        return provided_seq.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(provided_seq))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided_seq)
    return np.clip(a, 0.01, None)

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse LP-based local optimization with
    gradient polishing and simulated annealing.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    n_prov = len(provided_seq) if provided_seq is not None else 256

    # === Phase 1: Multi-start LP optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: Multi-start LP-based optimization")
    print("=" * 60)

    # Generate diverse initializations
    inits = []
    if provided_seq is not None:
        inits.append(("provided", make_provided_init(provided_seq, n_prov, rng)))
    inits.append(("chirp1", make_chirp_init(n_prov, rng)))
    inits.append(("chirp2", make_chirp_init(n_prov, rng)))
    inits.append(("multitone1", make_multitone_init(n_prov, rng)))
    inits.append(("splash1", make_splash_init(n_prov, rng)))
    inits.append(("binary1", make_binary_init(n_prov, rng)))
    inits.append(("random1", make_random_init(n_prov, rng)))
    inits.append(("multitone2", make_multitone_init(n_prov, rng)))
    inits.append(("chirp3", make_chirp_init(n_prov, rng)))

    # Also try a larger length
    if n_prov <= 512:
        inits.append(("chirp_large", make_chirp_init(512, rng)))
        inits.append(("multitone_large", make_multitone_init(512, rng)))

    # Sort by initial score (best first)
    init_scores = []
    for name, a in inits:
        s = local_score(a)
        init_scores.append((s, name, a))
    init_scores.sort(key=lambda x: x[0])

    print(f"  {len(init_scores)} initializations generated")
    for s, name, a in init_scores[:5]:
        print(f"    {name}: n={len(a)}, initial_score={s:.6f}")

    # Run LP optimization on each, with time budget proportional to quality
    total_time_p1 = min(900.0, deadline - time.monotonic() - 300)  # Reserve time for later phases
    if total_time_p1 < 100:
        total_time_p1 = max(100, deadline - time.monotonic() - 50)

    # Allocate more time to better initializations
    weights = []
    for s, name, a in init_scores:
        w = max(0.5, 3.0 - s)  # Lower score = higher weight
        weights.append(w)
    total_w = sum(weights)

    for i, (s, name, a) in enumerate(init_scores):
        if time.monotonic() > deadline - 250:
            break
        time_alloc = total_time_p1 * weights[i] / total_w
        time_alloc = max(20.0, min(time_alloc, deadline - time.monotonic() - 200))
        if time_alloc < 15:
            break

        print(f"  [{i+1}/{len(init_scores)}] n={len(a)}, init='{name}', alloc={time_alloc:.0f}s")
        f_best, score = lp_optimize(a, time_alloc, t_init=0.4, verbose=False)
        print(f"    LP result: score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.6f} (n={len(f_best)}) ***")

    # === Phase 2: Crossover between top solutions ===
    print("\n" + "=" * 60)
    print("Phase 2: Crossover and re-optimization")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 200:
        # Create crossover variants of the best solution
        n_best = len(best_seq)
        crossover_inits = []
        
        # Mix with original initializations
        for s, name, a in init_scores[:3]:
            if len(a) == n_best:
                mixed = 0.5 * best_seq + 0.5 * a
                mixed = np.clip(mixed, 0.01, None)
                mixed = mixed * (np.sum(best_seq) / np.sum(mixed))
                crossover_inits.append(mixed)
        
        # Random mixture
        for _ in range(3):
            a_rand = make_random_init(n_best, rng)
            mixed = 0.7 * best_seq + 0.3 * a_rand
            mixed = np.clip(mixed, 0.01, None)
            mixed = mixed * (np.sum(best_seq) / np.sum(mixed))
            crossover_inits.append(mixed)

        total_time_p2 = min(300.0, deadline - time.monotonic() - 200)
        if total_time_p2 > 50:
            per_xo = total_time_p2 / len(crossover_inits)
            for i, a_xo in enumerate(crossover_inits):
                if time.monotonic() > deadline - 200:
                    break
                print(f"  Crossover {i+1}/{len(crossover_inits)}, alloc={per_xo:.0f}s")
                f_xo, score_xo = lp_optimize(a_xo, min(per_xo, 100), t_init=0.3)
                print(f"    Score: {score_xo:.6f}")
                if score_xo < best_score:
                    best_score = score_xo
                    best_seq = f_xo.copy()
                    print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 3: Gradient polishing ===
    print("\n" + "=" * 60)
    print("Phase 3: Gradient-based polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(200.0, remaining * 0.4)
        if grad_time > 30:
            print(f"  Polishing n={len(best_seq)}, time={grad_time:.0f}s")
            a_grad, score_grad = gradient_polish(best_seq, grad_time, verbose=False)
            print(f"  Gradient polish: {score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Simulated annealing polish")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(250.0, remaining * 0.6)
        if sa_time > 30:

Multi-phase optimization for minimizing normalized peak self-convolution
of non-negative step functions. Combines diverse multi-start LP-based
alternating optimization, gradient polishing with adaptive soft-max,
crossover between top solutions, and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f, n):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. sum_j g[j]*f[k-j] <= M for all k, g >= 0."""
    A = build_constraint_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 20})
    if result.success:
        return result.x
    return None


def lp_optimize(f_init, time_limit, t_init=0.4, verbose=False):
    """LP-based iterative optimization with adaptive step size and backtracking."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(77777)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        if M < 1e-15:
            break

        g = solve_lp(f, M, n)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        # Backtracking line search
        accepted = False
        for trial_t in [t, t * 0.6, t * 0.4, t * 0.2, t * 0.1]:
            g_rescaled = g * (sum_f / sum_g)
            f_new = (1.0 - trial_t) * f + trial_t * g_rescaled
            f_new = np.clip(f_new, 0.0, None)
            sum_new = np.sum(f_new)
            if sum_new < 1e-10:
                continue
            f_new = f_new * (sum_f / sum_new)
            score_new = local_score(f_new)
            if score_new <= best_score + 1e-9:
                f = f_new
                t = trial_t
                accepted = True
                if score_new < best_score - 1e-12:
                    best_score = score_new
                    best_f = f.copy()
                stagnant = 0
                break

        if not accepted:
            # Accept with small step to avoid getting stuck
            g_rescaled = g * (sum_f / sum_g)
            f = 0.97 * f + 0.03 * g_rescaled
            f = np.clip(f, 0.0, None)
            s = np.sum(f)
            if s > 1e-10:
                f = f * (sum_f / s)

                f = f * (sum_f / np.sum(f))

                stagnant = 0

            t = min(0.65, t * 1.015)

        if verbose and iteration % 20 == 0:

                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=False):
    """Adam gradient descent with annealed soft-max on log-parameterization."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.9
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()

    tau_max = 15.0 / n
    tau_min = 0.05 / n

    # Precompute FFT size for gradient
    fft_size = 1
    while fft_size < 3 * n:
        fft_size <<= 1

    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_max * (tau_min / tau_max) ** (elapsed_frac ** 0.6)

        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        conv = fft_convolve(a, a)
        conv_max = np.max(conv)

        # Soft-max weights
        shifted = (conv - conv_max) / tau
        shifted = np.clip(shifted, -60, 0)
        exp_conv = np.exp(shifted)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp

        # Gradient: grad_a[i] = 2 * sum_j w[i+j] * a[j]
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        corr = np.fft.irfft(W * np.conj(A), fft_size)
        g_a = 2.0 * corr[:n]

        # Project to log-parameterization
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.003 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart mechanism: if stuck, add perturbation
        if no_improve > 500:
            theta += 0.1 * np.random.standard_normal(n)
            no_improve = 0

        if verbose and step_count % 200 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}")

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)


def sa_polish(f_init, time_limit, rng):
    """Simulated annealing with peak-targeting and diversity moves."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0a = time.monotonic()
    temperature = 0.004
    cooling = 0.99985
    iter_count = 0

    while time.monotonic() - t0a < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.40:
            # Peak-targeting: reduce peak contributors, increase others
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(12, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif r < 0.60:
            # Block modification
            width = max(2, int(rng.integers(2, max(3, n // 4))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.80:
            # Pair swap (move mass from high to low)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                transfer = 0.02 * sum_f
                candidate[i] -= transfer
                candidate[j] += transfer
        elif r < 0.90:
            # Asymmetric shift
            shift = int(rng.integers(1, max(2, n // 25)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))
        else:
            # Global tiny perturbation
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            f = candidate
            current_score = new_score

        temperature *= cooling

    return best_f, best_score


def make_chirp_init(n, rng):

        if time_alloc < 12:
            break

        print(f"  [{i+1}/{len(init_scores)}] '{name}' n={len(a)}, t={time_alloc:.0f}s")
        f_best, score = lp_optimize(a, time_alloc, t_init=0.35, verbose=False)
        print(f"    -> score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"    *** NEW BEST: {best_score:.6f} ***")

    # === Phase 2: Crossover ===
    print("\n" + "=" * 60)
    print("Phase 2: Crossover and re-optimization")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 250:
        n_best = len(best_seq)
        xover_inits = []

        # Mix best with other good initializations of same length
        for s, name, a in init_scores[:5]:
            if len(a) == n_best and a is not best_seq:
                mixed = 0.5 * best_seq + 0.5 * a
                mixed = np.clip(mixed, 0.01, None)
                mixed = mixed * (np.sum(best_seq) / np.sum(mixed))
                xover_inits.append(mixed)

        # Random mix
        for _ in range(3):
            a_r = make_random_init(n_best, rng)
            mixed = 0.6 * best_seq + 0.4 * a_r
            mixed = np.clip(mixed, 0.01, None)
            mixed = mixed * (np.sum(best_seq) / np.sum(mixed))
            xover_inits.append(mixed)

        total_time_p2 = min(250.0, deadline - time.monotonic() - 200)
        if total_time_p2 > 60 and xover_inits:
            per_xo = total_time_p2 / len(xover_inits)
            for i, a_xo in enumerate(xover_inits):
                if time.monotonic() > deadline - 200:
                    break
                ta = min(per_xo, 80.0)
                if ta < 10:
                    break
                print(f"  Xover {i+1}/{len(xover_inits)}, t={ta:.0f}s")
                f_xo, score_xo = lp_optimize(a_xo, ta, t_init=0.3)
                print(f"    -> score={score_xo:.6f}")
                if score_xo < best_score:
                    best_score = score_xo
                    best_seq = f_xo.copy()
                    print(f"    *** NEW BEST: {best_score:.6f} ***")

    # === Phase 3: Gradient polishing ===
    print("\n" + "=" * 60)
    print("Phase 3: Gradient polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(180.0, remaining * 0.35)
        if grad_time > 25:
            print(f"  n={len(best_seq)}, time={grad_time:.0f}s")
            a_grad, score_grad = gradient_polish(best_seq, grad_time, verbose=False)
            print(f"  -> score={score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === Phase 4: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Simulated annealing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(200.0, remaining * 0.55)
        if sa_time > 25:
            print(f"  n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  -> score={score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === Phase 5: Final LP ===
    print("\n" + "=" * 60)
    print("Phase 5: Final LP refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        final_time = min(80.0, remaining)
        if final_time > 8:
            print(f"  n={len(best_seq)}, time={final_time:.0f}s")
            a_final, score_final = lp_optimize(best_seq, final_time, t_init=0.15)
            print(f"  -> score={score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
