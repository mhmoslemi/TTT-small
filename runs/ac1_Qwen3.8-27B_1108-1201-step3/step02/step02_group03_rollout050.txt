"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam optimizer with soft-max smoothing, LP-based dual refinement,
simulated annealing explore-exploit, and L-BFGS polishing.

import numpy as np
import time
from scipy.optimize import linprog, minimize


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=1.0):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target / len(a))
    return a * (target / S)


def softmax_grad(a, tau):
    """
    Compute gradient of log-sum-exp(tau * conv(a,a)) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    # grad[i] = tau * 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = tau * 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def hardmax_grad(a):
    """Compute gradient at the argmax lag of the autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_solve(f, M):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > 1200:
        return None
    try:
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            m_start = j
            m_end = min(j + n, 2 * n - 1)
            A[m_start:m_end, j] = f[m_start - j:m_end - j]
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine_step(a, M, blending_choices=(0.5, 0.3, 0.15)):
    """Try LP-based refinement, return (new_a, improved)."""
    n = len(a)
    g = lp_solve(a, M)
    if g is None:
        return a, False
    g_sum = g.sum()
    if g_sum < 1e-6:
        return a, False
    g = normalize(g, 1.0)
    best_a = a.copy()
    best_s = compute_obj(a)
    for t_blend in blending_choices:
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, 1.0)
        s_new = compute_obj(a_new)
        if s_new < best_s - 1e-12:
            best_a = a_new
            best_s = s_new
    if best_s < compute_obj(a) - 1e-12:
        return best_a, True
    return a, False


def coordinate_polish(a, deadline, max_rounds=5):
    """Coordinate descent: try perturbing individual coordinates."""
    n = len(a)
    a = a.copy()
    best_score = compute_obj(a)
    patience = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-6:
                candidates = [0.0, 0.001, 0.01]
            else:
                j = k_star - i
                if 0 <= j < n and a[j] > orig:
                    candidates = [0.0, orig * 0.3, orig * 0.5, orig * 0.7, orig * 1.3]
                else:
                    candidates = [0.0, orig * 0.5, orig * 1.5]

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = max(val, 0.0)
                a_norm = normalize(a, 1.0)
                s = compute_obj(a_norm)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                    best_a_copy = a_norm.copy()
                a[i] = orig

            if best_val != orig:
                a = best_a_copy.copy()
                best_score = best_s
                improved = True
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))

        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0

    return a, best_score


def lbfgs_polish(a, deadline, tau=15.0):
    """Use L-BFGS-B to polish the sequence (without sum constraint, using normalization)."""
    n = len(a)
    if n > 800:
        return a, compute_obj(a)

    def objective(x):
        a_loc = np.maximum(x, 0.0)
        s = a_loc.sum()
        if s < 1e-12:
            return 1e10, np.zeros(n)
        a_loc = a_loc / s
        c = np.convolve(a_loc, a_loc)
        c_shifted = tau * (c - c.max())
        max_exp = np.max(c_shifted)
        log_sum = max_exp + np.log(np.exp(c_shifted - max_exp).sum())
        val = tau * log_sum / tau
        # Gradient
        w = np.exp(c_shifted - max_exp)
        w_sum = w.sum()
        w = w / w_sum
        a_rev = a_loc[::-1]
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * tau * conv_result[n - 1:n - 1 + n]
        return val, grad

    try:
        x0 = a.copy()
        result = minimize(objective, x0, method='L-BFGS-B',
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-12,
                                   'maxfun': 20000})
        if result.success or result.fun < compute_obj(a):
            a_new = np.maximum(result.x, 0.0)
            a_new = normalize(a_new, 1.0)
            return a_new, compute_obj(a_new)
    except Exception:
        pass
    return a, compute_obj(a)

        M = float(c.max())
        # Try with slightly relaxed M
        for m_factor in [1.0, 1.02, 1.05]:
            M_try = M * m_factor
            a_lp, improved = lp_refine_step(best, M_try)
            if improved:
                s = compute_obj(a_lp)
                if s < best_score:
                    best = a_lp
                    best_score = s

    return best, best_score


def generate_initializations(rng, n, use_prev=True):
    """Generate diverse initializations for length n."""
    starts = []

    # Previous best (if available)
    if use_prev:
        try:
            gs = np.asarray(height_sequence_1, dtype=np.float64)
            if gs.size >= 8 and np.all(np.isfinite(gs)):
                if len(gs) == n:
                    starts.append(gs.copy())
                elif len(gs) > n:
                    starts.append(gs[:n].copy())
                else:
                    # Pad or interpolate
                    from scipy.interpolate import interp1d
                    x_old = np.linspace(0, 1, len(gs))
                    x_new = np.linspace(0, 1, n)
                    f = interp1d(x_old, gs, kind='cubic', fill_value='extrapolate')
                    starts.append(np.clip(f(x_new), 0, None))
                # Reversed
                starts.append(gs[:n][::-1].copy() if len(gs) >= n else gs[::-1].copy())
        except (NameError, Exception):
            pass

    # Uniform
    starts.append(np.ones(n) * 0.5)

    # Random
    for _ in range(2):
        starts.append(rng.random(n) * 0.8 + 0.1)

    # Structured: smooth periodic
    i = np.arange(n, dtype=np.float64)
    starts.append(0.3 + 0.3 * np.sin(2 * np.pi * i * 3.0 / n))
    starts.append(0.3 + 0.3 * np.cos(2 * np.pi * i * 5.0 / n + 0.5))
    starts.append(0.2 + 0.4 * np.abs(np.sin(2 * np.pi * i * 2.0 / n)))

    # Gaussian-like
    center = n / 2.0
    starts.append(np.exp(-0.5 * ((i - center) / (n * 0.15)) ** 2) + 0.05)

    # Two peaks
    starts.append(0.1 + np.exp(-0.5 * ((i - n * 0.25) / (n * 0.08)) ** 2) +
                  np.exp(-0.5 * ((i - n * 0.75) / (n * 0.08)) ** 2))

    # "Random smooth" - low-pass filtered random
    r = rng.random(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kernel /= kernel.sum()
    starts.append(np.convolve(r, kernel, mode='same') + 0.05)

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization with Adam, LP, L-BFGS,
    and coordinate descent from diverse starting points.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    print(f"=== Starting optimization, budget={budget_s}s ===")
    print(f"Deadline in {deadline - t_start:.1f}s")

    # Try multiple lengths, ordered by promise
    lengths_to_try = [384, 256, 512, 448]

    # First, quick evaluation of previous best if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size > 8 and np.all(np.isfinite(gs)):
            s_prev = compute_obj(gs)
            best_score = s_prev
            best_seq = gs.copy()
            print(f"Previous best: n={len(gs)}, score={s_prev:.6f}")
    except (NameError, Exception):
        print("No previous sequence available.")

    # Allocate time among lengths
    # Give more time to the length of the current best
    n_total_starts = 0
    for n in lengths_to_try:
        n_total_starts += 4  # roughly 4 starts per length

    # First pass: quick optimization from previous best
    if best_seq is not None and len(best_seq) in lengths_to_try:
        n_best = len(best_seq)
        t_alloc = min(300.0, (deadline - t_start) * 0.25)
        print(f"\n--- Refining previous best (n={n_best}) with {t_alloc:.0f}s ---")
        seq, score = optimize_single(best_seq, t_start + t_alloc, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Improved to {best_score:.6f}")

    # Second pass: try other lengths and initializations
    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        if n == len(best_seq) if best_seq is not None else False:
            continue  # Already tried this length with prev best

        remaining = deadline - time.monotonic() - 30
        starts = generate_initializations(rng, n, use_prev=(n == len(best_seq) if best_seq is not None else True))

        # Deduplicate
        unique_starts = []
        for s in starts:
            key = round(float(s.sum()), 2)
            if not any(abs(key - round(float(us.sum()), 2)) < 0.1 and len(s) == len(us) for us in unique_starts):
                unique_starts.append(s)
        starts = unique_starts

        # Allocate time per start
        t_per_start = min(120.0, remaining / max(1, len(starts)) * 0.8)
        t_per_start = max(30.0, t_per_start)

        print(f"\n--- Length n={n}: {len(starts)} starts, ~{t_per_start:.0f}s each ---")

        for idx, a0 in enumerate(starts):
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(t_per_start, deadline - time.monotonic() - 30)
            if t_alloc < 20:
                break

            seq, score = optimize_single(a0, t_start + t_alloc, verbose=False)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [n={n}, start {idx}] NEW BEST: {best_score:.6f}")
            else:
                print(f"  [n={n}, start {idx}] score={score:.6f}")

    # Final intensive refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        n_final = len(best_seq)
        print(f"\n=== Final refinement (n={n_final}, score={best_score:.6f}, {remaining:.0f}s) ===")

        # Try multiple perturbation + re-optimization
        for trial in range(5):
            if time.monotonic() > deadline - 20:
                break
            t_alloc = min(200.0, (deadline - time.monotonic() - 10) / (5 - trial))
            if t_alloc < 20:
                break

            # Different perturbations
            if trial == 0:
                a0 = best_seq.copy()
            elif trial == 1:
                noise = rng.normal(0, 0.05, n_final)
                a0 = np.maximum(best_seq + noise, 0)
            elif trial == 2:
                a0 = best_seq[::-1].copy()  # Reversed
            elif trial == 3:
                # Random projective perturbation
                direction = rng.normal(0, 1, n_final)
                direction -= direction.mean()
                a0 = np.maximum(best_seq + 0.1 * direction, 0)
            else:
                # Sparse perturbation
                a0 = best_seq.copy()
                idx_pert = rng.choice(n_final, size=n_final // 4, replace=False)
                a0[idx_pert] *= rng.uniform(0.5, 1.5, n_final // 4)

            seq, score = optimize_single(a0, t_start + t_alloc, verbose=False)
            if score < best_score - 1e-10:
                best_score = score
                best_seq = seq.copy()
                print(f"  Final trial {trial}: NEW BEST {best_score:.6f}")
            else:
                print(f"  Final trial {trial}: score={score:.6f}")

        # Final L-BFGS + coordinate polish
        if time.monotonic() < deadline - 10 and n_final <= 700:
            a_lb, s_lb = lbfgs_polish(best_seq, deadline - 5)
            if s_lb < best_score - 1e-10:
                best_score = s_lb
                best_seq = a_lb
                print(f"  Final L-BFGS: {best_score:.6f}")

        if time.monotonic() < deadline - 5 and n_final <= 512:
            a_cd, s_cd = coordinate_polish(best_seq, deadline - 2)
            if s_cd < best_score - 1e-10:
                best_score = s_cd
                best_seq = a_cd
                print(f"  Final coord: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback: uniform sequence
    print("Fallback: uniform sequence")
    return [1.0] * 256
