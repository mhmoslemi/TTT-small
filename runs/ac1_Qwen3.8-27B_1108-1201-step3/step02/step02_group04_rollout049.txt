"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines adaptive mirror descent with line search, coordinate descent polishing,
LP-guided exploration, and simulated annealing for escaping local minima.
Improves upon previous LP-only approach with more diverse local search strategies.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fft_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgrad(a, w):
    """Compute grad[i] = sum_j a[j]*w[i+j] for i in [0,n-1], j in [0,n-1]."""
    n = len(a)
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    return W @ a


def mirror_descent(a0, time_limit, t_start=30.0, eta0=0.15, use_lass=False):
    """Adaptive mirror descent with line search and temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    temp = t_start
    patience = 0
    stall_count = 0
    t0 = time.monotonic()
    prev_best = best_obj

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        # Compute importance weights with current temperature
        scaled = temp * conv / M
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        grad = compute_subgrad(a, w)
        grad -= grad.mean()

        # Line search: try several step sizes
        best_step_obj = best_obj
        best_step_a = None
        candidates = [eta, eta * 0.5, eta * 0.25, eta * 2.0, eta * 0.75]
        for eta_try in candidates:
            a_new = a * np.exp(-eta_try * grad)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new /= s_new
            conv_new = fft_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_step_obj:
                best_step_obj = obj_new
                best_step_a = a_new
                best_eta = eta_try

        if best_step_a is not None:
            a = best_step_a
            eta = best_eta
            if best_obj - best_step_obj > 1e-10:
                best_obj = best_step_obj
                best_a = a.copy()
                patience = 0
                stall_count = 0
            else:
                patience += 1
                if best_obj < prev_best - 1e-8:
                    prev_best = best_obj
                    stall_count = 0
                else:
                    stall_count += 1
        else:
            patience += 1
            eta *= 0.8

        # Adaptive temperature and step size
        if patience % 20 == 0:
            temp = min(temp * 1.25, 5000.0)
            eta = max(eta * 0.85, 0.01)
            if patience > 100:
                # Annealing escape: add small random perturbation
                noise = np.random.randn(n) * 0.01
                a_pert = a * np.exp(noise)
                a_pert = np.maximum(a_pert, 0.0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a_pert /= sp
                    conv_p = fft_convolve(a_pert)
                    obj_p = 2.0 * n * conv_p.max()
                    if obj_p < best_obj * 1.05:  # accept if not too bad
                        a = a_pert
                patience = 50  # reset to give it a chance
                stall_count = 0

        if patience > 300:
            break

    return best_a, best_obj


def coordinate_descend(a0, time_limit, max_sweeps=50):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            a_old = a[i]
            if a_old < 1e-15:
                continue
            # Try several new values for a[i]
            for frac in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0]:
                a[i] = a_old * frac
                s = a.sum()
                if s < 1e-15:
                    a[i] = a_old
                    continue
                a_norm = a / s
                conv_new = fft_convolve(a_norm)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj - 1e-12:
                    a = a_norm
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = best_a[i]

        if not improved:
            break

    return best_a, best_obj


def make_lp_indices(n):
    """Precompute sparse structure for LP constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, n, rows, cols, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_step(f, n, rows, cols, max_iters=5):
    """LP-guided step: find direction and take a step."""
    conv = fft_convolve(f)
    M = conv.max()
    f_best = f.copy()
    obj_best = 2.0 * n * M

    for _ in range(max_iters):
        g = solve_lp(f, M, n, rows, cols)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs

        if np.max(np.abs(g - f)) < 1e-12:
            break

        # Line search
        for t in [0.5, 0.3, 0.1, 0.7, 0.8]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                continue
            f_new /= s
            conv_new = fft_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < obj_best - 1e-13:
                f = f_new
                f_best = f.copy()
                obj_best = obj_new
                M = conv_new.max()
                break
    return f_best, obj_best


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.02)
    inits['tri'] = tri / tri.sum()

    # Alternating pattern
    alt = np.ones(n)
    alt[::2] *= 0.3
    inits['alt'] = alt / alt.sum()

    return inits
