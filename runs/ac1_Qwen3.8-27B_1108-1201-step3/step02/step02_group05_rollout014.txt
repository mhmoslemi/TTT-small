```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric initializations,
LP-based iterative optimization, and targeted local search.
Key improvements: asymmetric starting points (ramps, exponentials), longer sequences,
reduced-constraint LP for speed, and multi-scale local search.

import numpy as np
import scipy.optimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s * s)

def solve_lp_reduced(f, M, n_constraints=200):
    """Solve LP with only the top n_constraints most binding constraints."""
    n = len(f)
    conv_f = fast_convolve(f)
    # Get the indices of the largest convolution values
    if n_constraints >= 2 * n - 1:
        k_indices = np.arange(2 * n - 1)
    else:
        k_indices = np.argsort(conv_f)[::-1][:n_constraints]
    
    A = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[idx, j] = f[i]
    
    c = -np.ones(n)
    b = np.full(len(k_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_full(f, M):
    """Solve the full LP with all constraints."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def line_search(f, g, max_iter=40):
    """Golden section search for best t in [0,1]."""
    def f_of_t(t):
        a = (1.0 - t) * f + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return np.inf
        a = a / s
        return obj_val(a)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)
    t = (lo + hi) / 2.0
    return t, f_of_t(t)

def lp_optimize(f0, time_limit, use_reduced=False, n_constraints=200):

    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 80

        if use_reduced:

        else:
            g = solve_lp_full(f, M)
        
        if g is None:
            no_improve += 5
            continue
        
        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            t, obj_t = line_search(f, g)
            if obj_t < best_obj - 1e-12:
                a_new = (1.0 - t) * f + t * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns < 1e-12:
                    break
                a_new = a_new / ns
                f = a_new
                best_obj = obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= max_no_improve:
            break
    
    return best_f, best_obj
