def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    result = np.fft.irfft(A * A, n=fft_size)
    return result[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution f*g using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, n=fft_size)
    G = np.fft.rfft(g, n=fft_size)
    result = np.fft.irfft(F * G, n=fft_size)
    return result[:size]

def peak_shave(f0, time_limit):
    """
    Directly reduce the peak by shaving coordinates that contribute most to it.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.time()
    iter_count = 0
    patience = 0
    
    while time.time() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        k_peak = int(np.argmax(conv))
        
        # Find coordinates that contribute most to the peak
        # b[k_peak] = sum_j f[j] f[k_peak - j]
        # Contribution of f[i] to b[k_peak]: f[i]*f[k_peak-i] (if valid) + f[k_peak-i]*f[i] (if valid and i≠k_peak-i)
        # = f[i]*(f[k_peak-i] + [i≠k_peak-i]*f[i-k_peak]... 
        # Actually the "sensitivity" of b[k_peak] to f[i] is:
        # db/d f[i] = f[k_peak-i] (if 0<=k_peak-i<n) + f[i-k_peak] (if 0<=i-k_peak<n)
        # For i in 0..n-1:
        #   f[k_peak-i] is valid when 0<=k_peak-i<=n-1, i.e., max(0,k_peak-n+1)<=i<=k_peak
        #   f[i-k_peak] is valid when 0<=i-k_peak<=n-1, i.e., k_peak<=i<=k_peak+n-1
        
        sens = np.zeros(n)
        for i in range(n):
            contrib = 0.0
            if 0 <= k_peak - i < n:
                contrib += f[k_peak - i]
            if 0 <= i - k_peak < n and i != k_peak - i:  # avoid double count
                contrib += f[i - k_peak]
            sens[i] = contrib
        
        # Also check neighboring peaks
        # Try reducing top-sensitive coordinates and see if it helps
        top_k = min(10, n // 4)
        top_indices = np.argsort(sens)[-top_k:]
        
        improved = False
        for frac in [0.1, 0.2, 0.3, 0.5]:
            f_new = f.copy()
            for idx in top_indices:
                f_new[idx] *= (1 - frac)
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                improved = True
                break
        
        if not improved:
            patience += 1
            if patience > 30:
                break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based alternating minimization (directly minimize cross-conv peak)
2. Soft-max gradient descent with multiplicative updates on the simplex
3. Peak-targeted coordinate shaving
4. Spectral flattening heuristic (Gerchberg-Saxton style)
5. Diverse initializations with adaptive time budgeting
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    result = np.fft.irfft(A * A, n=fft_size)
    return result[:size]


def make_cross_conv_indices(n):
    """Precompute (row, col) indices for the cross-conv constraint entries."""
    rows_list = []
    cols_list = []
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 2, i + n - 1)
        if k_start <= k_end:
            rows_list.extend(range(k_start, k_end + 1))
            cols_list.extend([i] * (k_end - k_start + 1))
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_min_peak_lp(f, n, rows, cols, timeout=8):
    """
    Solve: min M' s.t. sum(g)=1, g>=0, (f*g)[k] <= M' for all k.
    Returns (g, M_prime) or (None, None).
    """
    # Variables: x = [g_0, ..., g_{n-1}, M']
    c = np.zeros(n + 1)
    c[n] = 1.0

    # Inequality: (f*g)[k] - M' <= 0
    vals = np.tile(f, n)
    all_rows = np.concatenate([rows, np.arange(2 * n - 1, dtype=np.int32)])
    all_cols = np.concatenate([cols, np.full(2 * n - 1, n, dtype=np.int32)])
    all_vals = np.concatenate([vals, -np.ones(2 * n - 1)])

    A_ub = sparse.csc_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub,
        A_eq=A_eq, b_eq=b_eq,
        bounds=bounds, method='highs',
        options={'disp': False, 'time_limit': timeout}
    )

    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        return g, res.x[n]
    return None, None


def lp_alternate(f0, time_limit, n):
    """LP-guided alternating: at each step, find g minimizing max(f*g) subject to sum(g)=1."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_cross_conv_indices(n)

    t0 = time.time()
    t_interp = 1.0
    patience = 0
    iter_count = 0

    while time.time() - t0 < time_limit:
        iter_count += 1
        g, g_peak = solve_min_peak_lp(f, n, rows, cols, timeout=6)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            break

        improved = False
        for t_try in [t_interp, 0.7 * t_interp, 0.5 * t_interp, 0.3 * t_interp, 0.1 * t_interp]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t_interp = min(1.0, t_interp * 1.02)
                improved = True
                break

        if not improved:
            patience += 1
            t_interp *= 0.92
            if patience > 100:
                # Try a random perturbation to escape
                pert = np.random.default_rng(iter_count).normal(0, 0.02, n)
                f_pert = f + pert
                f_pert = np.maximum(f_pert, 0)
                sp = f_pert.sum()
                if sp > 1e-12:
                    f_pert /= sp
                    conv_pert = fast_convolve(f_pert)
                    obj_pert = 2.0 * n * conv_pert.max()
                    if obj_pert < best_obj:
                        f = f_pert
                        best_obj = obj_pert
                        best_f = f.copy()
                        patience = 0
                        t_interp = 0.5
                    else:
                        patience = 50  # accelerate termination
                if patience > 120:
                    break

        if iter_count % 50 == 0:
            elapsed = time.time() - t0
            print(f"    LP-alt iter {iter_count}, obj={best_obj:.8f}, t={t_interp:.3f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def softmax_gradient_descent(f0, time_limit, n, t_param=30.0, lr_base=0.05):
    """
    Minimize soft-max approximation of auto-conv peak using multiplicative updates (mirror descent).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_count = 0
    lr = lr_base
    curr_t = t_param

    t0 = time.time()
    patience = 0

    while time.time() - t0 < time_limit:
        step_count += 1
        conv = fast_convolve(f)
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Soft-max weights (temperature = 1/t_param)
        scaled = curr_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Gradient of soft-max objective w.r.t. f
        # d/d f[i] sum_k w[k] b[k] = sum_k w[k] (f[k-i] + f[i-k]) [with f[x]=0 outside [0,n-1]]
        grad1 = np.correlate(w, f, mode='full')[:n]
        grad2 = np.convolve(f, w, mode='full')[:n]
        grad = grad1 + grad2

        # Center for simplex invariance
        grad_c = grad - grad.mean()

        # Adam
        m = beta1 * m + (1 - beta1) * grad_c
        v = beta2 * v + (1 - beta2) * grad_c ** 2
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)

        update = m_hat / (np.sqrt(v_hat) + eps)
        update_norm = np.sqrt(np.mean(update ** 2))
        if update_norm > 0:
            update = update / update_norm

        # Multiplicative update
        f_new = f * np.exp(-lr * update)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-12:
            f_new = np.ones(n) / n
        else:
            f_new /= s

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            patience = 0
            lr = min(0.2, lr * 1.01)
        else:
            patience += 1
            if patience % 40 == 0:
                lr *= 0.75
            if patience % 100 == 0:
                curr_t = min(curr_t * 1.3, 200.0)
            if patience > 400:
                # Random restart from current best with perturbation
                rng = np.random.default_rng(step_count)
                f = best_f + rng.normal(0, 0.01, n)
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs > 1e-12:
                    f /= fs
                else:
                    f = best_f.copy()
                patience = 0
                lr = lr_base * 0.5
                curr_t = t_param
            if patience > 800:
                break

        if step_count % 200 == 0:
            elapsed = time.time() - t0
            print(f"    SoftMax iter {step_count}, obj={best_obj:.8f}, lr={lr:.5f}, t={curr_t:.1f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def peak_shave(f0, time_limit, n):
    """Reduce the auto-conv peak by selectively reducing high-contribution coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.time()
    patience = 0
    iter_count = 0

    while time.time() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        k_peak = int(np.argmax(conv))

        # Sensitivity: how much each f[i] contributes to b[k_peak]
        sens = np.zeros(n)
        for i in range(n):
            c = 0.0
            if 0 <= k_peak - i < n:
                c += f[k_peak - i]
            if 0 <= i - k_peak < n:
                c += f[i - k_peak]
            sens[i] = c

        top_k = min(15, n // 3)
        top_idx = np.argsort(sens)[-top_k:]

        improved = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]:
            f_new = f.copy()
            for idx in top_idx:
                f_new[idx] *= (1.0 - frac)
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                improved = True
                break

        if not improved:
            patience += 1
            if patience > 40:
                break

    return best_f, best_obj


def spectral_flatten(f0, time_limit, n, n_iter_max=200):
    """
    Gerchberg-Saxton style: alternate between time-domain (non-negativity) 
    and frequency-domain (flat magnitude) constraints to flatten the auto-conv.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    # Target: flat auto-conv => |F|^2 should be proportional to a delta
    # But we want the LINEAR conv to be flat, so use zero-padded FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    t0 = time.time()
    patience = 0

    for it in range(n_iter_max):
        if time.time() - t0 > time_limit:
            break

        # FFT of zero-padded f
        F = np.fft.rfft(f, n=fft_size)

        # Target magnitude: we want b = f*f_zero_padded to be flat
        # b_flat[k] = sum(b)/(2n-1) = 1/(2n-1) (since sum(f)=1, sum(b)=1)
        # B_target = FFT(b_flat) = [1, 0, 0, ...]
        # So |F_target|^2 = B_target => |F_target| = [1, 0, 0, ...]
        # That's too restrictive. Instead, use a soft target:
        # We want |F|^2 to be as "peaky" at DC as possible (concentrated energy at DC)
        # Actually for flat b, we need B to be a delta, meaning F has all energy at DC.
        # But F[0] = sum(f) = 1, so F[0]^2 = 1 = sum(b). For b flat, all B[k]=0 for k!=0.
        # This means |F[k]|^2 = 0 for k != 0, which is impossible for a length-n signal
        # unless f is all at one point (a delta), which gives b = delta^2 = delta (not flat).
        
        # Alternative: minimize max(b) - min(b) by shaping the spectrum.
        # Pragmatic approach: reduce the high-frequency content of |F|^2.
        # This makes b smoother (lower peaks).
        
        # Decay high frequencies:
        freqs = np.arange(len(F))
        # Gentle roll-off
        decay = np.exp(-0.1 * (freqs / n) ** 2)
        F_new = F * decay

        # Inverse FFT
        f_new = np.fft.irfft(F_new, n=fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = best_f.copy()
        else:
            f_new /= s

        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            patience = 0
        else:
            patience += 1
            if patience > 50:
                break

    return best_f, best_obj


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()

    # Chirp-like
    chirp = 1.0 + 0.3 * np.cos(2 * np.pi * x ** 2 / n)
    chirp = np.maximum(chirp, 0.1)
    inits['chirp'] = chirp / chirp.sum()

    # Sine modulated
    sine = 1.0 + 0.4 * np.sin(2 * np.pi * x / (n / 3))
    sine = np.maximum(sine, 0.1)
    inits['sine'] = sine / sine.sum()

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()

    return inits

            # Longer LP run on best
            t_lp = remaining * 0.4
            f_lp, obj_lp = lp_alternate(best_seq, time_limit=t_lp, n=n)
            sc_lp = float(evaluate_sequence(f_lp.tolist()))
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = f_lp.copy()
                print(f"  Deep LP: {sc_lp:.6f} (IMPROVED)")

            # Spectral flattening
            if time.time() < deadline - 20:
                t_spec = min(remaining * 0.2, deadline - time.time() - 20)
                f_spec, obj_spec = spectral_flatten(best_seq, time_limit=t_spec, n=n)
                sc_spec = float(evaluate_sequence(f_spec.tolist()))
                if np.isfinite(sc_spec) and sc_spec < best_score:
                    best_score = sc_spec
                    best_seq = f_spec.copy()
                    print(f"  Spectral: {sc_spec:.6f} (IMPROVED)")

            # Final soft-max polish with different params
            if time.time() < deadline - 10:
                t_final = min(remaining * 0.25, deadline - time.time() - 10)
                for tp, lr in [(60.0, 0.02), (150.0, 0.01), (80.0, 0.015)]:
                    if time.time() > deadline - 5:
                        break
                    tf = min(t_final / 3.0, deadline - time.time() - 5)
                    if tf < 2:
                        break
                    f_fn, _ = softmax_gradient_descent(best_seq, time_limit=tf, n=n, t_param=tp, lr_base=lr)
                    sc_fn = float(evaluate_sequence(f_fn.tolist()))
                    if np.isfinite(sc_fn) and sc_fn < best_score:
                        best_score = sc_fn
                        best_seq = f_fn.copy()
                        print(f"  Final SM tp={tp:.0f}: {sc_fn:.6f} (IMPROVED)")

            # Final peak shave
            if time.time() < deadline - 3:
                t_ps = min(15.0, deadline - time.time() - 3)
                f_ps, _ = peak_shave(best_seq, time_limit=t_ps, n=n)
                sc_ps = float(evaluate_sequence(f_ps.tolist()))
                if np.isfinite(sc_ps) and sc_ps < best_score:
                    best_score = sc_ps
                    best_seq = f_ps.copy()
                    print(f"  Final PS: {sc_ps:.6f} (IMPROVED)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
