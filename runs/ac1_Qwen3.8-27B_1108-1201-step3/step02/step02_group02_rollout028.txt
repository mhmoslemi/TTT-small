"""
Hybrid optimization combining fast coordinate descent with LP-based direction
finding and spectral initialization to minimize peak autocorrelation of
non-negative step functions. Uses incremental convolution updates for
efficiency and multiple diverse starting points to avoid local minima.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute evaluation score quickly."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def coord_descent_optimize(a_init, max_time=60.0, seed=42, verbose=False):
    """Fast coordinate descent with incremental convolution updates."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n) * (np.sqrt(2.0 * n) / n)
        S = np.sum(a)
    c = fft_convolve(a, a)
    score = 2.0 * n * np.max(c) / (S * S)
    best_score = score
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve_count = 0

    while time.monotonic() - t0 < max_time:
        sweep += 1
        order = rng.permutation(n)
        improved_this_sweep = False

        for i in order:
            v_old = a[i]
            S_prime = S - v_old
            if S_prime < 0.01 and v_old < 1e-10:
                continue

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            if lo > hi:
                continue

            ks = np.arange(lo, hi + 1)
            ki_vals = ks - i

            # Compute M_const from unaffected positions
            if lo == 0 and hi == 2 * n - 2:
                M_const = 0.0
            else:
                M_const = 0.0
                if lo > 0:
                    M_const = max(M_const, float(np.max(c[0:lo])))
                if hi < 2 * n - 2:
                    M_const = max(M_const, float(np.max(c[hi + 1:2 * n - 1])))

            # Old contributions to c[ks]
            old_contrib = 2.0 * v_old * a[ki_vals].copy()
            idx_2i = (ks == 2 * i)
            if np.any(idx_2i):
                old_contrib[idx_2i] = v_old * v_old
            base_c = c[ks] - old_contrib

            # Linear coefficients for non-2i positions
            linear_mask = ~idx_2i
            lin_coeff = a[ki_vals]  # a[k-i] for each k in ks

            def M_eval(v):
                if v < 0.0:
                    v = 0.0
                m = M_const
                if np.any(linear_mask):
                    lv = base_c[linear_mask] + 2.0 * v * lin_coeff[linear_mask]
                    m = max(m, float(np.max(lv)))
                if np.any(idx_2i):
                    m = max(m, float(base_c[idx_2i][0] + v * v))
                return m

            def score_eval(v):
                if v < 0.0:
                    v = 0.0
                S_new = S_prime + v
                if S_new < 0.01:
                    return 1e10
                return 2.0 * n * M_eval(v) / (S_new * S_new)

            # Find optimal v
            v_max = max(5.0 * v_old + 1.0, 3.0 * (S / n) * 2.0, 2.0)
            best_v = v_old
            best_s = score_eval(v_old)

            # Try v=0
            s0 = score_eval(0.0)
            if s0 < best_s:
                best_s = s0
                best_v = 0.0

            # Bounded scalar minimization
            try:
                res = minimize_scalar(score_eval, bounds=(0.0, v_max),
                                      method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 40})
                if res.success and res.fun < best_s:
                    best_s = res.fun
                    best_v = float(res.x)
            except Exception:
                pass

            # Try a few more candidate values around the optimum
            if best_v > 1e-8:
                for frac in [0.5, 1.5, 0.75, 1.25]:
                    v_try = best_v * frac
                    if v_try >= 0 and v_try <= v_max:
                        st = score_eval(v_try)
                        if st < best_s:
                            best_s = st
                            best_v = v_try

            v_new = max(0.0, best_v)

            if abs(v_new - v_old) > 1e-12:
                delta = v_new - v_old
                a[i] = v_new
                S += delta
                # Update convolution incrementally
                dc = 2.0 * delta * a[ki_vals].copy()
                if np.any(idx_2i):
                    dc[idx_2i] = v_new * v_new - v_old * v_old
                c[ks] += dc

                new_score = 2.0 * n * float(np.max(c)) / (S * S)
                if new_score < score:
                    score = new_score
                    improved_this_sweep = True
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD sweep {sweep}: score={score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")

        if not improved_this_sweep:
            no_improve_count += 1
            if no_improve_count >= 4:
                break
        else:
            no_improve_count = 0

    return best_a, best_score


def lp_direction(f, M, n):
    """Solve LP to find best descent direction."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def lp_iterate(f0, time_limit=120.0, seed=42, verbose=False):
    """LP-based iterative optimization."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    S_target = np.sqrt(2.0 * n)
    f = np.clip(f0.astype(np.float64), 0, None).copy()
    sum_f = np.sum(f)
    if sum_f < 0.01:
        f = np.ones(n) * (S_target / n)
    else:
        f = f * (S_target / sum_f)

    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.06
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = float(np.max(conv))
        b_0 = lp_direction(f, M, n)
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = (S_target / sum_b) * b_0

        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.125]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            sf_new = np.sum(f_new)
            if sf_new < 0.01:
                continue
            f_new = f_new * (S_target / sf_new)
            new_score = fast_score(f_new)
            if new_score < current_score - 1e-7:
                f = f_new
                current_score = new_score
                t = min(trial_t * 1.1, 0.2)
                improved = True
                stall_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break

        if not improved:
            stall_count += 1
            t *= 0.75
            if stall_count > 60:
                # Perturb
                f = f * np.exp(rng.normal(0, 0.04, n))
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * (S_target / sf)
                    current_score = fast_score(f)
                stall_count = 0
                t = 0.04

        if verbose and stall_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP: score={current_score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_f, best_score


def simulated_annealing_polish(a_init, max_time=120.0, seed=42):
    """Simulated annealing for final polish."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    current = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    current_score = fast_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.99985

    while time.monotonic() - t0 < max_time:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.012, size=width))
        elif r < 0.82:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.12, 0.12) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        elif r < 0.92:
            # Asymmetry adjustment (key insight: non-symmetric sequences can do better)
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            delta = rng.uniform(-0.08, 0.08) * max(current[i], current[j])
            candidate[i] = max(0, current[i] + delta)
            candidate[j] = max(0, current[j] - delta)
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.004, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = fast_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            accept_prob = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if rng.random() < accept_prob:
                current = candidate
                current_score = new_score

        temperature *= cooling

    return best, best_score


def make_asymmetric_init(n, rng, asymmetry=0.3):
    """Create asymmetric initialization to exploit non-symmetric structure."""
    t = np.arange(n, dtype=np.float64)
    # Linear ramp (asymmetric)
    a = 1.0 + asymmetry * (t / n - 0.5) * 2.0
    # Add some structure
    a += 0.15 * asymmetry * np.sin(2.0 * np.pi * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (1.5 * n))
    a = np.clip(a, 0.01, None)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 5):
        amp = 0.2 / k
        freq = rng.uniform(0.4, 2.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_exponential_init(n, rng, rate=1.0):
    """Exponentially weighted initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * t / n * 2.0)
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
